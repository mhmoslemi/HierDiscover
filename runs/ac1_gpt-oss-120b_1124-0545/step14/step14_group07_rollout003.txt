def _cosine_bump(n, alpha):
    # shape: a_i = (cos(pi*(i - (n-1)/2) / (alpha*(n-1)/2)) + 1)/2 for |i - (n-1)/2| <= alpha*(n-1)/2 else 0

    i = np.arange(n, dtype=np.float64)
    distance = np.abs(i - (n - 1) / 2.0)

    s = a.sum()
    if s > 0:

5. **Extended EG** – a long‐run EG (up to the remaining time budget) for
   fine‑grained improvement.

6. **Final evaluation** – the best vector is handed to `evaluate_sequence`
   and returned as a plain list.  The routine respects the 1000 s deadline
   and stores the result globally as ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq,seq)) for a unit‑ℓ₁ vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_bump(n: int, width_frac: float) -> np.ndarray:
    """
    Cosine‑bump window.
    Width fraction ∈ (0,1] gives the half‑width relative to the full length.
    """
    if width_frac <= 0.0:
        return _uniform(n)
    half_width = width_frac * (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - (n - 1) / 2.0)
    a = np.where(dist <= half_width,
                 0.5 * (1 + np.cos(np.pi * dist / half_width)),
                 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (diverse ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in (0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))

    for sigma in (0.12, 0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))

    seeds.append(_exponential_decay(n, 0.9995))

    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))

    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    for width in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_cosine_bump(n, width))

    # random Dirichlet draws (wide variety of concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (adds a flat bias)
    for mix_alpha in (0.1, 0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity normalisation (guard against accidental zero‑sum)
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Multiplicative EG on the exact max‑convolution objective."""
    x = np.asarray(start, dtype=np.float64).copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step‑size schedule (1/√t)
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)        # avoid under/overflow
        x = x * np.exp(delta)                  # multiplicative update

        # simplex projection (via renormalisation)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth approximation of max(conv) and gradient wrt logits y."""
    a = _softmax(y)                     # lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback to exact argmax
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule: ∂F/∂y = diag(a)·(∂F/∂a – (a·∂F/∂a)·1)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth surrogate value: (1/p)·log Σ exp(p·cₖ)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    p: float = 20.0,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth‑max surrogate (logits parametrisation).
    Returns the best vector found and its *exact* surrogate value
    (2·n·max(conv)).
    """
    y0 = np.log(np.maximum(start_a, 1e-15))
    # `scipy.optimize.minimize` may call the function many times;
    # we respect the global deadline inside the wrapper.
    def wrapped(y):
        return _lbfgsb_func(y, n, p)
    res = _opt.minimize(
        wrapped,
        y0,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "disp": False},
    )
    # In the rare case the optimiser exceeded the deadline, we abort early.
    if time.time() > deadline:
        # Return the start vector – it is still a feasible point.
        a_opt = start_a
    else:
        a_opt = _softmax(res.x)

    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Sequence length – a trade‑off between expressiveness and speed.
    # --------------------------------------------------------------
    n = 4000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence (n) = {n}")

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Evaluate the exact surrogate for every seed and keep the best few.
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 15
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick Exponentiated‑Gradient polishing on the top seeds.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinement ------------------------------------------")
        # run a short EG on a few promising seeds
        for idx, seed_vec in enumerate(best_seeds[:7]):
            if time.time() >= deadline:
                break
            cand, val = _expgrad_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=200_000,
                eta0=0.30,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B on a smooth surrogate with a geometric p‑schedule.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        # geometric schedule for the smoothing parameter p
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            # main optimisation starting from the current best
            cand, val = _lbfgsb_optimize(best_seq, n, deadline, p=p, max_iter=2000)
            print(f"[L‑BFGS] p={p:7.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** improved global best to {best_val:.6f}")

            # one random restart per p value
            if time.time() < deadline:
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.3, size=n)
                perturbed = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(perturbed, n, deadline, p=p, max_iter=2000)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart] p={p:7.1f} → improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – extended EG polishing (high‑iteration budget).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Extended EG polishing ----------------------------------------")
        remaining = deadline - time.time()
        # roughly 1500 updates per second on a modest machine
        max_iters = int(min(3_000_000, remaining * 1500))
        print(f"Running EG for up to {max_iters:,} iterations.")
        cand, val = _expgrad_optimize(
            best_seq,
            n,
            deadline,
            max_iters=max_iters,
            eta0=0.25,
        )
        print(f"[Extended EG] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, cand

    # --------------------------------------------------------------
    # Stage 4 – simple random‑perturbation hill‑climbing (any time left).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random hill‑climbing (final sweep) --------------------------")
        # work in log‑it space: y = log a  (softmax will renormalise)
        y_curr = np.log(np.maximum(best_seq, 1e-15))
        while time.time() < deadline:
            sigma = 0.05 * (1.0 + rng.random())   # modest noise scale
            y_new = y_curr + rng.normal(scale=sigma, size=n)
            a_new = _softmax(y_new)
            val_new = _objective_surrogate(a_new)
            if val_new < best_val:
                best_val, best_seq, y_curr = val_new, a_new, y_new
                print(f"[Hill‑climb] improved to {best_val:.6f}")
            else:
                # occasionally accept a worsening move to escape shallow traps
                if rng.random() < 0.001:
                    y_curr = y_new

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
