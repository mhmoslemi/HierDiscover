"""
Search for a non‑negative step‑function that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)
using a rich pool of analytic seeds, many random Dirichlet draws,
a multi‑stage Adam optimisation with a log‑sum‑exp smoothing schedule,
and a final long simulated‑annealing polish.  The total wall‑clock
budget is 1000 s; all timings are checked regularly and the best
sequence found (stored in ``height_sequence_1``) is returned.
"""

import time, math, sys
import numpy as np

# ----------------------------------------------------------------------
# low‑level utilities – unchanged from the reference implementation
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# ----------------------------------------------------------------------
# analytic families – return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   betas: list[float] | None = None,
                   max_iters_per_beta: int = 5000) -> np.ndarray:
    """
    Adam optimisation of the smoothed max‑convolution.
    The parameters are stored in log‑softmax space (logits).
    Returns an ℓ₁‑normalised vector.
    """
    if betas is None:
        betas = [1.0, 5.0, 20.0, 80.0, 200.0,
                 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)          # logits
    m = np.zeros_like(v)
    u = np.zeros_like(v)                   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for b in betas:
        if time.time() > deadline:
            break
        # allocate a modest number of iterations for this β
        remaining = deadline - time.time()
        # we aim for at most 0.04 s per 1000 iterations ≈ 0.04 s?? Hard to guess.
        # Use a safe cap, but also ensure at least a few hundred steps.
        max_iter = int(min(max_iters_per_beta,
                           max(500, remaining * 30)))   # about 30 it/s budget
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)                     # ℓ₁‑normalised
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp smoothing
            shift = max_c
            exp_b = np.exp(b * (conv - shift))
            sum_exp = exp_b.sum()
            w = exp_b / sum_exp                  # weights that sum to 1

            # gradient wrt a (from smoothed max)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            u = beta2 * u + (1.0 - beta2) * (grad_v * grad_v)
            m_hat = m / (1.0 - beta1 ** t)
            u_hat = u / (1.0 - beta2 ** t)
            step = 0.1 / (np.sqrt(u_hat) + eps)
            v = v - step * m_hat

    return _softmax(v)

# ----------------------------------------------------------------------
# simple simulated‑annealing in logit space (final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> np.ndarray:
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng(42)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.02 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.15 * (1.0 - progress) + 1e-5      # proposal width

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _objective_surrogate(prop_seq)

        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq

# ----------------------------------------------------------------------
# candidate pool generation (analytic + random)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []
    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.15, 0.18, 0.22):
        seeds.append(_gaussian(n, sigma))

    for a in (0.3, 0.6, 1.0, 2.0):
        for b in (0.3, 0.7, 1.2, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet with a range of concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    return seeds

# main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of length ~4000 that minimises the
    evaluation function.  The routine respects the time budget,
    periodically prints progress, and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1.  Seed generation (analytic + random)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    seeds = _generate_seeds(n, rng)

    # If a previous solution exists, add it as a seed (re‑normalised)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size == n and prev_arr.sum() > 0:
            prev_arr = np.maximum(prev_arr, 0.0)
            prev_arr = prev_arr / prev_arr.sum()
            seeds.append(prev_arr)
            print("Added previous solution as a seed.")

    # ------------------------------------------------------------------
    # 2.  Quick evaluation of seeds and keep the best few
    # ------------------------------------------------------------------
    scored = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        scored.append((val, s))
    scored.sort(key=lambda x: x[0])
    top_k = 10
    top_seeds = [s for _, s in scored[:top_k]]
    print(f"Evaluated {len(seeds)} seeds, keeping the best {top_k}.")

    # ------------------------------------------------------------------
    # 3.  Adam optimisation (smooth max) from each top seed
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    for i, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all Adam runs.")
            break
        print(f"\n--- Adam optimisation from seed #{i+1} ---")
        # allocate a share of the remaining time (roughly equal per seed)
        remaining = deadline - time.time()
        share = remaining / (len(top_seeds) - i)
        seq_opt = _adam_optimize(seed_vec, n, time.time() + share,
                                 max_iters_per_beta=8000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[Adam] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Long Adam refinement on the current best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Long Adam refinement ---")
        seq_ref = _adam_optimize(best_seq, n, deadline,
                                 max_iters_per_beta=20000)
        val_ref = _objective_surrogate(seq_ref)
        print(f"[Long Adam] surrogate = {val_ref:.6f}")
        if val_ref < best_val:
            best_seq = seq_ref.copy()
            best_val = val_ref
            print(f"[Update] new global best after long Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(800_000, remaining * 1500))   # a few hundred‑k steps
        if max_sa > 0:
            print("\n--- Simulated annealing polish ---")
            seq_sa = _simulated_annealing(best_seq, n, deadline, max_iters=max_sa)
            val_sa = _objective_surrogate(seq_sa)
            print(f"[SA] final surrogate = {val_sa:.6f}")
            if val_sa < best_val:

                best_val = val_sa

    return best_seq.tolist()

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of

1. Generation of a diverse pool of ℓ₁‑normalised seeds (analytic families,
   several Dirichlet draws and the previous best solution if present).

2. Quick evaluation of all seeds and selection of the top‑k for deeper
   optimisation.

3. Multi‑stage Adam optimisation on a log‑sum‑exp surrogate of the
   max‑convolution.  The parameters are stored in log‑softmax (logit) space,
   a schedule of increasing β (smoothness) is used and a modest number of
   iterations is performed for each β.

4. A longer Adam refinement started from the current best seed (uses the
   full budget that is still available).

5. Final simulated‑annealing polishing in logit space.

All steps respect the 1000 s wall‑clock budget; intermediate progress and
objective values are printed.  The best sequence is stored in the global
variable ``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   betas: list[float] | None = None,
                   max_iters_per_beta: int = 5000) -> np.ndarray:
    """
    Adam optimisation of the smoothed max‑convolution.
    Parameters live in log‑softmax space (logits) and are projected back
    to the probability simplex by the softmax at every step.
    Returns an ℓ₁‑normalised vector.
    """
    if betas is None:
        betas = [1.0, 5.0, 20.0, 80.0, 200.0,
                 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)          # logits
    m = np.zeros_like(v)                   # 1st moment
    u = np.zeros_like(v)                   # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for b in betas:
        if time.time() > deadline:
            break
        # Allocate a limited number of iterations for this β.
        # We never exceed max_iters_per_beta and also respect the remaining time.
        remaining = deadline - time.time()
        max_iter = int(min(max_iters_per_beta,
                           max(500, remaining * 30)))   # rough 30 it/s budget
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)                     # ℓ₁‑normalised probabilities
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing of the max
            shift = max_c
            exp_b = np.exp(b * (conv - shift))
            sum_exp = exp_b.sum()
            w = exp_b / sum_exp                  # soft‑weights, sum to 1

            # gradient wrt a (derived from the surrogate)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            u = beta2 * u + (1.0 - beta2) * (grad_v * grad_v)
            m_hat = m / (1.0 - beta1 ** t)
            u_hat = u / (1.0 - beta2 ** t)
            step = 0.1 / (np.sqrt(u_hat) + eps)
            v = v - step * m_hat

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (logit + softmax)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> np.ndarray:
    """
    Simple SA in logit space.  Uses a linear cooling schedule and a
    decreasing proposal scale.
    """
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng(42)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.02 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.15 * (1.0 - progress) + 1e-5      # proposal width

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _objective_surrogate(prop_seq)

        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Seed generation (analytic + random)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.15, 0.18, 0.22):
        seeds.append(_gaussian(n, sigma))

    for a in (0.3, 0.6, 1.0, 2.0):
        for b in (0.3, 0.7, 1.2, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws with a variety of concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    return seeds
