def _local_annealing(start, n, deadline):

   cur = start.copy()
   cur_val = _objective_surrogate(cur)  # 2n * max(conv)
   best = cur.copy()
   best_val = cur_val
   T = cur_val * 0.05 + 1e-12 (some schedule)
   rng = np.random.default_rng()

   while time.time() < deadline - 0.5:

        # generate candidate by random perturbation: pick random index i, random epsilon (small) to transfer mass to j
        i, j = rng.choice(n, size=2, replace=False)
        epsilon = rng.random() * cur[i] * 0.1  # move up to 10% of i's mass
        cand = cur.copy()
        cand[i] -= epsilon
        cand[j] += epsilon
        cand = _proj_simplex(cand)  # ensure non-negative sum
        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best = cand.copy()
                best_val = new_val
        # cooling
        T *= 0.9999

   return best, best_val

"""
Hybrid sub‑gradient + momentum descent followed by aggressive simulated‑annealing.
The search works on a probability vector (sum = 1) of length n (default 4000).
The objective is 2·n·max(conv(a,a)), which equals the official
`evaluate_sequence` value for sum‑normalised vectors.  The algorithm:

1. Produce a diverse pool of seeds (uniform, triangular, Gaussian,
   power‑law, Beta, Tukey, sparse Dirichlet draws, random low‑frequency
   cosine waves, mixtures, and the previous best if any).
2. Keep the best few seeds w.r.t. the surrogate objective.
3. For each seed run a *momentum sub‑gradient* optimisation:
      - compute the full convolution via FFT,
      - locate the index K where the convolution attains its maximum,
      - the sub‑gradient of the max‑norm is 2·a_{K‑i},
        we use g = a_{K‑i} − max(conv) (the constant term only shifts the
        vector on the simplex and does not affect the descent direction).
      - momentum = β·momentum + (1‑β)·g, step = η/√t·momentum.
      - after the step clip negatives and renormalise to the simplex.
   The step size η and momentum β are chosen conservatively; the loop
   runs until the global deadline is reached.
4. The best vector from step 3 is then refined by a simulated‑annealing
   phase that repeatedly reshuffles mass on random subsets or transfers
   a small amount between two positions.  The temperature starts at a
   fraction of the current objective and decays geometrically.
5. Finally the vector is evaluated with the official `evaluate_sequence`
   function and stored in the global name `height_sequence_1`.

The whole routine respects a user supplied `budget_s` (default 1000 s)
and returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(p,p)) for a distribution (∑p = 1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.2, 0.3, 0.4):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_tukey_window(n, 0.5))

    # power‑law shapes
    for p in (0.5, 1.0, 1.5):
        seeds.append(_powerlaw(n, p))

    # sparse Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # exact‑size sparse supports
    for m in (120, 200, 300, 400):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency random cosine waves
    for nf in (2, 3):
        for _ in range(4):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                seeds.append(a / a.sum())

    # mixtures of uniform and a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previous best, if any
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up (ensure proper normalisation)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Sub‑gradient with momentum
# ----------------------------------------------------------------------
def _shift_vector(a: np.ndarray, k: int) -> np.ndarray:
    """Return b where b[i] = a[k‑i] if in range, else 0."""
    n = a.shape[0]
    src = k - np.arange(n)
    mask = (src >= 0) & (src < n)
    b = np.zeros_like(a)
    b[mask] = a[src[mask]]
    return b


def _momentum_subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    beta: float = 0.9,
    max_iters: int = 10**9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Nesterov‑style momentum.
    Works on the simplex (sum = 1, non‑negative).  Returns the best
    sequence (as a probability vector) and its surrogate value.
    """
    a = start.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    momentum = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        M = conv[k]

        # sub‑gradient of max‑norm (ignoring the constant 4n factor)
        b = _shift_vector(a, k)          # b[i] = a[k‑i] (or 0)
        grad = b - M                     # scalar shift removes the ‑M term after projection

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule (classic 1/√t)
        step = (eta0 / math.sqrt(it)) * momentum

        # simple clipping to keep updates moderate
        step = np.clip(step, -0.02, 0.02)

        a = a - step
        # enforce non‑negativity
        a = np.maximum(a, 0.0)

        # renormalise onto the simplex (cheap – just divide)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress print (every 50000 iters)
            if it % 50000 == 0:
                print(f"[Momentum] it={it:7d} best={best_val:.6f}")
    return best_seq, best_val

# Simulated annealing (fine‑tuning)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10**9,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Neighbourhood moves that reshuffle mass on small random subsets."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.99995, min_temp)

        # Two alternative move types
        if rng.random() < 0.6:                     # reshuffle on a random subset
            size = rng.integers(8, 20)             # 8‑19 indices
            idx = rng.choice(n, size=size, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(size, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:                                       # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # up to 5 % of source mass
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        # re‑normalise onto the simplex (cheap)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val
