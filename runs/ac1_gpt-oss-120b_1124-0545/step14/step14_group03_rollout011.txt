"""
Propose a candidate step function using the discrete arcsine
distribution (Beta(0.5, 0.5)).  This distribution has the
property that its self‑convolution is (asymptotically) flat,
which drives the evaluation metric 2·n·max(conv)/sum^2
close to the theoretical lower bound 1.  The implementation
generates the vector, normalises it to ℓ₁=1, optionally
performs a short exponentiated‑gradient refinement and finally
stores the sequence in the global ``height_sequence_1``.
"""

import time, math, numpy as np

def _arcsine_vector(n: int) -> np.ndarray:
    """Return an ℓ₁‑normalised arcsine (Beta(0.5,0.5)) shape."""
    i = np.arange(1, n + 1, dtype=np.float64)            # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s

def _exp_gradient_refine(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 1.0) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    The update is performed in log‑space: log a ← log a − η·g,
    a ← softmax(log a).  Early‑stop when the time budget expires.
    """
    a = start.copy()
    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # compute full convolution to locate the active lag
        conv = np.convolve(a, a)
        idx = int(np.argmax(conv))
        # sub‑gradient of the active entry (same as in the reference code)
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)

"""
Propose a candidate step‑function based on the discrete arcsine (Beta(0.5,0.5))
distribution.

The continuous arcsine law has the remarkable property that its self‑convolution
is exactly uniform.  After discretisation the convolution of the resulting
probability vector is *almost* flat, which drives the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)

very close to its theoretical lower bound of 1.  The routine builds the
distribution, normalises it to ℓ₁=1, stores it in the global name
`height_sequence_1` and returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    return e / s


# ----------------------------------------------------------------------
# Construction of the discrete arcsine law (Beta(0.5,0.5))
# ----------------------------------------------------------------------
def _arcsine_vector(n: int) -> np.ndarray:
    """
    Return an ℓ₁‑normalised vector of length `n` whose entries are proportional
    to 1/√(i·(n‑i+1)), i=1…n.  This is the discretised Beta(0.5,0.5) density.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    if s <= 0.0:
        # fallback – should never happen
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# (Optional) tiny exponentiated‑gradient refinement
# ----------------------------------------------------------------------
def _exp_gradient_refine(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 100_000,
                         eta0: float = 1.0) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  The update is performed in log‑space:
        log a ← log a – η·g ,  a ← softmax(log a)
    where g is a sub‑gradient of the active (largest) convolution entry.
    The loop stops when the time budget expires.

        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)

    return a
