"""
Sidon‑set based construction of a step function.

The convolution max of a non‑negative vector p (∑p=1) is
   max_k (p*p)[k] = max_k ( #ordered pairs i+j=k ) / m²
where m = number of non‑zero entries.
If the support S ⊂ {0,…,n‑1} is a *Sidon* (B₂) set,
i.e. all unordered sums i+j (i≤j) are distinct,
then every sum receives at most two ordered contributions
(self‑pair contributes one, any other pair contributes two).
Hence max_k (p*p)[k] = 2 / m² and the official objective becomes

    2·n·max(conv) = 4·n / m².

To beat the target 1.503 we need m ≥ 104 for n=4000.
The code below repeatedly builds random Sidon sets by a simple greedy
procedure, keeps the largest one found before the time budget expires,
and returns the corresponding equal‑mass step function.
If the greedy search fails to reach the target it falls back to the
previous best sequence (if any) or to a uniform distribution.
"""

import time
import math
import numpy as np
from typing import List, Set, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_ordered_count(support: Set[int]) -> int:
    """
    For a given support set S, compute the largest number of ordered
    pairs (i, j) with i, j ∈ S and i+j = k.
    Returns that maximum count.
    """
    # count[k] will hold the number of ordered pairs giving sum k
    count = {}
    elems = list(support)
    m = len(elems)
    maxc = 0
    for i in range(m):
        a = elems[i]
        # self‑pair contributes one ordered pair
        s = a + a
        cnt = count.get(s, 0) + 1
        count[s] = cnt
        if cnt > maxc:
            maxc = cnt
        for j in range(i + 1, m):
            b = elems[j]
            s = a + b
            # (a,b) and (b,a) both contribute
            cnt = count.get(s, 0) + 2
            count[s] = cnt
            if cnt > maxc:
                maxc = cnt
    return maxc


def _objective_from_support(support: Set[int], n: int) -> float:
    """
    Compute the official objective value for an equal‑mass vector whose
    non‑zero positions are given by *support*.
    """
    m = len(support)
    if m == 0:
        return float("inf")
    maxc = _max_ordered_count(support)          # ordered‑pair count
    # each weight = 1/m, so max(conv) = maxc / m²
    return 2.0 * n * maxc / (m * m)


def _greedy_sid_set(n: int, rng: np.random.Generator) -> Set[int]:
    """
    Build a random Sidon (B₂) set by scanning the points in a random
    order and adding a point whenever it does not create a duplicate
    unordered sum.
    """
    cur_set: Set[int] = set()
    sumset: Set[int] = set()                     # stores unordered sums i+j
    candidates = list(range(n))
    rng.shuffle(candidates)

    for x in candidates:
        # test if x would clash with an existing sum
        if (x + x) in sumset:
            continue
        clash = False
        for y in cur_set:
            if (x + y) in sumset:
                clash = True
                break
        if clash:
            continue
        # safe to add x
        for y in cur_set:
            sumset.add(x + y)
        sumset.add(x + x)
        cur_set.add(x)
    return cur_set

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step function that minimizes ``evaluate_sequence``.
    The search is based on constructing large Sidon sets (sets with

    support the convolution peak is provably 2/m², giving an objective
    4·n/m².  For n=4000 this falls below 1.5 as soon as m≥104.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Problem size – we keep the same length as the previous runs.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Target size for the Sidon set (derived from the bound 4·n/m² < 1.503)
    # ------------------------------------------------------------------
    target_m = int(math.ceil(math.sqrt(4.0 * n / 1.503))) + 1
    print(f"Target support size to beat 1.503: m ≥ {target_m}")

    best_support: Set[int] = set()
    best_val = float("inf")
    best_m = 0

    attempt = 0
    # ------------------------------------------------------------------
    # Repeated random greedy construction – stop when time runs out or
    # when we have already crossed the target threshold.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1
        support = _greedy_sid_set(n, rng)

        m = len(support)
        if m <= best_m:
            # we already have a larger set; skip expensive evaluation
            continue

        val = _objective_from_support(support, n)

        if val < best_val:
            best_val = val
            best_support = support
            best_m = m
            print(f"[Attempt {attempt:4d}] new best: |S|={m}, value={val:.6f}")

        # early exit if we have comfortably beaten the target
        if best_val < 1.503:
            print("Target reached – stopping early.")
            break

    # ------------------------------------------------------------------
    # If the greedy search did not succeed (unlikely), fall back to the
    # previously discovered best sequence or a uniform distribution.
    # ------------------------------------------------------------------
    if best_m == 0:
        print("Greedy Sidon search failed – using previous or uniform fallback.")
        previous = globals().get("height_sequence_1")
        if previous is not None:
            try:
                seq = np.asarray(previous, dtype=np.float64)
                if seq.size == n and seq.sum() > 0.0:
                    globals()["height_sequence_1"] = seq.tolist()
                    return seq.tolist()
            except Exception:
                pass
        # uniform fallback
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        globals()["height_sequence_1"] = uniform.tolist()
        return uniform.tolist()

    # ------------------------------------------------------------------
    # Build the final sequence from the best Sidon support.
    # Equal masses (1/|S|) are used – this is optimal for a Sidon set.
    # ------------------------------------------------------------------
    seq_array = np.zeros(n, dtype=np.float64)
    mass = 1.0 / best_m
    for idx in best_support:
        seq_array[idx] = mass

    # Final official evaluation (just to report the exact value)
    true_val = evaluate_sequence(seq_array.tolist())
    print("\n=== Final result ===")
    print(f"Support size m = {best_m}")
    print(f"Surrogate value = {best_val:.6f}")
    print(f"evaluate_sequence = {true_val:.6f}")

    globals()["height_sequence_1"] = seq_array.tolist()
    return seq_array.tolist()
