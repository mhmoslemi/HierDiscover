"""
Propose a step function (non‑negative heights) aiming at a low value of
`evaluate_sequence`.  The search works as follows:

1. **Rich seed library** – deterministic families (uniform, triangular,
   Gaussian, Beta, arcsine, etc.), random Dirichlet draws, sparse
   supports and a copy of the previous best (if available).  All seeds are
   ℓ₁‑normalised.

2. **Fast filtering** – each seed is evaluated with the exact surrogate
   `2·n·max(conv(p,p))` (the same quantity used by `evaluate_sequence`);
   the best 10 seeds are kept.

3. **Three optimisation stages**
   * *Soft‑max Adam* – smooth approximation of the max via a log‑sum‑exp,
     annealing the temperature from 20 to 5000.  Several starts are run.
   * *Sparse exponential‑gradient* – mirror‑descent (multiplicative) on the
     true (non‑smooth) objective, with an optional sparsity projection.
   * *Simulated‑annealing* – random mass reshuffles on small subsets and
     tiny two‑point transfers, driven by a temperature schedule.

4. **Greedy local refinement** – while time remains we perform cheap
   random two‑point transfers and keep any improvement.

All updates are projected onto the probability simplex (ensuring a
valid non‑negative step function).  The code respects the 1000 s budget,
printing progress and the final true value obtained from
`evaluate_sequence`.  The best sequence is stored in the global name
`height_sequence_1` for reuse.
"""

import time

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray, n: int) -> float:
    """
    Exact surrogate for the official objective:
    2·n·max(conv(p,p))  (p is ℓ₁‑normalised, i.e. sum(p)=1).
    """
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _arcsine(n: int) -> np.ndarray:
    """Beta(½,½) – the classic U‑shaped arcsine density."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` equal non‑zero entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 5000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        L(p) = (1/α)·log Σₖ exp(α·(p*p)[k]).
    The temperature α grows linearly from `alpha_start` to `alpha_end`.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x, n)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N_conv = _next_pow2(3 * n - 2)      # size needed for correlation (p * soft)
    N_full = _next_pow2(2 * n - 1)      # size for the plain convolution

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- convolution p * p (full) -----
        fx = np.fft.rfft(x, n=N_full)
        conv_full = np.fft.irfft(fx * fx, n=N_full)[: 2 * n - 1]

        # ----- soft‑max of the convolution -----
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)          # ∑ soft = 1

        # ----- gradient of the smooth surrogate -----
        # gradient = 2 * correlation(p, soft)
        fs = np.fft.rfft(soft, n=N_conv)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N_conv)[:n]
        grad = 2.0 * corr

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # ----- projection onto the simplex -----
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x, n)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
            if t % 20000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.25,
    sparsity_thr: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) on the true objective.
    Optional sparsity projection zeros very small entries.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow in exp
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            low = x < sparsity_thr
            if np.any(low):
                x[low] = 0.0

        # simplex projection (still ℓ₁‑normalised)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x, n)
        if cur_val < best_val:
            best_val, best_x = cur_val, x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Classic SA with two move types:
      * reshuffle a random small subset (Dirichlet draw)
      * a tiny mass transfer between two positions
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur, n)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.5:
            # ---- reshuffle a random subset (size 5‑15) ----
            k_sub = rng.integers(5, 16)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # ---- tiny transfer between two positions ----
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand, n)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val, best = new_val, cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


def _local_greedy_refine(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    step_frac: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a small amount of mass from a
    random source index to a random target index.  Accept only improving
    moves (occasional random acceptance omitted for speed).
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur, n)
    best = cur.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        eps = rng.random() * step_frac * min(cur[i], 0.1)
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # ensure feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        val = _objective_surrogate(cand, n)
        if val < cur_val:
            cur = cand
            cur_val = val
            if val < best_val:
                best_val, best = val, cand.copy()
                if it % 50_000 == 0:
                    print(f"[Local] it={it:7d} best={best_val:.6f}")

    return best, best_val

def _collect_seeds(

        seeds.append(_gaussian(n, sigma))
    seeds.append(_beta_shape(n, 2.0, 2.0))          # bell‑shaped
    seeds.append(_beta_shape(n, 0.5, 0.5))          # arcsine (U‑shaped)
    seeds.append(_beta_shape(n, 0.3, 0.3))          # more extreme U
    seeds.append(_beta_shape(n, 0.8, 0.8))          # mild U
    seeds.append(_arcsine(n))
    seeds.append(_tukey_window(n, 0.5))

    # asymmetric Beta's (to adjust left/right bias)
    for a, b in ((0.3, 0.7), (0.7, 0.3), (0.2, 0.8), (0.8, 0.2)):
        seeds.append(_beta_shape(n, a, b))

    # very sparse Dirichlet draws (low concentration → spikier)
    for conc in (0.01, 0.03, 0.05, 0.1):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # exact‑size sparse supports (moderate sparsity)
    for m in (120, 200, 300, 400):
        for _ in range(2):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency random cosine‑series curves
    for nf in (2, 3):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                seeds.append(a / a.sum())

    # mixtures of uniform and a sparse support
    for mix_alpha in (0.2, 0.4, 0.6):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up: enforce proper normalisation
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with minimal `evaluate_sequence`
    value within `budget_s` seconds.  The best sequence is stored in the
    global variable `height_sequence_1` and also returned as a plain list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 4000 is a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – generate a large and very diverse seed library
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # Evaluate surrogate on each seed and keep the top performers
    scored = [(s, _objective_surrogate(s, n)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq, n)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            iters = int(min(180_000, remaining * 130))   # generous cap
            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline, max_iters=iters,
                alpha_start=20.0, alpha_end=5000.0, lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        iters = int(min(900_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=iters,
            eta0=0.25, sparsity_thr=1e-8,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        iters = int(min(800_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=iters,
        )
        print(f"[SA] surrogate after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap greedy local refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Greedy local refinement ---")
        remaining = deadline - time.time()
        iters = int(min(400_000, remaining * 2500))
        best_seq, best_val = _local_greedy_refine(
            best_seq, n, deadline, max_iters=iters, step_frac=0.03,
        )
        print(f"[Local] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
