def smooth_obj_and_grad(y, n, p):
    a = softmax(y)
    conv = conv_fft(a,a)
    max_c = conv.max()
    shifted = conv - max_c
    exp_arg = p*shifted
    exp_arg = np.clip(exp_arg, -700, 0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        mask = conv == max_c
        w = mask.astype(float)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = conv_fft(w, a_rev)
    grad_a = 2.0 * grad_full[n-1:n-1+n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y

def local_hill_climb(seq, n, deadline, steps=20000, init_step=1e-3, decay=0.999):
    best_seq = seq.copy()
    best_val = surrogate_objective(best_seq)
    step_size = init_step
    while steps > 0 and time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = step_size * rng.uniform(-1, 1)

        if delta > 0:
            # increase i, decrease j
            if best_seq[j] < delta:
                continue

            new_seq[i] += delta
            new_seq[j] -= delta

Propose a non‑negative step function that minimises the `evaluate_sequence`
objective.

Algorithm
---------
* Choose a (moderately large) length `n` and work on the ℓ₁‑simplex
  (vectors sum to 1).  The true objective is then
  `evaluate_sequence(a) = 2·n·max(conv(a,a))`.
* **Seed generation** – a rich pool of ℓ₁‑normalised vectors built from
  analytic windows (uniform, triangular, sine, Gaussian, beta, Tukey,
  exponential, log‑normal, Weibull) together with many random Dirichlet
  draws and low‑frequency cosine curves.
* **Quick Exponentiated‑Gradient (EG)** – a fast mirror‑descent on the
  simplex for a few seeds.  The sub‑gradient is the exact gradient of the
  active convolution shift.
* **Smooth‑max L‑BFGS‑B** – minimise a smooth surrogate
  `F_p(a) = (1/p)·log Σₖ exp(p·convₖ(a))` using `scipy.optimize.minimize`.
  The schedule `p = 5, 15, 30, 60, 120, 240` gradually approaches the true
  max, and after each `p` a random‑restart is performed.
* **Local hill‑climbing** – a cheap stochastic coordinate‑swap that moves a
  tiny mass from one index to another while staying on the simplex;
  accepts only improvements.
* **Final EG polishing** – a short extra EG run to stabilise the solution.
* The best vector is evaluated with the official `evaluate_sequence`,
  stored globally as ``height_sequence_1`` and returned as a plain list.
* The whole pipeline stops automatically before the time budget expires
  (default 1000 s).

The routine is deliberately aggressive with restarts and iteration limits,
yet respects the 1000 s deadline.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def _surrogate_objective(seq: np.ndarray) -> float:
    """
    Exact surrogate when Σseq = 1 : 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lognormal_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    # avoid log(0) by offsetting index by 1
    a = np.exp(-((np.log(i) - mu) ** 2) / (2 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _weibull_shape(n: int, k: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64) / n
    a = (k / np.math.gamma(1 + 1 / k)) * (i ** (k - 1)) * np.exp(-(i ** k))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.12, 0.15, 0.2, 0.3, 0.4):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.3, 0.6, 1.0, 2.0):
        for b in (0.3, 0.6, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.15, 0.3, 0.6, 0.85):
        seeds.append(_tukey_window(n, alpha))
    # log‑normal variants
    for mu, sigma in [(-1.0, 0.6), (-0.5, 0.4), (0.0, 0.3), (0.5, 0.5)]:
        seeds.append(_lognormal_shape(n, mu, sigma))
    # Weibull variants
    for k in (0.5, 0.8, 1.2, 1.8):
        seeds.append(_weibull_shape(n, k))

    # random Dirichlet draws
    for conc in (0.03, 0.07, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with a flat bias
    for mix_alpha in (0.1, 0.3, 0.7):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep the previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanitisation (guard against accidental zero‑sum)
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent (KL‑projection) on the simplex."""
    x = start.copy()
    best_x = x.copy()
    best_val = _surrogate_objective(x)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)          # √t schedule
        delta = -eta * g
        delta = np.clip(delta, -50, 50)     # protect against overflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _surrogate_objective(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_grad(y: np.ndarray, n: int, p: float):
    """Return value and gradient of the smooth surrogate wrt logits y."""
    a = _softmax(y)                         # a lies on the simplex
    conv = _conv_fft(a, a)
    max_c = conv.max()

    shifted = conv - max_c                  # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)     # keep in floating range
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)        # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


# ----------------------------------------------------------------------
# L‑BFGS‑B optimiser of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 500) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate with logits parametrisation.
    Returns the (exact) surrogate value 2·n·max(conv) of the optimiser.
    """
    y0 = np.log(np.maximum(start_a, 1e-15))

    res = _opt.minimize(_smooth_obj_grad, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})

    y_opt = res.x
    a_opt = _softmax(y_opt)

    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (mass‑swap) on the simplex
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, n: int, deadline: float,
                steps: int = 30_000, init_step: float = 0.005,
                decay: float = 0.9995, rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """Swap a tiny amount of mass between two random coordinates if it improves."""
    if rng is None:
        rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _surrogate_objective(best_seq)
    step = init_step

    while steps > 0 and time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small epsilon from i to j (or reverse)
        eps = step * min(best_seq[i], best_seq[j])
        if eps <= 0.0:
            steps -= 1
            continue
        if rng.random() < 0.5:
            # i -> j
            new_seq = best_seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
        else:
            # j -> i
            new_seq = best_seq.copy()
            new_seq[i] += eps
            new_seq[j] -= eps

        # safety: clip tiny negatives caused by rounding
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()          # re‑normalise (sum stays 1 up to numerical noise)

        new_val = _surrogate_objective(new_seq)
        if new_val < best_val:
            best_seq = new_seq
            best_val = new_val
        step *= decay
        steps -= 1
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a length: large enough to be expressive, still fast.
    # --------------------------------------------------------------
    n = 5000
    print("\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:

        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Keep the best few according to the simple surrogate.
    scored = [(s, _surrogate_objective(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – quick Exponentiated‑Gradient refinement.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _surrogate_objective(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:

        for idx, seed_vec in enumerate(best_seeds[:5]):  # a handful of seeds

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
