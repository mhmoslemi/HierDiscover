"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of the following stages (all time‑controlled):

1. **Seed pool** – a large and diverse collection of ℓ₁‑normalised vectors:
   deterministic windows, random Dirichlet / exponential draws, low‑frequency
   cosine curves and many random linear mixtures.  The previously best
   solution (if any) is also injected.

2. **Continuation Adam** – instead of L‑BFGS we run an Adam optimiser on a
   smooth log‑sum‑exp surrogate of the max‑convolution.  The temperature
   (β) is increased gradually (β up to 10⁶) and a generous per‑β iteration
   budget is used.  Adam’s adaptive step‑size often finds better local
   minima than L‑BFGS on this non‑convex landscape.

3. **Exponentiated‑gradient (mirror‑descent) refinement** – a long EG run on
   the *exact* max‑convolution objective with a slower decay of the step‑size
   (ηₜ = η₀ / t**0.55) and occasional random perturbations to escape plateaus.

4. **Local stochastic search** – after EG we perform a cheap simulated‑annealing
   style perturbation that swaps tiny amounts of mass between two random
   coordinates if it improves the true objective.

5. **Final polishing** – a short second Adam run (high β) starting from the
   best vector found.

All heavy work checks the global deadline (≈ 1000 s) and aborts cleanly.
The best sequence is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:

    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ------------------------------------------------------------
# Surrogate objective (smooth max) used by Adam / L‑BFGS
# ------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Function + gradient for the log‑sum‑exp surrogate."""
    a = _softmax(v)                     # probability simplex
    conv = _conv_fft(a, a)               # full convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_term / sum_exp               # soft weights over each lag
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   iters_per_beta: int = 2000,
                   lr0: float = 0.1) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate.
    Variable = logits v (softmax(v) = a).
    The learning‑rate is scaled by 1/√β so that higher β (sharper
    approximation) gets a smaller step size.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)        # initialise logits
    m = np.zeros_like(v)                 # 1st moment
    vexp = np.zeros_like(v)              # 2nd moment (named vexp to avoid clash)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = _softmax(v)
    best_val = 2.0 * n * np.max(_conv_fft(best_a, best_a))

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # step size adapts with β (larger β → smaller steps)
        lr = lr0 / np.sqrt(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad_v = _smooth_objective(v, n, beta)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            vexp = beta2 * vexp + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            vexp_hat = vexp / (1 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(vexp_hat) + eps)

            if t % 500 == 0:
                a = _softmax(v)
                cur = 2.0 * n * np.max(_conv_fft(a, a))
                if cur < best_val:
                    best_val = cur
                    best_a = a.copy()

    return best_a

# Exact objective & sub‑gradient (used by EG)
# ------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the active index.
    Returns ∂/∂a  max(conv) .
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


def _exp_gradient_optimize(start_a: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 5_000_000,
                          eta0: float = 1.5,
                          power: float = 0.55,
                          rng: np.random.Generator | None = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses exponentiated‑gradient updates:
        log a ← log a − η·g,  a ← softmax(log a)
    ηₜ = η₀ / t**power.
    Occasionally perturbs the solution to escape flat plateaus.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    last_improvement = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / (it ** power)

        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 5000 == 0:
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                last_improvement = it

            # if we have not improved for a while, inject a tiny random kick
            if it - last_improvement > 200_000:
                perturb = rng.normal(scale=0.005, size=n)
                log_a = np.log(best_a + 1e-30) + perturb
                a = _softmax(log_a)
                last_improvement = it

    # final check
    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ------------------------------------------------------------
# Simple stochastic local search (tiny swaps)
# ------------------------------------------------------------
def _local_swap_search(start_a: np.ndarray,
                       n: int,
                       deadline: float,
                       max_steps: int = 200_000,
                       rng: np.random.Generator | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny random mass swaps between two coordinates.
    Accepts only improving moves (hill‑climbing).
    """
    if rng is None:
        rng = np.random.default_rng()
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for step in range(1, max_steps + 1):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small amount δ from i to j
        delta = (rng.random() - 0.5) * 1e-3   # ±5e‑4
        if a[i] + delta < 0 or a[j] - delta < 0:
            continue
        a[i] += delta
        a[j] -= delta
        cur = _objective_exact(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
        else:
            # revert
            a[i] -= delta
            a[j] += delta

        # occasional reporting
        if step % 20000 == 0:
            print(f"[LocalSwap] step={step:,}, best={best_val:.6f}")

    return best_a, best_val


# ------------------------------------------------------------
# Seed generation (including mixtures)
# ------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian
    for sigma in np.linspace(0.08, 0.40, 9):
        seeds.append(_gaussian(n, sigma))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 17):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.5, 3.0, 6):
        for b in np.linspace(0.5, 3.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for al in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, al))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of pairs of already‑generated seeds
    num_mix = 250
    base_idx = np.arange(len(seeds))
    for _ in range(num_mix):
        i, j = rng.choice(base_idx, size=2, replace=False)
        alpha = rng.random()
        mix = _mix_shapes(seeds[i], seeds[j], alpha)
        seeds.append(mix)

    # Previous best (if present)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # clean‑up (remove possible tiny negatives, renormalise)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size -------------------------------------------------
    n = 4000
    print(f"\n=== Search started (n={n}) ===")

    # -----------------------------------------------------------------
    # Stage 0 – seed pool
    # -----------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # -----------------------------------------------------------------
    # Stage 1 – quick exact evaluation, keep top k
    # -----------------------------------------------------------------
    def exact_obj(v):
        return _objective_exact(v)

    evaluated = [(vec, exact_obj(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Best exact values among seeds (top {top_k}):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # -----------------------------------------------------------------
    # Stage 2 – continuation Adam on the smooth surrogate
    # -----------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = exact_obj(global_best)
    print(f"\n[Init] exact objective = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing Adam runs.")
            break
        remaining = deadline - time.time()
        # allocate a modest per‑beta iteration budget; bounded by remaining time
        per_beta_iter = max(800, int(min(3000, remaining * 0.06)))
        print(f"\n--- Adam run on seed #{idx+1} (per‑β iters={per_beta_iter}) ---")
        opt_vec = _adam_optimize(seed_vec,
                                 n,
                                 deadline,
                                 iters_per_beta=per_beta_iter,
                                 lr0=0.12)
        val = exact_obj(opt_vec)
        print(f"[Adam] exact = {val:.6f}")
        if val < global_best_val:
            global_best = opt_vec.copy()
            global_best_val = val
            print(f"[Update] New global best = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 3 – long exponentiated‑gradient refinement (exact objective)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        remain = deadline - time.time()
        max_exp_iters = int(min(5_000_000, remain * 0.25))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact) ---")
            eg_vec, eg_val = _exp_gradient_optimize(global_best,
                                                    n,
                                                    deadline,
                                                    max_iters=max_exp_iters,
                                                    eta0=1.5,
                                                    power=0.55,
                                                    rng=rng)
            print(f"[EG] best = {eg_val:.6f}")
            if eg_val < global_best_val:
                global_best = eg_vec.copy()
                global_best_val = eg_val
                print(f"[Update] New best after EG = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 4 – stochastic local swap search (tiny mass moves)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        remain = deadline - time.time()
        max_swap_steps = int(min(300_000, remain * 0.15))
        if max_swap_steps > 0:
            print("\n--- Local swap search ---")
            swap_vec, swap_val = _local_swap_search(global_best,
                                                    n,
                                                    deadline,
                                                    max_steps=max_swap_steps,
                                                    rng=rng)
            print(f"[Swap] best = {swap_val:.6f}")
            if swap_val < global_best_val:
                global_best = swap_vec.copy()
                global_best_val = swap_val
                print(f"[Update] New best after swap = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 5 – final high‑β Adam polishing
    # -----------------------------------------------------------------
    if time.time() < deadline:
        remain = deadline - time.time()
        per_beta_iter = max(500, int(min(2000, remain * 0.07)))
        print("\n--- Final high‑β Adam polishing ---")
        final_vec = _adam_optimize(global_best,
                                   n,
                                   deadline,
                                   beta_schedule=[200000.0, 500000.0, 1000000.0],
                                   iters_per_beta=per_beta_iter,
                                   lr0=0.04)
        final_val = exact_obj(final_vec)
        print(f"[Final‑Adam] exact = {final_val:.6f}")
        if final_val < global_best_val:
            global_best = final_vec.copy()
            global_best_val = final_val
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # -----------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging only)
    # -----------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nBest exact ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
