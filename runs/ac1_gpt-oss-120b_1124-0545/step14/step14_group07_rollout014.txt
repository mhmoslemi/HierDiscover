"""
Propose a non‑negative step function that minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Seed pool** – generate a diverse set of ℓ₁‑normalised vectors:
   analytic windows, random Dirichlet draws, random permutations of
   those draws and a few handcrafted “spike” patterns.
2. **Fast EG refinement** – a long exponentiated‑gradient (mirror‑descent
   on the simplex) run on the best few seeds.  The EG step is cheap
   (single FFT per iteration) so we can afford several hundred‑thousand
   updates.
3. **Smooth‑max L‑BFGS‑B** – minimise the smooth surrogate
   ``Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))`` with the logits parametrisation.
   The smoothing parameter p is increased geometrically
   (5 → 15 → 45 → 135 → 405 → 1215) and a short random‑restart is
   performed for each p.  The optimiser is allowed more internal
   iterations than before (up to 500).
4. **Swap‑local‑search** – with the remaining time we try random
   swaps of two entries (projection is automatic because swaps keep the
   ℓ₁‑norm).  Only swaps that improve the exact surrogate are kept.
5. **Final EG polishing** – a last EG run (shorter) from the best
   candidate.
6. **Official evaluation** – the final vector is handed to
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.

All stages respect the global deadline; the routine always returns the
best sequence found before the time limit expires.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective(seq: np.ndarray, n: int) -> float:
    """
    Exact objective that evaluate_sequence returns for a
    ℓ₁‑normalised vector (sum(seq)=1):
        2·n·max(conv(seq,seq))
    """
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _spike_two_peaks(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two‑spike pattern: distribute the ℓ₁ mass on two random positions."""
    a = np.zeros(n, dtype=np.float64)
    i, j = rng.choice(n, size=2, replace=False)
    a[i] = 0.5
    a[j] = 0.5
    return a  # already sums to 1


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else np.full_like(mixed, 1.0 / len(mixed))


# ----------------------------------------------------------------------
# Seed collection (augmented)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int, rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (adds slight flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # two‑spike handcrafted patterns (many random placements)
    for _ in range(4):
        seeds.append(_spike_two_peaks(n, rng))

    # random permutations of Dirichlet draws (order matters for convolution)
    for conc in (0.3, 0.7, 1.5):
        for _ in range(3):
            v = _random_dirichlet(n, conc, rng)
            seeds.append(v)                              # original order
            seeds.append(v[rng.permutation(n)])        # shuffled order
            seeds.append(v[::-1])                       # reversed order

    # keep previous solution (if any) as a seed
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # final sanity normalisation
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 400_000,
                      eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective(x, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize schedule √t
        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)     # avoid overflow
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective(x, n)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate (log‑sum‑exp) and its gradient w.r.t. logits y."""
    a = _softmax(y)               # stays on the simplex
    conv = _conv_fft(a, a)        # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c                # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)   # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                     # fallback: hard max
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)       # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    # chain rule for logits
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 500) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the smooth surrogate; returns exact objective."""
    y0 = np.log(np.maximum(start_a, 1e-15))

    # SciPy minimise will call with (y, n, p) via args
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})

    a_opt = _softmax(res.x)
    val = _objective(a_opt, n)   # exact surrogate (no smoothing)
    return a_opt, val


# ----------------------------------------------------------------------
# Simple swap‑based local search
# ----------------------------------------------------------------------
def _swap_local_search(start: np.ndarray, n: int, deadline: float,
                       max_attempts: int = 5000) -> tuple[np.ndarray, float]:
    """Randomly swap two entries if it improves the exact objective."""
    cur_seq = start.copy()
    cur_val = _objective(cur_seq, n)
    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        cand = cur_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]   # swap preserves ℓ₁‑norm
        cand_val = _objective(cand, n)
        if cand_val < cur_val:
            cur_seq, cur_val = cand, cand_val
            # occasional progress report
            if attempts % 500 == 0:
                print(f"[Swap] improvement after {attempts} attempts → {cur_val:.6f}")
        attempts += 1
    return cur_seq, cur_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a length that balances expressiveness and speed.
    # --------------------------------------------------------------
    n = 4000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Keep the best few according to the exact surrogate
    scored = [(s, _objective(s, n)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – long Exponentiated‑Gradient refinement.
    # --------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective(best_seq, n)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Long EG refinement -------------------------------------------")
        # we give each of the top 5 seeds a generous budget
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # give at most 250 000 iterations or about half of the remaining time
            max_iters = min(250_000, int(remaining * 4000))  # crude estimate
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_iters, eta0=0.30)
            print(f"[EG] seed #{idx+1} → {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** new global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B on a smoothed max‑convolution.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            max_iter = 500
            cand, val = _lbfgsb_optimize(best_seq, n, deadline, p=p,
                                         max_iter=max_iter)
            print(f"[L‑BFGS] p={p:6.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** improved global best to {best_val:.6f}")

            # one or two random‑restart per p
            for restart in range(2):
                if time.time() >= deadline:
                    break
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.5, size=n)
                pert = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(pert, n, deadline, p=p,
                                               max_iter=max_iter)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart p={p:.1f}] → improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Swap‑based local search (cheap but sometimes effective)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Swap‑local‑search -------------------------------------------")
        swap_attempts = 8000
        cand, val = _swap_local_search(best_seq, n, deadline,
                                       max_attempts=swap_attempts)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[Swap] improved global best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final EG polishing (very cheap, helps stabilise)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ------------------------------------------")
        remaining = deadline - time.time()
        # allocate a modest number of iterations proportional to remaining time
        max_iters = int(min(100_000, remaining * 4000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            print(f"[Final EG] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
