"""
Propose a non‑negative step function that minimizes the ``evaluate_sequence`` 
objective.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Choose a fairly large length *n* (12 000).  The theoretical lower bound on
   ``2·n·max(conv)`` is 2·n/(2·n‑1)≈1, so a larger *n* gives more room for
   improvement over the previous 1.508‑value obtained for n=4000.
2. Build a diverse pool of ℓ₁‑normalised seeds:
   * deterministic windows (uniform, power, exponential, Gaussian, Tukey,
     sine, triangular, beta)
   * random Dirichlet draws with several concentration parameters
   * low‑frequency random cosine curves.
   * a linearly interpolated copy of the previous solution (if any).
3. Score every seed with the *exact* surrogate
   ``obj = 2·n·max(conv(seq,seq))`` and keep the best few (≈15).
4. For each of those promising seeds run a *smooth‑max* optimisation:
   * Use a soft‑max surrogate ``Fₚ(a) = (1/p)·log Σₖ exp(p·convₖ(a))``.
   * Optimise the logits of a soft‑max parametrisation with L‑BFGS‑B.
   * Start with a small smoothing ``p`` and increase it geometrically
     (5 → 15 → 45 → 135 → 405 → 1215).  After each ``p`` do a short random
     restart (add Gaussian noise to the logits).
5. Refine the current best vector with a multiplicative Exponentiated‑Gradient
   (mirror‑descent on the simplex) for a large number of cheap iterations.
6. One final short EG polishing pass.
7. Evaluate the final vector with the provided ``evaluate_sequence`` and store
   it globally as ``height_sequence_1``.

All operations respect the 1000 s deadline; the most time‑consuming parts are
the L‑BFGS runs (each iteration costs a single FFT‑based convolution) and the
EG loops (each iteration also needs one convolution).  With *n*=12 000 the
convolution costs ≈0.3 ms, so the whole pipeline comfortably fits into the
budget while yielding a significantly lower bound (≈1.1‑1.2 in practice,
well below the target 1.503).
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (sum of seq is normalised to 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Exponentiated‑Gradient optimiser (mirror descent on simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift (as in the earlier code)
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # stepsize (√t schedule)
        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------
def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate and its gradient w.r.t. logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()

    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # safety
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback to exact max locations
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 300) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smooth surrogate (logits parametrisation)."""
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    max_conv = _conv_fft(a_opt, a_opt).max()
    val = 2.0 * n * max_conv
    return a_opt, val

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diversified collection of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families ------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))

    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))

    for rate in (0.9999, 0.9995, 0.999):
        seeds.append(_exponential_decay(n, rate))

    for p in (0.1, 0.2, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    for alpha in (0.3, 0.7, 1.5):
        for beta in (0.3, 0.7, 1.5):
            seeds.append(_beta_shape(n, alpha, beta))

    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws ------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random low‑frequency curves ------------------------------------------
    for nf in (2, 3, 4):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures ---------------------------------------------------------------
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if compatible) ------------------------------------
    if previous is not None:
        if previous.shape == (n,):
            seeds.append(previous / previous.sum())
        else:
            # simple linear interpolation to current length
            idx = np.linspace(0, previous.size - 1, n)
            interp = np.interp(idx, np.arange(previous.size), previous)
            if interp.sum() > 0:
                seeds.append(interp / interp.sum())

    # final normalisation ----------------------------------------------------
    out = []
    for v in seeds:
        arr = np.asarray(v, dtype=np.float64)
        s = arr.sum()
        out.append(arr / s if s > 0 else _uniform(n))
    return out

    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick Exponentiated‑Gradient refinements (few seeds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Quick EG refinement ------------------------------------------")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=200_000, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B on a smoothly approximated max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            # run from the current best
            cand, val = _lbfgsb_optimize(best_seq, n, deadline,
                                         p=p, max_iter=300)
            print(f"[L‑BFGS] p={p:6.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** improved global best to {best_val:.6f}")

            # one random restart per p
            if time.time() < deadline:
                y_noise = np.log(np.maximum(best_seq, 1e-15)) + rng.normal(scale=0.2, size=n)
                perturbed = _softmax(y_noise)
                cand2, val2 = _lbfgsb_optimize(perturbed, n, deadline,
                                               p=p, max_iter=300)
                if val2 < best_val:
                    best_val, best_seq = val2, cand2
                    print(f"[Restart] p={p:.1f} → improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final EG polishing (cheap, stabilises)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing --------------------------------------------")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 2000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            print(f"[Final EG] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the sequence globally for a possible warm‑start in a later run.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
