"""
Search for a step‑function that minimises the official ``evaluate_sequence``
objective.  The optimiser works directly on the L∞‑norm of the self‑convolution:

* start from the previously best sequence (or a uniform seed);
* repeatedly locate the shift k where the convolution is maximal;
* move a small amount of probability mass from an index that contributes a lot
  to that shift to an index that contributes very little;
* accept the move if it improves the surrogate objective or, with a
  Simulated‑Annealing probability, if it is a mild deterioration (to escape
  local minima);
* keep the best solution ever seen and return it.

All operations stay on the probability simplex (∑a_i = 1) so the objective is
scale‑invariant and the clipping performed by ``evaluate_sequence`` never
affects the optimisation.  The algorithm is lightweight, uses only NumPy and
fits comfortably inside the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Surrogate for ``evaluate_sequence``.
    The sequence is assumed to be non‑negative; the scale factor cancels out.
    """
    n = len(a)
    # max of a*a (full convolution)
    max_conv = np.max(_conv_fft(a, a))
    # denominator = (∑a)^2 ; here we keep the exact scaling because a may not be normalised
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (sum_a ** 2)

    accepting moves according to a simple simulated‑annealing schedule.

    rng = np.random.default_rng(seed)

    else:

    # normalise to the simplex (scale does not matter for the objective)
    s = a.sum()
    if s <= 0.0:

    else:

    cur_val = _objective_surrogate(a)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Annealing schedule
    T = max(best_val * 0.15, 1e-12)    # start temperature (a fraction of the current objective)
