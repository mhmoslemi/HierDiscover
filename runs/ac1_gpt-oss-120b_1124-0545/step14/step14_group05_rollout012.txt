"""
Propose a step‑function (non‑negative heights) that minimises the official
``evaluate_sequence`` objective.

The search consists of four stages:
  1. Generate a very diverse pool of normalised seeds (uniform, triangular,
     Gaussian, Beta, Tukey, several random Dirichlet draws, sparse supports,
     low‑frequency random curves, power‑law and exponential decay).
  2. Refine the best few seeds with a soft‑max Adam optimiser that smoothes the
     max‑convolution using a log‑sum‑exp surrogate (α is annealed from 20 to
     10 000).
  3. Perform a second Adam phase that directly minimises the L2‑distance between
     the self‑convolution and the flat target (1/(2·n‑1) for each entry).  The
     gradient of the MSE loss is obtained analytically via FFT‑based
     convolution/correlation and the update is projected onto the probability
     simplex.
  4. Optional polishing with the original sparse exponential‑gradient update and
     a simulated‑annealing fine‑tuning.

All stages respect the global time budget (default 1000 s) and the best
sequence found is stored in the global name ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(p,p)) for a distribution p
    (∑p = 1).  The vector p is assumed non‑negative.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(
                np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Vector with exactly ``m`` non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _power_law(n: int, exponent: float) -> np.ndarray:
    """i⁻ᵉ power‑law (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** exponent)
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i = exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    return a / a_sum


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    alpha_start: float = 20.0,
    alpha_end: float = 5000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth log‑sum‑exp surrogate of max(conv)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0,
                                                              t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _mse_adam_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L² loss between the self‑convolution and a flat
    target (value = 1/(2·n‑1) for each entry).  The gradient is obtained via
    FFT‑based convolution/correlation and the iterate is projected onto the
    probability simplex.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    # Fixed FFT size large enough for linear convolutions of length 3·n‑2
    N = _next_pow2(3 * n - 2)

    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val, dtype=np.float64)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution p * p (full) via FFT
        fx = np.fft.rfft(x, n=N)
        conv = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        diff = conv - target          # shape (2·n‑1,)

        # gradient of (1/2)||diff||² w.r.t. x:
        #   g = 2 * (conv(diff, x) + shifted version for the symmetric term)
        diff_fft = np.fft.rfft(diff, n=N)

        # conv(diff, x)  – linear convolution, length 3·n‑2
        conv1 = np.fft.irfft(diff_fft * fx, n=N)[: 3 * n - 2]

        # conv(diff, reversed x)
        xr = x[::-1]
        fxr = np.fft.rfft(xr, n=N)
        conv2 = np.fft.irfft(diff_fft * fxr, n=N)[: 3 * n - 2]

        # Assemble the full gradient (see derivation in analysis)
        grad = 2.0 * (conv1[:n] + conv2[n - 1:n - 1 + n])

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20000 == 0:
                print(f"[MSEAdam] it={t:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.25,
    sparsity_thr: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Sparse exponential‑gradient (mirror descent) optimisation."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # optional sparsity projection
        if sparsity_thr > 0.0:
            mask = x < sparsity_thr
            if np.any(mask):
                x[mask] = 0.0

        # simplex projection
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the surrogate objective."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.5:
            # reshuffle a random subset (size 5‑15)
            k_sub = rng.integers(5, 16)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_tukey_window(n, 0.5))
    # additional analytic families
    seeds.append(_power_law(n, 0.5))
    seeds.append(_power_law(n, 1.0))
    seeds.append(_power_law(n, 2.0))
    seeds.append(_exp_decay(n, 0.001))
    seeds.append(_exp_decay(n, 0.005))
    seeds.append(_exp_decay(n, 0.01))

    # Very sparse Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Exact‑size sparse supports
    for m in (120, 200, 300, 400):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # Low‑frequency random curves (as in the reference implementation)
    for nf in (2, 3):
        for _ in range(4):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                seeds.append(a / s)

    # Mixtures of uniform and a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # Previous best (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Final clean‑up (ensure proper normalisation)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search – within ``budget_s`` seconds – for a non‑negative step function
    that minimises the official ``evaluate_sequence`` objective.
    The pipeline consists of:
      * diverse seed generation,
      * soft‑max Adam polishing,
      * MSE‑Adam flattening of the convolution,
      * sparse exponential‑gradient polishing, and
      * simulated‑annealing fine‑tuning.
    The best sequence is returned as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Length of the step function – 4000 works well for the time budget.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – generate a large and very diverse seed library
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # Evaluate surrogate and keep the top performing ones
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            max_iters = int(min(200_000, remaining * 120))  # a bit more time per start
            cand, val = _adam_softmax_optimize(
                seed_vec, n, deadline,
                max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=10000.0,
                lr=0.12,
            )
            if val < best_val:
                best_seq, best_val = cand, val
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – MSE Adam optimisation (flatten the convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- MSE Adam optimisation (flatten conv) ---")
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 150))
        cand, val = _mse_adam_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            lr=0.05,
        )
        if val < best_val:
            best_seq, best_val = cand, val
            print(f"[MSEAdam] improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – sparse exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sparse Exponential‑Gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline,
            max_iters=max_iters,
            eta0=0.25,
            sparsity_thr=1e-8,
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(700_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
