"""
Propose a better non‑negative step‑function (list of heights).

The algorithm works in several stages:

1. **Seed collection** – A large and diverse pool of ℓ₁‑normalised vectors is
   generated from many analytic families (Gaussian, Kaiser, Tukey, etc.),
   random Dirichlet / exponential draws and a set of low‑frequency curves.
   Randomised parameters are sampled to explore the families more thoroughly
   than the previous “fixed‑grid” approach.

2. **Pre‑selection** – The surrogate objective   2·n·max(conv(a,a))   (exact
   for ℓ₁‑normalised vectors) is evaluated for every seed.  The best ``top_k``
   candidates are retained.

3. **L‑BFGS refinement** – The best candidate is refined with a continuation
   L‑BFGS optimiser that minimises a smooth log‑sum‑exp approximation of the
   max‑convolution.  The schedule of smoothing parameters (β) is extended to
   higher values than before and the number of L‑BFGS iterations per β is
   increased.

4. **Adam polishing** – Starting from the L‑BFGS output, a custom Adam
   optimiser runs directly in log‑softmax space on the same smoothed objective.
   The β‑schedule is continued to still larger values (up to 2·10⁴) so that
   the smooth objective becomes essentially the true max‑convolution.  Adam’s
   adaptive learning‑rate quickly fine‑tunes the solution.

5. **Final evaluation** – The best vector found is finally evaluated with the
   official ``evaluate_sequence`` function and stored in the global variable
   ``height_sequence_1`` for possible reuse.

The whole search respects the given time budget (default 1000 s) by checking
the deadline continuously.  All helper functions are defined at top level,
use only NumPy (and SciPy if available), and avoid closures or lambda
expressions.
"""

import time
import math
import numpy as np

# optional SciPy import for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a ℓ₁‑normalised vector:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0 -
         a1 * np.cos(2 * np.pi * i / (n - 1)) +
         a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0 -
         a1 * np.cos(2 * np.pi * i / (n - 1)) +
         a2 * np.cos(4 * np.pi * i / (n - 1)) -
         a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)            # Modified Bessel function (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    v = np.log(start_vec + 1e-30)      # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights for gradient: softmax on β·conv
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

            # gradient wrt logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_refine(start_seq: np.ndarray,
                 n: int,
                 deadline: float) -> np.ndarray:
    """
    Adam optimiser in log‑softmax space on the log‑sum‑exp smoothed max.
    Returns the best ℓ₁‑normalised vector found before the deadline.
    """
    # initialise logits
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    vhat = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # schedule of smoothing parameters (larger → tighter approximation)
    schedule = [(10, 12000),
                (100, 10000),
                (500, 8000),
                (2000, 6000),
                (8000, 4000),
                (20000, 3000)]

    for beta, steps in schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        for _ in range(steps):
            if time.time() > deadline:
                break
            t += 1

            a = _softmax(v)                      # current probabilities
            conv = _conv_fft(a, a)
            shift = np.max(conv)                 # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth objective (unused for gradient, kept for possible logging)
            # obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                # softmax on β·conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad_v
            vhat = beta2 * vhat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat_corr = vhat / (1.0 - beta2 ** t)

            # learning‑rate scales roughly as 1/√β to stay stable for large β
            lr = 0.05 / math.sqrt(beta_val + 1e-12)

            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # periodic evaluation of the true surrogate
            if (t % 2000) == 0:
                cur_a = _softmax(v)
                cur_val = _objective_surrogate(cur_a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_a.copy()
    return best_seq

        sigma = rng.uniform(0.05, 0.5)
        seeds.append(_gaussian(n, sigma))
    # – Sine window
    for _ in range(120):
        gamma = rng.uniform(0.2, 3.0)
        seeds.append(_sine_window(n, gamma))
    # – Kaiser
    for _ in range(120):
        beta = rng.uniform(1.0, 14.0)
        seeds.append(_kaiser_window(n, beta))
    # – Tukey
    for _ in range(120):
        alpha = rng.uniform(0.05, 0.95)
        seeds.append(_tukey_window(n, alpha))
    # – Power law
    for _ in range(120):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))
    # – Beta shape
    for _ in range(120):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        seeds.append(_beta_shape(n, a, b))
    # – Double Gaussian
    for _ in range(80):
        sigma_frac = rng.uniform(0.05, 0.2)
        shift_frac = rng.uniform(0.1, 0.4)
        seeds.append(_double_gaussian(n, sigma_frac, shift_frac))

    # previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanity check / normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000                     # number of points (adjustable)

    # ------------------------------------------------------------------
    # 2. Gather seeds (analytic families + randomised parameters)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seed vectors.")

    # ------------------------------------------------------------------
    # 3. Keep the best few candidates (by surrogate)
    # ------------------------------------------------------------------
    top_k = 8
    best_candidates = []   # list of (seq, val)

    def _consider(seq: np.ndarray):
        val = _objective_surrogate(seq)
        if len(best_candidates) < top_k:
            best_candidates.append((seq, val))
            best_candidates.sort(key=lambda x: x[1])
        else:
            if val < best_candidates[-1][1]:
                best_candidates[-1] = (seq, val)
                best_candidates.sort(key=lambda x: x[1])

    for vec in seeds:
        _consider(vec)

    print("\nBest surrogate values among seeds (top {}):".format(top_k))
    for i, (_, val) in enumerate(best_candidates):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement on the best seed
    # ------------------------------------------------------------------
    global_best_seq = best_candidates[0][0].copy()
    global_best_val = best_candidates[0][1]
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    if time.time() < deadline and _opt is not None:
        try:
            # give a relatively generous iteration budget per β
            seq_opt = _lbfgs_smooth_optimize(global_best_seq,
                                             n,
                                             deadline,
                                             beta_schedule=[1.0, 5.0, 20.0,
                                                            80.0, 200.0,
                                                            500.0, 1000.0,
                                                            2000.0, 5000.0,
                                                            10000.0, 20000.0],
                                             max_iter_per_beta=12000)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] failed:", e)

    # ------------------------------------------------------------------
    # 5. Adam polishing (smooth‑max continuation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam polishing (smooth‑max continuation) ---")
        seq_adam = _adam_refine(global_best_seq, n, deadline)
        val_adam = _objective_surrogate(seq_adam)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] new global best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
