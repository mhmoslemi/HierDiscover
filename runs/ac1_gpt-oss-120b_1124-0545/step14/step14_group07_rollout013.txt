def build_lp_matrix(a):
    n = len(a)
    # Build main part A with shape (2n-1, n)
    # Coefficient for column j is 2*a[(row - j)] for row j <= row < j+n
    # Use diags: diag_data = [2*a[i]*np.ones(n) for i in range(n)]
    # offsets = range(n)
    A_main = diags([2*a[i] * np.ones(n) for i in range(n)], offsets=range(n),
                   shape=(2*n-1, n), format='csr')
    # Create column for t: -1 for each row
    t_col = -np.ones((2*n-1, 1))
    # Combine into A_ub (2n-1, n+1)
    A_ub = sp.hstack([A_main, t_col], format='csr')
    return A_ub

def downsample(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return np.array(seq, dtype=np.float64)
    # simple: use interpolation
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n-1, new_n)
    return np.interp(x_new, x_old, seq)

"""
Propose a non‑negative step function that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Initialisation** – generate a diverse pool of ℓ₁‑normalised vectors
   (uniform, sine windows, Gaussian, random Dirichlet, low‑frequency
   curves).  Keep the best few (according to the exact objective) as
   starting points.

2. **Fast EG polishing** – run a few rounds of exponentiated‑gradient
   (mirror‑descent on the simplex) on each seed.  This quickly reduces
   the maximal auto‑convolution.

3. **Iterative linear‑program (LP) refinement** – starting from the
   current best vector `a`, form the linearised constraints

        2·conv(a, a_current)_k  –  t ≤ conv(a_current, a_current)_k

   (conv(a, a_current) is linear in the unknown vector).  Solving the
   LP (minimise `t`) yields a new candidate that strictly reduces the
   true maximum when the linearisation is tight.  Repeat a few times.

   The LP is built with a *sparse* Toeplitz matrix, which makes each
   iteration cheap even for a few thousand coefficients.

4. **Final EG polishing** – a short EG run from the best LP solution
   removes any remaining small irregularities.

The whole search respects the 1000 s deadline, automatically adapting
the amount of work to the remaining time.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective (2·n·max(conv(seq,seq))) for a sequence with Σ=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 150_000, eta0: float = 0.25):
    """Multiplicative updates with decreasing step size."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # projection onto simplex (just normalise, all entries ≥0)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of a smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


def _lbfgsb_func(y: np.ndarray, n: int, p: float):
    """Smooth surrogate (log‑sum‑exp) and its gradient."""
    a = _softmax(y)
    conv = _conv_fft(a, a)
    max_c = conv.max()
    shifted = conv - max_c
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:               # fallback (should not happen)
        w = (conv == max_c).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)       # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1: n - 1 + n]

    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


def _lbfgsb_optimize(start_a: np.ndarray, n: int, deadline: float,
                     p: float = 20.0, max_iter: int = 250):
    """L‑BFGS‑B on the smooth surrogate (log‑sum‑exp)."""
    y0 = np.log(np.maximum(start_a, 1e-15))
    res = _opt.minimize(_lbfgsb_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)
    return a_opt, val

# LP‑based linearisation refinement
# ----------------------------------------------------------------------
def _lp_refine(start_a: np.ndarray, deadline: float, max_iters: int = 5):
    """
    Repeatedly solve a linear programme that linearises the quadratic
    constraints  a·a ≤ t  around the current vector.
    Returns a (ℓ₁‑normalised) vector with a lower exact objective if possible.
    """
    n = len(start_a)
    a = start_a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv_cur = _conv_fft(a, a)                     # length 2n‑1
        # Build sparse matrix for 2·conv(a,·) ≤ t + conv_cur
        # Each diagonal offset i contains 2·a[i].
        diag_data = [2.0 * a[i] * np.ones(n) for i in range(n)]
        offsets = list(range(n))
        A_main = _sp.diags(diag_data, offsets,
                           shape=(2 * n - 1, n), format='csr')
        # column for the t‑variable (coefficient –1)
        t_col = -np.ones((2 * n - 1, 1))
        A_ub = _sp.hstack([A_main, t_col], format='csr')
        b_ub = conv_cur                               # RHS of the inequalities

        # Equality: sum a_i = 1  (t has coefficient 0)
        A_eq = np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1)
        b_eq = np.array([1.0])

        # Objective: minimise t
        c = np.concatenate([np.zeros(n), np.array([1.0])])

        bounds = [(0, None)] * n + [(0, None)]

        res = _opt.linprog(c,
                           A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds,
                           method='highs')
        if not res.success:
            # LP failure – abort refinement
            print(f"[LP iter {it+1}] linprog failed: {res.message}")
            break

        x = res.x
        new_a = x[:n]
        # Numerical safeguard
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s == 0.0:
            break
        new_a = new_a / s

        new_val = _objective_surrogate(new_a)
        if new_val < best_val - 1e-12:
            best_val = new_val
            a = new_a
            print(f"[LP iter {it+1}] improved surrogate → {best_val:.6f}")
        else:
            # No progress – stop early
            break

    return a, best_val


# ----------------------------------------------------------------------
# Helper: down‑sample / up‑sample a vector to a target length
# ----------------------------------------------------------------------
def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to obtain a vector of length ``new_n``."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, seq)

    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Generated {len(scored)} seeds, keeping top {top_k} with surrogate values:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Quick EG polishing of the best seeds.
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (quick) ---------------------")
        for idx, seed_vec in enumerate(best_seeds[:5]):   # a handful of promising seeds
            if time.time() >= deadline:
                break
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=120_000, eta0=0.30)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS‑B on the smooth max‑convolution surrogate (more thorough).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B refinement ------------------------------------------")
        p_schedule = [5.0, 15.0, 45.0, 135.0, 405.0, 1215.0]
        for p in p_schedule:
            if time.time() >= deadline:
                break
            cand, val = _lbfgsb_optimize(best_seq, n, deadline, p=p, max_iter=500)
            print(f"[L‑BFGS] p={p:7.1f} → surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"    ** global best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # LP‑based linearisation refinement (the main novelty).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP‑based linearisation refinement ---------------------------")
        max_lp_iters = 6          # a few LP solves are cheap for n≈2000
        cand, val = _lp_refine(best_seq, deadline, max_iters=max_lp_iters)
        if val < best_val:
            best_val, best_seq = val, cand
            print(f"[LP refinement] improved surrogate → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final EG polishing (cheap, stabilises the solution).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing -------------------------------------------")
        remaining = deadline - time.time()
        max_iters = int(min(80_000, remaining * 2000))
        if max_iters > 0:
            cand, val = _expgrad_optimize(best_seq, n, deadline,
                                          max_iters=max_iters, eta0=0.25)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
