def _hill_climb(start_seq, n, deadline):
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)
    it = 0
    max_iters = 500_000
    # parameters
    start_temp = cur_val * 0.01  # maybe 0.01 fraction
    temp_decay = 0.99995  # decays per iteration
    eps_factor = 0.1  # initial epsilon factor of min(a[i], 0.01)
    eps_decay = 0.9999

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # choose i where a[i] > 0
        i = rng.integers(n)
        # ensure we have mass to move
        if a[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # epsilon adaptively
        eps_max = min(a[i], eps_factor)
        epsilon = rng.random() * eps_max
        # propose move
        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # maybe renorm small drift
        # compute surrogate
        val_new = _objective_surrogate(a_new)
        # acceptance
        # temperature schedule
        temp = start_temp * (temp_decay ** it)
        if val_new < cur_val or rng.random() < np.exp(-(val_new - cur_val) / max(temp, 1e-12)):
            a = a_new
            cur_val = val_new
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # update eps_decay gradually reduce epsilon
        if it % 1000 == 0 and it > 0:
            eps_factor *= eps_decay
            # re-normalize to sum 1 to avoid drift
            sum_a = a.sum()
            if abs(sum_a - 1.0) > 1e-12:
                a = a / sum_a
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    return best_seq

    best_seq = a.copy()

        2·n·max(conv(a, a)) / (sum(a)²)

Algorithm

        S(a)=2·n·max(conv(a,a))     (sum(a)=1 ⇒ true objective=S.
    Keep the best few.
3.  Refine the elite seed with a continuation L‑BFGS optimiser
    (smooth max via log‑sum‑exp).  This gives a high‑quality local
    minimum.
4.  Perform a long exact‑subgradient (exponentiated‑gradient) run
    starting from the L‑BFGS output (the same routine used in the

5.  **Hill‑climbing / simulated‑annealing phase** – starting from the
    point obtained in step 4, repeatedly move a small amount of mass
    from the currently largest entry to a random other entry.
    Accept improvements always; accept worsening moves with a
    temperature that decays linearly to zero.  This non‑gradient
    exploration often escapes the plateau left by the sub‑gradient
    method and yields a modest but consistent improvement.
6.  An optional up‑sampling (if >30 s remain) doubles the grid,
    repeats the L‑BFGS + exponentiated‑gradient + hill‑climbing
    refinement on the finer grid, and finally keeps the better of
    the two lengths.
7.  Return the best sequence as a plain list and store it in the

import time

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^p  (increasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30  # avoid exact zero before normalisation
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture, then ℓ₁‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises a smooth surrogate of max‑convolution using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stay on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # numeric guard
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 600_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 500_000) -> np.ndarray:
    """
    Simple simulated‑annealing style hill‑climber.
    Moves a small amount of mass from the current largest entry to a
    random other entry, accepting improvements always and uphill moves
    with a linearly decaying temperature.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(0xC0FFEE)
    total_iters = max_iters

    for it in range(total_iters):
        if time.time() > deadline:
            break

        # linear schedules for temperature and step size
        progress = it / total_iters
        temp = (cur_val * 0.05) * (1.0 - progress) + (cur_val * 1e-6) * progress
        eps = 0.10 * (1.0 - progress) + 1e-6 * progress

        # donor = index of the largest component (greedy reduction of peaks)
        i = int(np.argmax(a))
        if a[i] <= 1e-15:
            continue
        # random target, different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to shift
        epsilon = eps * rng.random()
        epsilon = min(epsilon, a[i])           # never go negative

        # apply move
        a[i] -= epsilon
        a[j] += epsilon

        new_val = _objective_surrogate(a)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            # record best if improved
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                if it % 20000 == 0:
                    print(f"[HillClimb] it={it:,} best≈{best_val:.6f}")
        else:
            # revert the move
            a[j] -= epsilon
            a[i] += epsilon

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for termination

    # ---------------------------------
    # Stage 0 – problem size
    # ---------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # ---------------------------------
    # Stage 1 – seed generation & cheap ranking
    # ---------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ---------------------------------
    # Stage 2 – continuation L‑BFGS on the elite seed
    # ---------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 5000.0]

    # we only spend time on the single best seed – most powerful refinement
    if time.time() < deadline:
        max_iter_beta = 8000
        print("\n--- L‑BFGS refinement of the best seed ---")
        seq_opt = _lbfgs_smooth_optimize(global_best_seq, n, deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # ---------------------------------
    # Stage 3 – exact sub‑gradient (mirror‑descent) refinement
    # ---------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize(global_best_seq, n, deadline,
                                            max_iters=max_exp_iters)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ---------------------------------
    # Stage 4 – hill‑climbing (simulated annealing) phase
    # ---------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(800_000, remaining * 0.30 * 1e5))
        if max_hc_iters > 0:
            print("\n--- Hill‑climbing (simulated annealing) ---")
            seq_hc = _hill_climb(global_best_seq, n, deadline,
                                 max_iters=max_hc_iters)
            val_hc = _objective_surrogate(seq_hc)
            if val_hc < global_best_val:
                global_best_seq = seq_hc.copy()
                global_best_val = val_hc
                print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ---------------------------------
    # Stage 5 – optional up‑sampling & fine‑tuning on a finer grid
    # ---------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # L‑BFGS on the finer grid (shorter schedule)
            up_opt = _lbfgs_smooth_optimize(up_seq, new_n, deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=4000)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # another round of exact mirror‑descent
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp2 = int(min(800_000, rem * 0.10 * 1e5))
                if max_exp2 > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_final = _exp_gradient_optimize(global_best_seq, n,
                                                       deadline,
                                                       max_iters=max_exp2)
                    val_final = _objective_surrogate(seq_final)
                    if val_final < global_best_val:
                        global_best_seq = seq_final.copy()
                        global_best_val = val_final
                        print(f"[Final‑ExpGrad] improved to {global_best_val:.6f}")

    # ---------------------------------------------------------
    # Final evaluation with the official evaluator and return
    # ---------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
