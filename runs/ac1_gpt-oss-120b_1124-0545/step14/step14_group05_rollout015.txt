"""
Propose a step‑function (non‑negative heights) with a low official
`evaluate_sequence` score.

Strategy
--------
1. Build a large and very diverse pool of candidate vectors.
   * Analytic families (uniform, triangular, Gaussian, Beta, Tukey …)
   * Edge‑heavy beta (α,β < 1) – mass concentrated near the ends.
   * Random binary masks with various densities (≈30‑70 % active).
   * Random sparse supports with size up to ≈½ n.
2. Evaluate the cheap surrogate  2·n·max(conv)  on all seeds and keep the
   best few.
3. Refine each selected seed with a **mirror‑descent (exponential‑gradient)**
   optimiser that directly minimises the maximum convolution entry.
   The sub‑gradient of the active convolution index *k* is
        g_i = 2·a_{k‑i}   (zero outside the valid range).
   Multiplicative updates preserve positivity; after each step we project
   back onto the probability simplex (∑a_i = 1).
4. If time permits, a short simulated‑annealing polishing pass is run.
5. Return the best sequence found (stored in the global name
   `height_sequence_1` for reuse).

The whole search respects the time budget (`budget_s` seconds) and is
fully deterministic given the seed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, ≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(p,p)) for a probability vector
    (∑p = 1).  Faster than the official ``evaluate_sequence``.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _inverse_triangular(n: int) -> np.ndarray:
    # high on the edges, low in the centre
    a = 1.0 - _triangular(n)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Vector with exactly `m` non‑zero equal entries (∑=1)."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _random_binary_mask(n: int, p: float, rng: np.random.Generator) -> np.ndarray:
    """Bernoulli mask with probability `p`; uniform weight on the selected entries."""
    mask = rng.random(n) < p
    active = mask.sum()
    if active == 0:
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[mask] = 1.0 / active
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Optimiser – mirror descent on max‑convolution
# ----------------------------------------------------------------------
def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.5,
    sparsity_thr: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponential) mirror descent on the exact objective
    ``max(conv)``.  The algorithm works on the probability simplex, i.e.
    ``∑a_i = 1`` and ``a_i ≥ 0``.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active quadratic component:
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        # avoid overflow in the exponential
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        if sparsity_thr > 0.0:
            x[x < sparsity_thr] = 0.0

        # projection onto the simplex (performs the normalisation)
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best surrogate={best_val:.6f}")

    return best_x, best_val


# ----------------------------------------------------------------------
# Optional simulated annealing for fine‑tuning (kept lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        T = max(T * 0.99996, min_temp)

        # propose a tiny mass transfer between two random indices
        i, j = rng.choice(n, size=2, replace=False)
        eps = rng.random() * min(cur[i], 0.02)
        if eps <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:,} best surrogate={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diversified set of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # classical families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverse_triangular(n))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_beta_shape(n, 2.0, 2.0))          # bell shaped
    seeds.append(_beta_shape(n, 0.5, 0.5))          # U‑shaped (mass at ends)
    seeds.append(_tukey_window(n, 0.5))

    # random binary masks with a range of densities
    for p in (0.3, 0.4, 0.5, 0.6, 0.7):
        for _ in range(3):
            seeds.append(_random_binary_mask(n, p, rng))

    # exact‑size sparse supports (up to half the length)
    for m in (800, 1200, 1600, 2000, 2500):
        for _ in range(2):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency random curves (as in the reference code)
    for nf in (2, 3):
        for _ in range(4):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                seeds.append(a / a.sum())

    # mixtures of uniform with a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (800, 1600):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previous best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final clean‑up – ensure proper normalisation and finite entries
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        # clip any unexpected NaNs / infinities
        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the official
    ``evaluate_sequence`` objective.  The search runs for at most
    ``budget_s`` seconds and stores the best found sequence in the global
    name ``height_sequence_1``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin for final I/O

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – diverse seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # Evaluate the cheap surrogate and keep the top few seeds
    # ------------------------------------------------------------------
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – mirror‑descent (exponential‑gradient) from each seed
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a fraction of remaining time to this seed
        max_iters = int(min(300_000, remaining * 1300))
        print(f"\n[ExpGrad] start seed #{idx+1} (max_iters={max_iters})")
        cand_seq, cand_val = _expgrad_optimize(
            seed_vec, n, deadline,
            max_iters=max_iters,
            eta0=0.5,
            sparsity_thr=1e-12,
        )
        print(f"[ExpGrad] finished seed #{idx+1} → surrogate={cand_val:.6f}")
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – optional simulated‑annealing fine‑tuning (time‑permitting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑Annealing polishing (optional) ---")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
