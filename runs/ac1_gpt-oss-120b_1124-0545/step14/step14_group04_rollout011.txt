def random_phase_seed(n, rng):
    L = 2*n - 1
    N = _next_pow2(L)
    half = N // 2
    # For real-valued IFFT we need conj-symmetric.
    # random phases for k=1..half-1
    phi = rng.uniform(0.0, 2*np.pi, size=half-1)
    X = np.empty(N, dtype=complex)
    X[0] = 1.0  # amplitude 1, phase 0
    X[1:half] = np.exp(1j * phi)
    # for even N: Nyquist frequency must be real.
    if N % 2 == 0:
        X[half] = 1.0
        X[half+1:] = np.conj(X[1:half][::-1])
    else:
        X[half+1:] = np.conj(X[1:half][::-1])
    y = np.fft.ifft(X).real
    # take absolute squared for non-negative
    a = np.abs(y)**2
    a = a[:n]  # first n entries (rest are zero pad)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = _uniform(n)
    return a
```

"""
Simple random‑search + simulated‑annealing optimizer for the
step‑function minimisation problem.

The algorithm works in three phases:

0️⃣  Generate a huge, diverse pool of ℓ₁‑normalised candidates
    (uniform, Dirichlet draws, exponential draws, random‑phase
    constructions and the previous best solution).

1️⃣  Keep the top‑K candidates according to the exact objective
    2·n·max(conv(a,a)) (the surrogate used by ``evaluate_sequence``).

2️⃣  Starting from the best candidate run a lightweight simulated‑annealing
    that repeatedly reshuffles a random subset of entries or transfers a
    tiny amount of mass between two positions.  Accept worsening moves
    with a temperature that decays geometrically; keep the best sequence
    ever seen.

The whole routine respects a global time budget (default 1000 s) and
always returns a valid (non‑negative, ℓ₁‑normalised) list, storing it
in the global name ``height_sequence_1`` for reuse in subsequent runs.
"""

import time
import math
import numpy as np
import heapq

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for long vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    This matches the value returned by ``evaluate_sequence`` after the
    internal safety clamping because the sum is 1 and all entries are
    already in the allowed range.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) and normalise (α = conc)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    # dirichlet already sums to 1, but guard against pathological zero‑sum
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_phase_squared(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a vector a_i = |ifft(exp(i·φ))|², where φ_k are uniform random
    phases and the spectrum is forced to be Hermitian‑symmetric so that
    the inverse FFT is real.  The resulting a is non‑negative.
    """
    L = 2 * n - 1
    N = _next_pow2(L)

    # Build a Hermitian‑symmetric spectrum with unit magnitude.
    half = N // 2
    phi = rng.uniform(0.0, 2.0 * np.pi, size=half - 1)
    X = np.empty(N, dtype=np.complex128)
    X[0] = 1.0 + 0j                                   # DC component (real)
    X[1:half] = np.exp(1j * phi)                     # random phases
    if N % 2 == 0:
        X[half] = 1.0 + 0j                           # Nyquist (real)
        X[half + 1:] = np.conj(X[1:half][::-1])
    else:
        X[half + 1:] = np.conj(X[1:half][::-1])

    y = np.fft.ifft(X).real
    a = np.abs(y) ** 2
    a = a[:n]
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Simulated annealing (tiny local moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-13) -> np.ndarray:
    """
    Very cheap random walk that either reshuffles a small random subset
    (Dirichlet re‑allocation) or transfers a tiny amount of mass between two
    coordinates.  The temperature decays geometrically; the best sequence
    seen so far is retained.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)

    best = cur.copy()
    best_val = cur_val

    # Initial temperature proportional to the current value.
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.99994, min_temp)

        if rng.random() < 0.55:
            #  --- Subset reshuffle -------------------------------------------------
            k_sub = rng.integers(3, 11)               # size of the random subset
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total > 0.0:
                # re‑allocate the total mass uniformly at random (Dirichlet)
                new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
                cand = cur.copy()
                cand[idx] = new_alloc
            else:
                continue
        else:
            #  --- Tiny transfer between two positions -------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.1)      # move at most 10 % of the source
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best = cand.copy()
                best_val = new_val
                # occasional progress report
                if (it % 100_000) == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    print(f"[SA] finished – best surrogate = {best_val:.6f}")
    return best

    # 1️⃣  Simulated‑annealing polishing (most of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        # Allocate roughly the remaining time to SA (iters ≈ remaining·1500)
        max_iters = int(min(800_000, remaining * 1500))
        best_seq = _simulated_annealing(best_seq,
                                        n,
                                        deadline,
                                        max_iters=max_iters,
                                        init_temp_factor=0.08)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
