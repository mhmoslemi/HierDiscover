def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step sequence that minimizes evaluate_sequence.
    Strategy:
    1️⃣ Generate a large pool of candidate vectors from analytic families,
       random Dirichlet draws and parameter sweeps (exponential, power, Gaussian).
    2️⃣ Keep the best few (by surrogate objective) in a heap.
    3️⃣ Run a long sub‑gradient descent from the current best vector,
       followed by an exponential‑gradient polishing and a brief simulated‑annealing
       fine‑tune.
    Returns the best sequence as a list of floats.
    """
    import time, math, numpy as np, heapq

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    n = 4000
    print(f"\n--- Sequence length: n = {n} ---")

    top_k = 30
    heap = []  # (-value, vector)

    def push_candidate(vec, val):
        """Maintain a max‑heap of size <= top_k with smallest values."""
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))

    # ------------------------------------------------------------------
    # 0️⃣  Deterministic analytic families (as in previous solution)
    # ------------------------------------------------------------------
    print("\n--- Deterministic analytic seeds ---")
    analytic_seeds = []
    analytic_seeds.append(_uniform(n))
    analytic_seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        analytic_seeds.append(_sine_window(n, gamma))
    analytic_seeds.append(_gaussian(n, 0.2))
    analytic_seeds.append(_gaussian(n, 0.3))
    analytic_seeds.append(_gaussian(n, 0.4))
    analytic_seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        analytic_seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            analytic_seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        analytic_seeds.append(_tukey_window(n, alpha))

    # add previous solution if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size == n and arr.sum() > 0:
                analytic_seeds.append(arr / arr.sum())
        except Exception:
            pass

    for vec in analytic_seeds:
        val = _objective_surrogate(vec)
        push_candidate(vec, val)
    print(f"Seeded {len(analytic_seeds)} analytic candidates. Best so far ≈ { -heap[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parameter sweeps for exponential, power, Gaussian families
    # ------------------------------------------------------------------
    print("\n--- Parameter sweep (exponential) ---")
    for t in np.logspace(-5, -2, num=30):   # t = 1 - rate
        rate = 1.0 - t
        vec = _exponential_decay(n, rate)
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    print("\n--- Parameter sweep (power) ---")
    for p in np.linspace(0.2, 5.0, num=30):
        vec = _power(n, p)
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    print("\n--- Parameter sweep (gaussian) ---")
    for sigma_frac in np.linspace(0.05, 0.5, num=30):
        vec = _gaussian(n, sigma_frac)
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    print("\n--- Parameter sweep (beta) ---")
    for _ in range(30):
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        vec = _beta_shape(n, alpha, beta)
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws (many)
    # ------------------------------------------------------------------
    print("\n--- Random Dirichlet draws ---")
    dirichlet_draws = 2000
    for i in range(dirichlet_draws):
        if time.time() > deadline * 0.6:   # keep a margin for local optimisation
            break
        conc = rng.uniform(0.02, 2.0)
        vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    # ------------------------------------------------------------------
    # 3️⃣  Low‑frequency random curves (fast) and mixtures
    # ------------------------------------------------------------------
    print("\n--- Low‑frequency random curves ---")
    lowfreq_draws = 500
    for i in range(lowfreq_draws):
        if time.time() > deadline * 0.6:
            break
        nf = rng.integers(2, 6)
        vec = _lowfreq_curve(n, nf, rng)
        # random mixture with uniform for extra variety
        if rng.random() < 0.3:
            vec = _mix_shapes(_uniform(n), vec, rng.random())
        val = _objective_surrogate(vec)
        push_candidate(vec, val)

    # ------------------------------------------------------------------
    # 4️⃣  Extract the best few seeds
    # ------------------------------------------------------------------
    # heap stores (-val, vec), the most negative (largest -val) is worst.
    # The best candidate corresponds to the largest -val.
    best_entries = sorted(heap, key=lambda x: x[0], reverse=True)  # descending -val -> ascending val
    top_seeds = [entry[1] for entry in best_entries[:5]]
    best_val = -best_entries[0][0]
    best_vec = best_entries[0][1]
    print(f"\nBest surrogate after seeding ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Sub‑gradient descent (long) from the best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (long) ---")
        # allocate roughly half the remaining budget to sub‑grad
        remain = deadline - time.time()
        max_iters = int(min(300_000, remain * 250))  # heuristic: 250 iters per second
        best_vec, best_val = _subgrad_optimize(best_vec, n, deadline, max_iters=max_iters, eta0=0.8)
        print(f"[SubGrad] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remain = deadline - time.time()
        max_iters = int(min(800_000, remain * 1500))
        best_vec, best_val = _expgrad_optimize(best_vec, n, deadline, max_iters=max_iters, eta0=0.25)
        print(f"[ExpGrad] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final simulated‑annealing tweak (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final tweak ---")
        remain = deadline - time.time()
        max_iters = int(min(600_000, remain * 1800))
        best_vec, best_val = _simulated_annealing(best_vec, n, deadline, max_iters=max_iters)
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣  Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()

"""
Hybrid optimizer for the non‑negative step‑function minimisation problem.

Strategy
--------
0️⃣  Generate a very diverse pool of ℓ₁‑normalised candidates:
    - deterministic analytic families (uniform, triangular, Gaussian, …)
    - fine‑grained parameter sweeps for exponential, power, Gaussian, beta
    - thousands of random Dirichlet draws
    - low‑frequency random curves and random mixtures thereof
    - mixtures of two random base shapes (exponential/power/gaussian/β)
    Each candidate is scored with the exact surrogate objective
    (FFT‑based max‑convolution) and only the best ~30 are kept.

1️⃣  From the kept candidates run a *long* additive sub‑gradient descent
    on the exact objective (projected onto the probability simplex).

2️⃣  Polish the result with the multiplicative exponential‑gradient (mirror
    descent) method, which works well on the max‑convolution surrogate.

3️⃣  Finish with a short simulated‑annealing tweak.

All stages respect the global time budget (default 1000 s).  The final
sequence is stored in ``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact surrogate objective for an ℓ₁‑normalised vector (∑p = 1):
        f(p) = 2·n·max(conv(p,p)).
    This matches ``evaluate_sequence`` after the internal clamping.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _subgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 150_000,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a)),
    with Euclidean projection onto the simplex.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 25_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 250_000,
                      eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)

        delta = -eta * g
        delta = np.clip(delta, -50, 50)          # avoid overflow/underflow
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Tiny random transfers + subset reshuffles, accepting worsening moves."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

def _random_base_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random analytic family and return a normalised instance."""
    typ = rng.integers(0, 5)
    if typ == 0:                     # exponential
        rate = 1.0 - rng.uniform(5e-5, 5e-3)   # rate ∈ [0.99995, 0.995]
        vec = _exponential_decay(n, rate)
    elif typ == 1:                    # power law
        p = rng.uniform(0.05, 5.0)
        vec = _power(n, p)
    elif typ == 2:                    # Gaussian
        sigma = rng.uniform(0.05, 0.5)
        vec = _gaussian(n, sigma)
    elif typ == 3:                    # Beta
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        vec = _beta_shape(n, a, b)
    else:                             # low‑frequency cosine mixture
        nf = rng.integers(2, 6)
        vec = _lowfreq_curve(n, nf, rng)
    # optional mixture with uniform for extra diversity
    if rng.random() < 0.3:
        mix = rng.random()
        vec = _mix_shapes(_uniform(n), vec, mix)
    return vec


def _random_mixed_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mix two independently drawn base shapes."""
    a = _random_base_shape(n, rng)
    b = _random_base_shape(n, rng)
    alpha = rng.random()
    return _mix_shapes(a, b, alpha)

        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        vec = _beta_shape(n, a, b)
        val = _objective_surrogate(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))

    # ------------------------------------------------------------------
    # 2️⃣  Large random pool (Dirichlet draws, low‑freq curves, mixed shapes)
    # ------------------------------------------------------------------
    print("\n--- Random Dirichlet draws ---")
    dirichlet_draws = 4000
    for i in range(dirichlet_draws):
        if time.time() > deadline * 0.6:   # keep margin for refinement
            break
        conc = rng.uniform(0.02, 2.0)
        vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        val = _objective_surrogate(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))

    print("\n--- Low‑frequency random curves ---")
    lf_draws = 1200
    for i in range(lf_draws):
        if time.time() > deadline * 0.6:
            break
        nf = rng.integers(2, 6)
        vec = _lowfreq_curve(n, nf, rng)
        if rng.random() < 0.25:
            vec = _mix_shapes(_uniform(n), vec, rng.random())
        val = _objective_surrogate(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))

    print("\n--- Mixed‑shape random draws ---")
    mixed_draws = 1200
    for i in range(mixed_draws):
        if time.time() > deadline * 0.6:
            break
        vec = _random_mixed_shape(n, rng)
        val = _objective_surrogate(vec)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, vec))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, vec))

    # ------------------------------------------------------------------
    # 3️⃣  Extract a few promising seeds
    # ------------------------------------------------------------------
    best_entries = sorted(heap, key=lambda x: x[0], reverse=True)   # highest –val → lowest value
    top_seeds = [entry[1] for entry in best_entries[:5]]
    best_vec = best_entries[0][1]
    best_val = -best_entries[0][0]
    print(f"\nBest surrogate after seeding: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Sub‑gradient descent from the top seeds (allocate ≈½ time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient descent (long) ---")
        remaining = deadline - time.time()
        # roughly 200 000 iterations per remaining 80 s – adapt to actual time left
        max_iters = int(min(300_000, remaining * 250))
        # start from the very best seed, but also try a couple of the others
        candidates = [best_vec] + top_seeds[1:3]
        for idx, seed_vec in enumerate(candidates):
            if time.time() > deadline:
                break
            print(f"   Sub‑grad run #{idx+1}")
            cand, val = _subgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_iters,
                                          eta0=0.8)
            if val < best_val:
                best_val, best_vec = val, cand
                print(f"   New best after sub‑grad #{idx+1}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exponential‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_vec, best_val = _expgrad_optimize(best_vec, n, deadline,
                                               max_iters=max_iters,
                                               eta0=0.25)
        print(f"ExpGrad best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final simulated‑annealing tweak (optional, if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final tweak ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_vec, best_val = _simulated_annealing(best_vec, n, deadline,
                                                  max_iters=max_iters)
        print(f"SA final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
