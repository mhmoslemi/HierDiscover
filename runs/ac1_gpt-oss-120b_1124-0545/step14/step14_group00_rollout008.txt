            best_seq = a.copy()
        # optionally print progress every some iterations
        if it % 100_000 == 0:
            print(f"[Adam] it={it} best={best_val:.6f}")

"""Step‑function optimiser.

Searches for a non‑negative sequence a_0…a_{n‑1} (ℓ1‑normalised) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv is the full linear convolution.  The algorithm

* builds a rich set of analytic seeds (beta, edge‑biased, cosine‑inverted,
  double‑Gaussian, etc.), including a very fine beta‑grid down to α,β=0.05;
* evaluates the cheap surrogate 2·n·max(conv) (identical to the true
  objective for ℓ1‑normalised vectors);
* keeps the best few seeds and refines them with a continuation L‑BFGS
  optimiser (log‑sum‑exp smoothing, β up to 20000);
* performs long‑run Adam (mirrored‑gradient) optimisation on the exact
  max‑convolution;
* finishes with a simulated‑annealing polish.

All time‑checks respect the 1000 s wall‑clock budget.  The best found
sequence is stored in the global ``height_sequence_1`` and returned as a
plain list of floats.
"""
import time
import math
import numpy as np

# optional SciPy import for L‑BFGS‑B
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact / surrogate objective (ℓ1‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  – same as the true objective when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_triangular(n: int) -> np.ndarray:
    """V‑shape: 0 at the centre, 1 at the ends (linear)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.abs(i - c) / c
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_cosine(n: int) -> np.ndarray:
    """High at the ends, zero in the middle: 1+cos(2πi/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_power_shape(n: int, p: float) -> np.ndarray:
    """Heavy edges: weight ∝ ((i+1)*(n‑i))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    # avoid zero‑division – product is never zero for i∈[0,n‑1]
    a = ((i + 1) * (n - i)) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]

    # Initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing of the max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp                     # weights, sum=1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam (mirrored‑gradient) optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 400_000,
              alpha: float = 0.08,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Adam optimiser in log‑softmax space (exact max‑convolution)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam updates on the logits
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 100_000 == 0:
            print(f"[Adam] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _simulated_annealing(start_seq: np.ndarray,

                         deadline: float,

    """Simple SA in logit space."""

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.06 * (1.0 - progress) + 1e-7
        sigma = 0.20 * (1.0 - progress) + 1e-5

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting optimisation, n={n} ===")

    # ------------------------------------------------------------------
    # 0. Previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            p = np.asarray(previous, dtype=np.float64)
            if p.size == n and np.sum(p) > 0:
                prev_arr = p / np.sum(p)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Populate a diverse seed pool
    # ------------------------------------------------------------------
    seed_pool = []

    # deterministic analytic families
    seed_pool.append((_uniform(n), ('uniform',)))
    seed_pool.append((_inverse_triangular(n), ('inv_tri',)))
    seed_pool.append((_inverse_cosine(n), ('inv_cos',)))

    if prev_arr is not None:
        seed_pool.append((prev_arr, ('previous',)))

    # fine beta‑grid – include very U‑shaped parameters
    alphas = np.concatenate([
        np.linspace(0.05, 0.3, 7),
        np.linspace(0.3, 1.5, 7),
        np.linspace(1.5, 5.0, 5)
    ])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            seed_pool.append((vec, ('beta', a, b)))

    # edge‑power shapes (vary p)
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        seed_pool.append((_edge_power_shape(n, p), ('edge_pow', p)))

    # double Gaussian with edge‑centred means
    for sigma_frac in [0.10, 0.15, 0.20]:
        for shift_frac in [0.05, 0.10, 0.15, 0.20]:
            seed_pool.append((_double_gaussian(n, sigma_frac, shift_frac),
                              ('dbl_gauss', sigma_frac, shift_frac)))

    # random Dirichlet draws (few)
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            seed_pool.append((vec, ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Evaluate and keep the top‑k seeds
    # ------------------------------------------------------------------
    scored = []
    for vec, meta in seed_pool:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        scored.append((vec, val, meta))

    scored.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in scored[:top_k]]
    print(f"Seed pool: {len(scored)} candidates, best surrogate = {scored[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS on the elite seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{len(top_seeds)} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000,2000,
                                                        5000,10000,20000],
                                         max_iter_per_beta=2000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long‑run Adam (exact max) optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Guess ~0.00002 s per iteration → give a generous budget
        max_iters = int(min(600_000, remaining * 30_000))
        max_iters = max(200_000, max_iters)  # at least a decent run
        print(f"\n--- Adam optimisation (≤{max_iters} iters) ---")
        seq_adam, val_adam = _adam_opt(best_seq, n, deadline, max_iters=max_iters)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"[Update] after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(500_000, remaining * 25_000))
        max_iters = max(100_000, max_iters)
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, max_iters=max_iters)
        print(f"[SA] best surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq = seq_sa.copy()
            best_val = val_sa
            print(f"[Update] after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    print(f"\n=== Finished ===")
    print(f"Surrogate ≈ {best_val:.6f}")
    if true_val is not None:
        print(f"true evaluate_sequence = {true_val:.6f}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
