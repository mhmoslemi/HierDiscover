"""
Propose a step function (non‑negative heights) that minimizes
the official ``evaluate_sequence`` objective.

Approach
--------
1. Build a very diverse pool of candidate probability vectors (the
   sequence is later scaled to have sum 1 – the evaluation is
   scale‑invariant).  In addition to the analytic families used before
   we add exponential and power‑law families, many beta‑shape
   parametrisations and a broad sweep of Dirichlet concentrations.

2. Pick the best few seeds w.r.t. the exact objective
   (2·n·max(conv) / sum²).

3. Refine each of those seeds with a soft‑max Adam optimiser.
   The surrogate loss is the log‑sum‑exp of the convolution,
   which smoothly approximates the max.  The temperature α is
   annealed from 10 to 20000.

4. Fine‑tune the best result with a longer simulated‑annealing run
   that repeatedly reshuffles a random subset of the mass and also
   performs tiny transfers between two positions.  The schedule is
   much slower than before, allowing the algorithm to explore more.

5. Return the best sequence found.  The resulting list is stored in the
   global name ``height_sequence_1`` for possible reuse in the next
   optimisation round.
"""

import time, math, numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    n = len(x)
    L = 2 * n - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(seq: np.ndarray) -> float:
    """Exact evaluation of the official objective (without clamping)."""
    n = len(seq)
    conv = _conv_full(seq, seq)
    max_b = np.max(conv)
    s = np.sum(seq)
    # protective guard – the official function would return inf for
    # s < 0.01; we simply avoid that case.
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (lam > 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _powerlaw_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p > 0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_softmax_optimize(start: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           alpha_start: float = 10.0,
                           alpha_end: float = 20000.0,
                           lr: float = 0.12) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate:
       L(p) = (1/α)·log Σₖ exp(α·(p*p)[k])  """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_exact(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(3 * n - 2)          # length needed for convolutions

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Full self‑convolution via FFT
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # Temperature for the smooth max
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        # log‑sum‑exp (stable)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # Gradient of the surrogate: 2·corr(x, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # Projection back onto the probability simplex
        x = _proj_simplex(x)

        cur_val = _objective_exact(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            # occasional progress report
            if t % 25000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _simulated_annealing(start: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """Fine‑tuning with a slow‐cooling simulated annealing."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_exact(cur)
    best = cur.copy()
    best_val = cur_val

    # Initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.99996, min_temp)

        # ---- move proposal -------------------------------------------------
        if rng.random() < 0.6:
            # reshuffle a random subset (size 6‑20)
            k_sub = rng.integers(6, 21)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total == 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps_transfer = rng.random() * min(cur[i], 0.04)
            if eps_transfer == 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps_transfer
            cand[j] += eps_transfer

        new_val = _objective_exact(cand)
        delta = new_val - cur_val

        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 200_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation – a very broad library of analytic families
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.15, 0.25, 0.35, 0.45):
        seeds.append(_gaussian(n, sigma))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.3, 0.5, 0.7):
        seeds.append(_tukey_window(n, alpha))

    # exponential and power‑law families
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        seeds.append(_exponential_decay(n, lam))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        seeds.append(_powerlaw_decay(n, p))

    # random Dirichlet draws – both sparse and dense
    for conc in (0.01, 0.03, 0.07, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse supports of various sizes
    for m in (80, 120, 160, 200):
        for _ in range(2):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency random cosine series (non‑negative)
    for nf in (2, 3, 4):
        for _ in range(3):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                seeds.append(a / s)

    # mixtures of uniform with a sparse support
    for mix_alpha in (0.25, 0.4, 0.6):
        for m in (150, 300):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previous best (if provided)
    if previous is not None and previous.shape == (n,) and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final clean‑up – ensure proper normalisation and positivity
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            v = _uniform(n)
        clean.append(v / s)
    return clean

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search – within ``budget_s`` seconds – for a non‑negative step function
    that minimises the official ``evaluate_sequence`` objective.
    The algorithm builds a rich set of seeds, runs a soft‑max Adam
    optimiser on the best few, and finishes with a long simulated‑annealing
    refinement.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # small safety margin

    # ------------------------------------------------------------------
    # Length of the step function – a compromise between flexibility
    # and speed.  4000 works well for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n--- Length n={n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – generate a very diverse seed library
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # ------------------------------------------------------------------
    # Evaluate all seeds with the exact objective and keep the top few
    # ------------------------------------------------------------------
    scored = [(s, _objective_exact(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_exact(best_seq)
    print(f"\n[Init] best exact = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        # allocate roughly half of the remaining budget to Adam
        adam_deadline = time.time() + (deadline - time.time()) * 0.55
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= adam_deadline:
                break
            cand, val = _adam_softmax_optimize(
                seed_vec, n, adam_deadline,
                max_iters=300_000,
                alpha_start=10.0,
                alpha_end=20000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – simulated‑annealing polishing (the bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        sa_deadline = deadline
        best_seq, best_val = _simulated_annealing(
            best_seq, n, sa_deadline,
            max_iters=1_200_000,
            init_temp_factor=0.08,
            min_temp=1e-14,
        )
        print(f"[SA] final exact = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nExact ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
