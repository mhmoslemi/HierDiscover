def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Improved search for a non‑negative step‑function that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm expands the seed pool (dense Beta‑grid, many Dirichlet draws,
low‑frequency curves, deterministic families), ranks seeds by a cheap surrogate,
optimises the elite seeds with a continuation L‑BFGS on a smooth log‑sum‑exp
approximation of “max”, and finally refines the best candidate with a fast
Adam‑style exponentiated‑gradient on the exact nonsmooth objective.
Optional up‑sampling is performed if time permits.

All vectors are ℓ₁‑normalised, never contain negatives and never exceed 1000,
so the returned list can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as _opt
from itertools import product

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax yielding a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two ℓ₁‑vectors."""
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation – deterministic families + large random sweep
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))

    # ---- exhaustive Beta grid (log‑space) ---------------------------------
    alphas = np.logspace(-3, 1, 13)   # 0.001 … 10  (13 points)
    betas = np.logspace(-3, 1, 13)
    for a, b in product(alphas, betas):
        seeds.append(_beta_shape(n, a, b))

    # ---- random parametric families ---------------------------------------
    for _ in range(max_random // 5):
        # random Beta (skewed)
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # random power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # random exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # random linear‑power (decreasing)
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture uniform + Beta
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # ---- random Dirichlet draws -------------------------------------------
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ---- low‑frequency random curves ---------------------------------------
    for nf in (2, 3, 4, 5, 6):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # ---- previous solution -------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee non‑negative and ℓ₁‑normalised
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500) -> np.ndarray:
    """Optimise a smooth surrogate of max‑convolution using L‑BFGS‑B."""
    if beta_schedule is None:
        beta_schedule = [0.1, 1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0, 10000.0, 20000.0]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                      # shifting trick for numerical
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (exact derivative of smoothed max)
            w = exp_beta / sum_exp                     # weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] time limit reached (beta={beta_val})")
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} (beta={beta_val})")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   alpha: float = 0.02) -> np.ndarray:
    """Exponentiated‑gradient / Adam on the *exact* max‑convolution objective."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v_hat = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (softmax‑reparametrisation)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_corr = v_hat / (1.0 - beta2 ** t)
        log_a = log_a - alpha * m_hat / (np.sqrt(v_corr) + eps)

        # bring back to the simplex
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 100_000 == 0:
                print(f"[Adam] it={t:,} best≈{best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – optional fine‑tuning on larger grid
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()
