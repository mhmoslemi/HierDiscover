"""
Sidon‑set based attack.

The evaluation objective is
    2·n·max(conv(a,a)) / (sum(a)²)

with non‑negative a and scale invariance (the sum can be fixed to 1).
For a vector whose non‑zero indices form a Sidon set (all pairwise sums are
distinct) the convolution at any lag contains at most one product term.
Hence

    max(conv) = max( a_i² , 2·a_i·a_j ) = 2·max_i a_i² .

With a uniform allocation a_i = 1/m on m active positions this gives

    max(conv) = 2 / m²  →  objective = 4·n / m².

If m is as large as possible (the classic Sidon bound
    m(m+1)/2 ≤ 2·n‑1  →  m ≈ 2·√n ),
the objective approaches 1.  For n≈4000 the theoretical optimum is
≈1.02, well below the current 1.5085.

The algorithm repeatedly builds random Sidon sets by a greedy insertion
(using a Boolean array to forbid duplicate sums), keeps the largest set
found, builds a uniform‑weight step function on that set and records the
best objective value obtained from the official ``evaluate_sequence``.
The search runs until the time budget (≈ 1000 s) expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Sidon set construction (greedy, random order)
# ----------------------------------------------------------------------
def _greedy_sidon_set(n: int, order: np.ndarray) -> list[int]:
    """
    Build a Sidon set on {0,…,n‑1} by scanning indices in ``order``.
    ``order`` must be a permutation of ``np.arange(n)``.
    Returns a list of selected indices.
    """
    # Boolean mask for sums 0 … 2·(n‑1)
    used = np.zeros(2 * n, dtype=bool)
    pos = []
    for idx in order:
        # check the self‑sum first
        if used[idx + idx]:
            continue
        # check sum with already selected positions
        conflict = False
        for p in pos:
            if used[idx + p]:
                conflict = True
                break
        if conflict:
            continue
        # accept idx: mark all new sums
        for p in pos:
            used[idx + p] = True
            # used[p + idx] is the same entry, no need to set twice
        used[idx + idx] = True
        pos.append(int(idx))
    return pos


# ----------------------------------------------------------------------
# Helper to build the probability vector from a Sidon set
# ----------------------------------------------------------------------
def _build_sequence(n: int, positions: list[int]) -> np.ndarray:
    """
    Return a length‑n vector with uniform weight on the given ``positions``.
    The weights sum to 1.
    """
    m = len(positions)
    if m == 0:
        # fallback – uniform (should never happen)
        return np.full(n, 1.0 / n, dtype=np.float64)

    w = 1.0 / m
    seq = np.zeros(n, dtype=np.float64)
    seq[positions] = w
    return seq


# ----------------------------------------------------------------------
# Main entry point used by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function using Sidon‑set constructions.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # problem size – larger n gives a better theoretical lower bound,
    # but also a larger search space.  n=8000 is a good trade‑off.
    n = 8000
    print(f"\n=== Sidon‑set search start (n={n}) ===")

    # theoretical maximum cardinality of a Sidon set in {0,…,n‑1}
    max_m_theory = int((np.sqrt(1.0 + 8.0 * (2 * n - 1)) - 1) // 2)
    print(f"Theoretical maximum number of points: {max_m_theory}")

    best_seq = None
    best_val = float("inf")
    best_m = 0
    attempts = 0

    # first deterministic run (ascending order) – cheap baseline
    order0 = np.arange(n, dtype=np.int64)
    pos0 = _greedy_sidon_set(n, order0)
    seq0 = _build_sequence(n, pos0)
    val0 = evaluate_sequence(seq0.tolist())
    best_seq, best_val, best_m = seq0, val0, len(pos0)
    print(f"[Init] det. order: m={best_m}, value={best_val:.6f}")

    # main random‑greedy loop
    while time.time() < deadline:
        attempts += 1

        # random permutation of the index set
        order = rng.permutation(n)
        pos = _greedy_sidon_set(n, order)
        m = len(pos)

        # we are interested only in sets that can beat the current best.
        # The objective for a uniform Sidon set is 4·n / m².
        # If that lower bound already exceeds best_val, skip the expensive eval.
        theoretical_val = 4.0 * n / (m * m) if m > 0 else float("inf")
        if theoretical_val >= best_val:
            continue

        seq = _build_sequence(n, pos)
        val = evaluate_sequence(seq.tolist())

        if val < best_val:
            best_seq, best_val, best_m = seq, val, m
            print(f"[Update] attempt #{attempts}: m={m}, value={val:.6f}")

            # early exit if we hit the theoretical optimum for this n
            if best_m >= max_m_theory:
                print("[Info] reached theoretical maximum cardinality – stopping.")
                break

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[Fallback] using uniform distribution.")

    print(f"\n=== Finished (attempts={attempts}) ===")
    print(f"Best |S| = {best_m}, evaluate_sequence = {best_val:.6f}")

    # store globally for possible warm start in future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
