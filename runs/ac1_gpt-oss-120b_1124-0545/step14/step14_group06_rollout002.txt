"""
Optimise a non‑negative step function for the objective

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm works on L1‑normalised vectors (sum = 1) so the objective
reduces to 2·n·max(conv).  It proceeds in four stages:

1.   A rich but bounded set of analytic “window’’ shapes (uniform,
     triangular, exponential, power, Gaussian, logistic, Tukey, beta)
     and a few random Dirichlet draws are generated for a relatively large
     length n = 8000.  The 10 best seeds are kept.

2.   A short “exponential‑gradient’’ (mirror‑descent on the simplex)
     refinement is run on the best seeds.

3.   A smooth Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution
     further lowers the bound.

4.   A simulated‑annealing polishing step explores larger “jumps’’,
     followed by a long exponential‑gradient polish and finally a tiny
     hill‑climbing pass that performs random mass transfers and accepts
     any improvement.

All stages stop as soon as the global deadline (budget_s – 2 s safety margin)
is reached.  The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate used during the search (exact when Σ seq = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.999) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 10.0, shift: int | None = None) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    if shift is None:
        shift = n // 2
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - shift) / scale))
    a = a - a.min() + 1e-15
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families (a compact but still varied set)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.20, 0.25):
        seeds.append(_gaussian(n, sigma))
    for rate in (0.998, 0.999, 0.9995):
        seeds.append(_exponential_decay(n, rate))
    for p in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))
    for alpha, beta in ((0.8, 1.5), (1.0, 2.0), (1.2, 3.0)):
        seeds.append(_beta_shape(n, alpha, beta))
    for scale in (5.0, 10.0, 20.0):
        seeds.append(_logistic_window(n, scale))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # a handful of random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few exponential draws
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # include previous solution if available
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanity – ensure all seeds are ℓ₁‑normalised
    clean: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _expgrad_optimize(start: np.ndarray, n: int, deadline: float,
                      max_iters: int = 250_000, eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL‑mirror) descent on the simplex.
    The sub‑gradient uses the active shift of the current max‑convolution.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # step size – √t schedule (slightly slower decay)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        # Euclidean projection onto the simplex
        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
    return best_x, best_val


def _adam_smooth_optimize(start: np.ndarray, n: int, deadline: float,
                          init_lr: float = 0.04,
                          beta1: float = 0.9, beta2: float = 0.999,
                          eps: float = 1e-8,
                          p0: float = 1.0, p_factor: float = 1.5,
                          p_step: int = 3000,
                          max_total_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate
        F_p(a) = (1/p)·log Σ_k exp(p·conv_k(a)).
    The smoothing parameter p grows gradually.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    p = p0

    while time.time() < deadline and t < max_total_iters:
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        shifted = conv - max_conv               # ≤ 0
        exp_arg = p * shifted
        exp_arg = np.clip(exp_arg, -700, 0)     # safety
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the soft‑max surrogate
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _uniform(n)

        # increase smoothing
        if t % p_step == 0:
            p = min(p * p_factor, 400.0)

        # occasional fallback to a very cheap EG step when p is large
        if p > 300.0 and t % (p_step * 4) == 0:
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] = 2.0 * a[rev]
            eta = 0.02
            delta = -eta * g
            delta = np.clip(delta, -50, 50)
            a = a * np.exp(delta)
            a = a / a.sum()

    final_val = _objective_surrogate(best_a)
    return best_a, final_val


def _simulated_annealing(start: np.ndarray, n: int, deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Classic SA with two move families (block reshuffle & tiny transfer)."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99993, min_temp)   # slower cooling than before

        if rng.random() < 0.55:
            # reshuffle a random subset (size 5‑25)
            k = rng.integers(5, 26)
            idx = rng.choice(n, size=k, replace=False)
            tot = cur[idx].sum()
            if tot <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * tot
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            # transfer a small fraction (up to 5 % of source)
            eps = rng.random() * min(cur[i], 0.05 * cur[i])
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                # occasional progress report
                if it % 200_000 == 0:
                    print(f"[SA] it={it:8d} best={best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin so that we can still call evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # Length – a larger n usually gives a better bound
    n = 8000
    print(f"\n=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds.")

    # evaluate surrogate for all seeds (fast FFT based) and keep the best few
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, v) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick EG refinements on the very best seeds
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Quick EG refinements (few seeds) ---")
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            max_it = int(min(250_000, remaining * 1800))
            cand, val = _expgrad_optimize(seed_vec, n, deadline,
                                          max_iters=max_it, eta0=0.33)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Quick EG] seed #{idx+1} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smooth optimisation ---")
        remaining = deadline - time.time()
        max_adam_iters = int(min(1_800_000, remaining * 1600))
        best_seq, best_val = _adam_smooth_optimize(best_seq, n, deadline,
                                                   init_lr=0.045,
                                                   p0=1.0, p_factor=1.6,
                                                   p_step=2500,
                                                   max_total_iters=max_adam_iters)
        print(f"[Adam] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 1900))
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  max_iters=max_iters)
        print(f"[SA] best surrogate after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final long EG polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(3_000_000, remaining * 2100))
        if max_iters > 0:
            best_seq, best_val = _expgrad_optimize(best_seq, n, deadline,
                                                   max_iters=max_iters,
                                                   eta0=0.22)
            print(f"[Final EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – tiny stochastic hill‑climbing (mass‑transfer) pass
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Tiny hill‑climbing pass ---")
        cur = best_seq.copy()
        cur_val = _objective_surrogate(cur)
        while time.time() < deadline:
            i, j = rng.integers(n, size=2, replace=False)
            eps = cur[i] * rng.random() * 0.02  # move up to 2 % of source
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps
            new_val = _objective_surrogate(cand)
            if new_val < cur_val:
                cur = cand
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, cand.copy()
        print(f"[Hill‑Climb] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
