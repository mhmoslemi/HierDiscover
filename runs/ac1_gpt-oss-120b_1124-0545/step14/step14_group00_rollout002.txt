"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

within a 1000 s wall‑clock budget.

The search proceeds in three phases:

1. **Rich seeding** – many analytic families (uniform, beta, power‑law,
   Gaussian, Tukey), random Dirichlet draws (including an alternating
   ordering), mixtures of two beta shapes and low‑frequency cosine
   sums.  All candidates are ℓ₁‑normalised and the best few are kept.

2. **Continuation optimisation** – a smoothed surrogate of the
   ℓ∞‑objective (log‑sum‑exp of the convolution) is minimised with
   L‑BFGS‑B followed by a custom Adam optimiser that slowly drives the
   smoothing parameter β up to 2·10⁴.  The result stays on the probability
   simplex via a soft‑max parametrisation.

3. **Fine polishing** – exact sub‑gradient descent (projected) and a
   final simulated‑annealing walk (in logit space) are applied to the
   best vector found so far.

All heavy work respects the global deadline and the best sequence is
saved in ``height_sequence_1`` and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv)/sum²."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate for a ℓ₁‑normalised vector – equals 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    a = (a0

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # modified Bessel (numpy)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Create a Dirichlet draw and assign largest/smallest alternatingly."""
    raw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_two_betas(n: int, a1: float, b1: float,
                  a2: float, b2: float, w: float) -> np.ndarray:
    """Convex mixture of two beta‑shape vectors."""
    v1 = _beta_shape(n, a1, b1)
    v2 = _beta_shape(n, a2, b2)
    mix = (1.0 - w) * v1 + w * v2
    s = mix.sum()
    return mix / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS continuation optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # log‑sum‑exp smoothing
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (derived from the soft‑max representation)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 150_000,
                          max_beta: float = 20000.0) -> np.ndarray:
    """
    Adam optimisation of the smoothed max‑convolution (log‑sum‑exp)
    surrogate.  The smoothing parameter β is increased linearly from 1
    to ``max_beta``.
    Returns the best ℓ₁‑normalised vector seen.
    """
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.14

    v = np.log(start_vec + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    start_time = time.time()
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear β schedule
        beta = 1.0 + (max_beta - 1.0) * (it / max_iters)

        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # smoothed objective components
        shift = max_c
        exp_beta = np.exp(beta * (conv - shift))        # ≤ 1
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # occasional exact evaluation (every 2000 steps)
        if it % 2000 == 0:
            seq = _softmax(v)
            cur_val = _objective_surrogate(seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = seq.copy()
                print(f"[Adam] it={it:6d} new best surrogate={best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing (logit + softmax) for final polishing
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple SA in logit space.  Uses a linear cooling schedule and a
    decreasing proposal width.
    """
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-7          # temperature
        sigma = 0.18 * (1.0 - progress) + 1e-5       # proposal width

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generation helpers
# ----------------------------------------------------------------------
def _generate_candidates(n: int, rng: np.random.Generator, previous: np.ndarray = None):
    """Return a list of (vector, surrogate_score, meta) tuples."""
    cand = []

    # Uniform baseline
    cand.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Previous best, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        vec = previous / previous.sum()
        cand.append((vec, _objective_surrogate(vec), ('previous',)))

    # Dirichlet (plain and alternating)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            v = _random_dirichlet(n, conc, rng)
            cand.append((v, _objective_surrogate(v), ('dirichlet', conc)))
            alt = _alternating_dirichlet(n, conc, rng)
            cand.append((alt, _objective_surrogate(alt), ('alternating', conc)))

    # Beta shapes (random grid)
    for _ in range(30):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        v = _beta_shape(n, a, b)
        cand.append((v, _objective_surrogate(v), ('beta', a, b)))

    # Mixtures of two β‑shapes
    for _ in range(20):
        a1 = rng.uniform(0.2, 5.0)
        b1 = rng.uniform(0.2, 5.0)
        a2 = rng.uniform(0.2, 5.0)
        b2 = rng.uniform(0.2, 5.0)
        w = rng.uniform(0.0, 1.0)
        v = _mix_two_betas(n, a1, b1, a2, b2, w)
        cand.append((v, _objective_surrogate(v), ('mixbeta',)))

    # Low‑frequency cosine curves
    for _ in range(12):
        num_freq = rng.integers(1, 5)
        v = _lowfreq_curve(n, num_freq, rng)
        cand.append((v, _objective_surrogate(v), ('lowfreq', num_freq)))

    # Gaussian family
    for sigma in np.linspace(0.06, 0.45, 9):
        v = _gaussian(n, sigma)
        cand.append((v, _objective_surrogate(v), ('gauss', sigma)))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        v = _power(n, p)
        cand.append((v, _objective_surrogate(v), ('power', p)))

    return cand

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size (feel free to adjust; 4000 works well for the budget)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    print("\n--- Generating candidate seeds ---")
    candidates = _generate_candidates(n, rng, previous)

    # ------------------------------------------------------------------
    # 2. Keep top‑k seeds (by surrogate)
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [vec for vec, _, _ in candidates[:top_k]]
    print(f"Collected {len(candidates)} seeds, retaining top {top_k} for optimisation.")

    # ------------------------------------------------------------------
    # 3. Continuation optimisation: L‑BFGS + Adam
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all seeds.")
            break

        remaining = deadline - time.time()
        # Allocate a slice of the remaining budget to this seed
        # (roughly equally among the remaining seeds)
        est_per_seed = remaining / max(1, (len(top_seeds) - idx))
        # L‑BFGS stage
        max_iter_beta = int(min(4000, est_per_seed * 0.04))
        max_iter_beta = max(1000, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx+1} (max_iter/beta={max_iter_beta}) ---")
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                           max_iter_per_beta=max_iter_beta)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")

        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"[Update] new global best after L‑BFGS = {global_best_val:.6f}")

        # Adam refinement – allocate remaining budget for this seed
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        max_iters_adam = int(min(140_000, remaining * 0.05))
        max_iters_adam = max(30_000, max_iters_adam)

        print(f"[Adam] seed #{idx+1}, max_iters={max_iters_adam}")
        seq_adam = _adam_smooth_optimize(global_best_seq, n, deadline,
                                         max_iters=max_iters_adam)
        val_adam = _objective_surrogate(seq_adam)
        print(f"[Adam] surrogate = {val_adam:.6f}")

        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] new global best after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(700_000, remaining * 0.07))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline,
                                                max_iters=max_sub)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(600_000, remaining * 0.06))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline,
                                                  max_iters=max_sa)
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[SA] final best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
