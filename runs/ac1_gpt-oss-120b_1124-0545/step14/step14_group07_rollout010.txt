def subgrad_descent(a, n, deadline, max_iters, eta0):
    best = a.copy()
    best_val = _objective_surrogate(a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        conv = _conv_fft(a, a)
        k = int(conv.argmax())
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev = k - idx
        g[idx] = 2.0 * a[rev]
        eta = eta0 / math.sqrt(t)
        a = a - eta * g
        # project onto simplex
        a = _proj_simplex(a)   # Euclidean projection onto simplex
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best = a.copy()
    return best, best_val

def _smooth_objective(y, n, p):
    a = _softmax(y)
    conv = _conv_fft(a, a)
    max_c = conv.max()
    shifted = conv - max_c
    exp_arg = p*shifted
    exp_arg = np.clip(exp_arg, -700, 0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_c).astype(float)
        w_sum = w.sum()
    w /= w_sum
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)
    grad_a = 2.0 * grad_full[n-1:n-1+n]   # length n (center part)
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y
```

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Approach
--------
1. **Seed generation** – a large variety of Ⓛ₁‑normalised analytic windows
   (uniform, triangular, sine, Gaussian, power, beta, Tukey, exponential)
   together with random Dirichlet draws provide a diverse starting pool.
2. **Smooth max optimisation** – the true objective is
   ``2·n·max(conv(a,a))`` (the ℓ∞‑norm of the self‑convolution).
   We minimise a *smooth* surrogate
   ``F_p(a) = (1/p)·log Σₖ exp(p·convₖ(a,a))``,
   whose gradient is cheap to compute via FFTs.
   A schedule of increasing smoothing parameters ``p`` (5 → … → 1280)
   is used; at each stage an L‑BFGS‑B optimisation (logits → softmax)
   is performed with a generous iteration budget.
   After every stage a short random restart is tried.
3. **Sub‑gradient polishing** – after the smooth optimisation a
   classic projected sub‑gradient descent on the *exact* max‑convolution
   (multiplicative / additive updates on the simplex) is run for the
   remaining time.  This often pushes the max a little lower.
4. **Final evaluation** – the best sequence found (as a plain Python list)
   is stored globally as ``height_sequence_1`` and returned.

The whole pipeline respects the 1000 s deadline.  The code is fully
self‑contained, uses only NumPy / SciPy, works on a single thread
(but may safely use up to two CPUs) and prints progress information.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact surrogate (the objective we ultimately want to minimise)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when Σseq = 1 : 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components, clip to non‑negative and ℓ₁‑normalise."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (large, diverse pool)
# ----------------------------------------------------------------------
def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma))
    for sigma in (0.15, 0.2, 0.3, 0.4, 0.6):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform (adds slight flat bias)
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # keep previous solution (if any) as a seed
    if previous is not None and previous.shape == (n,) and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # final sanity normalisation
    out = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        out.append(v / s if s > 0 else _uniform(n))
    return out


# ----------------------------------------------------------------------
# Smooth max surrogate (used inside L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_max_func(y: np.ndarray, n: int, p: float):
    """Smooth approximation of max(conv) and its gradient w.r.t logits y."""
    a = _softmax(y)                     # a lies on the simplex
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c               # ≤ 0
    exp_arg = p * shifted
    exp_arg = np.clip(exp_arg, -700, 0)  # avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:                    # fallback – exact max locations only
        mask = conv == max_c
        w = mask.astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)     # length 3n‑2
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

    # chain rule for logits
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    # smooth value  (1/p)·log Σ exp(p·conv)  =  (log w_sum + p·max_c)/p
    obj = (np.log(w_sum) + p * max_c) / p
    return obj, grad_y


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate (logits → simplex)
# ----------------------------------------------------------------------
def _lbfgsb_smooth_opt(start_a: np.ndarray,
                       n: int,
                       deadline: float,
                       p: float,
                       max_iter: int = 800) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smooth surrogate for a given smoothing p."""
    y0 = np.log(np.maximum(start_a, 1e-15))

    # SciPy will call our function with the extra args (n, p)
    res = _opt.minimize(_smooth_max_func, y0,
                        args=(n, p),
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': max_iter, 'disp': False})
    a_opt = _softmax(res.x)
    val = _objective_surrogate(a_opt)   # exact surrogate (2·n·max(conv))
    return a_opt, val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int,
                         eta0: float = 0.10) -> tuple[np.ndarray, float]:
    """Simple Euclidean‑projected sub‑gradient descent on max(conv)."""
    a = start_a.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient for the active shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # stepsize ≈ 1/√t
        eta = eta0 / math.sqrt(it)
        a = a - eta * g

        # simplex projection (clip‑then‑renormalise – cheap and keeps non‑negativity)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Problem size (4000 works well: strong enough expressiveness,
    # fast enough FFTs)
    # --------------------------------------------------------------
    n = 4000
    print(f"\n--- Problem size -------------------------------------------------")
    print(f"Length of the sequence: n = {n}")

    # --------------------------------------------------------------
    # Stage 0 – generate a diverse pool of seeds.
    # --------------------------------------------------------------
    print("\n--- Seed generation ------------------------------------------------")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # Rate seeds by the *exact* surrogate (fast FFT based)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # Initialise global best
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – smooth‑max L‑BFGS‑B optimisation with increasing p.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS‑B optimisation (smooth max) ---------------------------")
        p_schedule = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0, 1280.0]
        max_iter_per_p = 800          # generous for each smoothing level

        for seed_idx, seed_vec in enumerate(best_seeds):
            if time.time() >= deadline:
                break
            cur_a = seed_vec.copy()
            for p in p_schedule:
                if time.time() >= deadline:
                    break
                cand, val = _lbfgsb_smooth_opt(cur_a, n, deadline, p, max_iter=max_iter_per_p)
                print(f"[L‑BFGS] seed #{seed_idx+1} p={p:6.1f} → surrogate = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    print(f"    ** new global best = {best_val:.6f}")

                # Random restart after each p (helps escape shallow local minima)
                if time.time() < deadline:
                    y_noise = np.log(np.maximum(cand, 1e-15)) + rng.normal(scale=0.12, size=n)
                    perturbed = _softmax(y_noise)
                    cand2, val2 = _lbfgsb_smooth_opt(perturbed, n, deadline, p,
                                                     max_iter=max_iter_per_p // 2)
                    if val2 < best_val:
                        best_val, best_seq = val2, cand2.copy()
                        print(f"[Restart] seed #{seed_idx+1} p={p:.1f} → improved to {best_val:.6f}")
                cur_a = cand  # continue optimisation from the current point

    # --------------------------------------------------------------
    # Stage 2 – final projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient polishing ----------------------------------------")
        remaining = deadline - time.time()
        # Estimate a safe number of iterations (roughly 1500 per second on a typical CPU)
        max_iters = int(min(600_000, remaining * 1500))
        if max_iters > 0:
            cand, val = _subgradient_descent(best_seq, n, deadline, max_iters=max_iters,
                                             eta0=0.08)
            print(f"[Subgradient] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"    ** improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
