Algorithm
---------

2.  Evaluate the cheap surrogate `2·n·max(conv(a,a))` (the denominator is 1
    because each seed is ℓ₁‑normalised).  Keep the best few seeds.
3.  Refine every elite seed with a continuation L‑BFGS optimiser that
    minimises a smooth log‑sum‑exp approximation of the max‑convolution.
    The schedule of the smoothing parameter β grows up to 2·10⁴.
4.  Run a long exponentiated‑gradient (mirror‑descent) optimisation on

    convolution indices within a small relative tolerance of the current
    maximum (smooth active‑set).  Step‑size ηₜ = 0.5/√t.
5.  If time permits, linearly up‑sample the current best sequence to a

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv(seq,seq)) for an ℓ₁‑normalised vector.
    (If sum(seq)=1 then the true objective equals this value.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape (skewed polynomial)."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _linear_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** p
    return a / a.sum()


def _linear_increasing_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = i ** p
    a[0] = 1e-30
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta≥0)."""
    if beta == 0.0:
        return _uniform(n)
    # use scipy.signal.kaiser if available; otherwise fallback to i0 formula
    try:
        w = _signal.kaiser(n, beta).astype(np.float64)
    except Exception:  # pragma: no cover
        a = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
        denom = (n - 1) / 2.0
        arg = 1.0 - (a / denom) ** 2
        arg = np.clip(arg, 0.0, None)
        w = _special.i0(beta * np.sqrt(arg)) / _special.i0(beta)
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑like density a_i ∝ 1/√(x(1−x))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    a = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), eps))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Seed generation – large random sweep over parametrised families
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None,
                   max_random: int = 2500) -> list[np.ndarray]:
    """Return a diverse collection of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, 0.0))
    seeds.append(_power(n, 1.0))
    seeds.append(_power(n, 2.0))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    # a few Kaiser windows with different β
    for beta in (2.0, 5.0, 8.0, 12.0):
        seeds.append(_kaiser_window(n, beta))
    seeds.append(_arcsine_shape(n))

    # parametric families – random draws
    for _ in range(max_random // 5):
        # random Beta
        a = 10 ** rng.uniform(-3, 1)
        b = 10 ** rng.uniform(-3, 1)
        seeds.append(_beta_shape(n, a, b))

        # random power‑law
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

        # random exponential decay
        lam = rng.exponential(scale=0.001) + 0.0001
        seeds.append(_exp_decay(n, lam))

        # random linear‑power (decreasing)
        p_lin = rng.uniform(0.0, 4.0)
        seeds.append(_linear_power(n, p_lin))

        # mixture of uniform with a Beta
        mix_alpha = rng.uniform(0.0, 1.0)
        a_beta = _beta_shape(n,
                             10 ** rng.uniform(-2, 1),
                             10 ** rng.uniform(-2, 1))
        seeds.append(_mix_shapes(_uniform(n), a_beta, mix_alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous solution
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # final sanitisation – guarantee positivity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
        else:
            v = v / s
        cleaned.append(v)
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_obj_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Objective + gradient for the smooth approximation of max(conv(a,a)).
    v_local are logits (softmax will produce a on the simplex).
    """
    a = _softmax(v_local)                     # stay on simplex
    conv = _conv_fft(a, a)
    shift = float(conv.max())                  # shift for numerical stability
    # smooth max via log‑sum‑exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight vector (length 2n‑1)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimises the smooth surrogate using L‑BFGS‑B.  Returns a
    ℓ₁‑normalised vector (soft‑max of the logits).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 3000.0, 5000.0, 10000.0, 20000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # cap the iteration budget to what we can afford
        max_iter = min(max_iter_per_beta, int(remaining * 0.04 * 1e5))
        if max_iter <= 0:
            break
        try:
            res = _opt.minimize(fun=_lbfgs_obj_grad,
                                x0=v,
                                args=(n, beta),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except Exception as e:               # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) optimiser – multi‑active set
# ----------------------------------------------------------------------
def _exp_gradient_optimize_multi(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 max_iters: int,
                                 rng: np.random.Generator,
                                 epsilon_rel: float = 1e-4) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    The sub‑gradient is averaged over all indices whose convolution value
    lies within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    eps_abs = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        thresh = max_val - max(max_val * epsilon_rel, eps_abs)
        active = np.where(conv >= thresh)[0]

        if active.size == 0:
            active = np.array([int(conv.argmax())])

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient w.r.t. a
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

        # step‑size ηₜ = 0.5 / √t
        eta = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * grad_a
        a = _softmax(log_a)

        # occasional random perturbation to escape plateaus
        if it % 200_000 == 0:
            noise = rng.normal(scale=0.02, size=n)
            log_a = np.log(a + 1e-30) + noise
            a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:,} best≈{cur_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – optional fine‑grid refinement
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid and ℓ₁‑renormalise."""
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return new_seq / new_seq.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we can finish cleanly
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Stage 0 – problem size
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Starting search (n={n}) ===")

    # --------------------------------------------------------------
    # Stage 1 – seed generation & cheap surrogate ranking
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:               # pragma: no cover
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr, max_random=2500)
    print(f"Generated {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Best surrogate values (top {top_k} seeds):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – continuation L‑BFGS on elite seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Current best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 3000.0, 5000.0,
                  10000.0, 20000.0]

    # limit to the very best few seeds to keep total time reasonable
    elite_count = min(5, len(top_seeds))
    for idx in range(elite_count):
        if time.time() > deadline:
            print("[Time] deadline reached before processing all elite seeds.")
            break

        seed_vec = top_seeds[idx]
        remaining = deadline - time.time()
        max_iter_per_beta = max(1200, int(min(8000, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS on elite seed #{idx+1} (max_iter/beta={max_iter_per_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient refinement (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(400_000, remaining * 0.30 * 1e5))
        if max_exp_iters > 0:
            print("\n--- Exponentiated‑gradient refinement (exact max) ---")
            seq_exp = _exp_gradient_optimize_multi(global_best_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=max_exp_iters,
                                                   rng=rng,
                                                   epsilon_rel=1e-4)
            val_exp = _objective_surrogate(seq_exp)
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – optional up‑sampling and fine‑tuning on a finer grid
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 30.0:
            new_n = n * 2
            print(f"\n--- Upsampling to n={new_n} ---")
            up_seq = _upsample(global_best_seq, new_n)

            # A quick L‑BFGS on the finer grid (lighter budget)
            up_deadline = time.time() + min(120.0, time_left * 0.5)
            up_opt = _lbfgs_smooth_optimize(up_seq,
                                            new_n,
                                            up_deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=1500)
            up_val = _objective_surrogate(up_opt)
            print(f"[Upsample‑L‑BFGS] surrogate = {up_val:.6f}")

            if up_val < global_best_val:
                global_best_seq = up_opt.copy()
                global_best_val = up_val
                n = new_n
                print(f"[Update] new best after up‑sampling = {global_best_val:.6f}")

            # Final exponentiated‑gradient on the fine grid
            if time.time() < deadline:
                rem = deadline - time.time()
                max_exp_fine = int(min(300_000, rem * 0.20 * 1e5))
                if max_exp_fine > 0:
                    print("\n--- Final exponentiated‑gradient on fine grid ---")
                    seq_final = _exp_gradient_optimize_multi(global_best_seq,
                                                             n,
                                                             deadline,
                                                             max_iters=max_exp_fine,
                                                             rng=rng,
                                                             epsilon_rel=5e-5)
                    val_final = _objective_surrogate(seq_final)
                    if val_final < global_best_val:
                        global_best_seq = seq_final.copy()
                        global_best_val = val_final
                        print(f"[Final‑ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
