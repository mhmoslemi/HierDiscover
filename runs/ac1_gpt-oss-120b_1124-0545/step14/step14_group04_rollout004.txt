"""
Hybrid differential‑evolution optimiser for the non‑negative step‑function
minimisation problem.

The algorithm consists of three stages:

0️⃣  Build a diverse initial population of ℓ₁‑normalised vectors
    (analytic windows, Dirichlet draws, low‑frequency cosine mixtures,
    and the previous best if present).

1️⃣  Run a classic DE (mutation = a + F·(b−c), binomial crossover) on the
    *exact* objective  f(a)=2·n·max(conv(a,a)).
    After each mutation the candidate is projected onto the probability
    simplex, guaranteeing non‑negativity and Σa=1 (the objective is
    scale‑invariant).  The population size is modest (≈50) so many
    generations fit well into the 1000 s budget.

2️⃣  Finish with a short additive sub‑gradient polish (the same routine used
    in the reference solution) to squeeze out any remaining gain.

The best vector found is returned as a plain Python list and stored in the
global variable ``height_sequence_1`` for warm‑starting the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σp = 1):
        2·n·max(conv(p,p)).
    This matches ``evaluate_sequence`` for vectors already clamped to
    [0,1000] and with Σp>0.01.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitudes/phases; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Differential Evolution optimiser
# ----------------------------------------------------------------------
def _de_optimize(
    population: list[np.ndarray],
    n: int,
    deadline: float,
    F: float = 0.5,
    CR: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Classic DE with simplex projection.  Returns the best vector seen."""
    pop = np.stack(population)               # shape (NP, n)
    NP = pop.shape[0]

    # pre‑evaluate
    values = np.array([_objective_surrogate(v) for v in pop])
    best_idx = int(np.argmin(values))
    best_vec = pop[best_idx].copy()
    best_val = float(values[best_idx])

    it = 0
    while time.time() < deadline:
        it += 1
        for i in range(NP):
            if time.time() >= deadline:
                break
            # pick three distinct indices a,b,c ≠ i
            idxs = list(range(NP))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # enforce non‑negativity before crossover
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross_mask = np.random.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[np.random.randint(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])

            # simplex projection (keeps Σ=1 automatically)
            trial = _proj_simplex(trial)

            # evaluate
            trial_val = _objective_surrogate(trial)

            # selection
            if trial_val < values[i]:
                pop[i] = trial
                values[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_vec = trial.copy()

        # occasional progress report
        if it % 5 == 0:
            print(f"[DE] generation {it:4d}  best surrogate = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (short polish)
# ----------------------------------------------------------------------
def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 100_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent on the exact objective with simplex projection."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 20_000 == 0:
                print(f"[SubGrad‑Polish] it={it:7d}  best={best_val:.6f}")

    return best_x, best_val

        if prev_arr.shape == (n,) and prev_arr.sum() > 0:
            seeds.append(prev_arr / prev_arr.sum())

    # Ensure every seed is ℓ₁‑normalised and non‑negative
    seeds = [np.maximum(s, 0.0) for s in seeds]
    seeds = [_proj_simplex(s) for s in seeds]

    # ------------------------------------------------------------------
    # Evaluate seeds and keep the top NP of them
    # ------------------------------------------------------------------
    NP = 48                                   # population size (multiple of 3)
    surrogate_vals = [_objective_surrogate(s) for s in seeds]
    ordered = sorted(zip(seeds, surrogate_vals), key=lambda kv: kv[1])
    population = [vec for vec, _ in ordered[:NP]]
    print(f"Initial population size: {len(population)} (best seed value ≈ {ordered[0][1]:.6f})")

    # ------------------------------------------------------------------
    # 1️⃣  Differential Evolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 1] Differential Evolution")
        best_vec, best_val = _de_optimize(population, n, deadline, F=0.45, CR=0.7)
        print(f"Best after DE: {best_val:.6f}")
    else:
        # fallback – use the single best seed we already have
        best_vec = population[0].copy()
        best_val = _objective_surrogate(best_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Short sub‑gradient polish (helps exploit the exact non‑smooth landscape)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 2] Sub‑gradient polishing")
        remaining = deadline - time.time()
        # allocate a modest number of iterations proportionally to remaining time
        max_iters = int(min(120_000, remaining * 500))
        best_vec, best_val = _subgrad_optimize(best_vec, n, deadline, max_iters=max_iters, eta0=0.6)
        print(f"Polish result: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (uses the real security‑clamped function)
    # ------------------------------------------------------------------
    print("\n[Final] Evaluating with official routine …")
    try:
        true_val = evaluate_sequence(best_vec.tolist())
        print(f"Surrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("WARNING: evaluate_sequence raised:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
