from scipy.sparse import lil_matrix

def convolution_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = lil_matrix((rows, n))
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over f index
        for i in range(i_start, i_end+1):
            j = k - i
            A[k, j] = f[i]
    return A.tocsr()
```

def _lbfgs_optimize(start, n, deadline, maxiter=500, alpha_start=20.0, alpha_end=2000.0):
    # schedule: we can loop over a few increasing alpha values, each performing minimize with warm start.
    x0 = np.log(start + 1e-12)  # ensure positive; treat as log weights.
    best_x = start.copy()
    best_val = _objective_surrogate(best_x)
    for stage_alpha in np.linspace(alpha_start, alpha_end, num=5):
        if time.time()>deadline: break
        # define closure capturing alpha
        def fun_x(x):
            w = np.exp(x - np.max(x))
            w = w / w.sum()
            conv = _conv_fft(w, w)
            max_c = np.max(conv)
            exp_vals = np.exp(stage_alpha * (conv - max_c))
            sum_exp = exp_vals.sum()
            val = max_c + np.log(sum_exp)/stage_alpha
            # gradient
            p = exp_vals / sum_exp
            grad_w = 2.0 * np.correlate(w, p, mode='valid')
            dot = np.dot(grad_w, w)
            grad_x = w * (grad_w - dot)
            # compute gradient of value w.r.t x
            return val, grad_x
        res = scipy.optimize.minimize(fun_x, x0, method='L-BFGS-B', jac=True,
                                      options={'maxiter': maxiter, 'disp': False})
        if not res.success:
            continue
        w_opt = np.exp(res.x - np.max(res.x))
        w_opt = w_opt / w_opt.sum()
        val_opt = _objective_surrogate(w_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_x = w_opt.copy()
        # update x0 to current solution for next stage
        x0 = res.x
    return best_x, best_val

def _lbfgs_optimize(start, n, deadline, maxiter=200, alpha=1000):
    # start is array with sum=1.
    x0 = np.log(start + 1e-12)
    def fun(x):
        w = np.exp(x - np.max(x))
        w = w / w.sum()
        conv = _conv_fft(w, w)
        max_c = np.max(conv)
        shifted = conv - max_c
        # compute smooth approximation: val = max_c + log(sum exp(alpha * shifted))/alpha
        exp_vals = np.exp(alpha * shifted)
        sum_exp = exp_vals.sum()
        val = max_c + np.log(sum_exp) / alpha
        # gradient
        p = exp_vals / sum_exp
        # compute gradient wrt w
        # compute g = 2* (each j) sum_i w_i p_{i+j}
        # use sliding window view
        windows = np.lib.stride_tricks.sliding_window_view(p, n)
        grad_w = 2.0 * windows @ w
        # chain rule for softmax
        dot = np.dot(grad_w, w)
        grad_x = w * (grad_w - dot)
        return val, grad_x
    res = scipy.optimize.minimize(fun, x0, method='L-BFGS-B', jac=True,
                                  options={'maxiter': maxiter, 'disp': False})
    if not res.success:
        # fallback to start
        return start, _objective_surrogate(start)
    w_opt = np.exp(res.x - np.max(res.x))
    w_opt = w_opt / w_opt.sum()
    return w_opt, _objective_surrogate(w_opt)
```

"""
Hybrid optimizer with a second‑order L‑BFGS refinement.

The search proceeds in stages (all respect a global time budget):
  0️⃣  Build a diverse pool of ℓ₁‑normalised seeds (analytic families,
      Dirichlet draws, low‑frequency mixtures, previous best).
  1️⃣  Refine the few best seeds with a *smooth* max‑convolution surrogate
      using a full L‑BFGS optimizer (soft‑max relaxation).  The soft‑max
      temperature (α) is increased linearly, letting the optimiser focus
      on the true max while still being smooth enough for second‑order
      steps.
  2️⃣  A short Adam run (smooth surrogate) from the best L‑BFGS result.
  3️⃣  Exact sub‑gradient descent on the true objective (max‑conv).
  4️⃣  Long‑run mirror‑descent (exponential‑gradient) polish.
  5️⃣  Simulated‑annealing fine‑tune.

All intermediate results are kept; the best sequence found before the
deadline is returned.  The implementation lives completely in pure
Python/NumPy/SciPy (no file I/O, no closures, no lambdas) and fits well
inside the 1000 s limit.
"""

import time
import math
import numpy as np
import scipy.optimize
from scipy.sparse import csr_matrix, lil_matrix

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑p = 1):
        2·n·max(conv(p,p)).
    This is exactly the value returned by ``evaluate_sequence`` after the
    security clamping (because the sum is 1 and the values are already in range).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 2000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth soft‑max surrogate."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(best_x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 20_000 == 0:
                print(f"[Adam] it={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val


def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent on the exact objective."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 25_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _expgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with KL divergence (multiplicative updates)."""
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        x = x * np.exp(delta)

        s = x.sum()
        if s <= 0:
            x = _uniform(n)
        else:
            x = x / s

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Tiny random transfers + subset reshuffles, accepting worsening moves."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 100_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# L‑BFGS refinement (smooth surrogate)
# ----------------------------------------------------------------------
def _lbfgs_obj(x: np.ndarray, n: int, alpha: float):
    """Smooth surrogate (log‑sum‑exp) and its gradient w.r.t the log‑variables x."""
    # softmax to obtain a valid probability vector
    x = x - np.max(x)                # for numerical stability
    w = np.exp(x)
    w = w / w.sum()

    conv = _conv_fft(w, w)
    max_c = np.max(conv)
    shifted = conv - max_c

    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    val = max_c + np.log(sum_exp) / alpha

    # probabilities of each shift in the soft‑max
    p = exp_vals / sum_exp

    # gradient w.r.t. w: g_j = 2 * Σ_i w_i * p_{i+j}
    # Compute using a sliding window view (no explicit Python loops)
    windows = np.lib.stride_tricks.sliding_window_view(p, n)  # shape (n,n)
    grad_w = 2.0 * windows @ w

    # chain rule through the soft‑max
    dot = np.dot(grad_w, w)
    grad_x = w * (grad_w - dot)
    return val, grad_x


def _lbfgs_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    maxiter: int = 200,
    alpha: float = 1000.0,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the soft‑max surrogate (alpha controls smoothness)."""
    x0 = np.log(start + 1e-12)       # start in log‑space (guarantees positivity)
    res = scipy.optimize.minimize(
        _lbfgs_obj,
        x0,
        args=(n, alpha),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'disp': False},
    )
    if not res.success:
        # fall back to the start vector
        return start, _objective_surrogate(start)

    w_opt = np.exp(res.x - np.max(res.x))
    w_opt = w_opt / w_opt.sum()
    return w_opt, _objective_surrogate(w_opt)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Length: a compromise between expressiveness and speed.
    n = 4000
    print(f"\n--- Sequence length: n = {n} ---")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate each seed with the *exact* objective (they are already ℓ₁‑normalised)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0.5 – L‑BFGS refinement on the smooth surrogate
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- L‑BFGS refinement (smooth surrogate) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):   # a few promising starts
            if time.time() >= deadline:
                break
            # two passes with increasing α
            cand, val = _lbfgs_optimize(seed_vec, n, deadline,
                                        maxiter=200, alpha=500.0)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[L‑BFGS] seed #{idx+1} improved → {best_val:.6f}")

            # a second tighter pass
            cand, val = _lbfgs_optimize(best_seq, n, deadline,
                                        maxiter=200, alpha=2000.0)
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[L‑BFGS] tighter pass improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            max_iters = int(min(120_000, remaining * 120))
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=max_iters,
                alpha_start=20.0,
                alpha_end=1800.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Additive sub‑gradient descent on the *exact* objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additive sub‑gradient optimisation ---")
        remaining = deadline - time.time()
        max_iters = int(min(150_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.7
        )
        print(f"[SubGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Long‑run Exponential‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Exponential‑gradient polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 1500))
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, deadline, max_iters=max_iters, eta0=0.25
        )
        print(f"[ExpGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        remaining = deadline - time.time()
        max_iters = int(min(600_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=max_iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
