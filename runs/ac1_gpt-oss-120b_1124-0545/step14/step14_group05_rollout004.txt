"""
propose_candidate
=================

A self‑contained optimiser for the ``evaluate_sequence`` objective.

The objective is scale‑invariant:
O(p) = 2·n·max(p∗p) / (∑p)² = 2·n·max(p∗p)     when ∑p = 1.
Hence we optimise a probability vector ``p`` on the simplex.

The algorithm works in three stages:

1️⃣  **Diverse seeds** – analytic families (uniform, triangular,
    Gaussian, Beta, Tukey, arcsine, semicircle) plus random Dirichlet
    draws and sparse supports.  The best few (by the smooth surrogate
    2·n·max(p∗p)) are kept.

2️⃣  **Smooth optimisation** – a soft‑max Adam optimiser that minimises
    the differentiable surrogate  Lα(p) = (1/α)·log∑ₖexp(α·(p∗p)ₖ).
    The temperature α is increased from 20 to 10 000, giving a
    gradually tighter approximation of the ``max``.
    The best result of the Adam passes is taken forward.

3️⃣  **Fine‑tuning** – (a) projected sub‑gradient descent on the true
    ``max(p∗p)`` (additive updates with a diminishing stepsize) and
    (b) an aggressive simulated‑annealing stage that shuffles mass,
    performs tiny transfers, and also moves probability away from the
    indices that dominate the current maximal convolution coefficient.

All stages respect the global time budget (default 1000 s) and the best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain Python list.

The implementation is deliberately lightweight, uses only ``numpy`` and
the standard library, and runs comfortably within the 2‑CPU limit.
"""

import time

# Linear‑algebra / geometry helpers

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗b using a real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Surrogate 2·n·max(p∗p) for a probability vector p (∑p=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    return a / a.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – very heavy at the ends, zero in the centre."""
    # shift indices by 0.5 to avoid division by zero at the exact ends
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return a / a.sum()


def _semicircle_shape(n: int) -> np.ndarray:
    """Continuous semicircle law discretised on n points."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    a = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_support(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Exactly `m` equal non‑zero entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()

def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 20.0,
    alpha_end: float = 10_000.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
          Lα(p) = (1/α)·log Σₖ exp(α·(p∗p)ₖ)
    with a linearly increasing temperature α.
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    N = _next_pow2(2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution via FFT
        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        # temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)

        # log‑sum‑exp smooth max
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        # gradient of the smooth surrogate = 2·corr(p, soft)
        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # projection onto simplex
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if t % 30_000 == 0:
                print(f"[Adam] iter={t:7d} best={best_val:.6f} α={alpha:.1f}")

    return best_x, best_val

) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent on max(p∗p).
    The sub‑gradient at the maximal shift k is 2·p shifted by (k).
    """
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    N = _next_pow2(2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]
        k = int(conv_full.argmax())

        # sub‑gradient of max at shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.08,
    min_temp: float = 1e-13,
) -> tuple[np.ndarray, float]:
    """
    Aggressive SA with two move types:
      * reshuffle a random small subset (Dirichlet redistribution)
      * transfer a tiny amount of probability from a “heavy” contributor
        of the current peak to a random location.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # exponential cooling
        T = max(T * 0.99997, min_temp)

        if rng.random() < 0.55:
            # targeted move from a heavy contributor of the current peak
            conv = _conv_fft(cur, cur)
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            contrib = cur[idx] * cur[rev]
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                src = rng.choice(idx, p=probs)
                dst = rng.integers(n)
                if src != dst and cur[src] > 1e-12:
                    eps = rng.random() * min(cur[src], 0.10)  # up to 10 % of source mass
                    cand = cur.copy()
                    cand[src] -= eps
                    cand[dst] += eps
                else:
                    cand = cur.copy()
            else:
                cand = cur.copy()
        else:
            # small random redistribution on a subset of 5‑12 entries
            k_sub = rng.integers(5, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()
                if it % 150_000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best, best_val

# Seed generation

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidate vectors."""
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_gaussian(n, 0.25))
    seeds.append(_gaussian(n, 0.35))
    seeds.append(_beta_shape(n, 2.0, 2.0))
    seeds.append(_tukey_window(n, 0.5))
    seeds.append(_arcsine_shape(n))
    seeds.append(_semicircle_shape(n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse exact‑size supports
    for m in (120, 200, 300, 400):
        for _ in range(3):
            seeds.append(_random_sparse_support(n, m, rng))

    # low‑frequency cosine mixtures
    for nf in (2, 3):
        for _ in range(4):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                seeds.append(a / s)

    # mixtures of uniform and a sparse support
    for mix_alpha in (0.3, 0.5):
        for m in (250, 350):
            sparse = _random_sparse_support(n, m, rng)
            mixed = _mix_two(_uniform(n), sparse, mix_alpha)
            seeds.append(mixed)

    # previously found best (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # final normalisation guard
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search – within ``budget_s`` seconds – for a non‑negative step function
    that minimises the official ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose the length of the step function.
    # Larger n gives a finer discretisation for the same theoretical bound.
    # ------------------------------------------------------------------
    n = 5000
    print(f"\n=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # Stage 0 – generate a large, diverse seed library
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Created {len(seeds)} seeds.")

    # evaluate surrogate and keep the top performers
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    best_seeds = [vec for vec, _ in scored[:top_k]]
    print(f"Top {top_k} surrogate values after seeding:")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max Adam optimisation (multiple starts)
    # ------------------------------------------------------------------
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Soft‑max Adam optimisation (multiple starts) ---")
        # allocate roughly 35 % of the remaining time to Adam
        adam_deadline = time.time() + 0.35 * (deadline - time.time())
        for idx, seed_vec in enumerate(best_seeds[:5]):
            if time.time() >= adam_deadline:
                break
            # give each start a fair share of the Adam budget
            per_start_deadline = time.time() + (adam_deadline - time.time()) / (5 - idx)
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                per_start_deadline,
                max_iters=250_000,
                alpha_start=20.0,
                alpha_end=10_000.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected Sub‑gradient descent ---")
        subgrad_deadline = deadline - 5.0  # keep a few seconds for the final SA
        remaining = subgrad_deadline - time.time()
        max_iters = int(min(1_200_000, remaining * 1500))
        best_seq, best_val = _subgrad_optimize(
            best_seq,
            n,
            subgrad_deadline,
            max_iters=max_iters,
            eta0=0.6,
        )
        print(f"[Sub‑grad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        sa_deadline = deadline
        remaining = sa_deadline - time.time()
        max_iters = int(min(1_500_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq,
            n,
            sa_deadline,
            max_iters=max_iters,
            init_temp_factor=0.08,
            min_temp=1e-13,
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
