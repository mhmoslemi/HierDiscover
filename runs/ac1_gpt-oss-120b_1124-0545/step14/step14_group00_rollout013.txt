"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a simple Adam optimiser on a smoothed max‑convolution objective.
The algorithm:

* generate a few diverse seeds (uniform, random Dirichlet, beta shapes)
* for each seed run Adam on the log‑softmax parametrisation.  The loss is a
  log‑sum‑exp surrogate of the max‑convolution; the schedule of the smoothing
  temperature β grows from 1 to 10 000.
* keep the best vector (in terms of the exact objective)
* finally polish the survivor with a short projected sub‑gradient run and
  return it.

The implementation respects the 1000 s wall‑clock budget, prints a few
progress messages and stores the result in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# optional SciPy for optional L‑BFGS polishing (not required for correctness)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    ``seq`` may be any non‑negative vector; the function is scale‑invariant.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _smooth_max_and_grad(logits: np.ndarray,
                         n: int,
                         beta: float):
    """
    Returns (smooth_obj, grad_logits) where ``smooth_obj`` is the
    log‑sum‑exp surrogate of the max‑convolution (scaled exactly as the
    real objective) and ``grad_logits`` is the gradient w.r.t. the logits.
    """
    a = _softmax(logits)                 # stays on the simplex
    conv = _conv_fft(a, a)               # length 2n‑1
    shift = np.max(conv)                 # for numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth approximation of max(conv)
    smooth_max = shift + np.log(sum_exp) / beta
    # scaled objective (identical to true objective when sum(a)=1)
    obj = 2.0 * n * smooth_max

    # gradient w.r.t. a  ( ∂/∂a  ∑ w_k * conv_k )
    w = exp_beta / sum_exp               # weights that sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # derivative of ∑ w_k conv_k

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 150_000,
                   beta_schedule=None,
                   lr: float = 0.10) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best probability vector (ℓ₁‑normalised) found.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    v = np.log(start_vec + 1e-30)        # log‑softmax parameters
    m = np.zeros_like(v)                # first moment
    v2 = np.zeros_like(v)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta1_pow = 1.0
    beta2_pow = 1.0

    best_seq = _softmax(v)
    best_val = _objective(best_seq)

    # How many iterations per β‑stage?
    stage_len = max(1, max_iters // len(beta_schedule))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- choose current β ----
        stage_idx = min((it - 1) // stage_len, len(beta_schedule) - 1)
        beta = float(beta_schedule[stage_idx])

        # ---- compute surrogate loss & gradient ----
        obj_smooth, grad = _smooth_max_and_grad(v, n, beta)

        # ---- Adam update ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v2 / (1.0 - beta2_pow)

        step = lr / (np.sqrt(v_hat) + eps)
        v -= step * m_hat

        # ---- evaluate the **exact** objective ----
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if it % 20_000 == 0:
            print(f"[Adam] it={it:7d} β={beta:5.0f} best={best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Simple projected sub‑gradient optimiser (fine polishing)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns a (potentially) improved sequence.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic diminishing step‑size
        lr = 0.30 / math.sqrt(it)
        a = a - lr * g
        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Analytic families (used only for seeding)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # problem size – 4000 works well and matches the previous runs
    n = 4000

    print("\n--- Seeding stage ---")
    seeds = []

    # 1) Uniform baseline
    seeds.append(_uniform(n))

    # 2) A few random Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 0.8, 1.5, 3.0):
        if time.time() > deadline:
            break
        vec = _random_dirichlet(n, conc, rng)
        seeds.append(vec)

    # 3) A coarse grid of beta‑shapes (only a handful – cheap)
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            if time.time() > deadline:
                break
            seeds.append(_beta_shape(n, a, b))

    # keep the best few seeds after a cheap surrogate evaluation
    surrogate_vals = [(vec, 2.0 * n * np.max(_conv_fft(vec, vec))) for vec in seeds]
    surrogate_vals.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in surrogate_vals[:5]]
    print(f"Selected {len(top_seeds)} promising seeds for Adam optimisation.")

    global_best_seq = None
    global_best_val = math.inf

    # ------------------------------------------------------------------
    # 4) Adam optimisation from each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n[Adam] optimisation from seed #{idx + 1}")
        # allocate a slice of remaining budget (roughly equal per seed)
        remaining = deadline - time.time()
        # aim for about 150 k iterations per seed unless time runs out
        max_iters = int(min(150_000, remaining * 0.12 * 1e5))
        max_iters = max(30_000, max_iters)

        seq_opt = _adam_optimize(seed_vec, n, deadline,
                                 max_iters=max_iters,
                                 lr=0.12)
        val_opt = _objective(seq_opt)
        print(f"[Adam] best surrogate value from this run = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Short projected sub‑gradient polishing (if time left)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(300_000, remaining * 0.15 * 1e5))
        if max_sub > 0:
            print("\n--- Sub‑gradient polishing ---")
            seq_polished = _subgradient_opt(global_best_seq, n, deadline,
                                            max_iters=max_sub)
            val_polished = _objective(seq_polished)
            if val_polished < global_best_val:
                global_best_val = val_polished
                global_best_seq = seq_polished.copy()
                print(f"[Polish] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – should never happen, but keep a safe return
        global_best_seq = _uniform(n)

    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
