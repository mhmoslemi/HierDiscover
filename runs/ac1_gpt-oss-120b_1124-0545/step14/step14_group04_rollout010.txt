def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_conv_matrix(f)  # shape (2n-1, n)
    b = np.full(2*n-1, M, dtype=np.float64)

"""
Hybrid optimizer with a dedicated linear‑programming refinement stage.

The algorithm proceeds as follows (all steps respect the overall time budget):

0️⃣  Generate a diverse set of ℓ₁‑normalised seeds (analytic windows,
    random Dirichlet draws, low‑frequency cosine mixes and the previous
    best solution if one exists).  Keep the best few seeds.

1️⃣  Perform a short Adam optimisation on a smooth log‑sum‑exp surrogate
    of the max‑convolution – this quickly reaches a good basin.

2️⃣  **LP‑refinement** – given the current sequence f, solve the linear
    program

        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M  for all k,
                    g ≥ 0 ,

    where M = max_k (f*f)[k].  This finds directions that move mass to
    positions where f is small while respecting the current worst‑case
    convolution bound.  A small convex combination of f and the LP
    solution is taken and projected onto the probability simplex.
    Repeating this 10‑20 times typically pushes the bound noticeably
    lower than the pure gradient‑based approach.

3️⃣  Fine‑tune the result with a few thousand sub‑gradient steps on the
    exact objective, followed by a short simulated‑annealing polish.

4️⃣  Return the best sequence found (and store it as ``height_sequence_1``
    for the next run).

The implementation is fully self‑contained, uses only NumPy and SciPy,
and stays well within the 1000 s budget for typical hardware.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact objective for a normalised vector (Σp = 1):
       2·n·max(conv(p,p)).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, rate: float = 0.998) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = rate ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed handling
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma))
    seeds.append(_gaussian(n, 0.2))
    seeds.append(_gaussian(n, 0.3))
    seeds.append(_gaussian(n, 0.4))
    seeds.append(_exponential_decay(n, 0.9995))
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for a in (0.5, 1.0, 2.0, 3.0):
        for b in (0.5, 1.0, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous solution (if any) – down‑sample / up‑sample if needed
    if previous is not None:
        prev = np.asarray(previous, dtype=np.float64)
        if prev.size != n:
            # simple linear resampling via nearest neighbour
            old_n = prev.size
            step = old_n / n
            idx = (np.arange(n) * step).astype(int)
            idx = np.clip(idx, 0, old_n - 1)
            prev = prev[idx]
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    # final sanity‑check (ensure ℓ₁‑normalised)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean

# Linear‑programming refinement
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A where A·g = conv(f,g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains a copy of f starting at row j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j  subject to  (f * g)[k] ≤ M,  g ≥ 0.
    Returns the optimal g (may be the zero vector) or None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)        # linprog minimises, so negate
    bounds = [(0, None) for _ in range(n)]

    # The problem size is modest (≈8 M entries for n=2000), the default
    # HiGHS interior‑point method works well.
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
    except Exception as exc:
        print("[LP] linprog raised:", exc)
        return None

    if not res.success:
        # In pathological cases the LP may be infeasible (should not happen).
        print("[LP] solver failure:", res.message)
        return None

    g = np.maximum(res.x, 0.0)   # clean tiny negatives
    return g


def _lp_refine(start: np.ndarray,
               n: int,
               deadline: float,
               max_outer: int = 20,
               step_size: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP refinement described above.
    Each outer iteration mixes a small amount of the LP direction
    into the current vector and projects onto the simplex.
    """
    f = start.copy()
    best_seq = f.copy()
    best_val = _objective_surrogate(f)

    for outer in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        # current worst‑case convolution value
        M = np.max(_conv_fft(f, f))

        g = _lp_max_sum(f, M)
        if g is None or g.sum() < 1e-14:
            # nothing useful found – terminate early
            break

        # a modest convex combination (step_size) keeps us safely inside the feasible region
        new_f = f + step_size * g
        new_f = _proj_simplex(new_f)

        cur_val = _objective_surrogate(new_f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new_f.copy()
            f = new_f               # continue from the improved point
            print(f"[LP] iter={outer:2d} improved → {best_val:.6f}")
            # gradually increase step size (but stay ≤0.2)
            step_size = min(step_size * 1.15, 0.20)
        else:
            # no improvement – shrink step size
            step_size *= 0.70
            if step_size < 1e-4:
                break

    return best_seq, best_val

# iteration limits that adapt to the remaining time budget)

def _adam_softmax_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    alpha_start: float = 20.0,
    alpha_end: float = 1800.0,
    lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    # Adam on the smooth log‑sum‑exp surrogate (identical to the reference)
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        fx = np.fft.rfft(x, n=N)
        conv_full = np.fft.irfft(fx * fx, n=N)[: 2 * n - 1]

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
        max_c = np.max(conv_full)
        exp_vals = np.exp(alpha * (conv_full - max_c))
        soft = exp_vals / np.sum(exp_vals)

        fs = np.fft.rfft(soft, n=N)
        corr = np.fft.irfft(np.conj(fx) * fs, n=N)[:n]
        grad = 2.0 * corr

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        s = x.sum()
        if s <= 0.0:
            x = _uniform(n)
        else:
            x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val


def _subgrad_optimize(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    # additive sub‑gradient descent on the exact objective
    x = start.copy()
    best_x = x.copy()
    best_val = _objective_surrogate(x)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(x, x)
        k = int(np.argmax(conv))

        # sub‑gradient of the active shift k
        g = np.zeros_like(x)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * x[rev]

        eta = eta0 / math.sqrt(it)
        x = x - eta * g
        x = _proj_simplex(x)

        cur_val = _objective_surrogate(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

    return best_x, best_val


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.99995, min_temp)

        if rng.random() < 0.6:
            # reshuffle a random subset
            k_sub = rng.integers(3, 13)
            idx = rng.choice(n, size=k_sub, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k_sub, 1.0)) * total
            cand = cur.copy()
            cand[idx] = new_alloc
        else:
            # tiny transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.07)
            if eps <= 0.0:
                continue
            cand = cur.copy()
            cand[i] -= eps
            cand[j] += eps

        new_val = _objective_surrogate(cand)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = cand
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cand.copy()

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the official ``evaluate_sequence`` objective within ``budget_s`` seconds.
    The function returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # tiny safety margin

    # length – a compromise between expressive power and speed
    n = 2000
    print(f"\n=== Optimisation started – length {n} ===")

    # ------------------------------------------------------------------
    # Stage 0 – seed generation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None

            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    print(f"Generated {len(seeds)} seed vectors.")

    # evaluate seeds with the *exact* surrogate (they are ℓ₁‑normalised)
    scored = [(s, _objective_surrogate(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, _ in scored[:top_k]]
    best_seq = best_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam on smooth surrogate (few short runs)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation (multiple starts) ---")
        for idx, seed_vec in enumerate(best_seeds[:4]):
            remaining = deadline - time.time()
            if remaining <= 0:

            iters = int(min(120_000, remaining * 120))
            cand, val = _adam_softmax_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=iters,
                alpha_start=20.0,
                alpha_end=1800.0,
                lr=0.12,
            )
            if val < best_val:
                best_val, best_seq = val, cand
                print(f"[Adam] seed #{idx+1} improved → {best_val:.6f}")

    # Stage 2 – LP refinement (the novelty of this version)

    if time.time() < deadline:

    # ------------------------------------------------------------------
    # Stage 3 – Sub‑gradient polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient polishing ---")
        remaining = deadline - time.time()
        iters = int(min(150_000, remaining * 300))
        best_seq, best_val = _subgrad_optimize(
            best_seq, n, deadline, max_iters=iters, eta0=0.7
        )
        print(f"[SubGrad] best surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Simulated‑annealing final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final polish ---")
        remaining = deadline - time.time()
        iters = int(min(300_000, remaining * 1800))
        best_seq, best_val = _simulated_annealing(
            best_seq, n, deadline, max_iters=iters
        )
        print(f"[SA] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, official value = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
