def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    V = np.concatenate([cos_feat, sin_feat], axis=1)  # shape (n,2M)

    # initial c = zero
    c = np.zeros(V.shape[1], dtype=np.float64)

    # base logits:
    base_log = np.log(base_seq + 1e-30)

    # Adam state:
    m = np.zeros_like(c)
    v = np.zeros_like(c)

    best_val = float('inf')
    best_a = None

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute log_a
        log_a = base_log + V @ c
        a = _softmax(log_a)
        # compute smoothed gradient via _grad_softmax_max
        # need gradient w.r.t log_a

        # Adam update

        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        c -= step

        # evaluate surrogate exact

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        if t % 50000 == 0:
            print(f"[LowDim Adam] iter {t}, best {best_val:.6f}")
    return best_a if best_a is not None else a

"""
Efficient search for a non‑negative step‑function (probability vector) that
minimises the evaluation function

    2·n·max(conv(a,a)) / (sum a)^2   (sum a = 1 after normalisation)

The algorithm proceeds in three stages:

1) **Low‑dimensional Adam** – the distribution is written as a soft‑max of a
   base (arcsine) shape plus a small Fourier correction.
   Adam optimisation on a smoothed log‑sum‑exp approximation of the max
   quickly finds a good neighbourhood.

2) **Multiplicative (exponentiated‑gradient) polishing** – a classic mirror‑
   descent on the exact max‑convolution objective, using a decreasing step
   size ηₜ = η₀ / √t.

3) **Projected sub‑gradient descent** – after the polishing we switch to a
   Euclidean projected‑gradient method on the true max‑convolution (sub‑
   gradient obtained from the active convolution entry).  This often pushes
   the value a little further down.

The whole routine respects a hard 1000 s wall‑time budget.  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective (surrogate) used internally: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seeds
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of  (1/β)·log Σₖ exp(β·conv(a,a)[k])  w.r.t. the log‑soft‑max
    parameters ``log_a``.  Returns ∂F/∂log_a (shape n).
    """
    a = _softmax(log_a)                 # probability vector, sum 1
    conv = _conv_fft(a, a)               # full self‑convolution (len 2n‑1)

    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                         # softmax weight, sum 1

    # g_a = 2 Σₖ wₖ · shift(a)  (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)            # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]

    # Chain rule through the soft‑max
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (full‑dimensional)
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max‑convolution objective.
    Returns the best simplex vector encountered and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for the smoothing temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

        if t % 50_000 == 0:
            print(f"[Adam] iter {t:,}, best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirror descent on the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror‑descent) optimiser on the exact
    max‑convolution objective.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (size n)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 500_000 == 0:
            print(f"[ExpGrad] iter {it:,}, best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto {x ≥ 0, Σx = 1} (the probability simplex)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 1_000_000,
                             step0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        eta = step0 / math.sqrt(it)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()

        a = a_new

        if it % 100_000 == 0:
            print(f"[ProjGrad] iter {it:,}, best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional Fourier‑corrected Adam optimiser
# ----------------------------------------------------------------------
def _lowdim_adam_opt(base_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     M: int = 12,
                     lr: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Optimises a soft‑max of ``log(base_seq) + V·c`` where V contains the first
    M cosine and M sine Fourier modes.  This restricts the search to a low‑
    dimensional manifold around the base (arcsine) shape.
    Returns the best full sequence found.
    """
    # feature matrix V (n × 2M)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i / (n - 1)
    cos_feat = np.cos(np.outer(theta, np.arange(1, M + 1)))
    sin_feat = np.sin(np.outer(theta, np.arange(1, M + 1)))
    V = np.concatenate([cos_feat, sin_feat], axis=1)        # shape (n, 2M)

    c = np.zeros(V.shape[1], dtype=np.float64)            # low‑dim parameters
    m = np.zeros_like(c)
    v = np.zeros_like(c)

    base_log = np.log(base_seq + 1e-30)

    best_seq = None
    best_val = float("inf")

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        beta = 0.5 + 7.5 * (t / max_iters)               # same schedule as Adam
        log_a = base_log + V @ c
        a = _softmax(log_a)

        grad_log = _grad_softmax_max(log_a, n, beta)    # ∂F/∂log_a
        grad_c = V.T @ grad_log                         # chain rule

        # Adam on the low‑dim parameters c
        m = 0.9 * m + 0.1 * grad_c
        v = 0.999 * v + 0.001 * (grad_c ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        c -= step

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if t % 40_000 == 0:
            print(f"[LowDimAdam] iter {t:,}, best surrogate = {best_val:.6f}")

    # fall‑back in the unlikely case no improvement happened
    if best_seq is None:
        best_seq = _softmax(base_log)

    return best_seq, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence()
    value.  The routine builds a strong arcsine seed, refines it with a
    low‑dimensional Fourier Adam, then applies a long exponentiated‑gradient
    polishing pass followed by a projected sub‑gradient descent.
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin before the hard 1000 s limit
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 4000 works well and matches the previous runs
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Initialise with the discrete arcsine distribution
    # ------------------------------------------------------------------
    a = _arcsine_distribution(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # ------------------------------------------------------------------
    # 2️⃣  Low‑dimensional Fourier‑corrected Adam optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Low‑dimensional Fourier Adam ===")
        a, val = _lowdim_adam_opt(
            base_seq=a,
            n=n,
            deadline=deadline,
            max_iters=200_000,
            M=14,               # a few cosine/sine modes
            lr=0.12,
        )
        print(f"   surrogate after low‑dim Adam = {val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Full‑dimensional Adam on the smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Full‑dimensional Adam (smoothed) ===")
        a, val = _adam_softmax_opt(

            n=n,
            deadline=deadline,
            max_iters=300_000,
            lr0=0.5,
            beta_start=0.5,
            beta_end=12.0,
        )
        print(f"   surrogate after full‑dim Adam = {val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient polishing (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:

        a, val = _exp_gradient_opt(
            start_seq=a,
            n=n,
            deadline=deadline,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            deadline=deadline,
            max_iters=1_000_000,
            step0=1.0,
        )
        print(f"   surrogate after proj‑grad = {val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(a.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = a.tolist()
    return a.tolist()
