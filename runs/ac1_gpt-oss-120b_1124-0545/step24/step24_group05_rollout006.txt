"""
Hybrid multi–stage search for a non‑negative step function that minimizes
evaluate_sequence(seq).

Algorithm
---------
1. **Seed generation** – several symmetric probability vectors are built:
   - discrete arcsine law (theoretical optimum),
   - uniform distribution,
   - convex combinations of arcsine & uniform,
   - a few random Dirichlet draws (symmetrised).

2. **Smoothed max‑convolution descent** – each seed is refined by Adam on the
   smooth approximation of the max‑convolution
       F_β(p) = (1/β)·log Σ_k exp(β·conv(p,p)[k]),
   with a temperature β that is raised from 0.5 to 8.  The parameters are kept
   in log‑softmax space, guaranteeing non‑negativity and sum‑to‑one after each
   update.  The best surrogate (2·n·max(conv)) over all seeds is retained.

3. **High‑β refinement** – the current best vector is further optimised by a
   second Adam run where β grows up to 30.  This pushes the smooth max ever
   closer to the true max.

4. **Multi‑active mirror descent polishing** – a final exponentiated‑gradient
   pass uses *all* near‑maximal convolution entries (within 1 % of the
   current maximum) to compute a sub‑gradient, instead of only the single
   arg‑max.  This typically lowers the final objective a little further.

5. **Symmetrisation** – after every optimisation phase the vector is forced
   to be symmetric (a_i = a_{n‑1‑i}) and re‑normalised; the optimum is known
   to be symmetric, and the projection reduces the search space.

6. **Time management** – the routine respects the 1000 s budget, allocating
   most of the time to the Adam phases and using the remainder for polishing.
   The best sequence found is stored in the global ``height_sequence_1`` for
   possible continuation runs.

The function returns the best list of heights (a probability vector with
∑a_i = 1) found within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (outputs a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _symmetrize(p: np.ndarray) -> np.ndarray:
    """Force symmetry p_i = p_{n-1-i} and renormalise."""
    p = np.asarray(p, dtype=np.float64)
    p = (p + p[::-1]) / 2.0
    s = p.sum()
    if s == 0.0:
        return np.full_like(p, 1.0 / len(p))
    return p / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used for optimisation:
        2·n·max(conv(seq,seq)),
    assuming seq sums to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑softmax params)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters `log_a`. Returns ∂F/∂log_a.
    """
    a = _softmax(log_a)                 # probability vector, ∑a=1
    conv = _conv_fft(a, a)               # length 2n‑1

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                         # w sums to 1, shape (2n‑1,)

    # g_a = 2 Σ_k w_k · shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)            # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]    # slice → length n

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed max‑convolution objective.
    The parameters are kept in log‑softmax space; after each iteration the
    current vector is symmetrised and renormalised.  Returns the best
    probability vector and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _symmetrize(_softmax(log_a))
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase the temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # gradient of the smoothed objective
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # Evaluate the current (exact) surrogate; enforce symmetry
        cur_seq = _symmetrize(_softmax(log_a))
        cur_val = _objective_surrogate(cur_seq)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing with a multi‑active set
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           eta0: float = 12.0,
                           eps_rel: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    The sub‑gradient is formed from *all* entries whose value lies within
    eps_rel of the current maximum, i.e.

        w_k = 1 / |A|   for k ∈ A = {k | conv[k] ≥ (1‑eps_rel)·max(conv)}.

    This often yields a smoother descent than using only the single arg‑max.
    """
    a = _symmetrize(start_seq.copy())
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # active set – all entries close to the current maximum
        thresh = max_val * (1.0 - eps_rel)
        active = np.where(conv >= thresh)[0]
        if active.size == 0:          # fallback – should never happen
            active = np.array([int(conv.argmax())])

        # uniform weights over the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient w.r.t. a (still in primal space)
        a_rev = a[::-1]
        w_a = _conv_fft(w, a_rev)                     # length 3n‑2
        g = 2.0 * w_a[n - 1 : n - 1 + n]               # size n

        # Mirror‑descent step in log‑space
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g

        a = _symmetrize(_softmax(log_a))

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence
    value.  The routine builds a handful of analytic and random seeds,
    refines each with a two‑stage Adam optimiser (moderate and high β),
    and finishes with a multi‑active mirror‑descent polishing pass.
    The best sequence found (as a probability vector) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a few seconds safety margin before the hard deadline
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose the problem size.  4000 is a good compromise between
    # quality and speed; the code will also try 8000 if time permits.
    # ------------------------------------------------------------------
    n_choices = [4000, 8000]   # increase to 16000 if we still have time later
    best_overall_seq = None
    best_overall_val = float("inf")
    best_n = None

    for n in n_choices:
        if time.time() > deadline - 20:
            break

        # ------------------------------------------------------------------
        # Build a compact pool of symmetric seeds for the current length.
        # ------------------------------------------------------------------
        seeds = []

        # 1) arcsine (theoretical champion)
        seeds.append(_arcsine_distribution(n))

        # 2) uniform
        seeds.append(_uniform(n))

        # 3) convex combinations arcsine + uniform (coarse sweep)
        arcs = _arcsine_distribution(n)
        uni = _uniform(n)
        for alpha in np.linspace(0.0, 1.0, 5):      # 0,0.25,…,1
            combo = alpha * arcs + (1.0 - alpha) * uni
            combo = combo / combo.sum()
            seeds.append(combo)

        # 4) a few random Dirichlet draws (symmetrised)
        for _ in range(5):
            cand = rng.dirichlet(alpha=np.full(n, 0.5))
            seeds.append(_symmetrize(cand))

        # ------------------------------------------------------------------
        # Optimise each seed with the first‑stage Adam optimiser.
        # ------------------------------------------------------------------
        remaining = len(seeds)
        for idx, seed_vec in enumerate(seeds, start=1):
            if time.time() > deadline - 10:
                print("[Info] Time almost exhausted – breaking seed loop.")
                break

            seed_vec = np.maximum(seed_vec, 0.0)
            if seed_vec.sum() == 0.0:
                continue
            seed_vec = seed_vec / seed_vec.sum()
            seed_vec = _symmetrize(seed_vec)

            # allocate iterations roughly proportional to remaining time
            time_left = deadline - time.time()
            # conservative estimate: ~2000 Adam iterations per second
            max_iters = int(min(300_000,
                                max(30_000, (time_left - 10.0) /
                                      max(1, remaining - idx + 1) * 2000)))
            if max_iters < 30_000:
                max_iters = 30_000

            print(f"[n={n}] Seed {idx}/{len(seeds)} – max_iters={max_iters}")

            seq, val = _adam_softmax_opt(seed_vec, n, deadline,
                                         max_iters=max_iters,
                                         lr0=0.5,
                                         beta_start=0.5,
                                         beta_end=8.0)

            print(f"   surrogate after low‑β Adam = {val:.6f}")

            if val < best_overall_val:
                best_overall_val = val
                best_overall_seq = seq.copy()
                best_n = n
                print(f"   *** new global best (surrogate) = {best_overall_val:.6f}")

        # ------------------------------------------------------------------
        # High‑β refinement for the best sequence of this length (if any)
        # ------------------------------------------------------------------
        if best_overall_seq is not None and best_n == n:
            if time.time() < deadline - 5:
                time_left = deadline - time.time()
                max_iters = int(min(500_000, time_left * 1500))
                if max_iters < 50_000:
                    max_iters = 50_000
                print(f"\n[Refine n={n}] high‑β Adam with max_iters={max_iters}")
                seq_ref, val_ref = _adam_softmax_opt(best_overall_seq, n, deadline,
                                                     max_iters=max_iters,
                                                     lr0=0.2,
                                                     beta_start=8.0,
                                                     beta_end=30.0)
                if val_ref < best_overall_val:
                    best_overall_val = val_ref
                    best_overall_seq = seq_ref.copy()
                    print(f"   *** improved after high‑β = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with multi‑active mirror descent
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline - 2:
        time_left = deadline - time.time()
        max_iters = int(min(3_000_000, time_left * 1200))
        print(f"\n[Polish] multi‑active mirror descent – max_iters={max_iters}")
        polished_seq, polished_val = _exp_gradient_opt_multi(
            best_overall_seq, best_n, deadline,
            max_iters=max_iters,
            eta0=12.0,
            eps_rel=1e-4)
        if polished_val < best_overall_val:
            best_overall_val = polished_val
            best_overall_seq = polished_seq.copy()
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Polish] no improvement over previous best.")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
