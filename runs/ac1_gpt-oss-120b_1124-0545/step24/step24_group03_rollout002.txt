"""
Step‑function optimizer (2026 edition)

The algorithm tries to beat the previous bound (≈1.5085) by
1. Constructing a very diverse seed set that includes
   * classic analytic families (uniform, 1/√i, β‑distribution, …)
   * two‑parameter families (i⁻ᵖ·e⁻ˡᵃᵐᵇᵈᵃ·i) with an extra offset δ
   * mixtures of two exponentials
2. Randomly sampling the three low‑dimensional families (≈1500 draws)
   and polishing the best few with a cheap Powell optimizer.
3. Selecting the top‑K seeds (K≈5) according to the exact surrogate
   2·n·max(conv) and refining each of them with a short but
   powerful “full‑optimisation” pipeline:
   – Exponentiated‑gradient on the true max‑convolution,
   – L‑BFGS on a smooth log‑sum‑exp surrogate,
   – Gradient descent that flattens the self‑convolution,
   – Simulated‑annealing polishing.
   The pipeline respects the global time‑budget and automatically
   reduces iteration counts when the deadline approaches.
4. The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned as a plain Python list.

The method works completely on the simplex (sum of coefficients = 1)
and never violates the non‑negativity constraints – the final call to
``evaluate_sequence`` returns the genuine objective value.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT. O(n log n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (exact after L1‑normalisation)
#   2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# analytic families (all return L1‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp_family(n: int, p: float, lam: float, delta: float) -> np.ndarray:
    """
    Two‑parameter power‑exponential family with an extra offset δ:
        a_i ∝ (i+δ)^{-p} · exp(-λ·i) , i = 0 … n‑1 .
    Returned vector is L1‑normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64) + delta
    a = (i ** (-p)) * np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential_family(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of two exponentials:
        a_i = (1‑α)·e^{-r1·i} + α·e^{-r2·i}.
    Returned vector is L1‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    e1 = np.exp(-r1 * i)
    e2 = np.exp(-r2 * i)
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Symmetric Dirichlet draw.
    """
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def _explore_families(n: int, rng: np.random.Generator, deadline: float):
    """
    Produce a rich set of seeds:
    - Analytic families (uniform, inverse‑sqrt, powers, exponentials, beta)
    - Parameter families via random search
    Returns a list of L1‑normalised vectors.
    """
    seeds = []

    # Classic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))
    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))
    for rate in np.logspace(-5, -0.2, 15):
        seeds.append(_exp_decay(n, rate))
    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (different concentrations)
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Parameter families
    param_deadline = deadline - 30.0  # reserve a few seconds for later stages
    if param_deadline < time.time():
        param_deadline = time.time() + 1.0
    random_params = _random_param_search(n, rng, param_deadline)
    # keep only the best 20 from the random sweep
    random_params.sort(key=lambda x: x[0])
    for val, vec in random_params[:20]:
        seeds.append(vec)

    # Include previous global result if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # Small perturbations of the current best seed (will be added later)
    return seeds

            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogates:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deep optimisation of a few top seeds (share remaining time)
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        # allocate roughly equal slices to the best two seeds
        slots = min(2, top_k)
        slot_time = remaining_time * 0.85 / slots
        for idx in range(slots):
            seed_vec = top_seeds[idx]
            sub_deadline = time.time() + slot_time
            print(f"\n[Refine] seed #{idx+1} (surrogate {evaluated[idx][1]:.6f}) – "
                  f"budget ≈ {slot_time:.1f}s")
            seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)
            if val_opt < global_best_val:
                global_best_seq = seq_opt
                global_best_val = val_opt
                print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with whatever time is left
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Final polish] remaining time")
        seq_fin, val_fin = _full_optimisation(global_best_seq, n, deadline, rng)
        if val_fin < global_best_val:
            global_best_seq = seq_fin
            global_best_val = val_fin
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
