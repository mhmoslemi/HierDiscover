def _simulated_annealing_local_search(start_seq, n, deadline, sigma_start, T_start, rng):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    start_local = time.time()
    min_sigma = 0.005
    last_recalc = 0
    accepted = 0
    attempts = 0

    while time.time() < deadline:
        attempts += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        # sigma schedule linearly decays
        progress = (time.time() - start_local) / max(deadline - start_local, 1e-12)
        sigma = sigma_start * (1 - progress) + min_sigma * progress

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # update conv copy
        cand_conv = conv.copy()
        # shift contributions
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()

        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            T = T_start * max(0.0, 1 - progress)  # linear decay to 0
            # avoid division by zero
            if T > 0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True
        if not accept:
            continue

        # apply move
        a[i] -= delta
        a[j] += delta
        conv = cand_conv
        cur_val = cand_val
        accepted += 1

        if cand_val < best_val:
            best_val = cand_val
            best_a = a.copy()

        if accepted - last_recalc >= 2000:
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            last_recalc = accepted

        # occasional prints
        if accepted % 5000 == 0:
            print(f"[SA] acc={accepted}, best={best_val:.6f}, cur={cur_val:.6f}, sigma={sigma:.5f}")

    # End while
    return best_a, best_val
```

"""
Enhanced step‑function optimizer.

Algorithm:
1️⃣ Build a large pool of diverse probability‑vector seeds
   (uniform, arcsine, triangular, spikes, many Beta‑α, Dirichlet draws,
   and the previous best solution if any).

2️⃣ Run Adam on a smooth‑max surrogate (log‑sum‑exp) to quickly
   descend into a good basin.

3️⃣ Polishing with exact exponentiated‑gradient updates on the true
   max‑convolution.

4️⃣ Parallel simulated‑annealing local search:
   – two independent processes explore the simplex by moving a random
     amount δ from one coordinate to another.
   – large initial σ (≈0.25) allows big jumps; σ decays linearly.
   – a temperature T (≈0.01·best_val) permits occasional uphill moves,
     also decaying linearly.
   – incremental convolution updates keep each trial O(n).

5️⃣ Final deterministic greedy walk (the classic incremental local search)
   for fine‑grained polishing.

6️⃣ Return the best sequence found (global variable ``height_sequence_1``)
   and report the exact value from ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import multiprocessing as mp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during the search: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # broadened beta schedule (more gradual ramps)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1000.0, 2000.0, 5000.0, 2e4, 5e4,
        1e5, 2e5, 5e5, 1e6, 2e6, 5e6,
        1e7, 2e7, 5e7,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner iterations – longer while there is time
        max_iter_beta = 12000 if (deadline - time.time()) > 30 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # convert to gradient on the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing incremental local search (parallel friendly)
# ----------------------------------------------------------------------


def _simulated_annealing_local_search(start_seq: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      sigma_start: float,
                                      T_start: float,
                                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climbing with occasional uphill moves.
    Moves transfer a random δ from i→j; the convolution is updated
    incrementally (O(n) slice ops).  σ and the temperature decay
    linearly with elapsed time.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    start_local = time.time()
    min_sigma = 0.005
    last_recalc = 0
    accepted = 0
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # pick distinct i, j
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)

        if a[i] <= 1e-15:
            continue

        # linear decay of σ
        progress = (time.time() - start_local) / max(deadline - start_local, 1e-12)
        sigma = sigma_start * (1 - progress) + min_sigma * progress

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()

        # acceptance rule (Metropolis)
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            T = T_start * max(0.0, 1 - progress)  # linear cooling to 0
            if T > 0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if not accept:
            continue

        # apply move
        a[i] -= delta
        a[j] += delta
        conv = cand_conv
        cur_val = cand_val
        accepted += 1

        if cand_val < best_val:
            best_val = cand_val
            best_a = a.copy()

        # periodic full recomputation to avoid drift
        if accepted - last_recalc >= 2000:
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            last_recalc = accepted

        if accepted % 5000 == 0:
            print(f"[SA] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, T≈{T:.6f}, best={best_val:.6f}")

    return best_a, best_val


def _local_search_worker(seq_list,
                         n,
                         deadline,
                         sigma_start,
                         T_start,
                         rng_seed,
                         conn):
    """Worker for multiprocessing – runs SA and sends back the result."""
    rng = np.random.default_rng(rng_seed)
    start_seq = np.asarray(seq_list, dtype=np.float64)
    best_seq, best_val = _simulated_annealing_local_search(start_seq,
                                                          n,
                                                          deadline,
                                                          sigma_start,
                                                          T_start,
                                                          rng)
    conn.send((best_seq.tolist(), float(best_val)))
    conn.close()


# ----------------------------------------------------------------------
# Intensive deterministic local refinement (same as previous code)
# ----------------------------------------------------------------------


def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)
    best_a = a.copy()
    best_val = 2.0 * n * np.max(conv)

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()

        if cand_val < best_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_val = cand_val
            accepted += 1

            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                last_recalc = accepted

        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found (as a plain list) and stores it in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load previous solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # Seed pool (all vectors normalised to sum=1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with many concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – richer α list
    beta_alphas = [
        0.005, 0.01, 0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25,
        0.30, 0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00, 3.00, 5.00,
    ]
    for a_alpha in beta_alphas:
        seeds.append(_beta_distribution(n, a_alpha))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 0️⃣ Evaluate seeds (fast surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Adam smoothing on smooth‑max surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 2️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        polishing_deadline = time.time() + 0.30 * remaining
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 3️⃣ Parallel simulated‑annealing local search
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 15:                              # enough time for parallel phase
            # split the remaining budget equally among two workers
            worker_deadline = time.time() + 0.50 * remaining

            rng_seed1 = rng.integers(1_000_000_000)
            rng_seed2 = rng.integers(1_000_000_000)

            sigma1 = 0.25
            sigma2 = 0.15
            T1 = best_val * 0.01   # larger temperature for the first worker
            T2 = best_val * 0.005  # smaller for the second

            ctx = mp.get_context('spawn')
            parent1, child1 = ctx.Pipe()
            parent2, child2 = ctx.Pipe()
            p1 = ctx.Process(target=_local_search_worker,
                             args=(best_seq.tolist(), n, worker_deadline,
                                   sigma1, T1, int(rng_seed1), child1))
            p2 = ctx.Process(target=_local_search_worker,
                             args=(best_seq.tolist(), n, worker_deadline,
                                   sigma2, T2, int(rng_seed2), child2))
            p1.start()
            p2.start()

            seq1, val1 = parent1.recv()
            seq2, val2 = parent2.recv()
            p1.join()
            p2.join()

            if val1 < best_val:
                best_val, best_seq = val1, np.asarray(seq1, dtype=np.float64)
                print(f"[Parallel SA] worker 1 improved best to {best_val:.6f}")
            if val2 < best_val:
                best_val, best_seq = val2, np.asarray(seq2, dtype=np.float64)
                print(f"[Parallel SA] worker 2 improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final deterministic greedy refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_deadline = deadline
        print("\n--- Final intensive greedy refinement ---")
        seq_loc, val_loc = _intensive_local_search(best_seq, n,
                                                   final_deadline,
                                                   sigma_start=0.05,
                                                   rng=rng)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (final greedy)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store result for potential next invocations
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
