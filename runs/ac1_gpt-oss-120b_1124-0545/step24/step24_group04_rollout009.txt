assistantfinal**Solution Explanation**

The task is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative step‑function, the sum is forced to be `1`
(by the normalisation inside the optimiser) and `*` denotes the linear
convolution.  
Only the shape of the vector matters – scaling does not change the
value of `F`.

The optimiser that was used before already performs a rather good
search, but a few extra ideas give the remaining ≈ 0.3 % improvement:

1. **Richer seed pool** – power‑law, highly asymmetric Beta‑distributions
   and *sparse* vectors (few non‑zero entries).  Some of those shapes have
   much smaller additive energy, i.e. fewer pairs of indices that add up
   to the same value, which directly reduces the convolution peak.

2. **Longer Adam phase** – the smooth‑max schedule is exercised with a
   denser schedule and larger inner‑iteration budgets.

3. **Simulated‑Annealing local refinement** – the previous local walk
   accepted only strictly improving moves, which may get stuck in a
   local minimum.  A Metropolis accept rule with a slowly decreasing
   temperature lets the optimiser climb out of shallow basins and find
   slightly better vectors.

4. **Dynamic time allocation** – the global deadline (≈ 1000 s) is split
   into three stages (seed → Adam → exact gradient → annealing) depending
   on the remaining time, so the algorithm always makes the most of the
   available budget.

All heavy work is done on the *probability* simplex (the vector always
sums to 1).  The surrogate objective used during optimisation is the
exact `max(conv)` computed by an FFT, which is fast enough for a few
hundred thousand evaluations.

The final sequence is stored in the global variable `height_sequence_1`
(and also returned) so that a later run can start from the best known
solution.

```python
"""
Non‑negative step‑function optimiser with extra seeds and simulated annealing.

The search consists of four phases:
  1️⃣ Build a very diverse set of seeds (uniform, arcsine, triangular,
      symmetric Beta, asymmetric Beta, power‑law, random sparse vectors,
      Dirichlet draws, Gaussian mixtures, and permutations).
  2️⃣ Adam optimiser on a smooth‑max surrogate (more iterations and a
      finer beta‑schedule than the baseline).
  3️⃣ Exact exponentiated‑gradient polishing on the true max‑convolution.
  4️⃣ Simulated‑annealing local walk (mass‑transfer, swaps, whole‑vector
      permutations) that accepts uphill moves with a temperature schedule.

All phases respect the overall 1000 s time‑budget.  The best sequence found
is saved in the global variable ``height_sequence_1`` and also returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, softmax, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2 * n * max(conv(seq, seq)).
    ``seq`` must already be normalised (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha,alpha) on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _asymmetric_beta(n: int, alpha_l: float, alpha_r: float) -> np.ndarray:
    """Beta(alpha_l,alpha_r) – left and right shape may differ."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha_l - 1.0) * np.power(1.0 - x, alpha_r - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p)  (symmetric power‑law)."""
    i = np.arange(n, dtype=np.float64)
    left = np.power(i + 1.0, -p)
    right = np.power(n - i, -p)
    a = left + right
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few (2‑5) Gaussian bumps, truncated to the interval.
    """
    m = rng.integers(2, 6)                # number of components
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)                     # centre
        sigma = rng.uniform(0.5, max(1.0, n * 0.05))   # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose ``m`` distinct positions and put a Dirichlet‐distributed
    mass on them.  The remaining entries stay zero.
    """
    idx = rng.choice(n, size=m, replace=False)
    probs = rng.dirichlet(np.full(m, 1.0))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = probs
    return a

def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser in log‑softmax space on a smooth‑max surrogate."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a very fine beta‑schedule – the larger the beta the closer the smooth‑max
    # to the true max.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
        5e7, 1e8, 2e8, 5e8,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # spend more inner iterations when we have a lot of time
        remaining = deadline - time.time()
        inner_iters = 18000 if remaining > 60 else 6000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 500 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Exact exponentiated‑gradient polishing (sub‑gradient on true max)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk on the simplex.
    Three move types are used:
      • mass transfer between two indices,
      • full swap of two values,
      • random permutation of the whole vector.
    Uphill moves are accepted with probability exp(-Δ/T).
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # Temperature and step‑size schedules
    T = cur_val * 0.05               # start temperature ≈ 5 % of the score
    sigma = 0.12                     # relative mass‑transfer amplitude
    decay_T = 0.9999
    decay_sigma = 0.99995
    min_sigma = 1e-5
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # pick move type
        move_type = rng.random()
        if move_type < 0.75:          # mass‑transfer (most frequent)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0 or delta > cur[i]:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        elif move_type < 0.95:        # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:                         # global permutation (rare)
            perm = rng.permutation(n)
            new = cur[perm]

        new_val = _surrogate(new)
        dE = new_val - cur_val

        # Metropolis accept rule
        if dE < 0.0 or rng.random() < math.exp(-dE / max(T, 1e-14)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()

        # schedule updates
        T *= decay_T
        sigma = max(min_sigma, sigma * decay_sigma)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a global time budget (≈ 1000 s) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a massive pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with a range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions (edge‑heavy)
    for a in (0.02, 0.05, 0.08, 0.12, 0.20, 0.30, 0.50, 0.80, 1.0):
        seeds.append(_beta_distribution(n, a))

    # Asymmetric Beta (different left/right exponents)
    for al, ar in ((0.02, 0.5), (0.05, 0.5), (0.5, 0.02), (0.1, 0.8)):
        seeds.append(_asymmetric_beta(n, al, ar))

    # Power‑law families
    for p in (0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
        seeds.append(_power_law(n, p))

    # Random Gaussian mixtures

    # Random sparse vectors – investigate low‑collision supports

            seeds.append(_random_sparse(n, m, rng))

    # Include a few random permutations of a few seeds (break symmetry)
    for base in seeds[:8]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Quick seed evaluation (surrogate objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing (smooth‑max surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = deadline - 5.0   # keep a few seconds for the final stage
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated annealing local refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, rng)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq, best_val = seq_sa.copy(), val_sa
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
