"""
Propose a step‑function that (approximately) minimizes ``evaluate_sequence``.

Algorithm (single run, up to ``budget_s`` seconds)
--------------------------------------------------
1. Build a diverse seed pool:
   * flat, arcsine, triangular.
   * a fine grid of symmetric Beta(α,α) densities (α∈[0.05,2]).

   All seeds are normalised to sum = 1 and evaluated with the cheap
   surrogate ``2·n·max(conv(a,a))``.  The five best seeds are kept.

2. For each seed run a *smooth* Adam‑optimiser in the probability‑simplex:
   * Objective = 2·n·soft‑maxβ(conv(a,a))  (β is increased linearly from 1 to 200
     during optimisation – a continuation method).
   * Variables are logits ``y`` (a = softmax(y)).  Adam updates are applied
     on the mirror‑gradient ``∇_y L = a·(∇_a L – <∇_a L,a>)``.
   * Every few thousand steps we evaluate the exact surrogate
     ``2·n·max(conv)`` and keep the best iterate.

3. Greedy refinement:
   * Repeatedly move a small fraction of mass from the index with the
     largest sub‑gradient (active max‑convolution) to the one with the
     smallest sub‑gradient, accepting only improving moves.
   * The loop runs as long as time permits (≈10⁵ – 10⁶ attempts).

4. Final stochastic walk:
   * Randomly transfer tiny amounts of mass between two coordinates,
     accepting improvements (local random search).

5. The best sequence seen is evaluated with the supplied
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT (O(N log N)).
    Returns a real array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _logsumexp(x: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (log_sum_exp, softmax_weights) for a vector ``x``.
    ``softmax_weights`` sums to 1 and equals exp(x - log_sum_exp).
    """
    m = np.max(x)
    exp_shifted = np.exp(x - m)
    sum_exp = exp_shifted.sum()
    log_sum = m + np.log(sum_exp)
    return log_sum, exp_shifted / sum_exp


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq,seq)), ``seq`` is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_start: float = 1.0,
                     beta_end: float = 200.0,
                     lr: float = 0.5,
                     eval_every: int = 4000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *smooth* surrogate
        L(a) = 2·n·log‑sum‑exp(β·conv(a,a)) / β,
    where β grows linearly from ``beta_start`` to ``beta_end``.
    ``start_seq`` must sum to 1.
    Returns the best sequence (according to the exact surrogate) and its value.
    """
    n = len(start_seq)
    # logits for the simplex
    y = np.log(start_seq + 1e-30)
    m = np.zeros_like(y)          # first moment
    v = np.zeros_like(y)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    it = 0
    start_time = time.time()

    while time.time() < deadline:
        it += 1
        # ----- current distribution -----
        a = _softmax(y)

        # ----- convolution -----
        conv = _conv_fft(a, a)                # length 2n‑1

        # ----- smooth max (log‑sum‑exp) -----
        # linearly increase β
        frac = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + frac * (beta_end - beta_start)

        beta_conv = beta * conv
        logZ, w = _logsumexp(beta_conv)       # w: softmax weights, sum(w)=1

        # ----- gradient wrt a -----
        # compute sum_i w_i a_{i-m} for every m using convolution
        conv_w_rev = _conv_fft(w, a[::-1])    # length 3n‑2
        slice_start = n - 1
        weighted_corr = conv_w_rev[slice_start:slice_start + n]  # size n
        # gradient of L w.r.t a: 4·n·weighted_corr
        grad_a = 4.0 * n * weighted_corr

        # ----- mirror (log‑space) gradient -----
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # ----- Adam update on y -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_opt)

        # ----- occasional exact evaluation -----
        if it % eval_every == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # early exit if we already beat the target bound
                if best_val < 1.503:
                    break

    # final check
    final_a = _softmax(y)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val, best_seq = final_val, final_a.copy()
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer: move a small fraction of probability mass from
    the index with the largest sub‑gradient (active max‑conv) to the one
    with the smallest sub‑gradient, accepting only improvements.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-14:
            # nothing to move
            continue

        # try a random fraction, but keep it bounded
        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a          # continue from the improvement
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex – keep moves that improve the exact surrogate.
    """
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence`` within ``budget_s`` seconds.
    The best sequence found is stored globally as ``height_sequence_1``
    and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Retrieve previously stored solution (if any)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # fine grid of symmetric Beta(α,α) densities
    alphas = np.linspace(0.05, 2.0, 200)   # 200 candidates – cheap to evaluate
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:03d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate among seeds = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation on the smooth surrogate (continuation)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Adam optimisation (smoothed max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f"  Adam run #{sidx} (time slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _adam_smooth_opt(seed_vec, per_seed_deadline)
            print(f"   → surrogate after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (Adam)")

            # ------------------------------------------------
            # 2️⃣‑b  Greedy refinement of the Adam result
            # ------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy after Adam)")

    # --------------------------------------------------------------
    # 3️⃣  Final random local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local random walk)")

    # --------------------------------------------------------------
    # 4️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
