"""
A robust optimiser for the black‑box ``evaluate_sequence``.

Strategy
--------
1️⃣  Build a very rich seed pool.
    • Uniform, triangular.
    • The generalized arcsine family a_i ∝ ((i+½)(n‑½‑i))^{‑p}
      – we sweep p with fine resolution (0.30 … 0.70, step 0.005).
    • Symmetric Beta densities a_i ∝ x_i^{α‑1}(1‑x_i)^{α‑1},
      – we sweep α (0.20 … 1.20, step 0.02).
    • Random Dirichlet draws (several concentrations).

2️⃣  Evaluate every seed *exactly* (the surrogate 2·n·max(conv(a,a))
    is identical to ``evaluate_sequence`` for a normalised vector).
    Keep the best few (top 5) for the next phases.

3️⃣  Adam on a log‑softmax (mirror) parametrisation.
    The smooth surrogate uses a log‑sum‑exp with a gradually
    increasing smoothness β.  We devote ~30 % of the remaining
    budget to this stage and run many inner iterations.

4️⃣  Mirror‑descent (exponentiated gradient) on the *exact* max‑convolution.
    The step‑size decays as η₀/√t with a relatively large η₀.
    This step consumes the bulk of the time.

5️⃣  A short simulated‑annealing walk (mass‑transfer moves) to escape
    possible shallow minima and polish the final answer.

All intermediate vectors are kept on the simplex (non‑negative,
unit‑sum).  The final candidate is passed to the supplied
``evaluate_sequence`` and stored globally as ``height_sequence_1``.
"""

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import minimize  # only for the optional L‑BFGS warm‑start (not strictly needed)


# ----------------------------------------------------------------------
# Low‑level helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (≈ O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used throughout the optimiser:
        2·n·max(conv(seq, seq)),   seq is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families of seeds

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """Generalised arcsine family a_i ∝ ((i+½)(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Discrete symmetric Beta(α,α) density."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp smooth approximation of the max‑convolution.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a reasonably long schedule – we let the outer loop run until the deadline
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # allow many inner iterations per beta – capped by remaining time
        max_inner = 15_000
        for _ in range(max_inner):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current point on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a  – the same derivation as in the original code
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient in logits (mirror space)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 500 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 8.0,
                     max_iters: int = 5_000_000) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient w.r.t. the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.
    A random fraction of mass is moved from i → j.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling schedule
        temp = max(final_temp, init_temp * (1.0 - elapsed / total))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)

        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            accept = rng.random() < prob

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence discovered within the allotted budget is stored globally
    as ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Try to reuse a previously found solution (if it matches the length)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a diverse seed pool (rich sweep)
    # --------------------------------------------------------------
    seeds: List[np.ndarray] = [_uniform(n), _triangular(n)]

    # ----- fine sweep over the arcsine exponent p -----
    p_grid = np.arange(0.30, 0.71, 0.005)   # 0.30 … 0.70 inclusive
    for p in p_grid:
        seeds.append(_power_arcsine(n, float(p)))

    # ----- fine sweep over symmetric Beta parameter α -----
    alpha_grid = np.arange(0.20, 1.21, 0.02)
    for a in alpha_grid:
        seeds.append(_beta_symmetric(n, float(a)))

    # ----- a few Dirichlet draws with various concentrations -----
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.6):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ----- previous solution (if any) -----
    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Cheap exact evaluation of all seeds (surrogate = true objective)
    # --------------------------------------------------------------
    seed_infos: List[Tuple[float, np.ndarray]] = []
    print("\n--- Initial seed evaluation (exact) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    # keep the best few seeds for later refinement
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(6, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam smoothing (≈30 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.30
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best after Adam")

    # --------------------------------------------------------------
    # 3️⃣  Exact exponentiated‑gradient polishing (majority of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 4.0                     # keep a few seconds for final walk
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=8.0,
                                           max_iters=10_000_000)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best after EG")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny annealing walk (last seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # --------------------------------------------------------------
    # 5️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
