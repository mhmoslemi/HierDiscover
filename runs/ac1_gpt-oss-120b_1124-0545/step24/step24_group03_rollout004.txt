"""
Step‑function optimizer – “param‑search + SA” version.

The algorithm builds a rich collection of analytic seeds (uniform,
inverse‑sqrt, power‑law, exponentials, beta‑shapes, two‑parameter
power‑exponential families, piecewise‑exponential, gaussian bumps,
sin‑power etc.).  All seeds are L1‑normalised (sum = 1).  Their
surrogate objective

    2·n·max(conv(a,a))

is evaluated with a fast FFT‑based convolution.  The best few seeds are
kept, then a tiny hill‑climb in the (p, λ, δ) parameter space of the
two‑parameter family a_i ∝ (i+δ)^{‑p}·e^{‑λ·i} refines the best candidate.
Finally a lightweight simulated‑annealing polish is applied.

The routine respects a wall‑clock budget (default 1000 s), stores the
best sequence in the global variable ``height_sequence_1`` and returns it
as a plain Python list.  All numbers are non‑negative and the final
sequence has L1‑norm = 1, so the true value returned by
``evaluate_sequence`` equals the surrogate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    L = a.shape[0] * 2 - 1
    N = 1 << ((L - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact, after normalising sum=1)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for a L1‑normalised seq."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families – all return a *normalised* (sum=1) vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    e1 = np.exp(-r1 * i)
    e2 = np.exp(-r2 * i)
    a = (1.0 - alpha) * e1 + alpha * e2
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    a /= a.sum()
    return a


def _power_exp_shift(n: int, p: float, lam: float, delta: float) -> np.ndarray:
    """Two‑parameter family with an additional shift δ ≥ 0."""
    i = np.arange(1, n + 1, dtype=np.float64) + delta
    a = (i ** (-p)) * np.exp(-lam * (i - 1 - delta))
    # note: the exponential still uses the original index (i‑1) for the rate,
    # but the shift only changes the power‑law denominator.
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_exp(n: int, m: int, lam: float) -> np.ndarray:
    """
    First m entries are constant, the remainder decays exponentially.
    m may be zero (pure exponential) or n (pure uniform).
    """
    if m <= 0:
        # pure exponential
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i)
        a /= a.sum()
        return a
    if m >= n:
        return _uniform(n)

    # geometric series for the tail
    tail_len = n - m
    if lam == 0.0:
        geo_sum = tail_len
    else:
        geo_sum = (1.0 - np.exp(-lam * tail_len)) / (1.0 - np.exp(-lam))

    # constant c satisfies: m*c + c*geo_sum = 1  ->  c = 1/(m+geo_sum)
    c = 1.0 / (m + geo_sum)

    a = np.empty(n, dtype=np.float64)
    a[:m] = c
    if lam == 0.0:
        a[m:] = c
    else:
        tail_idx = np.arange(tail_len, dtype=np.float64)
        a[m:] = c * np.exp(-lam * tail_idx)
    # normalise just in case numeric errors accumulated
    a /= a.sum()
    return a


def _gaussian_bump(n: int, mu: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at mu with standard deviation sigma."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _sin_power(n: int, q: float) -> np.ndarray:
    """a_i ∝ sin^q(π·(i+1)/(n+1)), zero at the boundaries."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** q
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    mixed_sum = mixed.sum()
    if mixed_sum == 0.0:
        return a.copy()
    mixed /= mixed_sum
    return mixed


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (Gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_attempts: int = 250_000,
    start_sigma: float = 5e-4,
    T_start: float = 1e-2,
    T_end: float = 1e-9,
) -> np.ndarray:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)
    cur_val = best_val
    attempts = 0

    n = a.shape[0]

    while attempts < max_attempts and time.time() < deadline:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq

    # least‑squares on the log‑profile (very cheap).
    def _fit_initial_params(seq: np.ndarray):
        i = np.arange(1, n + 1, dtype=np.float64)
        # avoid zeros for log‑fit
        eps = 1e-12
        log_a = np.log(seq + eps)
        # linear model: log a_i ≈ -p·log(i+δ) - λ·i
        # We'll fix δ = 0 for the quick fit and solve for p, λ via linear regression.
        X = np.column_stack((-np.log(i), -i))
        coeffs, _, _, _ = np.linalg.lstsq(X, log_a, rcond=None)
        p0, lam0 = coeffs
        delta0 = 0.0
        return max(p0, 0.01), max(lam0, 0.0), delta0

    # Initialise hill‑climb parameters
    p_cur, lam_cur, delta_cur = _fit_initial_params(best_seq)
    sigma_p, sigma_lam, sigma_delta = 0.05, 5e-4, 0.2
    hill_deadline = deadline - 5.0  # keep a small buffer for polishing

    while time.time() < hill_deadline:
        # propose a small random perturbation
        p_new = max(0.01, p_cur + sigma_p * rng.standard_normal())
        lam_new = max(0.0, lam_cur + sigma_lam * rng.standard_normal())
        delta_new = max(0.0, delta_cur + sigma_delta * rng.standard_normal())

        cand = _power_exp_shift(n, p_new, lam_new, delta_new)
        val_cand = _surrogate(cand)

        if val_cand < best_val:
            best_val = val_cand
            best_seq = cand
            p_cur, lam_cur, delta_cur = p_new, lam_new, delta_new
        else:
            # occasional acceptance (simulated annealing style) to escape flat regions
            if rng.random() < 0.005:
                p_cur, lam_cur, delta_cur = p_new, lam_new, delta_new

        # slowly shrink the proposal widths
        sigma_p *= 0.9995
        sigma_lam *= 0.9995
        sigma_delta *= 0.9995

        # break early if proposal widths become negligible
        if sigma_p < 1e-5 and sigma_lam < 1e-7 and sigma_delta < 1e-4:
            break

    print(f"\n[Hill‑climb] best surrogate = {best_val:.6f} (p={p_cur:.4f}, λ={lam_cur:.6f}, δ={delta_cur:.4f})")

    # ------------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polish (fast, limited budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate remaining time (but not more than 20 % of the original budget)
        remaining = deadline - time.time()
        polish_deadline = time.time() + min(remaining, budget_s * 0.15)
        best_seq = _simulated_annealing(best_seq, polish_deadline, rng)
        best_val = _surrogate(best_seq)
        print(f"[Polish] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Compute the true evaluation (may be a tiny bit slower)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.9f}, true evaluate_sequence = {true_val:.9f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
