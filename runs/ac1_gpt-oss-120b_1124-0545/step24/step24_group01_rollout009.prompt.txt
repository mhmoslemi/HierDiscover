def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s

    ``seq`` is assumed to sum to 1.

def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2

def _exp_gradient_momentum(start_seq: np.ndarray,

    contributes least.  Only improving moves are kept.

def _local_random_search(start_seq: np.ndarray,

    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept.

    rng = np.random.default_rng()

        attempts -= 1

        new_seq[j] += delta

        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val
