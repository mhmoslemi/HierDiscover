def subgrad_exact(a):
    conv = _conv_fft(a, a)
    max_b = conv.max()
    active = np.where(conv >= max_b - 1e-12)[0]  # tolerance
    w = np.zeros_like(conv)

* Finally performs a few random‑perturbation restarts of the hard‑max EG
  to escape shallow local minima.

All work respects the 1000 s wall‑clock budget.  The best sequence is stored
in the global variable ``height_sequence_1`` and also returned as a list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (sequences are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for sum(seq) = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – good starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (conc < 1 → sparse)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser (same idea as the reference code)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
):
    """Minimise the smoothed max‑convolution using L‑BFGS‑B."""
    try:
        import scipy.optimize as _opt
    except Exception:  # pragma: no cover
        _opt = None

    if _opt is None:                     # fallback – should never happen
        return start_a.copy(), _objective_surrogate(start_a)

    # Log‑softmax parameters – ensures positivity and sum‑to‑1
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_f = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # stable log‑sum‑exp of β·(conv‑shift)
            exp_beta = np.exp(beta_f * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_f)

            # Gradient of the smoothed surrogate (derived in the
            # reference implementation)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # tiny callback – abort on timeout
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_f}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hard‑max exponentiated‑gradient optimiser (exact max, single active index)
# ----------------------------------------------------------------------
def _hard_eg_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 8.0,
):
    """
    Mirror‑descent (exponentiated‑gradient) optimiser that uses the *exact*
    max‑convolution sub‑gradient (single active index).  Returns the best
    sequence seen and its surrogate value.
    """
    a = start_a.copy()
    v = np.log(a + 1e-30)               # log‑parameters (softmax)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(np.argmax(conv))        # single active shift

        # gradient of the active term  b_k = a·B_k·a  (B_k is the anti‑diag matrix)
        rev = a[::-1]
        shift = k - (n - 1)
        if shift >= 0:
            grad_a = np.concatenate([np.zeros(shift), rev[: n - shift]])
        else:
            grad_a = np.concatenate([rev[-shift :], np.zeros(-shift)])
        grad_a = 2.0 * grad_a

        # mirror‑descent gradient in the log‑space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        step = eta0 / math.sqrt(it)
        v = v - step * grad_v
        a = _softmax(v)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random perturbation + short hard‑EG refinement (helps escape plateaus)
# ----------------------------------------------------------------------
def _perturb_and_refine(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 80_000,
    sigma: float = 0.02,
    eta0: float = 6.0,
):
    """
    Create a Gaussian perturbation of ``base_seq`` in log‑space,
    run a short hard‑EG phase and keep the best result.
    """
    v0 = np.log(base_seq + 1e-30)
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(base_seq)

    while time.time() < deadline:
        # generate perturbation
        v_pert = v0 + sigma * rng.normal(size=n)
        a_pert = _softmax(v_pert)

        sub_deadline = deadline
        seq, val = _hard_eg_opt(
            start_a=a_pert,
            n=n,
            deadline=sub_deadline,
            max_iters=max_iters,
            eta0=eta0,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        # shrink perturbation amplitude for next round
        sigma *= 0.7
        if sigma < 1e-4:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point (required by the platform)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a deep smoothed L‑BFGS optimisation from several seeds,
    followed by a dedicated hard‑max EG polish and a few random‑perturbation
    restarts, all within the supplied time budget.  The best sequence is stored
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length is a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best (if any) and adapt its length
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.shape == (n,):
            base_seed = prev_arr / prev_arr.sum()
        else:
            # truncate or pad with zeros to fit length n
            if prev_arr.size > n:
                base_seed = prev_arr[:n]
            else:
                pad = np.zeros(n, dtype=np.float64)
                pad[: prev_arr.size] = prev_arr
                base_seed = pad
            base_seed = base_seed / base_seed.sum()
    else:
        base_seed = None

    # --------------------------------------------------------------
    # Assemble a small set of promising seeds
    # --------------------------------------------------------------
    seeds = []

    if base_seed is not None:
        seeds.append(base_seed)

    seeds.append(_inverse_sqrt(n))
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))
    seeds.append(_random_dirichlet(n, conc=0.01, rng=rng))

    if base_seed is not None:
        pert = base_seed + 0.004 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        if pert.sum() > 0:
            seeds.append(pert / pert.sum())

    # normalise everything (soft‑max will keep it anyway)
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation of seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[: max(1, len(evaluated))]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[: len(top_seeds)], 1):
        print(f" Seed #{i}: {v:.8f}")

    # --------------------------------------------------------------
    # β‑schedule for the smooth L‑BFGS phase (more points at the end)
    # --------------------------------------------------------------
    beta_schedule = [
        0.5,
        1.0,
        2.0,
        5.0,
        10.0,
        20.0,
        40.0,
        80.0,
        200.0,
        500.0,
        1_000.0,
        2_000.0,
        5_000.0,
        10_000.0,
        20_000.0,
        50_000.0,
        100_000.0,
        200_000.0,
        500_000.0,
        1_000_000.0,
        2_000_000.0,
        5_000_000.0,
        10_000_000.0,
        20_000_000.0,
        50_000_000.0,
        100_000_000.0,
        200_000_000.0,
        500_000_000.0,
        1_000_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Run (deep) L‑BFGS for each seed – time is split dynamically
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        # allocate up to ~20 % of the remaining budget (minimum 5 s)
        remaining = deadline - time.time()
        share = max(5.0, remaining * 0.20)
        sub_deadline = time.time() + share

        print(f"\n[Seed {si}/{len(top_seeds)}] L‑BFGS budget ≈ {share:.1f}s")

        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3000,
        )
        print(f"  → L‑BFGS surrogate = {val_lbfgs:.8f}")

        # keep the best result from this seed
        if val_lbfgs < global_best_val:
            global_best_val = val_lbfgs
            global_best_seq = seq_lbfgs.copy()
            print(f"   *** New global best after L‑BFGS = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final polishing on the current global best (hard‑max EG)
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final hard‑max EG polishing ---")
        seq_hard, val_hard = _hard_eg_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=500_000,
            eta0=5.0,
        )
        if val_hard < global_best_val:
            global_best_val = val_hard
            global_best_seq = seq_hard.copy()
            print(f"[Hard‑EG] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # A few random‑perturbation restarts (if time permits)
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Random perturbation restarts ---")
        seq_pert, val_pert = _perturb_and_refine(
            base_seq=global_best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=80_000,
            sigma=0.02,
            eta0=6.0,
        )
        if val_pert < global_best_val:
            global_best_val = val_pert
            global_best_seq = seq_pert.copy()
            print(f"[Perturb] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
