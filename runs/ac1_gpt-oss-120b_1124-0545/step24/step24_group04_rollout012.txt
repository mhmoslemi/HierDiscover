"""
non‑negative step‑function optimiser – second generation.

The algorithm builds a very diverse pool of seeds (uniform, arcsine,
triangular, power‑law, Dirichlet, symmetric Beta, sorted Dirichlet,
random Gaussian mixtures).  It keeps the few best seeds (by the exact
surrogate 2·n·max(conv)) and refines each of them with three phases:

1️⃣ Adam optimisation on a smooth‑max surrogate (long inner schedules,
   many beta‑values, learning‑rate decay).
2️⃣ Exact exponentiated‑gradient polishing on the true max‑convolution.
3️⃣ Aggressive stochastic‑annealing local search (mass‑transfer,
   swaps, occasional full permutations, uphill‑acceptance with a
   decreasing temperature).

All phases respect the global deadline (≈ 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` and also
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used throughout the optimisation.
    ``seq`` must be a probability vector (sum = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ 1/[(i+½)(n‑½‑i)]^p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    eps = 1e-12
    a = 1.0 / ((i * (n - i)) ** p + eps)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps (truncated to the interval).
    The resulting vector is normalised to sum 1.
    """
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)               # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)       # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    Uses a dense schedule of smooth‑max parameters (beta).
    Returns the best probability vector (sum = 1) seen and its surrogate.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # more aggressive schedule than the baseline
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 5e7, 1e8, 5e8, 1e9
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        # a generous inner budget – more iterations when we have time
        inner_iters = 18000 if remaining > 30 else 6000

        for _ in range(int(inner_iters)):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning‑rate decay – more stable late in the run
            lr = 0.15 / (1.0 + 0.00001 * it)

            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 700 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# ----------------------------------------------------------------------


def _local_annealing(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk with simulated‑annealing acceptance.
    Proposes mass‑transfer, swaps and occasional full permutations.
    Only moves that improve or are accepted probabilistically are kept.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # schedule parameters
    sigma = 0.12               # initial mass‑transfer step size (relative)
    sigma_min = 0.001
    sigma_decay = 0.985

    temp = cur_val * 0.02      # initial temperature for uphill moves
    temp_min = 1e-6
    temp_decay = 0.998

    while time.time() < deadline:
        # choose move type
        move_type = rng.integers(0, 100)

        if move_type < 60:          # 60 % – mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            if new[i] < 0.0:
                continue

        elif move_type < 90:        # 30 % – swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        else:                       # 10 % – whole permutation
            perm = rng.permutation(n)
            new = cur[perm]

        new_val = _surrogate(new)

        # simulated‑annealing acceptance
        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / max(temp, 1e-12)):
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_seq, best_val = new.copy(), new_val

        # update schedules
        sigma = max(sigma_min, sigma * sigma_decay)
        temp = max(temp_min, temp * temp_decay)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (≈ 1000 s) and stores the best sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the evaluation call
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # 0️⃣  Load a previous solution (if any)
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1️⃣  Build a very diverse pool of seeds (all normalised to sum = 1)
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # power‑law / arcsine‑like shapes with various exponents
    for p in (0.5, 0.75, 1.0, 1.25):
        seeds.append(_power_law(n, p))

    # Dirichlet draws with many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            vec = _random_dirichlet(n, conc, rng)
            seeds.append(vec)
            # also try the sorted version (anti‑symmetric)
            seeds.append(np.sort(vec))               # ascending
            seeds.append(np.sort(vec)[::-1])         # descending

    # Symmetric Beta distributions (important extreme shapes)
    for a in (0.05, 0.07, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50,
              0.70, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Random Gaussian mixtures – often produce interesting edge‑heavy shapes
    for _ in range(4):
        seeds.append(_gaussian_mixture(n, rng))

    # A few random permutations of the first few seeds (breaks symmetry)
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 2️⃣  Evaluate all seeds – keep the few best (exact surrogate)
    # --------------------------------------------------------------
    top_k = 5
    best_candidates = []        # list of (value, vector)

    print("\n--- Seed evaluation (exact surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:04d}: surrogate = {val:.6f}")
        # keep a tiny sorted list of the best seeds
        if len(best_candidates) < top_k:
            best_candidates.append((val, vec.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, vec.copy())
                best_candidates.sort(key=lambda x: x[0])

    print(f"[Init] best {top_k} surrogates:", [f"{v:.6f}" for v, _ in best_candidates])

    # --------------------------------------------------------------
    # 3️⃣  Refine each of the top seeds with Adam + exact polishing
    # --------------------------------------------------------------
    global_best_seq = None
    global_best_val = np.inf

    # allocate roughly half the remaining time for the refinement phase
    remaining_total = deadline - time.time()
    if remaining_total > 20 and best_candidates:
        per_seed_time = remaining_total * 0.45 / len(best_candidates)
        print("\n--- Refinement of top seeds ---")
        for rank, (seed_val, seed_vec) in enumerate(best_candidates, start=1):
            if time.time() > deadline:
                break
            print(f"\n[Refine #{rank}] start surrogate = {seed_val:.6f}")

            # Adam smoothing (smooth‑max)
            adam_deadline = time.time() + per_seed_time * 0.55
            seq_adam, val_adam = _adam_smoothing(seed_vec, n, adam_deadline)
            print(f"   [Adam] surrogate = {val_adam:.6f}")

            # Exact exponentiated‑gradient polishing
            eg_deadline = time.time() + per_seed_time * 0.40
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, n, eg_deadline, eta0=5.0)
            print(f"   [EG]   surrogate = {val_eg:.6f}")

            # keep the best of the three
            cand_seq, cand_val = seq_eg, val_eg
            if val_adam < cand_val:
                cand_seq, cand_val = seq_adam, val_adam
            if seed_val < cand_val:
                cand_seq, cand_val = seed_vec, seed_val

            if cand_val < global_best_val:
                global_best_val = cand_val
                global_best_seq = cand_seq.copy()
                print(f"   >>> Updated GLOBAL best (value={global_best_val:.6f})")

    # --------------------------------------------------------------
    # 4️⃣  Aggressive stochastic‑annealing local refinement
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Stochastic‑annealing local refinement ---")
        seq_loc, val_loc = _local_annealing(global_best_seq, n, deadline, rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < global_best_val:
            global_best_seq, global_best_val = seq_loc.copy(), val_loc
            print("   >>> Updated GLOBAL best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Final evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
