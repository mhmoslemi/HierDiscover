The goal is to find a non‑negative vector `a` (the heights of a step function) that minimises  

```
E(a) = 2·n·max(conv(a,a)) / (sum(a)²)          (  n = len(a) )
```

`E(a)` is *scale invariant* – we can normalise every candidate to `sum(a)=1` and then simply minimise  

```
F(a) = 2·n·max(conv(a,a))                (∑a = 1)
```

The maximum of the linear convolution is a piece‑wise quadratic function of `a`.  
Its sub‑gradient for the active index `k* = argmax(conv(a,a))` is

```
g_i = 2·a_{k*‑i}    (i such that 0 ≤ k*‑i < n) , 0 otherwise
```

A projected **exponentiated‑gradient (EG)** step in the log‑domain stays inside the probability simplex automatically.  
To speed‑up convergence we add Adam‑style momentum and a simple *mix‑with‑uniform* post‑processing that smooths the distribution (the uniform component is known to flatten the convolution).

The whole search pipeline is:

1. **Seed generation** – a large, diverse pool (uniform, arcsine, triangular, many asymmetric Beta‑distributions, Dirichlet draws, the previously stored best vector).  
   All seeds are normalised to sum = 1 and evaluated with the cheap surrogate `F`.

2. **Select the best seeds** – keep the `top_k` (default 5) candidates.

3. **Exact optimisation** – for each selected seed run an Adam‑EG optimiser on the *exact* surrogate `F`.  
   The optimiser works entirely in log‑space, respects the simplex, and is given a time‑slice of the remaining budget.

4. **Greedy refinement** – after each EG run move a tiny fraction of mass from the index with the largest sub‑gradient to the one with the smallest (only if the move improves `F`).  This quickly removes the last “bottleneck”.

5. **Local random walk** – a short stochastic walk on the simplex that accepts only improving moves.

6. **Uniform‑mix polishing** – line‑search a mixture `(1‑λ)·a + λ·uniform` (λ ∈ [0, 0.2]) and keep the best λ.

7. **Final evaluation** – call the provided `evaluate_sequence` on the best vector and store it globally.

The algorithm respects the 1000 s wall‑time, works with any `n` (default 4000), uses only NumPy (no external files), and prints progress information.

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Search for a step‑function (probability vector) that minimises
`evaluate_sequence`.  The method builds a diverse seed pool,
optimises the exact max‑convolution using Adam‑style exponentiated
gradient descent, refines with a greedy mass‑transfer, performs a
tiny stochastic walk and finally polishes the result by mixing with
the uniform distribution.
"""

from __future__ import annotations
import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # extremely unlikely, but safe
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq,seq)).
    ``seq`` is assumed to be normalised (sum == 1).
    """
    n = seq.size
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_exp_gradient(start_seq: np.ndarray,
                       deadline: float,
                       lr0: float = 0.8,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the exact surrogate.
    The optimisation stays on the simplex via the softmax mapping.
    Returns the best sequence (as a NumPy array) and its surrogate value.
    """
    n = start_seq.size
    a = start_seq.copy()
    # initialise log‑probabilities (mirror variables)
    logit = np.log(a + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    # main loop
    while time.time() < deadline:
        t += 1

        # --- convolution & active index -------------------------------------------------
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # --- sub‑gradient of the active max‑convolution --------------------------------
        # g_i = 2 * a_{k-i}  for indices where k-i is valid
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # --- mirror‑gradient (gradient w.r.t. log‑probabilities) -------------------------
        dot = np.dot(g, a)
        grad_log = a * (g - dot)            # shape (n,)

        # --- Adam update ---------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (sqrt decay)
        lr = lr0 / math.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logit -= step
        a = _softmax(logit)

        # periodic exact evaluation (every few thousand iterations)
        if t % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # optional early‑stop if we already beat the target
                # (remove the break – we keep searching for the absolute best)
    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Greedy transfer of a tiny fraction of mass from the coordinate that
    contributes most to the current max‑convolution towards the one that
    contributes least.  Only improving moves are kept.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a      # continue from the improved point
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> Tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept.
    """
    n = start_seq.size
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


def _mix_with_uniform(seq: np.ndarray,
                      max_lambda: float = 0.20,
                      steps: int = 41) -> Tuple[np.ndarray, float]:
    """
    Line‑search a convex mixture of ``seq`` with the uniform distribution.
    Returns the mixture with the lowest surrogate value.
    """
    n = seq.size
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = seq
    best_val = _objective_surrogate(seq)

    for lam in np.linspace(0.0, max_lambda, steps):
        mix = (1.0 - lam) * seq + lam * uniform
        mix /= mix.sum()       # renormalise (numerical safety)
        val = _objective_surrogate(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     top_k: int = 5,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored globally as ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0️⃣  Retrieve the previously stored solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = [_uniform(n), _arcsine(n), _triangular(n)]

    # many asymmetric Beta(α,β) shapes – parameters sampled logarithmically
    for _ in range(120):
        # sample log‑uniform in [1e-3, 10]
        log_a = rng.uniform(-3.0, 1.0)
        log_b = rng.uniform(-3.0, 1.0)
        alpha = 10 ** log_a
        beta = 10 ** log_b
        seeds.append(_beta_asymmetric(n, alpha, beta))

    # Dirichlet draws with different concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Cheap surrogate evaluation of all seeds (max‑conv only)
    # ------------------------------------------------------------------
    seed_infos: List[Tuple[float, np.ndarray]] = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:03d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(top_k, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate among seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact optimisation (Adam‑EG) on the top seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / max(1, top_k)
        print("\n--- Exact optimisation with Adam‑EG ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            # a fresh slice of the remaining budget for this seed
            slice_deadline = time.time() + (deadline - time.time()) / (top_k - sidx + 1)
            print(f"  EG run #{sidx}/{top_k}, time slice ≈ {slice_deadline - time.time():.1f}s")
            seq_opt, val_opt = _adam_exp_gradient(seed_vec, slice_deadline)
            print(f"   → surrogate after EG = {val_opt:.6f}")

            if val_opt < best_val:
                best_val, best_seq = val_opt, seq_opt.copy()
                print("   >>> Updated global best (EG)")

            # --------------------------------------------------------------
            # 3️⃣‑b  Greedy refinement of the EG result
            # --------------------------------------------------------------
            if time.time() < deadline:
                seq_ref, val_ref = _greedy_refine(seq_opt, deadline)
                if val_ref < best_val:
                    best_val, best_seq = val_ref, seq_ref.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 4️⃣  Mixing with uniform (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing by mixing with uniform ---")
        seq_mix, val_mix = _mix_with_uniform(best_seq)
        if val_mix < best_val:
            best_val, best_seq = val_mix, seq_mix.copy()
            print(f"   >>> Updated global best after uniform mix = {val_mix:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final random local refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final local stochastic refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=250_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call and return
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
