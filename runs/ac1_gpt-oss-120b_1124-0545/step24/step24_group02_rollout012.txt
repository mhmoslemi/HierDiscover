"""
Propose a step‑function (non‑negative heights) that minimises
    2·n·max(conv(a,a))   with   sum(a)=1.

Algorithm
---------

* Problem size: use a moderately large n (default 8000).  A larger n
  gives a finer discretisation of the continuous optimisation problem
  and typically allows a lower max‑convolution ratio.

* Build a diverse set of initial candidates:
    – the previously best sequence (if it matches the current n)
    – the classic inverse‑sqrt shape a_i ∝ 1/√(i+1)
    – a uniform sequence
    – a very sparse Dirichlet draw
    – symmetric beta‑shaped families   a_i ∝ x^{α‑1}(1‑x)^{α‑1}
      for several values of α∈[0.3,0.9].

  All candidates are ℓ₁‑normalised (∑a_i=1).

* For each candidate we run a *deep smoothed L‑BFGS* optimiser.
  The hard max in the objective is approximated by a log‑sum‑exp
  with a schedule of increasing β.  The optimisation works on the
  log‑softmax parameters, guaranteeing non‑negativity and the simplex
  constraint automatically.  The best sequence among the seeds is kept.

* The best L‑BFGS result is polished with a long *exponentiated‑gradient*
  (mirror‑descent) run that directly minimises the exact max‑convolution.
  Only the (single) most‑active index is used for the sub‑gradient
  (argmax of the convolution) – this gives a sharper descent direction
  than averaging over a tolerance band.

* The final sequence is evaluated with the supplied ``evaluate_sequence``
  function, stored in the global variable ``height_sequence_1`` and
  returned.

All steps respect the supplied wall‑clock budget (default 1000 s) and
print intermediate progress.
"""

import time
import math
import numpy as np

# Optional SciPy – used for L‑BFGS‑B; if unavailable we skip that phase.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (variables are ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))  with sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric beta(α,α) density on the discrete grid.
    a_i ∝ x^{α‑1}(1‑x)^{α‑1} with x = (i+0.5)/n .
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid log(0)
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    raw = np.exp((alpha - 1.0) * (np.log(x) + np.log(1.0 - x)))
    # normalise to sum = 1
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser (surrogate objective)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
):
    """
    Minimise the smoothed max‑convolution using L‑BFGS‑B.
    Returns (best_seq, best_surrogate_val) where best_seq is ℓ₁‑normalised.
    """
    if _opt is None:
        # No SciPy – fallback to a tiny EG run.
        return _exp_gradient_opt(start_a, n, deadline, max_iters=200_000)

    # initialise log‑softmax parameters
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # stable log‑sum‑exp over the convolution values
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient of the smooth surrogate w.r.t. a
            w = exp_beta / sum_exp                     # shape (2n‑1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 2n‑1
            grad_a = 2.0 * full[n - 1 : n - 1 + n]     # gradient wrt a

            # chain rule for the log‑softmax parametrisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 8.0,
    tol: float = 1e-12,
):
    """
    Mirror‑descent (exponentiated gradient) directly on the true max‑convolution.
    Returns (best_seq, best_surrogate_val).
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))                     # single active index

        # sub‑gradient for the max‑convolution (2·J_k a)
        g = np.zeros_like(a)
        rev = k_max - idx
        mask = (rev >= 0) & (rev < n)
        g[mask] = a[rev[mask]]
        g = 2.0 * g

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines analytic seeds, a smoothed L‑BFGS phase and a
    long exponentiated‑gradient polish, all respecting the time budget.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives finer resolution and usually a lower
    # bound for the max‑convolution ratio.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Retrieve previous best (if it matches the current length)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if (
        previous is not None
        and isinstance(previous, list)
        and len(previous) == n
    ):
        prev_arr = np.asarray(previous, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        if prev_arr.sum() > 0:
            prev_arr /= prev_arr.sum()
        else:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_inverse_sqrt(n))
    seeds.append(_uniform(n))
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))

    # Symmetric beta families – the most promising range of α is [0.3, 0.9]
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9]:
        seeds.append(_beta_shape(n, alpha))

    # Ensure all are normalised
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # ------------------------------------------------------------------
    # Quick surrogate evaluation of the raw seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:min(5, len(evaluated))]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:len(top_seeds)], 1):
        print(f" Seed #{i}: surrogate = {v:.8f}")

    # ------------------------------------------------------------------
    # Smoothed L‑BFGS schedule (beta → ∞ recovers the hard max)
    # ------------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        5_000_000.0, 1_000_0000.0, 5_000_0000.0, 1_000_00000.0,
        5_000_00000.0, 1_000_000000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # ------------------------------------------------------------------
    # Run L‑BFGS from each promising seed (time is split dynamically)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate at least 5 s or 15 % of the remaining budget to each seed
        share = max(5.0, remaining * 0.15)
        sub_deadline = time.time() + share

        print(f"\n[Seed {si}/{len(top_seeds)}] L‑BFGS budget ≈ {share:.1f}s")

        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3000,
        )
        print(f"  → L‑BFGS surrogate = {val_lbfgs:.8f}")

        if val_lbfgs < global_best_val:
            global_best_val = val_lbfgs
            global_best_seq = seq_lbfgs.copy()
            print(f"   *** New global best (L‑BFGS) = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Final intensive polishing using exponentiated‑gradient (exact max)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final EG polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=600_000,
            eta0=6.0,
            tol=1e-12,
        )
        if val_eg < global_best_val:
            global_best_val = val_eg
            global_best_seq = seq_eg
            print(f"   *** Improved after EG = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Official evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, "
            f"true evaluate_sequence = {true_val:.8f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

**Explanation of why this should improve the bound**

* **Larger discretisation (n = 8000).**  
  The continuous optimisation problem has a theoretical lower bound of 1 (ratio
  max / average).  A finer grid lets the algorithm approximate the optimal
  continuous shape more accurately, which typically lowers the max‑convolution
  ratio compared with the earlier n = 4000 run.

* **Richer seed pool.**  
  Besides the previous best and the classic inverse‑sqrt shape we inject
  uniform, Dirichlet and a whole family of symmetric beta‑distributions
  (including the arcsine‑type α = 0.5).  One of these analytically‑motivated
  shapes is already close to the optimum; the optimiser therefore starts
  much nearer to a good basin.

* **More aggressive L‑BFGS schedule.**  
  The β‑schedule is extended to larger values and each β receives
  up to 3000 iterations, giving the quasi‑Newton method ample opportunity to
  minimise the smoothed surrogate before it sharply approaches the true
  max‑operator.

* **Sharper EG polishing.**  
  The exponentiated‑gradient phase now uses the *exact* sub‑gradient
  (single arg‑max) rather than an average over a tolerance band, which
  provides a more accurate descent direction when the convolution is already
  nearly flat.  A longer run (up to 600 k iterations) is performed on the
  best L‑BFGS candidate.

* **Time‑budget aware.**  
  All stages monitor the wall‑clock and split the remaining budget dynamically,
  guaranteeing termination well before the 1000 s limit.

Together these refinements typically push the surrogate (and therefore the
true) objective below the previous 1.5085 value, aiming for the target ≈ 1.503
or better.
