def project_simplex(v):
    """Projects v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        if time.time() > deadline:

        max_idx = int(np.argmax(conv))
        # gradient: g_i = 2 * a[max_idx - i] for i in appropriate range else 0
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
        a = a_new
    return best_a, best_val

def sidon_init(n):
    # find prime p such that p^2 < n, largest prime <= sqrt(n)
    limit = int(math.isqrt(n))
    # find prime <= limit
    for cand in range(limit, 1, -1):
        if is_prime(cand):
            p = cand
            break
    else:
        p = 2
    # generate Sidon set
    indices = np.empty(p, dtype=int)
    for i in range(p):
        indices[i] = i * p + (i * i) % p
    # ensure within n
    indices = indices[indices < n]
    s = len(indices)
    if s == 0:
        return None
    a = np.zeros(n)
    a[indices] = 1.0 / s
    # sum = 1
    return a

def _adam_opt(start_seq, n, deadline, max_iters=200_000, beta_start=0.5, beta_end=1e9, alpha=0.3):
    v = np.log(start_seq + 1e-30)  # log prob
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        t = (it-1)/max_iters
        beta = beta_start * ((beta_end/beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        m = beta1 * m + (1-beta1) * grad
        v2 = beta2 * v2 + (1-beta2) * (grad*grad)
        m_hat = m / (1-beta1**it)
        v2_hat = v2 / (1-beta2**it)
        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _smooth_obj_and_grad(v_local, n, beta):
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    # compute exp(beta * (conv - max_c))
    exp_beta = np.exp(beta * (conv - max_c))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]  # length n
    # chain rule for softmax: grad_v = a * (grad_a - np.dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def hill_climb(start_seq, n, deadline, rng, max_iters=3_000_000):

    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # define temperature schedule
    T0 = 0.01 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        move_type = rng.random()
        if move_type < 0.45:  # swap

            if i == j:

            ns = cur_seq.copy()

        elif move_type < 0.85:  # gaussian perturb
            sigma = 0.001 / math.sqrt(it+1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.clip(ns, 0.0, None)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s
        else:  # targeted mass shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])  # donor
            q = int(rng.integers(0, n))
            if p == q:
                continue
            delta = 0.1 * cur_seq[p]  # move 10% of mass from p
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            # ensure nonnegativity
            if ns[p] < 0.0:
                ns[p] = 0.0
            ns = ns / ns.sum()
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # metropolis accept
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in several stages:

1.  Build a diverse set of initial probability vectors:
    – previous best (if any)
    – uniform, exponential decays
    – a Sidon set (pairwise sums are unique)
    – random Dirichlet draws (sparse)

2.  Fast projected sub‑gradient descent on the *exact* max‑convolution.
    The iterate stays on the simplex via a O(n) projection.
    Short runs are performed for each initialisation.

3.  Refine the current best with Adam on a smooth (log‑sum‑exp) surrogate
    of the max‑convolution.  The smoothing parameter β is increased from
    0.5 to 1e9.

4.  Exact exponentiated‑gradient (mirror descent) on the true max
    convolution – the step size decays as 1/√t.

5.  An aggressive simulated‑annealing hill‑climb (swap / Gaussian /
    targeted mass‑shift moves).

All stages respect the global time budget (default 1000 s) and keep the
best sequence seen.  The final sequence is evaluated with the provided
``evaluate_sequence`` black‑box and stored globally for possible
continuations.
"""

import time
import math
import numpy as np

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1  →  2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Smoothed max‑convolution objective.
    Returns (objective, gradient w.r.t. the log‑softmax variables v_local).
    """
    a = _softmax(v_local)                 # probability vector (sum=1)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # log‑sum‑exp smoothing
    exp_beta = np.exp(beta * (conv - max_c))    # max term → 1
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    # gradient w.r.t. a (exact for the smoothed max)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) and normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(math.isqrt(num))
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True


def _sidon_initialisation(n: int) -> np.ndarray | None:
    """
    Construct a Sidon set (pairwise sums distinct) using the classic
    i·p + i² mod p construction where p is the largest prime ≤ √n.
    Returns a probability vector that puts equal weight on the set.
    """
    limit = int(math.isqrt(n))
    p = None
    for cand in range(limit, 1, -1):
        if _is_prime(cand):
            p = cand
            break
    if p is None:
        return None
    idx = []
    for i in range(p):
        val = i * p + (i * i) % p
        if val < n:
            idx.append(val)
    if not idx:
        return None
    idx = np.array(idx, dtype=int)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / len(idx)
    return a

def _projected_subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 200_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    The step size decays as η₀ / √t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a_new = a - eta * g
        a_new = project_simplex(a_new)

        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
        a = a_new

        # occasional random jitter to escape plateaus
        if (it % 5000) == 0 and rng.random() < 0.1:
            jitter = 0.001 * rng.normal(size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    return best_seq, best_val


def _adam_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
    beta_end: float = 1e9,
    alpha: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate (log‑sum‑exp of the convolution).
    The variables are the log‑softmax parameters; the iterate always stays
    inside the probability simplex.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax parametrisation
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam updates (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8_000_000,
    eta0: float = 6.0,
) -> tuple[np.ndarray, float]:

    Exact exponentiated‑gradient descent (mirror descent) on the true max‑
    convolution.  The iterate stays on the simplex by working in log‑space.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:

    return best_seq, best_val


def _hill_climb(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 3_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb with three move types:
    – swap two positions,
    – Gaussian perturbation of all entries,
    – targeted mass‑shift from the pair that contributes most to the current
      maximal convolution entry.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = 0.01 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue

        elif move_type < 0.85:                   # Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.clip(ns, 0.0, None)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                    # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))

            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = 0.1 * cur_seq[donor]      # move 10 % of donor's mass
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            ns = np.clip(ns, 0.0, None)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val

    inits = []

    # a) reuse previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    arr = arr / s
                    inits.append(arr)
        except Exception:
            pass

    # b) uniform and exponential decays
    inits.append(np.full(n, 1.0 / n))
    inits.append(_exp_decay(n, rate=0.001))
    inits.append(_exp_decay(n, rate=0.004))

    # c) Sidon‑set based sparse initialise
    sid = _sidon_initialisation(n)
    if sid is not None:
        inits.append(sid)

    # d) random Dirichlet draws (α in [0.3,0.8] for sparsity)
    for _ in range(4):
        alpha = 0.3 + 0.5 * rng.random()
        dir_vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        dir_vec = dir_vec / dir_vec.sum()
        inits.append(dir_vec)

    # keep only unique vectors (hash based)
    uniq_inits = []
    seen = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen:
            uniq_inits.append(vec)
            seen.add(h)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – fast projected sub‑gradient runs (one per init)
    # ------------------------------------------------------------------
    print("[Phase 1] Projected sub‑gradient screening")
    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + 8.0)     # ~8 s per candidate
        seq, val = _projected_subgradient_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            rng=rng,
            max_iters=180_000,
            eta0=0.6,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"  Init {idx+1}/{len(uniq_inits)} new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smooth surrogate (refine current best)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Phase 2] Adam refinement (smooth max)")
        sub_deadline = deadline
        seq, val = _adam_opt(
            start_seq=best_seq,
            n=n,
            deadline=sub_deadline,
            max_iters=350_000,
            beta_end=1e9,
            alpha=0.28,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"  Adam improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Phase 3] Exact exp‑gradient descent")
        seq, val = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=12_000_000,
            eta0=5.0,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"  Exp‑gradient improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – aggressive hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Phase 4] Hill‑climb local search")
        seq, val = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=4_500_000,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"  Hill‑climb improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
