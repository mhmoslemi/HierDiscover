    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    G = np.zeros_like(a)  # sum squares of gradients
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # update AdaGrad accumulator
        G += g * g
        step = eta0 * g / np.sqrt(G + 1e-12)
        log_a -= step
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Generate a few diverse starting vectors (previous best,
   exponential decays, uniform, a couple of Dirichlet draws).
2. Very short AdaGrad exponentiated‑gradient runs on each start – this
   quickly discards poor candidates and yields a decent seed.
3. Run a long AdaGrad mirror‑descent on the remaining time.  The update is
   performed in log‑space (the exponentiated‑gradient trick) with an
   AdaGrad per‑coordinate step‑size; this supplies many cheap sub‑gradient
   steps (one FFT per iteration) and drives the max‑convolution down.
4. Refine the obtained point with a few L‑BFGS steps on the *smooth*
   max‑convolution (log‑sum‑exp) objective – this gives a second‑order
   push towards the optimum.
5. Finish with an aggressive simulated‑annealing hill‑climb that uses
   swaps, Gaussian noise and a targeted “mass‑shift” move aimed at the
   most offending pair of the active convolution offset.
6. Return the best vector found (as a plain Python list) and store it in
   the global variable ``height_sequence_1`` for possible continuation
   runs.

All heavy work is done with FFT‑based convolutions (O(n log n) per
iteration).  The schedule leaves a small safety margin before the 1000 s
deadline.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sum = 1, all entries > 0)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective when sum(seq)=1,
    i.e. 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by the L‑BFGS refinement
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  The variable v_local lives in log‑softmax space;
    a = softmax(v_local) is a probability vector (sum=1).
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# AdaGrad exponentiated‑gradient optimiser (sub‑gradient on true max)
# ----------------------------------------------------------------------
def _adagrad_exp_gradient_opt(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 10**9,
                              eta0: float = 5.0):
    """
    Mirror‑descent on the *true* max‑convolution using an AdaGrad
    per‑coordinate step size.
    Returns the best probability vector (sum=1) seen and its surrogate.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # log‑softmax variables
    G = np.zeros_like(a)              # accumulated squared gradients
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # AdaGrad accumulation and per‑coordinate step
        G += g * g
        step = eta0 * g / np.sqrt(G + 1e-12)
        log_a -= step
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS refinement on the smooth max‑objective
# ----------------------------------------------------------------------
class _SmoothObjGrad:
    """Callable object required by scipy.optimize.minimize."""
    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v):
        obj, grad = _smooth_obj_and_grad(v, self.n, self.beta)
        return obj, grad


def _lbfgs_refine(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iter: int = 500):
    """
    A short L‑BFGS optimisation of the smoothed max‑convolution.
    Returns the refined probability vector and its surrogate value.
    """
    v0 = np.log(start_seq + 1e-30)
    # a very large β makes the smooth max very close to the true max
    beta = 1e7
    obj_grad = _SmoothObjGrad(n, beta)

    # Run minimise – it will stop either after max_iter or when time is out.
    options = {"maxiter": max_iter, "disp": False}
    res = optimize.minimize(obj_grad,
                            v0,
                            method="L-BFGS-B",
                            jac=True,
                            options=options)

    a = _softmax(res.x)
    val = _objective_surrogate(a)
    return a, val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / perturb / targeted mass‑shift)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with three move types.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule (from ~1% of current value down to 1e‑6)
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturbation
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))

            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]

            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])   # donor index
            q = int(rng.integers(0, n))           # receiver
            if p == q:
                continue
            delta = 0.07 * cur_seq[p]            # move 7 % of donor mass
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            # renormalise (numerical drift)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – keep the classic 4000 entries
    n = 4000

    # 1. Build a small portfolio of initial vectors
    # ------------------------------------------------------------------
    inits = []

    # a) reuse previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:

            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()

        except Exception:

    # b) a few exponential decays
    for rate in (0.001, 0.003, 0.006):

        dir_vec = dir_vec / dir_vec.sum()
        inits.append(dir_vec)

    # ------------------------------------------------------------------
    # 2. Quick screening – a few short AdaGrad EG runs
    # ------------------------------------------------------------------
    print("[Screening] short AdaGrad runs")
    best_seq = None
    best_val = float("inf")

    per_candidate_time = 8.0      # seconds per candidate (adjustable)
    for idx, init in enumerate(inits):
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + per_candidate_time)
        seq, val = _adagrad_exp_gradient_opt(

        if val < best_val:

    # safety fallback
    if best_seq is None:

        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"  Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement on the smooth max‑objective
    # ------------------------------------------------------------------
    if time.time() + 30.0 < deadline:
        print("\n[Phase 3] L‑BFGS refinement")
        seq, val = _lbfgs_refine(best_seq, n, deadline - 30.0, max_iter=500)

    # ------------------------------------------------------------------

        seq, val = _hill_climb(best_seq, n, deadline, rng, max_iters=5_000_000)

    # 6. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    final_val = None

              f"true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs

    return best_seq.tolist()
