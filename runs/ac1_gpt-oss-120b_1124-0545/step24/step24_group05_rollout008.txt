"""
Propose a step‑function (non‑negative vector) that minimizes evaluate_sequence.

Algorithm
---------
1. Build a small library of analytic families (uniform, arcsine‑type,
   generalized Beta laws) and evaluate each one using a fast surrogate
   (2·n·max(conv(a,a))) to obtain a handful of promising seeds.
2. Run a short Adam‑style optimisation on each promising seed,
   minimising a smooth max‑convolution objective (log‑sum‑exp
   approximation).  The optimisation works in the log‑softmax
   parametrisation so the simplex constraints are satisfied automatically.
3. Take the best of the short runs and continue a long‑run Adam optimisation
   with a higher temperature (β up to 32) to approach the true max.
4. Finish with an exponentiated‑gradient (mirror‑descent) pass that uses the
   exact sub‑gradient of the max‑convolution.
5. Return the best sequence found and store it globally for future runs.

The routine respects a hard time‑budget (default 1000 s) and prints
intermediate surrogate values for monitoring.
"""

import time
import math
import numpy as np

# ======================================================================
# 1. Convolution helpers (FFT based, O(n log n))
# ======================================================================
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ======================================================================
# 2. Simple probability / distribution helpers
# ======================================================================
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _power_arcsine_distribution(n: int, p: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) * (n - 0.5 - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _uniform_distribution(n)
    return a / s


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β): a_i ∝ (i+½)^{α‑1} (n‑½‑i)^{β‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (beta - 1.0)
    s = a.sum()
    if s == 0.0:
        return _uniform_distribution(n)
    return a / s


# ======================================================================
# 3. Gradient of the smoothed max‑convolution (log‑softmax parametrisation)
# ======================================================================
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters `log_a`. Returned gradient has shape (n,).
    """
    a = _softmax(log_a)                # probability vector (sum=1)
    conv = _conv_fft(a, a)              # full self‑convolution, length 2n‑1

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                       # w sums to 1

    # g_a = 2 Σ_k w_k · shift(a); computed efficiently using a reversed a
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)          # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]  # centre slice → length n

    # Chain rule through the softmax (Jacobian of softmax)
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ======================================================================
# 4. Optimisers
# ======================================================================
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   beta_start: float = 0.5,
                   beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    The parameters live in log‑softmax space, guaranteeing positivity
    and unit sum.  Returns the best simplex vector seen and its surrogate
    value.
    """
    log_a = np.log(start_seq + 1e-30)   # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Linear schedule for the smoothing temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # Gradient of the smoothed objective w.r.t. log‑softmax params
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam updates (in log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # Evaluate the exact (non‑smoothed) surrogate
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses the classic sub‑gradient of the active entry (the index where
    the self‑convolution attains its maximum).  Returns the best vector
    and the surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry:
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    2·n·max(self‑convolution)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _power_arcsine_distribution(n: int, p: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) * (n - 0.5 - i)) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _uniform_distribution(n)
    return a / s


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on the index set."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (beta - 1.0)
    s = a.sum()
    if s == 0.0:
        return _uniform_distribution(n)
    return a / s


# ----------------------------------------------------------------------
# 3. Gradient of the smoothed max‑convolution (log‑softmax params)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters.  Returns a vector of length n.
    """
    a = _softmax(log_a)                      # probability vector, Σa = 1
    conv = _conv_fft(a, a)                    # length 2n‑1

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # g_a = 2 Σ_k w_k·shift(a)  (computed via convolution with reversed a)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)                # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]        # centre slice → length n

    # Chain rule through the softmax Jacobian
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# 4. Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   beta_start: float = 0.5,
                   beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max‑convolution objective.
    Works in log‑softmax space, thus the simplex constraints are always
    satisfied.  Returns the best vector seen and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses the sub‑gradient corresponding to the current maximal entry.
    Returns the best vector encountered and its exact surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (indices that contribute to k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# 5. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects a hard wall‑clock budget (default 1000 s) and
    prints diagnostic information.  The best sequence is stored in the
    global variable ``height_sequence_1`` for possible continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    n = 4000

    # -----------------------------------------------------------------
    # 5.1 Assemble a pool of analytic / random seeds
    # -----------------------------------------------------------------
    seeds = []

    # classic arcsine (p = 0.5)
    seeds.append(_power_arcsine_distribution(n, 0.5))

    # uniform
    seeds.append(_uniform_distribution(n))

    # power‑arcsine sweep
    for p in np.linspace(0.35, 0.65, 9):
        seeds.append(_power_arcsine_distribution(n, float(p)))

    # Beta‑family sweep (both symmetric and asymmetric)
    alphas = np.linspace(0.3, 2.0, 7)
    betas = np.linspace(0.3, 2.0, 7)
    for a_val in alphas:
        for b_val in betas:
            seeds.append(_beta_distribution(n, float(a_val), float(b_val)))

    # random Dirichlet draws (lightweight)
    for _ in range(5):
        seeds.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # -----------------------------------------------------------------
    # 5.2 Quick surrogate evaluation of all seeds
    # -----------------------------------------------------------------
    seed_infos = []   # (surrogate, vector)
    for idx, vec in enumerate(seeds, start=1):
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"[Seed {idx}/{len(seeds)}] surrogate = {val:.6f}")

    # Keep the K best seeds for a short Adam refinement
    seed_infos.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [vec for _, vec in seed_infos[:top_k]]
    print(f"\n[Info] Selected {len(top_candidates)} top seeds for quick Adam runs.")

    # -----------------------------------------------------------------
    # 5.3 Short Adam optimisation on each top candidate
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for i, cand in enumerate(top_candidates, start=1):
        if deadline - time.time() < 30.0:
            print("[Info] Insufficient time for further quick Adam runs.")
            break

        max_iters = 60_000
        print(f"[Quick Adam {i}/{len(top_candidates)}] max_iters={max_iters}")
        seq, val = _adam_optimize(cand, n, deadline,
                                  max_iters=max_iters,
                                  lr0=0.5,
                                  beta_start=0.5,
                                  beta_end=4.0)
        print(f"   quick‑Adam surrogate = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   *** new best after quick Adam (surrogate) = {best_val:.6f}")

    if best_seq is None:
        best_seq = _uniform_distribution(n)

    # -----------------------------------------------------------------
    # 5.4 Long Adam run on the current best seed
    # -----------------------------------------------------------------
    if deadline - time.time() > 60.0:
        # Rough estimate of feasible iterations based on remaining wall‑clock time
        remaining = deadline - time.time()
        max_iters_long = int(min(400_000,
                                 max(100_000, remaining / 0.0005)))
        print(f"\n[Long Adam] max_iters={max_iters_long}")
        best_seq, best_val = _adam_optimize(best_seq, n, deadline,
                                            max_iters=max_iters_long,
                                            lr0=0.4,
                                            beta_start=4.0,
                                            beta_end=32.0)
        print(f"[Long Adam] final surrogate = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 5.5 Final polishing with exponentiated‑gradient
    # -----------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        max_iters_poly = int(min(4_000_000,
                                 max(500_000, time_left / 0.0002)))
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_seq, n, deadline,
            max_iters=max_iters_poly,
            eta0=12.0)
        if polished_val < best_val:
            best_seq = polished_seq
            best_val = polished_val
            print(f"[Polish] improved surrogate = {best_val:.6f}")
        else:
            print("[Polish] no improvement (best surrogate unchanged)")

    # -----------------------------------------------------------------
    # 5.6 Evaluate using the provided evaluate_sequence function
    # -----------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
