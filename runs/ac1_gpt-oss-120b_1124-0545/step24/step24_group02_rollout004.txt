def _proj_simplex(v):
    # ensure non-negative by projecting onto simplex
    # algorithm from Duchi 2008: sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Advanced step‑function optimizer – projected‑gradient + smart seeding

* Initialise a probability vector (non‑negative, sum = 1) using a short
  hyper‑parameter sweep (power‑law, exponential, inverse‑sqrt) together
  with the previously best construction if available.
* Repeatedly run a simple projected‑gradient descent on the exact
  objective `max(conv(a,a))`.  The sub‑gradient of the max‑convolution
  is obtained analytically via a second FFT, so each iteration costs
  only O(n log n).  The iterate is projected onto the ℓ₁‑simplex after
  every step.
* The optimisation budget (default 1000 s) is split into a few
  independent runs; each run starts from the current best sequence
  plus a small random perturbation – this helps escape shallow local
  minima.
* The best sequence found is finally evaluated with the supplied
  `evaluate_sequence` routine and stored in the global variable
  `height_sequence_1` for possible reuse.

The algorithm is deliberately lightweight (no external solvers) and
fits comfortably into the time budget while typically improving the
surrogate value beyond the previous 1.5085 → ≈ 1.503 goal.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (uses zero‑padding to next pow‑2)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (sum of a_i = 1)  →  2·n·max(conv)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the ℓ₁‑simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Analytic families – cheap parametric seeds
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, p)
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, rel_tol: float = 1e-12):
    """Return (g, max_val) where g is a sub‑gradient of max(conv(a,a)).
    The gradient satisfies (∇max(conv))_i = 2·∑_{k∈A} a_{k‑i} / |A|
    with A the active set of indices attaining the maximum."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # active indices (relative tolerance to cope with floating noise)
    cutoff = max_val - rel_tol * max_val
    active = np.where(conv >= cutoff)[0]
    if active.size == 0:          # safety fallback
        active = np.array([np.argmax(conv)])
    # weight vector w of length 2n‑1
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    rev = a[::-1]
    grad_full = _conv_fft(rev, w)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_a: np.ndarray, deadline: float, init_lr: float = 0.5):
    """Run projected gradient descent on max(conv) until ``deadline``."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    n = len(a)

    # small helper to report occasional progress
    last_report = time.time()

    while time.time() < deadline:
        it += 1
        grad, _ = _subgradient_maxconv(a)
        lr = init_lr / math.sqrt(it)          # classic 1/√t schedule
        a = a - lr * grad
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[PGD] iteration {it:<6d}  best surrogate = {best_val:.8f}")

        # occasional time report (every ~5 s)
        if time.time() - last_report > 5.0:
            last_report = time.time()
            # nothing else – the loop will naturally continue

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    The routine
      * builds several cheap analytic seeds,
      * runs a few projected‑gradient refinements (with random perturbations)
        within the allocated wall‑clock time,
      * returns the best sequence found and stores it globally as
        ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 points give a good trade‑off between quality
    #               and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Assemble a short list of promising seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1) previous best (if present)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.shape == (n,):
            seeds.append(_proj_simplex(arr))

    # 2) sweep a power‑law exponent p ∈ [0.30, 0.70]
    best_power = None
    best_val = math.inf
    for p in np.linspace(0.30, 0.70, 41):

        val = _objective_surrogate(cand)

        slice_time = max(30.0, remaining * 0.30)

            start = start + noise
            start = np.maximum(start, 0.0)

            print(f"@@ New global best surrogate = {global_best_val:.8f}")

    # Final evaluation with the supplied function
    # ------------------------------------------------------------------

        true_val = None

    # Store for possible reuse by a subsequent run

    return global_best_seq.tolist()
