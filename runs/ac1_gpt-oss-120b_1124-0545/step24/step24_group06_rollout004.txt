"""
Enhanced step‑function optimiser (2024‑11‑25).

The algorithm follows a multi‑phase pipeline:

1. **Diverse seed generation** – uniform, arcsine, triangular, spike,
   many Dirichlet draws and symmetric Beta(α,α) densities (α from 0.02 to 2).
   The previous best solution is also added as a seed.

2. **Adam on a smooth‑max surrogate** – optimisation in the
   log‑softmax space of a soft‑max approximation of the L∞‑norm of the
   self‑convolution.  The β‑schedule is generous and the best exact
   surrogate seen is kept.

3. **Exact exponentiated‑gradient polishing** – mirror‑descent on the
   true max‑convolution objective using the sub‑gradient at the
   currently maximal index.

4. **Simulated‑annealing local refinement** – a fast incremental update
   of the full convolution is maintained.  Random moves transfer a
   small amount δ from a “source’’ index i (chosen with probability
   proportional to its contribution to the current maximal convolution)
   to a random “target’’ index j.  Moves that improve the objective are
   always accepted; uphill moves are accepted with probability
   exp(−Δ/T) where the temperature T is gradually cooled.  The schedule
   periodically re‑computes the convolution from scratch to avoid
   round‑off drift and restarts from a fresh random seed if progress
   stalls.

5. **Final evaluation** – the best sequence found is evaluated with the
   supplied ``evaluate_sequence`` and stored as ``height_sequence_1``
   for possible reuse in a subsequent run.

The code respects the 1000 s time budget, uses only NumPy (no external
solvers), and works for any problem size *n* (default 4000).  All
numbers are non‑negative and the sequence is automatically normalised
to sum to one – the objective is invariant to the overall scale.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a

# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # increasingly aggressive β values (smooth‑max temperature)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner iterations for this β (longer while we have time)
        max_iter_beta = 12000 if (deadline - time.time()) > 30 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val

# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Simulated‑annealing incremental local refinement
# ----------------------------------------------------------------------


def _simulated_annealing_local_search(start_seq: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      rng: np.random.Generator,
                                      seed_pool: list[np.ndarray]) -> tuple[np.ndarray, float]:
    """
    Fast SA walk on the simplex.
    Moves transfer a small amount δ from a source index i (chosen
    preferentially by its contribution to the current maximal convolution)
    to a random target index j.  The full convolution is updated incrementally.
    The temperature is geometrically cooled; occasional restarts from a
    fresh seed avoid stagnation.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)

    best_a = a.copy()
    best_val = cur_val

    # temperature parameters (relative to current objective)
    init_T = cur_val * 0.02
    T = init_T
    min_T = cur_val * 1e-6

    sigma = 0.07               # controls typical move size
    attempts = 0
    accepts = 0
    no_improve = 0
    last_restart = 0

    # pre‑compute full index array for fast selection of source indices
    indices = np.arange(n, dtype=int)

    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # 1) pick source i with probability proportional to its
        #    contribution to the current maximal convolution.
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))
        # contribution_i = a[i] * a[k_max - i]  (where defined)
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        valid_i = np.arange(start_i, end_i + 1, dtype=int)
        partner = k_max - valid_i
        contrib = a[valid_i] * a[partner]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            i = rng.integers(0, n)
        else:
            probs = contrib / total_contrib
            i = rng.choice(valid_i, p=probs)

        if a[i] <= 1e-15:            # nothing to move – skip this attempt
            continue

        # --------------------------------------------------------------
        # 2) pick a distinct target j uniformly
        # --------------------------------------------------------------
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # --------------------------------------------------------------
        # 3) proposal size δ
        # --------------------------------------------------------------
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # 4) incremental convolution update
        # --------------------------------------------------------------
        cand_conv = conv.copy()
        # linear contribution of moving δ from i to j
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        # diagonal / cross‑term corrections
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()
        delta_val = cand_val - cur_val

        # --------------------------------------------------------------
        # 5) Acceptance test (SA)
        # --------------------------------------------------------------
        accept = False
        if delta_val <= 0.0:
            accept = True
        else:
            # probability exp(-Δ/T)  (guard against overflow)
            prob = math.exp(-delta_val / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            # apply move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepts += 1
            no_improve = 0

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 6) Schedule updates
        # --------------------------------------------------------------
        if attempts % 3000 == 0:
            # geometric cooling
            T = max(min_T, T * 0.93)
            # slowly shrink sigma (but keep a floor)
            sigma = max(0.003, sigma * 0.97)

        # recompute exact convolution periodically to avoid drift
        if accepts % 5000 == 0 and accepts > 0:
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)

        # restart if we have been stuck for a while
        if no_improve >= 120000 and (attempts - last_restart) > 200000:
            # pick a fresh seed
            fresh = rng.choice(seed_pool)
            a = fresh.copy()
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            sigma = 0.07
            T = init_T
            no_improve = 0
            last_restart = attempts
            print(f"[SA] restart #{(last_restart // 200000)} after {attempts} attempts, "
                  f"best_val={best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found (as a plain list) and stores it in
    the global variable ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for final eval

    # problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load previously found solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a rich seed pool (all vectors will be normalised to sum=1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with several concentration parameters (including very spiky)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – a wide range of α values
    beta_alphas = [
        0.02, 0.03, 0.04, 0.05, 0.07, 0.10, 0.15, 0.20,
        0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.60,
        0.80, 1.00, 1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Seed evaluation – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()          # safety normalisation
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth‑max) ---")
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0            # keep a few seconds for final phases
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing intensive local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        local_deadline = deadline
        seq_loc, val_loc = _simulated_annealing_local_search(best_seq,
                                                             n,
                                                             local_deadline,
                                                             rng=rng,
                                                             seed_pool=seeds)
        print(f"[SA] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation (if any)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
