"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm
---------
1. **Parametric seeds** – generate families
   a_i ∝ ((i+δ)*(n‑δ‑i))⁻ᵍ   (δ≈0.5, γ≈0.5)   plus
   the classic arcsine, uniform and a few exponential decays.
   A small grid search over (γ,δ) yields the most promising shape.

2. **Smoothed max‑convolution descent** – minimise the smooth surrogate
   F_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]) using Adam on the log‑softmax
   parameters.  The temperature β is raised from 0.5 to 12, making the
   surrogate converge to the true max‑convolution.

3. **Exact max‑convolution polishing** – classic exponentiated‑gradient
   (mirror descent) on the genuine max‑convolution objective.

4. **Local random refinements** – cheap Monte‑Carlo moves that transfer a
   tiny amount of mass between two coordinates, always keeping the
   vector on the probability simplex.  Whenever the surrogate improves,
   the move is kept; otherwise it is discarded.

All phases respect the 1000 s budget and constantly check the deadline.
The best sequence found is stored in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Elementary helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used for optimisation:
        2·n·max(conv(seq,seq))
    (sum(seq) = 1, therefore this equals evaluate_sequence(seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Simple exponential tail a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _shape_gamma_delta(n: int, gamma: float, delta: float) -> np.ndarray:
    """
    Generalised arcsine‑like shape:
          a_i ∝ ((i+δ)*(n‑δ‑i))⁻ᵍ .
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + delta) * (n - delta - i)
    # avoid division by zero – the denominator is positive for 0<δ<n
    a = np.where(denom > 0.0, denom ** (-gamma), 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution surrogate
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
        (1/β)·log Σ_k exp(β·conv(a,a)[k])
    w.r.t. the log‑softmax parameters `log_a`.  Returns ∂F/∂log_a.
    """
    a = _softmax(log_a)                      # a lies on the simplex
    conv = _conv_fft(a, a)                    # length 2n‑1

    # softmax weights over convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                              # shape (2n‑1,)

    # g_a = 2 Σ_k w_k·shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)                # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]       # central n entries

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 400_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed max‑convolution objective.
    Returns the best probability vector encountered and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam updates (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) polishing on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 6_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient pass on the true objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (hard max)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local random search (tiny mass transfers between two coordinates)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         step_scale: float = 0.02,
                         rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Very cheap Monte‑Carlo refinement: repeatedly move a
    small fraction of mass from one index to another.  The new
    candidate is kept only if it improves the surrogate.
    """
    if rng is None:
        rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)

        src_mass = cur_seq[i]
        if src_mass <= 0.0:
            continue

        # move a tiny proportion of src_mass
        delta = rng.uniform(0.0, step_scale) * src_mass
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # re‑normalise (does not change the objective but keeps numerical stability)
        new_seq = np.maximum(new_seq, 0.0)
        s = new_seq.sum()
        if s <= 0.0:
            continue
        new_seq /= s

        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq.copy()
            best_val = val
            cur_seq = new_seq.copy()
            cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine performs:
      * a quick parametric grid search,
      * a long Adam optimisation on a smoothed max‑convolution,
      * an exponentiated‑gradient polishing pass,
      * a cheap random‑move local search.
    The best sequence found (as a plain Python list) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 4000                                        # length of the vector

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of seeds (grid + analytic families)
    # ------------------------------------------------------------------
    candidates = []

    # a) classic arcsine (the baseline)
    candidates.append(_arcsine_distribution(n))

    # b) uniform
    candidates.append(_uniform(n))

    # c) several exponential decays
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    # d) parametric shape a_i ∝ ((i+δ)(n−δ−i))⁻ᵍ
    gammas = np.linspace(0.43, 0.57, 15)            # around the arcsine exponent 0.5
    deltas = np.linspace(0.30, 0.90, 13)            # shift away from the exact 0.5
    for gamma in gammas:
        for delta in deltas:
            cand = _shape_gamma_delta(n, gamma, delta)
            candidates.append(cand)

    # e) convex combinations of arcsine and uniform (coarse sweep)
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for alpha in np.linspace(0.0, 1.0, 11):
        cand = alpha * arcs + (1.0 - alpha) * uni
        candidates.append(cand / cand.sum())

    # f) a few random Dirichlet draws (lightweight)
    for _ in range(6):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    # g) the previous best, if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(np.maximum(arr, 0.0) / np.maximum(arr.sum(), 1e-30))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Quick surrogate evaluation to pick the most promising seed
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = float("inf")
    for idx, cand in enumerate(candidates, start=1):
        if time.time() > deadline - 2.0:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand = cand / s
        val = _objective_surrogate(cand)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = cand
        # occasional progress print
        if idx % 30 == 0:
            print(f"[Grid] evaluated {idx}/{len(candidates)} candidates – "
                  f"current best surrogate = {best_seed_val:.6f}")

    if best_seed is None:
        best_seed = _uniform(n)

    # ------------------------------------------------------------------
    # 3) Smoothed Adam optimisation on the chosen seed
    # ------------------------------------------------------------------
    print("\n--- Adam optimisation (smooth max) ---")
    seq_adam, val_adam = _adam_softmax_opt(
        best_seed,
        n,
        deadline,
        max_iters=400_000,
        lr0=0.7,
        beta_start=0.5,
        beta_end=12.0,
    )
    print(f"Adam finished – surrogate = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    print("\n--- Exponentiated‑gradient polishing ---")
    seq_polish, val_polish = _exp_gradient_opt(
        seq_adam,
        n,
        deadline,
        max_iters=6_000_000,
        eta0=12.0,
    )
    print(f"Polish finished – surrogate = {val_polish:.6f}")

    # ------------------------------------------------------------------
    # 5) Tiny‑mass random local search
    # ------------------------------------------------------------------
    print("\n--- Local random refinements ---")
    seq_local, val_local = _local_random_search(
        seq_polish,
        n,
        deadline,
        max_iters=300_000,
        step_scale=0.02,
        rng=rng,
    )
    print(f"Local search finished – surrogate = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Choose the best among the three refined candidates
    # ------------------------------------------------------------------
    final_seq = seq_local
    final_val = val_local
    for candidate, cand_val in [(seq_adam, val_adam),
                                (seq_polish, val_polish)]:
        if cand_val < final_val:
            final_seq = candidate
            final_val = cand_val

    # ------------------------------------------------------------------
    # 6) Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {final_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store globally for continuation runs
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
