    deadline = start + budget_s - 2.0

"""
Step‑function optimizer for the “max‑convolution” objective.

The script builds a very diverse set of analytic seeds, evaluates a fast
FFT‑based surrogate (2·n·max(conv)) and then performs a multi‑stage non‑convex
search on the probability simplex:

1. A short exponentiated‑gradient (EG) pass on the exact max‑convolution.
2. A smooth‑max L‑BFGS phase with a homotopy over the smoothing parameter β.
3. A long “flattening’’ gradient descent that minimises the variance of the
   self‑convolution (making it as flat as possible).
4. A simulated‑annealing polish.

The routine explores many seeds (top‑k ≈ 7) with a quick EG refinement,
keeps the few most promising and invests the bulk of the remaining time in a
full optimisation of each.  Finally a high‑β L‑BFGS sweep is applied to push
the smooth surrogate as close as possible to the true max‑convolution.
The best discovered sequence is stored globally in ``height_sequence_1`` and
returned as a plain Python list.

All operations respect the supplied wall‑clock budget (default 1000 s) and
use only NumPy (SciPy is optional for the L‑BFGS stage).
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# optional SciPy imports – the code works without them
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ------------------------------------------------------------
# basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max giving a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# surrogate objective (exact for L1‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic seed families (all return L1‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ i^{‑p}·exp(‑λ·i)   (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (multi‑active)
# ------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient: average contributions from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Flattening gradient descent (minimise variance of the convolution)
# ------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom for normalised vectors
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Mirror‑descent on simplex via log‑softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    if _opt is None:
        # SciPy not available – return the start vector unchanged
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1_000_000_0.0,
        ]

    v = np.log(start_vec + 1e-30)   # work in log‑softmax space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Simulated‑annealing refinement (accept occasional worsenings)
# ------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Full optimisation pipeline for a single seed
# ------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=14.0,
        tol=1e-11,
    )

    # 2) smooth‑max L‑BFGS (more iterations per β)
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=6000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient (generous budget)
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=6_000_000,
            eta0=10.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=500_000,
            start_sigma=0.0008,
            T_start=6e-3,
            T_end=4e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ------------------------------------------------------------
# Helper: generate a rich seed set (including two‑parameter families)
# ------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    p_vals = np.linspace(0.3, 1.2, 10)
    lam_vals = np.linspace(0.0, 0.008, 8)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # mirrored versions
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include previously stored best (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # small random perturbations of the current best seed
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds

    """

    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: surrogate = {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Quick EG refinement on the top seeds (few hundred‑k iterations)
    # ------------------------------------------------------------------
    quick_refined = []

        if time.time() > quick_deadline:
            break
        seq, val = _exp_gradient_opt_multi(
            seed_vec,

    print("\nSelected seeds for deep optimisation (surrogate values):")

    # ------------------------------------------------------------------

        if time.time() > deadline:

        seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"  >>> Updated global best (deep) surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final high‑β L‑BFGS sweep (push surrogate as close as possible to true max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq = _lbfgs_smooth_optimize(

            max_iter_per_beta=8000,

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function (once) and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    return global_best_seq.tolist()
