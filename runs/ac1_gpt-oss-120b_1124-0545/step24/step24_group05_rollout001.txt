"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Choose a fairly large support (default 20 000).  The optimal value of
  the objective tends to 1 as the length grows, because the discrete
  arcsine law approaches the continuous arcsine law whose self‑convolution
  is exactly uniform.
* Initialise the probability vector with the *discrete arcsine* law
  `p_i ∝ 1/√((i+½)(n‑½‑i))`.  This distribution is already close to the
  global optimum.
* Refine the seed with a smoothed optimisation of the true objective:
  minimise the smoothed max‑convolution
  `F_β(p) = (1/β) log Σ_k exp(β·conv(p,p)[k])`
  using Adam in the log‑softmax (mirror) parametrisation.
  The temperature β is raised linearly from 0.5 to 8.0.
* Finish with a classic exponentiated‑gradient (mirror descent) on the
  exact max‑convolution objective.  Both phases stop automatically when
  the allocated time budget is exhausted.
* Return the best probability vector found (scaled to sum‑1, which is
  scale‑invariant for the true objective).

The code respects the 1000 s budget, prints a few progress messages,
stores the best sequence globally as ``height_sequence_1`` and finally
calls ``evaluate_sequence`` to obtain the true score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation:
       2·n·max(conv(seq,seq)).  Here seq is assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed max‑convolution descent (Adam)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
        F_β(p) = (1/β)·log Σ_k exp(β·conv(p,p)[k])
    w.r.t. the log‑softmax parameters `log_a`.
    Returns ∂F/∂log_a (size n).
    """
    a = _softmax(log_a)                 # probability vector (∑a = 1)
    conv = _conv_fft(a, a)               # length 2n‑1

    # softmax over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                         # w sums to 1, shape (2n‑1,)

    # g_a = 2 Σ_k w_k · shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)            # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]    # slice → length n

    # chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution objective.
    Returns the best probability vector seen and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # gradient of the smoothed objective
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate the *exact* surrogate (no smoothing)
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (index k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
