def _spike_mix(n, rng):
    base = _arcsine(n)
    k = rng.integers(5, 30) # number of spikes
    spikes = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    spikes[idx] = rng.random(k)
    spikes /= spikes.sum()
    mix = 0.8*base + 0.2*spikes
    mix /= mix.sum()
    return mix

"""
Improved step‑function optimiser.

The routine follows a multi‑stage scheme:

1️⃣ Build a very diverse set of seed probability vectors (uniform,
   arcsine, triangular, many symmetric Beta shapes, Dirichlet draws,
   Gaussian mixtures, β‑mixtures and spike‑mixtures).  All vectors are
   normalised to sum = 1.

2️⃣ Evaluate the surrogate  (2·n·max(conv(a,a))) on the seeds and keep the
   best K (default 5) candidates.

3️⃣ For every retained seed run a *full optimisation* consisting of:
   • Adam optimisation on a smooth‑max surrogate (many β‑steps,
     extended inner iterations).
   • Exponentiated‑gradient (mirror‑descent) on the true max‑convolution,
     with a larger initial step‑size and aggressive decay.
   • Aggressive multi‑scale stochastic local search that mixes
     small mass‑transfer moves, swaps, global permutations and occasional
     convex combinations with random seeds.

4️⃣ Keep the best sequence found over all phases, evaluate it with the
   provided ``evaluate_sequence`` and store it in the global variable
   ``height_sequence_1`` for possible reuse.

All helper functions are top‑level (no closures) and only NumPy,
time, math and random generators are used.  The algorithm respects the
overall time budget (≈ 1000 s) and always returns a valid non‑negative
step‑function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence must sum to 1 (probability vector).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(alpha,alpha) on {0,…,n‑1}.  The continuous density is
    proportional to x^{α‑1}(1‑x)^{α‑1}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _beta_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Mixture of two symmetric Beta densities with random parameters.
    """
    a1 = rng.uniform(0.05, 0.9)
    a2 = rng.uniform(0.9, 3.0)
    w = rng.random()
    b1 = _beta_distribution(n, a1)
    b2 = _beta_distribution(n, a2)
    a = w * b1 + (1.0 - w) * b2
    a /= a.sum()
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps (truncated to the interval).
    The resulting vector is normalised to sum 1.
    """
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _spike_mix(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Combine a smooth baseline (arcsine) with a few random spikes.
    """
    base = _arcsine(n)
    k = rng.integers(5, 30)                     # number of spikes
    spikes = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    spikes[idx] = rng.random(k) + 1e-6         # ensure positivity
    spikes /= spikes.sum()
    mix = 0.8 * base + 0.2 * spikes
    mix /= mix.sum()
    return mix

def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    Uses a dense schedule of smooth‑max β‑values and long inner loops.
    Returns the best probability vector (sum = 1) seen and its exact
    surrogate value.
    """
    logit = np.log(start_seq + 1e-30)          # initialise logits

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # extended β‑schedule (more fine‑grained at the high‑end)
    beta_schedule = [
        1.0, 3.0, 8.0, 20.0, 50.0,
        120.0, 300.0, 800.0, 2000.0,
        5000.0, 12000.0, 30000.0,
        80000.0, 200000.0, 500000.0,
        1_200_000.0, 2_500_000.0, 5_000_000.0,
        1_000_0000.0, 2_000_0000.0, 5_000_0000.0,
        10_000_0000.0
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # allocate inner iterations based on remaining time
        remaining = deadline - time.time()
        inner_iters = 16000 if remaining > 30 else 6000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient in log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.13
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 800 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val

# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Uses a larger initial step‑size and decay ~1/√t.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a (sum of contributions from all i)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Aggressive stochastic local search (mass transfer, swaps, perms,
#   and occasional convex mixing with random seeds)
# ----------------------------------------------------------------------


def _local_multi_search(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        seed_pool: list[np.ndarray]) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk on the simplex.
    Alternates three kinds of proposals:
      * geometric (mass‑transfer) moves,
      * swaps,
      * random permutation,
      * convex combination with a random seed from `seed_pool`.
    Only strictly improving moves are kept.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    sigma_start = 0.18          # relative size of mass‑transfer steps
    sigma_end = 0.003
    sigma = sigma_start
    decay = 0.96                # geometric decay per block
    block_iters = 5000          # attempts per sigma value
    mix_interval = 1200         # every N iterations try mixing with a seed

    it_total = 0

    while time.time() < deadline:
        # ---- mass‑transfer block ----
        for _ in range(block_iters):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            # guard against round‑off negativity
            if new[i] < -1e-15:
                continue
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- swap block ----
        for _ in range(block_iters // 2):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- occasional global permutation ----
        if rng.random() < 0.1:   # roughly one permutation every 10 blocks
            perm = rng.permutation(n)
            new = cur[perm]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- occasional convex mixing with a random seed ----
        if rng.random() < 0.12:
            seed_vec = rng.choice(seed_pool)
            t = rng.uniform(0.05, 0.25)          # mixing proportion
            mixed = t * cur + (1.0 - t) * seed_vec
            mixed /= mixed.sum()
            val = _surrogate(mixed)
            if val < cur_val:
                cur, cur_val = mixed, val

        it_total += 1

        # decay sigma
        sigma = max(sigma_end, sigma * decay)

        # after sigma reaches its floor, reduce block size to focus on swaps
        if sigma <= sigma_end + 1e-12:

    return cur, cur_val

def _full_optimize(start_seq: np.ndarray,
