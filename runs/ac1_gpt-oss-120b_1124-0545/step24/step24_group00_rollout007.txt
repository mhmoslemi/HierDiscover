def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    n = 4000

    # initializations
    inits = []
    # previous best
    prev = globals().get('height_sequence_1')
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        arr = np.maximum(np.asarray(prev, dtype=np.float64), 0.0)
        arr = _normalize(arr)
        inits.append(arr)

    # uniform
    inits.append(np.full(n, 1.0/n))
    # exponential
    for rate in [0.001, 0.0025, 0.004]:
        inits.append(_exp_decay(n, rate))

    # Dirichlet
    for alpha in [0.2, 0.5, 0.8, 1.0]:
        vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        vec = _normalize(vec)
        inits.append(vec)

    # unique inits as before
    uniq_inits = []
    seen = set()
    for v in inits:
        h = hash(v.tobytes())
        if h not in seen:
            uniq_inits.append(v)
            seen.add(h)

    best_seq = None
    best_val = float('inf')

    # Stage 1: Screening with moderate time per init
    print("[Stage 1] Screening init candidates")
    per_init_budget = max(0.0, (deadline - time.time()) * 0.2 / len(uniq_inits))
    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        sub_deadline = time.time() + per_init_budget
        # Adam
        seq_a, val_a = _adam_opt(start_seq=init, n=n, deadline=sub_deadline, max_iters=200_000, beta_end=5e7, alpha=0.4)
        # EG
        seq_e, val_e = _exp_gradient_opt(start_seq=seq_a, n=n, deadline=sub_deadline, max_iters=5_000_000, eta0=5.0)
        if val_e < best_val:
            best_val = val_e
            best_seq = seq_e.copy()
            print(f"  New best after init {idx+1}: surrogate={best_val:.6f}")

    # Stage 2: Long EG refinement on best so far
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Long EG refinement")
        # allocate majority of remaining time: ~60%
        time_left = deadline - time.time()
        sub_deadline = time.time() + time_left * 0.6
        seq_eg, val_eg = _exp_gradient_opt(start_seq=best_seq, n=n, deadline=sub_deadline, max_iters=20_000_000, eta0=3.0)

"""
Improved step‑function optimizer
================================

The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

for a non‑negative vector ``a`` of length ``n`` (default 4000).  The
objective is invariant to scaling, therefore we work with probability
vectors (sum = 1) and minimise the surrogate

    S(a) = 2·n·max(conv(a,a))

The optimiser consists of four phases

1. **Diverse initialisations** – uniform, exponential decays, several
   Dirichlet draws and (if available) the previous best sequence.
2. **Screening** – a short Adam optimisation on a smoothed version of
   the max‑convolution, followed by a modest exact exponentiated‑gradient
   (EG) run.  The best candidate is kept.
3. **Long EG refinement** – mirror‑descent on the true max‑convolution
   with a decaying step size.  This phase consumes the bulk of the
   time‑budget.
4. **Aggressive hill‑climb** – a simulated‑annealing loop that favours
   a targeted “mass‑shift’’ move which directly reduces the most
   offending convolution entry.  Finally a few short Adam polishing
   runs are performed.

All phases respect the global deadline (default 1000 s) and the best
sequence found is stored in ``height_sequence_1`` for possible
continuation runs.  The routine prints progress and returns the best
sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return _normalize(a)


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    v_local lives in log‑softmax space; a = softmax(v_local) has sum=1.
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                    # length = 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp smoothing of the max
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each conv entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 5e7,
              alpha: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 15_000_000,
                     eta0: float = 4.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.
    The iterate stays on the simplex via a log‑space representation.
    Step size decays as η0 / sqrt(t).
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types.
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 12_000_000) -> tuple[np.ndarray, float]:
    """
    Aggressive local search.  Three move types:
      * swap two positions,
      * add Gaussian noise,
      * targeted mass‑shift from the dominant pair of the current max.
    The mass‑shift move is used most often (≈70 %).
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling (more gentle than exponential)
        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        # ----- swap ----------------------------------------------------
        if move_type < 0.20:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        # ----- Gaussian perturb ----------------------------------------
        elif move_type < 0.40:
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        # ----- targeted mass‑shift ------------------------------------
        else:
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))

            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]

            if contrib.size == 0:
                continue
            # donor = index with largest contribution
            donor_rel = int(np.argmax(contrib))
            donor = int(idx_i[donor_rel])

            # receiver = random position (different from donor)
            receiver = int(rng.integers(0, n))
            if receiver == donor:
                continue

            # fraction to move – decays with iteration count
            frac = 0.05 * (1.0 - it / max_iters) + 0.001
            delta = cur_seq[donor] * frac
            if delta <= 0.0:
                continue

            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta

            # renormalise (numerical drift)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """

    start_time = time.time()

    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    n = 4000

    inits = []

        arr = np.asarray(prev, dtype=np.float64)

        inits.append(_normalize(arr))

    # uniform
    inits.append(np.full(n, 1.0 / n))

    # exponential decays
    for rate in (0.001, 0.0025, 0.004):

    # Dirichlet draws with various concentration parameters

        vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        inits.append(_normalize(vec))

    # keep only unique initial vectors (hash based)
    uniq_inits = []
    seen_hashes = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen_hashes:
            uniq_inits.append(vec)

    print(f"[Setup] {len(uniq_inits)} unique initial vectors prepared.")

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    print("[Stage 1] Screening candidates")

    per_init_budget = screening_budget / max(1, len(uniq_inits))

    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break

                                         deadline=sub_deadline,

        if val_e < best_val:
            best_val = val_e
            best_seq = seq_e.copy()

    # ------------------------------------------------------------------
    # 3. Long EG refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Long EG refinement")
        remaining = deadline - time.time()
        # give ~60 % of the remaining time to this phase

                                           deadline=sub_deadline,
                                           max_iters=30_000_000,

        if val_eg < best_val:
