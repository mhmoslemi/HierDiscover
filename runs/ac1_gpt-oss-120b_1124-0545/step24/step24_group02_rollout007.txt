def project_simplex(v):
    # v: numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Advanced step‑function optimizer (Projected Gradient + smooth L‑BFGS)

Main ideas
----------
* Work directly on the ℓ₁‑simplex (∑a_i = 1, a_i ≥ 0) using a Euclidean
  projected‑gradient descent (PGD).  The objective is
       2·n·max(conv(a,a))               (the true surrogate of evaluate_sequence)
  and the sub‑gradient of max(conv) is obtained by convolving the reversed
  sequence with a uniform weight on the *active* indices (those whose
  convolution value is close to the current maximum).  A simple momentum
  term (β≈0.9) speeds up convergence.

* The active‑set tolerance ε starts relatively large (≈0.2) so many
  indices contribute to the sub‑gradient, which makes the early phase
  very exploratory.  ε is multiplied by 0.9995 each iteration – it
  shrinks slowly to ≈10⁻¹², at which point the method behaves like a
  true sub‑gradient descent on the hard max.

* After the cheap PGD phase we invoke a “soft‑max” L‑BFGS‑B optimiser
  (identical to the one used in the previous solution) to polish the
  result on a smooth surrogate of the max‑convolution.  Finally we run
  a short PGD polish.

* Several diverse seeds are tried (previous best, inverse‑sqrt shape,
  sparse Dirichlet draws, uniform, perturbed versions).  The available
  time budget is split dynamically among the seeds; the best sequence
  found over all runs is returned and stored in the global variable
  ``height_sequence_1`` for possible reuse.

The whole routine respects the 1000 s wall‑clock limit and prints a
few status messages so you can see the progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution, sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, ∑x = 1}.
    Returns a new array (does not modify v).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _pgd_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10 ** 9,
    alpha0: float = 1.0,
    beta_mom: float = 0.9,
    eps_start: float = 0.2,
    eps_decay: float = 0.9995,
    eps_min: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) gradient descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_a.copy()
    # assure exact ℓ1‑normalisation
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)          # momentum buffer
    eps = eps_start                # active‑set tolerance
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # full convolution (needed for max and active set)
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set (relative tolerance)
        thresh = max_val * (1.0 - eps)
        active = np.where(conv >= thresh)[0]
        if active.size == 0:
            # fallback – should never happen
            active = np.array([int(np.argmax(conv))])

        # build uniform weight on the active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # sub‑gradient of the max‑convolution:
        #   g_i = 2 * (rev_a * w)[i]   (central slice)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)                # length = n + (2n‑1) – 1 = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # central n entries

        # momentum update
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # step size (standard 1/√t schedule)
        alpha = alpha0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = a - alpha * m
        a = _project_simplex(a)

        # evaluate surrogate
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

        # decay active‑set tolerance
        eps = max(eps * eps_decay, eps_min)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser (identical to the previous solution)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 2500,
):
    """
    Minimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns the best ℓ₁‑normalised sequence seen and its exact surrogate.
    """
    if _opt is None:
        # fall back to a tiny PGD if SciPy is missing
        return _pgd_optimize(start_a, n, deadline)

    # initialise logits for the soft‑max parametrisation
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max (see paper for derivation)
            w = exp_beta / sum_exp                     # soft‑max weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]     # gradient w.r.t. a

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain‑rule through soft‑max
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


def _default_beta_schedule():
    """A generous schedule of β values for the smoothed max."""
    return [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine uses several seeds, a fast
    projected‑gradient descent on the true max‑convolution, optional
    smoothed L‑BFGS polishing, and a final PGD polish.  The best
    sequence is stored in ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best (if any) to use as a seed
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    base_prev = None
    if isinstance(previous, list) and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.sum() > 0:
            base_prev = arr / arr.sum()

    # --------------------------------------------------------------
    # Build a diverse list of seeds
    # --------------------------------------------------------------
    seeds = []

    if base_prev is not None:
        seeds.append(base_prev)                                   # previous best

    seeds.append(_inverse_sqrt(n))                                # classic 1/√i
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))        # very sparse
    seeds.append(_random_dirichlet(n, conc=0.01, rng=rng))         # moderately sparse
    seeds.append(rng.random(n) + 1e-12)                           # plain random uniform

    if base_prev is not None:
        pert = base_prev + 0.003 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        if pert.sum() > 0:
            seeds.append(pert / pert.sum())                       # perturbed previous

    # normalise all seeds (guard against zero‑sum)
    clean_seeds = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        tot = s.sum()
        if tot > 0:
            clean_seeds.append(s / tot)
    seeds = clean_seeds

    # --------------------------------------------------------------
    # Quick surrogate evaluation of seeds (exact max‑convolution)
    # --------------------------------------------------------------
    seed_vals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in seed_vals[:max(1, len(seed_vals) // 2)]]

    print("\n--- Seed evaluation (exact surrogate) ---")
    for i, (_, val) in enumerate(seed_vals[:len(top_seeds)], 1):
        print(f" Seed #{i}: surrogate = {val:.8f}")

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Optimise each seed – the time budget is split dynamically
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly 20 % of the remaining time to this seed (min 8 s)
        share = max(8.0, remaining * 0.20)
        sub_deadline = time.time() + share
        print(f"\n[Seed {si}/{len(top_seeds)}] optimisation budget ≈ {share:.1f}s")

        # 1) fast PGD optimisation
        seq_pgd, val_pgd = _pgd_optimize(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            max_iters=500_000,
            alpha0=1.0,
            beta_mom=0.9,
        )
        print(f"   → PGD surrogate = {val_pgd:.8f}")

        # 2) optional L‑BFGS smooth polishing (if time permits)
        if time.time() < sub_deadline:
            seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
                start_a=seq_pgd,
                n=n,
                deadline=sub_deadline,
                beta_schedule=_default_beta_schedule(),
                max_iter_per_beta=1500,
            )
            print(f"   → L‑BFGS surrogate = {val_lbfgs:.8f}")
            if val_lbfgs < val_pgd:
                seq_pgd, val_pgd = seq_lbfgs, val_lbfgs

        # 3) short final PGD polish (few 10⁵ iters)
        if time.time() < sub_deadline:
            seq_final, val_final = _pgd_optimize(
                start_a=seq_pgd,
                n=n,
                deadline=sub_deadline,
                max_iters=200_000,
                alpha0=0.8,
                beta_mom=0.9,
            )
            print(f"   → PGD fine‑polish surrogate = {val_final:.8f}")
            if val_final < val_pgd:
                seq_pgd, val_pgd = seq_final, val_final

        # keep the best across seeds
        if val_pgd < global_best_val:
            global_best_val = val_pgd
            global_best_seq = seq_pgd.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing on the current global best
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final intensive PGD polishing on global best ---")
        final_seq, final_val = _pgd_optimize(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            alpha0=0.6,
            beta_mom=0.9,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
