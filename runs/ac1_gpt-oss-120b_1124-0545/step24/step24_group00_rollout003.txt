def proj_simplex(v):
    # v can be any real vector
    # algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved optimizer for the step‑function “max‑convolution” attack.

The objective (after normalising sum(a)=1) is

    Obj(a) = 2·n·max(conv(a,a))

where conv(a,a) is the linear convolution of a with itself.
The code builds a diverse pool of initial candidates (previous best,
exponential decays, Dirichlet draws, a Sidon‑set construction, uniform)
and runs several complementary optimisation phases:

1. **Screening** – short Adam runs on a log‑softmax
   parameterisation together with a smoothed‑max (log‑sum‑exp)
   surrogate, followed by a short exact exponentiated‑gradient (EG)
   pass.

2. **Long EG refinement** – a prolonged EG optimisation that works on the
   true max‑convolution sub‑gradient.

3. **Projected Gradient Descent (PGD)** – an Euclidean‑projection
   gradient descent that directly follows the sub‑gradient of the
   (nonsmooth) max‑convolution.  The step size decays as η₀/√t and the
   iterate is always projected onto the probability simplex.

4. **Hill‑climbing (simulated annealing)** – three move types
   (swap, Gaussian perturbation, targeted “mass‑shift”) with an
   exponential cooling schedule.

5. **Random‑restart Adam polishing** – a final few seconds of Adam on
   noisy variants of the current best.

All phases respect the global time budget (default 1000 s) and keep the
best sequence seen.  The final sequence is stored in the global variable
``height_sequence_1`` for possible continuation runs and returned as a
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (sum(seq)=1) → 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  Variable v lives in log‑softmax space; a = softmax(v)
    is a probability vector with ∑a = 1.
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              beta_start: float = 0.5,
              beta_end: float = 1e8,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 12_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.
    The iterate stays on the simplex via a log‑space representation (softmax).
    Step‑size decays as η₀ / √t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, sum(x)=1} (Duchi et al.)."""
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected Gradient Descent on the true max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 10_000_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Euclidean PGD on the true max‑convolution objective.
    Sub‑gradient is taken from all active maximal indices.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # pick all indices that are (almost) maximal
        active = np.where(conv >= max_val - 1e-12)[0]

        g = np.zeros_like(a)                # sub‑gradient of max(conv)
        for m in active:
            i_start = max(0, m - (n - 1))
            i_end = min(n - 1, m)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = m - idx_i
            g[idx_i] += 2.0 * a[rev_idx]   # ordered pairs

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        # evaluate every 100 iterations only (speed)
        if it % 100 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same moves as the reference code)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move that directly attacks the active convolution entry.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])     # index that contributes most
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = min(cur_seq[donor], 0.5 * cur_seq[donor]) * 0.1
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # problem size (fixed, but can be changed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of initial vectors
    # ------------------------------------------------------------------
    inits = []

    # a) reuse previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                inits.append(arr)
        except Exception:
            pass

    # b) exponential decays (several rates)
    for rate in (0.0008, 0.003, 0.01):
        inits.append(_exp_decay(n, rate=rate))

    # c) Dirichlet draws with various concentration parameters
    for alpha in (0.2, 0.5, 1.0):
        vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        vec = vec / vec.sum()
        inits.append(vec)

    # d) uniform distribution on a Singer Sidon set (size 62)
    q = 61                                   # prime
    N = q * q + q + 1                        # 3783 ≤ n
    sid_set = [(i * (q + 1) + i * i) % N for i in range(q + 1)]
    sid_vec = np.zeros(n, dtype=np.float64)
    sid_vec[sid_set] = 1.0
    sid_vec = sid_vec / sid_vec.sum()
    inits.append(sid_vec)

    # e) plain uniform vector (for completeness)
    inits.append(np.full(n, 1.0 / n))

    # retain only unique initial vectors (hashing the raw bytes)
    uniq_inits = []
    seen_hashes = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen_hashes:
            uniq_inits.append(vec)
            seen_hashes.add(h)

    # ------------------------------------------------------------------
    # 2. Short screening phase – Adam + EG on each candidate
    # ------------------------------------------------------------------
    print("[Stage 1] Screening candidates")
    best_seq = None
    best_val = float("inf")
    per_candidate_time = 25.0   # seconds allocated per candidate

    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        sub_deadline = time.time() + per_candidate_time
        print(f"  candidate {idx+1}/{len(uniq_inits)}")

        seq_a, _ = _adam_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            max_iters=200_000,
            beta_end=5e7,
            alpha=0.35,
        )
        seq_e, val_e = _exp_gradient_opt(
            start_seq=seq_a,
            n=n,
            deadline=sub_deadline,
            max_iters=5_000_000,
            eta0=9.0,
        )
        if val_e < best_val:
            best_val, best_seq = val_e, seq_e.copy()
            print(f"    ** new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Long exact EG refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Long EG refinement")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=20_000_000,
            eta0=4.0,
        )
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"    improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected Gradient Descent on true max‑convolution
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] Projected Gradient Descent")
        seq_pg, val_pg = _pgd_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=8_000_000,
            eta0=0.8,
        )
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()
            print(f"    PGD improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Aggressive hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 4] Hill‑climb (simulated annealing)")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=7_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"    HC improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing with short random‑restart Adam passes
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 5] Random‑restart Adam polishing")
        noisy = best_seq + 0.006 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy /= s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=120_000,
            beta_end=8e7,
            alpha=0.30,
        )
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print(f"    polishing improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
