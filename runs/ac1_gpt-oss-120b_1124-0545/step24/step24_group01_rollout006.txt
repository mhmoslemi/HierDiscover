"""
Hybrid search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm proceeds in four stages:

1.  **Diverse seeds** – generate many probability vectors:
    uniform, arcsine, triangular, several symmetric Beta(α,α) PDFs,
    Dirichlet draws with different concentrations and mixtures of the
    above.  All are normalised to sum = 1.

2.  **Mirror‑descent (exponentiated‑gradient)** – starting from the best
    seeds, optimise the surrogate 2·n·max(conv(a,a)) on the simplex.
    The update is performed in log‑space (relative‑entropy mirror) with
    Adam‑style momentum and a decreasing step size.  The best iterate
    seen for each seed is kept.

3.  **Greedy refinement** – repeatedly move a small fraction of mass
    from the coordinate that contributes most to the active
    convolution index towards the coordinate that contributes least.
    Only improving moves are retained.

4.  **Local stochastic walk** – a final random walk on the simplex to
    escape any remaining shallow minima.

All stages respect the overall time budget (default 1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` and returned
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used by ``evaluate_sequence`` when ``seq`` sums to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density (β(½,½))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _exp_gradient_momentum(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using exponentiated‑gradient with
    Adam‑style momentum.  ``start_seq`` is assumed to sum to 1.

    Returns the best sequence encountered and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    # log‑coordinates (mirror variables)
    logit = np.log(a + 1e-30)
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # mirror‑gradient (log‑space) compatible with KL‑divergence
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # momentum update (Adam‑style, no bias‑correction for speed)
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # update in log‑space and map back to simplex
        logit = logit - eta * mom
        a = _softmax(logit)

        # periodic exact check
        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # early exit if we already beat the target known from literature
        if best_val < 1.503:
            break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a tiny fraction of mass from the coordinate with the
    largest sub‑gradient to the one with the smallest.  Only improving moves
    are kept.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # active‑set sub‑gradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a  # continue from the improved point
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored globally as ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Load a previously stored solution (if it matches the requested size)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣ Assemble a diversified pool of seeds
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Symmetric Beta(α,α) with a fine grid of α values
    alphas = np.concatenate([
        np.linspace(0.20, 0.55, 15),   # U‑shaped region
        np.linspace(0.60, 1.30, 15),   # around arcsine
        np.linspace(1.40, 2.20, 10)    # bell‑shaped region
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of arcsine and uniform – a cheap way to explore the line
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * _uniform(n) + (1.0 - lam) * _arcsine(n)
        mix = mix / mix.sum()
        seeds.append(mix)

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣ Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(7, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Momentum exponentiated‑gradient optimisation (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed = remaining / top_k if top_k > 0 else 0.0
        print("\n--- Momentum EG optimisation (exact max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            per_deadline = time.time() + per_seed
            print(f"  EG run #{sidx} (time slice ≈ {per_deadline - time.time():.1f}s)")
            seq, val = _exp_gradient_momentum(seed_vec, per_deadline,
                                              eta0=4.0, beta_mom=0.9)
            print(f"   → surrogate after EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (EG)")

            # ------------------------------------------------------
            # 2️⃣‑b  Greedy refinement of the EG result
            # ------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline,
                                            attempts=80_000,
                                            delta_frac=0.015)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # --------------------------------------------------------------
    # 3️⃣ Repeated greedy passes (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Additional greedy refinement passes ---")
        while time.time() < deadline:
            seq2, val2 = _greedy_refine(best_seq, deadline,
                                        attempts=60_000,
                                        delta_frac=0.01)
            if val2 < best_val - 1e-8:
                best_val, best_seq = val2, seq2.copy()
                print(f"   >>> Improved to {best_val:.6f}")
            else:
                break

    # --------------------------------------------------------------
    # 4️⃣ Final random local walk (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final stochastic local search ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=300_000,
                                                sigma=0.012)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣ Official evaluation with ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
