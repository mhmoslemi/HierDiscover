"""
Improved search for a low‑value step‑function.

Algorithm Overview
------------------
1. **Seeding** – Start from a small pool of promising shapes:
   a discrete arcsine law (the analytically good shape), the previous
   best vector (if any) and a few random Dirichlet draws.

2. **Smoothed Adam** – Minimise the smooth approximation of
   `max(conv(a,a))` using Adam on the log‑softmax parameters.
   The temperature β is increased from 0.5 up to 64, which makes the
   surrogate converge to the true max while keeping the landscape
   smooth enough for fast progress.  The best simplex vector from all
   seeds is kept.

3. **Exponentiated‑gradient polishing** – Using the exact sub‑gradient
   of the max‑convolution (the active entry) we perform a very long
   mirror‑descent (AdaGrad‑style) pass.  The learning‑rate constant is
   enlarged (η₀=30) and the iteration budget is limited only by the
   remaining wall‑clock time.

4. **Random local search** – A cheap hill‑climbing / simulated‑annealing
   phase that transfers a tiny amount of mass between two random
   coordinates, keeps the simplex constraint by construction and
   accepts moves that strictly improve the surrogate.  This phase can
   eke out the last few percent of improvement.

5. **Final evaluation** – The obtained vector is passed to the
   provided `evaluate_sequence` function; the best vector is stored
   globally (`height_sequence_1`) for possible continuation runs.

The whole routine respects the 1000 s budget, stopping a few seconds
before the deadline so that the environment can safely terminate the
process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(N log N))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate that the optimiser minimises:
       2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s > 0:
        return a / s
    return np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (soft‑max temperature β)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k])
    with respect to the log‑softmax parameters log_a.
    Returns ∂F/∂log_a (size n).
    """
    a = _softmax(log_a)                 # probability vector (sum=1)
    conv = _conv_fft(a, a)               # length 2n‑1

    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                         # softmax weights over conv entries

    # g_a = 2 * (w * a) convolved with reversed a
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)           # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]   # slice → length n

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 64.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed max‑convolution.
    Returns the best simplex vector found and its (exact) surrogate value.
    """
    # initialise log‑probabilities
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update in the log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 6_000_000,
                     eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used for local search)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random local hill‑climbing / simulated‑annealing phase
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic refinement.  At each step a random amount of
    mass is transferred between two coordinates (keeping the sum
    exactly one).  Moves that strictly improve the surrogate are kept.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    no_improve = 0
    # scale of transferred mass – we decay it slowly so that we can
    # make fine adjustments later on.
    delta_factor = 0.05

    while time.time() < deadline and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a small fraction of mass from i to j
        delta = delta_factor * a[i]
        if delta > a[i]:
            delta = a[i]
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # avoid numerical drift (project back to simplex)
        a_prop = _project_simplex(a_prop)

        val = _objective_surrogate(a_prop)
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a_prop.copy()
            a = a_prop
            no_improve = 0
            # tighten the step size a little after a successful move
            delta_factor = max(1e-4, delta_factor * 0.98)
        else:
            no_improve += 1
            # occasionally increase step size to escape plateaus
            if no_improve % 1000 == 0:
                delta_factor = min(0.1, delta_factor * 1.05)

    return best_seq, best_val

    if time.time() < deadline:
        print("\n--- Local stochastic refinement ---")
        local_deadline = deadline - 2.0
        refined_seq, refined_val = _local_search(best_seq, n,
                                                 local_deadline,
                                                 rng,
                                                 max_no_improve=5000)
        if refined_val < best_val:
            best_seq = refined_seq
            best_val = refined_val
            print(f"[Local] improved surrogate = {best_val:.6f}")
        else:
            print("[Local] no improvement")

    # ------------------------------------------------------------------
    # Final evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
