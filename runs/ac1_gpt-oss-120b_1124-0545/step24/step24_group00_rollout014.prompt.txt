import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The variable v_local lives in log‑softmax space; a = softmax(v_local)
    is a probability vector with sum=1.
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

def _adam_opt(start_seq: np.ndarray,

def _exp_gradient_opt(start_seq: np.ndarray,

# Simulated‑annealing hill‑climb with three move types.
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move that tries to directly reduce the active convolution
    entry.  Returns the best sequence seen.
    """

    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            # Identify most offending pair for the current max offset
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])     # donor index
            q = int(rng.integers(0, n))           # random receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1   # small fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            # renormalise (the move preserves sum, but rounding may drift)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val
