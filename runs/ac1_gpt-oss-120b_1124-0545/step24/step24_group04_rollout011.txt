def _simulated_annealing(start_seq, n, deadline, rng):
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    start_temp = 0.01  # scale
    end_temp = 1e-6
    # linear cooling schedule based on time
    init_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - init_time
        total = deadline - init_time
        t_frac = elapsed / total
        T = start_temp * (1 - t_frac) + end_temp * t_frac

        if rng.random() < 0.5:
            i, j = rng.integers(0, n, size=2)
            if i == j:

            delta = cur[i] * rng.random() * T / (start_temp)  # scale with T relative to start_temp

            new[j] += delta

"""
Non‑negative step‑function optimiser – version 2.0.

The optimiser builds a set of parameterised “seed” distributions (uniform,
arcsine, triangular, symmetric and asymmetric Beta, Dirichlet, Gaussian
mixtures).  All seeds are normalised to sum = 1 (probability vectors).

Starting from the best seed we perform three optimisation phases, each
respecting the global 1000 s deadline:

1️⃣  Adam optimiser on a smooth‑max surrogate (log‑softmax parametrisation).
2️⃣  Exact exponentiated‑gradient descent on the true max‑convolution,
    using the whole *active set* of indices whose value is within a tiny
    tolerance of the current maximum (multiple sub‑gradients at once).
3️⃣  A temperature‑controlled simulated‑annealing walk that proposes
    small mass‑transfer, full swaps and occasional global permutations.
    Only moves that improve the surrogate are accepted, but uphill moves
    are also accepted with a probability exp(−Δ/Temp) to escape shallow
    plateaus.

The best vector is finally passed to the supplied ``evaluate_sequence``
function, stored in the global variable ``height_sequence_1`` and returned
as a plain Python list.  All steps are written without closures or lambdas
and use only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths used)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence has to sum to 1.  Returns ``2 * n * max(conv(seq, seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density (α,β > 0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a few (2‑5) Gaussian bumps, truncated to the interval."""
    m = rng.integers(2, 6)                # number of components
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)                # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)         # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    Uses a schedule of increasing smooth‑max “β” parameters.
    Returns the best probability vector (sum = 1) seen and the corresponding
    surrogate value.
    """
    # initialise logits
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a fairly dense schedule – the larger the β the closer to the true max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # Allow more work when we are still far from the deadline.
        remaining = deadline - time.time()
        inner_iters = 20000 if remaining > 100 else 8000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a (same formula as in the original code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax space)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 800 == 0:
                cur_val = _surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_multi(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        eta0: float = 5.0,
                        tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.
    Uses the full active set of indices within ``tol`` of the current maximum.
    Returns the best probability vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # active indices (within a relative tolerance of the maximum)
        active = np.where(conv >= max_c - max(tol, tol * max_c))[0]

        # sub‑gradient for the max‐convolution (sum of contributions of all active ks)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]      # 2·a_{k‑i}

        # learning‐rate schedule (standard diminishing step size)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement (mass transfer / swaps / permutations)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk with a simple temperature schedule.
    Only moves that improve the surrogate are kept, but uphill moves can be
    accepted with probability exp(-Δ/Temp) to escape flat plateaus.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    best_seq = cur.copy()
    best_val = cur_val

    # Temperature schedule: start relatively high, decay to near zero.
    T_start = 0.02
    T_end = 1e-6
    t0 = time.time()

    while time.time() < deadline:
        elapsed = time.time() - t0
        total = deadline - t0
        frac = elapsed / total
        T = T_start * (1.0 - frac) + T_end * frac   # linear cooling

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.55:          # mass‑transfer move (most frequent)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # step size proportional to current temperature
            delta = cur[i] * rng.random() * T / T_start
            delta = min(delta, cur[i])   # keep non‑negative
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:                             # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        # ---- occasional global permutation (to break symmetry) ------------
        if rng.random() < 0.004:   # roughly one permutation every ~250 steps
            perm = rng.permutation(n)
            new = new[perm]

        new_val = _surrogate(new)
        Δ = new_val - cur_val

        # Metropolis acceptance
        if Δ < 0.0 or rng.random() < math.exp(-Δ / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (≈ 1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Safety margin so that we never overshoot the external timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length of the candidate vector (feel free to increase for higher
    # asymptotic accuracy, but keep it reasonable for the 1000 s budget).
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as an additional seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Symmetric Beta with a dense grid of α values
    for alpha in np.linspace(0.05, 0.80, 16):          # 0.05 … 0.80 step 0.05
        seeds.append(_beta_symmetric(n, float(alpha)))

    # finer sampling around the region where the optimum is expected
    for alpha in np.linspace(0.20, 0.60, 21):          # step 0.02
        seeds.append(_beta_symmetric(n, float(alpha)))

    # Asymmetric Beta families (α ≠ β) – a small collection
    for a, b in [(0.15, 0.45), (0.30, 0.50), (0.40, 0.30), (0.60, 0.20)]:
        seeds.append(_beta_asymmetric(n, float(a), float(b)))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random Gaussian mixtures
    for _ in range(5):
        seeds.append(_gaussian_mixture(n, rng))

    # Include a few random permutations of some early seeds (breaks symmetry)
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate all seeds on the surrogate – keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing on the smooth‑max surrogate (≈ 45 % of remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient optimisation (active‑set) (≈ 35 %)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        eg_deadline = time.time() + 0.35 * (deadline - time.time())
        print("\n--- Exponentiated‑gradient optimisation (active set) ---")
        seq_eg, val_eg = _exp_gradient_multi(best_seq, n, eg_deadline,
                                             eta0=5.0, tol=1e-12)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing local refinement (use all remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_loc, val_loc = _simulated_annealing(best_seq, n, deadline, rng)
        print(f"[SA] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Final exact evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation expands the search space (fine‑grained symmetric and asymmetric Beta
distributions, extra Gaussian mixtures), adds a more aggressive Adam phase,
uses a true sub‑gradient that accounts for *all* active convolution indices,
and finishes with a temperature‑controlled stochastic walk that can escape
shallow local minima.  The code respects the 1000 s budget, reports progress,
and finally returns the best step‑function found.
