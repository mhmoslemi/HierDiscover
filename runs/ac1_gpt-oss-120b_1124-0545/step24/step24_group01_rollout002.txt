"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s budget)
--------------------------------
1. **Seed generation** – build a diverse pool of probability vectors:
   uniform, triangular, the standard arcsine law, many symmetric
   Beta(α,α) densities (α finely sampled), and mixtures of each of those
   with a uniform component.  All vectors are normalised to sum = 1.

2. **Cheap surrogate screening** – compute the exact surrogate
   ``2·n·max(conv(a,a))`` for every seed and keep the few best (≤ 5).

3. **Exponentiated‑gradient (mirror) descent** – starting from each top
   seed, run a long mirror‑descent on the true max‑convolution objective.
   The update is performed in log‑space (softmax) with Adam‑style
   momentum and a diminishing step‑size ηₜ = η₀/√t.  The run is
   time‑controlled: each seed receives an equal share of the remaining
   budget.  The best iterate (lowest surrogate) over all seeds is kept.

4. **Greedy refinement** – repeatedly move a small fraction of mass from
   the coordinate with the largest sub‑gradient (the active index of the
   max‑convolution) to the one with the smallest sub‑gradient, keeping
   only improving moves.  The loop runs until the time slice allocated to
   it expires.

5. **Local stochastic walk** – a final random‑walk on the simplex that
   perturbs two coordinates at a time; improvements are accepted.
   This exploits any remaining time budget.

6. **Final evaluation** – the best sequence is evaluated with the
   supplied ``evaluate_sequence`` and stored globally as
   ``height_sequence_1``.  The returned value is a plain list of ``float``.

The code respects the security checks of ``evaluate_sequence`` (all entries
are non‑negative and the vector is normalised).  The whole search fits
within the given 1000 s budget and uses at most one additional CPU core.
"""

import time
import math

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (independent of the final scaling):
    2·n·max(conv(seq, seq)), where ``seq`` sums to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x≥0, sum x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the points {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _mixture_with_uniform(base: np.ndarray, weight: float) -> np.ndarray:
    """Return (1‑w)*base + w*uniform, already normalised."""
    n = base.size
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - weight) * base + weight * uni
    # numerical safety
    mixed = np.maximum(mixed, 0.0)
    mixed /= mixed.sum()
    return mixed

def _exp_gradient_descent(start_seq: np.ndarray,
                          deadline: float,
                          eta0: float = 10.0,
                          beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Returns the best sequence observed and its surrogate value.
    """
    n = len(start_seq)
    a = np.copy(start_seq)
    a = np.maximum(a, 1e-30)          # avoid log(0)
    a = a / a.sum()
    logit = np.log(a)
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev]

        # mirror‑gradient (log‑space) for exponentiated gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam‑style momentum (no bias correction for speed)
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)

        # update and map back to the simplex
        logit -= eta * mom
        a = _softmax(logit)

        # periodic exact evaluation
        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            if best_val < 1.503:   # early stop if target already reached
                break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   delta_frac: float = 0.05,
                   rng: np.random.Generator = np.random.default_rng()) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small fraction of mass from the coordinate with
    the largest sub‑gradient to the one with the smallest sub‑gradient.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    n = len(a)
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            # nothing to move
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new_a.copy()
            a = new_a  # continue from the improved point
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         sigma: float = 0.015,
                         rng: np.random.Generator = np.random.default_rng()) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: move a small random fraction of
    mass between two random coordinates and keep the move only if it
    improves the exact surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    n = len(a)
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new_a.copy()
            a = new_a
    return best_seq, best_val

    # 0️⃣ Assemble a rich seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n), _beta_distribution(n, 0.5)]  # arcsine = Beta(0.5,0.5)

    # dense sweep of symmetric Beta(α,α)
    alphas = np.linspace(0.05, 3.0, 120)                # 120 different shapes
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # mixtures of the above with a uniform component (weights 0.05 … 0.5)
    mixture_weights = [0.05, 0.10, 0.20, 0.30, 0.40, 0.50]
    base_seeds = list(seeds)  # copy of current pool for mixing
    for base in base_seeds:
        for w in mixture_weights:
            seeds.append(_mixture_with_uniform(base, w))

    # Dirichlet draws (if time permits)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (exact surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:03d}: surrogate = {val:.6f}")

    # keep the best few seeds for expensive optimisation
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate from seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Exponentiated‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient optimisation (exact max) ---")
        # Give each top seed an equal share of the *remaining* time
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            seeds_left = top_k - sidx + 1
            sub_deadline = time.time() + remaining / seeds_left
            print(f"  EG run #{sidx} – time slice ≈ {sub_deadline - time.time():.1f}s")
            seq, val = _exp_gradient_descent(seed_vec, sub_deadline)
            print(f"   → surrogate after EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (EG)")

            # ----------------------------------------------------------
            #   2️⃣‑b Greedy refinement of the EG result
            # ----------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 3️⃣ Greedy refinement (more aggressive) on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Aggressive greedy refinement ---")
        seq_greedy, val_greedy = _greedy_refine(best_seq, deadline,
                                                delta_frac=0.08,
                                                rng=rng)
        if val_greedy < best_val:
            best_val, best_seq = val_greedy, seq_greedy.copy()
            print(f"   >>> Updated global best after aggressive greedy: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final random local walk (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final stochastic local search ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                sigma=0.02,
                                                rng=rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
