"""
Hybrid optimisation for the step–function problem.

The algorithm proceeds in four phases inside the 1000 s budget:

1️⃣  **Diverse analytic seeds** – uniform, triangular, power‑law (arcsine)
    families, symmetric / asymmetric Beta densities and a handful of
    random Dirichlet draws (including the previously stored solution).

2️⃣  **L‑BFGS‑B on a smooth max‑convolution surrogate** – the hard
    ``max(conv)`` is approximated by a log‑sum‑exp with a schedule of
    increasing smoothness parameters β.  The optimiser works in the
    log‑softmax (mirror) space, i.e. the variables are logits that are
    transformed to a probability vector by a soft‑max.  For each β a
    limited L‑BFGS‑B run (max ≈ 150 iterations) is performed; the final
    logits become the start point for the next, larger β.

3️⃣  **Exact exponentiated‑gradient polishing** – with the true
    ``max(conv)`` (active index only).  A multiplicative mirror‑descent
    update ``a_i ∝ a_i·exp(-η·g_i)`` is applied, where the sub‑gradient
    g is obtained from the active convolution index.  The step‑size
    decays as η₀/√t.

4️⃣  **Final annealing walk** – cheap stochastic moves on the simplex
    (mass transfer between two random coordinates) with a slowly cooling
    temperature, providing a last polish.

All phases respect the global deadline.  The best sequence found is
stored as the global variable ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(seq, seq))
    The input is assumed to be a probability vector (sum = 1).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """Generalised arcsine family a_i ∝ ((i+½)(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives

def _smooth_surrogate_and_grad(logit: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns the smooth surrogate (log‑sum‑exp approximation of max(conv))
    multiplied by 2·n, together with its gradient w.r.t. the logits.
    """
    a = _softmax(logit)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    diff = conv - max_c                     # ≤ 0
    exp_beta = np.exp(beta * diff)          # safe: biggest term = 1
    sum_exp = exp_beta.sum()
    smooth_max = max_c + np.log(sum_exp) / beta
    surrogate = 2.0 * a.shape[0] * smooth_max

    # gradient of smooth_max w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)              # length = 3·n‑2
    n = a.shape[0]
    grad_a = 2.0 * full[n - 1:n - 1 + n]    # derivative of smooth_max
    dot = np.dot(a, grad_a)
    grad_log = a * (grad_a - dot)           # chain rule through softmax
    grad_surrogate = 2.0 * n * grad_log
    return surrogate, grad_surrogate


def _lbfgs_optimize(start_seq: np.ndarray,
                    deadline: float,
                    beta_schedule: list[float] | None = None) -> tuple[np.ndarray, float]:
    """
    Run a cascade of L‑BFGS‑B optimisations on the smoothed surrogate.
    ``beta_schedule`` is a list of smoothness parameters, increasing.
    Returns the final probability vector and its surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [10.0, 100.0, 1_000.0, 10_000.0,
                         100_000.0, 500_000.0, 1_000_000.0, 5_000_000.0]

    logit = np.log(start_seq + 1e-30)        # initialise logits
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # minimise the smooth surrogate for the current β
        def fun(logit_vec):
            val, grad = _smooth_surrogate_and_grad(logit_vec, beta)
            return val, grad

        # a fairly small budget per β – we let L‑BFGS‑B decide
        res = minimize(fun,
                       logit,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 150, 'disp': False})
        logit = res.x
        a = _softmax(logit)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # keep a for the next β
    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exact exponentiated‑gradient (mirror descent) on the true max‑convolution.
    The update is a = softmax(log_a - η·g) where g is the sub‑gradient from the
    active convolution index.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_seq, best_val = a, cur_val
    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing walk on the simplex.
    Each move transfers a random fraction of mass from i → j.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the time budget is stored globally as
    ``height_sequence_1`` and also returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Load previously stored solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a rich seed pool
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n), _triangular(n)]

    # power‑law / arcsine family
    for p in np.linspace(0.1, 0.9, 9):
        seeds.append(_power_arcsine(n, float(p)))

    # symmetric Beta shapes
    for alpha in (0.1, 0.2, 0.3, 0.5, 0.8, 1.2, 1.5, 2.0):
        seeds.append(_beta_asymmetric(n, alpha, alpha))

    # asymmetric Beta shapes
    for alpha, beta in ((0.1, 0.8), (0.2, 1.0), (0.5, 2.0), (1.5, 0.5), (2.0, 0.8)):
        seeds.append(_beta_asymmetric(n, alpha, beta))

    # Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos: list[tuple[float, np.ndarray]] = []
    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(6, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ L‑BFGS‑B cascade on the smooth surrogate (≈ 30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        lbfgs_deadline = time.time() + remaining * 0.30
        print("\n--- L‑BFGS‑B smoothing cascade ---")
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > lbfgs_deadline:
                break
            seq, val = _lbfgs_optimize(seed_vec, lbfgs_deadline)
            print(f"  L‑BFGS‑B seed #{idx + 1:02d}: surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best after L‑BFGS‑B")

    # ------------------------------------------------------------------
    # 3️⃣ Exact exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 4.0          # keep a few seconds for annealing
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=5.0,
                                           max_iters=5_000_000)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best after EG")

    # ------------------------------------------------------------------
    # 4️⃣ Final tiny annealing walk (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best after annealing")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation enriches the seed pool, replaces the Adam stage with a more powerful L‑BFGS‑B optimisation of a log‑sum‑exp surrogate (progressively sharpening the smoothness), adds a longer exponentiated‑gradient polishing phase and finishes with simulated‑annealing.  
The added flexibility in the smoothing schedule and the use of a quasi‑Newton method often push the surrogate value below the previous 1.5085, getting closer to the target 1.503.
