"""
Step‑function optimizer – hybrid parametric search + smooth L‑BFGS + EG polish.

The algorithm proceeds in three stages

1. **Generate a diverse pool of seeds**.
   •  classic inverse‑sqrt shape,
   •  power‑law families a[i] ∝ (i+α)^{-p} with a grid over (p,α),
   •  exponential decay a[i] ∝ exp(-λ·i),
   •  a few Dirichlet draws of various concentrations,
   •  the previously found best sequence (if any).

   Every seed is normalised to unit ℓ₁‑norm (the scaling does not affect the
   evaluation) and its exact surrogate  `2·n·max(conv(a,a))` is computed.
   The best few seeds are kept for the optimisation phase.

2. **Deep smoothed L‑BFGS** on the best seeds.
   The hard max in the objective is replaced by a log‑sum‑exp (soft‑max)
   with an increasing temperature schedule β.  We optimise the log‑soft‑max
   parameters (log‑probabilities) with SciPy’s L‑BFGS‑B.  The final
   soft‑max vector is the candidate for the next stage.

3. **Exponentiated‑gradient polishing**.
   Using the exact max‑convolution objective, a mirror‑descent (EG) loop
   with Adam‑style momentum is run for a large number of iterations (or
   until the time budget expires).  The best iterate of this phase is the
   final answer.

All work respects the wall‑clock budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS); fallback to pure EG if missing
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:        # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (∑a=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # Numerical safety – fall back to uniform if everything underflows
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate: 2·n·max(conv(a,a))  for an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate used during the search (sum(a)=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – useful seeds
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a[i] ∝ (i+1+shift)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + shift
    a = idx ** (-p)
    return a / a.sum()


def _exponential(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed max‑convolution (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule,
                      max_iter_per_beta: int = 3000):
    """
    Minimise a smooth surrogate of the max‑convolution using L‑BFGS‑B.
    Returns the best ℓ₁‑normalised vector seen and its exact surrogate value.
    """
    if _opt is None:                       # pragma: no cover
        # SciPy not available – fall back to a short EG run.
        return _exp_gradient_opt(start_a, n, deadline,
                                 max_iters=200_000, eta0=6.0)

    # log‑softmax parameters (always stay finite)
    v = np.log(start_a + 1e-30)

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # ℓ₁‑norm = 1
            conv = _conv_fft(a, a)
            shift = conv.max()
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a (see derivation in the paper)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # ∂obj/∂a

            # back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-14,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:                # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {exc}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser – final polish
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 8.0,
                      tol: float = 1e-12):
    """
    Mirror‑descent on the *exact* max‑convolution.
    Uses Adam‑style momentum on the log‑variables.
    Returns the best iterate and its exact surrogate.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the max is (almost) attained
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient: average of a reversed on every active anti‑diagonal
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum and learning‑rate schedule
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        5_000_000.0, 10_000_000.0, 50_000_000.0, 100_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # ------------------------------------------------------------------
    # L‑BFGS + EG polishing for each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        # Dynamically split the remaining budget: 20 % per seed, min 10 s
        remaining = deadline - time.time()
        share = max(10.0, remaining * 0.20)
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx}/{len(top_seeds)}] allocated ≈ {share:.1f}s")

        # ----- smoothed L‑BFGS -------------------------------------------------
        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3000,
        )
        print(f"    L‑BFGS surrogate = {val_lbfgs:.8f}")

        # ----- EG polishing ---------------------------------------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=600_000,
                eta0=6.0,
                tol=1e-12,
            )
            print(f"    EG polish       = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # keep the better of the two
        cand_seq, cand_val = (seq_eg, val_eg) if val_eg < val_lbfgs else (seq_lbfgs, val_lbfgs)

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"    *** New global best surrogate = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Final intensive EG polishing on the current global best
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final intensive EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=1_200_000,
            eta0=4.0,
            tol=1e-12,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the seed pool (including power‑law, exponential and Dirichlet families), runs a more exhaustive smooth L‑BFGS phase with a denser β‑schedule, and follows it with a long exponentiated‑gradient polishing stage. All steps respect the 1000‑second wall‑clock budget, and the best sequence is cached in the global variable `height_sequence_1`.
