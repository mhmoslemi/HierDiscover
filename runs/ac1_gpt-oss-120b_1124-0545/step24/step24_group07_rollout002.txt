assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{value}= \frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 ,
\]

where `a*a` denotes the linear convolution of the step‑function with itself.
Because the value is invariant under a global scaling of `a`,
we can restrict the search to probability vectors (`∑a_i = 1`).

The optimiser works in three phases

1. **Diverse seeding** – a large pool of analytic families (uniform,
   triangular, arcsine, power‑law, exponential, Beta, mixtures) and many
   Dirichlet draws.  All candidates are normalised and evaluated with the
   cheap surrogate `2·n·max(conv(a,a))`.  The ten best seeds are kept.

2. **Exact exponentiated‑gradient polishing** – for each seed we run a
   mirror‑descent on the *exact* max‑convolution.  
   The sub‑gradient is obtained from *all* almost‑active indices
   (those whose convolution value is within `1e-12` of the maximum).
   The update is `log a ← log a – η·g` with a diminishing step size
   `η₀/√t`.  A time‑budget is split between the seeds, and finally a
   longer run is performed on the current global best.

3. **Final stochastic annealing** – a tiny Metropolis walk on the simplex
   (mass transfer `i → j`) that can escape shallow local minima and gives a
   last polish.

All heavy work uses FFT‑based convolution (`_conv_fft`) which is
`O(n log n)`; with `n = 4000` a single iteration costs only a few
hundred micro‑seconds, allowing millions of updates inside the 1000 s
budget.

The code below implements the whole pipeline in a single
`propose_candidate` function as required by the platform.


```python
"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser consists of three stages:

1️⃣  A wide seed pool (analytic families, Dirichlet draws, previous
    solution) is generated and the cheap surrogate
    `2·n·max(conv(a,a))` is evaluated.  The ten best seeds are kept.

2️⃣  Exact exponentiated‑gradient (mirror descent) on the true max‑convolution
    is run for every seed with a modest time slice, then a longer run on the
    best candidate.  The sub‑gradient uses *all* near‑maximal convolution
    indices (active set) which is more informative than a single index.

3️⃣  A final tiny simulated‑annealing walk transfers a small random
    fraction of mass between two positions and accepts uphill moves with a
    Metropolis rule.  This gives a last polishing step.

All convolutions use a real FFT (`_conv_fft`) for speed.  The
resulting probability vector is returned as a plain Python list;
`evaluate_sequence` will automatically scale it correctly.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # kept for possible future extensions

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    The input is assumed to be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_density(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _decreasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _increasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{-p} (monotonically increasing)."""
    i = np.arange(n, 0, -1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _exponential_growth(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(lam * i) (mirrored version of decay)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(lam * i)
    return a / a.sum()


def _mixture_beta(n: int, components, rng: np.random.Generator) -> np.ndarray:
    """
    Mixture of several Beta components.
    ``components`` is an iterable of tuples (alpha, beta, weight)
    with weight ≥0 and sum of weights = 1.
    """
    a = np.zeros(n, dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    for alpha, beta, w in components:
        comp = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
        comp = comp / comp.sum()
        a += w * comp
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimisation (mirror descent)
# ----------------------------------------------------------------------


def _exp_gradient_opt_time(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           min_iters: int = 0,
                           max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    The active set consists of all indices whose convolution value is
    within `tol` of the maximum.
    """
    n = len(start_seq)
    a = start_seq.copy()
    # keep strictly positive entries for the log‑space
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    # tolerance for active set (absolute)
    tol = 1e-12

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(np.max(conv))

        # indices that are essentially active
        active = np.where(conv >= max_val - tol)[0]

        # build weight vector (average over the active set)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        # gradient = 2 * (rev_a * w)[n-1 : n-1+n]
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        # mirror‑descent update in log‑space
        log_a = log_a - eta * grad
        a = _softmax(log_a)

        # periodic check for improvement
        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_seq, best_val = a, cur_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing walk (tiny polishing)
# ----------------------------------------------------------------------


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing on the simplex.
    A random fraction `delta` of mass is moved from a source index to a
    destination index.  Metropolis acceptance is used.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_a)

        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            accept = rng.random() < prob

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best distribution (as a list) is stored in
    the global variable ``height_sequence_1`` for possible reuse on the
    next call.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Load previous solution (if it matches the requested length)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a rich pool of seed candidates
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # arcsine‑like power‑law family (coarse sweep)
    for p in np.linspace(0.2, 1.2, 11):
        seeds.append(_power_arcsine(n, float(p)))

    # monotone power‑law families
    for p in np.linspace(0.4, 2.0, 9):
        seeds.append(_decreasing_power(n, float(p)))
        seeds.append(_increasing_power(n, float(p)))

    # exponential families
    for lam in np.logspace(-4, -1, 8):
        seeds.append(_exponential_decay(n, float(lam)))
        seeds.append(_exponential_growth(n, float(lam)))

    # symmetric Beta families
    for a in (0.3, 0.5, 0.8, 1.2, 1.5, 2.0):
        seeds.append(_beta_density(n, a, a))

    # asymmetric Beta families
    for (alpha, beta) in ((0.3, 1.0), (0.5, 1.5), (1.5, 0.5), (2.0, 0.8)):
        seeds.append(_beta_density(n, alpha, beta))

    # random Beta mixtures (two components)
    for _ in range(8):
        w = rng.random()
        comp = [
            (rng.uniform(0.2, 2.5), rng.uniform(0.2, 2.5), w),
            (rng.uniform(0.2, 2.5), rng.uniform(0.2, 2.5), 1 - w),
        ]
        seeds.append(_mixture_beta(n, comp, rng))

    # Dirichlet draws with a wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.3, 0.8, 1.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Initial cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    # keep the ten best seeds
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(10, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Exact exponentiated‑gradient polishing on the top seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate ≈ 75 % of the remaining time to polishing passes
        polish_time = remaining * 0.75
        per_seed = polish_time / (len(top_seeds) + 1)

        print("\n--- Exponentiated‑gradient polishing (short passes) ---")
        for i, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seed_deadline = time.time() + per_seed
            seq, val = _exp_gradient_opt_time(seed_vec, seed_deadline,
                                              eta0=5.0,
                                              max_iters=5_000_000)
            print(f" [Polish {i+1}/{len(top_seeds)}] surrogate = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best after short polishing")

    # ------------------------------------------------------------------
    # 3️⃣ Long run on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        long_deadline = deadline - 4.0          # keep a few seconds for anneal
        print("\n--- Long exponentiated‑gradient run on global best ---")
        seq, val = _exp_gradient_opt_time(best_seq, long_deadline,
                                          eta0=5.0,
                                          max_iters=10_000_000)
        print(f"[Long EG] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("   >>> Updated global best after long EG")

    # ------------------------------------------------------------------
    # 4️⃣ Final simulated‑annealing walk (tiny polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq, val = _anneal_opt(best_seq, deadline,
                               init_temp=0.02,
                               final_temp=1e-7,
                               sigma=0.015)
        print(f"[Anneal] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("   >>> Updated global best after annealing")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
