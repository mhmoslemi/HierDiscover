"""
Step‑function optimizer – “enhanced‑search” version.

The routine builds a very diverse seed pool (including classic windows,
Gaussian bells, two‑parameter power‑exponential shapes and random Dirichlet
samples).  After a quick surrogate evaluation it keeps the best few seeds and
runs a multi‑stage non‑convex optimisation on each of them:

1. Exponentiated‑gradient on the exact max‑convolution (large iteration budget).
2. Smooth‑max L‑BFGS (fine‑grained β‑schedule, more L‑BFGS iterations).
3. Gradient‑descent flattening of the self‑convolution (quadratic loss).
4. Simulated‑annealing polish (long cooling schedule).

All stages run in the log‑softmax (probability‑simplex) space, automatically
respecting the non‑negativity and sum‑to‑one constraints.  The total wall‑clock
budget is 1000 s; the code uses the remaining time intelligently, first
optimising the top‑k seeds in parallel (time‑sliced) and finally polishing the
overall best solution.  The best sequence is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional imports (SciPy is used only if available)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector, sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # algorithm from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (keeps the L1‑norm)."""
    s = (a + a[::-1]) * 0.5
    total = s.sum()
    if total > 0:
        s /= total
    return s


# ----------------------------------------------------------------------
# Surrogate objective (after L1‑normalisation the true objective reduces to
#   2·n·max(conv) )
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a /= a.sum()
    return a


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    mixed /= mixed.sum()
    return mixed


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


def _gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """σ is given as a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient on exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 20.0,
                            tol: float = 1e-11) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient from all active shifts
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Flattening gradient (minimise variance of self‑convolution)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 12_000_000,
                         eta0: float = 12.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom for L1‑normalised a
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1_000_0000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accept occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 800_000,
                         start_sigma: float = 0.001,
                         T_start: float = 1e-2,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

        eta0=22.0,
        tol=1e-12,
    )
    seq_eg = _symmetrize(seq_eg)

    # 2) Smooth‑max L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline,
                                            beta_schedule=None,
                                            max_iter_per_beta=9000)
        seq_lbfgs = _symmetrize(seq_lbfgs)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) Flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs, n, deadline,
                                                   max_iters=14_000_000,
                                                   eta0=14.0)
        seq_flat = _symmetrize(seq_flat)
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) Simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq_flat, n, deadline, rng,
                                              max_attempts=1_200_000,
                                              start_sigma=0.0006,
                                              T_start=8e-3,
                                              T_end=3e-9)
        seq_sa = _symmetrize(seq_sa)
    else:
        seq_sa, val_sa = seq_flat, _objective_surrogate(seq_flat)

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, _objective_surrogate(seq_flat)),
        (seq_sa, _objective_surrogate(seq_sa)),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (rich collection of analytic shapes)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # elementary families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.35, 0.4, 0.45, 0.5, 0.6, 0.8, 1.0, 1.2]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 25):
        seeds.append(_exp_decay(n, rate))

    rates = np.logspace(-5, -0.2, 14)
    for r1 in rates[:7]:
        for r2 in rates[7:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.4))

    for a in [0.4, 0.6, 1.0, 1.5]:
        for b in [0.4, 0.6, 1.0, 1.5]:
            seeds.append(_beta_shape(n, a, b))

    # two‑parameter power‑exponential sweep (dense)
    p_vals = np.linspace(0.3, 1.3, 15)
    lam_vals = np.linspace(0.0, 0.012, 12)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    # window functions
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    for sigma_frac in [0.07, 0.12, 0.18, 0.25, 0.33]:
        seeds.append(_gaussian_window(n, sigma_frac))

    # random Dirichlet draws
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures with uniform to break symmetry
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 5e-4), _power(n, 0.9)]
    for base in base_shapes:
        for alpha in [0.2, 0.4, 0.6]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed copies (occasionally useful)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include the previous global best if defined
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    prev_arr = prev_arr / s
                    seeds.append(prev_arr)
        except Exception:
            pass

    # small random perturbations of the currently best seed (if any)
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(5):
            pert = best_seed + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a wall‑clock budget,
    stores the best sequence in the global variable ``height_sequence_1``
    and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    n = 4000                                    # problem dimension

    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Build an extensive seed pool
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds, evaluating surrogates...")

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the few best seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seed survived the quick evaluation")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deep optimisation of the top seeds (time‑sliced)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        per_seed_time = remaining * 0.78 / top_k
        for idx, seed_vec in enumerate(top_seeds):
            sub_deadline = time.time() + per_seed_time
            seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)
            if val_opt < global_best_val:
                global_best_seq = seq_opt
                global_best_val = val_opt
                print(f"  [Seed {idx+1}] new global best surrogate = {global_best_val:.6f}")
            remaining = deadline - time.time()
            if remaining < 12.0:
                break

    # ------------------------------------------------------------------
    # 5. Final polish on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Final polish] refining the best candidate")
        seq_fin, val_fin = _full_optimisation(global_best_seq, n, deadline, rng)
        if val_fin < global_best_val:
            global_best_seq = seq_fin
            global_best_val = val_fin
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Compute the true objective and store the answer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
