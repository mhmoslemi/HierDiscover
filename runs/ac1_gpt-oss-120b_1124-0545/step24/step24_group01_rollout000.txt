def adam_expgrad(a, deadline, eta=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    logit = np.log(a+1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    t = 0
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    while time.time() < deadline:
        t += 1
        a = _softmax(logit) # ensure a normalized
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient wrt a
        g = np.zeros_like(a)
        i_start = max(0, k - (len(a) - 1))
        i_end = min(len(a)-1, k)
        idx = np.arange(i_start, i_end+1)
        rev = k - idx
        g[idx] = 2 * a[rev]
        # Mirror gradient in log domain:
        dot = np.dot(a, g)
        grad_log = a * (g - dot)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logit = logit - eta * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate
        if t % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        # early stop if best_val < target
        if best_val < 1.503:
            break
    return best_a, best_val
```

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that
minimises the supplied ``evaluate_sequence``.  The algorithm consists of
four stages:

1. **Broad seeding** – generate a large and diverse pool of candidate
   probability vectors (uniform, arcsine, triangular, symmetric / asymmetric
   Beta, random Dirichlet draws, and a parametric family
   ``a_i ∝ ((i+½)(n‑½‑i))^{-p}`` with many exponents *p*).  Each vector is
   normalised to sum 1 and evaluated with the exact surrogate
   ``2·n·max(conv(a,a))`` (the same objective that ``evaluate_sequence``
   uses, up to the normalisation).  The best few seeds are retained.

2. **Adam‑mirrored gradient descent** – starting from each retained seed we
   perform exponentiated‑gradient descent on the *exact* max‑convolution.
   The sub‑gradient of ``max(conv)`` at the current active shift is
   ``g_i = 2·a_{k‑i}``.  In log‑space (mirrored coordinates) we apply a
   full Adam optimiser with bias‑correction and a diminishing step size.
   The best iterate of every run is kept.

3. **Deterministic greedy refinement** – using the sub‑gradient we move a
   small fraction of mass from the coordinate with the largest gradient
   to the one with the smallest, accepting only improving moves.  This
   removes the last “bottleneck’’ of the max‑convolution.

4. **Stochastic local walk** – a final random walk on the simplex (small
   random transfers of mass) explores the remaining neighbourhood.

All steps respect the time limit (default 1000 s).  The best vector found
is stored in the global ``height_sequence_1`` variable and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used internally: 2·n·max(conv(seq, seq)).
    ``seq`` is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _max_conv_and_subgrad(a: np.ndarray):
    """Return (max_val, subgradient, argmax_index) for the max‑convolution."""
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    max_val = conv[k]

    n = len(a)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - idx
    g[idx] = 2.0 * a[rev]
    return max_val, g, k


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric when alpha==beta, otherwise asymmetric Beta(α,β)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_arcsine_family(n: int, p: float) -> np.ndarray:
    """Family a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    val = ((i + 0.5) * (n - 0.5 - i))
    # Guard against zero (should never happen for interior points)
    val = np.maximum(val, 1e-12)
    a = np.power(val, -p)
    a = a / a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_expgrad(start_seq: np.ndarray,
                  deadline: float,
                  eta: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam with exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best sequence seen (as a probability vector) and its surrogate.
    """
    a = start_seq.copy()
    n = len(a)

    # initialise log‑probabilities (mirror variables)
    logit = np.log(a + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    while time.time() < deadline:
        t += 1
        # map back to the simplex
        a = _softmax(logit)

        max_val, g, _ = _max_conv_and_subgrad(a)

        # mirror‑gradient (log‑space) as required by exponentiated‑gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update (with bias correction)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step in log‑space
        logit = logit - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # early stop condition
                if best_val < 1.503:
                    break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   delta_frac: float = 0.02,
                   max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy improvement: repeatedly move a small fraction of
    mass from the coordinate with the largest sub‑gradient to the one with
    the smallest (if it improves the surrogate).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(a)
    rng = np.random.default_rng()

    steps = 0
    while steps < max_steps and time.time() < deadline:
        steps += 1

        _, g, _ = _max_conv_and_subgrad(a)

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = new_a.copy()
            a = new_a  # continue from improved point

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         sigma: float = 0.015,
                         max_attempts: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex. Moves that improve the exact
    surrogate are kept.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        cur_val = _objective_surrogate(new_seq)
        if cur_val < best_val:
            best_seq = new_seq
            best_val = cur_val
    return best_seq, best_val
