def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()

def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.

def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast greedy walk on the simplex:
    repeatedly move a random amount δ from one coordinate to another.
    The convolution is updated by an inexpensive O(n) slice operation,
    allowing hundreds of thousands of trials.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    best_a = a.copy()
    best_max = np.max(conv)
    best_val = 2.0 * n * best_max

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 1e-15:
            continue

        # random transfer size, bounded by sigma * a[i]
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # propose new convolution (exact update)
        cand_conv = conv.copy()
        # shift contributions
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        # diagonal / cross terms
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # accept only improving moves
        if cand_val < best_val - 1e-12:
            # apply the move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_max = cand_max
            best_val = cand_val
            accepted += 1

            # periodically re‑compute the convolution from scratch to
            # eliminate accumulated rounding error
            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                last_recalc = accepted

        # schedule: slowly cool sigma
        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val
