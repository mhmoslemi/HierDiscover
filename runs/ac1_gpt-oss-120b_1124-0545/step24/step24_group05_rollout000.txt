"""
Propose a step‑function (non‑negative vector) with a low value of
evaluate_sequence().

Algorithm
---------
1. **Rich seed pool** – In addition to the arcsine shape used previously we
   generate a variety of Beta‑distribution shapes (symmetric and skewed),
   a few exponential decays, uniform, and random Dirichlet draws.
2. **Smoothed Adam descent** – Each seed is refined with Adam on the
   temperature‑smoothed max‑convolution objective
        F_β(p) = (1/β)·log Σ_k exp(β·(p∗p)[k]).
   The temperature β is raised linearly from 0.5 to 20 (instead of 8) to
   obtain a sharper approximation of the true max.  Learning‑rate and
   iteration budget are adapted to the remaining time.
3. **Mirrored‑gradient polishing** – The best sequence from the Adam phase
   receives a long exponentiated‑gradient (mirror‑descent) pass on the
   exact max‑convolution objective.
4. **Local hill‑climbing** – A cheap stochastic neighbour search moves a
   small amount of mass between two randomly chosen indices; improvements
   are kept.  This exploits any remaining “wiggle‑room’’ left by the
   gradient steps.
5. **Final evaluation** – The surrogate objective
        2·n·max(conv)/sum²
   matches evaluate_sequence (apart from its clipping, which never
   activates for our normalised vectors).  The true function is called
   once at the very end.

The routine is time‑aware: a hard deadline (budget_s‑5 s) is respected,
and each phase stops as soon as the deadline is approached.
The best sequence found is stored in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    # sum_a is forced to be > 0 (we normalise to 1), so division safe.
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1} using mid‑point rule."""
    i = np.arange(n, dtype=np.float64)
    # (i+0.5)^(α‑1) * (n‑0.5‑i)^(β‑1)  – always positive for α,β>0
    a = ((i + 0.5) ** (alpha - 1.0)) * ((n - 0.5 - i) ** (beta - 1.0))
    # Guard against overflow/underflow
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Gradient of the temperature‑smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters `log_a`.  Returns ∂F/∂log_a, shape (n,).
    """
    a = _softmax(log_a)                 # probability vector (sums to 1)
    conv = _conv_fft(a, a)              # length 2n‑1

    # Softmax over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                        # shape (2n‑1,)

    # Convolution with reversed a to obtain gradient wrt a
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)          # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]  # slice of length n

    # Chain rule through the softmax (log‑softmax) mapping
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 150_000,
                      lr0: float = 0.8,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max‑convolution objective.
    Returns the best simplex vector found and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)   # initialise log‑softmax parameters
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (in log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # Project back onto the simplex via softmax and evaluate
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) polishing on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry (indices i where overlap occurs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the surrogate objective
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                n: int,
                deadline: float,
                max_moves: int = 30_000,
                max_step_frac: float = 0.05) -> np.ndarray:
    """
    Randomly move a tiny amount of probability mass between two indices.
    Accept only improving moves.  Very cheap – works directly on the
    surrogate objective, which equals evaluate_sequence for normalised
    vectors.
    """
    best = seq.copy()
    best_val = _objective_surrogate(best)

    rng = np.random.default_rng()

    for _ in range(max_moves):
        if time.time() > deadline:
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # determine a small transferable amount
        amount = best[i] * max_step_frac * rng.random()
        if amount <= 0.0:
            continue

        # propose new vector
        cand = best.copy()
        cand[i] -= amount
        cand[j] += amount

        # keep it on the simplex (sum unchanged, non‑negative ensured)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_val = cand_val
            best = cand
            # occasionally log progress
            # (printing inside the loop may be noisy, so do it sparsely)
            if rng.random() < 0.001:
                print(f"[HillClimb] improved surrogate = {best_val:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence()
    value.  The routine builds a large pool of analytic seeds,
    refines each with a temperature‑smoothed Adam optimiser,
    runs a long exponentiated‑gradient polishing, and finally
    performs a cheap stochastic hill‑climb.  The best sequence found
    is returned (and stored globally as ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a small safety margin before the hard deadline.
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the FFT size (8192) and the
    # time budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    # 1) Classic arcsine (Beta(0.5,0.5))
    candidates.append(_arcsine_distribution(n))

    # 2) Uniform
    candidates.append(_uniform(n))

    # 3) Exponential decays
    for rate in (0.0003, 0.0007, 0.0015, 0.003):
        candidates.append(_exp_decay(n, rate))

    # 4) Symmetric Beta families (including arcsine & uniform)
    for alpha in (0.3, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0):
        candidates.append(_beta_distribution(n, alpha, alpha))

    # 5) Skewed Beta families
    for a, b in ((0.3, 3.0), (0.5, 3.0), (0.8, 3.0), (3.0, 0.5)):
        candidates.append(_beta_distribution(n, a, b))

    # 6) Convex mixes of arcsine and uniform (coarse sweep)
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for alpha in np.linspace(0.0, 1.0, 9):
        cand = alpha * arcs + (1.0 - alpha) * uni
        candidates.append(cand / cand.sum())

    # 7) Random Dirichlet draws (lightweight)
    for _ in range(5):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    # 8) If a previous best is stored, reuse it as a seed
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(np.maximum(arr, 0.0) / np.maximum(arr.sum(), 1e-30))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Optimise each candidate with the smoothed Adam optimiser
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    total_cands = len(candidates)

    for idx, cand in enumerate(candidates, start=1):
        # Normalise to a probability vector (sum = 1)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        # Check remaining time – abort early if not enough left.
        time_left = deadline - time.time()
        if time_left < 8.0:
            print("[Info] Time exhausted, stopping candidate loop.")
            break

        # Allocate iterations proportionally.  We cap at 250 k for safety.
        remaining = total_cands - idx + 1
        # Aim for ≈ 0.6·time_left spread equally; enforce a floor of 15 k.
        max_iters = int(min(250_000,
                            max(15_000,
                                (time_left * 0.6) / remaining)))
        print(f"[Candidate {idx}/{total_cands}] max_iters={max_iters}")

        seq, val = _adam_softmax_opt(cand,
                                     n,
                                     deadline,
                                     max_iters=max_iters,
                                     lr0=0.8,
                                     beta_start=0.5,
                                     beta_end=20.0)

        print(f"   surrogate after Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   *** new global best (surrogate) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Polishing with exponentiated‑gradient (mirror‑descent) – long pass
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _uniform(n)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polish_seq, polish_val = _exp_gradient_opt(best_overall_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=5_000_000,
                                                   eta0=12.0)
        if polish_val < best_overall_val:
            best_overall_seq = polish_seq
            best_overall_val = polish_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Polish] no improvement (best surrogate unchanged)")

    # ------------------------------------------------------------------
    # Optional cheap stochastic hill‑climb (can squeeze a few more
    # percent out of the surrogate).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Hill‑climbing refinement ---")
        hc_seq = _hill_climb(best_overall_seq,
                             n,
                             deadline,
                             max_moves=30_000,
                             max_step_frac=0.04)
        hc_val = _objective_surrogate(hc_seq)
        if hc_val < best_overall_val:
            best_overall_seq = hc_seq
            best_overall_val = hc_val
            print(f"[HillClimb] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[HillClimb] no improvement")

    # ------------------------------------------------------------------
    # Final evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for future continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
