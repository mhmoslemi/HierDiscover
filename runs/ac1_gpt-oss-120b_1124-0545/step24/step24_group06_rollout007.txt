"""
Hybrid optimiser for the step–function “max‑convolution” problem.

The optimiser proceeds in three stages:

1️⃣  Diverse seeds (uniform, arcsine, triangular, spikes, many Beta‑ and
    Dirichlet‑draws) are generated and the cheap surrogate
    `2·n·max(conv(a,a))` is evaluated. The best seed is kept.

2️⃣  A smooth‑max Adam optimiser is run on the best seed (log‑softmax
    parametrisation).  This quickly drives the objective down.

3️⃣  An exact sub‑gradient optimiser (exponentiated‑gradient on the true
    max‑convolution) refines the solution further.

4️⃣  Finally a simulated‑annealing local search with an *incremental*
    convolution update explores uphill moves and drastically reduces the
    maximum autocorrelation.  The search accepts a move with probability
    `exp(-(Δ)/T)` where the temperature `T` is slowly cooled.

All phases respect the time budget (default 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` and also
returned as a plain list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    The sequence is assumed to be a probability vector (sum=1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return vectors that already sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) discretised on the centre of each bin."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the soft‑max surrogate of ``2·n·max(conv)``."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a small schedule of “temperatures” (β in the log‑sum‑exp)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # more inner iterations while we have time left
        max_iter_beta = 10000 if (deadline - time.time()) > 30 else 3000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 2n‑1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement (incremental convolution update)
# ----------------------------------------------------------------------


def _simulated_annealing_local_search(start_seq: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast local walk with occasional uphill acceptance.
    The full convolution is kept explicitly and updated in O(n) per trial.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (relative to current objective)
    T_init = cur_val * 0.02
    T_final = cur_val * 1e-6
    T = T_init

    sigma = 0.08                               # initial relative step size
    attempts = 0
    accepted = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # pick source index – favour those that contribute most to the
        # current maximum (k = argmax conv)
        # --------------------------------------------------------------
        kmax = int(conv.argmax())
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        contrib = a[idx_range] * a[kmax - idx_range]
        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            i = rng.choice(idx_range, p=probs)
        else:
            i = rng.integers(0, n)

        # pick destination index (different from i)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)

        # size of the transfer
        max_delta = sigma * a[i]
        if max_delta <= 1e-15:
            continue
        delta = rng.random() * max_delta  # 0 < delta ≤ sigma·a[i]

        # --------------------------------------------------------------
        # Incremental convolution update
        # --------------------------------------------------------------
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()
        diff = cand_val - cur_val

        # accept if improvement or with Metropolis probability
        accept = False
        if diff < 0.0:
            accept = True
        else:
            prob = math.exp(-diff / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move permanently
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1

            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()

            # cool down slowly
            T = max(T_final, T * 0.9999)

            # periodic re‑normalisation of the convolution to avoid drift
            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                cur_val = 2.0 * n * conv.max()
                last_recalc = accepted

        # --------------------------------------------------------------
        # schedule updates (temperature and sigma)
        # --------------------------------------------------------------
        if attempts % 30000 == 0:
            sigma = max(0.002, sigma * 0.90)          # shrink step size
            T = max(T_final, T * 0.95)                # extra cooling

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣  Load previously found solution, if any
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich set of seeds (all are normalised to sum=1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta families – we sweep a finer grid of alphas
    beta_alphas = [
        0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15, 0.20,
        0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.60, 0.80,
        1.00, 1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # a simple mixture of arcsine and uniform (50‑50)
    mix = 0.5 * _arcsine(n) + 0.5 * _uniform(n)
    mix /= mix.sum()
    seeds.append(mix)

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate all seeds (cheap surrogate) and keep the best few
    # ------------------------------------------------------------------
    print("\n--- Seed evaluation (surrogate) ---")
    best_seq = None
    best_val = np.inf
    top_seeds = []                              # (val, vec)
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # just in case of numerical drift
        vec = np.maximum(vec, 0.0)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
        # keep the 5 best seeds for possible restarts
        top_seeds.append((val, vec.copy()))
        top_seeds.sort(key=lambda x: x[0])
        if len(top_seeds) > 5:
            top_seeds.pop()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam optimisation (first half of remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining   # ~35 % of the rest
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polishing (≈20 % of remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + 0.20 * remaining
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing intensive local refinement (remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_loc, val_loc = _simulated_annealing_local_search(best_seq, n, deadline, rng)
        print(f"[SA] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣  Final (official) evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible subsequent calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
