"""
step_function_optimizer – fast heuristic for minimise
evaluate_sequence(sequence).

The algorithm works in three phases:

1️⃣  Broad seed generation
    – deterministic seeds (uniform, arcsine, triangular)
    – a dense sweep of symmetric Beta(α,α) and symmetric
      exponential‑decay shapes (parameter α or rate λ is
      searched on a fine grid)
    – random Dirichlet draws with many concentration values
      and a handful of random Gaussian mixtures.

    All seeds are normalised to sum = 1 (the objective is
    homogeneous, therefore the absolute scale does not matter).

2️⃣  Continuous refinement
    – Adam optimiser on a smooth‑max surrogate of the true
      max‑convolution (log‑sum‑exp with a schedule of β’s).
    – A mirror‑descent step on the *exact* max‑convolution, but
      after each update the vector is symmetrised
      (a←(a+rev(a))/2) – the optimal solution is known to be
      symmetric, this accelerates convergence.

3️⃣  Multi‑scale stochastic local search
    – small mass‑transfer moves, swaps and occasional full
      permutations, accepting only strict improvements.
    – the search continues until the time budget (≈ 1000 s) is
      exhausted.

The best vector found is stored globally as ``height_sequence_1``
and returned as a plain Python list.  All helper functions are
defined at module level – no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (fast surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (same numerator as evaluate_sequence)
    – works on a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} while preserving the total mass."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution_param(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {1,…,n} (i+0.5)/n as the support."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        return _uniform(n)
    a /= tot
    return a


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """One‑sided exponential decay a_i ∝ exp(-rate·i)."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _sym_exponential(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-rate·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n, dtype=np.float64),
                    np.arange(n - 1, -1, -1, dtype=np.float64))
    a = np.exp(-rate * d)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps (truncated to the interval).
    The resulting vector is normalised to sum 1.
    """
    m = rng.integers(2, 6)                # 2‑5 components
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.5, n * 0.05)
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        return _uniform(n)
    a /= tot
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate (log‑sum‑exp).
    Returns the best probability vector seen and its surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a dense schedule of β values (larger → sharper surrogate)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner Adam steps – more if time permits
        remaining = deadline - time.time()
        inner_iters = 14000 if remaining > 40 else 5000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                 # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient w.r.t. a (derived in the reference code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # push‑forward through softmax (mirror‑gradient)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 500 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent on the true max‑convolution (with symmetrisation)
# ----------------------------------------------------------------------


def _sym_exp_gradient_opt(start_seq: np.ndarray,
                          deadline: float,
                          eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution.
    After each update the vector is symmetrised (a←(a+rev(a))/2).
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    n = len(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # contribution 2·a[j] (j = rev_idx)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # enforce symmetry – this keeps the feasible set small and
        # encourages the known symmetric optimum.
        a = _symmetrize(a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive stochastic local refinement (mass transfer, swaps, perms)
# ----------------------------------------------------------------------


def _local_multi_search(start_seq: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk on the simplex.
    Proposals:
      * small mass‑transfer from i→j,
      * full swap a[i] ↔ a[j],
      * random permutation of the whole vector.
    Only strict improvements are kept.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    sigma_start = 0.12          # relative size of mass‑transfer steps
    sigma_end = 0.004
    sigma = sigma_start
    decay = 0.96                # a slightly slower decay than before
    block_iters = 4000

    while time.time() < deadline:
        # ---- mass‑transfer block ----
        for _ in range(block_iters):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, len(cur), size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            if new[i] < 0.0:
                continue
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- swap block ----
        for _ in range(block_iters // 2):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, len(cur), size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- occasional whole‑vector permutation ----
        if rng.random() < 0.12:
            perm = rng.permutation(len(cur))
            new = cur[perm]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # decay sigma
        sigma = max(sigma_end, sigma * decay)
        if sigma <= sigma_end + 1e-9:
            block_iters = max(2000, block_iters // 2)

    return cur, cur_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the supplied
    time budget (≈ 1000 s) and saves the best sequence globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – we need a couple of seconds for the final call
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # Load a previously stored solution (if any) – useful for restarts
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev_seq, list) and len(prev_seq) == n:
        tmp = np.asarray(prev_seq, dtype=np.float64)
        if tmp.sum() > 0.0:
            prev_vec = tmp / tmp.sum()

    # --------------------------------------------------------------
    # 0️⃣  Build a wide pool of seeds (all normalised to sum = 1)
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # ----- symmetric Beta sweep (dense grid on log‑scale) -----
    for alpha in np.logspace(-2, 0.7, num=25):   # 0.01 … ~5
        seeds.append(_beta_distribution_param(n, float(alpha)))

    # ----- symmetric exponential sweep (log‑scale) -----
    for rate in np.logspace(-4, 0, num=20):     # 1e‑4 … 1
        seeds.append(_sym_exponential(n, float(rate)))

    # ----- random Dirichlet draws (many concentrations) -----
    conc_list = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0)
    for conc in conc_list:
        for _ in range(30):                     # 30 draws per concentration
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # ----- Gaussian mixtures -----
    for _ in range(8):
        if time.time() > deadline:
            break
        seeds.append(_gaussian_mixture(n, rng))

    # ----- add a few random permutations of the first few seeds -----
    for base in seeds[:8]:
        for _ in range(2):
            if time.time() > deadline:

            perm = rng.permutation(n)
            seeds.append(base[perm])

    # previously stored result
    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Evaluate all seeds with the exact surrogate (fast)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.asarray(vec, dtype=np.float64)
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam smoothing (smooth‑max) – ~45 % of the remaining budget
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 3️⃣  Symmetrised exponentiated‑gradient polishing (exact max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        eg_deadline = deadline - 3.0          # keep a few seconds for final steps
        print("\n--- Symmetrised exponentiated‑gradient polishing ---")
        seq_eg, val_eg = _sym_exp_gradient_opt(best_seq, eg_deadline,
                                               eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 4️⃣  Multi‑scale stochastic local refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Multi‑scale stochastic local refinement ---")
        seq_loc, val_loc = _local_multi_search(best_seq, deadline, rng)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Final evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None

              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
