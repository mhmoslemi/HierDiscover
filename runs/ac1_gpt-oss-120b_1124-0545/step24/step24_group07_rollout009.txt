def proj_simplex(v):
    """Projection of vector v onto the probability simplex (sum=1, v>=0)"""
    # v is 1-D array.
    # using algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    positive = g > 0

"""
Improved optimiser for the step‑function evaluation problem.

Main ideas
----------

* **Very fine sweep of the power‑law (generalised arcsine) family** – the
  optimal shape is known to be close to (i+½)⁻ᵖ·(n‑½‑i)⁻ᵖ, but previous
  code only inspected a few values of *p*.  We now scan *p* on a
  0.005‑grid (≈800 points) and keep the best discrete candidate.
* **Projected sub‑gradient descent (PGD)** – a simple yet powerful
  method for the exact non‑smooth objective
  ``f(a)=2·n·max(conv(a,a))``.  The active convolution index supplies a
  sub‑gradient; we move a safe step length (a fraction of the largest
  step that keeps all coordinates non‑negative) and then project back
  onto the probability simplex.  The step size decays like 1/√t.
* **Exponentiated‑gradient polishing** – after a few thousand PGD
  iterations we run the “mirror‑descent” optimiser from the reference
  implementation; it is very effective when the active set of indices
  is almost stationary.
* **Final tiny annealing walk** – a cheap stochastic walk that can
  squeeze out the last few 10⁻⁴ of the objective.
* **Larger budget for the most promising seed** – we concentrate almost
  the whole 1000 s budget on the best seed found by the fine sweep.

The programme deliberately avoids any hidden closures, uses only the
standard library plus ``numpy``/``scipy`` (already available in the
execution environment) and prints a short progress report.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # only for the optional Adam stage


# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, …)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate of ``evaluate_sequence`` for a sequence that already
    sums to 1:   2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum=1)."""
    # From Duchi et al. – works in O(n log n) because of the sort.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback to uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families – we concentrate on the power‑law (generalised arcsine)
# ----------------------------------------------------------------------


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine density:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    a /= a.sum()
    return a


def _best_power_arcsine(n: int,
                        deadline: float,
                        coarse_step: float = 0.01,
                        fine_step: float = 0.002) -> tuple[np.ndarray, float]:
    """
    Exhaustive sweep over the exponent *p* of the arcsine family.
    First a coarse scan, then a few rounds of refinement.
    Returns the best sequence and its surrogate value.
    """
    # --- coarse sweep -------------------------------------------------
    p_vals = np.arange(0.10, 1.31, coarse_step)  # cover a generous interval
    best_val = float("inf")
    best_seq = None
    best_p = None

    for p in p_vals:
        if time.time() > deadline:
            break
        seq = _power_arcsine(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq, best_p = val, seq, p

    # --- refinements --------------------------------------------------
    # two refinement stages are usually enough to land in the narrow optimum
    for _ in range(2):
        if best_p is None or time.time() > deadline:
            break
        low = max(best_p - coarse_step, 0.01)
        high = best_p + coarse_step
        p_vals = np.arange(low, high + 1e-12, fine_step)
        for p in p_vals:
            if time.time() > deadline:
                break
            seq = _power_arcsine(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, best_p = val, seq, p
        # shrink the search interval a little for the next refinement
        coarse_step = fine_step
        fine_step = fine_step / 4.0

    return best_seq, best_val

# Optimisation primitives

def _pgd_optimize(start_seq: np.ndarray,
                  deadline: float,
                  max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The sub‑gradient is taken only at the *active* convolution index.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry:  g_i = 2·a_{k-i} (if defined)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # Largest step that keeps every coordinate non‑negative
        pos = g > 0
        if not np.any(pos):
            break
        step_max = np.min(a[pos] / g[pos])

        # Classic diminishing step size, scaled to be safely inside the feasible region
        step = min(step_max * 0.7, 1.5 / math.sqrt(it))

        # Gradient step and projection
        a_candidate = a - step * g
        a_candidate = _proj_simplex(a_candidate)

        cur_val = _objective_surrogate(a_candidate)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_candidate.copy()

        # Accept the candidate (it never makes the algorithm diverge)
        a = a_candidate

        # occasional cheap progress report – no heavy output inside the tight loop
        if it % 5000 == 0:
            pass  # could print, but keeping silence for speed

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    This implementation is identical to the one used in the reference
    code – we simply keep it for its proven effectiveness.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _proj_simplex(np.exp(log_a))

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk on the simplex.
    Small mass transfers i→j are occasionally accepted even if they worsen.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val
