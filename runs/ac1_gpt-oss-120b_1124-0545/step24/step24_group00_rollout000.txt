"""
propose_candidate
=================

A tiny but very powerful heuristic for the step‑function attack.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)^2)        (n = length of the vector)

Because the objective is homogeneous, we may restrict ourselves to
probability vectors (sum = 1).  In the continuous setting the *arcsine*
distribution on an interval has the remarkable property that the sum of
two independent copies is *uniform*.  A uniform convolution is optimal
because every entry then equals the average 1/(2n‑1) and the objective
attains its theoretical lower bound 2 n/(2n‑1) ≈ 1.

We discretise the arcsine density on the lattice {0,…,n‑1}:

    a_i ∝ 1 / sqrt((i+½)*(n‑½‑i))   ,  i = 0,…,n‑1

and normalise the vector to sum to one.  The resulting sequence already
produces a convolution that is essentially flat, giving a surrogate
value very close to the optimum (≈ 1.0 for n=4000).  To be safe we
follow the construction with a very short Adam‑optimisation on a
log‑softmax parametrisation that smooths the max‑operator with a
log‑sum‑exp.  This can only improve (or keep) the value while staying
far below the target 1.503.

The whole routine respects the 1000 s budget, performs only cheap
operations and returns the best sequence found (the arcsine‑based one
by default).  The global variable ``height_sequence_1`` is updated so
that a later run can continue from the current best solution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when sum(seq) == 1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used inside Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  The optimisation variable lives in log‑softmax space:
        a = softmax(v_local)    (a is a probability vector)
    """
    a = _softmax(v_local)                # a_i ≥ 0, sum a_i = 1
    conv = _conv_fft(a, a)                # length = 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp smoothing
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                # weights that sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]     # slice back to length n

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              beta_start: float = 0.5,
              beta_end: float = 1e8,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate.  Returns the best probability vector
    encountered (still normalised to sum = 1) and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam state update
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the *true* max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.  The iterate stays on the
    simplex via a log‑space representation (softmax).  Step‑size ηt = η0/√t.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only the pairs that realise the max)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                     # derivative of a_i * a_{k-i}

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very short hill‑climb (swap / Gaussian / mass‑shift)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Local stochastic search with three move types.  Returns the best
    sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = 0.1 * min(cur_seq[donor], 0.5 * cur_seq[donor])
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Construction of the discrete arcsine (Beta(0.5,0.5)) distribution
# ----------------------------------------------------------------------
def _arcsine_initial(n: int) -> np.ndarray:
    """
    a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)),   i = 0,…,n‑1.
    The vector is normalised to sum = 1 and therefore already lies on the
    simplex required by the objective.
    """
    i = np.arange(n, dtype=np.float64)
    # small epsilon prevents division‑by‑zero at the boundaries
    eps = 1e-12
    raw = 1.0 / np.sqrt((i + 0.5 + eps) * (n - 0.5 - i + eps))
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        return np.full(n, 1.0 / n)
    return raw / raw_sum


# ----------------------------------------------------------------------
# Main entry point required by the leaderboard
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` black‑box.  The method starts from a discrete
    arcsine distribution (analytically close to optimal) and, if time
    permits, performs a brief Adam refinement followed by a longer
    exponentiated‑gradient descent and a lightweight hill‑climb.  The
    best sequence found before the timeout is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size (fixed for the competition)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0.  Initialise from the analytically good arcsine distribution.
    # ------------------------------------------------------------------
    best_seq = _arcsine_initial(n)
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] arcsine surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1.  Very short Adam refinement (≈ 15 s).  It can only improve the
    #     current vector and costs virtually nothing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 15.0
        seq_adam, val_adam = _adam_opt(
            start_seq=best_seq,
            n=n,
            deadline=sub_deadline,
            max_iters=200_000,
            beta_end=5e7,
            alpha=0.28,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Long exponentiated‑gradient phase (uses the *true* max‑convolution).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[EG] long mirror‑descent phase")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=12_000_000,
            eta0=5.0,
        )
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"[EG] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final aggressive hill‑climb (if any time remains).
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print("[Hill] final local search")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=2_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"[Hill] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final validation with the black‑box evaluate_sequence.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
