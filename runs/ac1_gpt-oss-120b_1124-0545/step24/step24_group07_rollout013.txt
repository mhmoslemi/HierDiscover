"""
Search for a non‑negative step function that minimizes ``evaluate_sequence``.

The optimiser follows several stages inside the 1000 s budget:

1️⃣  **Seed generation** – uniform, triangular, an extensive sweep of the
    power‑law (arcsine) family a_i ∝ ((i+½)(n‑½‑i))^{-p} with a fine
    grid for *p*, mixtures of the best arcsine with the uniform, a few
    Beta‑densities and Dirichlet draws.  The best seed (lowest surrogate
    2·n·max(conv)) becomes the starting point.

2️⃣  **LP‑based re‑weighting** – given the current sequence *f* (sum=1)
    compute M = max(conv(f,f)).  Choose a small set of “active’’ indices
    where the convolution attains M (the K largest values).  Solve a
    linear programme

        maximise   Σ g_i
        subject to (f * g)(k) ≤ M   for every active k
                   g_i ≥ 0

    (the constraints are linear in *g*).  Normalise the LP solution
    g → g/Σg and convex‑combine it with *f* using a short line‑search
    over mixing weights *t*.  Accept the move only if the surrogate
    objective improves.  Repeat until the time budget for this stage
    expires or no improvement is observed.

3️⃣  **Exact exponentiated‑gradient polishing** – mirror‑descent on the
    true max‑convolution (sub‑gradient taken at the active index) with a
    step‑size η_t = η₀/√t.  This stage consumes the bulk of the remaining
    time.

4️⃣  **Final annealing walk** – cheap stochastic moves on the simplex
    (transfer a small fraction of mass between random coordinates) with
    a decreasing temperature, to escape shallow minima.

All intermediate evaluations use the cheap surrogate
``2·n·max(conv)`` (exact for a normalised sequence).  The best sequence
found is stored globally as ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for a sequence with sum=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Log‑space softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Triangular “tent’’ shape."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    ``start_seq`` is assumed to sum to 1.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)

        accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / temp)
        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored globally as
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.all(arr >= 0):
                s = arr.sum()
                if s > 0.0:
                    prev_vec = arr / s
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣ Assemble a rich seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # ----- fine sweep of the arcsine exponent p -------------------
    best_arc = None
    best_val_arc = np.inf
    for p in np.linspace(0.35, 0.66, 321):   # 0.001 step ≈ 321 evaluations
        vec = _power_arcsine(n, p)
        val = _objective_surrogate(vec)
        if val < best_val_arc:
            best_val_arc = val
            best_arc = vec
    seeds.append(best_arc)

    # ----- mixtures of the best arcsine shape with the uniform -----
    for w in np.linspace(0.0, 1.0, 11):
        mix = w * best_arc + (1.0 - w) * _uniform(n)
        mix = mix / mix.sum()
        seeds.append(mix)

    # ----- additional analytic families ----------------------------
    for alpha in (0.3, 0.5, 0.8, 1.2, 1.5, 2.0):
        seeds.append(_beta_asymmetric(n, alpha, alpha))

    for alpha, beta in ((0.3, 1.0), (0.5, 1.5), (1.5, 0.5), (2.0, 0.8)):
        seeds.append(_beta_asymmetric(n, alpha, beta))

    # ----- Dirichlet draws (different concentrations) -------------
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣ Initial cheap evaluation of seeds (surrogate)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
    if best_seq is None:
        raise RuntimeError("No valid seed generated before deadline.")
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ LP‑based re‑weighting (active‑set optimisation)
    # --------------------------------------------------------------
    # Allocate roughly 35 % of the remaining budget for LP steps
    remaining = deadline - time.time()
    lp_deadline = time.time() + remaining * 0.35
    cur_seq = best_seq.copy()
    cur_val = best_val
    iteration = 0
    K_active = 12                     # number of active constraints
    candidate_ts = [0.2, 0.4, 0.6, 0.8]

    print("\n--- LP‑based re‑weighting ---")
    while time.time() < lp_deadline:
        iteration += 1
        conv = _conv_fft(cur_seq, cur_seq)
        M = np.max(conv)

        # Choose the K indices with the largest convolution values
        top_idx = np.argsort(-conv)[:K_active]

        # Build the constraint matrix A (shape K_active × n)
        A_rows = []
        for k in top_idx:
            row = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            row[idx_i] = cur_seq[k - idx_i]
            A_rows.append(row)
        A_ub = np.stack(A_rows)
        b_ub = np.full(K_active, M)

        # Linear programme: maximise Σ g_i  <=> minimise –Σ g_i
        res = linprog(c=-np.ones(n),
                      A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print("[LP] linprog failed – terminating LP phase")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            # nothing useful returned
            break
        g_norm = g / sum_g

        # Simple line‑search over a few mixing weights
        best_local_seq = cur_seq
        best_local_val = cur_val
        chosen_t = 0.0
        for t in candidate_ts:
            new_seq = (1.0 - t) * cur_seq + t * g_norm
            new_seq = new_seq / new_seq.sum()
            new_val = _objective_surrogate(new_seq)
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_seq = new_seq.copy()
                chosen_t = t

        if best_local_val < cur_val - 1e-9:
            cur_seq = best_local_seq
            cur_val = best_local_val
            print(f"[LP iter {iteration:03d}] improved surrogate = {cur_val:.6f} (t={chosen_t})")
            # Update overall best
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val
        else:
            # No improvement – exit LP loop early
            print("[LP] no improvement, stopping LP phase")
            break

    # --------------------------------------------------------------
    # 3️⃣ Exact exponentiated‑gradient polishing (majority of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 3.0               # keep a few seconds for annealing
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(cur_seq, eg_deadline,
                                           eta0=6.0,
                                           max_iters=5_000_000)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best after EG")

    # --------------------------------------------------------------
    # 4️⃣ Final tiny annealing walk (last seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # --------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for any future invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
