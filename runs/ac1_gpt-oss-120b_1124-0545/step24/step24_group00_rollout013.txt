"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a lightweight multi‑stage optimiser:

1.  Build a diverse pool of initial vectors (previous best,
    several exponential decays, Dirichlet draws, a triangular shape,
    a cosine‑bell shape).
2.  For each initial vector run a *smooth* Adam optimiser on the
    log‑softmax parametrisation.  The objective is the log‑sum‑exp
    smoothing of the max‑convolution; the smoothing parameter β is
    increased from 0.5 to ~1e7.
3.  Take the best candidate and refine it with an *exact* exponentiated‑
    gradient (mirror‑descent) on the true max‑convolution.  The step size
    decays as η₀/√t.
4.  Perform an aggressive simulated‑annealing hill‑climb that uses
    three move types (swap, Gaussian perturb, targeted mass‑shift) and
    a geometric temperature schedule.
5.  If time permits, do a few short random‑restart Adam sweeps around
    the current best solution.
6.  Return the best vector found and store it in ``globals()["height_sequence_1"]``
    for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  Variable v lives in log‑softmax space; a = softmax(v).
    """
    a = _softmax(v_local)                     # probability vector, sum=1
    conv = _conv_fft(a, a)                    # length 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 5e6,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.  The iterate stays on the
    simplex via a log‑space (softmax) representation.  The step‑size
    decays as η0 / sqrt(t).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types.
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 6_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                  # Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                   # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])         # donor index (large product)
            q = int(rng.integers(0, n))               # receiver index
            if p == q:
                continue
            delta = min(cur_seq[p], 0.4 * cur_seq[p]) * 0.15   # move a modest fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs several optimisation phases and returns the best
    sequence found before the allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # problem size (feel free to change – larger n gives a tighter theoretical bound)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build diverse initial vectors
    # ------------------------------------------------------------------
    inits = []

    # a) reuse previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                inits.append(arr)
        except Exception:
            pass

    # b) exponential decays with several rates
    for rate in (0.001, 0.003, 0.006):
        inits.append(_exp_decay(n, rate=rate))

    # c) random Dirichlet draws (α = 0.3 gives moderate sparsity)
    for _ in range(3):
        dir_vec = rng.gamma(shape=0.3, scale=1.0, size=n)
        dir_vec = dir_vec / dir_vec.sum()
        inits.append(dir_vec)

    # d) simple triangular shape (large values at the ends)
    i = np.arange(n, dtype=np.float64)
    tri = np.where(i < n / 2, i + 1, n - i)          # 1,2,…,n/2,…,2,1
    tri = tri / tri.sum()
    inits.append(tri)

    # e) cosine‑bell (smooth hump in the middle)
    cos = (np.cos(np.pi * i / (n - 1)) ** 2)
    cos = cos / cos.sum()
    inits.append(cos)

    # keep only unique vectors (by content hash)
    uniq_inits = []
    seen_hashes = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen_hashes:
            uniq_inits.append(vec)
            seen_hashes.add(h)

    # ------------------------------------------------------------------
    # 2. Quick screening with Adam on the smoothed surrogate
    # ------------------------------------------------------------------
    print("[Stage 1] Quick Adam screening")
    best_seq = None
    best_val = float("inf")

    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        per_deadline = time.time() + 10.0               # about 10 s per candidate
        seq_a, val_a = _adam_opt(
            start_seq=init,
            n=n,
            deadline=per_deadline,
            max_iters=80_000,
            beta_end=1e7,
            alpha=0.3,
        )
        if val_a < best_val:
            best_val = val_a
            best_seq = seq_a.copy()
            print(f"  New best (surrogate) = {best_val:.6f} after candidate {idx + 1}")

    # ------------------------------------------------------------------
    # 3. Exact exponentiated‑gradient refinement (max‑convolution)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Exact EG refinement")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=5_000_000,
            eta0=6.0,
        )
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive hill‑climb (swap / perturb / mass‑shift)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] Hill‑climb")
        # give the hill‑climber roughly half of the remaining budget
        remaining = deadline - time.time()
        hc_deadline = time.time() + remaining * 0.55
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=hc_deadline,
            rng=rng,
            max_iters=6_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"  Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional short random‑restart Adam refinements (final polishing)
    # ------------------------------------------------------------------
    while time.time() + 8.0 < deadline:
        print("\n[Stage 4] Random‑restart Adam polish")
        noisy = best_seq + 0.008 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 7.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=40_000,
            beta_end=3e7,
            alpha=0.25,
        )
        print(f"    restart surrogate = {val_rr:.6f}")
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print("    *** improvement found! ***")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
