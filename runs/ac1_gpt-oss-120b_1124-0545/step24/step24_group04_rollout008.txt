"""
Non‑negative step‑function optimiser – enhanced version.

The optimiser builds a very diverse pool of seeds (uniform, arcsine,
triangular, Dirichlet draws with many concentrations, symmetric Beta,
Gaussian mixtures and random permutations).  It keeps the best seed
according to a smooth‑max surrogate of the true objective and then
refines it in three timed phases:

1️⃣ Adam optimisation on a smooth‑max surrogate (soft‑max of the
   convolution).  The schedule of smoothness parameters is dense and
   the inner loop is allowed to run for many iterations.

2️⃣ A fast exponentiated‑gradient (mirror‑descent) stage that works
   directly on the true max‑convolution.  The stage is executed many
   times from random restarts to increase the chance of escaping
   local minima.

3️⃣ Gradient‑guided stochastic local search.  At each step the index
   where the convolution attains its maximum is identified; a mass‑
   transfer move is proposed from a position that contributes heavily
   to that peak to a random position that contributes little.
   Only strictly improving moves are kept, with occasional swap or
   permutation proposals to diversify the walk.

All phases respect a global deadline (≈1000 s).  The best sequence is
stored in the global variable ``height_sequence_1`` and also returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence must sum to 1 (probability vector).
    Returns 2 * n * max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps (truncated to the interval).
    The resulting vector is normalised to sum 1.
    """
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)        # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (more inner work)
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    Uses a schedule of increasingly sharp smooth‑max parameters.
    Returns the best probability vector (sum = 1) seen and its exact
    surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # dense schedule of smooth‑max sharpness
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        inner_iters = 12000 if remaining > 40 else 4000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 600 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑guided stochastic local search
# ----------------------------------------------------------------------


def _gradient_local_search(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex.
    At each step we locate the index k where the convolution attains its
    maximum and transfer a small amount of mass from a point that
    contributes heavily to that peak to a random other point.
    Only strictly improving moves are kept; occasional swap moves are
    attempted to diversify the search.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline:
        # compute full convolution and locate its maximum
        conv = _conv_fft(cur, cur)
        k = int(np.argmax(conv))

        # contribution of each coordinate to conv(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        contrib = cur[idx] * cur[rev]

        # choose a source index i with probability proportional to its contribution
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        # choose a destination index uniformly (could be improved)
        j = rng.integers(0, n)
        if i == j:
            continue

        # step size – decays roughly as 1/sqrt(iteration)
        scale = max(0.001, 0.5 / math.sqrt(it + 1))
        delta = cur[i] * rng.random() * scale
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < -1e-12:
            continue

        new_val = _surrogate(new)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if cur_val < best_val:
                best_seq, best_val = cur.copy(), cur_val

        # occasional swap move (helps break symmetry)
        if it % 250 == 0:
            i2, j2 = rng.integers(0, n, size=2)
            if i2 != j2:
                swap_seq = cur.copy()
                swap_seq[i2], swap_seq[j2] = swap_seq[j2], swap_seq[i2]
                swap_val = _surrogate(swap_seq)
                if swap_val < cur_val:
                    cur, cur_val = swap_seq, swap_val
                    if cur_val < best_val:
                        best_seq, best_val = cur.copy(), cur_val

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (≈1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any)
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1️⃣ Build a large, diverse pool of seeds   (all normalised to sum = 1)
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with a wide range of concentrations
    for _ in range(20):
        conc = rng.uniform(0.01, 2.0)
        seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta shapes (important extreme shapes)
    for a in (0.05, 0.10, 0.20, 0.30, 0.50, 0.80, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Random Gaussian mixtures – they often create interesting edge‑heavy shapes
    for _ in range(8):
        seeds.append(_gaussian_mixture(n, rng))

    # Include a few random permutations of a few seeds (breaks symmetry)
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 2️⃣ Evaluate seeds – keep the best (surrogate objective)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Adam smoothing (smooth‑max) – allocate ≈20 % of remaining time
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 30.0:
            adam_deadline = time.time() + 0.20 * remaining
            print("\n--- Adam smoothing (smooth max) ---")
            seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
            print(f"[Adam] best surrogate = {val_adam:.6f}")
            if val_adam < best_val:
                best_seq, best_val = seq_adam.copy(), val_adam
                print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient polishing with many random restarts
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 60.0:
            eg_total_deadline = time.time() + 0.35 * remaining
            print("\n--- Exponentiated‑gradient polishing (multiple restarts) ---")
            while time.time() < eg_total_deadline:
                # pick a start vector: 30 % chance to re‑use current best,
                # otherwise a fresh Dirichlet draw
                if rng.random() < 0.3:
                    start_vec = best_seq.copy()
                else:
                    conc = rng.uniform(0.01, 1.5)
                    start_vec = _random_dirichlet(n, conc, rng)

                slice_deadline = min(time.time() + 5.0,
                                     eg_total_deadline)
                seq_eg, val_eg = _exp_gradient_opt(start_vec, n,
                                                   slice_deadline,
                                                   eta0=5.0)
                if val_eg < best_val:
                    best_seq, best_val = seq_eg.copy(), val_eg
                    print(f"[EG] new best surrogate = {best_val:.6f}")

            # a final longer polishing on the current best
            polishing_deadline = deadline - 5.0
            if polishing_deadline > time.time():
                print("\n--- Final exponentiated‑gradient polishing ---")
                seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                                   polishing_deadline,
                                                   eta0=4.0)
                print(f"[EG‑final] surrogate = {val_eg:.6f}")
                if val_eg < best_val:
                    best_seq, best_val = seq_eg.copy(), val_eg
                    print("   >>> Updated global best (final EG)")

    # --------------------------------------------------------------
    # 5️⃣ Gradient‑guided stochastic local refinement (mass‑transfer, swaps)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Gradient‑guided stochastic local refinement ---")
        seq_loc, val_loc = _gradient_local_search(best_seq, n, deadline, rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 6️⃣ Final evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
