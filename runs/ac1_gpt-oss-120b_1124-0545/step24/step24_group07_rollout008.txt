Implement project_onto_simplex(v): algorithm by Duchi et al. Projects onto simplex S = {x >= 0, sum=1}.

```
def _project_simplex(v):
    # v is array (float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max j s.t. u_j - cssv_j / j > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _proj_gradient_opt(start_seq, deadline, eta0=0.2):

        idx_i = np.arange(i_start, i_end+1)

"""
search for a non‑negative step function that minimises ``evaluate_sequence``.

Strategy
========
1️⃣  Build a diverse pool of analytic seeds: uniform, triangular,
    a dense sweep of the power‑arcsine family, a grid of discrete Beta(α,β)
    shapes and several Dirichlet draws.  All vectors are normalised to sum = 1.

2️⃣  Cheap evaluation (exact max‑convolution) picks the few most promising
    seeds.

3️⃣  **Projected gradient descent (PGD)** on the *exact* objective.
    The sub‑gradient of ``max(conv(a,a))`` is 2·a shifted by the active index.
    After a step we project back to the probability simplex by clipping
    negatives and renormalising – a fast O(n) operation.
    The step‑size follows η₀/√t.  Time is split uniformly among the top seeds.

4️⃣  A short simulated‑annealing walk (tiny perturbations of the mass)
    finishes the polishing.

5️⃣  The final sequence is evaluated with the supplied ``evaluate_sequence``.
    The best sequence is stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    The input is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    a /= a.sum()
    return a


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    a /= a.sum()
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (clipping + renormalise)
# ----------------------------------------------------------------------


def _simplex_clip(v: np.ndarray) -> np.ndarray:
    """
    Fast projection onto the simplex:
        - set negatives to zero
        - renormalise to sum = 1 (if the sum becomes zero, fall back to uniform)
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _pgd_opt(start_seq: np.ndarray,
             deadline: float,
             eta0: float = 0.2,
             eval_every: int = 3000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  max(conv(a,a))  with projection onto the simplex.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution term
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size diminishing like 1/√t
        eta = eta0 / math.sqrt(it)

        # gradient step followed by cheap simplex projection
        a = _simplex_clip(a - eta * g)

        # occasional exact evaluation
        if it % eval_every == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[PGD] iter {it:7d} – surrogate = {cur_val:.6f}")

    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny simulated‑annealing walk (final polish)
# ----------------------------------------------------------------------


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing walk on the simplex.
    Moves transfer a random fraction of mass from i → j.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored globally as
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣ Assemble a rich seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # Power‑law (generalised arcsine) sweep
    for p in np.linspace(0.1, 2.0, 20):
        seeds.append(_power_arcsine(n, float(p)))

    # Grid of Beta shapes (α,β) – the most influential family
    alphas = np.linspace(0.2, 2.0, 20)
    betas = np.linspace(0.2, 2.0, 20)
    for alpha in alphas:
        for beta in betas:
            seeds.append(_beta_asymmetric(n, float(alpha), float(beta)))

    # Random Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣ Cheap evaluation of all seeds (exact max‑convolution)
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating seed pool (exact max‑conv) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        if (idx + 1) % 50 == 0 or idx == len(seeds) - 1:
            print(f" evaluated {idx + 1}/{len(seeds)} seeds – best so far = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(4, len(seed_infos))                   # a few elite seeds
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seed sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Projected Gradient Descent (exact max) on the elite seeds
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve a few seconds for the final annealing walk
        reserve = 5.0
        pgd_time = max(0.0, remaining - reserve)
        time_per_seed = pgd_time / max(1, len(top_seeds))

        print("\n--- Projected‑gradient descent on elite seeds ---")
        for i, seed_vec in enumerate(top_seeds):
            sub_deadline = time.time() + time_per_seed
            print(f"\n[PGD] start seed #{i + 1}/{len(top_seeds)} – "
                  f"time budget ≈ {time_per_seed:.1f}s")
            seq_opt, val_opt = _pgd_opt(seed_vec, sub_deadline,
                                        eta0=0.2, eval_every=4000)
            print(f"[PGD] seed #{i + 1}: surrogate = {val_opt:.6f}")
            if val_opt < best_val:
                best_val = val_opt
                best_seq = seq_opt.copy()
                print("   >>> Updated global best after PGD")

    # --------------------------------------------------------------
    # 3️⃣ Final tiny annealing walk (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk (polish) ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # --------------------------------------------------------------
    # 4️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
