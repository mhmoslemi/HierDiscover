def _topk_exp_gradient_opt(start_seq, n, deadline, max_iters, eta0, top_k):

- a = start_seq
- log_a = np.log(a + eps)
- best_seq = a, best_val = _objective_surrogate(a)
- for it in 1..max_iters:
  - if time > deadline: break
  - conv = _conv_fft(a,a)
  - # compute top_k indices
  - if top_k >= len(conv): use all.
  - top_idxs = np.argpartition(-conv, top_k-1)[:top_k]  # yields indices of top entries
  - gradient = np.zeros_like(a)
  - for k in top_idxs:
    - i_start = max(0, k-(n-1))
    - i_end = min(n-1, k)
    - idx_i = np.arange(i_start, i_end+1)
    - rev_idx = k - idx_i
    - gradient[idx_i] += 2.0 * a[rev_idx]
  - eta = eta0 / np.sqrt(it)
  - log_a = log_a - eta * gradient
  - a = _softmax(log_a)
  - cur_val = _objective_surrogate(a)
  - update best.

def _topk_exp_gradient_opt(start_seq, n, deadline, max_iters=10_000_000, eta0=10.0, top_k=5):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    L = 2*n -1
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        conv = _conv_fft(a, a)
        # find top_k offsets
        if top_k >= L:
            top_idxs = np.arange(L)
        else:
            # argpartition for speed
            top_idxs = np.argpartition(-conv, top_k-1)[:top_k]
        grad = np.zeros_like(a)
        for k in top_idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            grad[idx_i] += 2.0 * a[rev_idx]
        eta = eta0 / np.sqrt(it)
        log_a = log_a - eta * grad
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # optionally early termination if little improvement? skip.
    return best_seq, best_val

"""
Improved search for a low‑value step‑function.

The routine builds on the previous multi‑stage optimizer but adds a
*top‑K subgradient* method: instead of using only the single offset that
realises the current maximum convolution, we aggregate the sub‑gradients
of the K largest offsets.  This drives the whole convolution profile
towards a flatter shape and usually yields a lower max‑convolution value.
The algorithm consists of

1.  A cheap parametric sweep (exponential decay and power‑law shapes)
    to obtain a good starting point.
2.  Several short *top‑K EG* runs from diverse initialisations.
3.  A long refinement on the current best candidate.
4.  An aggressive hill‑climb (swap / Gaussian perturb / mass‑shift)
    as in the reference code.
5.  Optional short Adam polishing steps if time remains.

All optimisation works on the simplex (soft‑max / log‑space
representation) so the sum of the coefficients is always 1.
The final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation (smooth max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Smoothed max‑convolution objective with its gradient.
    The variable lives in log‑softmax space; a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              beta_start: float = 0.5,
              beta_end: float = 5e7,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient on the *true* max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 8_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution (single active offset).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Top‑K aggregated sub‑gradient exponentiated gradient optimizer
# ----------------------------------------------------------------------
def _topk_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 10_000_000,
                           eta0: float = 8.0,
                           top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the sum of sub‑gradients of the K largest
    convolution entries.  This tends to flatten the whole convolution
    profile rather than focussing on a single offset.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    L = 2 * n - 1

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # pick the K largest offsets (if K > L we simply use all)
        if top_k >= L:
            idxs = np.arange(L)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        grad = np.zeros_like(a)

        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            grad[idx_i] += 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as the reference implementation)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 4_500_000) -> tuple[np.ndarray, float]:
    """Local search with swap / Gaussian perturb / targeted mass‑shift."""
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:          # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                           # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])   # donor
            q = int(rng.integers(0, n))          # random receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Searches for a step‑function (non‑negative vector) that minimises
    `evaluate_sequence`.  The algorithm combines a cheap parametric sweep,
    several runs of a top‑K exponentiated‑gradient optimiser, a final
    long refinement, and an aggressive hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # safety margin

    # ------------------------------------------------------------------
    # problem size (fixed to 4000 as in the reference)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Very cheap parametric sweep (exponential & power‑law)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    def _eval_and_update(a):
        nonlocal best_seq, best_val
        a = a / a.sum()
        val = _objective_surrogate(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"    param‑sweep new best ≈ {val:.6f}")

    # exponential decays
    for rate in np.logspace(-4, -1, num=20):
        a = _exp_decay(n, rate)
        _eval_and_update(a)

    # power‑law shapes a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.1, 5.0, num=30):
        i = np.arange(n, dtype=np.float64) + 1.0
        a = i ** (-p)
        _eval_and_update(a)

    print(f"[Stage 1] Parametric sweep best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Gather diverse initial vectors
    # ------------------------------------------------------------------
    inits = []

    # a) previous best (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                inits.append(arr)
        except Exception:
            pass

    # b) the best found by the sweep
    if best_seq is not None:
        inits.append(best_seq.copy())

    # c) a few random Dirichlet draws (α = 0.5 for sparsity)
    for _ in range(3):
        dir_vec = rng.gamma(shape=0.5, scale=1.0, size=n)
        dir_vec = dir_vec / dir_vec.sum()
        inits.append(dir_vec)

    # d) a couple of fresh exponential decays with different rates
    inits.append(_exp_decay(n, rate=0.002))
    inits.append(_exp_decay(n, rate=0.003))

    # keep unique vectors (hash based)
    uniq = []
    seen = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen:
            uniq.append(vec)
            seen.add(h)

    # ------------------------------------------------------------------
    # 3. Short top‑K EG runs on each candidate (≈ 20 s per candidate)
    # ------------------------------------------------------------------
    print("[Stage 2] Short top‑K EG runs")
    for idx, init in enumerate(uniq):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(20.0, remaining)
        print(f"  Candidate {idx + 1}/{len(uniq)} (budget {sub_deadline - time.time():.1f}s)")

        seq_tk, val_tk = _topk_exp_gradient_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            max_iters=5_000_000,
            eta0=10.0,
            top_k=7,
        )
        if val_tk < best_val:
            best_val, best_seq = val_tk, seq_tk.copy()
            print(f"    New best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long refinement on the current best (top‑K EG)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] Long top‑K EG refinement")
        seq_tk, val_tk = _topk_exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=15_000_000,
            eta0=6.0,
            top_k=5,
        )
        if val_tk < best_val:
            best_val, best_seq = val_tk, seq_tk.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Aggressive hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 4] Hill‑climb")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=4_500_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional short Adam polishing (if time permits)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 5] Random‑restart Adam polishing")
        noisy = best_seq + 0.005 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s
        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=90_000,
            beta_end=1e8,
            alpha=0.28,
        )
        print(f"    restart surrogate = {val_rr:.6f}")
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print("    *** improvement found! ***")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store the best sequence for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
