"""
Step‑function optimiser – enhanced multi‑restart gradient + annealed local search.

The optimiser builds a large, diverse seed pool (uniform, arcsine, triangular,
symmetric beta, symmetric power‑law, several Gaussian shapes and Dirichlet
draws).  The best seed is refined in three timed stages:

1️⃣ Adam optimisation on a smooth‑max surrogate with an extended β‑schedule
   (the surrogate approaches the true max as β grows).  The Adam schedule
   uses many inner iterations and a slowly decaying learning rate.

2️⃣ Several restarts of exponentiated‑gradient (mirror descent) on the *exact*
   max‑convolution.  Each restart perturbs the current best vector with a tiny
   Dirichlet noise and runs the sub‑gradient update with a generous step size
   η₀ = 20, decreasing as 1/√t.

3️⃣ An aggressive simulated‑annealing local walk on the simplex.  Three move
   types are used – small mass‑transfer, full swaps and global permutations –
   and moves are accepted with probability exp(−Δ/τ) where the temperature τ
   decays geometrically.  The mass‑transfer scale σ also decays, allowing
   fine‑grained refinement.

All phases respect the overall deadline (≈ 1000 s).  The best sequence found
is stored in the global ``height_sequence_1`` variable and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate (the true objective when the sum of the vector is 1):
    2 * n * max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """
    Symmetric Gaussian bump centred at the middle of the interval.
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape (edges heavier):
    a_i ∝ (i+0.5)^(-p) + (n-i-0.5)^(-p)
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    left = (i) ** (-p)
    right = (n - i + 0.5) ** (-p)
    a = left + right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps (truncated to the interval)."""
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a

def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    A fine β‑schedule (smooth‑max → true max) together with many inner
    iterations yields a high‑quality approximation of the optimum.
    Returns the best probability vector (sum = 1) seen and its exact
    surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # Very dense β‑schedule, extending to extremely large values.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0,
        100_000_000.0, 200_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # More iterations while we still have time.
        remaining = deadline - time.time()
        inner_iters = 18000 if remaining > 30 else 6000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp trick)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Learning rate slowly decays with iteration count
            lr = 0.12 / (1.0 + 0.00001 * it)
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val

# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Multiple restarts are performed outside this function; it simply
    runs until the deadline.
    Returns the best vector seen and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _local_multi_search(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk on the simplex.
    Three move types are mixed:
      – tiny mass transfer i → j,
      – full swap a[i] ↔ a[j],
      – random permutation of the whole vector.
    Moves are accepted with a Metropolis‑type probability exp(−Δ/τ).
    The temperature τ and the mass‑transfer scale σ decay gradually.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    # annealing schedule
    temp = 0.02
    temp_decay = 0.9996

    sigma = 0.12          # relative size of mass‑transfer steps
    sigma_decay = 0.997

    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # decide which move to try
        r = rng.random()
        if r < 0.55:          # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            # guard against negative due to rounding
            if new[i] < 0.0:
                continue
        elif r < 0.85:        # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:                 # global permutation (rare)
            perm = rng.permutation(n)
            new = cur[perm]

        new_val = _surrogate(new)
        delta_val = new_val - cur_val

        # Metropolis acceptance
        if delta_val < 0 or rng.random() < math.exp(-delta_val / max(temp, 1e-12)):
            cur, cur_val = new, new_val

        # anneal temperature and sigma every few thousand attempts
        if attempts % 3000 == 0:
            temp = max(1e-6, temp * temp_decay)
            sigma = max(0.001, sigma * sigma_decay)

    return cur, cur_val

    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing on smooth‑max surrogate (major time share)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Multiple restarts of exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break
            # perturb current best with tiny Dirichlet noise
            eps = 1e-3
            noise = rng.dirichlet(np.full(n, eps))
            start = (1 - eps) * best_seq + eps * noise
            start /= start.sum()
            pol_deadline = deadline - 3.0   # leave a few seconds for later phases
            print(f"\n--- Exponentiated‑gradient polishing (restart {r + 1}) ---")
            seq_eg, val_eg = _exp_gradient_opt(start, n, pol_deadline, eta0=20.0)
            print(f"[EG {r + 1}] surrogate = {val_eg:.6f}")
            if val_eg < best_val:
                best_seq, best_val = seq_eg.copy(), val_eg
                print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Aggressive stochastic local refinement with simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Multi‑scale stochastic local refinement (annealed) ---")
        seq_loc, val_loc = _local_multi_search(best_seq, n, deadline, rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
