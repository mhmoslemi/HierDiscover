"""
Non‑negative step‑function optimiser – enhanced version.

The optimiser builds a very diverse pool of seeds (uniform, arcsine,
triangular, Dirichlet, symmetric & asymmetric Beta, Gaussian mixtures,
permutations) and keeps the best according to a fast surrogate
objective 2·n·max(conv(a,a)).  The best seed is then refined by

* Adam on a smooth‑max surrogate (more inner work and a longer β‑schedule);
* Exponentiated‑gradient polishing on the true max‑convolution;
* An aggressive multi‑scale stochastic walk (mass‑transfer, swaps,
  whole‑vector permutations);
* A deterministic “flatten‑max’’ phase that moves mass from indices
  that contribute most to the current maximal convolution value to
  indices that contribute the least (gradient‑guided mass transfer);
* A short simulated‑annealing walk that accepts occasional uphill moves.

All phases respect the global deadline (≈ 1000 s).  The best sequence is
stored in the global variable ``height_sequence_1`` and also returned as
a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(seq,seq)).
    ``seq`` is assumed to sum to 1 (probability vector).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _asymmetric_beta(n: int, alpha: float, beta: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Beta with separate left/right shape parameters."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps, truncated to the interval."""
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser in the log‑softmax space on a smooth‑max surrogate."""
    # initialise logits
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a dense schedule that ends at a very sharp smooth‑max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0, 100_000_000.0
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # adapt number of inner iterations to remaining time
        remaining = deadline - time.time()
        inner_iters = 18000 if remaining > 30 else 6000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a  (mirror‑gradient)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # length n slice
            grad_a = grad_weighted * (2.0 * n)            # factor 2·n from surrogate

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 400 == 0:
                cur_val = _surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive stochastic local search (mass transfer, swaps, permutations)
# ----------------------------------------------------------------------


def _local_multi_search(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk on the simplex.
    Alternates three kinds of proposals:
      * small mass transfer from i → j,
      * full swap of values a[i] ↔ a[j],
      * random permutation of the whole vector.
    Only strictly improving moves are kept.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)

    sigma_start = 0.15      # relative size of mass‑transfer steps
    sigma_end = 0.003
    sigma = sigma_start
    decay = 0.96            # slower geometric decay
    block_iters = 8000      # attempts per sigma value

    while time.time() < deadline:
        # ---- mass‑transfer block ----
        for _ in range(block_iters):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            if new[i] < 0.0:
                continue
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- swap block ----
        for _ in range(block_iters // 2):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- occasional global permutation ----
        if rng.random() < 0.10:   # roughly one permutation every 10 blocks
            perm = rng.permutation(n)
            new = cur[perm]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # decay sigma, keep looping
        sigma = max(sigma_end, sigma * decay)
        if sigma <= sigma_end + 1e-12:
            # after reaching minimum sigma we keep swapping only
            block_iters = max(2500, block_iters // 2)

    return cur, cur_val


# ----------------------------------------------------------------------
# Deterministic “flatten‑max’’ phase (gradient‑guided mass transfer)
# ----------------------------------------------------------------------


def _flatten_max(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a fraction of the mass from the index that contributes
    most to the current maximal convolution value to the index that contributes
    the least.  This reduces the peak of the convolution in a greedy fashion.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    step_factor = 0.25          # start with 25 % of the source mass
    min_step = 1e-5

    while time.time() < deadline:
        conv = _conv_fft(cur, cur)
        max_val = float(np.max(conv))
        k_max = int(np.argmax(conv))

        # sub‑gradient of the *active* term (same as in the EG step)
        g = np.zeros(n, dtype=np.float64)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_max - idx_i
        g[idx_i] = 2.0 * cur[rev_idx]   # 2·a_{k_max-i}

        # pick source (high) and destination (low) indices
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))

        if cur[i_src] < 1e-12:
            break

        delta = min(cur[i_src] * step_factor, cur[i_src])
        if delta <= 0.0:
            break

        new = cur.copy()
        new[i_src] -= delta
        new[i_dst] += delta

        new_val = _surrogate(new)

        if new_val < cur_val:
            # successful improvement – keep it and slightly increase step size
            cur, cur_val = new, new_val
            step_factor = min(0.5, step_factor * 1.08)
        else:
            # no improvement – shrink step size
            step_factor *= 0.65
            if step_factor < min_step:
                break

        # occasional random perturbation to escape tiny plateaus
        if rng.random() < 0.02:
            perm = rng.permutation(n)
            cur = cur[perm]
            cur_val = _surrogate(cur)

    return cur, cur_val


# ----------------------------------------------------------------------
# Short simulated‑annealing walk (allows occasional uphill moves)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very simple annealing: propose small random mass‑transfer moves and
    accept them with probability exp(-(Δ)/(T)) where the temperature T
    decays linearly to zero.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    t0 = 0.02 * cur_val      # initial temperature proportional to current value
    t_end = 1e-6
    total_time = deadline - time.time()
    if total_time <= 0.0:
        return cur, cur_val

    while time.time() < deadline:
        elapsed = time.time() - (deadline - total_time)
        T = t0 * max(0.0, 1.0 - elapsed / total_time) + t_end

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() * 0.05) * cur[i]   # up to 5 % of source mass
        if delta <= 0.0 or cur[i] < delta:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _surrogate(new)
        diff = new_val - cur_val
        if diff < 0.0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur, cur_val = new, new_val

    return cur, cur_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (≈ 1000 s) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Retrieve a previously stored solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a rich pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta
    for a in (0.02, 0.04, 0.07, 0.10, 0.15, 0.20, 0.30, 0.50, 0.70, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Asymmetric Beta (helps break symmetry)
    for a, b in ((0.1, 0.5), (0.5, 0.1), (0.05, 0.3), (0.3, 0.05)):
        seeds.append(_asymmetric_beta(n, a, b, rng))

    # Random Gaussian mixtures
    for _ in range(8):
        seeds.append(_gaussian_mixture(n, rng))

    # Permutations of a few seeds (break symmetries)
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate seeds – keep the best (according to the surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing (smooth‑max) – about 40 % of remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.40 * remaining
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing (sub‑gradient on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = deadline - 5.0
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline, eta0=8.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Aggressive stochastic local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Multi‑scale stochastic local refinement ---")
        seq_loc, val_loc = _local_multi_search(best_seq, n, deadline, rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Gradient‑guided “flatten‑max’’ phase
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Gradient‑guided flatten‑max phase ---")
        flatten_deadline = deadline - 2.0
        seq_flat, val_flat = _flatten_max(best_seq, n, flatten_deadline, rng)
        print(f"[Flatten] surrogate = {val_flat:.6f}")
        if val_flat < best_val:
            best_seq, best_val = seq_flat.copy(), val_flat
            print("   >>> Updated global best (flatten)")

    # ------------------------------------------------------------------
    # 7️⃣ Light simulated‑annealing walk (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing final walk ---")
        sa_deadline = deadline
        seq_sa, val_sa = _simulated_annealing(best_seq, n, sa_deadline, rng)
        print(f"[SA] surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq, best_val = seq_sa.copy(), val_sa
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 8️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
