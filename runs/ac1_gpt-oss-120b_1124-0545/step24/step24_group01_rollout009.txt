   surrogate ``2·n·max(conv(a,a))`` and keep the best few (default 5).

3. **Smooth max‑convolution optimisation** – for each selected seed run a
   mirror‑descent (exponentiated‑gradient) on a *smooth* approximation of
   the max‑convolution:
     *   ``F(a) = (1/β)·log( Σ_k exp(β·(a*a)[k]) )``
        with an increasing temperature β (start small → smooth,
        end large → close to the true max).
     *   Gradient of F is computed analytically:
           grad = 2·corr(a , softmax_β(conv(a,a)))
        where ``corr`` is the linear correlation and the softmax
        weights are the normalised exponentials of the convolved vector.
     *   Updates are performed in log‑space (mirrored variables) with
        Adam‑style momentum and a diminishing step‑size ηₜ = η₀/√t.
     *   The best iterate (lowest exact surrogate) is remembered.
   The loop runs until its time‑slice expires or the target value
   (≈1.503) is reached.

4. **Greedy refinement** – after the smooth optimiser a cheap greedy
   mass‑transfer step (as used in the previous state‑of‑the‑art) is applied
   while time permits.

5. **Local stochastic walk** – a final short random walk on the simplex
   (small swaps of mass) is performed to polish the solution.

6. The best sequence found is evaluated with the supplied
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used by the scoring function:
        2·n·max(conv(seq, seq))   (seq assumed to sum to 1)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _harmonic(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1) (normalised)."""
    a = 1.0 / (np.arange(1, n + 1, dtype=np.float64))
    return a / a.sum()


def _sqrt_diff(n: int) -> np.ndarray:
    """a_i ∝ √(i+1) - √i  (telescoping to √n)."""
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max‑convolution objective.
    Returns (grad, current_max), where ``grad`` is dF/da and ``current_max``
    is the exact max‑convolution value (needed for the surrogate).
    """
    n = len(a)
    conv = _conv_fft(a, a)                     # length 2n-1
    # softmax weights over the convolution values
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # stability
    w = exp_scaled / exp_scaled.sum()          # shape (2n-1,)

    # correlation g_i = 2 * Σ_j a_j * w_{i+j}
    # implement via convolution of reversed a with w
    corr = _conv_fft(a[::-1], w)               # length 3n-2
    grad = 2.0 * corr[n - 1: n - 1 + n]        # first n entries
    current_max = np.max(conv)
    return grad, current_max


def _smooth_exp_gradient(start_seq: np.ndarray,
                         deadline: float,
                         eta0: float = 0.5,
                         beta0: float = 5.0,
                         beta_max: float = 400.0,
                         beta_factor: float = 1.001,
                         beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smoothed max‑convolution.
    Returns the best probability vector encountered (summing to 1) and its
    exact surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)
    mom = np.zeros_like(logit)

    beta = beta0
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1

        grad, cur_max = _gradient_smooth(a, beta)
        cur_val = float(2.0 * n * cur_max)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # early exit when we beat the target bound
            if best_val < 1.503:
                break

        # mirror (log‑space) gradient
        avg = np.dot(grad, a)
        grad_log = a * (grad - avg)

        # Adam‑style momentum (no bias correction for speed)
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # diminishing stepsize
        eta = eta0 / math.sqrt(it)

        # update and map back to simplex
        logit = logit - eta * mom
        a = _softmax(logit)

        # increase temperature β slowly
        beta = min(beta_max, beta * beta_factor)

        # occasional exact surrogate check (cheap, already have cur_val)
        # (no extra work here)

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a tiny fraction of mass from the coordinate with the
    largest sub‑gradient to the one with the smallest (if it improves the
    exact surrogate).  Returns the best sequence seen.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex that keeps improving moves.
    """
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the allotted time
    is stored globally as ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Load a previously stored solution (if it exists and matches length)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Assemble a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n), _harmonic(n), _sqrt_diff(n)]

    # Symmetric Beta(α,α) with a spread of α values
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 7),   # very U‑shaped
        np.linspace(0.6, 1.4, 7),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with a variety of concentrations
    for conc in (0.02, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No seeds could be evaluated within the time budget.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth exponentiated‑gradient optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Smooth EG optimisation (exact max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f"  EG run #{sidx} (time slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _smooth_exp_gradient(seed_vec, per_seed_deadline)
            print(f"   → surrogate after EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (EG)")

            # ----------------------------------------------------------
            # 2️⃣‑b  Greedy refinement of the EG result (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 3️⃣  Final random local refinement (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=250_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 4️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
