def _adam_eg_opt(start_a, n, deadline, max_iters=200_000, eta0=6.0):
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active indices tolerance relative
        tol = 1e-12
        active = np.where(conv >= max_val - tol * max_val)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size
        else:
            # fallback: use full gradient (conv derivative)
            # but gradient small? Use all.
            pass
        
        # Adam update on log_a
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = eta0 / np.sqrt(it)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _lp_refine(seq, n, deadline, max_iters=30):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = _conv_fft(a, a).max()
        # Build sparse matrix
        rows = []
        cols = []
        data = []
        for i in range(n):
            f_i = a[i]
            # rows k = i...i+n-1
            start = i
            end = i + n
            rows.extend(range(start, end))
            cols.extend([i] * n)
            data.extend([f_i] * n)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Solve LP maximize sum(b) s.t. A @ b <= M, b >=0
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                    method='highs', options={'presolve': True, 'time_limit': min(10.0, deadline - time.time())})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sumb = b.sum()
        if sumb <= 0:
            break
        g = b / sumb
        # compute new max conv after mixing
        t = 0.2  # mixing factor
        # combine
        new_a = (1-t) * a + t * g
        new_a = new_a / new_a.sum()
        new_val = _objective_surrogate(new_a)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            print(f"[LP refine] iter {it}: improved surrogate {best_val:.8f}")
        else:
            # maybe try purely g if better
            g_val = _objective_surrogate(g)
            if g_val < best_val:
                best_val = g_val
                best_seq = g.copy()
                a = g
                print(f"[LP refine] iter {it}: improvement by g {best_val:.8f}")
            else:
                # no improvement; break
                print(f"[LP] iteration {it}: no improvement")
                break
    return best_seq, best_val

"""
Advanced step‑function optimizer (launches in < 1000 s).

Algorithm outline
-----------------
1. **Problem size** – we work with a moderate length (default 4000).  The
   objective for a distribution with sum = 1 is simply
   `obj = 2·n·max(conv(a,a))`.  All our internal optimisers keep the vector
   on the ℓ₁‑simplex by working with log‑softmax parameters.

2. **Multiple seeds** – the previous best (if any), the classic
   inverse‑sqrt shape and a very sparse Dirichlet draw are normalised and
   evaluated quickly.  The most promising seeds are taken forward.

3. **Smooth L‑BFGS** – a succession of smoothed‐max surrogates
   `softmax_β(conv)` is minimised with L‑BFGS‑B on the log‑softmax
   parameters.  The schedule of β grows exponentially; each stage runs a
   few thousand iterations.

4. **Adam‑style exponentiated‑gradient (EG)** – the exact non‑smooth
   objective is tackled by mirror‑descent on the log‑softmax variables.
   The sub‑gradient is obtained from the (few) *active* convolution indices
   where the maximum is attained.  Adam‑style momentum and a √t learning‑rate
   schedule give fast convergence.

5. **Linear‑programming refinement** – after the EG polish we solve a tiny
   LP:
        maximise   Σ b_j
        subject to  (a * b)_k ≤ M   for the *active* k,
                    b_j ≥ 0,
   where `M = max(conv(a,a))`.  The LP increases the total mass that can be
   placed under the same maximal convolution bound; after normalising the
   solution we mix it back with the current distribution.  Only the rows
   that are near‑tight are kept, so the LP is tiny (≤ 200 constraints).

6. **Final polish** – a last long EG run on the best candidate.

All stages respect the wall‑clock deadline (default 1000 s).  The best
sequence is stored in the global variable ``height_sequence_1`` and returned
as a list of Python floats.
"""

import time
import math
import numpy as np

# Optional imports – fall back gracefully if SciPy is missing

except Exception:  # pragma: no cover

    _sparse = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers (working with ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(a,a)) for ∑a = 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())


def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) – used inside L‑BFGS
# ----------------------------------------------------------------------
def _softmax_max(conv: np.ndarray, beta: float) -> float:
    """β‑softmax approximation of max(conv)."""
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    return shift + np.log(exp_beta.sum()) / beta


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_a: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule,
                       max_iter_per_beta: int = 2500):
    """Minimise the smoothed max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        # No SciPy – skip L‑BFGS entirely.
        return start_a.copy(), _objective_surrogate(start_a)

    # initialise log‑softmax parameters
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # smooth max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed objective w.r.t. a
            w = exp_beta / sum_exp                     # ∂softmax/∂conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # convolution a * w
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # chain rule

            # back‑prop to logits (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # early‑stop if we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-14,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient (EG) optimiser on the exact max.
# ----------------------------------------------------------------------
def _adam_eg_opt(start_a: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 500_000,
                 eta0: float = 8.0,
                 tol_rel: float = 1e-12):
    """Mirror descent (log‑softmax) with Adam momentum on the true max‑conv."""
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set – indices whose value is within tol_rel of the max
        active = np.where(conv >= max_val - tol_rel * max_val)[0]

        # sub‑gradient (average over active k)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size
        else:
            # fallback – full gradient (should never happen)
            g = 2.0 * _conv_fft(a, a)[n - 1:n - 1 + n]

        # Adam update on log‑softmax variables
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _lp_refine(seq: np.ndarray,
               n: int,
               deadline: float,
               max_outer_iters: int = 8,
               t_mix: float = 0.2):
    """
    One‑step linear‑programming refinement based on the active convolution
    constraints.  Returns an improved sequence if one is found.
    """
    if _opt is None or _sparse is None:
        # No LP capability – just return the input.
        return seq.copy(), _objective_surrogate(seq)

    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_outer_iters + 1):
        if time.time() > deadline:
            break

        conv_aa = _conv_fft(a, a)
        M = conv_aa.max()

        # ----- active constraints (top 0.1 % of the peak) -----
        rel_tol = 1e-3
        active = np.where(conv_aa >= M * (1 - rel_tol))[0]
        if active.size > 200:
            # keep at most 200 most‑relevant indices
            topk = np.argpartition(-conv_aa, 200 - 1)[:200]
            active = topk

        # ----- build sparse A (only active rows) -----
        rows = []
        cols = []
        data = []
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            length = i_end - i_start + 1
            rows.extend([k] * length)
            cols.extend(range(i_start, i_end + 1))
            data.extend(a[i_start:i_end + 1])
        A = _sparse.csr_matrix((data, (rows, cols)),
                               shape=(2 * n - 1, n))

        # ----- solve LP maximise Σ b_j subject to A·b ≤ M, b≥0 -----
        c = -np.ones(n)                      # minimise -Σb
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # give the LP a tiny internal time limit to stay safe
        lp_opts = {"presolve": True}
        try:
            res = _opt.linprog(c,
                               A_ub=A,
                               b_ub=b_ub,
                               bounds=bounds,
                               method="highs",
                               options=lp_opts)
        except Exception as e:  # pragma: no cover
            print(f"[LP] exception: {e}")
            break

        if not res.success:
            print(f"[LP] iteration {it} failed – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        g = b / sum_b   # normalise to sum = 1

        # ----- try pure replacement first -----
        g_val = _objective_surrogate(g)
        if g_val + 1e-12 < best_val:
            best_val = g_val
            best_seq = g.copy()
            a = g
            print(f"[LP] iter {it}: pure replacement improved to {best_val:.8f}")
            continue

        # ----- otherwise mix with current a -----
        new_a = (1.0 - t_mix) * a + t_mix * g
        new_a = new_a / new_a.sum()
        new_val = _objective_surrogate(new_a)

        if new_val + 1e-12 < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            print(f"[LP] iter {it}: mixed improvement to {best_val:.8f}")
        else:
            # no progress – stop early
            print(f"[LP] iter {it}: no improvement, terminating refinement")
            break

    return best_seq, best_val

        share = max(5.0, remaining * 0.20)
        sub_deadline = time.time() + share

        print(f"\n[Seed {si}/{len(top_seeds)}] budget ≈ {share:.1f}s")

        # 1) smooth L‑BFGS
        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=2500,
        )
        print(f"   L‑BFGS surrogate = {val_lbfgs:.8f}")

        # 2) Adam‑EG polish
        if time.time() < sub_deadline:
            seq_eg, val_eg = _adam_eg_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=400_000,
                eta0=6.0,
                tol_rel=1e-12,
            )
            print(f"   Adam‑EG polish = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # 3) LP‐based refinement
        if time.time() < sub_deadline:
            seq_lp, val_lp = _lp_refine(
                seq_eg,
                n=n,
                deadline=sub_deadline,
                max_outer_iters=6,
                t_mix=0.2,
            )
            print(f"   LP refinement   = {val_lp:.8f}")
        else:
            seq_lp, val_lp = seq_eg, val_eg

        # pick the best of the three phases
        cand_seq, cand_val = seq_lp, val_lp
        if cand_val > val_eg:
            cand_seq, cand_val = seq_eg, val_eg
        if cand_val > val_lbfgs:
            cand_seq, cand_val = seq_lbfgs, val_lbfgs

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing (Adam‑EG) on the globally best sequence
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        final_seq, final_val = _adam_eg_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=4.0,
            tol_rel=1e-12,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Evaluate with the provided black‑box function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
