    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000

    # Build seeds list.
    seeds = []

    # previous if exists
    previous = globals().get("height_sequence_1")
    if previous is not None and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        seeds.append(prev_arr / prev_arr.sum())

    # classic
    seeds.append(_inverse_sqrt(n))

    # random Dirichlet
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))

    # powerlaw grid
    for p in np.linspace(0.55, 1.0, 10):
        for offset in np.linspace(0.0, 5.0, 6):
            g = (np.arange(n, dtype=np.float64) + offset + 1) ** (-p)
            seq = g / g.sum()
            seeds.append(seq)

    # exponential grid
    for lam in [0.001, 0.003, 0.005, 0.008, 0.01]:
        g = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = g / g.sum()
        seeds.append(seq)

    # Mixed grid (power+exp)
    for p in [0.6, 0.7, 0.8]:
        for lam in [0.001, 0.003, 0.005]:
            g = (np.arange(n, dtype=np.float64) + 1) ** (-p) * np.exp(-lam * np.arange(n, dtype=np.float64))
            seq = g / g.sum()
            seeds.append(seq)

    # Ensure all seeds normalized
    seeds = [s / s.sum() for s in seeds if s.sum() > 0]

    # Evaluate seeds quickly using surrogate
    evaluated = []
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        evaluated.append((s, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:max(1, min(5, len(evaluated)))]]

    # Now run optimization for each top seed
    global_best_seq = None
    global_best_val = math.inf

    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        # allocate share for this seed: equally split remaining time
        remaining = deadline - time.time()
        share = max(5.0, remaining / (len(top_seeds) - idx + 1))
        sub_deadline = time.time() + share

        print(f"[Seed {idx}/{len(top_seeds)}] budget {share:.1f}s")
        # run L-BFGS tuning with larger iter

"""
Step‑function optimizer – low‑dimensional shape search + fine‑grained EG.

The idea
--------
1. Search a few compact families of monotone decreasing shapes
   (power‑law, exponential, and mixed power‑law × exponential).
   Each family is described by ≤3 parameters, so a dense grid
   (hundreds of candidates) can be evaluated cheaply using the exact
   surrogate  `2·n·max(conv(a,a))` (the true objective for sum(a)=1).

2. Keep the best few shapes as seeds and run the deep smoothed L‑BFGS
   optimiser from the previous state‑of‑the‑art code.  The schedule of
   β‑values is extended to larger values, and the iteration budget per
   β is slightly increased.

3. After the quasi‑Newton phase a long exponentiated‑gradient (EG)
   polishing phase (Adam‑style momentum) is performed.  This directly
   minimises the exact max‑convolution and is the main source of final
   improvement.

4. A final EG polishing using the remaining wall‑clock budget refines the
   best candidate further.

The routine respects the 1000 s budget, stores the best sequence in the
global variable ``height_sequence_1`` and returns it as a plain list.
"""

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (same as true objective for Σa = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a sequence with sum == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (used as seeds)
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse random draw on the simplex."""
    return rng.dirichlet(np.full(n, conc))


def _power_law(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + offset + 1.0) ** (-p)
    return a / a.sum()


def _exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    return a / a.sum()


def _mixed(n: int, p: float, lam: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset+1)^{-p}·exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + offset + 1.0) ** (-p) * np.exp(-lam * idx)
    return a / a.sum()


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser (identical to the previous version)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
):
    """Minimise the smoothed max‑convolution objective using L‑BFGS‑B."""
    if _opt is None:                # pragma: no cover
        # SciPy not available – fall back to a tiny EG loop.
        return _exp_gradient_opt(start_a, n, deadline)

    v = np.log(start_a + 1e-30)     # log‑softmax parameters
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (EG) optimiser – main polishing engine
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 8.0,
    tol: float = 1e-10,
):
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max‑convolution (average over active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The routine combines a low‑dimensional shape grid, a deep smoothed L‑BFGS
    stage and a long exponentiated‑gradient polishing phase.  The best
    sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the time budget
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Gather seeds (previous best + analytic families + randomness)
    # ------------------------------------------------------------------
    seeds = []

    # 1) previous best (if it matches the current length)
    previous = globals().get("height_sequence_1")
    if previous is not None and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0:
            seeds.append(prev_arr / prev_arr.sum())

    # 2) classic inverse‑sqrt shape
    seeds.append(_inverse_sqrt(n))

    # 3) a very sparse Dirichlet draw
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))

    # 4) power‑law grid (p ∈ [0.55, 1.0], offset ∈ [0,5])
    for p in np.linspace(0.55, 1.0, 10):
        for offset in np.linspace(0.0, 5.0, 6):
            seeds.append(_power_law(n, p, offset))

    # 5) exponential grid
    for lam in [0.001, 0.003, 0.005, 0.008, 0.010]:
        seeds.append(_exponential(n, lam))

    # 6) mixed power‑law × exponential
    for p in [0.6, 0.7, 0.8]:
        for lam in [0.001, 0.003, 0.005]:
            seeds.append(_mixed(n, p, lam))

    # Normalise all seeds (protect against accidental zero‑sum)
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # ------------------------------------------------------------------
    # Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    # keep the best few (up to 5) for the expensive optimisation phases
    top_seeds = [vec for vec, _ in evaluated[:min(5, len(evaluated))]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:len(top_seeds)], 1):
        print(f" Seed #{i}: surrogate = {v:.8f}")

    # ------------------------------------------------------------------
    # β‑schedule for the smoothed L‑BFGS phase (extended to very large β)
    # ------------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 5_000_000.0, 10_000_000.0,
        50_000_000.0, 100_000_000.0, 1_000_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # ------------------------------------------------------------------
    # Optimise each top seed – the time budget is split evenly
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # give the current seed a fair share of the remaining budget
        share = max(5.0, remaining / (len(top_seeds) - idx + 1))
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx}/{len(top_seeds)}] allocated ≈ {share:.1f}s")

        # 1) deep L‑BFGS on the smoothed surrogate
        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3000,
        )
        print(f"  → L‑BFGS surrogate = {val_lbfgs:.8f}")

        # 2) EG polishing (the heavy lifting)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=600_000,
                eta0=10.0,
                tol=1e-12,
            )
            print(f"  → EG polish   surrogate = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # choose the better of the two phases
        candidate_seq, candidate_val = (
            (seq_eg, val_eg) if val_eg < val_lbfgs else (seq_lbfgs, val_lbfgs)
        )

        if candidate_val < global_best_val:
            global_best_val = candidate_val
            global_best_seq = candidate_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Final intensive polishing on the overall best (uses remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        final_seq, final_val = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=6.0,
            tol=1e-12,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, "
            f"true evaluate_sequence = {true_val:.8f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
