"""
Improved step‑function optimizer.

The algorithm builds a very large and diverse seed set (analytic families,
window functions, fine grids of Beta‑ and power‑exponential families),
quickly evaluates the simple surrogate 2·n·max(conv) and keeps the few best
shapes.  For each of these top seeds we run a short exponentiated‑gradient
followed by a smooth‑max L‑BFGS refinement to obtain a reasonably good
candidate.  The best candidate is then fed to a full‑pipeline optimisation
(Exponentiated‑gradient → L‑BFGS → flattening gradient → simulated‑annealing)
which consumes most of the remaining time budget.  Finally, any time left
is spent on a lightweight coordinate‑descent (mass‑transfer) polishing step.
The best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional scipy imports
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.signal import windows as _sigwin
except Exception:  # pragma: no cover
    _sigwin = None

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max onto the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (exact surrogate after normalising sum=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for L1‑normalised seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# optimisation primitives
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    L‑BFGS‑B on the smooth‑max surrogate
        2·n·logsumexp(β·conv)/β .
    A schedule of β values is used; each stage is warm‑started from the
    previous solution.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1_000_0000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 3_000_000,
                         eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent minimising the variance of the self‑convolution
    (Σ (conv‑avg)²).  This encourages a flat convolution.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation SA with a linear cooling schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Full four‑stage pipeline (EG → L‑BFGS → flatten → SA) on a copy of
    *start_vec*.  All stages respect *deadline*.  Returns the best sequence
    and its surrogate value.
    """
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=14.0,
        tol=1e-11,
    )

    # 2) smooth L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=6000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=6_000_000,
            eta0=10.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=500_000,
            start_sigma=0.0008,
            T_start=6e-3,
            T_end=4e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# coordinate‑descent polishing (mass transfer)
# ----------------------------------------------------------------------
def _coordinate_descent_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a small amount of mass from one
    index to another while keeping the total sum fixed.  Accept only strict
    improvements (no uphill moves) – this is a simple hill‑climb that can
    still shave off a few percent from the surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    start = time.time()
    while time.time() < deadline:
        # linear reduction of the transfer size
        frac = (time.time() - start) / (deadline - start + 1e-12)
        eps = 0.004 * (1.0 - frac) + 1e-8   # between 4e‑3 and ~1e‑8
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose a signed transfer delta (from i to j)
        delta = (rng.random() - 0.5) * 2.0 * eps
        if delta > 0.0:
            delta = min(delta, a[i])          # cannot go negative
        else:
            d = -delta
            d = min(d, a[j])
            delta = -d

        if abs(delta) < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective_surrogate(a_new)
        if val < cur_val:
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# rich seed generation
# ----------------------------------------------------------------------
def _generate_rich_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a large, diverse set of L1‑normalised candidate vectors."""
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    # beta shapes – finer grid
    alphas = np.linspace(0.2, 3.0, 15)
    betas = np.linspace(0.2, 3.0, 15)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # two‑parameter power‑exponential family – dense sweep
    p_vals = np.linspace(0.3, 1.4, 13)
    lam_vals = np.linspace(0.0, 0.02, 11)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    # random Dirichlet draws
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # window functions from SciPy (if available)
    if _sigwin is not None:
        try:
            seeds.append(_sigwin.hann(n))
        except Exception:
            pass
        try:
            seeds.append(_sigwin.hamming(n))
        except Exception:
            pass
        try:
            seeds.append(_sigwin.blackman(n))
        except Exception:
            pass
        for beta in [2, 4, 6, 8, 10, 12]:
            try:
                w = _sigwin.kaiser(n, beta)
                seeds.append(w)
            except Exception:
                pass
        for alpha in [0.2, 0.5, 0.8]:
            try:
                w = _sigwin.tukey(n, alpha)
                seeds.append(w)
            except Exception:
                pass
        if hasattr(_sigwin, 'nuttall'):
            try:
                seeds.append(_sigwin.nuttall(n))
            except Exception:
                pass

    # mixture of uniform with a few base shapes
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed versions (to explore asymmetry)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include previous global best if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                    seeds.append(prev_arr)
        except Exception:
            pass

    # small Gaussian perturbations of the current global best (if any)
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # make sure everything is non‑negative and normalised
    clean = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
    return clean


# ----------------------------------------------------------------------
# quick optimisation for a seed (EG + L‑BFGS only)
# ----------------------------------------------------------------------
def _quick_opt(start_seq: np.ndarray,
               n: int,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a lightweight optimisation: short EG run followed by smooth‑max
    L‑BFGS.  Returns the refined sequence and its surrogate value.
    """
    # short EG
    seq_eg, _ = _exp_gradient_opt_multi(
        start_seq.copy(),
        n,
        deadline,
        max_iters=300_000,
        eta0=12.0,
        tol=1e-11,
    )

    if time.time() >= deadline:
        return seq_eg, _objective_surrogate(seq_eg)

    # brief L‑BFGS with a modest schedule
    seq_lbfgs = _lbfgs_smooth_optimize(
        seq_eg,
        n,
        deadline,
        beta_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 200.0, 500.0, 1000.0],
        max_iter_per_beta=2500,
    )
    return seq_lbfgs, _objective_surrogate(seq_lbfgs)

# main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000
    print(f"\n=== Optimiser start (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Generate a rich seed set and evaluate surrogate quickly
    # ------------------------------------------------------------------
    seeds = _generate_rich_seeds(n, rng)
    print(f"Generated {len(seeds)} seeds, evaluating ...")

    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2. Quick optimisation of the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    print(f"\n[Initial] best surrogate = {global_best_val:.6f}")

    # allocate a slice of the remaining budget for these cheap runs
    remaining = deadline - time.time()
    if remaining > 20.0 and top_k > 1:
        per_seed_deadline = time.time() + remaining * 0.4 / top_k
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq_opt, val_opt = _quick_opt(seed_vec, n, per_seed_deadline, rng)
            print(f"[Quick‑opt] seed #{idx+1} got surrogate {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()

    # ------------------------------------------------------------------
    # 3. Deep optimisation of the current best (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        deep_deadline = deadline
        print("\n[Deep optimisation] starting full pipeline")
        seq_opt, val_opt = _full_optimisation(global_best_seq, n, deep_deadline, rng)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** Updated global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final coordinate‑descent polishing (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("\n[Final polish] coordinate‑descent")
        seq_fin, val_fin = _coordinate_descent_opt(global_best_seq, n, final_deadline, rng)
        if val_fin < global_best_val:
            global_best_seq = seq_fin
            global_best_val = val_fin
            print(f"   *** Polished surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
