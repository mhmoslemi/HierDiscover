def local_search(seq, n, deadline):
    a = seq.copy()
    conv = conv_fft(a,a)
    cur_val = 2 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()

    # parameters
    sigma = 0.1
    start_temp = 0.005 # maybe 0.01 * cur_val
    temp = start_temp
    decay = 0.99995  # per iteration
    accept = 0
    iter = 0

    while time.time() < deadline:
        i = rng.integers(0,n)  # source
        if a[i] <= 1e-15: continue
        j = rng.integers(0,n)  # dest, ensure not i
        if j == i: continue

        # choose delta
        delta = sigma * a[i] * rng.random()
        if delta <= 0: continue

        # compute candidate convolution
        cand_conv = conv.copy()
        cand_conv[i:i+n] -= 2.0*delta*a
        cand_conv[j:j+n] += 2.0*delta*a
        idx2i = 2*i
        idx2j = 2*j
        idxij = i+j
        cand_conv[idx2i] += delta*delta
        cand_conv[idx2j] += delta*delta
        cand_conv[idxij] -= 2.0*delta*delta

        cand_val = 2.0 * n * cand_conv.max()

        delta_val = cand_val - cur_val

        # acceptance
        accept_move = False
        if delta_val < 0:
            accept_move = True
        else:
            if rng.random() < math.exp(-delta_val / (temp + 1e-12)):
                accept_move = True

        if accept_move:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accept += 1

            # update best if improved
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # maybe recompute conv from scratch every some steps
            if accept % 2000 == 0:
                conv = conv_fft(a, a)
                cur_val = 2.0 * n * conv.max()

        # update sigma adaptively: maybe decrease slowly
        if accept % 5000 == 0 and accept > 0:
            sigma = max(0.001, sigma * 0.98)
            temp = max(1e-6, temp * 0.99)

        iter += 1

    return best_a, best_val

def _local_search_simulated(start_seq, n, deadline, rng):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    sigma = 0.15 # start larger
    target_prob = 0.2 # with this prob we use targeted
    temp_start = max(0.01, cur_val * 0.01) # maybe 0.01 of value
    temp = temp_start
    decay = 0.9999
    accepted = 0
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # decide move type
        if rng.random() < target_prob:
            # targeted move
            p = int(np.argmax(conv))
            i_start = max(0, p - (n - 1))
            i_end = min(p, n - 1)
            idxs = np.arange(i_start, i_end + 1)
            rev_idxs = p - idxs
            contrib = a[idxs] * a[rev_idxs]
            i = idxs[np.argmax(contrib)]
            # choose destination random not i
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            i = rng.integers(0, n)
            if a[i] <= 1e-15:
                continue
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        if a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental conv update
        cand_conv = conv.copy()
        cand_conv[i:i+n] -= 2.0 * delta * a
        cand_conv[j:j+n] += 2.0 * delta * a
        cand_conv[2*i] += delta * delta
        cand_conv[2*j] += delta * delta
        cand_conv[i + j] -= 2.0 * delta * delta

        cand_val = 2.0 * n * np.max(cand_conv)
        delta_val = cand_val - cur_val

        accept_move = False
        if delta_val < 0:
            accept_move = True
        else:
            # simulated annealing probability
            if rng.random() < math.exp(-delta_val / (temp + 1e-12)):
                accept_move = True

        if accept_move:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            if accepted % 2000 == 0:
                conv = _conv_fft(a, a)
                cur_val = 2.0 * n * np.max(conv)

            if accepted % 5000 == 0:
                sigma = max(0.005, sigma * 0.95)
                temp = max(1e-6, temp * 0.95)

    return best_a, best_val

def _optimize_for_n(n, deadline, rng):
    # Build seeds (list of vectors)
    seeds = [_uniform(n), _arcsine(n), _triangular(n), _spike(n)]
    for conc in (0.05,0.1,0.2,0.5,1.0,2.0):
        seeds.append(_random_dirichlet(n, conc, rng))
    for alpha in (0.02,0.05,0.07,0.1,0.15,0.2,0.25,0.3,0.35,0.4,0.45,0.5,0.6,0.8,1.0,1.2,1.5,2.0):
        seeds.append(_beta_distribution(n, alpha))

    # incorporate previous solution upsampled if appropriate
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        # upsample if length differs
        if prev_arr.size != n:
            # interpolate
            old_n = prev_arr.size
            old_pos = np.linspace(0,1,old_n)
            new_pos = np.linspace(0,1,n)
            upsampled = np.interp(new_pos, old_pos, prev_arr)
            if upsampled.sum() > 0:
                upsampled = upsampled / upsampled.sum()
                seeds.append(upsampled)
        else:
            seeds.append(prev_arr / prev_arr.sum())

    best_seq = None
    best_val = np.inf

    # evaluate seeds
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        seq = vec / vec.sum()
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
    # Adam smoothing
    if best_seq is not None and time.time() < deadline:
        # allocate time for Adam ~30% of remaining
        remain = deadline - time.time()
        adam_deadline = time.time() + min(0.3 * remain, 30.0) # at most 30 sec
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
    # EG polishing
    if best_seq is not None and time.time() < deadline:
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline, eta0=5.0)
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
    # Local search
    if best_seq is not None and time.time() < deadline:
        seq_loc, val_loc = _local_search_simulated(best_seq, n, deadline, rng)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
    return best_seq, best_val

"""
Enhanced step‑function optimiser (new version).

The optimiser builds a diverse pool of probability vectors (Beta,
Dirichlet, arcsine, spike, uniform, …).  For each problem size *n* it
runs three phases:

1. **Adam on a smooth‑max surrogate** – fast gradient descent in the
   log‑softmax space with a continuation of the smoothness parameter β.
2. **Exact exponentiated‑gradient polishing** – mirror‑descent on the
   true max‑convolution using the sub‑gradient of the active index.
3. **Simulated‑annealing local walk** – an incremental O(n) update of the
   self‑convolution while moving a small amount δ from one coordinate to
   another.  The walk accepts improving moves and, with a slowly cooling
   temperature, also occasional uphill moves; every few thousand accepted
   moves the convolution is recomputed from scratch to avoid drift.
   A small fraction of the moves are *targeted*: the current peak index
   of the convolution is identified and mass is taken from the pair that
   contributes most to that peak.

The optimiser is run on one or two different lengths (default 4000 and,
if time permits, 2 × that size).  The best sequence found (minimum of
`evaluate_sequence`) is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of ``2·n·max(conv(a,a))``.
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a fairly fine schedule of smoothness parameters
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        1e5, 2e5, 5e5,
        1e6, 2e6, 5e6,
        1e7,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # more inner iterations while we have time left
        max_iter_beta = 8000 if (deadline - time.time()) > 30 else 2000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = np.exp(logit - np.max(logit))
            a = a / a.sum()                     # softmax → probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)            # outer factor 2·n

            # convert to gradient on the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = np.exp(logit - np.max(logit))
    final_a = final_a / final_a.sum()
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a - np.max(log_a))
        a = a / a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing incremental local refinement
# ----------------------------------------------------------------------


def _local_search_simulated(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            sigma_start: float = 0.15,
                            target_prob: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Fast greedy‑annealing walk on the simplex.
    Randomly moves a small amount δ from one coordinate to another.
    ``target_prob`` controls the fraction of moves that are *targeted*:
    they focus on the currently maximal convolution entry.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    sigma = sigma_start
    # temperature for uphill moves – start modest, decay slowly
    temp = max(1e-6, 0.02 * cur_val)
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        # ----- choose move type -----
        if rng.random() < target_prob:
            # --- targeted move ------------------------------------------------
            p = int(np.argmax(conv))                     # current peak index
            i_start = max(0, p - (n - 1))
            i_end = min(p, n - 1)
            idxs = np.arange(i_start, i_end + 1, dtype=int)
            rev_idxs = p - idxs
            contrib = a[idxs] * a[rev_idxs]
            i = idxs[np.argmax(contrib)]                 # index that contributes most
        else:
            # --- random source index -----------------------------------------
            i = rng.integers(0, n)
            if a[i] <= 1e-15:
                continue

        # choose a distinct destination index
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # propose a transfer size
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental convolution update (O(n)) -------------------------
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        cand_conv[2 * i] += delta * delta
        cand_conv[2 * j] += delta * delta
        cand_conv[i + j] -= 2.0 * delta * delta

        cand_val = 2.0 * n * np.max(cand_conv)
        delta_val = cand_val - cur_val

        # ----- acceptance test (simulated annealing) ----------------------
        accept_move = False
        if delta_val < 0.0:
            accept_move = True
        else:
            if rng.random() < math.exp(-delta_val / (temp + 1e-12)):
                accept_move = True

        if accept_move:
            # apply move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # re‑compute from scratch occasionally (numeric drift)
            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                cur_val = 2.0 * n * np.max(conv)
                last_recalc = accepted

            # cooling schedule
            if accepted % 5000 == 0:
                sigma = max(0.005, sigma * 0.94)
                temp = max(1e-8, temp * 0.94)

    return best_a, best_val

def _optimize_for_n(n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the three‑phase optimisation (Adam → EG → local) for a fixed length ``n``.
    Returns the best probability vector (sums to 1) and its exact surrogate value.
    """
    # ----- 1️⃣ build a seed pool (all vectors sum to 1) -----
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    beta_alphas = (
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00,
    )
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # reuse previous solution if it exists (interpolate if lengths differ)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size > 0 and prev_arr.sum() > 0.0:
            if prev_arr.size == n:
                seeds.append(prev_arr / prev_arr.sum())
            else:
                # linear interpolation to the new size
                old_n = prev_arr.size
                old_pos = np.linspace(0.0, 1.0, old_n)
                new_pos = np.linspace(0.0, 1.0, n)
                up = np.interp(new_pos, old_pos, prev_arr)
                if up.sum() > 0.0:
                    seeds.append(up / up.sum())

    best_seq = None
    best_val = np.inf

    # ----- 2️⃣ seed evaluation -----
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        seq = vec / vec.sum()
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # ----- 3️⃣ Adam smoothing (smooth‑max) -----
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # give at most 30 % of the remaining budget to Adam
        adam_deadline = time.time() + min(0.30 * remaining, 30.0)
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()

    # ----- 4️⃣ Exact exponentiated‑gradient polishing -----
    if best_seq is not None and time.time() < deadline:
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline, eta0=5.0)
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()

    # ----- 5️⃣ Intensive local refinement (simulated‑annealing) -----
    if best_seq is not None and time.time() < deadline:
        seq_loc, val_loc = _local_search_simulated(
            best_seq, n, deadline, rng,
            sigma_start=0.15,
            target_prob=0.15,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found is stored in the global variable ``height_sequence_1``
    and also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size – default 4000, can be overridden via kwargs
    n0 = int(kwargs.get("n", 4000))
    ns = [n0]

    # If time permits, also try a larger size (2×).  Upper bound 10000 to stay safe.
    n_big = n0 * 2
    if n_big <= 10000:
        ns.append(n_big)

    best_seq = None
    best_val = np.inf

    for n in ns:
        if time.time() > deadline:
            break

        # allocate a fair share of remaining time to this N
        remaining_time = deadline - time.time()
        # simple split: equal share per remaining size
        remaining_ns = len(ns) - ns.index(n)
        time_for_this = max(10.0, remaining_time / remaining_ns)
        sub_deadline = time.time() + time_for_this
        if sub_deadline > deadline:
            sub_deadline = deadline

        print(f"\n--- Optimising for n = {n} (budget ≈ {time_for_this:.1f}s) ---")
        seq_n, val_n = _optimize_for_n(n, sub_deadline, rng)
        print(f"[Result] n = {n} → surrogate = {val_n:.6f}")

        if val_n < best_val:
            best_val, best_seq = val_n, seq_n.copy()
            print(f"   >>> Updated global best (n = {n})")

    if best_seq is None:
        # Should never happen, but fallback to a uniform sequence of length n0
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)

    # Final official evaluation
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
