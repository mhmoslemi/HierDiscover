"""
Advanced search for a non‑negative step‑function minimising ``evaluate_sequence``.

Algorithm
---------
1. Initialise a diverse pool of seeds (uniform, arcsine, triangular,
   symmetric and asymmetric Beta, random Dirichlet draws).  All seeds are
   normalised to ℓ₁‑norm = 1 and the cheap surrogate
   ``2·n·max(conv(a,a))`` (i.e. the denominator of ``evaluate_sequence``)
   is evaluated.  The best few seeds are kept.

2. For each selected seed run *exponentiated‑gradient* (mirror descent on the
   simplex) with Adam‑style momentum on the **exact** max‑convolution.
   Several hyper‑parameter settings are tried (different learning‑rate
   prefactors and momentum values).  The best iterate of each run is retained.

3. Fine‑grained deterministic refinement:
   Repeatedly shift a tiny amount of mass from the coordinate that contributes
   most to the current max‑convolution (largest sub‑gradient) to the one that
   contributes the least.  The shift is evaluated *exactly* by an O(n) update
   of the convolution (no FFT per move).  The step‑size is steadily reduced
   when no improvement is observed, allowing the algorithm to settle into a
   very flat convolution.

4. A short stochastic walk (random pairwise swaps) is performed as a final
   polish.

5. The final candidate is evaluated with the provided ``evaluate_sequence``
   function and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    ``seq`` is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density (α,β > 0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives

def _exp_gradient_momentum(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           beta_mom: float = 0.9) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) with Adam‑style momentum on the
    exact max‑convolution.  ``start_seq`` must already be normalised to ℓ₁=1.
    Returns the best sequence encountered and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # mirror‑gradient (log‑space)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam‑style momentum (no bias‑correction for speed)
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # update in log‑space and map back to the simplex
        logit = logit - eta * mom
        a = _softmax(logit)

        # periodic check
        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
        # early exit if already very good
        if best_val < 1.503:
            break

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _fine_refine(start_seq: np.ndarray,
                 deadline: float,
                 init_step: float = 0.02,
                 min_step: float = 0.001,
                 stall_patience: int = 150_000) -> Tuple[np.ndarray, float]:
    """
    Deterministic pairwise mass transfer that directly updates the convolution.
    The method repeatedly moves a tiny fraction of mass from the coordinate with
    the largest sub‑gradient (most responsible for the current peak) to the
    coordinate with the smallest sub‑gradient, accepting only improvements.
    The step size is halved when a long streak without improvement occurs.
    """
    a = start_seq.copy()
    n = len(a)
    conv = _conv_fft(a, a)                       # length 2n‑1
    best_seq = a.copy()
    best_max = conv.max()
    step = init_step
    no_improve = 0
    rng = np.random.default_rng()

    L = len(conv)  # = 2*n‑1

    while time.time() < deadline:
        # active index and its sub‑gradient
        k = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:                     # fallback to random pair
            i_max = rng.integers(0, n)
            i_min = rng.integers(0, n)
            if i_max == i_min:
                continue

        delta = step * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        # --- incremental update of the convolution ---------------------------------
        cand_conv = conv.copy()
        # linear part: -2*delta*shift(a,i_max) + 2*delta*shift(a,i_min)
        cand_conv[i_max:i_max + n] -= 2.0 * delta * a
        cand_conv[i_min:i_min + n] += 2.0 * delta * a

        # quadratic corrections (at most three entries)
        idx2 = 2 * i_max
        if idx2 < L:
            cand_conv[idx2] += delta * delta
        idx2 = 2 * i_min
        if idx2 < L:
            cand_conv[idx2] += delta * delta
        idx2 = i_max + i_min
        if idx2 < L:
            cand_conv[idx2] -= 2.0 * delta * delta
        # ---------------------------------------------------------------------------

        new_max = cand_conv.max()
        if new_max < best_max - 1e-13:           # strict improvement
            a[i_max] -= delta
            a[i_min] += delta
            conv = cand_conv
            best_max = new_max
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= stall_patience:
                # reduce step size and reset counter
                step = max(min_step, step * 0.5)
                no_improve = 0

    return best_seq, best_max


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> Tuple[np.ndarray, float]:
    """
    Very light stochastic walk – moves are accepted only if they improve the
    exact surrogate.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget is
    stored globally as ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored solution (if any) – useful for warm starts
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Betas
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),
        np.linspace(0.6, 1.4, 9),
        np.linspace(1.5, 2.5, 7)
    ])
    for a in alphas:
        seeds.append(_beta_symmetric(n, float(a)))

    # asymmetric Betas – random draws for α≠β
    for _ in range(12):
        a = 10 ** rng.uniform(-2, 0.5)          # α in (0.01, ~3)
        b = 10 ** rng.uniform(-2, 0.5)
        seeds.append(_beta_asymmetric(n, a, b))

    # Dirichlet draws with a range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    # initialise the global best
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Momentum exponentiated‑gradient optimisation (exact max‑conv)
    # ------------------------------------------------------------------
    if time.time() < deadline and top_seeds:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Momentum EG optimisation (exact max‑conv) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            # use a couple of different hyper‑parameter settings
            for (eta0, mom) in [(5.0, 0.9), (8.0, 0.85), (3.0, 0.95)]:
                seq, val = _exp_gradient_momentum(seed_vec,
                                                  per_seed_deadline,
                                                  eta0=eta0,
                                                  beta_mom=mom)
                print(f"  EG run #{sidx} (η0={eta0}, β={mom}) → surrogate = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, seq.copy()
                    print("   >>> Updated global best (EG)")
                # quick greedy refinement after each EG run
                seq2, val2 = _fine_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (fine‑refine)")

    # ------------------------------------------------------------------
    # 3️⃣ Final stochastic walk (light random search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=300_000,
                                                sigma=0.01)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 4️⃣ Aggressive fine‑grained deterministic refinement (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Aggressive deterministic refinement ---")
        seq_fine, max_conv_fine = _fine_refine(best_seq, deadline,
                                               init_step=0.02,
                                               min_step=0.0008,
                                               stall_patience=100_000)
        print(f"[Fine] final max convolution = {max_conv_fine:.9f}")
        if max_conv_fine < best_val / (2 * n):
            best_seq = seq_fine.copy()
            best_val = 2 * n * max_conv_fine
            print("   >>> Updated global best (aggressive)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starting next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
