"""
Improved 4000‑dimensional step‑function optimizer.

The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

with non‑negative entries.  The algorithm proceeds in several
independent phases, each respecting the global time budget (default
1000 s):

1. **Diverse initialisations** – exponential decays, power‑law
   decays, triangular / sinusoidal profiles and many Dirichlet draws.
   The surrogate objective (sum(a)=1) is evaluated once for every
   candidate and the best few are kept.

2. **Fast Adam warm‑up + exact exponentiated‑gradient (EG) refinement**
   for each top candidate.  Adam works on a log‑softmax parametrisation
   of the smooth log‑sum‑exp surrogate; the EG phase uses the true
   max‑convolution sub‑gradient and a multiplicative update
   ``a_i ← a_i·exp(-η·g_i)`` with η ∝ 1/√t.

3. **Long EG run on the current best** – almost the whole remaining
   time is spent here, allowing millions of cheap updates.

4. **Random‑restart EG polishing** – occasional Gaussian perturbations
   of the best sequence followed by a short EG run, accepted only
   when they improve the surrogate.

5. **Aggressive simulated‑annealing hill‑climb** as a final local‑search
   stage (swap / Gaussian perturb / targeted mass‑shift moves).

All phases keep the best sequence seen so far.  At the end the exact
black‑box ``evaluate_sequence`` function is called once to obtain the
true score, the sequence is stored in the global ``height_sequence_1``
(for possible continuation) and returned.

The implementation uses only NumPy and the provided ``evaluate_sequence``
function, runs on a single process (≈2 CPU cores are allowed) and respects
the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate objective for a probability vector (sum = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _power_law_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  Variable v lives in log‑softmax space; a = softmax(v)
    is a probability vector with sum = 1.
    """
    a = _softmax(v_local)            # probability vector
    conv = _conv_fft(a, a)           # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                    # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp            # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)        # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              beta_start: float = 0.5,
              beta_end: float = 1e8,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 12_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.
    The iterate stays on the simplex via a multiplicative update
    a_i ← a_i·exp(-η·g_i) with η = η0/√t.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry (active offset = max_idx)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]          # 2·a_{k*‑i}

        eta = eta0 / math.sqrt(it)

        # multiplicative update (mirror descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types.
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move that tries to directly reduce the active convolution
    entry.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                   # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                    # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])   # index that contributes most
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = min(cur_seq[donor], 0.5 * cur_seq[donor]) * 0.1
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs a number of optimiser phases and returns the best
    sequence found before the allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size (feel free to change, but keep it reasonable)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a large pool of diverse initial vectors
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                candidates.append(arr)
        except Exception:
            pass

    # b) exponential decays (log‑spaced rates)
    for rate in np.logspace(-4, -1.3, num=15):
        candidates.append(_exp_decay(n, float(rate)))

    # c) power‑law decays
    for p in np.linspace(0.1, 3.0, num=20):
        candidates.append(_power_law_decay(n, float(p)))

    # d) simple deterministic shapes
    #   – linear decreasing
    lin = np.linspace(n, 1, n)
    candidates.append(lin / lin.sum())
    #   – triangular (mountain)
    half = n // 2
    tri = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    tri = tri[:n].astype(np.float64)
    candidates.append(tri / tri.sum())
    #   – sinusoidal (positive half‑wave)
    x = np.arange(n, dtype=np.float64) + 0.5
    sin = np.sin(np.pi * x / n)
    sin = np.maximum(sin, 0.0)
    candidates.append(sin / sin.sum())

    # e) random Dirichlet draws with several concentration parameters
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            vec = rng.gamma(shape=alpha, scale=1.0, size=n)
            vec = vec / vec.sum()
            candidates.append(vec)

    # keep only unique vectors (hash based)
    uniq = []
    seen = set()
    for vec in candidates:
        h = hash(vec.tobytes())
        if h not in seen:
            uniq.append(vec)
            seen.add(h)

    # ------------------------------------------------------------------
    # 2. One‑shot evaluation of all candidates – keep the best few
    # ------------------------------------------------------------------
    print("[Stage 1] Evaluating many diverse seeds (surrogate)")
    best_seq = None
    best_val = float("inf")
    scored = []

    for vec in uniq:
        if time.time() > deadline - 30.0:
            break
        val = _objective_surrogate(vec)
        scored.append((val, vec))
        if val < best_val:
            best_val, best_seq = val, vec.copy()

    # keep the top‑K candidates for deeper optimisation
    scored.sort(key=lambda t: t[0])
    top_k = 10
    top_candidates = [vec for (_, vec) in scored[:top_k]]

    # ------------------------------------------------------------------
    # 3. Refine each top candidate (Adam warm‑up + EG)
    # ------------------------------------------------------------------
    print("\n[Stage 2] Adam warm‑up + EG refinement of top seeds")
    for idx, init in enumerate(top_candidates):
        if time.time() > deadline - 10.0:
            break
        sub_deadline = min(deadline, time.time() + 40.0)

        # short Adam pass
        seq_a, _ = _adam_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            max_iters=80_000,
            beta_end=5e7,
            alpha=0.30,
        )

        # exact EG refinement (allow it to stop when time runs out)
        seq_e, val_e = _exp_gradient_opt(
            start_seq=seq_a,
            n=n,
            deadline=sub_deadline,
            max_iters=5_000_000,
            eta0=6.0,
        )
        if val_e < best_val:
            best_val, best_seq = val_e, seq_e.copy()
            print(f"  Candidate {idx+1}/{len(top_candidates)} → new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long EG run on the current best (consumes the bulk of the budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n[Stage 3] Long EG refinement on best sequence")
        seq_long, val_long = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=30_000_000,
            eta0=5.0,
        )
        if val_long < best_val:
            best_val, best_seq = val_long, seq_long.copy()
            print(f"  Improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random‑restart EG polishing (if any time left)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        eps = rng.uniform(0.001, 0.01)
        noisy = best_seq + eps * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _exp_gradient_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=7_000_000,
            eta0=8.0,
        )
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print(f"  Random‑restart improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final aggressive hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 4] Final hill‑climb")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=5_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"  Hill‑climb improved best → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
