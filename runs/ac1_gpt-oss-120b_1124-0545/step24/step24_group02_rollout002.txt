    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        conv = _conv_fft(a,a)
        max_val = conv.max()
        # Determine active indices

        mask = conv >= max_val - tol * max_val
        # Compute subgradient g in a-space
        g = np.zeros_like(a)
        idx = np.arange(n)

        # Implementation: loop over active k; may be many.
        for k in np.where(mask)[0]:
            rev = k - idx
            valid = (rev >= 0) & (rev < n)
            g[valid] += a[rev[valid]]
        # Multiply by 2 (since derivative of conv)
        g *= 2.0

        dot = np.dot(g, a)
        grad_logit = a * (g - dot)

        m = beta1 * m + (1 - beta1) * grad_logit
        v = beta2 * v + (1 - beta2) * (grad_logit ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        cur_val = _objective_surrogate(_softmax(log_a))

            best_seq = _softmax(log_a).copy()

"""
Advanced convex‑convolution minimiser.

The algorithm works in three stages:

1.  **Generate diverse seeds** – inverse‑sqrt, exponential decay,
    several Dirichlet draws (different sparsities) and the previous
    global best if available.

2.  **Deep smoothed L‑BFGS** – we optimise a smooth log‑sum‑exp
    approximation of the hard max‑convolution.  The optimiser works
    with log‑softmax parameters (so the variables stay on the simplex).
    A long β‑schedule and many L‑BFGS iterations per β give a high‑quality
    local optimum.

3.  **Exact Adam‑mirrored gradient descent** – starting from the best
    L‑BFGS output we minimise the true objective, i.e. the hard
    `max(conv(a,a))`.  The sub‑gradient for a given active set of
    maximising convolution indices is obtained by a single FFT, and an
    Adam‑style update is performed on the log‑variables.  The optimiser
    runs until the wall‑clock budget is exhausted, constantly keeping the
    best sequence seen.

All work respects the 1000 s budget (a small safety margin is kept).
The final sequence is stored in the global variable ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    Result length = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum == 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Simple families of candidates (all are ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _exp_decay(n: int, rate: float = 0.001) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (conc > 0, small → sparse)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised → sum == 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used in the optimiser: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser (log‑softmax parameters)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule,
                      max_iter_per_beta: int = 3000):
    """
    Minimise the smooth log‑sum‑exp surrogate.
    Returns the best ℓ₁‑normalised sequence and its exact surrogate value.
    """
    if _opt is None:
        # SciPy missing – fall back to a short EG run (unlikely in the platform).
        return start_a.copy(), _objective_surrogate(start_a)

    # initialise log‑softmax variables
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # stable soft‑max over conv values
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth surrogate w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # convert to gradient w.r.t. logits v (chain rule through softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # early termination when we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-14,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑mirrored exact‑max optimiser (works directly on the hard max)
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 500_000,
                           eta0: float = 6.0,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           tol: float = 1e-12):
    """
    Mirror‑descent on the true objective max(conv(a,a)) using Adam on the logits.
    Returns the best sequence encountered and its exact surrogate value.
    """
    # logits (log‑softmax) – initialise from the provided probability vector
    log_a = np.log(start_a + 1e-30)
    m = np.zeros_like(log_a)
    v_hat = np.zeros_like(log_a)

    best_seq = _softmax(log_a).copy()
    best_val = _objective_surrogate(best_seq)

    idx = np.arange(n, dtype=np.int64)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # ----- compute exact sub‑gradient of max(conv) -----
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set – indices where the convolution is essentially maximal
        active = np.where(conv >= max_val - tol * max_val)[0]

        if active.size == 0:
            # degenerate – treat as no gradient (should not happen)
            grad_a = np.zeros_like(a)
        else:
            # weight each active index uniformly – this is a valid sub‑gradient
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_logit = a * (grad_a - dot)

        # ----- Adam update on the logits -----
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        eta = eta0 / math.sqrt(t)          # classic √t decay
        log_a -= eta * m_hat / (np.sqrt(v_hat_corr) + eps)

        # ----- keep the best point seen so far -----
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

        # occasional progress report (very cheap)
        if t % 100_000 == 0:
            print(f"    Adam iter {t:,} – best surrogate {best_val:.8f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point used by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a deep smoothed L‑BFGS stage from several diverse seeds
    and finishes with an Adam‑mirrored gradient descent that directly reduces
    the true max‑convolution.  The best sequence is stored globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressivity
    #                and per‑iteration cost.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Assemble a varied list of seeds
    # --------------------------------------------------------------
    seeds = []

    # 1) previous best (if it matches the current length)
    previous = globals().get("height_sequence_1")
    if previous is not None and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)       # ensure non‑negative
        s = arr.sum()
        if s > 0:
            seeds.append(arr / s)

    # 2) classic inverse‑sqrt shape
    seeds.append(_inverse_sqrt(n))

    # 3) exponential decay (two different rates)
    seeds.append(_exp_decay(n, rate=0.001))
    seeds.append(_exp_decay(n, rate=0.002))

    # 4) random Dirichlet draws with several concentrations (very sparse ↔ dense)
    for conc in (0.001, 0.005, 0.02, 0.1):
        d = _random_dirichlet(n, conc, rng)
        seeds.append(d)
        # also its mirror – sometimes symmetry helps
        seeds.append(d[::-1])

    # 5) a small random perturbation of the best known (if any)
    if previous is not None:
        pert = np.maximum(previous + 0.005 * rng.normal(size=n), 0.0)
        s = pert.sum()
        if s > 0:
            seeds.append(pert / s)

    # Normalise everything to the simplex (sum == 1)
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation of all seeds – keep the most promising
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    # keep at most 6 best seeds for the expensive L‑BFGS stage
    top_seeds = [vec for vec, _ in evaluated[:6]]

    print("\n--- Seed quality (surrogate) ---")
    for i, (_, val) in enumerate(evaluated[:len(top_seeds)], 1):
        print(f"Seed #{i}: {val:.8f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule (smooth approximation of the max)
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        5_000_000.0, 10_000_000.0, 50_000_000.0, 100_000_000.0,
        500_000_000.0, 1_000_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Run the L‑BFGS optimiser from each promising seed.
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget (≈ 15 % per seed, ≥5 s)
        remaining = deadline - time.time()
        share = max(5.0, remaining * 0.15)
        sub_deadline = time.time() + share

        print(f"\n[Seed {si}/{len(top_seeds)}] – time slice ≈ {share:.1f}s")

        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3000,
        )
        print(f"  L‑BFGS surrogate = {val_lbfgs:.8f}")

        # ----------------------------------------------------------
        # Short Adam polishing (if we still have time)
        # ----------------------------------------------------------
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_exp_gradient_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=400_000,
                eta0=4.0,
            )
            print(f"  Adam polish   = {val_adam:.8f}")
        else:
            seq_adam, val_adam = seq_lbfgs, val_lbfgs

        # Keep the better of the two phases for this seed
        if val_adam < val_lbfgs:
            candidate_seq, candidate_val = seq_adam, val_adam
        else:
            candidate_seq, candidate_val = seq_lbfgs, val_lbfgs

        if candidate_val < global_best_val:
            global_best_val = candidate_val
            global_best_seq = candidate_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive Adam polishing on the current global best.
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final intensive Adam polishing ---")
        final_seq, final_val = _adam_exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=3.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"   *** Improved to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Report the true evaluation (the platform will also do this)
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
