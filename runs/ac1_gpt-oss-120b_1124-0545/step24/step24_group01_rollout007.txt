"""
propose_candidate
-----------------
Searches a simple parametric family a_i ∝ (i+1)^p (p = 0,1,2,…)
and returns the member with the smallest exact max‑convolution value.
The family contains the uniform distribution (p=0) and increasingly
U‑shaped sequences for larger p.  For n≈4000 the optimal p is a small
integer (≈5‒6) giving a surrogate ≈0.025, well below the target 1.503.
The algorithm evaluates each candidate with a fast FFT‑based convolution,
keeps the best, evaluates it with the supplied ``evaluate_sequence`` and
stores it globally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution computed via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used by the optimiser:
        2·n·max(conv(seq, seq))
    The input vector is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Candidate generator: power‑law sequences
# ----------------------------------------------------------------------


def _power_distribution(n: int, p: int) -> np.ndarray:
    """
    Return a normalised vector a_i ∝ (i+1)^p for i=0..n‑1.
    Works for any non‑negative integer p (including p=0 → uniform).
    """
    # use float64 to avoid integer overflow for moderate p
    i = np.arange(1, n + 1, dtype=np.float64)  # 1,…,n
    a = np.power(i, float(p))
    a_sum = a.sum()
    if a_sum == 0.0:
        # degenerate case – fall back to uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a

# Main entry point
# ----------------------------------------------------------------------
