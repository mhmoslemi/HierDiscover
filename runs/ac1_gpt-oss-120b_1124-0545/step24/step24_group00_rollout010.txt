def beta_distribution(n, a, b, eps=1e-12):
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    # compute unnormalized weights
    w = np.power(x, a-1) * np.power(1 - x, b-1)
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum
    return w

"""
Improved search for the step‐function attack.

The optimizer works in several stages

1.  **Rich initial pool** – we build a set of candidate vectors
    (previous best, exponential decays, Dirichlet draws and a dense
    family of symmetric Beta‑distributions a_i ∝ x^{α‑1}(1‑x)^{α‑1}).
    All candidates are normalised to sum = 1 and the exact surrogate
    2·n·max(conv(a,a)) is evaluated – the five best are kept.

2.  **Adam on a smoothed max** – each retained candidate is refined with
    Adam on the log‑softmax parametrisation while the “soft max’’ is
    approximated by a log‑sum‑exp with a geometric β‑schedule (β from
    0.5 to >10⁸).  This quickly lowers the surrogate without touching the
    expensive true max.

3.  **Exact exponentiated‑gradient (mirror descent)** – the result of
    Adam is fed to a sub‑gradient descent that works on the true
    max‑convolution (the active offset).  The step size decays like
    η₀/√t.  This phase is run for as long as the global time budget
    allows.

4.  **Aggressive hill‑climb** – a simulated‑annealing local search with
    three move types (swap, Gaussian perturb, targeted mass‑shift) is
    applied to the best vector found so far.  The temperature schedule
    is exponential and the routine stops only when the overall deadline
    is reached.

5.  **Random‑restart polishing** – a few short Adam runs are started
    from a noisy copy of the current best to escape possible local
    minima.

The whole routine stops a few seconds before the 1000 s limit,
stores the final vector in the global name ``height_sequence_1`` and
returns it as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution (full, not cyclic) via FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact (surrogate) objective for a vector that already sums to 1:
#   f(a) = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Families of initialisations
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta(α,β) on the grid points x_i = (i+0.5)/n.
    Returns a probability vector (sum = 1) with non‑negative entries.
    """
    # avoid exact 0/1 to keep the density finite
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # compute unnormalised pdf values
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # numerical safety
    a = np.where(np.isfinite(a), a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution
    objective.  The optimisation variable lives in log‑softmax space:
        a = softmax(v_local)   (sum(a)=1)
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # for numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weight for each convolution entry (derivative of log‑sum‑exp)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1e9,
              alpha: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns (best_seq, best_surrogate_value) seen before the deadline.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.  The iterate stays on the
    simplex via a log‑space representation (softmax).  Step‑size decays
    as η₀/√t.  Returns the best sequence found before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only pairs (i,j) with i+j=max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift’’ move.  Returns the best sequence seen before the deadline.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # temperature (geometric decay)
        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                  # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                    # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = idx_i[np.argmax(contrib)]          # donor index
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = min(cur_seq[donor], 0.5 * cur_seq[donor]) * 0.12
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            # renormalise (tiny drift possible)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val
