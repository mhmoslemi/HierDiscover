def refine_with_lp(f_initial, budget, n):
    f = f_initial.copy()
    best_val = compute_objective(f)  # surrogate 2*n*max_conv
    best_f = f.copy()
    start = time.time()
    while time.time() < budget:
        conv_f_f = _conv_fft(f, f)
        M = conv_f_f.max()
        # active constraints
        tol = max(M * 1e-3, 1e-8)
        active = np.where(conv_f_f >= M - tol)[0]
        if active.size == 0:
            # fallback
            active = [np.argmax(conv_f_f)]
        # build A_ub
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            # For each j, compute i = k - j
            # Equivalent to iterate i where 0 <= i < n then j = k - i
            i_min = max(0, k - (n-1))
            i_max = min(k, n-1)
            # For i in range(i_min, i_max+1):
            #   j = k - i
            #   coefficient for variable b_j is f_i
            # So we fill for each i, but that may be many loops; better to use vectorized method.
            i_range = np.arange(i_min, i_max+1, dtype=int)
            j_range = k - i_range
            rows.extend([row_idx]*len(i_range))
            cols.extend(j_range.tolist())
            data.extend(f[i_range].tolist())
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=float)
        c = -np.ones(n, dtype=float)
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': max(0, budget - time.time())})
        if not res.success:
            # break if fails
            break
        b_vec = res.x
        sum_b = b_vec.sum()
        if sum_b == 0:
            # can't improve, break
            break
        g = b_vec / sum_b
        # step size selection
        t = 0.2
        improved = False
        while t > 1e-4:
            f_candidate = (1 - t) * f + t * g
            conv_cand = _conv_fft(f_candidate, f_candidate)
            M_cand = conv_cand.max()
            val_cand = 2 * n * M_cand
            if val_cand < best_val - 1e-12:
                f = f_candidate
                best_val = val_cand
                best_f = f.copy()
                improved = True
                break
            t *= 0.5
        if not improved:
            # If no improvement, break iteration
            break
        # optional print
        print(f"Iteration improvement: val={best_val:.8f}")
    return best_f, best_val

"""
Step‑function optimizer – LP‑guided descent + final EG polish.

Idea
----
* The objective to minimise (for a normalised sequence Σa_i=1) is
      J(a) = 2·n·max(conv(a,a)).
  The “max’’ is the source of non‑convexity.
* For a current sequence `f` we compute M = max(conv(f,f)).
  The constraints   conv(f,b) ≤ M   are *linear* in the unknown `b`.
  Hence we can solve a linear programme which, under these constraints,
  maximises Σ b_i.  The resulting `b` points towards a distribution that
  spreads the mass more evenly than `f`.
* We then move a short step towards the normalised LP solution `g`
  (mixing `f ← (1‑t)f + t·g`).  If the move reduces J we keep it,
  otherwise we shrink `t`.  Repeating this yields a monotone descent
  (or stops when no improvement can be found).
* Several analytic families (inverse‑sqrt, power‑law, exponential,
  random Dirichlet) provide seeds.  The best seeds are refined with the
  LP‑descent.  Finally a short exponentiated‑gradient (mirror‑descent)
  polish is applied.
* The whole routine respects the supplied wall‑clock budget (default
  1000 s) and returns the best sequence found.

The implementation is completely self‑contained, uses only NumPy,
SciPy (linprog, sparse) and standard library modules.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs) – O(n log n)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (sequence is assumed to be ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for Σa_i = 1:  2·n·max(conv(a,a))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – provide diverse starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power_law(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, alpha)
    return a / a.sum()


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse random draw from Dirichlet(conc·1)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (mirror descent on the exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 8.0,
                     tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent (EG) directly on max(conv). Returns best seen."""
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the max is (almost) attained
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of max‑conv (average over active positions)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = np.exp(log_a)
        a /= a.sum()                     # keep ℓ₁‑norm = 1

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP‑guided refinement
# ----------------------------------------------------------------------
def _lp_refine(seed: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 2000,
               t_initial: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Starting from `seed` (Σ=1) repeatedly:
      1. compute M = max(conv(f,f));
      2. solve   max Σ b_i   s.t. conv(f,b) ≤ M, b ≥ 0;
      3. normalise b → g;
      4. mix f ← (1‑t)f + t g with back‑tracking on t.
    Stops when no improvement occurs or time expires.
    Returns the best sequence (ℓ₁‑normalised) and its surrogate value.
    """
    f = seed.copy()
    best_val = _objective_surrogate(f)
    best_f = f.copy()
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv_f = _conv_fft(f, f)
        M = conv_f.max()
        # active constraints – a modest relative slack
        tol = max(M * 1e-3, 1e-9)
        active = np.where(conv_f >= M - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv_f)], dtype=int)

        # Build sparse A_ub where (A_ub b)_k = Σ_i f_i * b_{k-i}
        rows, cols, data = [], [], []
        for row_idx, k in enumerate(active):
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
            j_range = k - i_range                     # corresponding b‑indices
            rows.extend([row_idx] * len(i_range))
            cols.extend(j_range.tolist())
            data.extend(f[i_range].tolist())
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=float)
        c = -np.ones(n, dtype=float)                 # maximise Σb_i ⇔ minimise -Σb_i

        # Linear programme (highs solver handles sparse matrices efficiently)
        res = scipy.optimize.linprog(
            c, A_ub=A_ub, b_ub=b_ub,
            bounds=(0, None), method='highs',
            options={'presolve': True}
        )
        if not res.success:
            # if LP fails we abort the refinement
            break

        b_vec = res.x
        sum_b = b_vec.sum()
        if sum_b == 0.0:
            break
        g = b_vec / sum_b

        # Back‑tracking line search on mixing weight t
        t = t_initial
        improved = False
        while t > 1e-5:
            f_candidate = (1.0 - t) * f + t * g
            val_candidate = _objective_surrogate(f_candidate)
            if val_candidate < best_val - 1e-14:
                f = f_candidate
                best_val = val_candidate
                best_f = f.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            # small random perturbation as a last resort
            noise = rng.normal(scale=1e-3, size=n)
            f_pert = f + noise
            f_pert = np.maximum(f_pert, 0.0)
            s = f_pert.sum()
            if s > 0:
                f_pert /= s
                val_pert = _objective_surrogate(f_pert)
                if val_pert < best_val - 1e-14:
                    f = f_pert
                    best_val = val_pert
                    best_f = f.copy()
                    continue
            # no improvement – stop the loop
            break

    return best_f, best_val
