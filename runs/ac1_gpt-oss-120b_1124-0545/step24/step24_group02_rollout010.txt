def _project_simplex(v):
    # ensure nonnegative sum=1
    # Using algorithm: sort v descending, find rho...
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1,len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def _projected_gradient_descent(start_a, n, deadline, max_iters=1_000_000, eta0=0.5, tol=1e-12):
    a = start_a.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    iter=0
    while time.time() < deadline and iter < max_iters:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]
        # compute gradient via convolution of rev_a with active weights
        w = np.zeros_like(conv)
        w[active] = 1.0
        if w.sum() > 0:
            w /= w.sum()
            grad = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]
        else:
            grad = np.zeros_like(a)
        eta = eta0 / math.sqrt(iter+1)
        a = a - eta * grad
        a = _project_simplex(a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        iter += 1
    return best_seq, best_val

"""
Improved step‑function optimizer.

The idea:
* Use many diverse initialisations (previous best, inverse‑sqrt, uniform,
  sparse Dirichlet, Gaussian, etc.).
* For each seed perform a large‐scale *projected gradient descent* (PGD) on
  the exact objective 2·n·max(conv(a,a)) (the “max‑convolution” problem).
  The sub‑gradient of max(conv) is obtained from the active shifts,
  and after each gradient step the iterate is projected onto the probability
  simplex (ℓ₁‑normalisation, non‑negativity).  The step‑size follows a
  1/√t schedule; the loop runs until the allocated time expires.
* A short *exponentiated‑gradient* (EG) polishing phase follows PGD
  (multiplicative updates on the simplex) – similar to the previous
  implementation but with a higher iteration budget.
* The best-found sequence over all seeds is stored in the global variable
  ``height_sequence_1`` and finally evaluated with the supplied
  ``evaluate_sequence`` routine.
* The whole search respects the wall‑clock budget (default 1000 s) and
  prints progress information.
"""

import time
import math
import numpy as np

# optional SciPy – not required for the core PGD/EG loop
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate objective 2·n·max(conv(seq, seq)) for a sequence
    normalised to sum = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – quick seeds
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with a common concentration parameter."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Projected Gradient Descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _proj_gradient_descent(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.8,
    tol: float = 1e-12,
):
    """
    Projected gradient descent on f(a) = 2·n·max(conv(a,a)).
    Uses a sub‑gradient built from the active shifts (where the convolution
    attains its maximum) and projects onto the simplex after each step.
    """
    a = start_a.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set – indices where conv is (almost) maximal
        active = np.where(conv >= max_val - tol * max_val)[0]

        # weight vector w = uniform over active shifts
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            # degenerate – treat whole vector (should never happen)
            w[:] = 1.0 / conv.size

        # gradient = 2 * (rev_a * w) restricted to size n
        grad = 2.0 * _conv_fft(a[::-1], w)[n - 1 : n - 1 + n]

        # diminishing step size
        eta = eta0 / math.sqrt(it + 1)
        a = a - eta * grad
        a = _project_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirrors previous code)
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 8.0,
    tol: float = 1e-12,
):
    """
    Mirror‑descent / exponentiated‑gradient optimisation on the exact max
    convolution.  This is a complementary method to the additive PGD.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)  # reuse soft‑max from earlier code (stable)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# (Optional) smoothed L‑BFGS optimiser – kept for extra power
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 4000,
):
    """Smoothed max‑convolution optimiser using L‑BFGS‑B on the soft‑max."""
    if _opt is None:
        # fall back to PGD if SciPy is unavailable
        return _proj_gradient_descent(start_a, n, deadline)

    v = np.log(start_a + 1e-30)                     # logits
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs several diverse seeds, each receiving a time slice for
    projected‑gradient descent (PGD) followed by an exponentiated‑gradient
    (EG) polish.  A final PGD polishing phase uses any remaining budget.
    The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Assemble a short list of promising seeds
    # --------------------------------------------------------------
    seeds = []

    # 1) previous best (if available and of correct length)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0:
            seeds.append(prev_arr / prev_arr.sum())

    # 2) classic inverse‑sqrt shape
    seeds.append(_inverse_sqrt(n))

    # 3) uniform distribution
    seeds.append(np.full(n, 1.0 / n))

    # 4) very sparse Dirichlet (conc=0.001)
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))

    # 5) moderately sparse Dirichlet (conc=0.05)
    seeds.append(_random_dirichlet(n, conc=0.05, rng=rng))

    # 6) absolute Gaussian draw
    gauss = np.abs(rng.normal(size=n))
    if gauss.sum() > 0:
        seeds.append(gauss / gauss.sum())

    # 7) sparse random support (≈1 % non‑zero)
    k = max(2, n // 100)
    sparse = np.zeros(n, dtype=np.float64)
    idxs = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 1e-3
    sparse[idxs] = vals
    if sparse.sum() > 0:
        seeds.append(sparse / sparse.sum())

    # Ensure all seeds are proper probability vectors
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation of the seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[: max(1, len(evaluated))]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[: len(top_seeds)], 1):
        print(f" Seed #{i}: {v:.8f}")

    # --------------------------------------------------------------
    # Optimisation schedule for the smoothed max‑convolution (optional)
    # --------------------------------------------------------------
    beta_schedule = [
        0.5,
        1.0,
        2.0,
        5.0,
        10.0,
        20.0,
        40.0,
        80.0,
        200.0,
        500.0,
        1_000.0,
        2_000.0,
        5_000.0,
        10_000.0,
        20_000.0,
        50_000.0,
        100_000.0,
        200_000.0,
        500_000.0,
        1_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Run the optimiser from each seed – time is split dynamically
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # give each seed roughly 20 % of remaining time, but at least 5 s
        share = max(5.0, remaining * 0.20)
        sub_deadline = time.time() + share
        print(f"\n[Seed {si}/{len(top_seeds)}] allocated time ≈ {share:.1f}s")

        # ---- optional L‑BFGS phase (gets a good warm‑start) ----
        seq_cur = seed_vec
        if _opt is not None:
            try:
                seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
                    start_a=seed_vec,
                    n=n,
                    deadline=sub_deadline,
                    beta_schedule=beta_schedule,
                    max_iter_per_beta=3000,
                )
                print(f"  L‑BFGS surrogate = {val_lbfgs:.8f}")
                seq_cur = seq_lbfgs
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed: {e}")

        # ---- projected gradient descent (core optimiser) ----
        if time.time() < sub_deadline:
            seq_pgd, val_pgd = _proj_gradient_descent(
                start_a=seq_cur,
                n=n,
                deadline=sub_deadline,
                max_iters=10_000_000,
                eta0=0.9,
                tol=1e-12,
            )
            print(f"  PGD surrogate = {val_pgd:.8f}")
        else:
            seq_pgd, val_pgd = seq_cur, _objective_surrogate(seq_cur)

        # ---- EG polishing (helps escape tiny plateaus) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_pgd,
                n=n,
                deadline=sub_deadline,
                max_iters=400_000,
                eta0=6.0,
                tol=1e-12,
            )
            print(f"  EG polish   surrogate = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_pgd, val_pgd

        # pick the better of PGD / EG for this seed
        cand_seq, cand_val = (seq_eg, val_eg) if val_eg < val_pgd else (seq_pgd, val_pgd)

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing on the current global best
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Final polishing on global best (PGD) ---")
        final_seq, final_val = _proj_gradient_descent(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=5_000_000,
            eta0=0.5,
            tol=1e-12,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
