            break

            best_seq = a.copy()

"""
Propose a high‑quality non‑negative step function (list of heights) that
minimises ``evaluate_sequence``.

The optimiser works in four stages within the allotted time budget:

1️⃣  **Diverse analytic seeds** – a rich pool of parameterised families
    (uniform, triangular, power‑arcsine, Beta, exponential, sine‑power,
    random Dirichlet draws and mixtures).  All members are normalised to
    sum = 1 and the surrogate ``2·n·max(conv(a,a))`` is evaluated.
    The best few seeds are kept.

2️⃣  **Smooth L‑BFGS refinement** – the non‑smooth objective is replaced by
    a log‑sum‑exp smoothed version
        fβ(a) = (2·n/β)·log∑ₖ exp(β·(a⋆a)ₖ) .
    L‑BFGS‑B optimisation is performed on the log‑softmax (mirror) space
    for a schedule of β values (1 → 200 000).  The softmax parametrisation
    guarantees the simplex constraints automatically.

3️⃣  **Exact exponentiated‑gradient polishing** – a mirror‑descent step on the
    true max‑convolution (the same routine used in the reference solution)
    further reduces the objective.

4️⃣  **Final simulated‑annealing walk** – a cheap stochastic walk on the
    simplex (mass transfer between two random indices) with a linear cooling
    schedule removes remaining shallow local minima.

Throughout the run the best sequence (according to the cheap surrogate)
is kept globally in ``height_sequence_1``.  At the end we call the
supplied ``evaluate_sequence`` to obtain the official score.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during optimisation.
    The input is assumed to be a probability vector (sum = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    a /= a.sum()
    return a


def _power_arcsine_shape(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    a /= a.sum()
    return a


def _exponential_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    a /= a.sum()
    return a


def _sine_power_shape(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ sin(i·π/n)^p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a /= a.sum()
    return a


def _random_dirichlet_shape(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _generate_seed_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a large and diverse pool of candidate distributions.
    """
    seeds: list[np.ndarray] = []

    # Simple hand‑crafted families
    seeds.append(_uniform_shape(n))
    seeds.append(_triangular_shape(n))

    # Power‑arcsine sweep
    for p in np.arange(0.2, 2.01, 0.1):
        seeds.append(_power_arcsine_shape(n, float(p)))

    # Symmetric Beta sweep (log‑space for better coverage)
    for alpha in np.geomspace(0.1, 5.0, num=20):
        seeds.append(_beta_shape(n, float(alpha), float(alpha)))

    # Asymmetric Beta random draws
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_shape(n, float(alpha), float(beta)))

    # Exponential decay family
    for lam in np.arange(0.0, 5.01, 0.25):
        seeds.append(_exponential_shape(n, float(lam)))

    # Sine‑power family
    for p in np.arange(0.1, 5.01, 0.2):
        seeds.append(_sine_power_shape(n, float(p)))

    # Dirichlet draws with several concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet_shape(n, float(conc), rng))

    # Re‑use the previous global solution if it matches the required length
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS optimisation on a smooth surrogate
# ----------------------------------------------------------------------


def _smooth_obj_full(logit: np.ndarray, n: int, beta: float):
    """
    Smoothed objective (log‑sum‑exp of the convolution) together with its
    gradient w.r.t. the log‑softmax parameters.
    Returns a tuple (value, gradient) expected by ``scipy.optimize.minimize``.
    """
    # map logits → probabilities on the simplex
    a = _softmax(logit)

    # convolution a⋆a
    conv = _conv_fft(a, a)

    # log‑sum‑exp of beta·conv (numerically stable)
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_shift = np.exp(beta_conv - max_bc)
    sum_exp = np.sum(exp_shift)
    lse = max_bc + np.log(sum_exp)                     # = log∑ exp(beta*conv)

    # smooth surrogate value: (2·n/β)·lse
    value = (2.0 * n / beta) * lse

    # softmax weights for the weighted sum of convolutions
    w = exp_shift / sum_exp                           # w_k = exp(beta*conv_k) / Σ

    # gradient of Σ w_k·conv_k w.r.t. a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_weighted = 2.0 * full[n - 1:n - 1 + n]       # length n
    grad_a = 2.0 * n * grad_weighted                  # chain rule for outer factor 2·n

    # gradient in the log‑softmax (mirror) space
    dot = np.dot(grad_a, a)
    grad_logit = a * (grad_a - dot)

    return float(value), grad_logit.astype(np.float64)


def _lbfgs_opt(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a sequence of L‑BFGS runs on the smooth surrogate with a
    schedule of β values.  Returns the best sequence (as a probability
    vector) and its surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # β schedule – start gentle, finish very sharp
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0,
    ]

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # L‑BFGS options – limit the number of objective evaluations
        # (each evaluation also yields the gradient)
        opts = {"maxfun": 250, "maxiter": 250, "gtol": 1e-12}

        try:
            res = minimize(_smooth_obj_full, logit,
                           args=(n, beta),
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as exc:
            print(f"[L‑BFGS] exception at β={beta}: {exc}")
            continue

        logit = res.x
        a = _softmax(logit)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.0e} improved surrogate = {cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (same as reference)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny simulated‑annealing walk (final polishing)
# ----------------------------------------------------------------------


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer walk on the simplex.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    The returned list is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    print("\n=== Starting optimisation (n =", n, ", budget =", budget_s, "s) ===")

    # --------------------------------------------------------------
    # 0️⃣ Build a large pool of analytic seeds
    # --------------------------------------------------------------
    seeds = _generate_seed_pool(n, rng)

    # --------------------------------------------------------------
    # 1️⃣ Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        if (idx + 1) % 50 == 0 or idx == len(seeds) - 1:
            print(f" evaluated {idx + 1}/{len(seeds)} seeds; current best surrogate = {seed_infos[0][0]:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(6, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate after seed sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ L‑BFGS refinement on the smooth surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        lbfgs_deadline = time.time() + remaining * 0.45   # allocate ~45 % of remaining time
        print("\n--- L‑BFGS smoothing (smooth max) ---")
        seq_lbfgs, val_lbfgs = _lbfgs_opt(best_seq, lbfgs_deadline)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_seq, best_val = seq_lbfgs.copy(), val_lbfgs
            print("   >>> Updated global best after L‑BFGS")

    # --------------------------------------------------------------
    # 3️⃣ Exact exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 4.0                       # leave a few seconds for anneal
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=5.0,
                                           max_iters=5_000_000)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best after EG")

    # --------------------------------------------------------------
    # 4️⃣ Final tiny annealing walk (last seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_ann, val_ann = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_ann:.6f}")
        if val_ann < best_val:
            best_seq, best_val = seq_ann.copy(), val_ann
            print("   >>> Updated global best after annealing")

    # --------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for potential warm‑starts in the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
