def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0
    n = 8000   # choose

    # generate seeds using families but with n dimension
    seeds = generate_seeds(n, rng)

    # evaluate surrogate for each seed (fast but still conv). Keep top_k=5
    evaluated = []
    for vec in seeds:
        if time.time() > deadline: break
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0: continue
        v = v / s
        val = _objective_surrogate(v)
        evaluated.append((v, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]
    best_seq = top_seeds[0]
    best_val = _objective_surrogate(best_seq)
    print(f"Initial best surrogate: {best_val}")

    # Determine remaining time
    remaining = deadline - time.time()
    # allocate sub-deadlines for each seed
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline: break
        # allocate time proportionally: (remaining * 0.8) / top_k maybe.
        sub_dead = time.time() + remaining * 0.75 / top_k # leaving some time for final polish
        print(f"Optimising seed #{idx+1} with time {sub_dead - time.time():.1f}s")
        seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_dead, rng)
        print(f"Seed #{idx+1} got surrogate {val_opt}")
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt
            print(f"New global best: {best_val}")

    # final polishing with remaining time
    if time.time() < deadline:
        final_dead = deadline
        print("Final polishing")
        best_seq, best_val = _full_optimisation(best_seq, n, final_dead, rng)
        print(f"Final best surrogate {best_val}")

    # compute true value via evaluate_sequence
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluation = {true_val}")
    except Exception as e:
        print("Error evaluating sequence:", e)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

def proj_simplex(v):
    # returns w that is the Euclidean projection onto simplex {x>=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _lbfgs_obj(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    # gradient w.r.t a:
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    # map to gradient wrt v (logit):
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

"""
Step‑function optimizer for the ``evaluate_sequence`` objective.

The method works on a probability simplex (∑a_i = 1, a_i≥0) because the
objective is scale‑invariant.  It

1. builds a rich set of analytic / random seeds,
2. keeps the few best seeds,
3. runs a short exponentiated‑gradient (EG) phase on each of them
   (cheap but moves quickly toward low max‑convolution),
4. selects the most promising intermediate vector and runs the full pipeline
   (EG → smooth L‑BFGS → flattening gradient → simulated annealing),
5. performs a final EG refinement,
6. tries a few convex mixtures of the current best with simple companions
   (uniform, reversed, a second‑best seed),
7. returns the best vector found (stored globally as ``height_sequence_1``).

All heavy steps respect the wall‑clock budget (default 1000 s).  The code
contains no lambda expressions and all helper functions are defined at the
module level, as required by the competition runner.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional scientific imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# global used by the L‑BFGS callback (cannot be a closure)
# ----------------------------------------------------------------------
_lbfgs_deadline = 0.0


def _lbfgs_callback(_):
    """Interrupt L‑BFGS when the global deadline is exceeded."""
    if time.time() > _lbfgs_deadline:
        raise KeyboardInterrupt


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) (∑a_i = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (sub‑gradient of the exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """EG on the exact max‑convolution (multi‑active indices)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the self‑convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 4_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on Σ (conv‑avg)² to flatten the autocorrelation."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)  # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)           # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # log‑space projected gradient (mirror descent)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Objective and gradient for the smooth max surrogate:
        2·n·logsumexp(β·conv)/β .
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = conv.max()
    # stable log‑sum‑exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """L‑BFGS‑B on the smooth‑max surrogate, homotoping over β."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    global _lbfgs_deadline
    _lbfgs_deadline = deadline

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_lbfgs_obj,
                x0=v,
                args=(n, beta),
                method='L-BFGS-B',
                jac=True,
                callback=_lbfgs_callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """Gaussian‑perturbation SA with a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       # configuration – can be tuned
                       eg_iters: int = 2_500_000,
                       eg_eta0: float = 14.0,
                       eg_tol: float = 1e-11,
                       lbfgs_iters: int = 6000,
                       flatten_iters: int = 6_000_000,
                       flatten_eta0: float = 10.0,
                       sa_attempts: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Run the four stages (EG → L‑BFGS → flatten → SA) on a copy of *start_vec*.
    All stages respect the shared *deadline*.
    Returns the best sequence encountered and its surrogate value.
    """
    # 1) exponentiated‑gradient on exact max‑convolution
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=eg_iters,
        eta0=eg_eta0,
        tol=eg_tol,
    )

    # 2) smooth L‑BFGS (more iterations per β)
    seq_lbfgs = seq_eg
    val_lbfgs = val_eg
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=lbfgs_iters,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)

    # 3) flattening gradient
    seq_flat, val_flat = seq_lbfgs, val_lbfgs
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=flatten_iters,
            eta0=flatten_eta0,
        )

    # 4) simulated annealing polish
    seq_sa, val_sa = seq_flat, val_flat
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=sa_attempts,
            start_sigma=0.0008,
            T_start=6e-3,
            T_end=4e-9,
        )

    # pick the best among the stages
    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val

# Helper: generate a rich seed set (including the 2‑parameter family)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of candidate seed vectors (already L1‑normalised)."""
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))
    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    p_vals = np.linspace(0.3, 1.2, 10)
    lam_vals = np.linspace(0.0, 0.008, 8)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed copies to increase diversity
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include the previous global best if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # small perturbations of the current best known seed
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds

def _mixture_exploration(base_seq: np.ndarray,
                         companions: list[np.ndarray],
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Try linear blends with a grid of alphas; keep the best feasible point."""
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq)

    alphas = np.linspace(0.0, 1.0, 21)  # 0,0.05,…,1

    for comp in companions:
        for alpha in alphas:
            if time.time() > deadline:
                return best_seq, best_val
            mixed = _mix_shapes(best_seq, comp, alpha)
            val = _objective_surrogate(mixed)
            if val < best_val:
                best_val = val
                best_seq = mixed.copy()
    return best_seq, best_val

    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # Allocate a fraction of the remaining budget to this quick sweep
    remaining = deadline - time.time()
    quick_budget = remaining * 0.2
    quick_deadline = time.time() + quick_budget

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > quick_deadline:
            break
        seq_tmp, val_tmp = _exp_gradient_opt_multi(
            seed_vec,
            n,
            quick_deadline,
            max_iters=300_000,
            eta0=10.0,
            tol=1e-11,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = seq_tmp
            print(f"[Quick EG] seed #{idx+1} improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deep optimisation of the most promising seed (majority of time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        deep_dead = time.time() + remaining * 0.75
        print(f"\n[Deep optimisation] allocating ≈{remaining*0.75:.1f}s")
        seq_opt, val_opt = _full_optimisation(
            best_seq,
            n,
            deep_dead,
            rng,
            # tuned parameters – a little more flattening and L‑BFGS work
            eg_iters=2_500_000,
            eg_eta0=14.0,
            eg_tol=1e-11,
            lbfgs_iters=8000,
            flatten_iters=8_000_000,
            flatten_eta0=10.0,
            sa_attempts=600_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt
            print(f"   *** Updated global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final EG refinement (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n[Final EG refinement]")
        seq_ref, val_ref = _refine_eg(best_seq, n, deadline, rng)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref
            print(f"[Refine] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Mixture exploration with a few simple companions
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        companions = [_uniform(n), best_seq[::-1].copy()]
        # also try the second‑best seed (if we have it)
        if len(evaluated) > 1:
            companions.append(evaluated[1][0])
        print("\n[Mixture exploration]")
        seq_mix, val_mix = _mixture_exploration(best_seq, companions, n, deadline)
        if val_mix < best_val:
            best_val = val_mix
            best_seq = seq_mix
            print(f"[Mix] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
