def exp_gradient_polish_topk(start_seq, n, deadline, max_iters, eta0=12.0, top_k=5):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # Get indices of top_k entries (largest)
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        # compute subgradient as sum over each k
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]  # sum contributions after all peaks
        # maybe average
        g = g / len(ks)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # ensure positivity via softmax
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a non‑negative step‑function (probability vector) with a low
evaluate_sequence value.

Algorithm
---------
1.  Build a small diverse pool of analytic seeds:
        * discrete arcsine law (Beta(0.5,0.5))
        * a few symmetric Beta(α,α) laws (α∈{0.3,0.4,0.6,0.8})
        * uniform
        * exponential decays
        * a handful of random Dirichlet draws
   All seeds are normalised to sum to 1.

2.  **Smoothed Adam** – minimise a smooth approximation of the max‑convolution
    (log‑sum‑exp) using Adam on the log‑softmax parameters.
    This quickly brings the candidate into a good basin.
    Temperature β is raised from 0.5 to 15.0.

3.  **Adam on the exact max‑convolution sub‑gradient** – a mirror‑descent
    (Adam) step is performed on the true objective.
    The sub‑gradient is taken as the average of the gradients of the
    `top_k` largest convolution entries (k=3 by default).  This
    simultaneously flattens several peaks and accelerates convergence.

4.  **Random local search** – a few cheap perturbations that move a tiny
    amount of mass from one coordinate to another are tried; any improvement
    is kept.

5.  The best sequence found is evaluated with the supplied
    `evaluate_sequence` function and stored in the global variable
    ``height_sequence_1`` for possible continuation runs.

The whole routine respects the supplied time‑budget (default 1000 s) and
stops as soon as the deadline is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (returns length len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)).  `seq` is assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seeds
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) law on {0,…,n‑1} discretised via the mid‑point rule.
    For α=0.5 this coincides (up to a constant) with the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n + 0.5 - i) ** (alpha - 1.0))
    # avoid infinities when α<1 and i hits 0 or n+0.5, but for our α>0 they are safe
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Pure exponential decay, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parameters)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 150_000,
                     lr0: float = 0.5,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     beta_start: float = 0.5,
                     beta_end: float = 15.0) -> tuple[np.ndarray, float]:
    """Adam on the smooth approximation (log‑sum‑exp) of the max‑convolution."""
    log_a = np.log(start_seq + 1e-30)       # work in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        a = _softmax(log_a)

        # -------- gradient of the smooth max ----------
        conv = _conv_fft(a, a)                     # length 2n‑1
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()                               # softmax weights, sum=1

        # g_a = 2 Σ_k w_k * shift(a)   (size n)
        a_rev = a[::-1]
        w_a = _conv_fft(w, a_rev)                 # length 3n‑2
        g_a = 2.0 * w_a[n - 1:n - 1 + n]           # slice → length n

        # chain rule through the softmax
        dot = np.dot(g_a, a)
        grad_log = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # keep best
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_max_subgrad_opt(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 10_000_000,
                          eta0: float = 12.0,
                          top_k: int = 3,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the true objective.
    The sub‑gradient is obtained by averaging the gradients of the `top_k`
    largest convolution entries.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)

        # indices of the `top_k` peaks
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient w.r.t. a (average of the contributions)
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        g = g / len(ks)

        # chain rule through the softmax
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam‑style adaptive steps
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta0 / math.sqrt(t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate and keep best
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val

# Tiny random local search (cheap perturbations)
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 5000,
                         eps: float = 1e-3,
                         rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Try to improve the candidate by moving a small amount of mass between
    two random coordinates.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective_surrogate(a)
    best_seq = a.copy()

    for _ in range(attempts):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] < eps:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            a = a_new
            best_seq = a_new.copy()
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns the best sequence
    discovered (as a Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  A power‑of‑two is convenient for FFT and
    # still modest enough for many iterations.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Build a modest pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1) discrete arcsine (Beta(0.5,0.5))
    seeds.append(_arcsine_distribution(n))

    # 2) a few symmetric Beta laws
    for alpha in (0.35, 0.45, 0.6, 0.8):
        seeds.append(_beta_distribution(n, alpha))

    # 3) uniform
    seeds.append(_uniform(n))

    # 4) exponential decays (several rates)
    for rate in (0.0008, 0.0015, 0.003):
        seeds.append(_exp_decay(n, rate))

    # 5) random Dirichlet draws
    for _ in range(3):
        seeds.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # 6) previous best, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(np.maximum(arr, 0.0) / np.maximum(np.sum(arr), 1e-12))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – smoothed Adam on each seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    remaining = len(seeds)

    for idx, cand in enumerate(seeds, 1):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        time_left = deadline - time.time()
        if time_left < 12.0:
            print("[Info] Time exhausted before finishing all seeds.")
            break

        # allocate a modest number of iterations – we let it shrink as we
        # progress through the list.
        max_iters = int(min(150_000,
                            max(30_000,
                                (time_left - 20.0) / max(1, remaining - idx + 1))))
        print(f"[Seed {idx}/{len(seeds)}] max_iters={max_iters}")

        seq, val = _adam_smooth_opt(cand, n, deadline,
                                    max_iters=max_iters,
                                    lr0=0.6,
                                    beta_start=0.5,
                                    beta_end=15.0)

        print(f"   surrogate after smoothed Adam = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   *** New global best (surrogate) = {best_val:.6f}")

    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------
    # Stage 2 – Adam on the exact max‑convolution sub‑gradient
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with Adam on true max‑convolution (top‑k) ---")
        polished_seq, polished_val = _adam_max_subgrad_opt(
            best_seq, n, deadline,
            max_iters=10_000_000,
            eta0=12.0,
            top_k=3)
        if polished_val < best_val:
            best_seq = polished_seq
            best_val = polished_val
            print(f"[Polish] improved surrogate = {best_val:.6f}")
        else:
            print(f"[Polish] no improvement (best surrogate unchanged)")

    # ------------------------------------------------------------------
    # Stage 3 – tiny random local search (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random local search ---")
        local_seq, local_val = _random_local_search(best_seq, n, deadline,
                                                    attempts=5000,
                                                    eps=1e-3,
                                                    rng=rng)
        if local_val < best_val:
            best_seq = local_seq
            best_val = local_val
            print(f"[Local] improved surrogate = {best_val:.6f}")
        else:
            print("[Local] no improvement")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
