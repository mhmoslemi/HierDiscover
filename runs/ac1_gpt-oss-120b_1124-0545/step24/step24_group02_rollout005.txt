def _exp_gradient_opt_full(start_a, n, deadline, max_iters=5_000_000, eta0=4.0, tol=1e-12):
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    N = _next_pow2(2*n-1)
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break
        # compute convolution via FFT
        af = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(af * af, n=N)[:2*n-1]
        max_val = conv.max()
        # tolerance relative to max
        active = np.where(conv >= max_val - tol*max_val)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)
        # compute gradient via conv with w
        w = np.zeros_like(a)
        w[active] = 1.0 / active.size
        rev_a = a[::-1]
        # conv_rev = conv of rev_a and w
        rev_f = np.fft.rfft(rev_a, n=N)
        w_f = np.fft.rfft(w, n=N)
        conv_rev = np.fft.irfft(rev_f * w_f, n=N)[:2*n-1]
        grad = 2.0 * conv_rev[n-1:n-1+n] # shape n
        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        eta = eta0 / math.sqrt(it)
        log_a -= eta * momentum
        # optionally add jitter every 50000 steps to escape flatness:
        if it % 50000 == 0:
            log_a += 0.001 * np.random.default_rng().normal(size=n)
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Optimiser for the step‑function evaluation problem.

Algorithm
---------
1.  Initialise a collection of diverse seeds (previous best, inverse‑sqrt,
    a few Dirichlet draws).  All seeds are ℓ₁‑normalised.
2.  Pick the seed with the smallest *exact* surrogate
    ``2·n·max(conv(a,a))`` (i.e. the true objective for ‖a‖₁=1).
3.  Run a long exponentiated‑gradient (EG) mirror‑descent on the *hard*
    max‑convolution objective.  The EG update keeps the iterates on the
    probability simplex, uses a momentum term (Adam‑style) and adds a tiny
    random “kick’’ every 50 k iterations to escape flat plateaus.
    The learning‑rate schedule ηₜ = η₀ / √t and a very small tolerance for
    the active set allow the optimisation to keep improving the peak
    convolution value.
4.  With the best sequence found so far, perform a few rounds of a
    smoothed‑max L‑BFGS‑B optimisation.  The smooth surrogate replaces the
    hard max by a log‑sum‑exp with an exponentially increasing β schedule,
    giving a precise gradient that fine‑tunes the solution.
5.  Finally polish the result with another (shorter) EG run.
6.  Return the best sequence and store it in the global variable
    ``height_sequence_1`` for possible reuse.

The whole routine respects the supplied wall‑clock budget (default 1000 s)
and prints progress information.
"""

import time
import math
import numpy as np

# Optional SciPy import – used only for the L‑BFGS phase.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum == 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – provide good starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic shape a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser – works directly on the hard max.
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 6.0,
    rel_tol: float = 1e-12,
    abs_tol: float = 1e-15,
    momentum_beta: float = 0.9,
    kick_interval: int = 50_000,
    kick_scale: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient on the *true* max‑convolution.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = np.asarray(start_a, dtype=np.float64).copy()
    if a.sum() == 0:
        raise ValueError("Start sequence has zero sum.")
    a /= a.sum()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        # --- self‑convolution -------------------------------------------------
        af = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(af * af, n=N)[: 2 * n - 1]
        max_val = conv.max()

        # --- active set (indices attaining the max, within tolerance) -------
        tol = max(abs_tol, rel_tol * max_val)
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        # --- sub‑gradient of the hard max (averaged over active indices) ---
        w = np.zeros_like(a)
        w[active] = 1.0 / active.size
        rev_a = a[::-1]
        rev_f = np.fft.rfft(rev_a, n=N)
        w_f = np.fft.rfft(w, n=N)
        conv_rev = np.fft.irfft(rev_f * w_f, n=N)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # shape (n,)

        # --- momentum + step -------------------------------------------------
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        eta = eta0 / math.sqrt(it)
        log_a -= eta * momentum

        # occasional random perturbation to escape plateaus
        if kick_interval and (it % kick_interval == 0):
            log_a += kick_scale * rng.normal(size=n)

        # --- project back onto the simplex -----------------------------------
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed‑max L‑BFGS optimiser (fallback when SciPy is available)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution using L‑BFGS‑B.
    Returns the best normalised sequence seen and its exact surrogate.
    """
    if _opt is None:                     # pragma: no cover
        # SciPy unavailable – fall back to EG.
        return _exp_gradient_opt_full(
            start_a=start_a,
            n=n,
            deadline=deadline,
            max_iters=max_iter_per_beta,
            eta0=6.0,
        )

    # work in the log‑softmax space
    v = np.log(start_a + 1e-30)

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of the max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth surrogate w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # chain rule for the log‑softmax parametrisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val

    # --------------------------------------------------------------
