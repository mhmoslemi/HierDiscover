def eg_opt(start_seq, n, deadline, max_iters=None):
    if max_iters is None: compute based on time remaining (maybe large)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 1
    while time < deadline and it <= max_iters:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set: offsets within rel tolerance
        # Considering numerical error, we can choose tolerance = max_val - 1e-12.
        active = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient contributions
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            # contribution 2*a[rev_idx]
            # Could weight each offset equally, or average across active set.
            g[idx_i] += 2.0 * a[rev_idx]
        # Average if multiple active offsets:
        if len(active) > 1:
            g /= len(active)
        # step
        eta = 6.0 / math.sqrt(it)  # or dynamic
        log_a -= eta * g
        a = _softmax(log_a)
        if (it & 0xFFF) == 0: # check objective occasionally
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        it += 1
    return best_seq, best_val

"""
search for a low‑value step‑function.

The objective is
    2·n·max(conv(a,a)) / (sum(a)**2)                     (evaluate_sequence)

All vectors are kept on the probability simplex (sum = 1), therefore the
objective reduces to 2·n·max(conv(a,a)).  The code follows three phases:

1.  Generate a handful of diversified initial vectors (previous best,
    exponential decays, random Dirichlet and uniform).
2.  For each initialisation run a mirror‑descent (exponentiated‑gradient)
    optimiser that works on the *exact* max‑convolution.  Instead of the
    usual single‑offset sub‑gradient we use the **active set** – all offsets
    whose convolution value is within a tiny tolerance of the current
    maximum – and average their sub‑gradients.  The update is performed in
    log‑space and the step‑size decays as η₀/√t.
    A short run for every start gives a pool of good candidates; the best
    one is then refined for the remainder of the time budget.
3.  A final simulated‑annealing hill‑climb (swap / Gaussian perturb /
    targeted mass‑shift) is applied to polish the solution.

The routine respects the 1000 s time budget, stores the best vector in the
global variable ``height_sequence_1`` and finally evaluates it with the
provided black‑box ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Mirror‑descent on the true max‑convolution (active‑set version)
# ----------------------------------------------------------------------
def _eg_opt_active(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 10_000_000,
                   eta0: float = 6.0,
                   eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
    Uses the average sub‑gradient over the *active set* of offsets whose
    convolution value is within ``eps`` of the current maximum.
    Returns the best probability vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # log‑softmax variables
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 1
    while it <= max_iters and time.time() < deadline:
        # current full convolution and the maximal entry
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_val:
            best_val = cur_obj
            best_seq = a.copy()

        # indices whose value is (almost) maximal
        active = np.where(conv >= max_val - eps)[0]
        # build weight vector w (same length as conv) for the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)          # average over the active offsets

        # sub‑gradient of the averaged active offsets:
        #   g_i = 2 * Σ_{k∈active} a_{k-i}
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1 : n - 1 + n]

        # mirror‑descent step in log‑space
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad
        a = _softmax(log_a)

        it += 1

    # final check (in case the loop stopped before evaluating the last a)
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (three move types)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Local search combining swaps, Gaussian perturbations and a targeted
    “mass‑shift” move that moves a small amount of probability away from the
    most offending pair of the current active offset.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (exponential cooling)
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturbation
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            # move a modest fraction (10 % of half the donor’s mass)
            delta = min(cur_seq[donor], 0.5 * cur_seq[donor]) * 0.1
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – propose a candidate sequence
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs three optimisation phases (screening, long EG refinement
    and hill‑climb) and returns the best sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin before the hard timeout
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------------------
    # problem size – feel free to change but the default works well for the budget
    # -----------------------------------------------------------------------------
    n = 4000

    # -------------------------------------------------------------------------
    # 1. build a set of diversified initial vectors
    # -------------------------------------------------------------------------
    inits = []

    # a) reuse previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                inits.append(arr)
        except Exception:
            pass

    # b) a few exponential decays
    for rate in (0.001, 0.003, 0.005):
        inits.append(_exp_decay(n, rate))

    # c) random Dirichlet draws (α = 0.5 → some sparsity)
    for _ in range(3):
        dir_vec = rng.gamma(shape=0.5, scale=1.0, size=n)
        dir_vec = dir_vec / dir_vec.sum()
        inits.append(dir_vec)

    # d) uniform (the obvious baseline)
    inits.append(np.full(n, 1.0 / n))

    # keep only truly distinct vectors
    uniq_inits = []
    seen_hashes = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen_hashes:
            uniq_inits.append(vec)
            seen_hashes.add(h)

    # -------------------------------------------------------------------------
    # 2. short screening: run a modest EG on *each* start – keep the best
    # -------------------------------------------------------------------------
    print("[Stage 1] Screening – short EG runs")
    best_seq = None
    best_val = float("inf")

    per_init_budget = 30.0                     # seconds allocated per start
    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + per_init_budget)
        print(f"  init {idx + 1}/{len(uniq_inits)} (budget {per_init_budget}s)")
        seq, val = _eg_opt_active(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            eta0=8.0,
            eps=1e-12,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"    New best surrogate = {best_val:.6f}")

    # -------------------------------------------------------------------------
    # 3. long refinement of the current best candidate
    # -------------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Long EG refinement on the current best")
        seq, val = _eg_opt_active(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            eta0=5.0,
            eps=1e-12,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # -------------------------------------------------------------------------
    # 4. Aggressive hill‑climb for final polishing
    # -------------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] Hill‑climb")
        seq, val = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=5_000_000,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # -------------------------------------------------------------------------
    # 5. Final evaluation with the black‑box function
    # -------------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
