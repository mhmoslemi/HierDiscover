def _project_simplex(v):
    """Projection of v onto simplex {x >=0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v

    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Simple projected subgradient descent optimizer for the step‑function problem.
    The algorithm starts from a known good sequence (or an inverse‑sqrt seed),
    then repeatedly moves opposite to a sub‑gradient of the max‑convolution
    objective while projecting onto the probability simplex.  A short
    random‑restart phase is added to escape shallow local minima.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine length
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        base = np.asarray(prev, dtype=np.float64)
        base = np.maximum(base, 0.0)
        if base.sum() > 0:
            base = base / base.sum()
    else:
        base = _inverse_sqrt(n)
    # Ensure sum=1
    base = base / base.sum()

    # We'll keep best global
    best_seq = base.copy()
    best_val = _objective_surrogate(best_seq)

    # Number of random restarts: maybe 3
    num_restarts = 3
    for r in range(num_restarts):
        if time.time() >= deadline:
            break
        # allocate budget fraction: half for first; else roughly equal
        remaining = deadline - time.time()
        # allocate a share (maybe 0.3 of remaining for each restart)
        share = remaining * 0.3 if r < num_restarts-1 else remaining
        sub_deadline = time.time() + share

        if r == 0:
            seed_vec = best_seq.copy()
        else:
            # random Dirichlet seed to explore new basin
            seed_vec = rng.dirichlet(np.full(n, 0.001)).astype(np.float64)
            seed_vec = seed_vec / seed_vec.sum()

        seq, val = _proj_grad_descent(seed_vec, n, sub_deadline,
                                      max_iters=300_000,
                                      eta0=1.5, mu=0.9, tol=1e-12)
        print(f"[Restart {r+1}] best surrogate = {val:.8f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   *** New global best surrogate = {best_val:.8f}")

    # Final polishing: run a few more iterations with smaller eta
    if time.time() < deadline:
        print("\n--- Final polishing ---")
        final_seq, final_val = _proj_grad_descent(best_seq, n, deadline,
                                                  max_iters=200_000,
                                                  eta0=0.5, mu=0.9, tol=1e-13)
        if final_val < best_val:
            best_seq = final_seq
            best_val = final_val
            print(f"[Final] improved surrogate to {best_val:.8f}")

    # Evaluate via provided function
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate {best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Save globally
    globals()["height_sequence_1"] = best_seq.tolist()

    best_seq = a.copy()

"""
Projected sub‑gradient descent optimizer for the step‑function max‑convolution
problem.

The core idea
--------------
* Work directly with the non‑negative ℓ₁‑simplex (sum of the heights = 1).
* For a vector a the objective is  f(a) = 2·n·max(conv(a,a)),
  i.e. the evaluation function when the sum is normalised to one.
* The (non‑smooth) sub‑gradient of max(conv) at a is obtained from the
  active convolution indices where the maximum is attained.
* A simple heavy‑ball/ momentum update is performed:
        v ← μ·v – η·grad
        a ← Π_simplex( a + v )
  where Π_simplex is the Euclidean projection onto the simplex.
* Several short random‑restart runs are carried out (inverse‑sqrt,
  previous best and a sparse Dirichlet draw) with a dynamic time budget.
* The best sequence discovered is finally polished with a longer run
  using a smaller stepsize.
* The resulting sequence (still normalised to sum = 1) is stored in the
  global variable ``height_sequence_1`` and returned as a plain ``list``.
"""

import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection of a vector onto the probability simplex:
        {x ∈ ℝⁿ | x ≥ 0, Σ x = 1}
    """
    if v.ndim != 1:
        raise ValueError("_project_simplex expects a 1‑D array")
    n = v.shape[0]

    # Fast path for already feasible vectors
    if v.min() >= 0 and np.isclose(v.sum(), 1.0):
        return v.copy()

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families – good deterministic seeds
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1) and ℓ₁‑normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Very sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact surrogate (hard max) – the value we actually want to minimise
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with ℓ₁‑norm = 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of the map a ↦ max(conv(a,a)) and the current
    maximum value.  The sub‑gradient is the average over all active
    convolution indices.
    Returns (grad, max_val).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = float(conv.max())
    # active indices (within absolute tolerance)
    active = np.where(conv >= max_val - tol)[0]

    # weight vector for the active constraints (average)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size

    # gradient = 2 * (rev_a * w)[n-1 : n-1+n]
    rev = a[::-1]
    full = _conv_fft(rev, w)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Projected (heavy‑ball) sub‑gradient descent
# ----------------------------------------------------------------------
def _proj_grad_descent(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 1.0,
    mu: float = 0.9,
    tol: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """
    Heavy‑ball projected sub‑gradient descent on the exact max‑convolution
    objective.  The iterate is always kept on the ℓ₁‑simplex.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_a.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # momentum buffer
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # sub‑gradient at current point
        grad, _ = _subgradient_max_conv(a, tol=tol)

        # stepsize schedule η_t = η₀ / √t
        eta = eta0 / math.sqrt(it)

        # heavy‑ball update (move opposite to the gradient)
        v = mu * v - eta * grad
        a = _project_simplex(a + v)

        # occasional exact evaluation (every 1000 iterations)
        if it % 1000 == 0 or it == max_iters:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"    iter {it:6d}: new best surrogate = {best_val:.8f}")

    # Final exact check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs several short heavy‑ball sub‑gradient descents from
    diverse seeds and a final long polishing phase, all within the supplied
    time budget.  The best sequence is stored in the global variable
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin for finalisation

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best (if any) to use as a seed
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if (
        isinstance(previous, list)
        and len(previous) == n
        and all(isinstance(x, (int, float)) for x in previous)
    ):
        base = np.asarray(previous, dtype=np.float64)
        base = np.maximum(base, 0.0)
        if base.sum() > 0.0:
            base = base / base.sum()
        else:
            base = _inverse_sqrt(n)
    else:
        base = _inverse_sqrt(n)

    # --------------------------------------------------------------
    # Prepare several diverse seeds (previous best, inverse‑sqrt,
    # sparse Dirichlet, slightly perturbed previous)
    # --------------------------------------------------------------
    seeds = [base, _inverse_sqrt(n), _random_dirichlet(n, conc=0.001, rng=rng)]
    # perturbed version of the current best, if we have one
    pert = base + 0.003 * rng.normal(size=n)
    pert = np.maximum(pert, 0.0)
    if pert.sum() > 0.0:
        seeds.append(pert / pert.sum())

    # Ensure all seeds are on the simplex
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation of the seeds
    # --------------------------------------------------------------
    seed_vals = []
    for s in seeds:
        val = _objective_surrogate(s)
        seed_vals.append((s, val))
    seed_vals.sort(key=lambda x: x[1])
    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(seed_vals, 1):

    # --------------------------------------------------------------
    # Run a short heavy‑ball descent from each seed, sharing the time‑budget

    global_best_seq = None

    for idx, (seed_vec, _) in enumerate(seed_vals, 1):

            eta0=1.5,
            mu=0.9,
            tol=1e-12,
        )
        print(f"  → heavy‑ball surrogate = {val:.8f}")

        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing on the current global best
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        final_seq, final_val = _proj_grad_descent(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=400_000,
            eta0=0.5,
            mu=0.9,
            tol=1e-13,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------

        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, "
            f"true evaluate_sequence = {true_val:.8f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
