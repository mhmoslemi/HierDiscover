"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.  The objective is equal to
``2·n·max(conv)`` when the vector is normalised to sum = 1, so the optimiser
works on the simplex.

The algorithm is a *population‑based memetic search*:

1.  Generate a diverse set of initial probability vectors (uniform, exponential,
    Dirichlet draws, sparse random patterns and the previously best solution).
2.  Run a *short* Adam optimisation on a log‑softmax parametrisation with a
    smoothed‑max surrogate, followed by a short exact exponentiated‑gradient
    phase.  This yields a cheap filtered population.
3.  Keep the K best individuals (K≈5) and repeatedly create *offspring* by
    convex‑combining two parents and adding a small Gaussian mutation.
    Each offspring is refined with a medium‑length exponentiated‑gradient run.
    If it improves the worst member of the population it replaces it.
    The loop runs until the time budget is almost exhausted.
4.  A final long exponentiated‑gradient refinement is performed on the current
    best member, followed by an aggressive simulated‑annealing hill‑climb.
5.  The best vector is evaluated with the black‑box ``evaluate_sequence``
    function, stored in ``height_sequence_1`` for possible continuation runs,
    and returned as a plain list.

All helper functions are top‑level, there are no closures, and the code
stays well within the 1000‑second budget on a single CPU (the FFT‑based
convolution is very cheap for n=4000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The variable v_local lives in log‑softmax space; a = softmax(v_local)
    is a probability vector with sum=1.
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp smoothing
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation (smoothed max)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 120_000,
              beta_start: float = 0.5,
              beta_end: float = 2e7,
              alpha: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate.
    Returns the best probability vector (sum = 1) seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution (sub‑gradient of the active entry).
    Returns the best probability vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # each participating index gets 2 * counterpart value
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / Gaussian / targeted mass‑shift)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with three move types.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])     # donor index (largest contribution)
            q = int(rng.integers(0, n))           # random receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1   # 10 % of a small fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Crossover + mutation helper (used inside the population loop)
# ----------------------------------------------------------------------
def _crossover_and_mutate(parent1: np.ndarray,
                          parent2: np.ndarray,
                          rng: np.random.Generator,
                          sigma: float) -> np.ndarray:
    """Convex combination of two parents + Gaussian noise, re‑normalised."""
    w = rng.random()
    child = w * parent1 + (1.0 - w) * parent2
    child = child + sigma * rng.normal(size=len(child))
    child = np.maximum(child, 0.0)
    s = child.sum()
    if s == 0.0:
        # fall back to one of the parents (should be extremely rare)
        child = parent1.copy()
        s = child.sum()
    child = child / s
    return child


# ----------------------------------------------------------------------
# Evolutionary / memetic optimisation driver
# ----------------------------------------------------------------------
def _evolutionary_opt(initial_population: list[np.ndarray],
                     n: int,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Runs a small population memetic algorithm:
        * quick Adam + EG refinement of each seed,
        * keep K best,
        * repeatedly generate offspring (crossover+mutation) and
          refine them with a medium‑length EG run,
        * replace the worst individual when improvement occurs.
    Returns the overall best vector and its surrogate value.
    """
    # ------------------------------------------------------------------
    # Stage 0 – cheap filtering of the raw seeds
    # ------------------------------------------------------------------
    filtered = []
    for seed_seq in initial_population:
        if time.time() > deadline:
            break
        sub_dead = time.time() + 12.0               # ~12 s per seed
        seq_a, _ = _adam_opt(seed_seq, n, sub_dead,
                             max_iters=70_000,
                             beta_end=2e7,
                             alpha=0.35)
        seq_e, best_val = _exp_gradient_opt(seq_a, n, sub_dead,
                                            max_iters=600_000,
                                            eta0=6.0)
        filtered.append((seq_e, best_val))

    if not filtered:
        # fallback – generate a uniform distribution
        uniform = np.full(n, 1.0 / n)
        filtered.append((_exp_gradient_opt(uniform, n, deadline)[0],
                         _objective_surrogate(uniform)))

    # keep K best individuals
    filtered.sort(key=lambda x: x[1])
    K = min(5, len(filtered))
    pop = [seq for seq, _ in filtered[:K]]
    pop_vals = [val for _, val in filtered[:K]]
    best_val = pop_vals[0]

    print(f"[Evolution] initial population size {K}, best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – evolutionary loop (crossover + short EG)
    # ------------------------------------------------------------------
    while time.time() + 12.0 < deadline:          # a little safety margin
        # pick two distinct parents
        i, j = rng.choice(len(pop), size=2, replace=False)
        sigma = 0.01 * max(1.0 - (time.time() - (deadline - 1000)) / 1000.0, 0.1)
        child = _crossover_and_mutate(pop[i], pop[j], rng, sigma)

        # refine offspring (medium length EG)
        sub_dead = time.time() + 6.0
        child_seq, child_val = _exp_gradient_opt(child, n, sub_dead,
                                                  max_iters=1_200_000,
                                                  eta0=5.0)

        # replace worst if we improved
        if child_val < pop_vals[-1]:
            pop[-1] = child_seq
            pop_vals[-1] = child_val
            # re‑sort
            order = np.argsort(pop_vals)
            pop = [pop[idx] for idx in order]
            pop_vals = [pop_vals[idx] for idx in order]
            best_val = pop_vals[0]
            print(f"[Evolution] new population best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long exact EG refinement of the current best
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print("[Final EG] long refinement of current best")
        best_seq, best_val = _exp_gradient_opt(pop[0], n, deadline,
                                               max_iters=6_000_000,
                                               eta0=5.0)
        print(f"    after long EG best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive hill‑climb on the final best
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        print("[Hill‑climb] final local search")
        best_seq, best_val = _hill_climb(best_seq, n, deadline, rng,
                                         max_iters=4_500_000)
        print(f"    after hill‑climb best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size (fixed, but can be changed if desired)
    n = 4000

    # ------------------------------------------------------------------
    # 1. Assemble a heterogeneous set of initial vectors (sum = 1)
    # ------------------------------------------------------------------
    init_vectors = []

    # a) reuse previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0.0:
                    init_vectors.append(arr / s)
        except Exception:
            pass

    # b) deterministic baselines
    init_vectors.append(np.full(n, 1.0 / n))                # uniform
    for rate in (0.0005, 0.001, 0.002, 0.004, 0.008):
        init_vectors.append(_exp_decay(n, rate))

    # c) random Dirichlet draws with varying concentration
    for _ in range(6):
        alpha = rng.uniform(0.1, 2.0)
        vec = rng.gamma(shape=alpha, scale=1.0, size=n)
        vec = vec / vec.sum()
        init_vectors.append(vec)

    # d) a few sparse random patterns (helps escape dense local minima)
    for _ in range(2):
        m = rng.integers(int(0.1 * n), int(0.3 * n))
        idx = rng.choice(n, size=m, replace=False)
        sparse = np.zeros(n)
        sparse[idx] = 1.0 / m
        init_vectors.append(sparse)

    # deduplicate (tiny chance of exact duplicates)
    uniq = []
    seen = set()
    for v in init_vectors:
        h = hash(v.tobytes())
        if h not in seen:
            uniq.append(v)
            seen.add(h)

    # ------------------------------------------------------------------
    # 2. Run the memetic evolutionary optimiser
    # ------------------------------------------------------------------
    best_seq, best_val = _evolutionary_opt(uniq, n, deadline, rng)

    # ------------------------------------------------------------------
    # 3. Final black‑box evaluation and bookkeeping
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
