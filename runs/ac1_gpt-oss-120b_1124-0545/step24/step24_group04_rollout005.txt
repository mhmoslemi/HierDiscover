def _proj_simplex(v):
    # Input: any 1D array
    # Output: projected onto simplex sum=1, v >=0
    # Implementation from Duchi: sort descending, compute cumulative sum, find rho etc.
    v_sorted = np.sort(v)[::-1]

    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_opt(start_seq, n, deadline, step0=0.5):

    best_seq = a.copy()
    best_val = _surrogate(a)
    t = 0

        # calculate convolution and max index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient
        g = np.zeros_like(a)
        # for i such that idx_i + idx_j = k => i = any, j=k-i
        # j in [0, n-1] => i in [max(0, k-(n-1)), min(n-1, k)]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        g[idx_i] = 2.0 * a[rev]
        # Determine step size
        eta = step0 / np.sqrt(t)
        # gradient descent step: a = a - eta * g
        a = a - eta * g
        # project onto simplex:
        a = _proj_simplex(a)
        # evaluate surrogate
        val = _surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

"""
Differential‑Evolution + Sub‑gradient optimiser for the step‑function task.

The optimiser builds a diverse seed pool, creates a small DE population and
evolves it using classic “DE/rand/1/bin”.  After the evolutionary phase a
projected sub‑gradient descent (the exact max‑convolution sub‑gradient) is
run to polish the best individual.  All work is performed on the probability
simplex (the sum of the heights is forced to 1) because the evaluation
function is scale‑invariant.  The final vector is returned as a plain list
and stored in the global ``height_sequence_1`` for possible warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    seq is assumed to be a probability vector (sum == 1).
    Returns 2 * n * max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps."""
    m = rng.integers(2, 6)                # 2‑5 components
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.5, n * 0.05)
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _sinusoidal(n: int) -> np.ndarray:
    """Shape ∝ sin²(π·(i+½)/n) – concentrated in the middle."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** 2
    a /= a.sum()
    return a


def _cosine_shape(n: int) -> np.ndarray:
    """Half‑cosine bell (cosine on [0,π])."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.cos(np.pi * (i / n - 0.5))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects an arbitrary vector onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        # Numerical safety – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Differential Evolution optimiser
# ----------------------------------------------------------------------


def _de_optimise(pop: np.ndarray,
                 fitness: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classic DE/rand/1/bin on the surrogate objective.
    ``pop`` has shape (popsize, n) and already satisfies the simplex.
    Returns the best individual found and its surrogate value.
    """
    popsize, n = pop.shape
    best_idx = int(np.argmin(fitness))
    best_seq = pop[best_idx].copy()
    best_val = float(fitness[best_idx])

    gen = 0
    while time.time() < deadline:
        gen += 1
        for i in range(popsize):
            if time.time() >= deadline:
                break

            # --- mutation -------------------------------------------------
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.maximum(mutant, 0.0)
            s = mutant.sum()
            if s == 0.0:
                mutant = _uniform(n)
            else:
                mutant /= s

            # --- crossover (binomial) ------------------------------------
            cross_mask = rng.random(n) < CR
            if not cross_mask.any():
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _proj_simplex(trial)

            # --- selection ------------------------------------------------
            val = _surrogate(trial)
            if val < fitness[i]:
                pop[i] = trial
                fitness[i] = val
                if val < best_val:
                    best_val = val
                    best_seq = trial.copy()
                    print(f"[DE] gen {gen:4d} – new global best surrogate = {best_val:.6f}")

        # occasional diversity injection (replace worst with fresh random)
        if gen % 25 == 0:
            worst = int(np.argmax(fitness))
            # draw a fresh Dirichlet candidate
            new_vec = _random_dirichlet(n, conc=0.5, rng=rng)
            fitness_w = _surrogate(new_vec)
            if fitness_w < fitness[worst]:
                pop[worst] = new_vec
                fitness[worst] = fitness_w
                if fitness_w < best_val:
                    best_val = fitness_w
                    best_seq = new_vec.copy()
                    print(f"[DE] injected new best (gen {gen}) surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     step0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient method for
        f(a) = max_k (a * a)[k]      (sum(a)=1, a_i≥0).

    The sub‑gradient at a is 2·a reflected about the active index.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # compute sub‑gradient g_i = 2 a_{k-i} for all feasible i
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        g[idx_i] = 2.0 * a[rev]

        eta = step0 / math.sqrt(t)
        a = a - eta * g
        a = _proj_simplex(a)

        val = _surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # occasional progress report
            if t % 200 == 0:
                print(f"[SubG] iter {t:5d} – new best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (≈ 1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for final evaluation / printing
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously stored solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of seed vectors (probability vectors)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _sinusoidal(n),
        _cosine_shape(n),
    ]

    # Dirichlet draws with a variety of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions (edge‑heavy / centre‑heavy)
    for a in (0.05, 0.10, 0.15, 0.20, 0.30, 0.50, 0.80, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Random Gaussian mixtures
    for _ in range(4):
        seeds.append(_gaussian_mixture(n, rng))

    # A few permutations of early seeds to break symmetry
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate seeds – keep the best (surrogate objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Differential Evolution (global exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        popsize = 20
        # initialise population with the best seed + random Dirichlet draws
        pop = np.empty((popsize, n), dtype=np.float64)
        pop[0] = best_seq
        for i in range(1, popsize):
            pop[i] = _random_dirichlet(n, conc=0.5, rng=rng)

        fitness = np.empty(popsize, dtype=np.float64)
        for i in range(popsize):
            fitness[i] = _surrogate(pop[i])

        de_deadline = time.time() + 0.55 * remaining
        print("\n--- Differential Evolution (global search) ---")
        seq_de, val_de = _de_optimise(pop, fitness, de_deadline, rng,
                                      F=0.5, CR=0.9)
        print(f"[DE] best surrogate after DE = {val_de:.6f}")
        if val_de < best_val:
            best_seq, best_val = seq_de.copy(), val_de
            print("   >>> Updated global best (DE)")

    # ------------------------------------------------------------------
    # 4️⃣ Projected sub‑gradient polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = deadline - 5.0
        print("\n--- Projected sub‑gradient polishing (exact max) ---")
        seq_sg, val_sg = _subgradient_opt(best_seq, n, polishing_deadline,
                                          step0=1.0)
        print(f"[SubG] best surrogate after polishing = {val_sg:.6f}")
        if val_sg < best_val:
            best_seq, best_val = seq_sg.copy(), val_sg
            print("   >>> Updated global best (sub‑gradient)")

    # ------------------------------------------------------------------
    # 5️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation augments the previous pipeline with a classic Differential‑Evolution global search followed by an exact projected sub‑gradient descent.  
* The DE stage explores the simplex broadly and regularly injects fresh individuals to keep diversity.  
* The sub‑gradient stage works on the true max‑convolution (no smooth‑max surrogate) and uses a simple Euclidean projection onto the probability simplex, allowing fine‑grained improvements.  

Both phases respect the 1000 s deadline, and the best sequence found is stored in `height_sequence_1` for warm‑starts in subsequent runs.
