"""
Enhanced step‑function optimiser with simulated‑annealing refinement.

The optimiser proceeds in several stages inside the allotted wall‑clock
budget (default 1000 s):

1. **Diversified seeding** – several deterministic and random
   probability vectors are generated (uniform, arcsine, triangular,
   spikes, Dirichlet draws, symmetric beta‑distributions and the
   previously best solution, if any).  The best seed according to the
   exact surrogate 2·n·max(conv) is kept.

2. **Adam on a smooth‑max surrogate** – a fast Adam optimiser works in
   the log‑softmax space on the log‑sum‑exp approximation of the true
   max.  This quickly drives the sequence into a good basin.

3. **Exact exponentiated‑gradient polishing** – the true max‑convolution
   is tackled with mirror‑descent (exponentiated‑gradient).  The step
   size decays as 1/√t.

4. **Greedy incremental local search** – many cheap “move δ from i to
   j’’ trials are evaluated by an O(n) incremental convolution update.
   Only strictly improving moves are kept.

5. **Simulated‑annealing refinement** – starting from the sequence
   obtained in step 4, a classic Metropolis‑Hastings walk is performed.
   The temperature and the transfer‑scale σ decay exponentially with
   elapsed time, allowing occasional uphill moves that help escape the
   local minimum found by the greedy walk.

After the last phase the best sequence is evaluated with the public
``evaluate_sequence`` function, stored in the global variable
``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a schedule of smooth‑max sharpening factors
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # adapt number of inner iterations to remaining time
        max_iter_beta = 12000 if (deadline - time.time()) > 30 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a (closed form via convolution)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_val = _objective_surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fast greedy walk (strictly improving moves)
# ----------------------------------------------------------------------


def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Greedy walk on the simplex: repeatedly move a small amount δ from one
    coordinate to another, keeping only strictly improving moves.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    best_a = a.copy()
    best_max = np.max(conv)
    best_val = 2.0 * n * best_max

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a

        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        if cand_val < best_val - 1e-12:
            # accept the improving move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_max = cand_max
            best_val = cand_val
            accepted += 1

            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                last_recalc = accepted

        # cooling of sigma
        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (allows uphill moves)
# ----------------------------------------------------------------------


def _simulated_annealing_refine(start_seq: np.ndarray,
                                n: int,
                                deadline: float,
                                rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk on the simplex.
    The temperature as well as the transfer scale σ decay exponentially.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()
    best_val = cur_val

    accepted = 0
    attempts = 0
    last_recalc = 0

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    # initial temperature: 1 % of the current value; final temperature ≈ 0
    T0 = cur_val * 0.01
    T_end = cur_val * 1e-9

    while time.time() < deadline:
        attempts += 1
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # progress based temperature (exponential cooling)
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        T = T0 * ((T_end / T0) ** progress) if T0 > 0 else 0.0

        # σ follows the same decay as temperature (but stays ≥ 0.001)
        sigma = max(0.001, 0.2 * (T / T0) if T0 > 0 else 0.001)

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a

        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # Metropolis acceptance
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

            # every few thousand accepted moves recompute from scratch
            if accepted - last_recalc >= 5000:
                conv = _conv_fft(a, a)
                cur_val = 2.0 * n * np.max(conv)
                last_recalc = accepted

        # occasional progress report (≈ every 30 s)
        if attempts % 10000 == 0:
            now = time.time()
            if now - start_time - elapsed >= 30.0:
                print(f"[SA] elapsed={elapsed:.1f}s, attempts={attempts}, "
                      f"accepted={accepted}, best_val={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the five‑phase pipeline described in the module
    docstring and respects the supplied time budget.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load previously found solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with several concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – a richer set of alphas
    beta_alphas = [
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of seeds – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth‑max) ---")
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.25 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0            # keep some seconds for final phases
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Intensive greedy local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Intensive local refinement (greedy) ---")
        local_deadline = deadline
        seq_loc, val_loc = _intensive_local_search(best_seq, n,
                                                   local_deadline,
                                                   sigma_start=0.05,
                                                   rng=rng)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Simulated‑annealing final refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(best_seq, n, deadline,
                                                    rng=rng)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 7️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
