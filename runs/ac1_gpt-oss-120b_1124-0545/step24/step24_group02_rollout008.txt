def _simulated_annealing(a, n, deadline, max_iters=200000, init_temp=0.01, cooling=0.999):
    best = a.copy()
    best_val = _objective_surrogate(best)
    cur = a.copy()
    cur_val = best_val
    temp = init_temp
    rng = np.random.default_rng()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # pick i with cur[i] > 0
        i = rng.choice(np.where(cur > 1e-12)[0])
        # pick j != i
        j = rng.integers(0, n-1)
        if i == j: continue
        # move epsilon fraction of cur[i]
        eps = cur[i] * 0.01 * rng.random()
        if eps <= 0: continue
        cur[i] -= eps
        cur[j] += eps
        # ensure nonneg
        new_val = _objective_surrogate(cur)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / temp):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i] += eps
            cur[j] -= eps
        temp *= cooling
    return best, best_val

"""
Hybrid optimizer for the non‑negative step‑function problem.

The algorithm proceeds in four stages:

1. **Seed generation** – a diverse collection of candidates is built from:
   * the previously stored best sequence (if any);
   * classic analytic families (inverse‑sqrt, power, product‑power);
   * a number of sparse Dirichlet draws.
   All candidates are ℓ₁‑normalised (sum = 1).

2. **Fast surrogate screening** – each seed is evaluated with the exact
   surrogate objective 2·n·max(conv(a,a)) (the sum is already 1), which is
   cheap thanks to FFT‑based convolution.  The best few seeds are kept.

3. **Gradient polishing** – each selected seed is refined with an
   exponentiated‑gradient (EG) mirror‑descent on the true max‑convolution.
   EG works on the logarithmic parameters (log‑softmax) so the simplex
   constraint is satisfied automatically.  Momentum and a decreasing step
   size are used for fast convergence.

4. **Final polishing** – the current best sequence undergoes a long EG run
   followed by a lightweight simulated‑annealing phase that moves tiny amounts
   of mass between random positions.  Both phases respect the ℓ₁‑simplex and
   improve the exact objective.

All work respects the supplied wall‑clock budget (default 1000 s).  The
resulting sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the sequence is assumed ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – provide good starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power_family(n: int, alpha: float) -> np.ndarray:
    """Monotone decreasing shape a_i ∝ (i+1)^{-alpha}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -alpha, where=i != 0)
    return a / a.sum()


def _product_power_family(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ ((i+1)*(n-i))^{-alpha}."""
    i = np.arange(0, n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    a = np.power(prod, -alpha, where=prod != 0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 8.0,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)          # keep in log‑space
    momentum = np.zeros_like(a)       # Adam‑style momentum
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx_array = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the maximum is almost attained
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of max(conv) – average over the active shifts
        g = np.zeros_like(a)
        for k in active:
            rev = k - idx_array
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size   # factor 2 from derivative of a_i a_{k-i}

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight simulated annealing on the simplex
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp: float = 0.01,
    cooling: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Simple annealing that moves a tiny fraction of mass between two random
    positions while preserving the ℓ₁‑norm.  The move is accepted with the
    usual Metropolis probability.
    """
    rng = np.random.default_rng()
    cur = start_a.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val
    temp = init_temp

    # pre‑compute a mask of viable donor indices (non‑zero mass)
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # pick a donor with non‑negligible mass
        donor_candidates = np.where(cur > 1e-12)[0]
        if donor_candidates.size == 0:
            break
        i = rng.choice(donor_candidates)
        # pick a different receiver
        j = rng.integers(0, n - 1)
        if j == i:
            continue

        # move a small random proportion of the donor's mass
        eps = cur[i] * rng.random() * 0.02   # at most 2 % of donor's weight
        if eps <= 0.0:
            continue

        cur[i] -= eps
        cur[j] += eps

        new_val = _objective_surrogate(cur)
        delta = new_val - cur_val

        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur.copy()
        else:
            # revert
            cur[i] += eps
            cur[j] -= eps

        temp *= cooling
        # avoid temperature becoming exactly zero
        if temp < 1e-12:
            temp = 1e-12

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    A mixture of analytic seeds, gradient descent, and simulated annealing
    is employed within the time budget.
    The best sequence is stored in ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length trades time vs quality
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Gather seeds (previous best, analytic families, random draws)
    # --------------------------------------------------------------
    seeds = []

    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.shape == (n,):
            seeds.append(prev_arr / prev_arr.sum())
        else:
            # if the stored sequence has a different length, ignore it
            pass

    # classic families
    seeds.append(_inverse_sqrt(n))
    for alpha in np.linspace(0.0, 0.5, 7):          # edge‑heavy family
        seeds.append(_product_power_family(n, alpha))
    for alpha in np.linspace(0.0, 1.0, 7):          # monotone power family
        seeds.append(_power_family(n, alpha))

    # random Dirichlet draws – a few with very sparse and a few with moderate sparsity
    for _ in range(20):
        seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))
        seeds.append(_random_dirichlet(n, conc=0.05, rng=rng))

    # Ensure all seeds are normalised (sum = 1) and valid
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Fast surrogate evaluation of the raw seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    # keep the top few (roughly a third or at least three)
    keep = max(3, len(evaluated) // 3)
    top_seeds = [vec for vec, _ in evaluated[:keep]]

    print("\n--- Initial seed screening (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:keep], 1):
        print(f" Seed #{i}: surrogate = {v:.8f}")

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Gradient polishing of the promising seeds
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        print(f"\n[Seed {idx}/{len(top_seeds)}] starting EG polishing")
        seq_refined, val_refined = _exp_gradient_opt(
            start_a=seed_vec,
            n=n,
            deadline=deadline,
            max_iters=250_000,          # a moderate budget per seed
            eta0=6.0,
            tol=1e-12,
        )
        print(f"  → EG surrogate = {val_refined:.8f}")

        if val_refined < global_best_val:
            global_best_val = val_refined
            global_best_seq = seq_refined.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing (long EG + tiny annealing)
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        # Long EG run
        seq_final, val_final = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=4.0,
            tol=1e-12,
        )
        print(f"[Final EG] surrogate = {val_final:.8f}")

        # Light simulated annealing to escape possible plateaus
        seq_final2, val_final2 = _simulated_annealing(
            start_a=seq_final,
            n=n,
            deadline=deadline,
            max_iters=200_000,
            init_temp=0.01,
            cooling=0.9995,
        )
        print(f"[Final annealing] surrogate = {val_final2:.8f}")

        if val_final2 < global_best_val:
            global_best_seq = seq_final2.copy()
            global_best_val = val_final2
            print(f"   *** Improved global best after annealing = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
