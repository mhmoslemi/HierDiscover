"""
Step‑function optimizer – “enhanced‑seed & multi‑restart” version.

Key ideas
---------
* Build a very rich seed pool (uniform, power families, exponentials,
  two‑parameter power‑exponential, beta shapes, block‑shapes and many
  Dirichlet draws).  The extra block‑shape family (high plateau + low tail)
  explores a region that the original code never visited.
* Keep the *top‑k* seeds (k≈5) and devote a fraction of the remaining budget
  to a full four‑stage optimisation of each seed:
    1. exponentiated‑gradient on the exact max‑convolution,
    2. L‑BFGS on a smooth‑max surrogate (if SciPy is available),
    3. gradient descent that flattens the convolution,
    4. a long simulated‑annealing polish.
* The best seed gets most of the time (≈70 % of the remaining budget) while
  the other top seeds receive a smaller share – this gives several
  independent chances to escape a local minimum.
* After the deep runs we perform a final “perturb‑and‑climb” phase:
  small Gaussian bumps around the current best solution, followed by a few
  hundred EG steps.  This cheap refinement often pushes the surrogate a
  little further.
* The whole routine respects the 1000 s wall‑clock limit, stores the best
  sequence in the global variable ``height_sequence_1`` and returns it as a
  plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional SciPy imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (true objective with L1‑normalised sequence)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for ∑a=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    return a / a.sum()


def _block_shape(n: int, b: int, w: float) -> np.ndarray:
    """High plateau of length b (value a_h) and low tail (value a_l = w·a_h)."""
    if b <= 0 or b >= n:
        return _uniform(n)
    a_h = 1.0 / (b + w * (n - b))
    a_l = w * a_h
    a = np.full(n, a_l, dtype=np.float64)
    a[:b] = a_h
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient on exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on active indices)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # gradient: average autocorrelation slices for all active k
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the self‑convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent minimising Σ_k (conv_k – avg)² (makes convolution flat)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when ∑a=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # log‑softmax projection (mirror descent on simplex)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """L‑BFGS‑B on 2·n·log‑sum‑exp(β·conv)/β with a homotopy over β."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # work in log‑softmax space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Jacobian of soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """Gaussian‑perturbation SA with linear cooling."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sequence → (exp‑grad → L‑BFGS → flatten → SA) → best found."""
    # 1) exponentiated‑gradient on the exact max‑convolution
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=14.0,
        tol=1e-11,
    )

    # 2) smooth‑max L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=6000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient descent
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=6_000_000,
            eta0=10.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=500_000,
            start_sigma=0.0008,
            T_start=6e-3,
            T_end=4e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: generate a rich set of seeds (including block‑shapes)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of L1‑normalised seed vectors."""
    seeds = []

    # classic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    # exponentials
    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    # mixed exponentials
    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    # beta‑distribution shapes
    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # two‑parameter power‑exponential sweep (dense)
    p_vals = np.linspace(0.3, 1.2, 12)
    lam_vals = np.linspace(0.0, 0.008, 10)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    # block‑shape family – high plateau + low tail
    b_ratios = np.linspace(0.01, 0.5, 12)          # fraction of indices with high value
    w_vals = np.logspace(-4, 0, 10)               # low/high ratio
    for br in b_ratios:
        b = max(1, int(br * n))
        for w in w_vals:
            seeds.append(_block_shape(n, b, w))

    # random Dirichlet draws (different concentrations)
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures with uniform for symmetry‑break
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed versions (to explore opposite ordering)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # previous global best (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # small perturbations of the current best seed (if we already have one)
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point – the function the judges will call
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, stores the
    best sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # 2 s safety margin – the external harness will kill the process after budget_s
    deadline = start_time + budget_s - 2.0

    # Problem size – 5000 gives a good trade‑off between expressiveness and speed
    n = 5000

    print(f"\n=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # 1.  Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} seed vectors, evaluating surrogates...")

    # ------------------------------------------------------------------
    # 2.  Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3.  Initialise global best with the very best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Deep optimisation of the top seeds (budget split)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Allocate ~70 % to the best seed, the rest evenly to the others
        best_share = 0.70
        other_share = (1.0 - best_share) / max(1, top_k - 1)

        for idx, seed_seq in enumerate(top_seeds):
            if time.time() > deadline:
                break
            share = best_share if idx == 0 else other_share
            sub_deadline = time.time() + remaining * share
            print(f"\n[Optimising seed #{idx+1}] allocated ≈{(remaining*share):.1f}s")
            seq_opt, val_opt = _full_optimisation(seed_seq, n, sub_deadline, rng)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"   *** New global best surrogate = {global_best_val:.6f}")
            remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 5.  Final perturb‑and‑climb polishing (cheap EG steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Final perturb‑and‑climb]")
        # a few hundred EG iterations on a slightly perturbed version
        for attempt in range(5):
            pert = global_best_seq + 0.001 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            sub_deadline = time.time() + min(5.0, deadline - time.time())
            seq_opt, val_opt = _exp_gradient_opt_multi(
                pert,
                n,
                sub_deadline,
                max_iters=300_000,
                eta0=10.0,
                tol=1e-11,
            )
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"   *** Improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Evaluate with the official function and store result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
