def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (returns length len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # Guard against divide‑by‑zero (should never happen)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:

    Exact surrogate: 2·n·max(conv(seq, seq)).
    ``seq`` must be normalised to sum = 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _triangular(n: int) -> np.ndarray:
    half = n // 2

    a = np.concatenate([left, right])
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:

    Generalised arcsine family:

    # add epsilon to avoid zero‑division for pathological p
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _power_mixture(n: int, p: float, lam: float) -> np.ndarray:
    """
    Mixture: lam * arcsine(p) + (1‑lam) * uniform.
    """
    a1 = _power_arcsine(n, p)
    a0 = _uniform(n)
    a = lam * a1 + (1.0 - lam) * a0
    return a / a.sum()


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1} (mid‑point rule).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _gaussian_like(n: int, gamma: float) -> np.ndarray:
    """
    Discrete Gaussian‑like shape:
        a_i ∝ exp(-γ·(i - μ)^2)
    μ is the centre of the interval.

    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * (i - mu) ** 2)
    return a / a.sum()


def _laplace_like(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Laplace‑like shape:
        a_i ∝ exp(-α·|i - μ|)

    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * np.abs(i - mu))
    return a / a.sum()

    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

        100_000.0, 200_000.0,
        500_000.0, 1_000_000.0,
        2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # allocate a generous number of inner iterations per β
        max_iter_beta = 12_000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current point on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv wrt a (via convolution with reversed a)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient in logits (mirror space)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Sub‑gradient is taken at the active index (argmax of the convolution).
    Returns the best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))          # active convolution index

        # Build sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Diminishing stepsize
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # occasional exact evaluation
        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk on the simplex.
    A random proportion of mass is transferred from one index to another.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)

    best_seq = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling schedule
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # Transfer a random fraction (scaled by sigma) of a[i] to j
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure no negative values due to numerical noise
        if new_a[i] < -1e-15:
            continue

        new_val = _objective_surrogate(new_a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length ``n``) that minimises
    ``evaluate_sequence`` under a 1000 s time budget.
    Returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin to avoid overflow

    # ------------------------------------------------------------------
    # 0️⃣ Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣ Assemble a very rich seed pool
    # ------------------------------------------------------------------
    seeds = []

    # Basic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Power‑law arcsine (coarse grid)
    for p in np.linspace(0.15, 1.0, 18):
        seeds.append(_power_arcsine(n, float(p)))

    # Mixture arcsine + uniform
    for p in (0.4, 0.5, 0.6):
        for lam in np.linspace(0.0, 1.0, 9):
            seeds.append(_power_mixture(n, float(p), float(lam)))

    # Symmetric Beta (α = β)
    for alpha in np.linspace(0.2, 2.0, 19):
        seeds.append(_beta_asymmetric(n, float(alpha), float(alpha)))

    # Asymmetric Beta
    for (alpha, beta) in ((0.3, 1.0), (0.5, 1.5), (1.5, 0.5), (2.0, 0.8)):
        seeds.append(_beta_asymmetric(n, float(alpha), float(beta)))

    # Gaussian‑like shapes
    for gamma in np.logspace(-6, -2, 12):
        seeds.append(_gaussian_like(n, float(gamma)))

    # Laplace‑like shapes
    for alpha in np.logspace(-4, -1, 12):
        seeds.append(_laplace_like(n, float(alpha)))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Warm‑start from previous run, if available
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial cheap evaluation of seeds (surrogate)
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    # Sort by surrogate value and keep a handful of the best
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(8, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing (≈ 15 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.15
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best after Adam")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing (majority of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 4.0   # keep a few seconds for final walk
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=5.0,
                                           max_iters=6_000_000)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best after EG")

    # ------------------------------------------------------------------
    # 5️⃣ Final tiny annealing walk (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation (warm‑start)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
