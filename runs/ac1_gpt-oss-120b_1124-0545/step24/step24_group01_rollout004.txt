"""
Proposes a step‑function (non‑negative vector whose sum is 1) that
minimises the supplied ``evaluate_sequence`` function.

The search proceeds in three phases:

1️⃣  **Seed generation & cheap evaluation** – a large, diverse pool of
    candidate probability vectors (uniform, arcsine, triangular,
    many symmetric Beta(α,α) densities and low‑concentration Dirichlet
    draws) is built.  The cheap surrogate 2·n·max(conv(a,a)) is used to
    keep the best few seeds.

2️⃣  **Mirror‑descent optimisation (exponentiated‑gradient / Adam‑style)**
    on the *exact* max‑convolution.  The sub‑gradient of the active
    convolution index is used, updates are performed in log‑space and
    stay on the probability simplex via a soft‑max.  Time is split
    evenly among the top seeds, fixing the bug of the previous
    implementation where later seeds received a negative time slice.

    After each EG run a deterministic greedy refinement (mass transfer
    from the coordinate with the largest sub‑gradient to the one with the
    smallest) is applied.

3️⃣  **Final stochastic local search** – a cheap random walk that moves a
    small fraction of mass between random coordinates while keeping
    improvements (occasionally accepting worse moves with a tiny
    temperature to escape shallow local minima).

When the allotted CPU time (``budget_s``) is exhausted the best vector
found is evaluated with ``evaluate_sequence`` and stored globally as
``height_sequence_1``.  The same routine can be called repeatedly; it
will reuse a previously found solution (if its length matches) as an
additional seed.

The algorithm is deliberately lightweight, uses only ``numpy`` and the
standard library and fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search: 2·n·max(conv(seq, seq)).
    ``seq`` is assumed to sum to 1 (or will be normalised by the caller).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density (Beta(½,½) on a grid)."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _exp_gradient_momentum(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 8.0,
                           beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) with Adam‑style momentum.
    Operates on the simplex via a soft‑max mapping.  Returns the best
    sequence seen (according to the exact surrogate) and its value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    # initialise log‑probabilities (mirror variables)
    logit = np.log(a + 1e-30)
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # find all indices that are (essentially) active
        # tolerance 1e-12 is enough for double precision
        active = np.where(conv >= max_val * (1.0 - 1e-14))[0]

        # sub‑gradient averaged over the active set
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * a[rev]

        g /= max(1, active.size)          # average if more than one active

        # mirror‑gradient (log‑space) for exponentiated‑gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam‑style momentum (first moment only – the second moment is omitted
        # for speed; the scalar step‑size already decays with sqrt(t)).
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        eta = eta0 / math.sqrt(it)

        # update in log‑space and map back to the simplex
        logit = logit - eta * mom
        a = _softmax(logit)

        # occasional exact evaluation
        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
        # early exit if we already beat the target
        if best_val < 1.503:
            break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 200_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy refinement: repeatedly move a small fraction of
    mass from the coordinate with the largest sub‑gradient to the one with
    the smallest sub‑gradient, accepting only improving moves.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val, best_seq, a = val, new_a.copy(), new_a
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept; a very small Metropolis acceptance rule is used
    to occasionally escape flat plateaus.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    a = best_seq.copy()
    cur_val = best_val
    T = 0.005  # tiny temperature
    cooling = 0.99995

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # occasional uphill move
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a, cur_val = new_a, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_a.copy()
        T *= cooling
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the allotted
    time is stored globally as ``height_sequence_1`` and returned as a
    plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣ Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Symmetric Beta(α,α) with many α values (U‑shapes, arcsine‑like and bell‑shapes)
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with several concentrations (highly irregular seeds)
    concs = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0)
    for conc in concs:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣ Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(8, len(seed_infos))          # a slightly larger elite set
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Momentum EG optimisation (exact max‑convolution)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Momentum EG optimisation (exact) ---")
        remaining_seeds = len(top_seeds)
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() >= deadline:
                break
            # recompute a fair time slice for the *current* seed
            time_left = deadline - time.time()
            # guarantee a minimum of 0.5 s for each remaining seed
            per_seed_deadline = time.time() + max(0.5, time_left / remaining_seeds)
            remaining_seeds -= 1

            print(f"  EG run #{sidx} (target slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _exp_gradient_momentum(seed_vec, per_seed_deadline)
            print(f"   → surrogate after EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (EG)")

            # ------------------------------------------------------
            # 2️⃣‑b Greedy refinement of the EG result
            # ------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # --------------------------------------------------------------
    # 3️⃣ Final random local refinement (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=350_000,
                                                sigma=0.012)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 4️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call (so that a later run can start from here)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
