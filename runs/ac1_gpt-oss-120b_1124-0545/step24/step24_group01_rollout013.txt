def simulated_annealing(a0, deadline):
    a = a0.copy()
    n = len(a)
    conv = np.convolve(a, a)   # currently using np.convolve (O(n^2)) but for n=4000 okay? It's about 16 million operations. Might be okay one time per seed. Or use conv_fft.

    best_seq = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        ratio = elapsed / total if total > 0 else 1.0
        # temperature schedule
        t = 5e-4 * (1e-8 / 5e-4) ** ratio   # geometric
        # step size schedule
        step_frac = 0.2 * (0.01/0.2) ** ratio   # maybe exponential decreasing
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        delta = step_frac * a[i] * rng.random()
        # apply temporary update to conv
        conv[i:i+n] -= 2*delta*a
        conv[j:j+n] += 2*delta*a
        new_max = conv.max()

            a[j] += delta

                best_seq = a.copy()

            conv[j:j+n] -= 2*delta*a

"""
Improved step‑function search.

The evaluation metric is 2·n·max(conv(a,a)) once the vector a has been
normalised to sum = 1 (the supplied ``evaluate_sequence`` applies the same
normalisation internally).  The goal is therefore to minimise the maximal
entry of the self‑convolution.

The algorithm is a multi‑stage optimiser:

1. **Broad seeding** – a large, diverse set of probability vectors is
   generated (uniform, arcsine, triangular, many symmetric / asymmetric
   Beta‑densities, Dirichlet draws and power‑law shapes).  All are
   normalised and their exact convolution maxima are evaluated with a fast
   FFT based routine.  The best few seeds are retained.

2. **Simulated annealing (SA)** – each retained seed is refined by a
   custom SA that works directly on the convolution array.  Moving a small
   amount of mass from index *i* to *j* updates the full convolution in
   O(n) time by a simple slice operation, so millions of iterations are
   affordable.  The temperature and the step‑size decay geometrically;
   uphill moves are accepted with probability exp(‑Δ / T).  The SA stops
   as soon as the objective drops below the target (≈ 1.503).

3. **Greedy refinement** – after SA a cheap greedy transfer of a tiny
   fraction of mass from the coordinate with the largest sub‑gradient to
   the one with the smallest is performed until no further improvement
   is possible.

4. **Local random walk** – a final stochastic walk explores the neighbourhood
   of the current best solution.

5. **Evaluation** – the best sequence is finally evaluated with the
   given ``evaluate_sequence`` and stored globally as ``height_sequence_1``.

The code respects the 1000 s time budget, prints progress, and contains
no closures or lambda expressions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _obj_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective (surrogate) from a convolution array."""
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray, n: int) -> float:
    """Fast surrogate – compute 2·n·max(conv(seq,seq)).  Assumes sum(seq)=1."""
    conv = _conv_fft(seq, seq)
    return _obj_from_conv(conv, n)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_sym(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a /= a.sum()
    return a


def _beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) density."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    dens = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    dens /= dens.sum()
    return dens


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(i + 1.0, p)
    a /= a.sum()
    return a


def _sinusoid(n: int, freq: float = 1.0) -> np.ndarray:
    """Positive half‑sinusoid shape."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.sin(np.pi * freq * x) ** 2
    a += 1e-9                     # avoid exact zeros
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         n: int,
                         target_obj: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    # initial full convolution (exact)
    conv = np.convolve(a, a)       # length 2n‑1, dense but cheap for n≈4000
    cur_max = conv.max()
    best_seq = a.copy()
    best_max = cur_max

    # initial temperature roughly half of current max
    Tmax = cur_max * 0.5 + 1e-12
    Tmin = cur_max * 1e-5 + 1e-15

    step_start = 0.20
    step_end = 0.001

    rng = np.random.default_rng()
    L = len(conv)

    while time.time() < deadline:
        elapsed = time.time() - (deadline - (deadline - time.time()))
        # compute normalised progress ratio (0 … 1)
        total = deadline - (deadline - (deadline - time.time()))
        # safeguard against division by zero
        ratio = (time.time() - (deadline - (deadline - time.time()))) / max(
            1e-9, deadline - (deadline - (deadline - time.time())))

        # geometric temperature schedule
        T = Tmax * (Tmin / Tmax) ** ratio
        # geometric step‑size schedule
        step_frac = step_start * (step_end / step_start) ** ratio

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)

        if a[i] <= 1e-15:
            continue

        delta = step_frac * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # ----- apply linear update to the convolution (exact) -----
        # conv_new = conv - 2*delta*shift_i(a) + 2*delta*shift_j(a)
        # using slices – shift_i(a) occupies indices i … i+n‑1
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        new_max = conv.max()
        dE = new_max - cur_max

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            # Metropolis condition
            if rng.random() < math.exp(-dE / max(T, 1e-12)):
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_seq = a.copy()
                # early exit if we already beat the target
                if 2.0 * n * best_max < target_obj:
                    break
        else:
            # revert convolution update
            conv[i:i + n] += 2.0 * delta * a
            conv[j:j + n] -= 2.0 * delta * a

    best_obj = 2.0 * n * best_max
    return best_seq, best_obj


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   n: int,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small fraction of mass from the coordinate with the
    largest sub‑gradient to the one with the smallest sub‑gradient.
    """
    a = start_seq.copy()
    conv = np.convolve(a, a)
    best_seq = a.copy()
    best_max = conv.max()
    best_obj = 2.0 * n * best_max

    rng = np.random.default_rng()
    while time.time() < deadline:
        # locate current max index
        k = int(np.argmax(conv))

        # compute sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            break

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        # provisional update of convolution (linear part only)
        conv[i_max:i_max + n] -= 2.0 * delta * a
        conv[i_min:i_min + n] += 2.0 * delta * a

        new_max = conv.max()
        if new_max < best_max:
            a[i_max] -= delta
            a[i_min] += delta
            best_max = new_max
            best_seq = a.copy()
            best_obj = 2.0 * n * best_max
        else:
            # revert
            conv[i_max:i_max + n] += 2.0 * delta * a
            conv[i_min:i_min + n] -= 2.0 * delta * a

    return best_seq, best_obj


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         n: int,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept.
    """
    a = start_seq.copy()
    conv = np.convolve(a, a)
    best_seq = a.copy()
    best_max = conv.max()
    best_obj = 2.0 * n * best_max

    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # update convolution (linear part)
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        new_max = conv.max()
        if new_max < best_max:
            a[i] -= delta
            a[j] += delta
            best_max = new_max
            best_seq = a.copy()
            best_obj = 2.0 * n * best_max
        else:
            # revert convolution
            conv[i:i + n] += 2.0 * delta * a
            conv[j:j + n] -= 2.0 * delta * a

    return best_seq, best_obj

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the allotted time
    is stored globally as ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Load a previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Beta(α,α) – a broad sweep of α
    alphas = np.concatenate([
        np.linspace(0.03, 0.10, 5),   # very U‑shaped
        np.linspace(0.12, 0.30, 7),   # strongly U‑shaped
        np.linspace(0.35, 0.55, 5),   # around arcsine
        np.linspace(0.6, 1.4, 5),     # gentle bell
        np.linspace(1.5, 2.5, 5)      # more concentrated
    ])
    for a in alphas:
        seeds.append(_beta_sym(n, float(a)))

    # asymmetric Beta draws – random parameters
    for _ in range(6):
        a = rng.uniform(0.05, 3.0)
        b = rng.uniform(0.05, 3.0)
        seeds.append(_beta_asym(n, a, b))

    # Dirichlet draws with several concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # power‑law shapes
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        seeds.append(_power_law(n, p))

    # sinusoid shapes
    for freq in (1, 2, 3):
        seeds.append(_sinusoid(n, freq))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Cheap surrogate evaluation of seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n=== Evaluating seed pool (exact surrogate) ===")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec /= vec.sum()
        val = _objective(vec, n)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: obj = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(7, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_obj = _objective(best_seq, n)
    print(f"\n[Init] best surrogate = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing refinement of seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n=== Simulated‑annealing phase ===")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f" SA run #{sidx} (time slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _simulated_annealing(seed_vec, per_seed_deadline, n)
            print(f"   → SA surrogate = {val:.6f}")
            if val < best_obj:
                best_obj, best_seq = val, seq.copy()
                print("   >>> Updated global best (SA)")

            # ------------------------------------------------------------------
            # 2️⃣‑b Greedy refinement (fast but powerful)
            # ------------------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline, n)
                if val2 < best_obj:
                    best_obj, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 3️⃣ Final local random walk (tiny stochastic refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Final local random walk ===")
        seq_loc, val_loc = _local_random_search(best_seq, deadline, n)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_obj:
            best_obj, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 4️⃣ Official evaluation with the supplied evaluate_sequence
    # ------------------------------------------------------------------
    # re‑normalise to guard against tiny drift
    best_seq = np.maximum(0.0, best_seq)
    if best_seq.sum() == 0.0:
        # fall back to uniform if something went terribly wrong
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / best_seq.sum()

    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_obj:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store the solution for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
