def isotonic_regression(y):
    n = len(y)
    solution = y.astype(float).copy()
    weight = np.ones(n)
    i = 0
    while i < n-1:
        if solution[i] < solution[i+1]:  # for decreasing we want solution[i] >= solution[i+1]
            # merge pools i and i+1
            tot_weight = weight[i] + weight[i+1]
            tot_solution = (solution[i] * weight[i] + solution[i+1] * weight[i+1]) / tot_weight
            solution[i] = tot_solution
            solution[i+1] = tot_solution
            weight[i] = tot_weight
            weight[i+1] = tot_weight
            # backtrack
            j = i-1
            while j >= 0 and solution[j] < solution[j+1]:
                tot_weight = weight[j] + weight[j+1]
                tot_solution = (solution[j] * weight[j] + solution[j+1] * weight[j+1]) / tot_weight
                solution[j] = tot_solution
                solution[j+1] = tot_solution
                weight[j] = tot_weight
                weight[j+1] = tot_weight
                j -= 1
            i = max(j, 0)
        else:
            i += 1
    return solution

                weight[j-1] = tot_weight
                weight[j] = tot_weight

def _isotonic_decreasing(v):
    # v is a 1-D numpy array, enforce decreasing (non-increasing)
    n = len(v)
    # Work on reversed array to get increasing isotonic regression
    y = v[::-1]
    # Initialize blocks
    blocks = []
    for i in range(n):
        blocks.append([y[i], 1])  # [avg, weight]
        # Merge while last two averages are decreasing (i.e., block[-2][0] > block[-1][0])
        while len(blocks) >= 2 and blocks[-2][0] > blocks[-1][0] - 1e-12:
            avg1, w1 = blocks[-2]
            avg2, w2 = blocks[-1]
            tot_w = w1 + w2
            new_avg = (avg1 * w1 + avg2 * w2) / tot_w
            blocks[-2] = [new_avg, tot_w]
            blocks.pop()
    # Expand blocks
    y_proj = np.empty(n, dtype=float)
    idx = 0
    for avg, w in blocks:
        cnt = int(round(w))
        y_proj[idx:idx+cnt] = avg
        idx += cnt
    return y_proj[::-1]
```

Propose a candidate step‑function.

The optimiser builds on the previous multi‑stage approach but adds two
important changes:

* The exact “exponentiated‑gradient” phase now uses **all active offsets**
  (all indices where the convolution attains the max up to a tiny tolerance)
  instead of a single offset.  This gives a richer sub‑gradient and lets the
  algorithm escape shallow plateaus.

* The initial pool of starting points is enlarged (previous best, two
  exponential decays, a random Dirichlet, a random decreasing vector) and the
  short Adam+EG screening time per candidate is reduced, allowing the final
  exact EG phase to run for many more iterations.

All stages respect the global 1000 s deadline.  The best sequence found is
stored in the global variable ``height_sequence_1`` for possible continuation
and returned as a plain list of floats.
"""

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when sum(seq) = 1:  2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – useful for Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad_a(a: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient wrt a) of the smoothed max‑convolution.
    The variable a is assumed to sum to 1.
    """
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weight for each convolution entry (softmax over conv values)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # exact gradient of the smooth surrogate
    return obj, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              beta_start: float = 0.5,
              beta_end: float = 5e7,
              alpha: float = 0.35) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad_a = _smooth_obj_and_grad_a(_softmax(v), n, beta)

        # chain rule: gradient w.r.t. v (log‑softmax vars)
        a = _softmax(v)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution
# (now using all active offsets)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 12_000_000,
                           eta0: float = 6.0,
                           eps_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.
    Uses a sub‑gradient that averages over all offsets whose value is within
    ``eps_active`` of the global maximum.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # active set – every index whose value is close to the maximum
        active = np.where(conv >= max_val - eps_active)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size                     # uniform weighting

        # sub‑gradient of (average) active entries
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1 : n - 1 + n]            # d/da of the (averaged) max

        # map to gradient w.r.t. log‑softmax variables
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_v
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (unchanged)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 4_500_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])     # donor index
            q = int(rng.integers(0, n))           # random receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1   # small fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs several increasingly intensive optimisation phases and
    returns the best sequence found before the allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # problem dimensionality (feel free to change, but 4000 works well)
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a diverse set of initial vectors
    # ------------------------------------------------------------------
    inits = []

    # a) reuse previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                inits.append(arr)
        except Exception:
            pass

    # b) exponential decays with a few rates
    for rate in (0.001, 0.003, 0.006):
        inits.append(_exp_decay(n, rate=rate))

    # c) random Dirichlet draws (α = 0.5) – then sort descending to obtain a
    #    decreasing shape (still a valid probability vector)
    for _ in range(2):
        vec = rng.gamma(shape=0.5, scale=1.0, size=n)
        vec = np.sort(vec)[::-1]               # decreasing order
        vec = vec / vec.sum()
        inits.append(vec)

    # d) completely random decreasing vector (uniform draws then sort)
    for _ in range(2):
        vec = rng.random(n)
        vec = np.sort(vec)[::-1]
        vec = vec / vec.sum()
        inits.append(vec)

    # keep unique candidates (hash based)
    uniq_inits = []
    seen = set()
    for vec in inits:
        h = hash(vec.tobytes())
        if h not in seen:
            uniq_inits.append(vec)
            seen.add(h)

    # ------------------------------------------------------------------
    # 2. Short screening: Adam + (multi‑active) EG for each init
    # ------------------------------------------------------------------
    print("[Stage 1] Screening short Adam+EG runs")
    best_seq = None
    best_val = float("inf")
    per_candidate_time = max(5.0, (deadline - time.time() - 300.0) / max(1, len(uniq_inits)))
    # keep a 5 s buffer for later stages and a 300 s reserve for the long EG

    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline - 300.0:
            break
        sub_deadline = time.time() + per_candidate_time
        print(f"  Candidate {idx+1}/{len(uniq_inits)} – time budget ≈ {per_candidate_time:.1f}s")
        seq_a, _ = _adam_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            max_iters=250_000,
            beta_end=5e7,
            alpha=0.35,
        )
        seq_e, val_e = _exp_gradient_opt_multi(
            start_seq=seq_a,
            n=n,
            deadline=sub_deadline,
            max_iters=6_000_000,
            eta0=8.0,
        )
        if val_e < best_val:
            best_val, best_seq = val_e, seq_e.copy()
            print(f"    New best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Long exact EG refinement (uses the whole remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] Long exact EG refinement (multi‑active)")
        seq_eg, val_eg = _exp_gradient_opt_multi(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=30_000_000,
            eta0=6.0,
        )
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive hill‑climb (swap / perturb / mass‑shift)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] Hill‑climb")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=5_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
