def _power_law_distribution(n, p):
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    a = np.power(prod, p)
    a[prod <= 0] = 0.0
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
        s = 1.0
    return a / s
```

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.
=============================================================
Algorithm
---------
1. **Problem size** – select a moderate length (default 4000).  The
   objective is scale‑invariant, so we keep the sum of the vector equal
   to 1 (soft‑max parametrisation).

2. **Analytic family** – a two‑parameter family  
   a_i ∝ ((i+δ)·(n‑δ‑i))^p ,  δ>0 , p<0.  The classical arcsine corresponds
   to (δ,p)=(0.5,‑0.5).  We perform a cheap grid search on (δ,p), then a
   simple hill‑climbing refinement, to obtain a very good seed.

3. **Smoothed max‑convolution descent** – minimise the smooth surrogate  
   F_β(a)= (1/β)·log Σ_k e^{β·(a*a)[k]} using Adam on the
   log‑softmax parameters.  β is increased linearly from a small value
   (≈0.2) to a large value (≈30) so that the optimiser finally sees the
   true max‑convolution.

4. **Exact max‑convolution polishing** – a long exponentiated‑gradient
   (mirror‑descent) pass on the exact objective, using the sub‑gradient
   of the active convolution entry.

5. **Time management** – the whole routine respects the 1000 s budget,
   keeping a 5‑second safety margin.  Progress is printed so that the
   experiment can be monitored.

The best sequence found is stored in the global variable ``height_sequence_1``,
allowing continuation across separate runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate_value(a: np.ndarray) -> float:
    """Exact surrogate (identical to evaluate_sequence when sum(a)=1):
       2·n·max(a*a)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _general_distribution(n: int, delta: float, p: float) -> np.ndarray:
    """Two‑parameter family a_i ∝ ((i+δ)(n‑δ‑i))^p with p<0."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + delta) * (n - delta - i)
    # protect against negative/zero products (should not happen for 0<δ<n)
    prod = np.where(prod > 0, prod, 1.0)
    a = prod ** p
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _resample_to_n(arr: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    if old_n == n_new:
        return arr.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, n_new, dtype=np.float64)
    return np.interp(x_new, x_old, arr)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (soft‑max) objective
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. the log‑softmax
    parameters `log_a`.  Returns ∂F/∂log_a of size n.
    """
    a = _softmax(log_a)                # probability vector (sum = 1)
    conv = _conv_fft(a, a)             # length 2n‑1

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                       # shape (2n‑1,)

    # g_a = 2·Σ_k w_k·shift(a)  (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)          # length 3n‑2
    g_a = 2.0 * w_a[n - 1: n - 1 + n]   # centre slice → length n

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.2,
                      beta_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smooth max‑convolution surrogate.
    Returns the best probability vector found and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)          # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _surrogate_value(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # gradient of the smoothed objective
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate exact surrogate (no smoothing)
        cur_seq = _softmax(log_a)
        cur_val = _surrogate_value(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max‑conv objective)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution
    objective.  Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_value(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_value(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (vector) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget,
    builds a small parametric family, performs a quick grid/hill‑climb,
    refines the best candidate with Adam on a smoothed max‑convolution,
    and finishes with an exponentiated‑gradient polishing pass.
    The best vector is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  4000 gives a good trade‑off between
    # solution quality and speed for the given budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1) Build an initial pool of seeds.
    # ------------------------------------------------------------------
    seeds = []

    # (a) Classical arcsine (already a very good starting point)
    seeds.append(_arcsine_distribution(n))

    # (b) Re‑use the sequence from a previous run (if it has the right size)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] != n:
                arr = _resample_to_n(arr, n)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                seeds.append(arr / s)
        except Exception:
            pass

    # (c) Uniform and a few exponentials
    seeds.append(_uniform(n))
    for rate in (0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exp_decay(n, rate))

    # (d) Random Dirichlet draws for diversity
    for _ in range(3):
        seeds.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # ------------------------------------------------------------------
    # 2) Scan a 2‑parameter family a_i ∝ ((i+δ)(n‑δ‑i))^p .
    # ------------------------------------------------------------------
    print("[Info] Scanning (δ,p) family...")
    delta_vals = np.linspace(0.25, 0.75, 11)    # 0.25 … 0.75
    p_vals = np.linspace(-0.65, -0.35, 13)     # -0.65 … -0.35
    best_family_seq = None
    best_family_val = float("inf")
    best_delta = best_p = None

    for delta in delta_vals:
        for p in p_vals:
            seq = _general_distribution(n, delta, p)
            val = _surrogate_value(seq)
            if val < best_family_val:
                best_family_val = val
                best_family_seq = seq
                best_delta, best_p = delta, p

    # Fine‑grained hill‑climbing around the best (δ,p)
    step_delta, step_p = 0.02, 0.01
    for _ in range(12):   # up to 12 refinement rounds
        improved = False
        for d_shift in (-step_delta, 0, step_delta):
            for p_shift in (-step_p, 0, step_p):
                if d_shift == 0 and p_shift == 0:
                    continue
                delta_c = best_delta + d_shift
                p_c = best_p + p_shift
                if not (0.0 < delta_c < n):
                    continue
                seq = _general_distribution(n, delta_c, p_c)
                val = _surrogate_value(seq)
                if val < best_family_val:
                    best_family_val = val
                    best_family_seq = seq
                    best_delta, best_p = delta_c, p_c
                    improved = True
        if not improved:
            # shrink the search steps
            step_delta *= 0.5
            step_p *= 0.5
        if step_delta < 1e-4:
            break

    print(f"[Info] Best (δ,p) = ({best_delta:.4f}, {best_p:.4f}) → surrogate = {best_family_val:.6f}")
    seeds.append(best_family_seq)

    # ------------------------------------------------------------------
    # 3) Quick evaluation of all seeds (exact surrogate) and keep the
    #    top‑k promising ones.
    # ------------------------------------------------------------------
    evaluated = []
    for s in seeds:
        seq = np.maximum(s, 0.0)
        if seq.sum() == 0.0:
            continue
        seq = seq / seq.sum()
        val = _surrogate_value(seq)
        evaluated.append((val, seq))

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [seq for _, seq in evaluated[:top_k]]

    # ------------------------------------------------------------------
    # 4) Adam‑softmax optimisation on the selected seeds.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # allocate a generous number of Adam iterations (will stop early if time runs out)
    max_total_adam_iters = 250_000
    it_per_candidate = max_total_adam_iters // max(1, len(top_candidates))

    for idx, cand in enumerate(top_candidates, start=1):
        if time.time() > deadline - 30:
            print("[Info] Approaching deadline – stopping further Adam runs.")
            break
        print(f"[Adam] Candidate {idx}/{len(top_candidates)} – max_iters={it_per_candidate}")
        seq, val = _adam_softmax_opt(cand,
                                     n,
                                     deadline,
                                     max_iters=it_per_candidate,
                                     lr0=0.5,
                                     beta_start=0.2,
                                     beta_end=30.0)
        print(f"       -> surrogate after Adam = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"       *** New global best (surrogate) = {best_val:.6f}")

    # If nothing succeeded, fall back to the best from the quick scan.
    if best_seq is None:
        best_seq = best_family_seq.copy()
        best_val = best_family_val
        print("[Info] Falling back to best analytic family seed.")

    # ------------------------------------------------------------------
    # 5) Polishing with exponentiated‑gradient (exact max‑conv).
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("\n[Polish] Starting exponentiated‑gradient polishing")
        # Allocate as many iterations as the remaining budget allows
        remaining = deadline - time.time()
        # Rough estimate: ~0.001 s per iteration → 1000 iters per second
        max_polish_iters = int(min(800_000, remaining * 900))
        best_seq, best_val = _exp_gradient_opt(best_seq,
                                               n,
                                               deadline,
                                               max_iters=max_polish_iters,
                                               eta0=12.0)
        print(f"[Polish] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final evaluation with the supplied evaluate_sequence() function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Final result ===")
        print(f"Surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
