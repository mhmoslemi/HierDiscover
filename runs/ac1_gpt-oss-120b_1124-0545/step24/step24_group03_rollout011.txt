"""
Propose a step‑function whose self‑convolution has a very small L∞‑norm.

Algorithm
---------

1. **Seed generation** – a few analytic families (uniform, inverse‑sqrt,
   power‑law, exponential, Dirichlet) are normalised to sum = 1.  
   The best few seeds (by the exact surrogate 2·n·max(conv)) are kept.

2. **LP–based flattening** – for a fixed sequence `f` (length *n*) we compute
   `M = max(conv(f,f))`.  The linear programme

        maximise   Σ g_i
        subject to Σ_i f_i·g_{k-i} ≤ M   for all shifts k
                    g_i ≥ 0

   finds a “partner’’ `g` whose overlap with `f` never exceeds the current
   worst autocorrelation `M`.  Because the constraints are linear in `g`,
   `scipy.optimize.linprog` can solve it quickly (the constraint matrix is
   a sparse Toeplitz matrix built with ``scipy.sparse.diags``).

   The new candidate is a convex combination `f←(1‑t)·f + t·ĝ` where `ĝ` is the
   normalised LP solution.  The step size `t` is tried from 0.5 down to
   0.03125; the first value that strictly reduces `M` is accepted.
   The procedure is repeated until no improvement is possible or the time
   budget is exhausted.

3. **Optional fine‑tuning** – after the LP phase a short L‑BFGS optimisation
   of the smooth‑max surrogate (log‑sum‑exp of the convolution) is performed
   to polish the result.

The whole routine respects the supplied time budget (default 1000 s), stores
the best sequence in the global variable ``height_sequence_1`` and returns it
as a plain Python list.  All numbers are non‑negative, the sequence is

# basic numeric utilities (FFT‑based convolution)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for a sum‑normalised seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all give L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    a /= a.sum()
    return a


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# LP‑based improvement of a single seed
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                deadline: float,
                max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the linear programme
          maximise Σ g_i
          s.t.    conv(f,g) ≤ M   (all shifts)
                  g_i ≥ 0
    where M = max(conv(f,f)).  After each solve a convex mixing step is
    performed.  Returns the best sequence found and its surrogate value.
    """
    if _opt is None or _sp is None:      # pragma: no cover
        return f, _objective_surrogate(f)

    n = len(f)
    best_seq = f.copy()
    best_conv = _conv_fft(best_seq, best_seq)
    best_M = best_conv.max()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # Build the sparse Toeplitz constraint matrix A (size (2n‑1) × n)
        #   A[k, j] = f_{k‑j}   whenever 0 ≤ k‑j < n, else 0
        # ------------------------------------------------------------------
        A = _sp.diags(f, offsets=range(n),
                      shape=(2 * n - 1, n),
                      format='csc')

        b = np.full(2 * n - 1, best_M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                 # maximise Σg_i  ⇔ minimise -Σg_i
        bounds = [(0.0, None)] * n

        # ------------------------------------------------------------------
        # Solve the LP
        # ------------------------------------------------------------------
        try:
            res = _opt.linprog(c,
                               A_ub=A,
                               b_ub=b,
                               bounds=bounds,
                               method='highs',
                               options={'presolve': True, 'time_limit': max(int(deadline - time.time()), 1)})
        except Exception as e:               # pragma: no cover
            print(f"[LP] exception at iteration {it}: {e}")
            break

        if not res.success:
            # LP failed – abort further LP attempts
            print(f"[LP] failed at iteration {it} (status {res.status})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            print("[LP] LP returned zero vector")
            break
        g /= sum_g                     # normalise to sum = 1

        # ------------------------------------------------------------------
        # Convex mixing – try a few step sizes, keep the first improvement
        # ------------------------------------------------------------------
        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = (1.0 - t) * best_seq + t * g
            # already normalised because both summands sum to 1
            conv_cand = _conv_fft(cand, cand)
            M_cand = conv_cand.max()
            if M_cand < best_M - 1e-14:          # strict improvement
                best_seq = cand
                best_M = M_cand
                best_conv = conv_cand
                improved = True
                break
        if not improved:
            # No improvement with any step size – stop the LP loop
            break

        # ------------------------------------------------------------------
        # Logging
        # ------------------------------------------------------------------
        obj = 2.0 * n * best_M
        print(f"[LP‑Iter {it:2d}] max‑conv = {best_M:.6e}  →  surrogate = {obj:.6f}")

    return best_seq, 2.0 * n * best_M

            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:                # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v - np.max(v))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Generate a modest seed set and keep the few best
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _inverse_sqrt(n),
        _power(n, 0.5),
        _power(n, 0.7),
        _exp_decay(n, 1e-3),
        _exp_decay(n, 5e-3),
        _random_dirichlet(n, 0.02, rng),
        _random_dirichlet(n, 0.1, rng),
    ]

    # include previously found sequence if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    seeds.append(arr / s)
        except Exception:                     # pragma: no cover
            pass

    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:2]]
    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:2]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based iterative improvement (main workhorse)
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # allocate the bulk of the time to the LP routine
    lp_deadline = time.time() + (deadline - time.time()) * 0.75
    seq_lp, val_lp = _lp_improve(best_seq, lp_deadline, max_iters=30)
    if val_lp < best_val:
        best_seq, best_val = seq_lp, val_lp
        print(f"[LP] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional LBFGS smoothing if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n[LBFGS] polishing step")
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=4000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_seq, best_val = seq_lbfgs, val_lbfgs
            print(f"[LBFGS] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                        # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
