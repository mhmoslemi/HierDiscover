"""
Optimiser for the “evaluate_sequence” black‑box.

The goal is to minimise

        2·n·max(conv(a,a)) / (∑a)²

which is scale‑invariant, so we work on the simplex
(∑a = 1, a_i ≥ 0).  The optimiser proceeds in four stages:

1️⃣  **Rich seed pool** – analytic families (uniform, triangular,
    power‑law a_i ∝ ((i+½)(n‑½‑i))^{‑p}, symmetric/asymmetric Beta,
    sinusoidal) and several Dirichlet draws.  A fine sweep over the
    shape parameters is performed and the best few seeds are kept.

2️⃣  **Fast gradient polishing** – a mirror‑descent (exponentiated‑gradient)
    method on the exact max‑convolution.  It works directly on the
    logits (log‑softmax) to keep the simplex constraints.  The step‑size
    decays as η₀/√t.  The majority of the time‑budget is spent here.

3️⃣  **Multi‑restart exploration** – after the long run, a few short
    EG runs are started from perturbed versions of the current best
    point (mixture with a low‑concentration Dirichlet).  This helps
    escape shallow basins.

4️⃣  **Final annealing** – a tiny Metropolis‑style walk on the simplex
    (mass transfer between two random coordinates) for the last few
    seconds to polish the solution.

All heavy operations (convolution, gradients) use a real FFT
implementation for speed (O(n log n)).  The best sequence found is
stored globally as ``height_sequence_1`` and also returned as a list of
floats.

The optimiser is deliberately lightweight: it avoids the costly
smooth‑max (log‑sum‑exp) Adam phase used previously and gives the
exponentiated‑gradient almost the whole 1000 s budget, which empirically
yields a lower bound than the reference implementation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a normalised sequence (∑seq = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) on {0,…,n‑1}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _sinusoidal(n: int, power: float = 1.0) -> np.ndarray:
    """
    Sinusoidal shape:
        a_i ∝ sin(π·(i+½)/n)^{power}
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** power
    a = a + 1e-12       # avoid exact zeros
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
    Optimises directly on the simplex via log‑softmax parametrisation.
    Returns the best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk on the simplex.
    Moves transfer a random fraction of mass from i → j.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_val = _objective_surrogate(new_a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

        seeds.append(_sinusoidal(n, pwr))

    # Dirichlet draws with several concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds generated before deadline.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(6, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient polishing (majority of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 5.0   # reserve a few seconds for the final walk
        print("\n--- Exponentiated‑gradient (main run) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=5.0,
                                           max_iters=5_000_000)
        print(f"[EG‑main] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best after main EG")

    # ------------------------------------------------------------------
    # 4️⃣ Short multi‑restart explorations (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most 20 % of the remaining time to a few short restarts
        restart_time_budget = min(remaining * 0.2, 30.0)
        n_restart = 3
        print("\n--- Multi‑restart EG explorations ---")
        for r in range(n_restart):
            if time.time() >= deadline:
                break
            # mixture of current best with a fresh low‑concentration Dirichlet
            pert = _random_dirichlet(n, 0.01, rng)
            alpha_mix = 0.07
            start_vec = (1.0 - alpha_mix) * best_seq + alpha_mix * pert
            start_vec = start_vec / start_vec.sum()
            sub_deadline = time.time() + restart_time_budget / n_restart
            seq_r, val_r = _exp_gradient_opt(start_vec, sub_deadline,
                                             eta0=5.0,
                                             max_iters=2_000_000)
            print(f"[Restart {r+1}] surrogate = {val_r:.6f}")
            if val_r < best_val:
                best_seq, best_val = seq_r.copy(), val_r
                print("   >>> Updated global best after restart")

    # ------------------------------------------------------------------
    # 5️⃣ Final annealing walk (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealing walk ---")
        seq_loc, val_loc = _anneal_opt(best_seq, deadline,
                                       init_temp=0.02,
                                       final_temp=1e-7,
                                       sigma=0.015)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for potential next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
