def _next_pow2(x: int) -> int:

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _local_multi_search(start_seq: np.ndarray,
                        n: int,
                        deadline: float,

    """
    Multi‑scale stochastic walk on the simplex.
    Alternates three kinds of proposals:
      * small mass transfer from i → j,
      * full swap of values a[i] ↔ a[j],
      * random permutation of the whole vector.
    Only strictly improving moves are kept.
    """

    cur_val = _surrogate(cur)

    # Parameters controlling the schedule
    sigma_start = 0.12      # relative size of mass‑transfer steps
    sigma_end = 0.004
    sigma = sigma_start
    decay = 0.97            # geometric decay per block
    block_iters = 4000      # attempts per sigma value

    while time.time() < deadline:
        # ---- mass‑transfer block ----
        for _ in range(block_iters):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * cur[i] * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            if new[i] < 0.0:
                continue
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- swap block ----
        for _ in range(block_iters // 2):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # ---- occasional global permutation ----
        if rng.random() < 0.12:   # roughly one permutation every 8 blocks
            perm = rng.permutation(n)
            new = cur[perm]
            val = _surrogate(new)
            if val < cur_val:
                cur, cur_val = new, val

        # decay sigma, stop if it becomes tiny
        sigma = max(sigma_end, sigma * decay)
        if sigma <= sigma_end + 1e-9:
            # after reaching minimum sigma we still continue with swaps only
            block_iters = max(2000, block_iters // 2)

    return cur, cur_val
