def _smooth_adam(start_seq, deadline, lr0=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
                  beta_start=1.0, beta_end=200.0, beta_iters=500000):
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute probs
        a = _softmax(logit)
        # compute conv
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # schedule beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_iters)
        # compute weights
        exp_arg = beta * conv - np.max(beta * conv)  # for stability
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0/len(w))
        else:
            w = w / w_sum

        # compute gradient g via correlation
        Lw = len(w)
        La = n
        L = Lw + La - 1
        Nfft = _next_pow2(L)
        Wfft = np.fft.rfft(w, n=Nfft)
        Afft = np.fft.rfft(a, n=Nfft)
        corr = np.fft.irfft(Wfft * np.conj(Afft), n=Nfft)[:L]
        g = 2.0 * corr[:n]

        # mirror gradient
        grad_log = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)  # decreasing learning rate
        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

        # if we reach time, break
        if time.time() > deadline:
            break

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq = final_a
        best_val = final_val
    return best_seq, best_val

def _annealed_hill_climb(start_seq, deadline, attempts=500_000, sigma_start=0.02, sigma_end=0.001):
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    rng = np.random.default_rng()
    # compute schedule: sigma = sigma_start * (sigma_end / sigma_start)^{progress}
    total_attempts = attempts
    for attempt in range(attempts):
        if time.time() > deadline:
            break
        progress = attempt / total_attempts
        sigma = sigma_start * ((sigma_end / sigma_start) ** progress)  # exponential decay
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

"""
Minimal‐max‑convolution optimiser

The objective to be minimised is

    E(a) = 2·n·max_k (a * a)[k]      with   a_i ≥ 0 ,   Σ a_i = 1

(the supplied ``evaluate_sequence`` does exactly this, after clamping).

We work on the probability simplex by keeping the *logits*  z  and using
``a = softmax(z)``.  A smoothed version of the ``max`` is obtained with a
log‑sum‑exp trick:

    Fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

For β → ∞,  Fβ → max_k (a*a)[k].
Its gradient on the simplex can be written in closed form using the
convolution of a and a weighting vector `w` (the softmax of the
convolution values).  The gradient w.r.t. the logits is

    g = a * ( 2·corr(w,a) – ⟨2·corr(w,a), a⟩ ),

where ``corr`` is the (linear) cross‑correlation.
Both the convolution and the correlation are computed by a single FFT,
so each iteration costs O(n log n) with a very small constant.

The optimiser runs an Adam‑style update with a *β* schedule:
we start with a smooth objective (small β) to obtain a good direction,
then increase β so that the optimisation focuses on the true maximum.
After the Adam phase a cheap greedy refinement and a long annealed
hill‑climb finish the search.

The whole routine respects the 1000 s budget and stores the best
sequence in the global ``height_sequence_1`` variable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation:  (w ⋆ a)[i] = Σ_j w[i+j]·a[j].
    Returns the full length L = len(w)+len(a)-1, of which
    the first ``len(a)`` entries are the needed values.
    """
    Lw = len(w)
    La = len(a)
    L = Lw + La - 1
    N = _next_pow2(L)
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(W * np.conj(A), n=N)[:L]
    return corr


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution, no safety clamping)
# ----------------------------------------------------------------------


def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)), a is assumed to sum to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_adam(start_seq: np.ndarray,
                 deadline: float,
                 lr0: float = 0.15,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 beta_start: float = 0.8,
                 beta_end: float = 150.0,
                 beta_ramp_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max‑convolution.
    Returns the best sequence seen and its exact surrogate value.
    """
    n = len(start_seq)
    # initialise logits (mirror variables)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = _softmax(logit).copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1

        # current probability vector
        a = _softmax(logit)

        # exact convolution (needed for the surrogate)
        conv = _conv_fft(a, a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothed max -----
        # schedule for beta (smoothness)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)

        # weights proportional to exp(beta*conv)
        exp_arg = beta * conv - np.max(beta * conv)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smoothed objective:
        #   g = 2 * corr(w, a)   (first n entries)
        corr = _corr_fft(w, a)          # length = 3n-2
        g = 2.0 * corr[:n]

        # mirror‑gradient on the simplex (log‑space)
        grad_log = a * (g - np.dot(g, a))

        # Adam update (bias‑correction omitted for speed)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        # diminishing step‑size, square‑root decay
        lr = lr0 / math.sqrt(it)

        # apply update in log‑space
        logit = logit - lr * m / (np.sqrt(v) + eps)

        # occasional early exit if we have already passed a known target
        if best_val < 1.503:
            break

    # final check (in case the very last iterate is best)
    a_final = _softmax(logit)
    final_val = _objective_surrogate(a_final)
    if final_val < best_val:
        best_seq, best_val = a_final, final_val

    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 130_000,
                   delta_frac: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Moves a small fraction of mass from the index with the largest
    sub‑gradient to the one with the smallest, keeping only improving moves.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # active sub‑gradient at k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
    return best_seq, best_val


def _annealed_hill_climb(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 400_000,
                         sigma_start: float = 0.025,
                         sigma_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing with an exponentially decaying step size.
    Only strictly improving moves are kept.
    """
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    for t in range(attempts):
        if time.time() > deadline:
            break
        # exponential cooling
        sigma = sigma_start * ((sigma_end / sigma_start) ** (t / attempts))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored globally as ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored solution of matching length (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Beta(α,α) – a wide family of U‑ and bell‑shapes
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Cheap surrogate evaluation of seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Smoothed Adam optimisation (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Smoothed Adam optimisation (exact max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f"  Run #{sidx} (time slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _smooth_adam(seed_vec, per_seed_deadline)
            print(f"   → surrogate after Adam = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (Adam)")

            # ----------------------------------------------------------
            # 2️⃣‑b  Greedy refinement of the Adam result
            # ----------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 3️⃣  Final annealed hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final annealed hill‑climb refinement ---")
        seq_loc, val_loc = _annealed_hill_climb(best_seq, deadline,
                                                attempts=400_000,
                                                sigma_start=0.025,
                                                sigma_end=0.001)
        print(f"[Hill‑climb] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (hill‑climb)")

    # ------------------------------------------------------------------
    # 4️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
