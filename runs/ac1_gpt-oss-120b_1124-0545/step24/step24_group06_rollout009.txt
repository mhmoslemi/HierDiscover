def _lp_expand(p, M, deadline):

    n = len(p)

    # Build constraint matrix
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)  # shape n*n
    cols = np.tile(np.arange(n), n)
    data = np.repeat(p, n)

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)

    # Objective: maximize sum(q) => minimize -sum(q)
    c = -np.ones(n)

    # Bounds:
    bounds = [(0, None)] * n

    # Use method='highs'
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})

    if res.success:
        q = res.x
        return q
    else:
        print("LP failed:", res.message)
        return None

def _lp_optimize(p, M, deadline):
    import numpy as np, scipy.sparse as sp, scipy.optimize as opt
    n = len(p)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(p, n)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    remaining = deadline - time.time()
    if remaining <= 0:
        return None
    # solve with high accuracy
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': max(1.0, remaining)})
    except Exception as e:
        print("LP exception:", e)
        return None
    if not res.success:
        print("LP not successful:", res.message)
        return None
    q = res.x
    # Ensure non-neg small thresholds
    q = np.maximum(q, 0.0)
    return q

"""
Enhanced step‑function optimiser with a linear‑programming “flatten‑convolution”
phase.

The optimiser works in three stages:

1️⃣ **Diverse seeding** – we generate many probability vectors:
   uniform, triangular, spike, several Beta(α,α) laws (both by point‑wise
   density and by exact bin‑integrals), Dirichlet draws and the classic
   arcsine law – both the simple midpoint version and the exact integral
   version.  The best seed according to the exact surrogate
   `2·n·max(conv(p,p))` is kept.

2️⃣ **LP‑based flattening** – given a current distribution *p* we solve the
   linear program  

        maximise   Σ q_i
        subject to conv(p,q) ≤ M,   q ≥ 0 ,

   where *M* = max(conv(p,p)).  The LP finds the most mass that can be added
   without raising the current worst convolution value.  After normalising
   the LP solution we move a fraction *t* towards it (mirror‑descent step).
   Repeating this a few times drives the self‑convolution towards a flat
   shape (in the limit the sum‑distribution would be uniform).

3️⃣ **Intensive local refinement** – a fast incremental walk that repeatedly
   transfers a small amount δ from one coordinate to another, updating the
   full convolution in O(n) time.  Hundreds of thousands of trial moves are
   evaluated inside the remaining time budget.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from scipy.special import betainc, betaincinv, beta, betainc, betaincinv  # betainc is enough

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(p, p))
    ``p`` is assumed to be a probability vector (sum = 1).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _arcsine_center(n: int) -> np.ndarray:
    """Mid‑point discretisation of the arcsine law (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _arcsine_integral(n: int) -> np.ndarray:
    """Exact bin‑integral of the arcsine density on [0,1]."""
    # CDF(x) = (2/π) * arcsin(sqrt(x))
    i = np.arange(n + 1, dtype=np.float64) / n
    cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(i))
    a = np.diff(cdf)                     # bin masses
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _beta_integral(n: int, alpha: float) -> np.ndarray:
    """Discrete symmetric Beta(α,α) obtained by integrating the density over bins."""
    # Regularised incomplete beta is the CDF for Beta(α,α)
    xs = np.arange(n + 1, dtype=np.float64) / n
    cdf = betainc(alpha, alpha, xs)        # shape (n+1,)
    a = np.diff(cdf)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# LP‑based flattening step
# ----------------------------------------------------------------------


def _lp_expand(p: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve   maximise Σ q_i   subject to   conv(p, q) ≤ M,   q ≥ 0.
    Returns the optimal q (may be all‑zero if the LP fails) or None.
    """
    n = len(p)
    # Ensure strict positivity to avoid unbounded variables
    eps = 1e-14
    p = np.maximum(p, eps)

    # Build the convolution matrix A where A @ q = conv(p, q)
    # The pattern (row = i + j, col = j, data = p_i) does NOT depend on p.
    rows = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(p, n)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)               # maximise → minimise –Σ q_i
    bounds = [(0.0, None)] * n

    # Give linprog a few seconds but respect the global deadline
    remaining = max(0.0, deadline - time.time())
    opts = {"presolve": True, "time_limit": max(1.0, remaining)}

    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method="highs", options=opts)
    except Exception as exc:                 # pragma: no cover
        print("[LP] exception:", exc)
        return None

    if not res.success:
        print("[LP] failed:", res.message)
        return None

    q = np.maximum(res.x, 0.0)               # clip small negatives
    return q


# ----------------------------------------------------------------------
# Intensive local search (same as the previous best implementation)
# ----------------------------------------------------------------------


def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast greedy walk on the simplex: repeatedly move a random amount δ from
    one coordinate to another, updating the full convolution in O(n) time.
    Returns the best sequence found and its exact surrogate value.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)
    best_a = a.copy()
    best_max = np.max(conv)
    best_val = 2.0 * n * best_max

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        if cand_val < best_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_max = cand_max
            best_val = cand_val
            accepted += 1

            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                last_recalc = accepted

        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget (seconds) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load previous solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a rich seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [
        _uniform(n),
        _triangular(n),
        _spike(n),
        _arcsine_center(n),
        _arcsine_integral(n),
    ]

    # Beta(α,α) by exact bin integration – a finer sweep around the optimal 0.5
    beta_alphas = [0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
                   0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80,
                   0.90, 1.00, 1.20, 1.50, 2.00]
    for a in beta_alphas:
        seeds.append(_beta_integral(n, a))

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Seed evaluation – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ LP‑based flattening phase (few aggressive updates)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- LP flattening phase ---")
        lp_deadline = time.time() + 0.30 * (deadline - time.time())   # about 30 % of the remaining budget
        p = best_seq.copy()
        for lp_iter in range(3):                # a handful of LP steps is enough
            if time.time() > lp_deadline:
                break
            conv = _conv_fft(p, p)
            M = conv.max()
            q = _lp_expand(p, M, lp_deadline)
            if q is None:
                break
            sum_q = q.sum()
            if sum_q <= 1.0 + 1e-12:
                break                     # no slack – we are essentially optimal
            g = q / sum_q                 # normalised direction
            # mixing weight – a moderate step keeps stability
            t = 0.2
            p = (1.0 - t) * p + t * g
            p = np.maximum(p, 0.0)
            p /= p.sum()                  # re‑normalise
            new_val = _objective_surrogate(p)
            print(f"[LP] iter {lp_iter + 1}: surrogate = {new_val:.6f}  (sum_q={sum_q:.3f})")
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_seq = p.copy()
        # after the loop we keep the best sequence found
        best_seq = p if best_seq is None else best_seq

    # ------------------------------------------------------------------
    # 4️⃣ Intensive local refinement (major time consumer)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Intensive local refinement (incremental) ---")
        seq_loc, val_loc = _intensive_local_search(best_seq, n,
                                                   deadline,
                                                   sigma_start=0.05,
                                                   rng=rng)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc

    # ------------------------------------------------------------------
    # 5️⃣ Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
