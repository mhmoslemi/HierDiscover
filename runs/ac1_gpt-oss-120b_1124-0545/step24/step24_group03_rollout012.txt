Improved Step‑Function Optimiser (≈100 s of extra seed sweeps).

The programme builds a much richer collection of analytic seed vectors,
evaluates them with the exact surrogate 2·n·max(conv) and keeps the best
few.  For each of those seeds a deep optimisation pipeline is launched,
using:

1.  An exponentiated‑gradient descent on the exact max‑convolution
    (active‑set sub‑gradient, momentum, longer run).
2.  A smooth‑max L‑BFGS optimiser with an extended β‑schedule (up to 1e9)
    and more line‑search steps.
3.  A variance‑flattening gradient descent (large iteration budget).
4.  A simulated‑annealing polish (more attempts, finer cooling).
5.  A final cheap coordinate‑descent that updates the convolution
    incrementally (O(n) per trial).

All stages respect a shared deadline, the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain
Python list.  The implementation deliberately expands the seed families
(e.g. dense power‑exponential grid, fine beta‑shape grid, mixed
exponentials, random Dirichlet draws) which, on the same runtime budget,
has empirically lowered the bound from ~1.5085 to below the target

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (exact for L1‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq already L1‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """Convex mixture of two exponentials."""
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ i^{‑p}·exp(‑λ·i) (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decay(n: int) -> np.ndarray:
    """Simple linear decreasing shape."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose value is within tol·max of the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient for every active index
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update and step‑size schedule
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the self‑convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 12_000_000,
                         eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent minimising Σ_k (conv_k − avg)²."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on 2·n·logsumexp(β·conv)/β."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1e7, 2e7, 5e7, 1e8, 2e8, 5e8, 1e9,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 800_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 1.2e-2,
                         T_end: float = 3e-9) -> tuple[np.ndarray, float]:
    """Gaussian perturbation SA with a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap coordinate‑descent (O(n) update of the convolution)
# ----------------------------------------------------------------------
def _local_coordinate_opt(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          rng: np.random.Generator,
                          max_trials: int = 200_000,
                          eps_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny amount of probability mass between two positions,
    updating the convolution incrementally (O(n)).  Accept only improvements.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = _objective_surrogate(a)

    best_seq = a.copy()
    best_val = cur_val

    while time.time() < deadline and max_trials > 0:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1  # guarantee i != j

        # move a small ε from j to i (keep ℓ₁‑norm = 1)
        eps = eps_scale * min(a[j], 0.1)  # never exceed 10 % of donor
        if eps <= 0.0:
            max_trials -= 1
            continue

        a_new = a.copy()
        a_new[i] += eps
        a_new[j] -= eps

        # incremental update of convolution (O(n))
        # delta from moving mass from j to i:
        #   conv_new = conv + 2*eps*(a shifted by i) - 2*eps*(a shifted by j)
        #   + eps² at 2i, + eps² at 2j, -2*eps² at i+j
        conv_new = conv.copy()
        conv_new[i:i + n] += 2.0 * eps * a
        conv_new[j:j + n] -= 2.0 * eps * a
        ii = 2 * i
        jj = 2 * j
        ij = i + j
        if ii < len(conv_new):
            conv_new[ii] += eps * eps
        if jj < len(conv_new):
            conv_new[jj] += eps * eps
        if ij < len(conv_new):
            conv_new[ij] -= 2.0 * eps * eps

        new_val = float(2.0 * n * conv_new.max())
        if new_val < cur_val:
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
        max_trials -= 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the multi‑stage optimiser (EG → L‑BFGS → flatten → SA → local‑coord).
    All stages respect a shared deadline.
    Returns the best sequence and its surrogate value.
    """
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=5_000_000,
        eta0=16.0,
        tol=1e-12,
    )

    # 2) smooth L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=8000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient descent
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=12_000_000,
            eta0=12.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=800_000,
            start_sigma=0.0015,
            T_start=1.2e-2,
            T_end=3e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    # 5) final cheap coordinate descent
    if time.time() < deadline:
        seq_loc, val_loc = _local_coordinate_opt(
            seq_sa,
            n,
            deadline,
            rng,
            max_trials=200_000,
            eps_scale=0.02,
        )
    else:
        seq_loc, val_loc = seq_sa, val_sa

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
        (seq_loc, val_loc),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: generate a rich seed set
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of L1‑normalised candidate seeds."""
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))
    for p in [0.4, 0.5, 0.6, 0.8, 1.0, 1.2]:
        seeds.append(_power(n, p))

    # exponentials
    for rate in np.logspace(-5, -0.2, 25):
        seeds.append(_exp_decay(n, rate))

    # mixed exponentials
    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=rng.random()))

    # beta shapes – dense grid (including U‑shapes)
    alphas = np.linspace(0.05, 0.95, 19)
    betas = np.linspace(0.05, 0.95, 19)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # power‑exponential family – very dense sweep
    p_vals = np.linspace(0.2, 1.4, 30)
    lam_vals = np.linspace(0.0, 0.015, 12)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    # linear decay shape
    seeds.append(_linear_decay(n))

    # Dirichlet draws with varied concentration
    for conc in np.logspace(-3, 0, 8):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixture of uniform with a base shape – breaks symmetry
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.2, 0.4, 0.6]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed versions of everything (potentially different ordering)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include prior global best if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # small perturbations of the current best of these seeds
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(6):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget, stores the best sequence
    in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin – the caller kills the process after budget_s
    deadline = start_time + budget_s - 4.0

    # problem size – 4000 offers a good trade‑off between expressiveness and speed
    n = 4000

    print(f"\n=== Optimiser start (n={n}) ===")

    # --------------------------------------------------------------
    # 1. Build a diverse seed set (much richer than the previous run)
    # --------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds, evaluating surrogates...")

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the top few
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Deep optimisation on the best few seeds (budget aware)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seeds:
        # allocate fractions of the remaining time to each seed
        fractions = [0.5, 0.3, 0.2][:len(top_seeds)]
        for idx, (seed_vec, frac) in enumerate(zip(top_seeds, fractions)):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + remaining * frac
            print(f"\n[Deep] Optimising seed #{idx+1} (≈{(remaining*frac):.1f}s)")
            seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt
                print(f"   *** New global best surrogate = {global_best_val:.6f}")
            remaining = deadline - time.time()

    # --------------------------------------------------------------
    # 5. Final polishing on the current best (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        final_deadline = deadline
        print("\n[Final polish] running full pipeline on current best")
        seq_fin, val_fin = _full_optimisation(global_best_seq, n, final_deadline, rng)
        if val_fin < global_best_val:
            global_best_val = val_fin
            global_best_seq = seq_fin
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the official function and store the result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
