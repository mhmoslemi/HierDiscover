"""
Propose a high‑quality non‑negative step‑function.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Diversified start vectors** – previous best, several exponentials,
   a triangular shape and Dirichlet draws with different concentration
   parameters.
2. **Smooth Adam optimisation** – the variables are the log‑softmax
   (a simplex representation).  A log‑sum‑exp surrogate of the max‑convolution
   is optimised with an Adam optimiser; the smoothing parameter β is
   ramped from ≈0.2 to 1e10, so the final stages see the true max.
3. **Exact mirror‑descent** – sub‑gradient descent on the true
   max‑convolution (exponentiated‑gradient / mirror descent) with a
   1/√t decaying step size.  Runs until almost the full remaining budget.
4. **Population mixing** – keep the best few candidates, repeatedly mix two
   of them, add a tiny Gaussian jitter and run a short Adam refinement.
   This helps escape local minima.
5. **Aggressive hill‑climb** – simulated‑annealing style local search with
   swap, Gaussian perturbation and a targeted “mass‑shift” move that pulls
   probability mass away from the most‑responsible pair for the active
   convolution offset.
6. **Final evaluation** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` function and stored globally for possible
   continuation runs.

All steps respect the global deadline (``budget_s`` seconds) and employ
FFT‑based convolution for speed.  The returned list sums to 1 (the
normalisation is irrelevant for the black‑box evaluator because the
objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a probability vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The variable v lives in log‑softmax space; a = softmax(v)
    is a probability vector (sum = 1).
    """
    a = _softmax(v_local)                     # simplex point
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp smoothing
    exp_beta = np.exp(beta * (conv - max_c))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in analysis)
    w = exp_beta / sum_exp                     # soft weights, sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # back‑prop through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              beta_start: float = 0.2,
              beta_end: float = 1e10,
              alpha: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_seq + eps)               # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + adam_eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) on the true max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 20_000_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The iterate stays on the simplex via a log‑space rep (softmax).
    Step size decays as η0 / sqrt(t).
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active offset
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / perturb / mass‑shift)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 6_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with three move types:
      * simple swap of two entries,
      * Gaussian perturbation of all entries,
      * targeted mass‑shift away from the most contributing pair.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue

        elif move_type < 0.90:                   # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                    # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])   # donor index (largest product)
            q = int(rng.integers(0, n))           # receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1   # small fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs several optimisation phases and returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size (fixed, but can be changed if desired)
    n = 4000

    # --------------------------------------------------------------
    # 0. Initialise a pool of promising candidates (seq, surrogate)
    # --------------------------------------------------------------
    pool = []   # list of (seq, surrogate)

    # a) reuse previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s == 0.0:
                    arr = np.full(n, 1.0 / n)
                else:
                    arr = arr / s
                pool.append((arr, _objective(arr)))
        except Exception:
            pass

    # b) a few hand‑crafted shapes
    init_vecs = []

    # exponential decays (different rates)
    for rate in (0.001, 0.003, 0.006):
        init_vecs.append(_exp_decay(n, rate))

    # simple triangular shape
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    tri = 1.0 - np.abs(i - centre) / (centre + 1.0)
    tri = np.maximum(tri, 0.0)
    tri = tri / tri.sum()
    init_vecs.append(tri)

    # Dirichlet draws with various concentration parameters
    for alpha in (0.3, 0.8, 2.0):
        for _ in range(2):
            vec = rng.gamma(shape=alpha, scale=1.0, size=n)
            vec = vec / vec.sum()
            init_vecs.append(vec)

    # ensure uniqueness (hash on raw bytes)
    uniq_inits = []
    seen = set()
    for vec in init_vecs:
        h = hash(vec.tobytes())
        if h not in seen:
            uniq_inits.append(vec)
            seen.add(h)

    # --------------------------------------------------------------
    # 1. Short Adam runs on each diverse start (screening)
    # --------------------------------------------------------------
    print("[Stage 1] short Adam runs (smooth surrogate)")
    best_seq = None
    best_val = float("inf")

    for idx, init in enumerate(uniq_inits):
        if time.time() > deadline:
            break
        sub_deadline = time.time() + 30.0
        print(f"  Adam start {idx + 1}/{len(uniq_inits)}")
        seq, val = _adam_opt(
            start_seq=init,
            n=n,
            deadline=sub_deadline,
            max_iters=300_000,
            beta_start=0.2,
            beta_end=1e10,
            alpha=0.4,
        )
        pool.append((seq, val))
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"    New best surrogate = {best_val:.6f}")

    # keep only the top few
    pool.sort(key=lambda t: t[1])
    pool = pool[:5]

    # --------------------------------------------------------------
    # 2. Exact mirror‑descent on the best candidate (runs to deadline)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 2] exact mirror‑descent (sub‑gradient) on best")
        seq, val = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=20_000_000,
            eta0=8.0,
        )
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"    Improved surrogate = {best_val:.6f}")
        # keep in pool
        pool.append((best_seq, best_val))
        pool.sort(key=lambda t: t[1])
        pool = pool[:5]

    # --------------------------------------------------------------
    # 3. Population mixing + short Adam refinements (making jumps)
    # --------------------------------------------------------------
    while time.time() + 20.0 < deadline:
        if len(pool) < 2:
            break
        # pick two distinct pool members uniformly
        idxs = rng.choice(len(pool), size=2, replace=False)
        seq1 = pool[idxs[0]][0]
        seq2 = pool[idxs[1]][0]

        lam = rng.random()
        mix_seq = lam * seq1 + (1.0 - lam) * seq2
        # tiny jitter to avoid exact linear combos
        mix_seq += 0.001 * rng.normal(size=n)
        mix_seq = np.maximum(mix_seq, 0.0)
        s = mix_seq.sum()
        if s == 0.0:
            continue
        mix_seq = mix_seq / s

        sub_deadline = time.time() + 8.0
        seq, val = _adam_opt(
            start_seq=mix_seq,
            n=n,
            deadline=sub_deadline,
            max_iters=80_000,
            beta_start=0.1,
            beta_end=1e9,
            alpha=0.3,
        )
        if val < pool[-1][1] or len(pool) < 5:
            if len(pool) < 5:
                pool.append((seq, val))
            else:
                pool[-1] = (seq, val)
            pool.sort(key=lambda t: t[1])
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"    Mix/Adam improved best → {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Aggressive hill‑climb (simulated annealing)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n[Stage 3] hill climbing")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=6_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"    Hill‑climb improved best → {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final evaluation with the black‑box function
    # --------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
