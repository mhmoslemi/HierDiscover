"""
Improved step‑function optimizer.

The routine consists of four phases:

1️⃣  **Exploratory parametric search** – random draws from several
    low‑dimensional families (pure exponential, truncated exponential,
    power‑law and a mixture of two exponentials).  Each draw is normalized
    to a probability vector and evaluated with the exact surrogate
    ``2·n·max(conv(p,p))``.  The best few candidates are retained.

2️⃣  **Exact mirror‑descent (exponentiated‑gradient)** – a sub‑gradient
    method on the true max‑convolution.  The gradient is the convex
    combination of all active constraints (all offsets whose value is
    within a tiny tolerance of the current maximum).  A simple
    1/√t step‑size schedule is employed; the iterate lives on the simplex
    via a log‑softmax representation.

3️⃣  **Aggressive simulated‑annealing hill‑climb** – three move operators
    (swap, Gaussian perturbation, and a targeted mass‑shift that drains
    mass from the most offending pair).  The temperature is cooled
    geometrically.

4️⃣  **Final polishing** – a few short Adam runs from a noisy copy of the
    current best solution.

All phases respect the global deadline (default 1000 s) and constantly
track the best sequence seen.  The final sequence is returned and stored
in the global variable ``height_sequence_1`` for possible continuation
runs.

The implementation avoids closures and lambda‑expressions, uses only
NumPy and the standard library, and works on a single thread (the time
budget already accounts for the full 2‑CPU limit).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real arrays using an FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    return np.fft.irfft(fx * fy, n=N)[:L]


# ----------------------------------------------------------------------
# Elementary utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(p,p)).
    The input is assumed to be a probability vector (sum = 1).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families for the exploratory stage
# ----------------------------------------------------------------------
def _exp_decay_seq(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _truncated_exp_seq(n: int, rate: float, L: int) -> np.ndarray:
    """a_i ∝ exp(-rate·i) for i<L, zero otherwise, then normalised."""
    a = np.empty(n, dtype=np.float64)
    a[:L] = np.exp(-rate * np.arange(L, dtype=np.float64))
    a[L:] = 0.0
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _mix_exp_seq(n: int, rate1: float, rate2: float, w: float) -> np.ndarray:
    """Mixture w·exp(-rate1·i) + (1‑w)·exp(-rate2·i)."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-rate1 * i) + (1.0 - w) * np.exp(-rate2 * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Exact mirror‑descent on the true max‑convolution (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent with a sub‑gradient of the *max* convolution.
    The iterate stays on the simplex via a log‑softmax parameterisation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space representation (softmax of log_a yields a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices that are (almost) active – tolerance is very small
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient: sum of contributions of all active offsets
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * a[rev]

        # step‑size schedule η = η0 / √t
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g                     # entropic mirror step
        a = _softmax(log_a)                  # back to the probability simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same moves as the reference code,
# with a few tuned parameters)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 6_000_000) -> tuple[np.ndarray, float]:
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (geometric)
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)

        move_type = rng.random()
        if move_type < 0.45:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue

        elif move_type < 0.90:                   # Gaussian perturbation
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns /= s

        else:                                     # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = 0.1 * min(cur_seq[donor], 0.5 * cur_seq[donor])
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns /= s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small Adam optimiser on the log‑softmax parametrisation (used only in
# a final polishing step)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 150_000,
              beta_start: float = 0.5,
              beta_end: float = 1e9,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for β (log‑space)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        # smooth surrogate (log‑sum‑exp) and its gradient
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = conv.max()
        shift = max_c
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v -= step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The function runs several
    optimisation phases within the allotted time and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # problem size – can be changed if desired
    n = 4000

    print(f"[+] Starting optimisation (n={n}, time budget ≈{budget_s}s)")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic parametric search
    # ------------------------------------------------------------------
    param_end = time.time() + min(180.0, budget_s * 0.2)   # at most 20 % of total time
    best_seq = None
    best_val = float("inf")
    top_k = []                     # keep the 5 best candidates for later refinement
    K = 5

    families = ["exp", "trunc_exp", "power", "mix"]
    while time.time() < param_end:
        fam = rng.choice(families)

        if fam == "exp":
            rate = rng.uniform(0.0001, 0.02)
            seq = _exp_decay_seq(n, rate)

        elif fam == "trunc_exp":
            rate = rng.uniform(0.0001, 0.04)
            L = rng.integers(int(0.05 * n), n)          # support length
            seq = _truncated_exp_seq(n, rate, L)

        elif fam == "power":
            p = rng.uniform(0.2, 5.0)
            seq = _power_law_seq(n, p)

        else:  # mix
            rate1 = rng.uniform(0.0001, 0.04)
            rate2 = rng.uniform(0.0001, 0.04)
            w = rng.random()
            seq = _mix_exp_seq(n, rate1, rate2, w)

        # evaluate surrogate (sequence already normalised)
        val = _objective_surrogate(seq)

        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Phase 1] New global best {best_val:.6f}")

        # maintain a small elite pool
        if len(top_k) < K:
            top_k.append((val, seq.copy()))
            top_k.sort(key=lambda x: x[0])
        else:
            if val < top_k[-1][0]:
                top_k[-1] = (val, seq.copy())
                top_k.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – exact mirror‑descent refinements from the elite pool
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective_surrogate(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        # split the remaining time evenly among the elite candidates
        per_candidate_deadline = time.time() + remaining / (len(top_k) + 1)
        for idx, (_, cand_seq) in enumerate(top_k):
            print(f"[Phase 2] Mirror‑descent start from elite #{idx + 1}")
            seq_opt, val_opt = _exp_gradient_opt(
                start_seq=cand_seq,
                n=n,
                deadline=per_candidate_deadline,
                max_iters=6_000_000,
                eta0=8.0,
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, seq_opt.copy()
                print(f"   --> improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase 3] Hill‑climb")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=5_000_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"   --> hill‑climb improvement {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam polishing (short random restarts)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("[Phase 4] Short Adam polish")
        noisy = best_seq + 0.005 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy /= s
        sub_dead = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_dead,
            max_iters=120_000,
            beta_end=5e8,
            alpha=0.28,
        )
        print(f"   restart surrogate = {val_rr:.6f}")
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print("   *** new best found ***")

    # ------------------------------------------------------------------
    # Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        final_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
