def _annealed_local_search(start_seq, n, deadline):
    rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    sigma = 0.1
    sigma_min = 0.001
    T = cur_val * 0.001  # small fraction of objective.
    T_min = cur_val * 1e-6
    dec_factor = 0.9995

    attempts = 0
    accepted = 0
    no_improve_counter = 0

    while time.time() < deadline:
        attempts += 1

        # source selection weighted by a
        src = rng.choice(n, p=a)  # ensures src likely non-zero
        # ensure source not zero
        if a[src] < 1e-12:
            continue
        # target selection uniform among indices not src
        tgt = rng.integers(0, n-1)
        if tgt >= src:
            tgt += 1  # ensures different

        delta = sigma * a[src] * rng.random()
        if delta <= 0:
            continue

        cand_conv = conv.copy()
        # update slices
        cand_conv[src:src+n] -= 2.0 * delta * a
        cand_conv[tgt:tgt+n] += 2.0 * delta * a
        idx2src = 2*src
        idx2tgt = 2*tgt
        idxst = src + tgt
        cand_conv[idx2src] += delta * delta
        cand_conv[idx2tgt] += delta * delta
        cand_conv[idxst] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max
        delta_val = cand_val - cur_val

        accept = False
        if delta_val < -1e-12:
            accept = True
        else:
            prob = math.exp(-delta_val / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1
            no_improve_counter = 0
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
        else:
            no_improve_counter += 1

        # periodic cooling
        if attempts % 1000 == 0:
            sigma = max(sigma_min, sigma * dec_factor)
            T = max(T_min, T * dec_factor)

        # reheat if stuck
        if no_improve_counter > 20000:
            # reheat
            sigma = min(0.2, sigma * 1.5)
            T = max(T_min, T * 1.5)
            no_improve_counter = 0
            print(f"[Anneal] Reheat at attempt {attempts}, sigma={sigma:.5f}, T={T:.6e}, best_val={best_val:.6f}")

        if accepted % 5000 == 0 and accepted > 0:
            conv = _conv_fft(a, a)  # recompute exactly
        # optionally log progress
        if attempts % 50000 == 0:
            print(f"[Anneal] attempts={attempts}, accepted={accepted}, best_val={best_val:.6f}, sigma={sigma:.5f}, T={T:.6e}")

"""
Annealed local‑search optimiser for the step‑function challenge.

The algorithm works in three stages:

1.  Load the best sequence from a previous run (global variable
    ``height_sequence_1``) or, if none exists, generate a handful of
    diverse seeds (uniform, arcsine, spike, Dirichlet draws,
    symmetric and asymmetric Beta distributions) and keep the one with
    the smallest surrogate value 2·n·max(conv(a,a)).

2.  Run a *simulated‑annealing* walk on the simplex.  A move transfers a
    small amount δ from a source index i to a target index j; the
    convolution is updated with a cheap O(n) slice operation (the same
    formula used in the reference implementation).  Improving moves are
    always accepted; uphill moves are accepted with probability
    ``exp(−Δ/T)`` where the temperature T is slowly cooled.  The step‑size
    factor σ is also annealed.  Periodic full recomputation of the
    convolution eliminates accumulated rounding error.

3.  The best sequence found is finally evaluated with the supplied
    ``evaluate_sequence`` function and stored in the global variable for
    the next invocation.

All helper functions are top‑level, there are no closures or lambdas,
and the routine respects a user‑specified time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate used during the search: 2·n·max(conv(seq,seq))."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on the centre of each bin."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _beta_distribution_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density discretised on the centre of each bin."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing incremental local search
# ----------------------------------------------------------------------


def _annealed_local_search(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           sigma_start: float = 0.10,
                           rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk on the simplex.
    Moves transfer a random δ from one coordinate to another.
    The convolution is updated incrementally; uphill moves may be
    accepted with probability exp(−Δ/T) where the temperature T is
    gradually cooled.
    Returns the best sequence encountered and its exact surrogate value.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    sigma = sigma_start
    sigma_min = 0.005
    sigma_decay = 0.9995

    # initial temperature – a small fraction of the objective
    T = max(1e-8, cur_val * 0.005)
    T_min = 1e-12
    T_decay = 0.9995

    attempts = 0
    accepted = 0
    last_full = 0
    no_improve = 0

    while time.time() < deadline:
        attempts += 1

        # --- pick source and target indices ---------------------------------
        src = rng.integers(0, n)
        if a[src] <= 1e-15:
            continue

        tgt = rng.integers(0, n - 1)
        if tgt >= src:
            tgt += 1  # ensure distinct

        # --- propose a transfer ------------------------------------------------
        delta = sigma * a[src] * rng.random()
        if delta <= 0.0:
            continue

        # cheap copy of current convolution
        cand_conv = conv.copy()
        # shift contributions
        cand_conv[src:src + n] -= 2.0 * delta * a
        cand_conv[tgt:tgt + n] += 2.0 * delta * a
        # diagonal / cross terms
        idx2src = 2 * src
        idx2tgt = 2 * tgt
        idxst = src + tgt
        cand_conv[idx2src] += delta * delta
        cand_conv[idx2tgt] += delta * delta
        cand_conv[idxst] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max
        delta_val = cand_val - cur_val

        # --- acceptance test -------------------------------------------------
        accept = False
        if delta_val < -1e-12:               # strict improvement
            accept = True
        else:
            prob = math.exp(-delta_val / max(T, 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1
            no_improve = 0
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
        else:
            no_improve += 1

        # --- cooling / reheating ---------------------------------------------
        if attempts % 1000 == 0:
            sigma = max(sigma_min, sigma * sigma_decay)
            T = max(T_min, T * T_decay)

        # reheat if we have been stuck for a long time
        if no_improve >= 20000:
            sigma = min(0.20, sigma * 1.5)
            T = max(T_min, T * 1.5)
            no_improve = 0
            print(f"[Anneal] Reheat at attempt {attempts:,} – sigma={sigma:.4f}, T={T:.2e}, best={best_val:.6f}")

        # occasional full recomputation to cure rounding errors
        if accepted - last_full >= 3000:
            conv = _conv_fft(a, a)
            last_full = accepted

        # progress reports (optional, harmless)
        if attempts % 50000 == 0:
            print(f"[Anneal] attempts={attempts:,} accepted={accepted} sigma={sigma:.4f} "
                  f"T={T:.2e} best_val={best_val:.6f}")

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – keep a couple of seconds free for the final evaluation
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load a previously found solution if it exists
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                start_seq = arr / arr.sum()
                print("[Init] Loaded previous solution.")
            else:
                start_seq = None
        except Exception:
            start_seq = None
    else:
        start_seq = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of seeds (each is a probability vector)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta families
    for a in (0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
              0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
              1.20, 1.50, 2.00):
        seeds.append(_beta_distribution(n, a))

    # Asymmetric Beta families (more expressive)
    for a, b in [(0.2, 1.5), (0.5, 2.0), (0.8, 0.3), (1.2, 0.6), (1.5, 0.2)]:
        seeds.append(_beta_distribution_asym(n, a, b))

    # Add the previously found vector if we have one
    if start_seq is not None:
        seeds.append(start_seq)

    # ------------------------------------------------------------------
    # 2️⃣ Choose the best seed according to the surrogate objective
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec, n)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    if best_seq is None:
        # This should never happen, but guard against it
        raise RuntimeError("No valid seed could be generated.")

    print(f"[Init] Best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Intensive simulated‑annealing local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_loc, val_loc = _annealed_local_search(best_seq, n, deadline,
                                                  sigma_start=0.10,
                                                  rng=rng)
        print(f"[Anneal] best surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (annealed)")

    # ------------------------------------------------------------------
    # 4️⃣ Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
