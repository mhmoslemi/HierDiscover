"""
Propose a step‑function (non‑negative vector) with a low
evaluate_sequence value.

Algorithm
---------
1. **Parametric sweep** – generate symmetric Beta‑type
   distributions p_i ∝ [(i+½)(n‑½‑i)]^e for many exponents *e*.
   This family interpolates between uniform (e=0) and strongly
   edge‑heavy shapes (e<0).  The best exponent is kept as a seed.

2. **Diverse seeding** – in addition to the best Beta‑seed we
   keep a few analytic seeds (arcsine, uniform, exponential) and
   a few random Dirichlet draws.

3. **Smoothed max‑convolution descent** – each seed is refined
   with an Adam optimiser applied to the smooth approximation

        F_β(p) = (1/β)·log Σ_k exp(β·(p∗p)_k)

   The temperature β is increased from 0.5 to 16.0 (larger than
   before) so the optimisation sees a sharper objective.
   The number of Adam iterations is adapted to the remaining
   time budget.

4. **Final polishing** – a classic exponentiated‑gradient (mirror
   descent) pass on the exact max‑convolution objective reduces
   any remaining bias.  The step size decays as η_t = η₀/√t.

5. **Time management** – the routine stops as soon as the hard
   deadline (budget_s seconds) is reached.  If it ever obtains
   a value ≤ 1.503 it returns immediately.

The best vector found is stored in the global name ``height_sequence_1``
so that a later run can continue from it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate  2·n·max(conv(seq,seq))  (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _beta_power_distribution(n: int, exponent: float) -> np.ndarray:
    """
    Symmetric Beta‑type distribution:
        p_i ∝ [(i+½)(n‑½‑i)]**exponent .
    exponent < 0 puts more mass on the edges, exponent > 0
    concentrates mass in the centre.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    a = prod ** exponent
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters `log_a`. Returns ∂F/∂log_a (size n).
    """
    a = _softmax(log_a)                        # probability vector
    conv = _conv_fft(a, a)                      # linear self‑convolution

    # soft‑max over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                                # w sums to 1

    # g_a = 2 Σ_k w_k · shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)                  # length 3n‑2
    g_a = 2.0 * w_a[n - 1: n - 1 + n]           # centre slice → length n

    # Chain rule through the soft‑max
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 16.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed max‑convolution objective.
    Returns the best probability vector found and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase the temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # gradient of the smoothed objective
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate the exact surrogate (not the smoothed version)
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirror descent on the exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (all contributions)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        # diagonal term should have weight 1, off‑diagonal weight 2
        # we use the same 2‑factor for simplicity (as in the reference)
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
