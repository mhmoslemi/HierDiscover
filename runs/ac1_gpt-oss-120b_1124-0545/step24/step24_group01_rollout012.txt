def _mirror_descent_opt(start_seq, deadline):
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)  # base
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)  # raw surrogate: 2*n*max(conv)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # select indices within epsilon tolerance of max
        # relative tolerance: 1e-9? perhaps using absolute
        tol = max_c * 1e-9 + 1e-12
        active = np.where(conv >= max_c - tol)[0]
        # compute avg gradient
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            rev = k - idx
            grad[idx] += 2.0 * a[rev]
        grad = grad / len(active)  # average

        # mirror gradient
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = 1.0 / (np.sqrt(v_hat) + eps)  # adapt step per coordinate

        # apply update: maybe scale by base learning rate
        base_lr = 0.5  # tune
        logit = logit - base_lr * eta * m_hat

        # map back to simplex
        a = _softmax(logit)

        if t % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        if best_val < 1.503:
            break
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq = a.copy(); best_val = final_val
    return best_seq, best_val

"""
Step‑function optimiser for ``evaluate_sequence``.

The optimiser works in three phases:

1. **Diversified seeding** – a collection of analytic probability vectors
   (uniform, arcsine, triangular, Beta(α,α), power‑law, exponential decay,
   random Dirichlet) are generated and normalised to sum = 1.  The five
   best seeds according to the exact surrogate 2·n·max(conv) are kept.

2. **Mirror‑descent on the max‑convolution** – an exponentiated‑gradient
   (mirror) descent with Adam‑style momentum minimises the exact
   ``max(conv(a,a))``.  Rather than using a single active index,
   the sub‑gradient is averaged over the *K* largest convolution entries,
   providing a smoother descent direction and better escape from plateaus.

3. **Refinement** – a greedy mass‑transfer step followed by a large
   stochastic walk on the simplex polish the solution.  The best
   sequence found is finally evaluated with ``evaluate_sequence`` and
   stored as ``height_sequence_1`` for reuse.

The routine respects the 1000 s budget, prints progress, and returns a
list of non‑negative heights that (empirically) improves on the previous
best bound (≈ 1.5085).  All operations are performed on the normalised
(simplex) representation; the final ``evaluate_sequence`` call is
scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used in the optimiser:
        2·n·max(conv(seq,seq)),  where seq sums to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector of length ``n``)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Symmetric (or asymmetric) Beta(α,β) on the discrete grid."""
    if beta is None:
        beta = alpha
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = a / a.sum()
    return a


def _powerlaw(n: int, exponent: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, exponent)
    a = a / a.sum()
    return a


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _mirror_descent_opt(start_seq: np.ndarray,
                        deadline: float,
                        eta0: float = 0.5,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8,
                        top_k: int = 8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent (exponentiated‑gradient) on the exact
    max‑convolution objective.  The sub‑gradient is the average of the
    gradients of the ``top_k`` largest convolution entries.
    """
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)            # initialise entropy mirror variables
    m = np.zeros_like(logit)            # first moment
    v = np.zeros_like(logit)            # second moment

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    while time.time() < deadline:
        t += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices of the K largest values (use argpartition for speed)
        k = min(top_k, conv.size)
        top_idx = np.argpartition(conv, -k)[-k:]

        grad = np.zeros_like(a)
        for kk in top_idx:
            i_start = max(0, kk - (n - 1))
            i_end = min(n - 1, kk)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = kk - idx
            grad[idx] += 2.0 * a[rev]
        grad = grad / float(len(top_idx))

        # mirror‑gradient (entropy Bregman)
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)

        # Adam updates on the dual variables (logits)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule – slowly decays with √t
        eta = eta0 / math.sqrt(t)

        # update in log‑space and pull back to simplex
        logit = logit - eta * (m_hat / (np.sqrt(v_hat) + eps))
        a = _softmax(logit)

        # occasional exact evaluation
        if t % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # early exit if target already beaten
                if best_val < 1.503:
                    break

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 200_000,
                   delta_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small mass from the coordinate with the largest
    sub‑gradient to the coordinate with the smallest one, keeping only
    improving moves.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(g.argmax())
        i_min = int(g.argmin())

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta

        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a  # continue from the improved point
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 400_000,
                         sigma: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex – moves that improve the exact
    surrogate are kept.  ``sigma`` controls the typical relative step size.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length ``n``) that minimises
    ``evaluate_sequence``.  The best sequence discovered before the time
    limit is stored globally as ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final absolute evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Assemble a rich and diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Betas
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))  # symmetric

    # asymmetric Betas (random a,b)
    for _ in range(6):
        a = rng.uniform(0.05, 2.5)
        b = rng.uniform(0.05, 2.5)
        seeds.append(_beta_distribution(n, a, b))

    # power‑law shapes
    for exp in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        seeds.append(_powerlaw(n, exp))

    # exponential decays
    for lam in (0.0005, 0.001, 0.002, 0.004, 0.008):
        seeds.append(_exponential_decay(n, lam))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (exact surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(8, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Mirror‑descent optimisation (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Mirror‑descent optimisation (exact max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            slice_end = time.time() + remaining / (top_k - sidx + 1)
            print(f"  Run #{sidx} – time slice ≈ {slice_end - time.time():.1f}s")
            seq, val = _mirror_descent_opt(seed_vec,
                                           deadline=min(slice_end, deadline),
                                           eta0=0.5,
                                           top_k=8)
            print(f"   → surrogate after descent = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (descent)")

            # ----- greedy refinement of the descent result -----
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final stochastic refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=600_000,
                                                sigma=0.012)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local walk)")

    # ------------------------------------------------------------------
    # 4️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
