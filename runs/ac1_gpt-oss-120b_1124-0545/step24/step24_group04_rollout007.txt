def _simulated_annealing(start_seq, n, deadline, rng):
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T0 = cur_val  # some initial temperature maybe
    T = T0
    alpha = 0.9995 (cooling factor per iteration)
    iter = 0
    while time.time() < deadline:
        # pick random move type
        move_type = rng.choice(['transfer', 'swap'])
        if move_type == 'transfer':
            i, j = rng.integers(n, size=2)
            if i == j: continue
            # delta as random fraction of cur[i]
            max_delta = cur[i] * 0.5  # move at most half
            if max_delta < 1e-12:
                continue
            delta = max_delta * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            i, j = rng.integers(n, size=2)
            if i == j: continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        # guarantee non-negatives
        if new[i] < -1e-12: continue
        new_val = _surrogate(new)
        d = new_val - cur_val
        if d < 0 or rng.random() < math.exp(-d / T):
            cur, cur_val = new, new_val
            if cur_val < best_val:
                best_seq, best_val = cur.copy(), cur_val
        # update temperature
        iter += 1

"""
Non‑negative step‑function optimiser (≈ 1000 s).

The routine builds a very diverse seed pool (uniform, arcsine, triangular,
Dirichlet, symmetric / asymmetric Beta, Gaussian mixtures, sparse “spike”
vectors and a geometric‑spacing seed).  The best seed (according to the
smooth surrogate 2·n·max(conv)) is refined through three phases:

1️⃣ Adam optimisation on a smooth‑max surrogate (log‑softmax parametrisation).
2️⃣ Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
3️⃣ Simulated‑annealing / stochastic local search:
   • small mass‑transfer moves,
   • full swaps,
   • targeted transfers that remove mass from the pair of indices that
     contribute most to the current peak of the convolution.
   The schedule accepts uphill moves with probability exp(‑Δ/T) and cools
   the temperature geometrically.  Periodic global permutations help to
   escape symmetry traps.

After the SA phase an extra short EG polish is performed (if time permits)
and the final vector is evaluated with the provided ``evaluate_sequence``.
The best sequence is stored in the global ``height_sequence_1`` variable
so that subsequent runs can reuse it as a seed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    n = len(x)
    L = 2 * n - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(fx * fx, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence must sum to 1 (probability vector).
    Returns 2 * n * max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _asymmetric_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.05, 5.0)
    beta = rng.uniform(0.05, 5.0)
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps (truncated to the interval)."""
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _sparse_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a handful of random positions and give them Dirichlet weights."""
    k = rng.integers(20, 200)          # number of spikes
    positions = rng.choice(n, size=k, replace=False)
    weights = rng.random(k)
    if weights.sum() == 0:
        weights = np.ones(k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=np.float64)
    a[positions] = weights
    return a


def _geometric_seed(n: int) -> np.ndarray:
    """Geometrically spaced positions – a Sidon‑like sparse pattern."""
    pos = []
    p = 0
    while p < n:
        pos.append(p)
        p = int(p * 1.7) + 1
    a = np.zeros(n, dtype=np.float64)
    if len(pos) == 0:
        a[0] = 1.0
    else:
        a[pos] = 1.0 / len(pos)
    return a

def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space using a schedule of smooth‑max
    parameters (beta).  Returns the best probability vector (sum = 1) seen
    and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        inner_iters = 14000 if remaining > 40 else 6000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a * w)        # convolution of rev_a with w
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 800 == 0:
                cur_val = _surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns (best_vector, best_surrogate).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing based stochastic local search
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multi‑scale stochastic walk with temperature‑controlled uphill moves.
    Three move types are mixed:
      – small mass transfer i → j,
      – full swap a[i] ↔ a[j],
      – targeted transfer from a high‑contributing index of the current peak.
    Returns (best_seq, best_surrogate).
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # Initial temperature – use current surrogate as a scale.
    T = cur_val if cur_val > 0 else 1.0
    decay = 0.99993

    it = 0
    conv_cache = None
    peak_idx = None

    while time.time() < deadline:
        it += 1

        # Refresh peak information every few steps (cheap relative to FFT)
        if it % 5 == 0 or conv_cache is None:
            conv_cache = _conv_fft(cur)
            peak_idx = int(np.argmax(conv_cache))

        # Choose move type
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # ---- small mass transfer ----
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_delta = cur[i] * 0.5
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        elif move_type == 1:
            # ---- full swap ----
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        else:
            # ---- targeted transfer ----
            # pick i that contributes most to the current peak
            i_min = max(0, peak_idx - (n - 1))
            i_max = min(peak_idx, n - 1)
            indices = np.arange(i_min, i_max + 1, dtype=int)
            contrib = cur[indices] * cur[peak_idx - indices]
            if contrib.sum() == 0:
                i = rng.integers(i_min, i_max + 1)
            else:
                i = indices[np.argmax(contrib)]

            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

            max_delta = cur[i] * 0.5
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        # Evaluate candidate
        new_val = _surrogate(new)
        delta_val = new_val - cur_val

        # Accept / reject
        accept = False
        if delta_val < 0:
            accept = True
        else:
            prob = math.exp(-delta_val / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_seq = cur.copy()
                best_val = cur_val

        # Temperature decay
        T *= decay

        # Occasionally shake the whole vector with a random permutation
        if it % 2500 == 0 and rng.random() < 0.08:
            perm = rng.permutation(n)
            cur = cur[perm]
            cur_val = _surrogate(cur)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The best sequence is stored in ``height_sequence_1`` for reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Try to reuse a previously saved solution as a seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣ Build a diverse pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta
    for a in (0.05, 0.07, 0.10, 0.15, 0.20, 0.30, 0.40,
              0.50, 0.70, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Asymmetric Beta
    for _ in range(6):
        seeds.append(_asymmetric_beta(n, rng))

    # Gaussian mixtures
    for _ in range(6):
        seeds.append(_gaussian_mixture(n, rng))

    # Sparse and geometric seeds
    for _ in range(4):
        seeds.append(_sparse_seed(n, rng))
    seeds.append(_geometric_seed(n))

    # Random permutations of a few early seeds (break symmetry)
    for base in seeds[:5]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate seeds – keep the best (surrogate objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam smoothing (smooth‑max) – about 40 % of remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.40 * remaining
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣ Exact exponentiated‑gradient polishing – ≈ 15 % of remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + 0.15 * remaining
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 4️⃣ Simulated‑annealing stochastic local refinement – rest of time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing stochastic local refinement ---")
        seq_local, val_local = _simulated_annealing(best_seq, n, deadline, rng)
        print(f"[SA] exact surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_seq, best_val = seq_local, val_local
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 5️⃣ Final short EG polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        polishing_deadline = deadline - 2.0
        print("\n--- Final exponentiated‑gradient polish ---")
        seq_final, val_final = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                                 eta0=2.0)
        print(f"[Final EG] exact surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print("   >>> Updated global best (final EG)")

    # ------------------------------------------------------------------
    # 6️⃣ Evaluate with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
