"""
Simulated‑annealing step‑function optimiser.

The optimiser builds a diverse set of seed probability vectors (they all sum
to 1) and keeps the best one w.r.t. the exact surrogate
    S(a) = 2·n·max(conv(a,a)) .
From that seed we run a fast *incremental* simulated‑annealing local
search.  A single move transfers a small amount δ from position i to
position j; the self‑convolution is updated by a handful of NumPy slice
operations, so hundreds of thousands of proposals can be examined per
second.  The temperature and the proposal scale σ decay exponentially,
allowing large exploratory steps early on and fine‑grained refinement
later.  The best sequence seen is returned and stored in the global
variable ``height_sequence_1`` for possible reuse by a future run.

The algorithm respects the 1000 s budget: a short seed‑evaluation phase
(~seconds), an optional cheap Adam polishing phase, and the main
simulated‑annealing phase which consumes the remaining time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray, n: int) -> float:
    """
    Exact surrogate 2·n·max(conv(seq,seq)).
    ``seq`` must be a non‑negative vector that sums to 1.
    """
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (optional)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp smooth approximation of the max."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0

    # a schedule of “temperature” values for the smooth‑max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective(best_seq, n)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # use more inner steps when we still have time
        max_iter = 12000 if (deadline - time.time()) > 30 else 4000
        for _ in range(max_iter):
            if time.time() > deadline:
                break

            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad = 2.0 * full[n - 1:n - 1 + n] * (2.0 * n)

            # convert to gradient on log‑softmax coordinates
            dot = np.dot(grad, a)
            grad_log = a * (grad - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasionally record exact surrogate
            if it % 800 == 0:
                cur_val = _objective(a, n)
                if cur_val < best_val:
                    best_val, best_seq = cur_val, a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective(final_a, n)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(best_seq, n)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search with incremental convolution updates
# ----------------------------------------------------------------------


def _simulated_annealing_local_search(start_seq: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      rng: np.random.Generator,
                                      sigma_start: float = 0.30,
                                      sigma_end: float = 0.005,
                                      T_start: float = 2e-4,
                                      T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Incremental SA.
    - Randomly transfers a fraction σ of the source mass.
    - Temperature T controls acceptance of worsening moves.
    - Both σ and T decay exponentially from the start to the end of the phase.
    Returns the best sequence and its exact surrogate value.
    """
    a = start_seq.copy()                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                    # current exact convolution
    cur_val = 2.0 * n * np.max(conv)

    best_a = a.copy()
    best_val = cur_val

    phase_start = time.time()
    total_phase = max(deadline - phase_start, 1e-6)

    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1
        # linear decay of σ and T based on elapsed fraction
        elapsed = time.time() - phase_start
        frac = min(elapsed / total_phase, 1.0)
        sigma = sigma_start * (sigma_end / sigma_start) ** frac
        T = T_start * (T_end / T_start) ** frac

        # propose a move
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update (see comments in previous version)
        cand_conv = conv.copy()
        # shift the contribution of the moved mass
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        # adjust the quadratic terms
        cand_conv[2 * i] += delta * delta
        cand_conv[2 * j] += delta * delta
        cand_conv[i + j] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max
        delta_val = cand_val - cur_val

        # Metropolis criterion
        if delta_val < 0 or rng.random() < math.exp(-delta_val / max(T, 1e-15)):
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1

            if cand_val < best_val:
                best_val = cand_val
                best_a = a.copy()

            # periodic full recomputation to fight numeric drift
            if accepted - last_recalc >= 2500:
                conv = _conv_fft(a, a)
                cur_val = 2.0 * n * np.max(conv)
                last_recalc = accepted

        # occasional progress report (lightweight)
        if attempts % 20000 == 0:
            print(f"[SA] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.5f}, T={T:.2e}, best={best_val:.6f}")

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The function respects the supplied time budget, stores the best result in
    the global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we don't overrun the judge
    deadline = start_time + budget_s - 2.0

    # problem size (default 4000, can be overridden)
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # 0️⃣  Possibly load the previous best solution
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            prev_vec = arr / arr.sum()

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds (all normalised to sum 1)
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta families
    beta_alphas = [
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25,
        0.30, 0.35, 0.40, 0.45, 0.50, 0.60,
        0.80, 1.00, 1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 2️⃣  Quick seed evaluation (surrogate)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective(vec, n)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Optional Adam smoothing (fast smooth‑max optimisation)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # devote up to 15 % of the remaining budget to Adam
        adam_deadline = time.time() + min(remaining * 0.15, 30.0)
        print("\n--- Adam smoothing (smooth‑max surrogate) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polishing (optional)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # keep a small slice for polishing
        eg_deadline = time.time() + min(remaining * 0.10, 25.0)
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 5️⃣  Main simulated‑annealing local refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_sa, val_sa = _simulated_annealing_local_search(
            best_seq,
            n,
            deadline,
            rng,
            sigma_start=0.30,
            sigma_end=0.005,
            T_start=2e-4,
            T_end=1e-9,
        )
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # --------------------------------------------------------------
    # 6️⃣  Final official evaluation using the provided evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
