"""
Enhanced step‑function optimiser with simulated‑annealing local refinement.

The optimiser works in several stages:

1. **Seed generation** – a diverse set of probability vectors (uniform,
   arcsine, triangular, spike, Dirichlet draws, discrete Beta, and the
   previously found solution) are created and normalised to sum to 1.

2. **Fast smooth‑max Adam optimisation** – each promising seed is refined
   with Adam on a soft‑max surrogate of the objective.  This quickly
   reaches a good neighbourhood.

3. **Exact exponentiated‑gradient polishing** – a mirror‑descent step
   using the true maximum of the self‑convolution pushes the solution
   further toward a (local) optimum.

4. **Simulated‑annealing incremental local search** – the core improvement.
   A move transfers a small amount δ from an index *i* (chosen with
   probability proportional to its contribution to the current maximum
   convolution) to a random index *j*.  The full convolution is updated
   in O(n) using a closed‑form slice update.  Moves that worsen the
   objective are occasionally accepted according to a temperature schedule,
   allowing the search to escape local minima.  Accepted moves are
   periodically recomputed from scratch to avoid accumulated floating‑point
   error.

5. **Optional multi‑restart** – while time permits, additional random
   seeds are explored with a short annealing run, keeping the best
   overall sequence.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))

    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a schedule of smooth‑max sharpnesses – larger values give a tighter
    # approximation to the true max.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        max_iter_beta = 10000 if (deadline - time.time()) > 30 else 3000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing incremental local search
# ----------------------------------------------------------------------


def _annealing_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.10,
                            init_temp: float = 0.005,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast stochastic walk on the simplex using simulated annealing.
    A transfer δ is moved from a high‑contributing index i (chosen
    proportionally to its contribution to the current maximum of the
    convolution) to a random index j.  The convolution is updated
    incrementally in O(n).  Worsening moves are occasionally accepted
    according to a linearly cooling temperature schedule.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    best_a = a.copy()
    best_val = 2.0 * n * np.max(conv)

    curr_a = a.copy()
    curr_conv = conv.copy()
    curr_val = best_val

    sigma = sigma_start
    accepted = 0
    attempts = 0
    recompute_every = 3000          # recompute full convolution periodically
    start_time = time.time()
    total_allowed = max(deadline - start_time, 1e-9)

    while time.time() < deadline:
        attempts += 1

        # locate the index where the convolution attains its maximum
        k_max = int(np.argmax(curr_conv))

        # build the gradient‑like contribution vector (a_rev) for this k_max
        idx = np.arange(n)
        rev_idx = k_max - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        a_rev = np.zeros_like(a)
        a_rev[mask] = a[rev_idx[mask]]

        # probability for picking i – put a little mass on all entries
        prob_i = a_rev + 1e-12
        prob_i_sum = prob_i.sum()
        prob_i = prob_i / prob_i_sum

        i = int(rng.choice(n, p=prob_i))
        # guard against tiny source mass
        if curr_a[i] < 1e-15:
            positive = np.where(curr_a > 1e-15)[0]
            if positive.size == 0:
                continue
            i = int(rng.choice(positive))

        # random destination index (different from i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # size of the transfer
        delta = sigma * curr_a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update (exact)
        cand_conv = curr_conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # decide acceptance
        if cand_val < curr_val:
            accept = True
        else:
            # linear cooling: temperature proportional to remaining time
            remaining = max(deadline - time.time(), 0.0)
            T = init_temp * (remaining / total_allowed) + 1e-12
            prob = math.exp(-(cand_val - curr_val) / T)
            accept = rng.random() < prob

        if accept:
            # commit the move
            curr_a[i] -= delta
            curr_a[j] += delta
            curr_conv = cand_conv
            curr_val = cand_val
            accepted += 1

            if curr_val < best_val:
                best_val = curr_val
                best_a = curr_a.copy()

            # periodic full recomputation to keep numerical errors bounded
            if accepted % recompute_every == 0:
                curr_conv = _conv_fft(curr_a, curr_a)

        # slowly cool sigma (helps fine‑tuning later on)
        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)

    return best_a, best_val

        adam_deadline = time.time() + 0.40 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0            # keep a few seconds for final phase
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing intensive local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_loc, val_loc = _annealing_local_search(
            best_seq,
            n,
            deadline,
            sigma_start=0.10,
            init_temp=0.005,
            rng=rng,
        )
        print(f"[SA] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Optional multi‑restart if time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:   # keep ~20 s for final evaluation & safety
        # create a fresh random Dirichlet seed and run a short SA
        conc = rng.choice([0.05, 0.1, 0.2, 0.5, 1.0])
        seed_vec = _random_dirichlet(n, conc, rng)
        seed_vec = seed_vec / seed_vec.sum()
        sub_deadline = time.time() + min(30.0, deadline - time.time() - 5.0)
        seq_tmp, val_tmp = _annealing_local_search(
            seed_vec,
            n,
            sub_deadline,
            sigma_start=0.12,
            init_temp=0.008,
            rng=rng,
        )
        print(f"[Restart] surrogate = {val_tmp:.6f}")
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, seq_tmp.copy()
            print("   >>> Updated global best (restart)")

    # ------------------------------------------------------------------
    # 7️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
