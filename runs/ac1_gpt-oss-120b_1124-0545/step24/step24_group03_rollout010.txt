def propose_candidate(seed=123, budget_s=1000, **kwargs):

def _exp_gradient_opt_multi(start_seq, n, deadline, max_iters=1000000, eta0=12.0, tol=1e-10):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # identify active indices near max
        active = np.where(conv >= max_val - tol * max_val)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size > 0:
            g = 2.0 * g / active.size
        # momentum update
        momentum = beta_mom * momentum + (1 - beta_mom) * g
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * momentum

"""
Optimiser for the step‑function bound.

The algorithm proceeds in three phases:

1. **Diverse analytic seeds** – families such as uniform,
   inverse‑sqrt, power‑law, exponential, power × exponential,
   beta‑distribution and random Dirichlet draws are generated on the
   chosen length `n` (≈ 6000).  They are normalised to a simplex and the
   surrogate objective `2·n·max(conv(a,a))` is evaluated quickly (via FFT).
2. **Parameter fine‑tuning** – a cheap differential‑evolution run
   optimises the two parameters `(p,λ)` of the family
   `a_i ∝ i^{‑p}·exp(‑λ·i)`.  The best shape replaces the current best seed.
3. **Non‑convex refinement** – a cascade of mirror‑descent stages
   (exponentiated‑gradient on the exact max‑convolution, a smooth‑max
   L‑BFGS, a variance‑flattening gradient descent and a short simulated‑annealing
   polishing) is applied to the current best sequence.
   All stages respect the shared deadline.

At the end a cheap symmetry‑exploitation step (averaging with the
reversed sequence) is tried; the best sequence (by the surrogate) is
returned and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional scientific imports (SciPy = optional, the code works without)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.optimize import differential_evolution as _de
except Exception:  # pragma: no cover
    _de = None

# Global placeholders used by callbacks (required because nested closures
# are forbidden by the challenge rules)
# ----------------------------------------------------------------------

_lbfgs_beta = None       # current smoothing parameter
_lbfgs_deadline = None   # deadline for L‑BFGS
_de_n = None             # problem size for differential evolution

# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (exact for L1‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# analytic families (all return L1‑normalised vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    """Two‑parameter family a_i ∝ i^{‑p}·exp(‑λ·i) (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of L1‑normalised seed vectors."""
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    # beta shapes
    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # two‑parameter power‑exponential family (coarse sweep)
    p_vals = np.linspace(0.3, 1.2, 10)
    lam_vals = np.linspace(0.0, 0.008, 8)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, p, lam))

    # random Dirichlet draws
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures with uniform to break symmetry
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # incorporate any previously found global best
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    arr = arr / s
                    seeds.append(arr)
        except Exception:  # pragma: no cover
            pass

    # small perturbations of the current best seed
    if seeds:
        best_local = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_local + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # add the reversed versions (often useful)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    return seeds


# ----------------------------------------------------------------------
# Exponentiated‑gradient on exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 1_500_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam‑style momentum
    momentum = np.zeros_like(a)
    beta_m = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices that are “almost active”
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update
        momentum = beta_m * momentum + (1.0 - beta_m) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on Σ_k (conv_k − avg)².
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)  # length of full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom  # = 1/denom for sum=1
        diff = conv - avg

        # gradient of Σ diff² w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (log‑softmax projection)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_objective(v: np.ndarray) -> tuple[float, np.ndarray]:
    """Objective and gradient for smooth‑max L‑BFGS (uses globals)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = conv.max()
    beta = _lbfgs_beta
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smoothed objective
    obj = 2.0 * _lbfgs_n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[_lbfgs_n - 1:_lbfgs_n - 1 + _lbfgs_n]

    # push gradient onto the simplex (log‑softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_callback(v: np.ndarray) -> bool:
    """Terminate L‑BFGS when the global deadline is passed."""
    if time.time() > _lbfgs_deadline:
        raise KeyboardInterrupt()
    return False

            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    global _lbfgs_n, _lbfgs_beta, _lbfgs_deadline
    _lbfgs_n = n
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        _lbfgs_beta = float(beta)
        _lbfgs_deadline = deadline
        try:
            res = _opt.minimize(_lbfgs_objective,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=_lbfgs_callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.001,
                         T_start: float = 6e-3,
                         T_end: float = 4e-9) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian‑perturbation SA with a linear cooling schedule.
    Returns best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run EG → L‑BFGS → flatten → SA on a copy of *start_vec*.
    Returns the best sequence observed and its surrogate value.
    """
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=1_200_000,
        eta0=14.0,
        tol=1e-11,
    )

    # 2) smooth L‑BFGS (if SciPy is available)
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline,
                                            max_iter_per_beta=5000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs, n,
                                                   deadline,
                                                   max_iters=3_000_000,
                                                   eta0=10.0)
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq_flat, n, deadline,
                                              rng,
                                              max_attempts=300_000,
                                              start_sigma=0.0008,
                                              T_start=5e-3,
                                              T_end=2e-9)
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Differential‑evolution objective (uses global _de_n)
# ----------------------------------------------------------------------
def _de_objective(params):
    """Surrogate for the power‑exponential family (global length = _de_n)."""
    p, lam = params
    seq = _power_exp(_de_n, p, lam)
    return _objective_surrogate(seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the allotted wall‑clock budget, stores the best
    found sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # problem size – 6000 yields a good trade‑off between quality and speed
    n = 6000
    print(f"\n=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Generate a rich set of analytic seeds
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(3, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best with the best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑tune the two‑parameter power‑exponential family via DE
    # ------------------------------------------------------------------
    if _de is not None and time.time() < deadline - 15.0:
        global _de_n
        _de_n = n
        bounds = [(0.2, 1.5), (0.0, 0.02)]
        # modest settings – fast but still enough to move away from the coarse grid
        result = _de(_de_objective,
                     bounds,
                     maxiter=30,
                     popsize=10,
                     seed=seed,
                     polish=False,
                     disp=False)
        seq_de = _power_exp(n, result.x[0], result.x[1])
        val_de = _objective_surrogate(seq_de)
        print(f"[DE] surrogate = {val_de:.6f}")
        if val_de < global_best_val:
            global_best_seq = seq_de
            global_best_val = val_de
            print("   *** Updated global best from DE")

    # ------------------------------------------------------------------
    # 5. Deep non‑convex optimisation (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.75
        print(f"\n[Deep optimisation] allocating ≈{remaining * 0.75:.1f}s")
        seq_opt, val_opt = _full_optimisation(global_best_seq, n, sub_deadline, rng)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** Updated global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Small stochastic local search (random perturbations)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        sigma = 0.001
        ns = global_best_seq + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < global_best_val:
            global_best_seq = ns
            global_best_val = val

    # ------------------------------------------------------------------
    # 7. Symmetrisation trick (often reduces max‑convolution)
    # ------------------------------------------------------------------
    sym_seq = (global_best_seq + global_best_seq[::-1]) / 2.0
    sym_seq = sym_seq / sym_seq.sum()
    sym_val = _objective_surrogate(sym_seq)
    if sym_val < global_best_val:
        global_best_seq = sym_seq
        global_best_val = sym_val
        print(f"[Symmetrisation] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
