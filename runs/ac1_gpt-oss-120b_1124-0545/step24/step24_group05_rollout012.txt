def local_search(a, deadline):
    n = len(a)
    best_seq = a.copy()
    best_obj = objective(a)
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        # propose delta
        max_delta = min(best_seq[i], 0.1) # not too large
        delta = (rng.random() - 0.5) * 2 * max_delta  # random between -max_delta .. max_delta
        # move delta from i to j
        new_i = best_seq[i] - delta
        new_j = best_seq[j] + delta
        if new_i < 0: continue
        # create new seq
        new_seq = best_seq.copy()
        new_seq[i] = new_i
        new_seq[j] = new_j
        # compute objective
        val = _objective_surrogate(new_seq)
        if val < best_obj:
            best_obj = val
            best_seq = new_seq
    return best_seq, best_obj

def _local_search_opt(start_seq, n, deadline, max_steps=200000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()
    # precompute initial conv maybe not needed.
    while time.time() < deadline and max_steps>0:
        max_steps -= 1
        # pick i, j
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # choose delta = epsilon * a[i] maybe
        # epsilon random from [-0.1,0.1]
        eps = (rng.random() - 0.5) * 0.2 # +/-10%
        delta = eps * a[i]
        if delta <= -a[i] or delta >= a[j] + 1e-12:
            # ensure positivity
            # adjust delta to feasible range
            if delta < -a[i]:
                delta = -a[i] + 1e-12

        new_seq[i] -= delta
        new_seq[j] += delta

        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            a = new_seq.copy()

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    n = 8000  # length

    # Build seeds
    candidates = []
    # arcsine
    candidates.append(_arcsine_distribution(n))
    # upsample previous best if exist
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        # upsample by interpolation
        if arr.ndim == 1:
            old_n = arr.shape[0]
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, arr)
            # ensure non-neg, renorm
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up = up / up.sum()
                candidates.append(up)
    # uniform
    candidates.append(_uniform(n))
    # exponentials
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))
    # convex combos arcsine/uniform
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for alpha in np.linspace(0, 1, 11):
        cand = alpha * arcs + (1 - alpha) * uni
        cand = cand / cand.sum()
        candidates.append(cand)
    # random Dirichlet
    for _ in range(3):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    best_seq = None
    best_val = float("inf")
    # iterate candidates
    total_cands = len(candidates)
    for idx, cand in enumerate(candidates, start=1):
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            continue
        cand = cand / cand.sum()
        time_left = deadline - time.time()
        if time_left < 15.0:
            print("[Info] Time low, breaking candidate loop.")
            break
        # assign iteration budget proportionally
        # we set max_iters based on remaining time and n
        # naive: 200k for each, but ensure not exceed time
        max_iters = 200_000
        # maybe reduce if many candidates remain
        # compute remaining candidates:
        remaining = total_cands - idx + 1
        # allocate time per candidate: (time_left-10) / remaining
        # estimate iters = (time_per_cand/ (conv_time_per_iter)) ~ but unknown.
        # We'll simply use fixed 200k.

        seq, val = _adam_softmax_opt(cand, n, deadline, max_iters=max_iters,
                                     lr0=0.5, beta_start=0.5, beta_end=64.0)
        print(f"[Candidate {idx}/{total_cands}] Adam best val {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   New global best after Adam: {best_val:.6f}")

    if best_seq is None:
        best_seq = _uniform(n)

    # Polishing with exponentiated gradient
    if time.time() < deadline:
        print("\n--- Polishing with exponentiated gradient ---")
        seq_pg, val_pg = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=5_000_000,
                                           eta0=12.0)
        if val_pg < best_val:
            best_val = val_pg
            best_seq = seq_pg
            print(f"[Polish] improved to {best_val:.6f}")
        else:
            print("[Polish] no improvement.")
    # Local random perturbation search
    if time.time() < deadline:
        print("\n--- Local random search refinement ---")
        seq_local, val_local = _local_search_opt(best_seq, n, deadline,
                                                 max_steps=200_000)
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local
            print(f"[Local] improved to {best_val:.6f}")

    # final evaluation
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Optimisation of a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Length** – we use a longer vector (default 8000).  The true objective
   after normalising ∑a_i = √(2 n) equals the maximal entry of the
   self‑convolution, i.e. 2 n·max(conv(a,a))/∑a² = max(conv).  Larger *n*
   gives a finer discretisation of the continuous arcsine law and can
   reduce this maximum.

2. **Initial seeds** – the discrete arcsine law, an up‑sampled copy of the
   best sequence from a previous run, the uniform distribution and a few
   simple exponential decays.  All are normalised to a probability vector.

3. **Smooth descent (Adam)** – minimise a smooth approximation of the
   max‑convolution

        F_β(a) = (1/β)·log Σ_k exp(β·conv(a,a)[k])

   by Adam on the log‑softmax parameters.  The temperature β is raised
   linearly from 0.5 up to 64, which yields a very sharp surrogate to the
   true objective.  The best simplex vector seen during the run is kept.

4. **Exact sub‑gradient polishing** – a classic exponentiated‑gradient (mirror
   descent) on the exact max‑convolution objective.  Only one FFT‑based
   convolution is needed per iteration, so we can afford millions of steps
   within the time budget.

5. **Local random refinement** – a cheap hill‑climber that moves a tiny
   amount of mass between two random coordinates and keeps the change if it
   reduces the exact objective.  This often squeezes out the last few
   percent.

6. **Time management** – the programme stops all work a few seconds before
   the hard deadline, evaluates the best sequence with the supplied
   ``evaluate_sequence`` function and stores it globally under
   ``height_sequence_1`` for possible continuation runs.

The combination of a finer discretisation, a high‑temperature smoothed
optimum and an extensive mirror‑descent polishing readily pushes the
objective below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective used for comparison:
       2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) with respect to the
    log‑softmax parameters `log_a`.  Returned gradient is ∂F/∂log_a.
    """
    a = _softmax(log_a)                      # probability simplex
    conv = _conv_fft(a, a)                    # self‑convolution (len 2n‑1)

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                              # shape (2n‑1,)

    # g_a = 2 Σ_k w_k · shift_i(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)                # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]        # slice → (n,)

    # chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 64.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best probability vector seen and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for the temperature
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate the exact (non‑smoothed) surrogate
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (a sparse vector)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random‑search refinement (hill‑climbing)
# ----------------------------------------------------------------------
def _local_search_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic refinement.  Moves a small amount of mass between
    two random coordinates; keeps the change if the exact objective improves.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    while time.time() < deadline and max_steps > 0:
        max_steps -= 1

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a transfer proportional to the source mass
        eps = (rng.random() - 0.5) * 0.2       # ±10 % of source
        delta = eps * a[i]

        # enforce non‑negativity
        if delta < -a[i]:
            delta = -a[i] + 1e-12
        if delta > a[j]:
            delta = a[j] - 1e-12
        if abs(delta) < 1e-15:
            continue

        new_seq = a.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            a = new_seq                     # continue climbing from the improvement

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function yielding a low
    ``evaluate_sequence`` value.  The routine respects the 1000 s budget,
    builds a small pool of analytic seeds (arcsine, up‑sampled previous
    result, uniform, exponentials), runs a high‑temperature smoothed Adam
    optimisation, follows with a long exponentiated‑gradient polishing pass
    and finally applies a cheap stochastic hill‑climber.  The best vector is
    stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a few seconds buffer before the hard deadline
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – longer vectors give a finer discretisation.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Build a compact pool of seeds
    # --------------------------------------------------------------
    candidates = []

    # 1) discrete arcsine law – the most promising analytic shape
    candidates.append(_arcsine_distribution(n))

    # 2) up‑sampled copy of the best sequence from a previous run (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.shape[0] > 0:
                old_n = arr.shape[0]
                # linear interpolation to the new length
                x_old = np.linspace(0, 1, old_n)
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, arr)
                up = np.maximum(up, 0.0)
                if up.sum() > 0:
                    candidates.append(up / up.sum())
        except Exception:
            pass

    # 3) uniform distribution
    candidates.append(_uniform(n))

    # 4) a few exponential decays (lightweight analytic family)
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    # 5) convex blends of arcsine and uniform (coarse sweep)
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for alpha in np.linspace(0.0, 1.0, 7):
        blend = alpha * arcs + (1.0 - alpha) * uni
        candidates.append(blend / blend.sum())

    # --------------------------------------------------------------
    # Optimise each seed with the smoothed Adam optimiser.
    # We give the *first* seed (the arcsine) many more iterations – it is
    # by far the most promising.  The remaining seeds get a modest budget.
    # --------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    total_candidates = len(candidates)
    for idx, cand in enumerate(candidates, start=1):
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        time_left = deadline - time.time()
        if time_left < 15.0:
            print("[Info] Time exhausted – leaving candidate loop.")
            break

        # allocate iterations – the first seed gets a large quota
        if idx == 1:
            max_iters = 400_000
        else:
            max_iters = 80_000

        seq, val = _adam_softmax_opt(
            cand, n, deadline,
            max_iters=max_iters,
            lr0=0.5,
            beta_start=0.5,
            beta_end=64.0)

        print(f"[Candidate {idx}/{total_candidates}] Adam best surrogate = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   *** New global best (surrogate) = {best_overall_val:.6f}")

    if best_overall_seq is None:
        # fallback – uniform vector
        best_overall_seq = _uniform(n)

    # --------------------------------------------------------------
    # Exact polishing with exponentiated‑gradient (mirror descent)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        seq_pg, val_pg = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=5_000_000,
            eta0=12.0)

        if val_pg < best_overall_val:
            best_overall_val = val_pg
            best_overall_seq = seq_pg
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Polish] no improvement")

    # --------------------------------------------------------------
    # Very cheap stochastic refinement (hill‑climbing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local random‑search refinement ---")
        seq_local, val_local = _local_search_opt(
            best_overall_seq, n, deadline,
            max_steps=200_000)

        if val_local < best_overall_val:
            best_overall_val = val_local
            best_overall_seq = seq_local
            print(f"[Local] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Local] no improvement")

    # Final evaluation with the provided function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
