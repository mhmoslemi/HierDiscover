"""
Improved search for a step–function minimising ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1️⃣  Build a diverse pool of seed distributions
    – uniform, arcsine, triangular,
    – symmetric Beta(α,α) for a fine grid of α,
    – a few Dirichlet draws,
    – the previously‑found best sequence (if any).

2️⃣  Keep the five seeds with the smallest *exact* surrogate
    ``2·n·max(conv(a,a))`` (the objective for a normalised sum‑1 vector).

3️⃣  For each top seed run a **smoothed exponentiated‑gradient descent**:
    * parametric form a = softmax(z) (mirror descent on the simplex);
    * use a smooth approximation of the max,
      Lτ(a)= (1/τ)·log ∑ exp(τ·conv(a,a)).
      The gradient of Lτ is a weighted average of the sub‑gradients
      of all convolution entries; this yields a much more informative
      direction than the sub‑gradient at a single active index.
    * τ is increased gradually (5 → ≈150) so the optimisation first
      follows a smooth landscape and then focuses on the true max.
    * Adam‑style momentum and a diminishing step size ηt=η0/√t
      are applied in the log‑space.

4️⃣  Greedy refinement – move a tiny fraction of mass from a
    coordinate with the largest sub‑gradient to the one with the
    smallest (only if the move improves the exact surrogate).

5️⃣  Final stochastic walk (local random search) for the remaining seconds.

6️⃣  Evaluate the best vector with the supplied ``evaluate_sequence`` and
    store it globally as ``height_sequence_1``.

The whole procedure fits comfortably into the 1000 s budget, repeatedly
updates the global best, and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate 2·n·max(conv(seq, seq)) for a vector that already
    sums to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives

def _smooth_exp_gradient(start_seq: np.ndarray,
                         deadline: float,
                         eta0: float = 5.0,
                         beta_mom: float = 0.9,
                         tau0: float = 5.0,
                         tau_max: float = 150.0,
                         tau_step: float = 1.5,
                         tau_inc_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated‑gradient descent.
    Returns the best sequence (sum‑1) seen and its exact surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)          # mirror variables
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # ----- exact convolution -----
        conv = _conv_fft(a, a)                 # length 2n‑1
        L = conv.size

        # ----- smoothed max (log‑sum‑exp) -----
        maxc = np.max(conv)
        shifted = conv - maxc                 # ≤0 → prevents overflow
        exp_vals = np.exp(tau * shifted)
        w = exp_vals / exp_vals.sum()         # weights, sum to 1

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        conv_aw = _conv_fft(a, w_rev)         # length n+L‑1
        idx = L - 1 - np.arange(n)            # indices that give the needed slice
        g = 2.0 * conv_aw[idx]                # Euclidean gradient ∂L/∂a

        # ----- mirror‑gradient (log‑space) -----
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # ----- momentum (Adam‑style) -----
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # ----- diminishing step size -----
        eta = eta0 / math.sqrt(it)

        # ----- update -----
        logit = logit - eta * mom
        a = _softmax(logit)

        # ----- periodic exact evaluation -----
        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # early stop when we are already below the target
                if best_val < 1.503:
                    break

        # ----- increase smoothing temperature -----
        if it % tau_inc_every == 0 and tau < tau_max:
            tau = min(tau * tau_step, tau_max)

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small amount of mass from the coordinate with the
    largest sub‑gradient to the one with the smallest (if it improves the
    exact surrogate).
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 250_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex; only improving moves are kept.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val
