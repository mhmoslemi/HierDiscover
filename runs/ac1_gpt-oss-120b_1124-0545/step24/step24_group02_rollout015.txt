class _LBFGSObjective:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
    def __call__(self, v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = conv.max()
        beta_val = self.beta
        exp_beta = np.exp(beta_val * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / beta_val)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

  ∑a_i = 1, which makes the objective 2·n·max(conv(a,a)).
* Stage 1 – “smooth‑max” L‑BFGS:
  minimise a log‑sum‑exp approximation of the max‑convolution with an
  increasing β schedule (continuation method).  The objective and its
  gradient are implemented in a tiny callable class – no closures.
* Stage 2 – exponentiated‑gradient (EG) mirror descent on the exact
  max‑convolution (sub‑gradient of the max).  Adam‑style momentum and a
  1/√t step‑size schedule are used.
* Stage 3 – variance‑flattening: minimise the L₂‑deviation of the full
  convolution from its theoretical mean (1/(2·n‑1)).  The gradient is
  obtained analytically via a single FFT; Adam updates are applied on
  the log‑parameters.
* Several diverse initial seeds (previous best, inverse‑sqrt, random
  Dirichlet, a U‑shaped exponential, a cosine “U” shape and a perturbed
  copy of the previous best) are evaluated quickly; the best few are
  handed to the optimiser.
* The available wall‑clock budget (default 1000 s) is split dynamically
  among the seeds, with a final intensive polishing on the current
  global best.  The best sequence found is stored in the global variable
  ``height_sequence_1`` and returned.

The implementation respects all restrictions (no file I/O, top‑level
helpers only, no lambda closures) and can run with or without SciPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS).  The optimiser falls back to
# the EG method if SciPy is unavailable.
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # extreme case – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective used during optimisation:
    2·n·max(conv(seq, seq))  (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – useful initial seeds
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Extremely sparse random draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _u_shape_exponential(n: int, beta: float) -> np.ndarray:
    """U‑shaped vector a_i = beta**min(i, n‑1‑i)   (0 < beta ≤ 1)."""
    i = np.arange(n, dtype=np.int64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(d * np.log(beta))
    return a


def _cosine_u_shape(n: int) -> np.ndarray:
    """U‑shaped cosine: high at both ends, low in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)
    # shift to be non‑negative (tiny negative due to rounding)
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
class _LBFGSObjective:
    """Callable used by SciPy's L‑BFGS‑B.

    Parameters
    ----------
    n : int
        Length of the vector.
    beta : float
        Smoothing parameter for the log‑sum‑exp approximation of max.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = conv.max()
        b = self.beta
        # stable soft‑max over the convolution values
        exp_beta = np.exp(b * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / b)

        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


def _lbfgs_smooth_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule,
                      max_iter_per_beta: int = 2500):
    """Minimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns the best ℓ₁‑normalised sequence seen and its exact surrogate.
    """
    if _opt is None:                                 # SciPy not present
        return _exp_gradient_opt(start_a, n, deadline)

    v = np.log(start_a + 1e-30)                      # initialise log‑softmax
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        obj_fun = _LBFGSObjective(n=n, beta=beta)

        try:
            res = _opt.minimize(obj_fun,
                                v,
                                method="L-BFGS-B",
                                jac=True,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-14,
                                         "disp": False})
            v = res.x
        except Exception as exc:                     # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {exc}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient (EG) optimiser – works on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 8.0,
                      tol: float = 1e-10):
    """Exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over the active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Variance‑flattening optimiser – tries to make conv(a,a) as constant as
# possible (minimise L₂ deviation from the theoretical mean).
# ----------------------------------------------------------------------
def _variance_flatten_opt(start_a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 5.0,
                         tol: float = 1e-12):
    """Adam‑style mirror descent on the L₂‑distance of conv(a,a) from its
    mean 1/(2·n‑1).  Returns the best sequence and its exact surrogate.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    avg = 1.0 / (2 * n - 1)      # exact mean of the convolution

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        residual = conv - avg                     # deviation from flat target
        rev_a = a[::-1]
        # gradient of ½‖conv‑avg‖² w.r.t. a (see derivation in analysis)
        grad_a = 4.0 * _conv_fft(rev_a, residual)[n - 1:n - 1 + n]

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_a
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the supplied wall‑clock budget and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length offers a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best (if any) to use as a seed.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.shape == (n,):
            previous = prev_arr / prev_arr.sum()
        else:
            previous = None

    # --------------------------------------------------------------
    # Assemble an initial *seed* pool.
    # --------------------------------------------------------------
    seeds = []

    if previous is not None:
        seeds.append(previous)                              # 1) previous best

    seeds.append(_inverse_sqrt(n))                         # 2) classic analytic shape
    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))  # 3) very sparse Dirichlet

    # 4) a few deterministic U‑shapes
    for beta in (0.90, 0.95, 0.98):
        u = _u_shape_exponential(n, beta)
        if u.sum() > 0:
            seeds.append(u / u.sum())

    # 5) cosine‑based U‑shape (high at both ends, low in the middle)
    c = _cosine_u_shape(n)
    if c.sum() > 0:
        seeds.append(c / c.sum())

    # 6) perturbed copy of the previous best (if any)
    if previous is not None:
        pert = previous + 0.003 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s > 0.0:
            seeds.append(pert / s)

    # Normalise all seeds (protect against numerical mishaps)
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation of the seeds (cheap, no time‑budget)
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    # keep the top half (or at least one) for the expensive optimisation
    keep = max(1, len(evaluated) // 2)
    top_seeds = [vec for vec, _ in evaluated[:keep]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:keep], 1):
        print(f" Seed #{i}: surrogate = {v:.8f}")

    # --------------------------------------------------------------
    # β schedule for the smooth‑max continuation.
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Optimise each seed – time is split dynamically.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give each seed at most 20 % of the remaining time (but never less than 5 s)
        share = max(5.0, remaining * 0.20)
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx}/{len(top_seeds)}] optimisation budget ≈ {share:.1f}s")

        # 1) deep L‑BFGS on the smoothed max‑convolution
        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=2500,
        )
        print(f"  → L‑BFGS surrogate = {val_lbfgs:.8f}")

        # 2) EG polishing (helps escape tiny plateaus)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=400_000,
                eta0=6.0,
                tol=1e-12,
            )
            print(f"  → EG polish   surrogate = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # keep the better of the two
        cand_seq, cand_val = (seq_eg, val_eg) if val_eg < val_lbfgs else (seq_lbfgs, val_lbfgs)

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing on the current global best.
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing: EG phase ---")
        seq_eg_final, val_eg_final = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=300_000,
            eta0=4.0,
            tol=1e-12,
        )
        if val_eg_final < global_best_val:
            global_best_val = val_eg_final
            global_best_seq = seq_eg_final.copy()
            print(f"   *** Improved after EG = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Variance‑flattening stage (if any time left).
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final polishing: variance‑flattening ---")
        seq_var, val_var = _variance_flatten_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=500_000,
            eta0=5.0,
        )
        if val_var < global_best_val:
            global_best_val = val_var
            global_best_seq = seq_var.copy()
            print(f"   *** Improved after variance flattening = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function.
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:                                 # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
