"""
Enhanced anti‑concentration optimiser.

The algorithm builds a diverse set of initial probability vectors,
selects the most promising few and refines them by three stages:

1.  Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) – fast
    gradient‑based improvement.
2.  Exact mirror‑descent (exponentiated‑gradient) on the true
    ∞‑norm of the self‑convolution.
3.  Simulated‑annealing local search that moves a small amount of mass
    between two indices, updating the convolution incrementally.
    This final phase can explore uphill moves and therefore escapes
    shallow local minima, often pushing the bound below the previous
    best (≈1.5085).

All phases respect the time budget; the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain
list of non‑negative floats (already normalised to sum = 1, which does
not affect the final evaluation).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))

    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α, α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _mixture_beta_spike(n: int, alpha: float, lam: float) -> np.ndarray:
    """Convex combination ``lam * Beta(α,α) + (1‑lam) * spike``."""
    beta = _beta_distribution(n, alpha)
    spike = _spike(n)
    mix = lam * beta + (1.0 - lam) * spike
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a richer schedule (more gradual increase of β)
    beta_schedule = [
        0.5, 2.0, 10.0, 30.0,
        100.0, 300.0, 800.0,
        2000.0, 5000.0,
        15000.0, 40000.0,
        100000.0, 250000.0,
        600000.0, 1500000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # dynamic inner‑iteration budget
        max_iter = 12000 if (deadline - time.time()) > 30 else 3000

        for _ in range(max_iter):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp with temperature β)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 900 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 3.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Stops when the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing incremental local refinement
# ----------------------------------------------------------------------


def _annealing_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast local walk with Metropolis acceptance:
    – move a random δ from index i to j,
    – update the convolution incrementally,
    – accept if the new value improves or with probability exp(-Δ/T).
    The temperature T slowly drops to zero, allowing occasional uphill
    steps early on and pure descent later.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)

    best_a = a.copy()
    best_val = cur_val

    sigma = sigma_start
    attempts = 0
    accepted = 0

    start_time = time.time()
    total_time = deadline - start_time
    # initial temperature – 1 % of current value (tunable)
    init_temp = max(1e-8, cur_val * 0.01)

    while time.time() < deadline:
        attempts += 1

        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 1e-15:
            continue

        # propose a transfer; magnitude is a fraction of a[i]
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # temperature schedule – linear decay
        elapsed = time.time() - start_time
        T = init_temp * max(0.0, (deadline - time.time()) / total_time)

        # Metropolis rule
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1

            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()

        # cooling of the step‑size σ
        if accepted and accepted % 5000 == 0:
            sigma = max(0.001, sigma * 0.90)
            print(f"[Anneal] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

        # periodic exact recomputation to avoid drift
        if accepted - (accepted // 3000) * 3000 == 0:
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)

    return best_a, best_val

        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"   >>> Updated global best (Anneal) => {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Final official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        # re‑normalise (tiny drift may have occurred)
        best_seq = best_seq / best_seq.sum()
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
