def _alternating_lp_refine(a, n, deadline, max_iters=5):
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(best_a, best_a)
        M = np.max(conv)

        # Build convolution matrix
        # each diagonal offset d in 0..n-1 gets a as entries
        import scipy.sparse as sp
        A = sp.diags([best_a] * n, np.arange(n), shape=(2*n-1, n), format='csc')
        # Solve LP
        from scipy.optimize import linprog
        c = -np.ones(n)  # maximize sum(b)
        ub = np.full(2*n-1, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1 + 1e-8:
            # no improvement possible
            print("[LP] Sum(b) <= 1, stop.")
            break
        # mixing factor schedule
        t = 0.6  # can experiment
        new_a = (1 - t) * best_a + t * (b / sum_b)
        # renormalize
        new_a = new_a / new_a.sum()
        # compute new objective
        new_val = 2.0 * n * np.max(_conv_fft(new_a, new_a))
        print(f"[LP] iter {it}: sum(b)={sum_b:.6f}, new_val={new_val:.6f}")
        if new_val < best_val - 1e-9:
            best_val = new_val
            best_a = new_a
        else:
            # if not improve, we could still accept? reduce t and continue maybe
            # break if stagnating
            print("[LP] No improvement, break.")
            break
    return best_a, best_val

"""
Optimiser for the step‑function “max‑self‑convolution” problem.

The algorithm combines several ideas that complement the previous
baseline:

1.  A diverse pool of seeds (uniform, arcsine, triangular, spike,
    Dirichlet draws, symmetric Beta PDFs, and the previously best
    solution if available).

2.  An *alternating linear‑program* phase:  for the current probability
    vector `a` we solve the LP

        maximise   Σ b_j
        subject to  (a * b)_k ≤ M   for all k,
                    b_j ≥ 0 ,

    where `M = max(a*a)`.  The solution `b` is rescaled to a probability
    vector and mixed with `a`.  Repeating this step strictly reduces
    the objective `2·n·max(conv(a,a))`.

    The convolution matrix `A` (the linear map `b → a*b`) is built once
    per iteration using a sparse Toeplitz representation with
    `scipy.sparse.diags`.  The LP is solved with `scipy.optimize.linprog`
    (HiGHS).

3.  A short exact exponentiated‑gradient polish (sub‑gradient on the
    true max).

4.  An intensive incremental local‑search that repeatedly transfers a
    small amount of mass between two coordinates while updating the
    self‑convolution in O(n) time.  This explores the neighbourhood of
    the LP‑refined point and frequently yields the final improvement.

All phases respect the overall time budget (default 1000 s).  The best
sequence found (as a list of non‑negative heights) is stored in the
global variable ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    `seq` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two endpoints."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on cell centres.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a = a / a.sum()                     # stay on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Incremental local search (fast greedy walk)
# ----------------------------------------------------------------------


def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast greedy walk on the simplex:
    repeatedly move a random amount δ from one coordinate to another.
    The convolution is updated by an inexpensive O(n) slice operation.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    best_a = a.copy()
    best_max = np.max(conv)
    best_val = 2.0 * n * best_max

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # Update convolution incrementally
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a

        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        if cand_val < best_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_max = cand_max
            best_val = cand_val
            accepted += 1

            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)   # recompute to wipe rounding error
                last_recalc = accepted

        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Alternating linear‑program refinement
# ----------------------------------------------------------------------


def _alternating_lp_refine(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 5,
                           mix_factor: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP
        maximise Σ b_j
        s.t.   (a * b)_k ≤ M   for all k,
               b_j ≥ 0
    where M = max(a*a).  The solution `b` is rescaled to a probability
    vector and mixed with `a`.  The process is monotone decreasing on
    the surrogate objective.
    """
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        M = np.max(conv)

        # Build the convolution matrix A (size (2n‑1) × n)
        # Each diagonal (offset d = 0..n‑1) contains the vector a.
        A = sp.diags([a] * n, offsets=np.arange(n), shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=ub,
                                     bounds=(0, None),
                                     method='highs',
                                     options={'disp': False})

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No improvement possible
            print(f"[LP] iteration {it}: sum(b)≈{sum_b:.6f} ≤ 1 → stop")
            break

        # Rescale to a probability vector
        g = b / sum_b

        # Mix (a ← (1‑t) a + t g)  – t is fixed but can be reduced later if desired
        t = mix_factor
        a = (1.0 - t) * a + t * g
        a = a / a.sum()                # re‑normalise to avoid drift

        # Evaluate the surrogate
        new_val = _objective_surrogate(a)
        print(f"[LP] iter {it}: sum(b)={sum_b:.6f}, new surrogate={new_val:.6f}")

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a.copy()
        else:
            # No improvement – early exit
            print("[LP] No improvement this iteration, terminating LP phase.")
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found (as a plain list) and stores it in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size (default 4000, can be overridden)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load previously found solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    beta_alphas = [
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Quick seed evaluation (surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec.astype(np.float64)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Alternating LP refinement (major improvement lever)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Alternating LP refinement ---")
        lp_deadline = deadline - 10.0                # keep a few seconds for later phases
        seq_lp, val_lp = _alternating_lp_refine(best_seq, n, lp_deadline,
                                                max_iters=6, mix_factor=0.6)
        print(f"[LP] best surrogate after LP = {val_lp:.6f}")
        if val_lp < best_val:
            best_val, best_seq = val_lp, seq_lp.copy()
            print("   >>> Updated global best (LP)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        eg_deadline = deadline - 5.0
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Intensive incremental local refinement (fine tuning)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Intensive local refinement (incremental) ---")
        seq_loc, val_loc = _intensive_local_search(best_seq, n,
                                                   deadline,
                                                   sigma_start=0.05,
                                                   rng=rng)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
