def _next_pow2(x):
    return 1 << ((x - 1).bit_length())
def _conv_fft(x, y):
    L = len(x)+len(y)-1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _local_search_sa(start_seq, n, deadline, sigma_start=0.2, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = np.max(conv)
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    best_max = cur_max

    total_time = deadline - time.time()
    T0 = 0.05 * cur_val  # initial temperature
    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_full = 0

    while time.time() < deadline:
        attempts += 1
        # remaining factor for sigma and ternperature
        rem = deadline - time.time()
        frac = rem / total_time if total_time > 0 else 0.0
        temp = T0 * frac  # linear cooling

        # choose source i weighted by a (high mass)
        if a.sum() == 0:
            break
        i = rng.choice(n, p=a)  # source with prob proportional to mass
        # ensure delta >0 maybe
        if a[i] < 1e-12:
            continue
        # choose destination j (could be uniform)
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta: up to sigma*a[i]
        delta = sigma * a[i] * rng.random()
        if delta <= 0:
            continue

        # compute cand_conv
        cand_conv = conv.copy()
        # vectorized updates
        # Note: slice indices might exceed conv length (2*n-1) but i+n <= 2n-1, same for j.
        cand_conv[i:i+n] -= 2.0 * delta * a
        cand_conv[j:j+n] += 2.0 * delta * a

        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        # ensure idx2i, idx2j within conv length
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # acceptance
        accept = False
        if cand_val <= cur_val:
            accept = True
        else:
            # probability
            if temp > 0 and rng.random() < math.exp(-(cand_val - cur_val) / temp):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            cur_max = cand_max
            accepted += 1
            # update best if improved
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                best_max = cur_max

            # periodic full recompute
            if accepted - last_full >= 5000:
                conv = _conv_fft(a, a)
                cur_max = np.max(conv)
                cur_val = 2.0 * n * cur_max
                last_full = accepted

        # sigma cooling 
        # we can gradually reduce sigma based on fraction of time.
        sigma = sigma_start * frac

    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

      2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full linear self‑convolution.
The optimiser consists of three phases:

1. **Diverse seeding** – a collection of probability vectors
   (uniform, arcsine, triangular, spikes, many symmetric Beta and Dirichlet draws).
2. **Smooth‑max Adam optimisation** – fast optimisation of the
   surrogate 2·n·max(conv) using Adam in the log‑softmax space.
3. **Exact sub‑gradient descent (exponentiated‑gradient)** – mirror‑descent on the
   true max‑convolution.
4. **Intensive local refinement with simulated‑annealing** – repeatedly move a
   small amount δ from a source index to a destination index, updating the
   convolution incrementally.  Non‑improving moves are occasionally accepted
   according to a temperature schedule, allowing the search to escape shallow
   plateaux.

All phases respect the time budget (default 1000 s) and the search result is
stored in the global `height_sequence_1` for warm‑starts in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    The result has length len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective (surrogate) for a probability vector:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (β=0.5,0.5)."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on the bin centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Operates in the log‑softmax (mirror) space.
    Returns the best probability vector seen and its exact objective.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a fairly rich schedule for the smoothness parameter β
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
        2000.0, 5000.0, 20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = np.exp(logit) / np.exp(logit).sum()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner steps – longer when we still have ample time
        max_iter_beta = 12000 if (deadline - time.time()) > 20 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = np.exp(logit)
            a /= a.sum()                     # current probability vector

            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp trick)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length = n + (2n‑1) – 1
            grad = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad *= 2.0 * n                     # outer factor 2·n

            # map to gradient on log‑softmax variables
            dot = np.dot(grad, a)
            grad_log = a * (grad - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 600 == 0:
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = np.exp(logit)
    final_a /= final_a.sum()
    final_val = _objective(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Keeps the best vector encountered.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Intensive local refinement with simulated annealing
# ----------------------------------------------------------------------


def _local_search_sa(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     sigma_start: float = 0.25,
                     rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast greedy walk that also accepts occasional uphill moves.
    The move transfers a small amount δ from a source index i to a destination j.
    Convolution is updated incrementally – O(n) work per trial.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = np.max(conv)
    cur_val = 2.0 * n * cur_max

    best_a = a.copy()
    best_val = cur_val

    total_time = max(deadline - time.time(), 1e-9)
    T0 = 0.02 * cur_val          # initial temperature (≈ 2 % of current objective)
    accepted = 0
    attempts = 0
    last_full = 0

    while time.time() < deadline:
        attempts += 1

        # linear cooling for temperature and sigma
        remaining = max(deadline - time.time(), 0.0)
        frac = remaining / total_time
        temp = T0 * frac
        sigma = sigma_start * frac

        # pick source proportional to its current mass (larger a → larger effect)
        if a.sum() == 0.0:
            break
        i = rng.choice(n, p=a)
        if a[i] <= 1e-15:
            continue

        # uniformly pick destination different from source
        j = rng.integers(0, n)
        if j == i:
            continue

        # how much to move
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the self‑convolution
        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cand_conv[idx2i] += delta * delta
        cand_conv[idx2j] += delta * delta
        cand_conv[idxij] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        # acceptance test (simulated annealing)
        accept = False
        if cand_val <= cur_val:
            accept = True
        else:
            if temp > 0 and rng.random() < math.exp(-(cand_val - cur_val) / temp):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # Numerical safety – keep non‑negative
            if a[i] < 0.0:
                a[i] = 0.0
            conv = cand_conv
            cur_val = cand_val
            cur_max = cand_max
            accepted += 1

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            # periodic full recompute to avoid drift
            if accepted - last_full >= 3000:
                conv = _conv_fft(a, a)
                cur_max = np.max(conv)
                cur_val = 2.0 * n * cur_max
                last_full = accepted

        # occasional progress report
        if attempts % 50000 == 0:
            print(f"[Local SA] attempts={attempts}, accepted={accepted}, "
                  f"temp={temp:.6f}, best_val={best_val:.6f}")

    # final clean‑up (full recompute)
    conv = _conv_fft(best_a, best_a)
    best_val = 2.0 * n * np.max(conv)
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found is stored in the global variable ``height_sequence_1``
    and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin of a couple of seconds for the final evaluation
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from the previous run (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a rich pool of seeds (all sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – a broad α‑grid
    beta_alphas = [
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00, 3.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Seed evaluation – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):

        vec = vec / vec.sum()
        val = _objective(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:

            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing (smooth‑max surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        # allocate roughly 30 % of the remaining time

        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        best_seq, best_val = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
        print(f"[EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Intensive local refinement with simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Intensive local refinement (simulated annealing) ---")
        best_seq, best_val = _local_search_sa(best_seq, n, deadline,

    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for warm‑start in the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
