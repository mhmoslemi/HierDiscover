"""
Improved step‑function optimizer.

The search builds a very rich collection of analytic “seed’’ shapes
(including dense sweeps of exponential‑power families, mixed exponentials,
geometric decays and power‑law shapes), selects the few best by the exact
surrogate  2·n·max(conv)  and then refines them with three stages:

1. Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
   A sub‑gradient is formed from *all* active indices (within a tiny
   tolerance) and Adam‑style momentum is used.  The optimisation is run
   for a large number of iterations while respecting the wall‑clock
   deadline.

2. Smooth‑max L‑BFGS (log‑sum‑exp surrogate) with a dense β‑schedule.
   This gives a high‑precision refinement of the max‑convolution peak.

3. A final simulated‑annealing polish that perturbs the logits with a
   Gaussian kernel while cooling the temperature linearly.

After the three stages a cheap “exponent‑tuning’’ step scans
γ∈[0.8,1.2] and keeps the best power‑transformed normalised sequence.
The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.

All stages continuously check the global deadline, therefore the
routine always produces a valid answer before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional scientific imports (SciPy = optional, the code works without)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact after L1‑normalisation)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for L1‑normalised seq."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, q: float) -> np.ndarray:
    """Geometric decay a_i ∝ q^i (0<q<1)."""
    i = np.arange(n, dtype=np.float64)
    a = q ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient on exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """Mirror‑descent with momentum on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # build sub‑gradient from all active indices
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev = kk - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the smooth max surrogate 2·n·log‑sum‑exp(β·conv)/β."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (Gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 600_000,
                         start_sigma: float = 0.0009,
                         T_start: float = 7e-3,
                         T_end: float = 4e-9) -> tuple[np.ndarray, float]:
    """Gaussian SA with linear cooling."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Power/Exponential sweep for seed creation

def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of diverse L1‑normalised candidate vectors."""
    seeds = []

    # simple analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    # power‑law sweep
    for p in np.linspace(0.1, 2.0, 25):
        seeds.append(_power(n, float(p)))

    # pure exponential decay (rates spanning many decades)
    for rate in np.logspace(-6, -0.1, 30):
        seeds.append(_exp_decay(n, float(rate)))

    # mixed exponentials (dense grid)
    rates = np.logspace(-6, -0.1, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            for alpha in [0.2, 0.5, 0.8]:
                seeds.append(_mixed_exponential(n, float(r1), float(r2), float(alpha)))

    # power‑exponential two‑parameter family (dense)
    p_vals = np.linspace(0.1, 1.5, 20)
    lam_vals = np.linspace(0.0, 0.05, 12)
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp(n, float(p), float(lam)))

    # geometric decay (q close to 1 is interesting)
    for q in np.linspace(0.9995, 0.90, 16):
        seeds.append(_geom_decay(n, float(q)))

    # beta‑distribution shapes
    for a in [0.5, 0.8, 1.2, 2.0]:
        for b in [0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # random Dirichlet draws
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # convex mixtures with uniform to break symmetry
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-4), _power_exp(n, 0.8, 0.01)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, float(alpha)))

    # reversed versions (feel free to explore asymmetry)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # incorporate previously known best if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:  # pragma: no cover
            pass

    # small perturbations of the current global best (if any)
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(5):
            pert = best_seed + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds

def _gamma_tuning(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  gammas=None) -> np.ndarray:
    """
    Scan a few power‑transform exponents γ and keep the best.
    This cheap scan often yields a tiny improvement after SA.
    """
    if gammas is None:
        gammas = np.linspace(0.80, 1.20, 41)

    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)

    for g in gammas:
        if time.time() > deadline:
            break
        transformed = np.power(seq, g)
        transformed = transformed / transformed.sum()
        val = _objective_surrogate(transformed)
        if val < best_val:
            best_val = val
            best_seq = transformed.copy()

    return best_seq

        seq_fin = _gamma_tuning(seq_fin, n, final_deadline, rng)

        val_fin = _objective_surrogate(seq_fin)
        if val_fin < global_best_val:
            global_best_seq = seq_fin
            global_best_val = val_fin
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
