def _beta_distribution(n, alpha):
    i = np.arange(n) + 0.5
    x = i / n
    # to avoid log(0), clip
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    log_a = (alpha - 1) * (np.log(x) + np.log(1 - x))
    a = np.exp(log_a)
    # if overflow (log_a > 700), set to np.exp(700) but else ok.
    # Actually np.exp can overflow to inf; we can replace inf with max finite.
    a[np.isinf(a)] = np.finfo(np.float64).max
    return a / a.sum()

budget and stores the best sequence in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, guarantees a sum of 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective used during optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seeds
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) on {i+½\over n}, i=0,…,n‑1.
    For α=0.5 this coincides with the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    # protect against overflow/underflow
    log_a = np.clip(log_a, -700, 700)
    a = np.exp(log_a)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax params.
    Returned shape is (n,).
    """
    a = _softmax(log_a)                # probability vector (∑ a = 1)
    conv = _conv_fft(a, a)              # length 2n‑1

    # soft‑max weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # g_a = 2 Σ_k w_k·shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)          # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]  # centre slice → length n

    # chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best probability vector encountered and its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)   # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) polish on the true max‑conv objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the k‑th convolution entry)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (probability vector) with a low evaluate_sequence
    value.  Combines analytic seeds, long Adam optimisation (temperature up to 30),
    an extensive exponentiated‑gradient polish and a cheap stochastic hill‑climbing
    refinement.  The best sequence found is stored in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    n = 4000  # length of the step function

    # ------------------------------------------------------------------
    # Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1) classic arcsine
    seeds.append(_arcsine_distribution(n))

    # 2) beta‑α family (α ranging from 0.15 to 0.95)
    for alpha in (0.15, 0.20, 0.25, 0.30, 0.35,
                  0.40, 0.45, 0.50, 0.55, 0.60,
                  0.65, 0.70, 0.80, 0.90):
        seeds.append(_beta_distribution(n, alpha))

    # 3) uniform
    seeds.append(_uniform(n))

    # 4) a few exponential decays
    for rate in (0.0003, 0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exp_decay(n, rate))

    # 5) random Dirichlet draws (concentration 0.5 gives moderately sparse vectors)
    for _ in range(5):
        seeds.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # 6) simple convex combos of arcsine and uniform (coarse)
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for mix in np.linspace(0.0, 1.0, 6):
        combo = mix * arcs + (1.0 - mix) * uni
        seeds.append(combo / combo.sum())

    # 7) previous best, if it exists and matches the current length
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(np.maximum(arr, 0.0) / np.maximum(arr.sum(), 1e-30))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Optimise each seed with a relatively long Adam run.
    # The temperature is pushed up to β = 30 to better approximate the true max.
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")
    total_seeds = len(seeds)

    for idx, cand in enumerate(seeds, start=1):
        if time.time() > deadline - 10.0:
            print("[Info] Time about to run out – stopping seed loop.")
            break

        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        # allocate a modest but non‑trivial number of iterations
        time_left = deadline - time.time()
        seeds_remaining = total_seeds - idx + 1
        # aim for at least 40 k iterations per seed, but respect time budget
        max_iters = int(min(300_000,
                            max(40_000,
                                (time_left - 5.0) / max(1, seeds_remaining) * 120)))
        max_iters = max(max_iters, 20_000)

        print(f"[Candidate {idx}/{total_seeds}] max_iters={max_iters}")

        seq, val = _adam_softmax_opt(cand, n, deadline,
                                     max_iters=max_iters,
                                     lr0=0.4,
                                     beta_start=0.5,
                                     beta_end=30.0)

        print(f"   surrogate after Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   *** new global best (surrogate) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _uniform(n)
        best_overall_val = _objective_surrogate(best_overall_seq)

    # ------------------------------------------------------------------
    # Polishing with a high‑iteration exponentiated‑gradient pass
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=5_000_000,
            eta0=24.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best unchanged)")

    # ------------------------------------------------------------------
    # Tiny stochastic hill‑climbing – move a small fraction of mass between
    # two random positions if it improves the exact objective.
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Local hill‑climbing refinement ---")
        seq = best_overall_seq.copy()
        val = best_overall_val
        step_factor = 0.02  # start with up‑to‑2 % of a[i] moved
        it = 0
        while time.time() < deadline - 0.5:
            it += 1
            if it % 2000 == 0:
                step_factor *= 0.95
                step_factor = max(step_factor, 1e-5)

            i = rng.integers(0, n)
            if seq[i] < 1e-12:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue

            delta = seq[i] * step_factor * rng.random()
            if delta <= 0.0:
                continue

            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            cur_val = _objective_surrogate(new_seq)
            if cur_val < val:
                seq = new_seq
                val = cur_val
                if it % 5000 == 0:
                    print(f"[HillClimb] iter {it}, new surrogate {val:.6f}")

        best_overall_seq = seq
        best_overall_val = val

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
