def _simulated_annealing(start_seq, n, deadline, init_temp, rng):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    best_a = a.copy()
    best_val = 2*n*conv.max()
    cur_val = best_val
    T = init_temp
    cooling = 0.995  # after each iteration maybe factor.
    attempts=0
    accepted=0
    while time.time() < deadline:
        attempts+=1
        i,j = rng.integers(0,n,2)
        if i==j or a[i] <= 1e-12: continue
        # propose delta as random fraction 0- sigma * a[i] where sigma maybe 0.2.
        delta = a[i]*rng.random()*0.2
        # update conv as before.
        cand_conv=conv.copy()
        cand_conv[i:i+n] -= 2*delta*a
        cand_conv[j:j+n] += 2*delta*a
        idx2i = 2*i
        idx2j = 2*j
        idxij = i+j
        cand_conv[idx2i] += delta*delta
        cand_conv[idx2j] += delta*delta
        cand_conv[idxij] -= 2*delta*delta
        cand_val = 2.0*n*cand_conv.max()
        dE = cand_val - cur_val
        if dE < 0 or rng.random() < math.exp(-dE/T):
            # accept move
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1
            if cand_val < best_val:
                best_val = cand_val
                best_a = a.copy()
        # cooling schedule per iteration maybe update T=init_temp* (1 - progress)
        # We can update after fixed steps, e.g., every 1000 attempts.
        if attempts % 2000 == 0:
            # compute progress fraction (time)
            elapsed = time.time() - start_time
            fraction = elapsed / (deadline - start_time)
            T = init_temp * (1 - fraction) # linear cooling to zero at the end
    return best_a, best_val

"""
Enhanced step‑function optimiser.

The optimiser builds on the previous three‑phase approach
(Adam → exponentiated‑gradient → greedy local walk) and adds two
new ingredients that give a noticeable improvement on the 4000‑length
problem:

1. **A richer seed set** – besides the uniform, arcsine, triangular and
   symmetric Beta families we also try *asymmetric* Beta(α,β) distributions
   on a dense grid of parameters.  These give many non‑symmetric
   candidates that can start the optimisation in a better basin.

2. **Simulated‑annealing refinement** – after the exact exponentiated‑gradient
   polishing we run a Metropolis‑style random walk that occasionally
   accepts uphill moves.  The walk uses the fast incremental convolution
   update (the same O(n) update as the greedy local search) and a
   linearly‑cooled temperature schedule.  This helps escape shallow local
   minima that the deterministic walk cannot leave.

The overall schedule (time limited by ``budget_s``) is:

   * seed evaluation (≈10 % of the budget)
   * Adam smoothing on the best seed (≈20 %)
   * exact exponentiated‑gradient polishing (≈20 %)
   * simulated‑annealing (≈30 %)
   * a short final greedy local refinement (≈10 %)
   * final call to ``evaluate_sequence``

All steps keep the vector on the probability simplex (sum = 1) and
maintain non‑negativity, so the returned sequence can be fed directly
to ``evaluate_sequence``.  The best sequence is stored in the global
variable ``height_sequence_1`` for possible reuse by subsequent runs.
"""

import time
import math
import itertools
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))

    ``seq`` must be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n bins."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _spike(n: int) -> np.ndarray:
    """All mass on the two ends (half on each side)."""
    a = np.zeros(n, dtype=np.float64)
    a[0] = a[-1] = 0.5
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # log‑space to avoid under/overflow
    log_a = (alpha - 1.0) * np.log(x) + (alpha - 1.0) * np.log(1.0 - x)
    log_a -= np.max(log_a)          # shift for numerical stability
    a = np.exp(log_a)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # log‑space to keep the numbers stable
    log_a = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a hand‑crafted schedule of “β” values for the smooth‑max;
    # larger β makes the surrogate tighter.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # adapt the inner iteration budget to the remaining time
        max_iter_beta = 10000 if (deadline - time.time()) > 30 else 3000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Intensive local search using an incremental convolution update
# ----------------------------------------------------------------------


def _intensive_local_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            sigma_start: float = 0.05,
                            rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast greedy walk on the simplex:
    repeatedly move a random amount δ from one coordinate to another.
    The convolution is updated by an inexpensive O(n) slice operation,
    allowing hundreds of thousands of trials.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # exact current convolution
    best_a = a.copy()
    best_max = np.max(conv)
    best_val = 2.0 * n * best_max

    sigma = sigma_start
    accepted = 0
    attempts = 0
    last_recalc = 0

    while time.time() < deadline:
        attempts += 1

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        # linear part of the update
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        # quadratic correction
        cand_conv[2 * i] += delta * delta
        cand_conv[2 * j] += delta * delta
        cand_conv[i + j] -= 2.0 * delta * delta

        cand_max = cand_conv.max()
        cand_val = 2.0 * n * cand_max

        if cand_val < best_val - 1e-12:
            # accept
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            best_a = a.copy()
            best_max = cand_max
            best_val = cand_val
            accepted += 1

            if accepted - last_recalc >= 2000:
                conv = _conv_fft(a, a)
                last_recalc = accepted

        # slow cooling of sigma
        if accepted and accepted % 5000 == 0:
            sigma = max(0.005, sigma * 0.90)
            print(f"[Local] attempts={attempts}, accepted={accepted}, "
                  f"sigma={sigma:.4f}, best_val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement (uphill moves allowed)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style random walk that can accept uphill moves.
    Uses the same O(n) incremental convolution update as the greedy walk.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)

    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    # linear cooling schedule: temperature goes to ~0 at the deadline
    start_time = time.time()
    attempts = 0
    accepted = 0
    sigma = 0.03  # typical move size (fraction of source coordinate)

    while time.time() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        cand_conv = conv.copy()
        cand_conv[i:i + n] -= 2.0 * delta * a
        cand_conv[j:j + n] += 2.0 * delta * a
        cand_conv[2 * i] += delta * delta
        cand_conv[2 * j] += delta * delta
        cand_conv[i + j] -= 2.0 * delta * delta

        cand_val = 2.0 * n * cand_conv.max()
        dE = cand_val - cur_val

        # update temperature (linear cooling)
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = max(1e-12, init_temp * (1.0 - elapsed / total))

        accept = False
        if dE < 0:
            accept = True
        else:
            if rng.random() < math.exp(-dE / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = cand_conv
            cur_val = cand_val
            accepted += 1
            if cand_val < best_val:
                best_val = cand_val
                best_a = a.copy()

        # occasional full recomputation to avoid drift
        if accepted % 3000 == 0:
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)

        # adapt sigma slowly (optional)
        if attempts % 5000 == 0:
            sigma = max(0.005, sigma * 0.95)

    return best_a, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The best sequence found is stored in the global variable
    ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a tiny safety margin for the final call to ``evaluate_sequence``.
    deadline = start_time + budget_s - 1.5

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load any previously found solution
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _spike(n),
    ]

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta family (already used previously)
    beta_alphas = [
        0.02, 0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # **New**: asymmetric Beta(α,β) grid – this gives many non‑symmetric candidates.
    asym_params = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for alpha, beta in itertools.product(asym_params, asym_params):
        # skip the symmetric case that we already have (optional)
        seeds.append(_beta_asymmetric(n, alpha, beta))

    # Include the previous best solution (if any) as a seed
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of seeds – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # Ensure non‑negativity and normalise
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:

        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:04d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------

    if time.time() < deadline:

        remaining = deadline - time.time()

        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:

            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0  # keep a few seconds for later phases

                                           eta0=8.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:

            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------

    if time.time() < deadline:

        # Temperature is set as a small fraction of the current best value.

        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              init_temp=init_temp,
                                              rng=rng)
        print(f"[SA] surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Final short greedy local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:

        # Use whatever time is left (a few seconds are enough for polishing)

        seq_loc, val_loc = _intensive_local_search(best_seq, n,
                                                   final_deadline,
                                                   sigma_start=0.02,
                                                   rng=rng)
        print(f"[Final Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (final local)")

    # ------------------------------------------------------------------
    # 7️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
