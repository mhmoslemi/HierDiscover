Propose a step function that minimises ``evaluate_sequence``.
--------------------------------------------------------------------
Algorithm
--------------------------------------------------------------------
1. **Diverse seeds** – generate a large pool of probability vectors:
   uniform, arcsine‑like, triangular, many symmetric Beta(α,α) shapes,
   Dirichlet draws with several concentrations and (if present) the
   previously best solution.

2. **Cheap screening** – evaluate the surrogate
   ``2·n·max(conv(p,p))`` for every seed (this is the objective when the
   vector is normalised to sum = 1).  Keep the K best seeds (K≈5).

3. **Smoothed exponentiated‑gradient optimisation** – for each kept
   seed run a mirror‑descent on a *smooth* approximation of the
   max‑convolution:
        •  λ(t) grows from λ₀≈5 to λ₁≈2000; the smooth max is
           `S_λ(p)= (1/λ)·log Σ exp(λ·(p*p))`.
        •  The gradient of S_λ is  2·corr( w , p )  where
           `w = softmax(λ·(p*p))` and `corr` is the linear cross‑correlation
           (valid mode).  We compute the correlation with the FFT for speed.
        •  Mirror‑descent in the KL‑geometry (exponentiated gradient):
           `logit ← logit – η·momentum·grad_log`,  `p = softmax(logit)`.
        •  Step‑size η decays as η₀/√t, momentum uses Adam‑style
           accumulation (β≈0.9).  The exact surrogate is evaluated

   with the largest sub‑gradient to the one with the smallest (only if

# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate (sum(p)=1): 2·n·max(conv(p,p))."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Smoothed exponentiated‑gradient on the *soft* max‑convolution

def _exp_grad_smooth(start_seq: np.ndarray,
                     start_time: float,
                     end_time: float,
                     eta0: float = 5.0,
                     beta_mom: float = 0.9,
                     lam_start: float = 5.0,
                     lam_end: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective

        S_λ(p) = (1/λ)·log Σ exp(λ·(p*p))

    The gradient of S_λ is 2·corr(w, p) where w = softmax(λ·(p*p)).
    Parameters
    ----------
    start_seq : np.ndarray
        Probability vector (sum = 1) to start from.
    start_time, end_time : float
        Wall‑clock times defining the allowed time window.
    eta0, beta_mom : float
        Adam‑style step‑size and momentum parameters.
    lam_start, lam_end : float
        Schedule for the smoothing parameter λ (grows linearly with time).
    Returns
    -------
    best_seq, best_val
        The best sequence seen (as a probability vector) and its exact
        surrogate value (2·n·max(conv)).
    """
    n = len(start_seq)
    a = start_seq.copy()
    # log‑probabilities (the mirror variables)
    logit = np.log(a + 1e-30)
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    # pre‑alloc buffers for FFT lengths
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len + n - 1)  # for correlation

    while time.time() < end_time:
        it += 1

        # ---------- smoothing schedule ----------
        elapsed = time.time() - start_time
        total = end_time - start_time
        lam = lam_start + (lam_end - lam_start) * (elapsed / total)

        # ---------- compute convolution ----------
        conv = _conv_fft(a, a)                     # length 2n‑1
        # ---------- softmax weights ----------
        # use stable softmax: exp(lam*(conv - max(conv)))
        max_c = np.max(conv)
        exp_scaled = np.exp(lam * (conv - max_c))
        w = exp_scaled / exp_scaled.sum()          # still sums to 1

        # ---------- gradient of the smoothed objective ----------
        # grad = 2 * corr(w, a)  (valid mode → length n)
        # compute correlation via FFT for speed
        # corr(w, a) = (w * a_rev)[n-1 : n-1+n]   where a_rev = a[::-1]
        a_rev = a[::-1]
        # FFT of both signals
        fw = np.fft.rfft(w, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)
        # extract the valid part (size n)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ---------- mirror‑gradient (KL) ----------
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)

        # ---------- momentum ----------
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # ---------- step‑size ----------
        eta = eta0 / math.sqrt(it)

        # ---------- update in log‑space ----------
        logit = logit - eta * mom
        a = _softmax(logit)          # stays on the simplex

        # ---------- occasional exact evaluation ----------
        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth EG] iter {it:6d}, λ={lam:.1f}, surrogate={cur_val:.6f}")

        # early exit if we already beat a strong target (≈1.503)
        if best_val < 1.503:
            break

    # final exact check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val

def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
    return best_seq, best_val

def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    the supplied ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and stores the best vector globally as
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0️⃣  Load previous solution (if any) as a possible seed
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Assemble a rich seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Betas
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate seeds cheaply (surrogate) and keep the best few
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed exponentiated‑gradient (major optimisation phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate 60 % of the remaining budget to the smoothed phase
        smooth_budget = remaining * 0.60
        per_seed_deadline = time.time() + smooth_budget / top_k
        print("\n--- Smoothed EG optimisation (soft max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            # ensure we do not exceed the global deadline
            local_end = min(per_seed_deadline, deadline)
            print(f"  Smooth EG run #{sidx} (time slice ≈ {local_end - time.time():.1f}s)")
            seq, val = _exp_grad_smooth(seed_vec,
                                        start_time=time.time(),
                                        end_time=local_end,
                                        eta0=5.0,
                                        beta_mom=0.9,
                                        lam_start=5.0,
                                        lam_end=2000.0)
            print(f"   → surrogate after smooth EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (smooth EG)")

            # reduce the per‑seed window for the next seed
            per_seed_deadline = time.time() + smooth_budget / max(1, top_k - sidx)

    # ------------------------------------------------------------------
    # 4️⃣  Exact max‑gradient polishing (short run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polishing_deadline = time.time() + (deadline - time.time()) * 0.20
        print("\n--- Exact max‑gradient polishing ---")
        seq, val = _exp_gradient_momentum(best_seq.copy(),
                                          polishing_deadline,
                                          eta0=4.0,
                                          beta_mom=0.9)
        print(f"   → surrogate after polishing = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("   >>> Updated global best (polish)")

    # ------------------------------------------------------------------
    # 5️⃣  Greedy refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        greedy_deadline = time.time() + (deadline - time.time()) * 0.10
        print("\n--- Greedy refinement ---")
        seq, val = _greedy_refine(best_seq.copy(), greedy_deadline)
        print(f"   → surrogate after greedy = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("   >>> Updated global best (greedy)")

    # ------------------------------------------------------------------
    # 6️⃣  Final tiny stochastic walk
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final local random walk ---")
        seq_loc, val_loc = _local_random_search(best_seq.copy(),
                                                deadline,
                                                attempts=250_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in a later call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
