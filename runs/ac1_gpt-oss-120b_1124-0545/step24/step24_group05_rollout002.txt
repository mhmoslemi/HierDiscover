def proj_simplex(v):
    # ensure sum=1 nonnegative
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and (v >= 0).all():
        return v
    # similar to https://github.com/chantalcircus/fast-projection-simplex
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence(seq).

Algorithm
---------
1. **Initial seed** – start from the best known sequence (global variable
   ``height_sequence_1``) if it exists, otherwise fall back to the
   discrete arcsine law (the analytic shape that already gives a very
   low value).

2. **Projected sub‑gradient descent** – the true objective is
   ``max(conv(a,a))`` (the max entry of the self‑convolution).  Its
   sub‑gradient at the active index *k* is
   ``g_i = 2·a_{k−i}`` (zero when the index is out of range).  We
   perform an Euclidean projected sub‑gradient method with momentum
   (β≈0.9) and a step size η_t = η₀/√t.  After each step the iterate is
   projected onto the probability simplex (non‑negative, sum = 1).

3. **Stochastic hill‑climbing** – when the sub‑gradient loop finishes,
   a light‑weight simulated‑annealing style perturbation phase
   transfers a tiny amount of mass from a random source index to a
   random target index.  Moves that improve the surrogate objective are
   kept; otherwise they are accepted with a temperature‑dependent
   probability.  This often finds tiny gains that the deterministic
   descent misses.

4. **Final evaluation** – the best vector found is evaluated with the
   provided ``evaluate_sequence`` function, stored in the global
   variable ``height_sequence_1`` and returned.

The whole routine respects the 1000 s budget, checking the deadline
regularly and stopping as soon as time runs out.  The code is fully
self‑contained, uses only NumPy (and optionally SciPy for additional
seeds) and runs on a single core (the 2‑CPU limit is respected by the
single‑threaded FFT implementation).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------
def _proj_subgradient_opt(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 1.0,
    beta_mom: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the exact max‑convolution
    objective.  Returns the best sequence seen and its surrogate value.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    # momentum buffer
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # current convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # momentum update
        v = beta_mom * v + (1.0 - beta_mom) * g

        # step size schedule η_t = η₀ / √t
        step = eta0 / math.sqrt(t)

        # gradient step + simplex projection
        a = _project_simplex(a - step * v)

        # evaluate surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional logging
            if t % 5000 == 0:
                print(f"[SubGrad] iter {t:,} – new best surrogate {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (simulated annealing style)
# ----------------------------------------------------------------------
def _hill_climb_opt(
    a0: np.ndarray,
    deadline: float,
    max_trials: int = 200_000,
    epsilon_initial: float = 0.02,
    temperature_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass between two coordinates.
    Accept improvements; otherwise accept with a probability that
    decays with the temperature schedule.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    # pre‑compute a list of index pairs for speed
    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline and t < max_trials:
        t += 1
        # linear cooling
        frac = t / max_trials
        eps = epsilon_initial * (1.0 - frac)  # shrink epsilon
        temperature = temperature_initial * (1.0 - frac)

        # pick a source with enough mass
        src = rng.integers(0, n)
        if a[src] < eps:
            # try again a few times before giving up this trial
            for _ in range(3):
                src = rng.integers(0, n)
                if a[src] >= eps:
                    break
        if a[src] < eps:
            continue

        # pick a distinct target
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        # propose new vector
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        # no need to re‑project – sum is preserved and non‑negativity holds

        cand_val = _objective_surrogate(cand)

        accept = False
        if cand_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_val - best_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                if t % 5000 == 0:
                    print(f"[HillClimb] trial {t:,} – new best surrogate {best_val:.6f}")

    return best_a, best_val

def propose_candidate(

    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative probability vector that minimises the
    supplied ``evaluate_sequence`` function.  The routine combines a
    projected sub‑gradient optimiser with a stochastic hill‑climbing
    refinement and respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin before the hard deadline
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Length of the step function – a moderate size that provides a good
    # trade‑off between quality and computational cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from the best known construction if available, else use
    # the analytic arcsine law.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape != (n,):
                # interpolate/resample to the required length
                x_old = np.linspace(0.0, 1.0, arr.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                arr = np.interp(x_new, x_old, arr)
            a0 = np.maximum(arr, 0.0)
            s = a0.sum()
            if s == 0.0:
                a0 = _arcsine_distribution(n)
            else:
                a0 = a0 / s
        except Exception:
            a0 = _arcsine_distribution(n)
    else:
        a0 = _arcsine_distribution(n)

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent with momentum.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    subgrad_iters = min(180_000, int(time_left * 30))  # rough budget
    print("[Info] Starting projected sub‑gradient phase")
    best_seq, best_val = _proj_subgradient_opt(
        a0,
        deadline,
        max_iters=subgrad_iters,
        eta0=1.0,
        beta_mom=0.9,
    )
    print(f"[Info] Sub‑gradient phase finished – surrogate {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic hill‑climbing refinement.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        hill_trials = int(min(300_000, time_left * 200))
        print("[Info] Starting hill‑climbing phase")
        best_seq, best_val = _hill_climb_opt(
            best_seq,
            deadline,
            max_trials=hill_trials,
            epsilon_initial=0.02,
            temperature_initial=0.5,
        )
        print(f"[Info] Hill‑climbing phase finished – surrogate {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
