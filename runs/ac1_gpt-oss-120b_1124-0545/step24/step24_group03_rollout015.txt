"""
Optimiser for the “step‑function” task.

The objective (after normalising the sum to 1) is 2·n·max(conv(a,a)).
The global lower bound is 2 (the uniform sequence) but the best known
construction reaches ≈1.5085 for n=4000.  This version tries to go
further by:

1. Generating a *very* rich set of analytic seeds (beta‑shapes,
   power‑exponential families, mixed exponentials, random Dirichlet,
   etc.) and keeping the few best according to the exact surrogate
   2·n·max(conv).

2. Running the four‑stage non‑convex optimiser (exponentiated‑gradient,
   smooth‑max L‑BFGS, flattening gradient and simulated‑annealing) on
   each of those top seeds.  The overall time‑budget is split evenly
   between the seeds.

3. (Optional) a few iterations of a linear‑programming “LP‑step” that
   solves the classic LP described in the statement:
   maximise Σ g_i  subject to  a∗g ≤ M  (M = max(a∗a)),
   then mix the resulting g back into a.  This step further flattens
   the self‑convolution and costs only a couple of seconds for n=4000.

All helpers are top‑level (no closures), the code uses only NumPy and
SciPy (when available) and respects the 1000 s wall‑clock limit.
The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional SciPy imports – the LP‑step works only if SciPy is present
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate: 2·n·max(conv)  (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# LP‑step (the classic linear‑program described in the statement)
# ----------------------------------------------------------------------
def _lp_step(seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    One iteration of the LP improvement:
        maximise Σ g_i  subject to  (seq * g)_k ≤ M  (all k),
    then mix the normalised solution back into ``seq``.
    Returns the new sequence and its surrogate value.
    If SciPy is not available, this function does nothing.
    """
    if _opt is None:
        return seq, _objective_surrogate(seq)

    # current bound
    conv = _conv_fft(seq, seq)
    M = conv.max()
    L = 2 * n - 1

    # build the (L × n) constraint matrix
    # column j contains seq shifted by j positions
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = seq

    # linear programme: maximise 1^T g  (i.e. minimise -1^T g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, M, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # solve – HiGHS is very fast for this size
    res = _opt.linprog(
        c=c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "disp": False},
    )
    if not res.success:
        # if the LP fails, just return the unchanged sequence
        return seq, _objective_surrogate(seq)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return seq, _objective_surrogate(seq)

    # normalise g to keep the simplex invariant
    g_norm = g / sum_g

    # convex combination – a modest step size keeps monotonicity
    t = 0.5
    new_seq = (1.0 - t) * seq + t * g_norm
    new_seq = new_seq / new_seq.sum()           # re‑normalise (numerical safety)

    # stop early if the global timer is exhausted
    if time.time() > deadline:
        return new_seq, _objective_surrogate(new_seq)

    return new_seq, _objective_surrogate(new_seq)


# ----------------------------------------------------------------------
# optimisation stages (identical to the “deep‑search” version)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg
        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent on the simplex (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    seq_eg, _ = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=14.0,
        tol=1e-11,
    )
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, _objective_surrogate(seq_eg)

    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs, n, deadline)
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq_flat, n, deadline, rng)
    else:
        seq_sa, val_sa = seq_flat, val_flat

    # pick the best amongst the stages
    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, _objective_surrogate(seq_eg)),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (much denser than the previous “deep‑search” version)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # a few deterministic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))
    for p in [0.4, 0.5, 0.6, 0.8, 1.0]:
        seeds.append(_power(n, p))
    for rate in np.logspace(-5, -0.2, 15):
        seeds.append(_exp_decay(n, rate))

    # dense random beta‑shapes
    for _ in range(300):
        a = 10 ** rng.uniform(-3, 1)          # α ∈ [1e‑3, 10]
        b = 10 ** rng.uniform(-3, 1)          # β ∈ [1e‑3, 10]
        seeds.append(_beta_shape(n, a, b))

    # dense power‑exponential family
    for _ in range(200):
        p = rng.uniform(0.2, 1.2)
        lam = rng.uniform(0.0, 0.01)
        seeds.append(_power_exp(n, p, lam))

    # mixed exponentials
    for _ in range(200):
        r1 = rng.uniform(1e-5, 0.5)
        r2 = rng.uniform(1e-5, 0.5)
        alpha = rng.uniform(0.0, 1.0)
        seeds.append(_mixed_exponential(n, r1, r2, alpha))

    # random Dirichlet draws
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5, 1.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixture of uniform + a base shape (break symmetry)
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # use the previous global best if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0:
                    seeds.append(prev_arr / s)
        except Exception:
            pass

    # reversed copies (different ordering)
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # small random perturbations of the current global best (if any)
    if seeds:
        best_seed = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = best_seed + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin – the caller will kill the process after budget_s
    deadline = start_time + budget_s - 2.0

    # problem size – 4000 works well with the FFT‑based surrogate
    n = 4000
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Massive seed generation & quick surrogate evaluation
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds – evaluating surrogates...")

    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogates:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2. Initialise global best with the best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Deep optimisation of each top seed (time‑split equally)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and top_k > 0:
        per_seed_budget = remaining * 0.85 / top_k
        for idx, seed_seq in enumerate(top_seeds):
            sub_deadline = time.time() + per_seed_budget
            print(f"\n[Deep opt] seed #{idx+1}/{top_k}, budget ≈{per_seed_budget:.1f}s")
            seq_opt, val_opt = _full_optimisation(seed_seq, n, sub_deadline, rng)
            if val_opt < global_best_val:
                global_best_seq = seq_opt
                global_best_val = val_opt
                print(f"   *** Updated global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional LP‑based polishing (a few iterations, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        print("\n[LP polishing] performing a few LP‑steps")
        lp_iters = 4
        for it in range(lp_iters):
            if time.time() > deadline:
                break
            global_best_seq, global_best_val = _lp_step(global_best_seq, n, deadline)

    # ------------------------------------------------------------------
    # 5. Final gradient‑based polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Final polish] remaining time used for full optimisation")
        seq_fin, val_fin = _full_optimisation(global_best_seq, n, deadline, rng)
        if val_fin < global_best_val:
            global_best_seq = seq_fin
            global_best_val = val_fin
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
