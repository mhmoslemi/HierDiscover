def _exp_gradient_opt2(start_seq, n, deadline, max_iters=20_000_000, eta0=5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # determine all active indices within eps tolerance
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient as average over active constraints
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2 * a[rev_idx]
        g = g / len(active)
        eta = eta0 / math.sqrt(it)

        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _adam_subgrad_opt_flexible(start_seq, n, deadline, max_iters=1_000_000,
                              eta0=1.0, beta1=0.9, beta2=0.999):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v2 = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    eps = 1e-15
    while time.time()<deadline and it<max_iters:
        it+=1
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # compute subgradient for active offsets
        max_idxs = np.where(conv >= max_val - eps)[0]
        g = np.zeros_like(a)
        # For each active offset
        for k in max_idxs:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            g[idx_i] += 2 * a[rev_idx]
        # average over active offsets if multiple
        g = g / len(max_idxs)
        # chain rule for log_softmax
        grad_log = a * (g - np.mean(g * a))
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_log
        v2 = beta2 * v2 + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**it)
        v2_hat = v2 / (1 - beta2**it)
        step = eta0 / math.sqrt(it) * m_hat / (np.sqrt(v2_hat) + 1e-9)
        log_a = log_a - step
        cur_val = max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Improved optimizer for the step‑function attack.

The search proceeds in four stages:

1. Smooth Adam (log‑softmax + log‑sum‑exp smoothing) – a quick
   high‑level refinement of any starting vector.
2. Sub‑gradient Adam – direct optimisation of the exact max‑convolution
   using an Adam‑style update on the true sub‑gradient.
3. Exponentiated‑gradient (mirror‑descent) – a long run with a decreasing
   1/√t step‑size that drives the distribution toward the optimum.
4. Aggressive simulated‑annealing hill‑climb – three move types
   (swap, Gaussian perturb, targeted mass‑shift) with a slower cooling
   schedule for final polishing.

The routine respects the 1000 s budget (leaving a 5 s safety margin) and
stores the best sequence in the global ``height_sequence_1`` so that a
future run can continue from it.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector of sum = 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate objective (sum(seq)=1): 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) and normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smooth Adam (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_smooth(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 500_000,
                 beta_start: float = 0.5,
                 beta_end: float = 1e9,
                 alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Variables are kept in log‑softmax space (v); a = softmax(v).
    """
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # schedule for smoothing parameter β (geometric)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        # smoothed objective & gradient (see original code)
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

        # gradient w.r.t. a (derived analytically)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient Adam (direct optimisation of max‑convolution)
# ----------------------------------------------------------------------
def _adam_subgrad(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 2_000_000,
                  eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that works directly with the true max‑convolution.
    The sub‑gradient is taken from all indices that are (approximately)
    active (i.e. within a tiny tolerance of the maximum).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v2 = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # collect all active offsets (within 1e-12 relative tolerance)
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of the (max) objective w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        g = g / float(len(active))          # average over the active set

        # chain rule through softmax: grad_log = a * (g - <g,a>)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam step on log‑space variables
        m = beta1 * m + (1.0 - beta1) * grad_log
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = (eta0 / math.sqrt(it)) * m_hat / (np.sqrt(v2_hat) + eps)
        log_a = log_a - step

        cur_val = max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) on true max
# ----------------------------------------------------------------------
def _exp_gradient(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 10_000_000,
                  eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a simple 1/√t step‑size, staying on the simplex
    via the exponentiated‑gradient (log‑softmax) parametrisation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set (as above)
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        g = g / float(len(active))

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with three move types:
        * swap two positions,
        * add a small Gaussian perturbation to every entry,
        * targeted “mass‑shift” from a dominant pair to a random receiver.
    The temperature schedule is slower than in the original code,
    allowing more thorough exploration.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    # a more gradual cooling schedule
    T0 = max(1e-2 * cur_val, 1e-4)
    T1 = 1e-7 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.35:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue

        elif move_type < 0.70:                   # Gaussian perturb
            sigma = 0.01 / math.sqrt(it + 1)    # start larger, decay
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                     # targeted mass‑shift
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            donor = int(idx_i[np.argmax(contrib)])
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                continue
            delta = min(cur_seq[donor], 0.5 * cur_seq[donor]) * 0.05   # 5 % of donor
            ns = cur_seq.copy()
            ns[donor] -= delta
            ns[receiver] += delta
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

"""
Improved optimizer for the step‑function attack.

The search proceeds in four stages:

1. **Smooth Adam** – optimises a log‑sum‑exp smoothed surrogate of the
   max‑convolution while the variables stay in log‑softmax space.
2. **Sub‑gradient Adam** – directly minimises the true max‑convolution
   using an Adam‑style update on the exact sub‑gradient (averaged over
   all active offsets).
3. **Exact exponentiated‑gradient** – a long mirror‑descent run with a
   1/√t step‑size that stays on the simplex via the exponentiated‑gradient
   (log‑softmax) parametrisation.
4. **Aggressive hill‑climb** – simulated‑annealing with three move types
   (swap, Gaussian perturbation, targeted mass‑shift) and a slower cooling
   schedule for final polishing.

The routine respects the 1000 s budget (leaving a 5 s safety margin) and
stores the best sequence in the global ``height_sequence_1`` so that a
future run can continue from it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector of sum = 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate objective (sum(seq)=1): 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) and normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smooth Adam (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_smooth(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 0.5,
    beta_end: float = 1e9,
    alpha: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max‑convolution surrogate."""
    v = np.log(start_seq + 1e-30)            # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        # smoothed objective (not used for the update, only for diagnostics)
        _ = 2.0 * n * (shift + np.log(sum_exp) / beta)

        # gradient w.r.t. a (derived analytically)
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v -= step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient Adam (direct max‑convolution optimisation)
# ----------------------------------------------------------------------
def _adam_subgrad(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 5.0,
) -> tuple[np.ndarray, float]:
    """Adam optimiser that works directly with the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v2 = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max_val * 1e-12 + 1e-15
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        g = g / float(len(active))

        # chain rule through softmax
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam dynamics on log‑space
        m = beta1 * m + (1.0 - beta1) * grad_log
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = (eta0 / math.sqrt(it)) * m_hat / (np.sqrt(v2_hat) + eps)
        log_a -= step

        cur_val = max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
