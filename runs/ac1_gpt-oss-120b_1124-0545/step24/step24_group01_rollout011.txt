import numpy as np, time, math

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft_full(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(logit):
    max_log = np.max(logit)
    e = np.exp(logit - max_log)
    s = e.sum()
    return e / s

def _smooth_max_grad(a, tau):
    # compute convolution a*a
    conv = _conv_fft_full(a, a)
    # smoothed max weights
    w = np.exp(tau * conv)
    w = w / w.sum()
    # gradient = 2 * conv(a, w) (first n entries)
    grad_full = _conv_fft_full(a, w)
    grad = 2.0 * grad_full[:len(a)]
    return grad

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm proceeds in three phases:

1. **Diverse initial seeds** – flat, arcsine, triangular, many
   symmetric Beta shapes and a few Dirichlet samples (all normalised
   to sum = 1).  The five best seeds (according to the cheap surrogate
   `2·n·max(conv(a,a))`) are kept.

2. **Smoothed‑max exponentiated‑gradient (Adam)** – we minimise the
   exact surrogate but replace the non‑smooth `max` by a
   temperature‑controlled softmax:
   ```
   Sτ(a) = (1/τ)·log Σk exp(τ·(a*a)[k]).
   ```
   Its gradient can be computed efficiently via two FFT‑based
   convolutions.  The optimisation runs in log‑space with Adam momentum,
   a diminishing learning‑rate and a linear increase of the temperature
   τ (from 0.5 up to 8).  After each iteration the iterate is projected
   back onto the simplex by a softmax; a tiny symmetrisation
   ``a←(a+a[::-1])/2`` is performed to exploit the inherent symmetry of
   the problem.

3. **Fine‑grained refinement** – greedy mass transfer between the most
   and least harmful coordinates, followed by a stochastic local walk.
   The best sequence found is finally evaluated with the supplied
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(logit: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(logit)
    e = np.exp(logit - m)
    return e / e.sum()


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(a,a)),   assuming ``a`` sums to 1.
    """
    n = len(a)
    conv = _conv_fft_full(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _gradient_smoothed_max(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum
        (1/τ)·log Σ exp(τ·(a*a)[k])
    w.r.t. ``a``.
    """
    n = len(a)
    conv_aa = _conv_fft_full(a, a)                  # length 2n‑1
    w = np.exp(tau * conv_aa)
    w /= w.sum()                                    # probability vector, length 2n‑1
    w_rev = w[::-1]                                 # reverse for cross‑correlation
    conv_rev = _conv_fft_full(a, w_rev)             # length 3n‑2
    start = n - 1
    # slice that corresponds to Σ a_j w_{i+j} for i=0..n‑1, then reverse
    slice_part = conv_rev[start:start + n][::-1]
    grad = 2.0 * slice_part
    return grad


def _adam_expgd(start_seq: np.ndarray,
                deadline: float,
                lr0: float = 0.05,
                beta1: float = 0.9,
                beta2: float = 0.999,
                tau_start: float = 0.5,
                tau_end: float = 8.0,
                tau_ramp_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror‑descent (exponentiated gradient) on the exact
    smoothed‑max surrogate.  Returns the best sequence seen and its
    surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    logit = np.log(a + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    eps = 1e-8

    while time.time() < deadline:
        it += 1

        # linear temperature schedule
        if it >= tau_ramp_iters:
            tau = tau_end
        else:
            tau = tau_start + (tau_end - tau_start) * it / tau_ramp_iters

        grad = _gradient_smoothed_max(a, tau)               # d/d a of smoothed max
        dot = np.dot(grad, a)                               # ⟨a,∇⟩
        grad_log = a * (grad - dot)                         # mirror gradient

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size
        eta = lr0 / math.sqrt(it)

        # update log‑parameters
        logit = logit - eta * m_hat / (np.sqrt(v_hat) + eps)

        # map back onto the simplex
        a = _softmax(logit)

        # tiny symmetrisation – helps to destroy asymmetric bottlenecks
        a = (a + a[::-1]) / 2.0
        a /= a.sum()

        # periodic exact evaluation
        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"Iter {it:6d}: surrogate = {cur_val:.6f}")
            if best_val < 1.503:          # target reached – stop early
                break

        # safe‑guard: leave a few seconds for the final refinement
        if time.time() > deadline - 1.0:
            break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest, keeping only improving
    moves.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        conv = _conv_fft_full(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active term (k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Only moves that improve the
    exact surrogate are kept.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val, best_seq = val, new_seq
    return best_seq, best_val

    """

    start_time = time.time()
    # safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Gather a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # symmetric Beta(α,α) – many shapes (U‑shaped → bell‑shaped)
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped

    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):

            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # 1️⃣  Fast surrogate evaluation of all seeds

    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Adam‑EG optimisation (exact surrogate) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):

                print("   >>> Updated global best (EG)")

            # 2️⃣‑b  Greedy refinement of the EG result
            # ----------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # 3️⃣  Final random local refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=300_000,
                                                sigma=0.015)
        print(f"[Local] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    true_val = None
    try:

    return best_seq.tolist()
