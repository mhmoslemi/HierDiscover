def _smooth_obj_and_grad(logits, beta):
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    maxc = np.max(conv)
    exp_beta = np.exp(beta * (conv - maxc))
    w = exp_beta / np.sum(exp_beta)   # shape len(conv)
    # objective: 2*n * sum(conv * w)
    obj = 2.0 * n * np.dot(conv, w)
    # gradient: For each a_i: grad_i = 2*n * sum_k w_k * d conv_k / d a_i
    # d conv_k / d a_i = a_{k-i} + a_{i - k} (when in range)
    # Equivalent to convolving w with reversed a (twice)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    # full length = len(rev_a)+len(w)-1 = n + (2n-1) -1 = 3n -2?
    # Actually len(conv)=2n-1, len(w)=2n-1. Then full length = n + (2n-1) -1 = 3n -2.
    # We need values at positions (n-1) to (n-1 + n-1) inclusive => slice [n-1:n-1+n]
    grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # shape n
    grad_a = grad_weighted * (2.0 * n)
    # Convert to gradient w.r.t logits:
    dot = np.dot(grad_a, a)
    grad_log = a * (grad_a - dot)
    return obj, grad_log
```

def _lbfgs_smooth(start_seq, deadline):
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    betas = [10.0, 200.0, 5000.0]
    for beta in betas:
        if time.time() > deadline: break
        # define objective function specific to beta:
        def fun(logits):
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            maxc = np.max(conv)
            exp_beta = np.exp(beta * (conv - maxc))
            w = exp_beta / np.sum(exp_beta)
            obj = 2.0 * n * np.dot(conv, w)
            # gradient
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n-1:n-1+n]
            grad_a = grad_weighted * (2.0 * n)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)
            return obj, grad_log
        # Use SciPy L-BFGS with gradient
        res = minimize(fun, logit, method='L-BFGS-B', jac=True,
                       options={'maxfun': 200, 'maxiter': 200, 'ftol':1e-12})
        logit = res.x
        a = _softmax(logit)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # early exit if time over
        if time.time() > deadline:
            break
    return best_seq, best_val
```

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm overview
-----------------
1️⃣ **Rich seed pool** – generate analytic families (uniform, triangular,
   power‑law (generalised arcsine) with a fine sweep of the exponent,
   symmetric/asymmetric Beta, Dirichlet draws) and also linear mixtures
   of the best analytic seed with the uniform distribution.  All seeds
   are normalised to sum‑to‑one.

2️⃣ **Cheap surrogate ranking** – compute the surrogate objective
   ``2·n·max(conv(a,a))`` (the same as ``evaluate_sequence`` for a
   normalised sequence).  Keep the top few seeds.

3️⃣ **Smooth optimisation (L‑BFGS‑B)** – minimise a smooth
   log‑sum‑exp approximation of the max‑convolution.  The optimisation
   is performed in the log‑softmax (mirror) space, keeping the iterate
   automatically on the simplex.  A small schedule of increasing smoothness
   parameters (β) is used.

4️⃣ **Exact polishing (exponentiated‑gradient / mirror descent)** – run a
   sub‑gradient descent on the true max‑convolution with a decaying step‑size
   ``η = η₀/√t``.  This stage dominates the remaining time‑budget.

5️⃣ **Final annealing walk** – a cheap simulated‑annealing walk that
   transfers a small random fraction of mass between two coordinates.
   It helps escape shallow local minima.

6️⃣ **Official evaluation** – call the supplied ``evaluate_sequence`` on the
   best candidate, store it globally for the next invocation and return it.

The whole routine respects the 1000 s budget; it adapts the time split
according to the remaining deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback (should not happen)
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    ``seq`` is assumed to already sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom + 1e-12, -p)
    return a / a.sum()


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mixture(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam*a + (1-lam)*b, renormalised to sum=1."""
    mixed = lam * a + (1.0 - lam) * b
    s = mixed.sum()
    if s == 0.0:
        return a.copy()  # fallback
    return mixed / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _lbfgs_smooth(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on a smooth log‑sum‑exp approximation of max‐convolution.
    The variable is the logits; the simplex constraint is enforced by the
    softmax transform.  A short schedule of increasing smoothness parameters
    (β) is used.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)  # initialise logits
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # schedule of smoothness values – start moderate, end very sharp
    beta_schedule = [20.0, 200.0, 2000.0, 20000.0, 200000.0, 2_000_000.0]

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(logits):
            """Return (objective, grad) for given logits."""
            a = _softmax(logits)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp weights (smooth max)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)

            # smooth surrogate (scaled exactly like the true one)
            obj = 2.0 * n * np.dot(conv, w)

            # gradient w.r.t. a: 2·n·Σ_k w_k·∂conv_k/∂a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = n + (2n‑1) – 1 = 3n‑2
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # slice size n
            grad_a = grad_weighted * (2.0 * n)

            # convert to gradient in logits space (mirror descent)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)
            return obj, grad_log

        # L‑BFGS‑B – we ask it to use both value and gradient
        res = minimize(fun,
                       logit,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'ftol': 1e-12,
                                'disp': False})

        logit = res.x
        a = _softmax(logit)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

        # early exit if we are out of time
        if time.time() > deadline:
            break

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Sub‑gradient is taken at the currently active index.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.02,
                final_temp: float = 1e-7,
                sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk on the simplex.
    Moves transfer a random fraction of mass from i → j.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_a)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val
