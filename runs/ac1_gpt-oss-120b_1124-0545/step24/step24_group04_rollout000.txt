def _gradient_local_search(a, n, deadline, rng):
    cur = a.copy()
    cur_obj = _surrogate(cur) # surrogate is exact objective (2*n*max conv)
    # schedule params
    step_factor_start = 0.5
    step_factor_end = 0.001
    decay_factor = 0.99
    step_factor = step_factor_start
    iteration = 0
    while time.time() < deadline:
        # compute conv and max index
        conv = _conv_fft(cur, cur)
        max_val = np.max(conv)
        k = int(np.argmax(conv))
        # compute gradient g
        g = np.zeros_like(cur)
        # indices i such that 0 <= k-i < n
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # compute rev_idx = k - i
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * cur[rev_idx]
        # pick i_high, i_low
        i_high = int(np.argmax(g))
        i_low = int(np.argmin(g))
        if cur[i_high] <= 0:
            # skip if no mass to transfer
            # fallback random move
            i_high, i_low = rng.choice(n, 2, replace=False)
        # step size
        delta = step_factor * cur[i_high]
        if delta <= 1e-12:
            # shrink step factor
            step_factor = max(step_factor_end, step_factor * decay_factor)
            continue
        # propose new
        new = cur.copy()
        new[i_high] -= delta
        new[i_low] += delta
        # ensure non-negative
        if new[i_high] < 0:
            continue
        new_obj = _surrogate(new)
        if new_obj < cur_obj:
            cur = new
            cur_obj = new_obj
            # optional: reset step factor maybe increase
            # keep same step_factor
        else:
            # optionally accept with small probability? Simulated annealing
            if rng.random() < 0.001:
                cur = new
                cur_obj = new_obj
        # decay step factor periodically
        iteration += 1
        if iteration % 200 == 0:
            step_factor = max(step_factor_end, step_factor * decay_factor)
    return cur, cur_obj
```

"""
Hybrid optimisation for the step‑function problem.

The optimiser proceeds in three stages:

1️⃣  A very diverse pool of seed vectors (uniform, arcsine, triangular,
    Dirichlet draws with many concentrations, random Gaussian mixtures,
    sparse “spike” vectors and the previous best solution) is generated.
    All seeds are normalised to sum = 1 and the best few are kept
    according to the exact objective 2 · n · max(conv(a,a)).

2️⃣  Starting from each retained seed we run an exponentiated‑gradient (mirror‑descent)
    optimiser on the true max‑convolution (the same routine used in the previous
    baseline).  This quickly reduces the objective and provides a good
    neighbourhood for the next phase.

3️⃣  A custom gradient‑driven stochastic local‑search is applied.
    At every iteration we:
        • compute the index k* where the convolution attains its maximum,
        • form the exact sub‑gradient g_i = 2·a_{k*‑i} (zero outside the support),
        • move a fraction of mass from the index with largest g_i to the index with
          smallest g_i (a “mass‑transfer” step).  The step‑size decays
          geometrically, allowing large early jumps and fine refinements later.
    The move is kept only if it strictly lowers the objective.
    The loop runs until the global deadline, performing many thousands of cheap
    iterations (each iteration only needs one FFT of length ≈ 2 n).

Finally the best sequence found is evaluated with the provided ``evaluate_sequence``
and stored in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when sum(a)==1:
        2 * n * max(conv(a,a))
    """
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of a few Gaussian bumps (truncated to the interval)."""
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _spike_vector(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Create a sparse vector with exactly m non‑zero equal entries."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on the exact max)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its exact objective.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑driven stochastic mass‑transfer local search
# ----------------------------------------------------------------------


def _gradient_mass_transfer(a_start: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Large‑scale stochastic walk guided by the exact sub‑gradient.
    At each step mass is moved from the index with the largest gradient
    to the index with the smallest gradient.  The step‑size decays
    geometrically, allowing a coarse‑to‑fine optimisation.
    Returns the best vector seen and its exact objective.
    """
    cur = a_start.copy()
    cur_val = _objective(cur, n)

    # step‑size schedule
    step_factor = 0.5          # start with up to 50 % of the source mass
    step_factor_end = 0.0005   # final granularity
    decay = 0.985              # geometric decay every 200 successful moves
    successful = 0

    while time.time() < deadline:
        # compute the max‑convolution and the index where it occurs
        conv = _conv_fft(cur, cur)
        max_val = np.max(conv)
        k = int(np.argmax(conv))

        # exact sub‑gradient of the active max‑convolution term
        g = np.zeros_like(cur)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * cur[rev_idx]

        # choose source (largest gradient) and target (smallest gradient)
        i_src = int(np.argmax(g))
        i_tgt = int(np.argmin(g))

        # if the source has essentially no mass we fall back to a random move
        if cur[i_src] <= 1e-14:
            i_src, i_tgt = rng.integers(0, n, size=2, endpoint=False)
            if i_src == i_tgt:
                continue

        delta = step_factor * cur[i_src]   # transfer a fraction of the source mass
        if delta <= 1e-15:
            # step became too tiny – stop early
            break

        # propose the move
        new = cur.copy()
        new[i_src] -= delta
        new[i_tgt] += delta

        # safety check (numerics)
        if new[i_src] < -1e-15:
            continue

        new_val = _objective(new, n)

        if new_val < cur_val:
            cur = new
            cur_val = new_val
            successful += 1
            # decay the step size occasionally
            if successful % 200 == 0:
                step_factor = max(step_factor_end, step_factor * decay)
        else:
            # occasional uphill acceptance (simulated‑annealing flavour)
            if rng.random() < 0.001:
                cur = new
                cur_val = new_val

    return cur, cur_val
