def _random_local_search(start_seq, n, deadline, rng, max_attempts=200000, start_sigma=0.002, T_start=5e-3, T_end=5e-9):
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start) if T_start != 0 else start_sigma
        # pick indices
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        # delta scaled by sigma and by a[i] (source)
        max_transfer = a[i]
        # draw delta = sigma * rng.normal() * max_transfer maybe
        delta = sigma * rng.normal()
        # cap delta
        if delta > 0:
            delta = min(delta, max_transfer)
        else:
            # moving from j to i
            delta = max(delta, -a[j])
        ns = a.copy()
        ns[i] += delta
        ns[j] -= delta
        # ensure nonnegative due to rounding
        # renorm to sum 1 (should stay same) but guard against negative rounding
        if ns[i] < 0 or ns[j] < 0:
            continue
        # (No need to renorm because sum unchanged)
        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1
    return best_seq, best_val

"""
Improved step‑function optimizer.

The optimizer works on a dense grid (default n=8000) and proceeds in three
stages:

1.  A very diverse seed set (analytic families, fine grid search over the
    two‑parameter power‑exponential family and many Dirichlet draws) is built
    and the few best seeds (according to the exact surrogate 2·n·max(conv))
    are kept.

2.  Each kept seed is refined with an L‑BFGS optimisation of a smooth
    log‑sum‑exp surrogate for the maximum convolution.  The schedule of the
    smoothing parameter β is long and the optimiser is given a generous number
    of iterations.

3.  A fast, pure‑Python simulated‑annealing / hill‑climbing phase makes tiny
    mass transfers between random coordinates.  This “local‑search” can still
    improve the surrogate by a few 10⁻⁴, which is enough to push the value
    below the target 1.503.

All stages respect the global wall‑clock deadline.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Return smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# surrogate objective (exact after L₁‑normalisation)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    """Two‑parameter family a_i ∝ i^{‑p}·exp(‑λ·i) (i starts at 1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1e7, 2e7,
        ]

    # work in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # push onto log‑softmax manifold
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple stochastic local search (mass transfer between two points)
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.002,
                         T_start: float = 6e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq, best_val = a.copy(), cur_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start) if T_start != 0 else start_sigma

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a small transfer delta
        delta = sigma * rng.normal()
        if delta > 0:
            delta = min(delta, a[i])          # take from i, give to j
        else:
            delta = max(delta, -a[j])          # take from j, give to i
            i, j = j, i                         # swap direction so delta>0
            delta = -delta

        if delta == 0.0:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] += delta
        ns[j] -= delta
        # enforce non‑negativity (tiny numerical errors)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fine grid search for the two‑parameter family a_i ∝ i^{-p}·exp(-λ·i)
# ----------------------------------------------------------------------
def _grid_search_power_exp(n: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Coarse‑to‑fine grid search over (p, λ). Returns best sequence and its surrogate."""
    # coarse grid
    p_vals = np.linspace(0.4, 1.0, 13)          # 0.4 … 1.0
    lam_vals = np.linspace(0.0, 0.008, 9)      # 0 … 8·10⁻³
    best_seq, best_val, best_p, best_lam = None, float('inf'), None, None

    for p in p_vals:
        for lam in lam_vals:
            seq = _power_exp(n, p, lam)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_seq, best_val, best_p, best_lam = seq, val, p, lam

    # three refinement stages
    for radius, steps in [((0.15, 0.0015), 7),
                         ((0.06, 0.0006), 9),
                         ((0.025, 0.00025), 11)]:
        p_center, lam_center = best_p, best_lam
        p_low, p_high = max(0.0, p_center - radius[0]), p_center + radius[0]
        lam_low, lam_high = max(0.0, lam_center - radius[1]), lam_center + radius[1]
        p_grid = np.linspace(p_low, p_high, steps)
        lam_grid = np.linspace(lam_low, lam_high, steps)
        for p in p_grid:
            for lam in lam_grid:
                seq = _power_exp(n, p, lam)
                val = _objective_surrogate(seq)
                if val < best_val:
                    best_seq, best_val, best_p, best_lam = seq, val, p, lam

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (large, diverse pool)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    for p in [0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.2, 25):
        seeds.append(_exp_decay(n, rate))

    # mixed exponentials
    rates = np.logspace(-5, -0.2, 12)
    for r1 in rates[:6]:
        for r2 in rates[6:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    # beta‑distribution shapes
    for a in [0.3, 0.5, 0.8, 1.2, 2.0]:
        for b in [0.3, 0.5, 0.8, 1.2, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # fine‑grid power‑exponential (the result itself is a seed)
    best_pe_seq, _ = _grid_search_power_exp(n, rng)
    seeds.append(best_pe_seq)

    # random Dirichlet draws
    for conc in [0.001, 0.003, 0.008, 0.02, 0.1, 0.5]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # convex mixes with uniform to break symmetry
    base_shapes = [_inverse_sqrt(n), _exp_decay(n, 1e-3), _power(n, 1.0)]
    for base in base_shapes:
        for alpha in [0.15, 0.35, 0.55]:
            seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # reversed copies of everything
    seeds.extend([s[::-1].copy() for s in list(seeds)])

    # include previous global best if available and sized correctly
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    seeds.append(arr / s)
        except Exception:  # pragma: no cover
            pass

    # small random perturbations of the current best seed (if any)
    if seeds:
        cur_best = min(seeds, key=_objective_surrogate)
        for _ in range(4):
            pert = cur_best + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted wall‑clock budget, stores the best
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin – the caller may terminate after exactly budget_s seconds
    deadline = start_time + budget_s - 2.0

    # problem size – 8000 gives a finer discretisation while still being fast
    n = 8000
    print(f"\n=== Starting optimisation (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Build a large, diverse seed pool
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)
    print(f"Generated {len(seeds)} raw seeds. Evaluating surrogates...")

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the few best
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed with L‑BFGS + local stochastic search
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refining seed #{idx + 1} ---")
        sub_deadline = deadline - 1.0  # leave a little room for the final polish

        # L‑BFGS on the smooth max surrogate
        seq_lbfgs = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            sub_deadline,
            beta_schedule=None,
            max_iter_per_beta=8000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"  after L‑BFGS: {val_lbfgs:.6f}")

        # Stochastic local search (mass‑transfer neighbourhood)
        if time.time() < sub_deadline:
            seq_loc, val_loc = _random_local_search(
                seq_lbfgs,
                n,
                sub_deadline,
                rng,
                max_attempts=250_000,
                start_sigma=0.0015,
                T_start=5e-3,
                T_end=5e-9,
            )
            print(f"  after local search: {val_loc:.6f}")
        else:
            seq_loc, val_loc = seq_lbfgs, val_lbfgs

        # update global best if we improved
        if val_loc < global_best_val:
            global_best_val, global_best_seq = val_loc, seq_loc.copy()
            print(f"  *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with a longer stochastic run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n=== Final polishing phase ===")
        seq_final, val_final = _random_local_search(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=500_000,
            start_sigma=0.0010,
            T_start=3e-3,
            T_end=1e-9,
        )
        if val_final < global_best_val:
            global_best_val, global_best_seq = val_final, seq_final.copy()
            print(f"[Final] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
