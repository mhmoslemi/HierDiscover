"""
Optimiser for ``evaluate_sequence`` – a multi‑stage simplex search.

The optimiser works inside a fixed wall‑clock budget (default 1000 s) and
consists of the following stages:

1️⃣ **Rich seed pool** – analytic families (uniform, decreasing power‑law,
   decreasing exponential, beta, triangular) and many Dirichlet draws.
   A cheap surrogate ``2·n·max(conv(a,a))`` (the true value for a
   normalised sequence) is used to pick the most promising few seeds.

2️⃣ **Extended Adam smoothing** – minimise a log‑sum‑exp smooth
   approximation of the max‑convolution.  The smoothing parameter β is
   increased gradually; the inner Adam loop now performs many more
   updates than the reference implementation.

3️⃣ **Multi‑start exponentiated‑gradient polishing** – exact sub‑gradient
   descent on the true max‑convolution for each of the top seeds.
   The step‑size uses a larger initial η₀ and the run is limited only
   by the remaining wall‑clock time.

4️⃣ **Final simulated‑annealing walk** – a cheap stochastic walk on the
   simplex that occasionally accepts worsening moves.  This helps escape
   shallow local minima.

All intermediate results are logged.  The best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    The input is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families – all produce a probability vector (sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _power_decreasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, i = 0 … n‑1."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(i, -p)
    a = a / a.sum()
    return a


def _exponential_decay(n: int, r: float) -> np.ndarray:
    """
    Geometric decay a_i ∝ r**i  (0 < r < 1).
    """
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    a = a / a.sum()
    return a


def _beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Single Dirichlet draw with scalar concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth (log‑sum‑exp) surrogate.
    The inner loop now runs up to 20 000 iterations per β value,
    which gives a noticeably tighter smoothing optimisation.
    """
    n = len(start_seq)
    # initialise logits (mirror‑space)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a longer, finer schedule for the smoothness parameter β
    beta_schedule = [
        1.0, 3.0, 10.0, 30.0, 100.0, 300.0,
        1_000.0, 3_000.0, 10_000.0, 30_000.0,
        100_000.0, 300_000.0, 1_000_000.0, 3_000_000.0,
        10_000_000.0, 30_000_000.0, 100_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # allow many more Adam steps per β value
        max_iter_beta = 20_000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                  # current point on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)    # outer factor 2·n

            # convert to gradient in logits (mirror space)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation (every few hundred steps)
            if it % 800 == 0:
                cur_val = _objective_surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 8.0,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    The step‑size schedule η₀ / sqrt(t) is retained, but we allow many
    more iterations (up to max_iters) subject to the time budget.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k (exact for max)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # keep track of the best ever seen
        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _anneal_opt(start_seq: np.ndarray,
                deadline: float,
                init_temp: float = 0.03,
                final_temp: float = 1e-8,
                sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk on the simplex.
    Transfers a tiny random fraction of mass from i → j.
    """
    n = len(start_seq)
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling schedule
        temp = init_temp * (1.0 - elapsed / total)
        temp = max(final_temp, temp)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = sigma * a[i] * rng.random()
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

                                       init_temp=0.03,

                                       sigma=0.02)
        print(f"[Anneal] surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best after annealing")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
