"""
Enhanced step‑function optimizer.

The routine builds a collection of diverse seeds (uniform, inverse‑sqrt,
random Dirichlet, power‑law, exponential), runs a smoothed L‑BFGS on
the soft‑max (log‑probability) parametrisation, follows it with a long
exponentiated‑gradient (mirror‑descent) polishing phase and finally
applies a light simulated‑annealing local search.  All work respects
the supplied time budget (default 1000 s).  The best sequence found is
stored in the global ``height_sequence_1`` variable and returned as a
list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the L‑BFGS optimiser.
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of size power‑of‑2."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (the result sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (the quantity we are really minimising)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return ``2·n·max(conv(seq,seq))`` where ``n=len(seq)``."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seed generation
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i (0<r<1)."""
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
):
    """Minimise the smoothed max‑convolution objective."""
    if _opt is None:  # pragma: no cover
        # No SciPy – fall back to a tiny EG run.
        return _exp_gradient_opt(start_a, deadline, max_iters=200_000)

    n = len(start_a)
    # initialise log‑softmax parameters
    v = np.log(start_a + 1e-30)

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # chain‑rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact surrogate, mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 8.0,
    tol: float = 1e-10,
):
    """Mirror‑descent on the true max‑convolution surrogate."""
    n = len(start_a)
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum (Adam‑style) and step‑size schedule
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (works on the exact surrogate)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(
    start_a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
):
    """Very simple SA that moves a small amount of mass between two indices."""
    n = len(start_a)
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    cur_val = best_val
    start_time = time.time()
    init_temp = 1e-3 * best_val  # scale with current objective
    final_temp = 1e-7 * best_val

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = init_temp * (1.0 - elapsed / total) + final_temp * (elapsed / total)

        # propose a move: transfer a random fraction of mass from i to j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 0.0:
            continue
        # at most 20 % of the source mass is moved, scaled by a random factor
        delta = a[i] * rng.random() * 0.2
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective_surrogate(a_new)
        delta_obj = val - cur_val
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / (T + 1e-12)):
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines several optimisation phases (L‑BFGS, EG, SA) and
    respects the supplied wall‑clock budget.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin; the harness will cut us off if we overrun.
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Collect candidate seeds.
    # --------------------------------------------------------------
    seeds = []

    # 0) previous best, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape == (n,):
            seeds.append(prev_arr / prev_arr.sum())

    # 1) uniform (nice baseline)
    seeds.append(np.full(n, 1.0 / n))

    # 2) classic inverse‑sqrt shape
    seeds.append(_inverse_sqrt(n))

    # 3) a few random Dirichlet draws (very sparse)
    for conc in (0.001, 0.01, 0.1):
        seeds.append(_random_dirichlet(n, conc, rng))

    # 4) power‑law families
    for p in (0.3, 0.5, 0.8, 1.2):
        seeds.append(_power_law(n, p))

    # 5) exponential families (r close to 1)
    for r in (0.999, 0.9995, 0.9999):
        seeds.append(_exponential(n, r))

    # Normalise all seeds (protect against rounding issues) and drop empties
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # --------------------------------------------------------------
    # Quick surrogate evaluation – keep the top few seeds.
    # --------------------------------------------------------------
    evaluated = [(s, _objective_surrogate(s)) for s in seeds if time.time() < deadline]
    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:min(5, len(evaluated))]]

    print("\n--- Seed summary (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:len(top_seeds)], 1):
        print(f" Seed #{i}:  {v:.8f}")

    # --------------------------------------------------------------
    # Optimisation schedule for the smoothed max‑convolution.
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        5_000_000.0, 10_000_000.0, 50_000_000.0, 100_000_000.0,
    ]

    global_best_seq = None
    global_best_val = math.inf

    # --------------------------------------------------------------
    # Run the optimiser from each promising seed.
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() >= deadline:
            break
        # give each seed a fair share of the remaining budget
        remaining = deadline - time.time()
        share = max(5.0, remaining * 0.20)  # at least 5 s per seed
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx}/{len(top_seeds)}] allocated ~{share:.1f}s")

        # 1) smoothed L‑BFGS
        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=seed_vec,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=3500,
        )
        print(f"  L‑BFGS surrogate = {val_lbfgs:.8f}")

        # 2) exponentiated‑gradient polishing
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_lbfgs,
                deadline=sub_deadline,
                max_iters=600_000,
                eta0=6.0,
                tol=1e-12,
            )
            print(f"  EG polish surrogate = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick the better of the two phases
        cand_seq, cand_val = (seq_eg, val_eg) if val_eg < val_lbfgs else (seq_lbfgs, val_lbfgs)

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final stochastic refinement (simulated annealing) if time permits.
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Final SA polishing ---")
        sa_deadline = deadline
        global_best_seq, global_best_val = _simulated_annealing_opt(
            start_a=global_best_seq,
            deadline=sa_deadline,
            rng=rng,
        )
        print(f"   SA improved surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Compute the *real* score using the provided evaluator.
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("evaluate_sequence raised an exception:", exc)

    # Store the best sequence for possible reuse in later runs.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
