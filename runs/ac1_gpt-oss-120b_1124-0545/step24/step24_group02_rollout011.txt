Implementation of simplex projection:

def project_simplex(v):
    """Projection of vector v onto probability simplex (sum=1, v>=0)."""
    # Implementation based on algorithm from
    # https://arxiv.org/abs/1309.1541
    # returns w
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
step_function_optimizer – high‑precision search for a low‑peak auto‑convolution.

The optimiser works in three stages (all respecting the supplied wall‑clock
budget):
1️⃣  Initialise the candidate using the previous best, the classic inverse‑sqrt
    shape and a few random Dirichlet draws (all ℓ₁‑normalised).
2️⃣  Refine the shape with a *deep smoothed* L‑BFGS run.  The hard max in the
    true objective `2·n·max(conv(a,a))` is replaced by a smooth log‑sum‑exp
    approximation with a schedule of β values that slowly turns the problem
    into the original one.  The optimiser works on log‑softmax parameters,
    guaranteeing non‑negativity and sum‑to‑one automatically.
3️⃣  Polish the result with a long exponentiated‑gradient (mirror‑descent)
    routine that directly minimises the exact max‑convolution.  A simple
    momentum (Adam‑style) and a decaying learning‑rate `η₀/√t` are used.
    The best sequence seen throughout the run is kept.

Both stages are deliberately performed on the *probability simplex* (the
variables are interpreted as a probability distribution, which is admissible
because the evaluation function is homogeneous of degree 0).  The final
sequence is stored in the global variable ``height_sequence_1`` for possible
reuse.

The implementation uses only NumPy and SciPy (HiGHS LP solver is optional
but not required).  All helper functions are defined at module level – no
closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Global deadline used by the L‑BFGS callback (no closures)
# ----------------------------------------------------------------------
_global_deadline = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic starting families
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1) and ℓ₁‑normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw (very sparse for tiny concentration)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS surrogate optimiser (smooth max‑convolution)
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, n: int, beta: float):
    """
    Objective and gradient for the smoothed max‑convolution.
    Parameters
    ----------
    v : np.ndarray
        Log‑softmax parameters (length n).
    n : int
        Sequence length.
    beta : float
        Smoothing parameter.
    Returns
    -------
    (obj, grad) : tuple[float, np.ndarray]
        Objective value and gradient w.r.t. ``v``.
    """
    a = _softmax(v)                     # ℓ₁‑normalised, non‑negative
    conv = _conv_fft(a, a)              # full convolution, length 2·n‑1
    shift = conv.max()
    # stable soft‑max over the convolution values
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see paper for derivation)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)         # chain rule through soft‑max
    return obj, grad_v


def _lbfgs_callback(_):
    """Interrupt L‑BFGS if the global deadline has been passed."""
    if _global_deadline is not None and time.time() > _global_deadline:
        raise KeyboardInterrupt()


def _lbfgs_smooth_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule,
                      max_iter_per_beta: int = 3000):
    """
    Minimise the smoothed max‑convolution by L‑BFGS‑B.
    Returns the best ℓ₁‑normalised sequence seen and its exact surrogate.
    """
    global _global_deadline
    _global_deadline = deadline

    v = np.log(start_a + 1e-30)           # initialise log‑softmax parameters
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        try:
            res = _opt.minimize(_lbfgs_obj,
                                v,
                                args=(n, beta_val),
                                method="L-BFGS-B",
                                jac=True,
                                callback=_lbfgs_callback,
                                options={"maxiter": max_iter_per_beta,
                                         "ftol": 1e-14,
                                         "disp": False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    _global_deadline = None
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (EG) optimiser – exact objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 8.0,
                      tol: float = 1e-12):
    """
    Mirror‑descent (exponentiated gradient) directly on the exact max‑convolution.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set (indices whose value is within relative tolerance)
        active_mask = conv >= max_val - tol * max_val
        active_size = active_mask.sum()
        if active_size == 0:
            # safety fall‑back – treat the global maximiser as active
            active_mask = conv == max_val
            active_size = 1

        # sub‑gradient – average over all active indices (vectorised via FFT)
        w = active_mask.astype(float)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n] / active_size

        # Adam‑style momentum on the logarithmic parameters
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional housekeeping output (lightweight)
        if it % 100_000 == 0:
            print(f"    EG iter {it:,} – current surrogate = {cur_val:.8f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (optional helper)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a deep smoothed L‑BFGS optimisation followed by a
    long exponentiated‑gradient polishing phase, all inside the supplied
    wall‑clock budget.  The best sequence is stored in the global variable
    ``height_sequence_1`` and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it modest to stay inside the time budget
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and prev:
        n = len(prev)
        prev_arr = np.asarray(prev, dtype=np.float64)
        if np.sum(prev_arr) > 0.0:
            prev_norm = prev_arr / np.sum(prev_arr)
        else:
            prev_norm = None
    else:
        n = 4000
        prev_norm = None

    # ------------------------------------------------------------------
    # Build a short list of promising seeds (all ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    if prev_norm is not None:
        seeds.append(prev_norm)

    seeds.append(_inverse_sqrt(n))

    seeds.append(_random_dirichlet(n, conc=0.001, rng=rng))
    seeds.append(_random_dirichlet(n, conc=0.05, rng=rng))

    # Ensure proper normalisation and drop any degenerate vectors
    seeds = [s / s.sum() for s in seeds if s.sum() > 0.0]

    # ------------------------------------------------------------------
    # Quick surrogate evaluation of seeds – pick the best as starting point
    # ------------------------------------------------------------------
    best_seed = None
    best_val = math.inf
    for s in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        if val < best_val:
            best_val = val
            best_seed = s

    print(f"\n--- Initial seed ---\nSurrogate = {best_val:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – deep smoothed L‑BFGS (focus on reducing the max‑convolution)
    # ------------------------------------------------------------------
    seq_lbfgs = best_seed
    val_lbfgs = best_val
    if _opt is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 35 % of the remaining budget to L‑BFGS
        lbfgs_deadline = time.time() + remaining * 0.35

        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0, 10_000_000.0, 20_000_000.0,
            50_000_000.0, 100_000_000.0, 200_000_000.0,
            500_000_000.0, 1_000_000_000.0,
        ]

        seq_lbfgs, val_lbfgs = _lbfgs_smooth_opt(
            start_a=best_seed,
            n=n,
            deadline=lbfgs_deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=4000,
        )
        print(f"\n[After L‑BFGS] surrogate = {val_lbfgs:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient polishing on the exact objective
    # ------------------------------------------------------------------
    seq_eg, val_eg = seq_lbfgs, val_lbfgs
    if time.time() < deadline:
        seq_eg, val_eg = _exp_gradient_opt(
            start_a=seq_lbfgs,
            n=n,
            deadline=deadline,
            max_iters=2_000_000,
            eta0=12.0,          # a slightly larger initial step‑size
            tol=1e-12,
        )
        print(f"\n[After EG] surrogate = {val_eg:.8f}")

    # ------------------------------------------------------------------
    # Choose the best of the two phases
    # ------------------------------------------------------------------
    if val_eg < val_lbfgs:
        final_seq = seq_eg
        final_surrogate = val_eg
    else:
        final_seq = seq_lbfgs
        final_surrogate = val_lbfgs

    # ------------------------------------------------------------------
    # Final official evaluation (may raise, but we catch it)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {final_surrogate:.8f}, "
              f"true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
