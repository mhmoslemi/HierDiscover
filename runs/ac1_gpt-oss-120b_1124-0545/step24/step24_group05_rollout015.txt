"""
Step‑function optimizer for `evaluate_sequence`.

The main ideas
----------------
* **Richer seed family** – besides the classic arcsine we try
  discrete beta laws `p_i ∝ (i+½)^{a‑1}·(n‑½‑i)^{b‑1}` with several
  (a,b).  These include the arcsine (a=b=½) and many asymmetric shapes;
  a few random Dirichlet draws are added for diversity.

* **More aggressive smoothing** – the Adam optimiser works on a
  smooth log‑sum‑exp surrogate of the max‑convolution.
  The temperature schedule is extended to a larger final β (≈20) so
  that the surrogate tracks the true maximum more closely.

* **Much longer mirror‑descent polishing** – after the Adam phase we run
  the exponentiated‑gradient (mirror descent) pass for as many iterations
  as the remaining time permits (up to several‑million steps).  This
  stage directly minimises the exact max‑convolution and is the most
  effective way to shave the last few hundredths off the objective.

* **Time‑aware allocation** – the total budget (≈1000 s) is split:
  ~20 % for the quick Adam refinements on a handful of seeds and the
  rest for the polishing pass.  The loop stops early if the deadline is
  close or if a solution better than the target `1.503` is found.

The function returns the best sequence (list of non‑negative floats)
found before the deadline.  The global variable ``height_sequence_1``
stores the result for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑law: p_i ∝ (i+½)^{a‑1}·(n‑½‑i)^{b‑1}."""
    i = np.arange(n, dtype=np.float64)
    weight = np.power(i + 0.5, a - 1.0) * np.power(n - 0.5 - i, b - 1.0)
    # guard against NaNs when a or b < 1 (0**negative)
    weight[np.isnan(weight)] = 0.0
    s = weight.sum()
    if s == 0.0:
        return _uniform(n)
    return weight / s


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of  Fβ(a) = (1/β)·log Σ_k exp(β·conv(a,a)[k])
    w.r.t. log‑softmax parameters `log_a`.  Returned gradient is
    ∂F/∂log_a (same shape as log_a).
    """
    a = _softmax(log_a)                     # probability vector
    conv = _conv_fft(a, a)                  # length 2n‑1

    # softmax over the convolution entries
    scaled = beta * conv
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()                            # w sums to 1

    # g_a = 2 Σ_k w_k·shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)               # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]       # slice → length n

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 1.0,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution objective.
    Returns the best probability vector found (sum=1) and its exact
    surrogate value (2·n·max(conv)).
    """
    log_a = np.log(start_seq + 1e-30)   # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max‑convolution objective)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 20_000_000,
                     eta0: float = 15.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution
    objective.  The loop stops when the deadline is reached or when the
    iteration budget expires.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (indices that contribute to k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low value of
    `evaluate_sequence`.  The routine builds a diverse pool of seeds
    (arcsine, beta‑laws, exponentials, random Dirichlet, previous best),
    optimises each seed with a short Adam pass on a smoothly‑approximated
    max‑convolution objective (high final temperature), and finally
    performs a long exponentiated‑gradient polishing run.  The best
    sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin before the hard deadline
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Construct a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) classic arcsine
    candidates.append(_arcsine_distribution(n))

    # 2) beta‑law variations around the arcsine (a,b) ∈ {0.3,0.4,0.5,0.6,0.7,0.8}
    exponents = [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]
    for a in exponents:
        for b in exponents:
            if abs(a - b) < 1e-12:
                # the symmetric case is already represented (a=b=0.5)
                continue
            candidates.append(_beta_distribution(n, a, b))

    # 3) previous best, if any (allows continuation across runs)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(np.maximum(arr, 0.0) / np.maximum(arr.sum(), 1e-30))
        except Exception:
            pass

    # 4) uniform
    candidates.append(_uniform(n))

    # 5) a few exponentials
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    # 6) a handful of random Dirichlet draws (lightweight)
    for _ in range(5):
        candidates.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # ------------------------------------------------------------------
    # Quick Adam refinements (≈20 % of the time budget)
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    # allocate a fixed small budget for the Adam phase
    adam_deadline = start_time + 0.20 * budget_s   # 20 % of total time
    for idx, cand in enumerate(candidates, start=1):
        if time.time() > adam_deadline:
            print("[Info] Adam phase time exhausted.")
            break

        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        # limit the number of iterations – a few‐hundred‑thousand are enough
        max_iters = 150_000
        print(f"[Adam seed {idx}/{len(candidates)}] max_iters={max_iters}")

        seq, val = _adam_softmax_opt(cand, n, adam_deadline,
                                     max_iters=max_iters,
                                     lr0=1.0,
                                     beta_start=0.5,
                                     beta_end=20.0)

        print(f"   surrogate after Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   *** new global best (surrogate) = {best_overall_val:.6f}")

        # early exit if we already beat the target
        if best_overall_val <= 1.503:
            print("[Info] Target already reached – skipping remaining seeds.")
            break

    # ------------------------------------------------------------------
    # Polishing with exponentiated‑gradient (the heavy part)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _uniform(n)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        # use as many iterations as the remaining wall‑clock time permits
        polish_iters = 20_000_000          # an upper bound, the loop checks the deadline
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=polish_iters,
            eta0=15.0)

        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best surrogate unchanged)")

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
