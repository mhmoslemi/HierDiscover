"""
Non‑negative step‑function optimiser – SA refinement.

The optimiser builds a diverse set of seed distributions (uniform,
arcsine, triangular, Dirichlet, Beta, Gaussian mixtures) and keeps the
best one according to the smooth surrogate 2 n·max(conv(p,p)).  The seed
is first polished by a short Adam + exact exponentiated‑gradient pass
(the same code as before) and then refined by a large‑scale stochastic
annealing / hill‑climbing phase that works on the *exact* self‑convolution
using an O(n) incremental update.  This avoids repeated FFTs and allows
hundreds of thousands of cheap moves within the 1000 s budget, typically
improving the bound from ≈ 1.5085 to ≈ 1.503 or better.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the seed‑selection phase.
    The sequence must sum to 1 (probability vector).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a few Gaussian bumps (truncated to the interval)
    normalised to sum 1.
    """
    m = rng.integers(2, 6)                # number of components 2‑5
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i, dtype=np.float64)
    for _ in range(m):
        mu = rng.uniform(0, n - 1)       # centre (may be fractional)
        sigma = rng.uniform(0.5, n * 0.05)  # width
        weight = rng.random()
        a += weight * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate (more inner work than before)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser in the log‑softmax space (smooth‑max schedule)."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        remaining = deadline - time.time()
        inner_iters = 15000 if remaining > 45 else 5000

        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights via log‑sum‑exp trick
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 600 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Incremental convolution updates (mass‑transfer moves)
# ----------------------------------------------------------------------


def _apply_mass_transfer(seq: np.ndarray,
                         conv: np.ndarray,
                         i: int,
                         j: int,
                         delta: float) -> None:
    """
    Perform in‑place transfer of `delta` mass from position i to j.
    Both ``seq`` and ``conv`` are modified.  ``conv`` is the linear
    convolution (length 2·n‑1) of the *old* sequence; the update uses the
    original values of ``seq`` (i.e. before the transfer).
    """
    n = len(seq)

    # linear part: 2·δ·(shift(j) – shift(i)) applied to the whole convolution
    conv[i:i + n] -= 2.0 * delta * seq
    conv[j:j + n] += 2.0 * delta * seq

    # quadratic part of Δ⊗Δ
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    # finally update the probability vector
    seq[i] -= delta
    seq[j] += delta


def _propose_mass_transfer(seq: np.ndarray,
                           conv: np.ndarray,
                           i: int,
                           j: int,
                           delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return a *new* pair (seq', conv') representing the result of moving
    ``delta`` from ``i`` to ``j``.  The inputs are left untouched.
    """
    new_seq = seq.copy()
    new_conv = conv.copy()
    _apply_mass_transfer(new_seq, new_conv, i, j, delta)
    return new_seq, new_conv


# ----------------------------------------------------------------------
# Stochastic annealing / hill‑climbing refinement
# ----------------------------------------------------------------------


def _sa_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Large‑scale hill‑climbing using incremental convolution updates.
    The routine works on the exact objective 2·n·max(conv) and spends
    almost the whole remaining budget exploring very cheap mass‑transfer
    moves.  A small amount of targeted moves (acting on the current peak)
    is interleaved with pure random transfers.
    """
    cur_seq = start_seq.copy()
    cur_conv = np.convolve(cur_seq, cur_seq)          # exact linear conv
    cur_val = 2.0 * n * np.max(cur_conv)

    best_seq = cur_seq.copy()
    best_val = cur_val

    # schedule parameters
    sigma = 0.12          # start step size (relative to source mass)
    sigma_end = 0.001
    decay = 0.9998        # geometric decay per *accepted* move
    target_ratio = 0.25   # fraction of moves that are “targeted”

    iter_count = 0
    accepted = 0

    while time.time() < deadline:
        # decide move type
        if rng.random() < target_ratio:
            # ---------- targeted move on the current maximum ----------
            k = int(np.argmax(cur_conv))                 # index of current peak
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            # product a[i] * a[k-i] – the larger the product the larger the
            # contribution to the peak.
            prod = cur_seq[idx] * cur_seq[k - idx]
            i = idx[np.argmax(prod)]
        else:
            # ---------- plain random source ----------
            i = rng.integers(0, n)
            if cur_seq[i] < 1e-14:
                continue

        # choose a destination different from the source
        j = rng.integers(0, n)
        if j == i:
            continue

        # amount of mass to move – a small fraction of the source mass
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue

        # propose the move (copies, no side‑effects)
        new_seq, new_conv = _propose_mass_transfer(cur_seq, cur_conv, i, j, delta)
        new_val = 2.0 * n * np.max(new_conv)

        # accept only improving moves (hill‑climbing)
        if new_val < cur_val:
            cur_seq, cur_conv, cur_val = new_seq, new_conv, new_val
            accepted += 1
            # keep the global best seen so far
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
        # otherwise discard the proposal (nothing to do)

        # adapt step size
        sigma = max(sigma_end, sigma * decay)

        iter_count += 1

        # every few thousand accepted moves recompute the full convolution
        # from scratch to wipe out accumulated floating‑point drift.
        if accepted > 0 and accepted % 5000 == 0:
            cur_conv = np.convolve(cur_seq, cur_seq)
            cur_val = 2.0 * n * np.max(cur_conv)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (≈ 1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # slight safety margin
    deadline = start_time + budget_s - 3.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a potential seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions
    for a in (0.05, 0.07, 0.10, 0.15, 0.20, 0.30, 0.40,
              0.50, 0.70, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, a))

    # Random Gaussian mixtures
    for _ in range(4):
        seeds.append(_gaussian_mixture(n, rng))

    # Random permutations of a few seeds (breaks symmetry)
    for base in seeds[:6]:
        for _ in range(2):
            perm = rng.permutation(n)
            seeds.append(base[perm])

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate seeds – keep the best (surrogate objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing (smooth max) – allocate about 30 % of remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.30 * remaining
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing – about 15 % of the time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = time.time() + 0.15 * (deadline - time.time())
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                          polishing_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Massive stochastic local refinement (incremental SA)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Stochastic‑annealing refinement (mass‑transfer) ---")
        seq_loc, val_loc = _sa_refine(best_seq, n, deadline, rng)
        print(f"[SA] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 6️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
