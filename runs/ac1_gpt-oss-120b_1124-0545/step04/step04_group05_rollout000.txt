"""
propose_candidate – hybrid search (beta‑shape init → exact‑max sub‑gradient Adam
→ aggressive simulated‑annealing).  

1. Initialise a symmetric Beta‑type distribution a_i ∝ (i+½)^{α‑1}(n‑i‑½)^{α‑1}
   and sweep α∈[0.30,0.80] (step 0.02) to pick the best start‑point for the
   given length *n*.  This already beats the classic arcsine shape.

2. Stage 1 – exact‑max sub‑gradient descent.
   * Variables are kept on the simplex via a soft‑max re‑parameterisation
     a = softmax(v).
   * The convex (sub‑)gradient of the *true* max‑convolution is used:
     for the index k where (a∗a)_k attains the maximum,
     ∂/∂a_i max = 2·a_{k‑i} (zero when k‑i is out of bounds).
   * Adam is applied on the unconstrained v‑vectors with a linearly
     decreasing learning‑rate.
   * Whenever no improvement is seen for 15 s we jitter v slightly,
     re‑initialising the momentum – this helps escape stagnation.

3. Stage 2 – simulated‑annealing local refinement.
   * Random mass transfers between two coordinates (up to 10 % of the
     source mass) are proposed.
   * Proposals are accepted if they improve the exact objective or with
     Metropolis probability exp(−Δ/T) where the temperature T is linearly
     cooled.
   * The SA phase consumes the remaining budget.

4. The best candidate found is normalised (sum = 1) and returned as a
   plain Python list.  All numbers are non‑negative, so `evaluate_sequence`
   will accept the result.

The algorithm runs solely on the CPU, uses only NumPy (and the provided
`evaluate_sequence`), and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the linear convolution a*a (exact)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv.max()


def conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (real part)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def subgradient_max(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max_i (a*a)_i with respect to a.
    If the maximum is attained at several indices, the average of the
    corresponding sub‑gradients is returned.
    """
    n = a.size
    conv = conv_full(a, fft_len)
    max_val = conv.max()
    # tolerance to collect all (almost) maximal indices
    tol = max_val * (1.0 + 1e-12) - max_val
    max_idx = np.where(conv >= max_val - tol)[0]

    # accumulate gradients for each maximal index
    g = np.zeros_like(a)
    for k in max_idx:
        # g_i = 2 * a_{k-i} when 0 <= k-i < n, else 0
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]
    g /= max_idx.size  # average if multiple maxima
    return g


def beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on {0,…,n‑1}: (i+½)^{α‑1}(n‑i‑½)^{α‑1}."""
    i = np.arange(n, dtype=float)
    left = i + 0.5
    right = n - i - 0.5
    a = np.power(left * right, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def initialise_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Choose the best Beta‑shape after a quick sweep, or reuse a previously
    stored good sequence.
    """
    # 1) reuse previous successful sequence if it exists and matches length
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.size == n and prev.sum() > 0.0:
            prev = prev / prev.sum()
            return prev

    # 2) quick grid search over α ∈ [0.30,0.80] with step 0.02
    best_score = np.inf
    best_a = None
    fft_len = 1 << ((3 * n - 2).bit_length())
    for alpha in np.arange(0.30, 0.81, 0.02):
        a = beta_shape(n, alpha)
        # direct objective for a probability vector: 2·n·max(conv)
        maxc = conv_max(a, fft_len)
        score = 2.0 * n * maxc
        if score < best_score:
            best_score = score
            best_a = a
    # 3) final small random perturbation to break symmetry a bit
    if best_a is None:
        raise RuntimeError("Failed to initialise a.")
    best_a += 1e-12 * rng.random(n)  # tiny jitter
    best_a = best_a / best_a.sum()
    return best_a


def exact_score_from_max(n: int, conv_max_val: float) -> float:
    """Exact objective when sum(a)=1."""
    return float(2.0 * n * conv_max_val)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine follows the hybrid scheme described
    in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # safety margin

    # Length of the sequence – default 2000, can be overridden via kwargs
    n = int(kwargs.get("n", 2000))

    # pre‑compute FFT size (next power of two ≥ 3·n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 0) initialise a (probability vector)
    # ------------------------------------------------------------------
    a = initialise_a(n, rng)
    v = np.log(a + 1e-12)          # log‑parameters for soft‑max

    # ------------------------------------------------------------------
    # 1) Stage 1 – exact‑max sub‑gradient Adam descent
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # learning‑rate schedule: start relatively high, decay to a small value
    lr_start, lr_end = 0.15, 0.001

    best_a = a.copy()
    conv = conv_full(a, fft_len)
    best_max = conv.max()
    best_score = exact_score_from_max(n, best_max)
    last_improve = start_time

    # allocate ~70 % of the total budget for this stage
    stage1_deadline = start_time + 0.70 * budget_s
    it = 0
    while time.time() < stage1_deadline:
        # progress fraction for schedules
        p = (time.time() - start_time) / (stage1_deadline - start_time)
        lr = lr_start * (1.0 - p) + lr_end * p

        # current point on simplex
        a = softmax(v)

        # compute convolution and its maximum
        conv = conv_full(a, fft_len)
        cur_max = conv.max()
        cur_score = exact_score_from_max(n, cur_max)

        # sub‑gradient of the exact max
        g_a = subgradient_max(a, fft_len)          # shape (n,)

        # chain rule through softmax (v → a)
        dot_ag = np.dot(a, g_a)
        grad_v = a * (g_a - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # keep best seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_max = cur_max
            last_improve = time.time()
            print(f"[{it:07d}] stage‑1 new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

        # occasional jitter if stagnated > 15 s
        if time.time() - last_improve > 15.0:
            # re‑center v around the current best and add tiny noise
            v = np.log(best_a + 1e-12) + rng.normal(scale=1e-3, size=n)
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[{it:07d}] restart after stagnation (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 2) Stage 2 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    remaining_time = overall_deadline - time.time()
    if remaining_time > 5.0:
        a = best_a.copy()
        best_conv_max = best_max
        # SA temperature schedule (linear cooling)
        T_start = best_score * 0.004          # 0.4 % of current score
        T_end = best_score * 1e-6

        sa_it = 0
        sa_start = time.time()
        while time.time() < overall_deadline:
            # update temperature
            phase_elapsed = time.time() - sa_start
            phase_frac = min(1.0, phase_elapsed / remaining_time)
            T = T_start * (1.0 - phase_frac) + T_end * phase_frac

            # propose a random mass transfer between two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move a fraction (up to 10 %) of a[i] to a[j]
            max_move = min(a[i], 0.10 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate exact objective for the proposal
            new_max = conv_max(a_new, fft_len)
            new_score = exact_score_from_max(n, new_max)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} "
                          f"(t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] total time={elapsed:.1f}s, best score={best_score:.6f}, length={n}")

    # expose the best sequence for possible warm‑starts in future calls
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
