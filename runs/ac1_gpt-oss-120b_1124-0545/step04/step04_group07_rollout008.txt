"""
Propose a low‑value step‑function.

Algorithm
---------
1.  **Scaled Cantor initialisation** –  For a grid of scaling
    factors `a ∈ [0.30, 0.38]` the (deterministic) Cantor‑type
    measure with `2**L` atoms (`L=18`) is generated, binned into the
    required number of bins `n` and normalised to total mass 1.
    The best scaling factor (lowest `evaluate_sequence` value) is kept
    as a starting point.  A random Dirichlet start is also tried as a
    fall‑back.

2.  **Active‑set sub‑gradient descent** –  In each iteration we
    compute the full convolution `c = a*a`.  All indices where `c`
    nearly attains its maximum (within a tiny tolerance) are taken as
    the *active set*.  The sub‑gradient of `max(c)` is the (negative)
    average of the gradients of these active constraints, which can be
    obtained by a single FFT‑based convolution of the reversed vector
    with a sparse weight vector.  A Nesterov‑style momentum term and a
    diminishing step‑size (`step ∝ 1/√t`) drive the update.  After each
    update we project onto the probability simplex (clip to ≥0 and
    renormalise).

3.  **Random jitter** –  Every few thousand iterations a small
    uniformly‑distributed jitter is added to the vector to help escape
    shallow plateaus.

4.  The best sequence seen (lowest objective) is returned.  The whole
    search respects the 1000 s budget.

The method differs from the previous smooth‑max descent by using a
hard active‑set weighting (sub‑gradient of the true max) and an
exhaustive low‑dimensional search over Cantor scaling factors, which
typically yields a lower final bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT.
    ``size`` must be ≥ len(u)+len(v)-1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size)


def _scaled_cantor_histogram(n: int, L: int, a_scale: float) -> np.ndarray:
    """
    Length‑`n` histogram approximating the self‑similar Cantor‑type
    measure defined by the scaling factor ``a_scale`` (0 < a_scale < 0.5).

    The construction uses ``2**L`` equally weighted atoms whose positions
    are   sum_{k=0}^{L-1} b_k·(1‑a)·a**k   with b_k∈{0,1}.

    The returned vector is normalised to sum = 1.
    """
    N = 1 << L                     # number of atoms (2**L)
    idx = np.arange(N, dtype=np.uint64)[:, None]   # (N,1)
    # binary digits of the atom index (0/1)
    bits = ((idx >> np.arange(L, dtype=np.uint64)) & 1).astype(np.float64)  # (N, L)

    # weights for each digit
    weights = (1.0 - a_scale) * (a_scale ** np.arange(L, dtype=np.float64))

    # positions of the atoms in [0,1)
    pos = bits @ weights           # (N,)

    # histogram on the required grid
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    seq = hist.astype(np.float64) / float(N)   # normalise → sum == 1
    return seq

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # 1️⃣  Scan a grid of Cantor scaling factors for a good start
    # --------------------------------------------------------------
    L = 18                                     # number of binary digits → 2**L atoms
    a_grid = np.linspace(0.30, 0.38, 81)       # coarse grid, step ≈ 0.001
    best_init_score = np.inf
    best_init_seq = None
    best_a = None

    # pre‑allocate FFT size for the forthcoming convolutions
    size1 = _next_pow2(2 * n - 1)               # for a*a
    size2 = _next_pow2(3 * n - 2)               # for a_rev * w

    for a_scale in a_grid:

        # full convolution (linear)
        conv_full = _full_convolution(seq, size1)
        max_b = conv_full.max()
        score = 2.0 * n * max_b               # sum(seq)=1 → exact objective
        if score < best_init_score:
            best_init_score = score
            best_init_seq = seq
            best_a = a_scale

    conv_full = _full_convolution(rand_seq, size1)

    # --------------------------------------------------------------
    while time.time() < deadline:

        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # keep the best ever seen

            best_score = cur_score

            last_improve = time.time()

        # tolerance relative to the maximum – start loose and tighten
        tol_rel = 1e-12

        # ----------------------------------------------------------
        g_full = _conv_fft(a_rev, w, size2)

        g = g_full[n - 1 : n - 1 + n]
