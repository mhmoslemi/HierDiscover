def random_swap(a, scale):
    # pick i (source) and j,k (dest) distinct; transfer some mass from i to j and k.
    n = len(a)

def adam_optimize(x, lam, n, steps, lr):
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, steps+1):
        phi, grad = compute_phi_and_grad(x, lam, n)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat)+eps)
    return x

def adam_optimize(x, lam, n, steps, lr, deadline):
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, steps+1):
        if time.time() >= deadline:
            break
        phi, grad = compute_phi_and_grad(x, lam, n)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat/(np.sqrt(v_hat)+eps)
        x = x - step
    return x

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Analytic grid search** – sweep several parametric families
   (product‑law, sum‑law, exponential, Gaussian) over a fine grid of
   parameters.  The best candidate becomes the starting point.

2. **Smooth‑max optimisation** – treat the objective as a smooth
   approximation Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).  Optimise the
   unconstrained log‑probability vector `x` (a = softmax(x)) with an
   Adam optimiser while gradually increasing λ up to a very large
   value (≈8·10⁴).  The Adam loop uses only NumPy/FFT, no external
   optimiser is required.

3. **Exact sub‑gradient descent** – after the smoothing stages the
   true objective `max(conv)` is tackled directly.  The sub‑gradient
   of `max(conv(a,a))` is obtained by an FFT‑based convolution with a
   binary mask of the active (maximal) lag indices; a projected
   gradient step on the probability simplex follows.

4. **Light random‑mass‑transfer local search** – occasional stochastic
   moves (transfer a tiny amount of probability mass from one index
   to another) explore the neighbourhood of the current best point.
   The step size is adapted according to recent improvements.

All phases are time‑boxed by the supplied `budget_s` (default 1000 s);
the best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is ignored,
    the function normalises internally).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

# Analytic families

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^(‑p)   (edge‑heavy shape)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^(‑p) + (n‑1‑i+off)^(‑p)   (two tails)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exp_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·|i‑center|)   (center‑peaked shape)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - center))
    a = a / a.sum()
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(‑0.5·((i‑center)/σ)²)   (smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a

def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given log‑probability vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # soft‑max over convolution entries
    shift = conv - max_c                   # non‑positive
    exp_term = np.exp(lam * shift)        # ≤ 1
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam    # Φ_λ

    w = exp_term / sum_exp                 # (2n‑1) weights, sum to 1

    # gradient of Φ_λ w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]      # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # map gradient to x‑space using the Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


def adam_optimize(x: np.ndarray, lam: float, n: int, steps: int,
                  lr: float, deadline: float) -> np.ndarray:
    """Simple Adam optimiser for the smooth max objective."""
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        _, grad = compute_phi_and_grad(x, lam, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
    return x


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max(convolution))
# ----------------------------------------------------------------------


def subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact objective
    max(conv(a,a)).  Uses an FFT‑based computation of the sub‑gradient.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    max_val = np.max(conv)
    active = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
    cnt = active.sum()
    if cnt == 0:
        return a
    w = active / cnt                                         # uniform over active indices
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]            # sub‑gradient w.r.t. a
    a_new = a - step * grad
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the supplied evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search – a solid deterministic start
    # ------------------------------------------------------------------
    best_a = np.ones(n) / n
    best_score = compute_score(best_a)

    # possibly reuse a previously discovered construction
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n and ref.sum() > 0:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_a = sc, ref.copy()

    # families to explore
    p_vals = np.linspace(0.30, 0.70, 81)           # fine grid for power‑laws
    offset_vals = np.linspace(0.0001, 2.0, 81)     # avoid literal zero to keep finite values
    c_vals = np.linspace(0.0005, 0.0200, 41)      # exponential decay
    sigma_vals = np.linspace(100.0, 800.0, 41)    # Gaussian width

    # product‑law and sum‑law families
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offset_vals:
                if time.time() >= deadline:
                    break
                if mode == "product":
                    cand = analytic_product(n, p, off)
                else:
                    cand = analytic_sum(n, p, off)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # exponential decay (center‑peaked) – kept for completeness
    for c in c_vals:
        if time.time() >= deadline:
            break
        cand = analytic_exp_decay(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # Gaussian bump family
    for sigma in sigma_vals:
        if time.time() >= deadline:
            break
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    print(f"[grid] best score after analytic sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max optimisation with Adam (continuation on λ)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0,
                2000.0, 8000.0, 20000.0, 80000.0]

    for lam in lam_vals:
        if time.time() >= deadline:
            break
        # step budget adapts to the remaining time (rough estimate:
        # ~0.002 s per Adam iteration for n≈2000)
        remaining = deadline - time.time()
        max_possible = int(remaining / 0.002)
        # a modest default; may be reduced by the time estimate
        steps = 3000
        if lam <= 10:
            steps = 1500
        if lam > 2000:
            steps = 5000
        steps = min(steps, max_possible)
        if steps <= 0:
            break

        x = adam_optimize(x, lam, n, steps, lr=0.05, deadline=deadline)
        a = softmax(x)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[adam λ={lam:.1f}] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient descent on max(conv)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        # step size decreases as 1/√t (standard for sub‑gradient)
        step = 0.5 / np.sqrt(it)
        a = subgrad_step(a, step)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[subgrad iter={it}] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02  # relative amount moved per trial
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[random] new best = {best_score:.6f} "
                  f"(t={time.time() - start_time:.1f}s)")
            a = cand
            scale = max(0.001, scale * 0.9)        # tighten after improvement
        else:
            # slowly relax the step size to keep exploring
            scale = min(0.05, scale * 1.001)
            a = cand

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # return plain Python list as required by the external harness
    return best_a.tolist()
