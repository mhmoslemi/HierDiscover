"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The method consists of three phases:

1. **Initialisation** – use the best known global sequence if it exists,
   otherwise start from a simple analytic “U‑shaped” family (product of
   distances to the ends) and a tiny grid search around plausible
   parameters.

2. **Smoothed‑max Adam optimisation** – minimise a smooth approximation
   of the max‑convolution:
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k) .
   The gradient of f_α can be computed in O(n log n) by a pair of FFT‑based
   convolutions.  An Adam optimiser with a slowly increasing α (from 1 up
   to ≈100) drives the iterates toward a local minimum of the true
   objective while the simplex projection keeps the vector non‑negative
   and normalised (Σa = 1).

3. **Final stochastic refinement** – small Gaussian perturbations around
   the best vector found are accepted when they improve the exact score.

All heavy work is done with FFTs (power‑of‑two padding) so the routine
runs comfortably inside the 1000 s budget for n≈2000–4000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is assumed to lie on the simplex (Σa = 1); a safety
    normalisation is performed nevertheless.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum of power‑laws from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tails: a_i ∝ exp(-lam·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine power family: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped family: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    using FFTs.  Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = conv_fft(a)                         # length 2n‑1
    # soft‑max weights (numerically stable)
    maxc = conv.max()
    exp_alpha = np.exp(alpha * (conv - maxc))
    w = exp_alpha / exp_alpha.sum()           # Σ w = 1

    # gradient = 2·(rev(a) ∗ w)[n‑1 : n‑1+n]
    rev_a = a[::-1]
    g_full = conv_fft_general(rev_a, w)       # length 3n‑2
    start = n - 1
    g = 2.0 * g_full[start:start + n]
    return g

def propose_candidate(seed: int = 42,

    Search for a step‑function (list of non‑negative floats) that minimises
    the given evaluation function.  The algorithm combines a quick analytic
    initialisation, a smoothed‑max Adam optimisation with a gradually
    increasing temperature (α), and a final stochastic refinement.
    """

                sc = compute_score(cand)

        a = np.clip(a, 0.0, None)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[{step:07d}] new best = {best_score:.6f} (α={alpha:.2f})")

        if time.time() - last_improve > 30.0:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{step:07d}] jitter restart")

        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total:.1f}s")
    return best_seq.tolist()
