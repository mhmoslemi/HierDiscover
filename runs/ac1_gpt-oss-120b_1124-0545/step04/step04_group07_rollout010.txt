assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the discrete linear convolution of the step‑function heights.
All numbers are projected onto the simplex (\(\sum_i a_i = 1\)) – this makes the
objective simply `2*n*max(conv)`.

The optimiser works in three stages:

1. **High‑quality start** – a histogram of the middle‑third Cantor measure
   (`L=18` atoms) is binned into `n` cells and normalised.
2. **Smooth‑max gradient descent** – a differentiable soft‑max
   approximation of the maximum is minimised with Nesterov momentum.
3. **Sub‑gradient descent** – the exact sub‑gradient at the index of the
   current maximum (`kmax = argmax(conv)`) is used for a fast
   additive / projected update.
4. **Final LP polishing** – a tiny linear programme (only the 20 largest
   convolution entries) is solved with `cvxpy` to obtain a locally optimal
   distribution.  The LP is inexpensive (≈ 20 × n constraints) and often
   yields a small extra drop in the objective.

All updates keep the vector non‑negative and re‑normalise it, guaranteeing
that `evaluate_sequence` will never reject the candidate.  
The whole routine respects a 1000 s budget; a safety margin of 0.5 s is kept
to avoid forced termination.

```python
"""
Propose a low‑value step‑function sequence.

The algorithm consists of four phases:
1. Cantor‑measure histogram (L=18) → feasible starting point.
2. Smooth‑max gradient descent with Nesterov momentum (fast early improvement).
3. Exact sub‑gradient descent on the current maximal convolution entry.
4. Small linear‑program polishing on the 20 largest convolution entries.

All iterates are projected onto the simplex (non‑negative, sum = 1), and the
best sequence seen during the run is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT (output length = len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a (length 2*n‑1)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 18) -> np.ndarray:
    """
    Histogram of the (1/3) Cantor measure on `n` bins.
    Uses 2**L atoms (L=18 → 262 144 atoms), gives a sum‑to‑1 vector.
    """
    N = 1 << L                                 # number of atoms
    # ternary digits of the atoms
    powers = 3.0 ** np.arange(1, L + 1)        # (L,)
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    # position of each atom in [0,1)
    pos = (bits * (2.0 / powers)).sum(axis=1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)    # normalise → sum == 1
    return a


def _smoothmax_step(
    a: np.ndarray,
    a_rev: np.ndarray,
    conv_full: np.ndarray,
    size2: int,
    step: float,
    tau: float,
) -> np.ndarray:
    """
    One projected gradient step for the smooth‑max of a*a.
    Implements the soft‑max weighting described in the reference code.
    """
    max_b = conv_full.max()
    # soft‑max weights (numerically stable)
    w = np.exp((conv_full - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient = -2·(a_rev ∗ w)   (central n entries)
    g_full = _conv_fft(a_rev, w, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]

    d = -2.0 * g               # descent direction
    d -= d.mean()              # keep Σd = 0  (mass unchanged)

    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _subgradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a*a) at the index k_max = argmax(conv).
    For i with 0 ≤ i ≤ n‑1 and 0 ≤ k_max-i ≤ n‑1:

        ∂(a*a)_kmax / ∂a_i = a_{kmax-i} + a_i

    All other components are zero.
    """
    n = a.shape[0]
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g = np.zeros_like(a)
    # a_{kmax-i} term
    g[idx] = a[k_max - idx] + a[idx]   # a_i term added in the same line
    return g


def _lp_polish(a: np.ndarray, n: int, max_b: float, size1: int, time_left: float):
    """
    Solve a tiny linear programme on the 20 largest convolution entries.
    Returns (new_a, new_max) if a strict improvement is found, otherwise (None,None).
    """
    try:
        import cvxpy as cp
    except Exception:            # cvxpy not available
        return None, None

    # enough time for a quick LP?
    if time_left < 5.0:
        return None, None

    conv_full = _full_convolution(a, size1)
    top_k = 20
    # indices of the largest entries (including ties)
    idxs = np.argpartition(conv_full, -top_k)[-top_k:]

    rows = len(idxs)
    A = np.zeros((rows, n))
    for r, k_val in enumerate(idxs):
        i_start = max(0, k_val - (n - 1))
        i_end = min(k_val, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        A[r, idx_range] = a[k_val - idx_range]  # weight a_i in (f*g)_k

    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [A @ g <= t, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(t), constraints)

    # a modest iteration limit keeps the LP cheap
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, feastol=1e-9, reltol=1e-9)
    except Exception:
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    t_opt = t.value
    g_opt = g.value
    if t_opt is None or g_opt is None:
        return None, None

    # Accept only a strict improvement
    if t_opt < max_b - 1e-12:
        return g_opt, t_opt
    return None, None

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    a = _cantor_histogram(n, L=18)            # already sums to 1

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b               # sum(a)==1 → same as evaluate_sequence
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Hyper‑parameters
    # phase 1 (smooth‑max)
    base_step = 0.9
    tau_factor = 0.6
    tau_decay = 0.9997
    momentum = 0.85
    v = np.zeros_like(a)

    # phase 2 (sub‑gradient)
    sub_base_step = 0.5
    sub_momentum = 0.9
    v_sub = np.zeros_like(a)

    it = 0
    # Phase cut‑offs (proportional to the total budget)
    phase1_end = start + 0.55 * budget_s
    phase2_end = start + 0.85 * budget_s

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:

        # Full convolution (FFT)
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            elapsed = now - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # --------------------------------------------------------------
        # Phase selection
        # --------------------------------------------------------------
        if now < phase1_end:
            # ----------- smooth‑max gradient descent ----------
            tau = max(tau_factor * max_b, 1e-12)
            tau_factor *= tau_decay
            step = base_step / np.sqrt(it + 1.0)

            a = _smoothmax_step(a, a_rev, conv_full, size2, step, tau)

            # momentum is already inside the smooth‑max step (via v)
            a_rev = a[::-1].copy()

        elif now < phase2_end:

            k_max = int(np.argmax(conv_full))
            g = _subgradient_at_max(a, k_max)

            # Nesterov‑style momentum on the sub‑gradient
            v_sub = sub_momentum * v_sub + (1.0 - sub_momentum) * g
            step = sub_base_step / np.sqrt(it + 1.0)

            a = a - step * v_sub
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.ones_like(a) / n
            else:
                a /= s

            # occasional random jitter to escape shallow plateaus
            if np.random.rand() < 0.001:
                jitter = 0.02 * np.random.randn(n) * a
                a = a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            # Continue a few smooth‑max steps but add random perturbations
            tau = max(tau_factor * max_b, 1e-12)
            tau_factor *= tau_decay
            step = base_step / np.sqrt(it + 1.0)

            a = _smoothmax_step(a, a_rev, conv_full, size2, step, tau)
            a_rev = a[::-1].copy()

            if np.random.rand() < 0.002:
```
