def build_active_matrix(a, active_indices):
    # shape: len(active_indices) x n
    rows = []
    for k in active_indices:
        # shift a by k: a_shifted[i] = a[k-i] if 0 <= k-i < n else 0

def refine(a, n, time_limit):

"""
Propose a sequence of non‑negative heights that (approximately) minimizes

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in two phases:

1.  Initialise from the historic best (or a smooth sine shape) and normalise
    to a probability vector (sum = 1).  For a probability vector the
    evaluation reduces to `2·n·max(conv)`, so we just try to lower the
    maximum entry of the self‑convolution.

2.  Repeatedly improve the current vector by solving a tiny linear programme.
    Let ``M = max(conv(a,a))`` and let ``K`` be the (small) set of indices
    where the convolution is within a tiny tolerance of ``M``.
    The LP maximises the sum of a new vector ``g`` under the linear constraints

        (a * g)[k] ≤ M   for k in K,   g ≥ 0

    This is exactly the “active‑set” step described in the statement.
    After solving, ``g`` is renormalised to sum = 1 and mixed with the
    current vector:  a ← (1−t)·a + t·g .  The mixing coefficient ``t`` is
    obtained by a quick back‑tracking line search; the new vector is kept
    only if it strictly reduces the maximum convolution value.

    If no improvement is seen for a while we jitter the current vector to
    escape stagnation.  The whole refinement runs until the time budget is
    exhausted (or no further progress can be made).  Multiple restarts
    (based on the best vector seen so far) are performed if time permits.

The implementation uses NumPy for fast FFT‑based convolutions and SciPy’s
`linprog` (HiGHS) to solve the tiny LPs.  All constraints are built only
for the active set, so each LP is tiny (≤ 30 constraints) and solved
quickly.  The returned list is a plain Python list of floats; it is
guaranteed to be non‑negative and to respect the time budget.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the (real) self‑convolution a*a of length 2·len(a)‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def _active_constraint_matrix(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|K| × n) for the active constraints
        (a * g)[k] ≤ M ,   k ∈ K .
    Each row k contains the shifted copy of ``a`` that appears in the
    convolution at position k.
    """
    n = a.size
    rows = []
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        row = np.zeros(n, dtype=float)
        row[i] = a[k - i]         # a_{k-i}
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        return np.empty((0, n), dtype=float)


def _solve_lp_max_sum(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximize   1ᵀg   subject to   (a * g)[k] ≤ M for k∈K ,   g ≥ 0 .
    Returns the optimal g (normalised to sum = 1) or ``None`` on failure.
    """
    if active_idx.size == 0:
        return None

    A_mat = _active_constraint_matrix(a, active_idx)
    b_vec = np.full(active_idx.size, M, dtype=float)

    # linprog solves a minimisation problem → minimise –1ᵀg
    c = -np.ones(a.size, dtype=float)
    bounds = [(0.0, None)] * a.size

    # use the HiGHS solver; the matrix is tiny (≤ 30 × n)
    res = opt.linprog(
        c,
        A_ub=A_mat,
        b_ub=b_vec,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    return g0 / s   # normalise to sum = 1


def _mix_and_improve(
    a: np.ndarray,
    g: np.ndarray,
    cur_M: float,
    fft_len: int,
    tol: float = 1e-15,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple back‑tracking line search for a mixing coefficient ``t``
    (starting from 1.0 and halving) and keep the best feasible mixture that
    strictly reduces the maximal convolution value.
    Returns the new vector and its new max‑convolution.
    """
    best_a = a
    best_M = cur_M

    # try a few geometrically decreasing steps
    for t in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
        a_new = (1.0 - t) * a + t * g
        # numerical safety: clip negatives and re‑normalise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s
        conv_new = _convolve_self(a_new, fft_len)
        M_new = conv_new.max()
        if M_new < best_M - tol:
            best_a = a_new
            best_M = M_new
            break
    return best_a, best_M


def _refine_once(a_start: np.ndarray, n: int, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Perform the active‑set LP refinement starting from ``a_start``.
    The routine runs until ``time_limit`` seconds have elapsed or no
    improvement is possible.  Returns the best vector found and its score
    (2·n·max(conv)).
    """
    start = time.time()
    deadline = start + time_limit
    fft_len = _next_pow2(2 * n - 1)

    a = a_start.copy()
    conv = _convolve_self(a, fft_len)
    cur_M = conv.max()
    best_a = a.copy()
    best_M = cur_M
    best_score = 2.0 * n * best_M

    # bookkeeping for stagnation detection
    last_improve = start
    it = 0
    max_active = 30  # cap on number of active constraints

    while time.time() < deadline:
        it += 1
        conv = _convolve_self(a, fft_len)
        cur_M = conv.max()
        eps = cur_M * 1e-12 + 1e-14
        # indices where the convolution is (almost) maximal
        active_idx = np.where(conv >= cur_M - eps)[0]
        if active_idx.size > max_active:
            # keep only the largest values – they are already sorted by index,
            # so we take the first ``max_active`` entries (any subset works).
            active_idx = active_idx[:max_active]

        g = _solve_lp_max_sum(a, cur_M, active_idx)
        if g is None:
            # LP failed – give up this refinement run
            break

        a_new, M_new = _mix_and_improve(a, g, cur_M, fft_len)
        if M_new < cur_M - 1e-12:
            a = a_new
            cur_M = M_new
            cur_score = 2.0 * n * cur_M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_M = cur_M
                best_a = a.copy()
                last_improve = time.time()
                print(f"[iter {it:06d}] new best M={best_M:.8e}  score={best_score:.6f}")
        else:
            # stagnation → jitter the current iterate
            if time.time() - last_improve > 30.0:
                rng = np.random.default_rng(int(time.time() * 1e6) & 0xFFFFFFFF)
                jitter = rng.normal(scale=0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()
                last_improve = time.time()
                print(f"[iter {it:06d}] jitter restart")
            else:
                # nothing improved and not yet time to jitter – stop
                break

    return best_a, best_score


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects a time budget (default 1000 s) and returns a plain
    ``list[float]``.  It starts from the historic best (if available) or from a
    smooth sine shape, then carries out several active‑set LP refinements
    with occasional jitter restarts.
    """
    rng_global = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # tiny safety margin

    # ------------------------------------------------------------------
    # 0) Choose sequence length
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Initialise a reasonable starting vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size), mode="constant")
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=float)
        else:
            a0 /= s
    else:
        # smooth sine shape – positive on the whole interval
        i = np.arange(n, dtype=float)
        a0 = np.sin(np.pi * (i + 0.5) / n)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # 2) Run several refinement attempts (restarts) within the budget
    # ------------------------------------------------------------------
    global_best_a = a0.copy()
    global_best_score = float("inf")
    attempts = 4                       # a few independent starts
    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        # allocate a fraction of the remaining time to this attempt
        alloc = time_left * 0.9 / (attempts - attempt)

        if attempt == 0:
            a_start = global_best_a.copy()
        else:
            # jitter the current global best to obtain a fresh start
            jitter = rng_global.normal(scale=0.02, size=n)
            a_start = np.maximum(global_best_a + jitter, 0.0)
            a_start_sum = a_start.sum()
            if a_start_sum == 0.0:
                a_start = np.full(n, 1.0 / n, dtype=float)
            else:
                a_start /= a_start_sum

        a_opt, score_opt = _refine_once(a_start, n, alloc)

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {attempt}] new global best score={global_best_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 3) Final safety: fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if global_best_a is None or global_best_a.size == 0:
        global_best_a = np.full(n, 1.0 / n, dtype=float)

    # ensure correct format for the caller
    global_best_a = np.maximum(global_best_a, 0.0)

    total_time = time.time() - start_time
    print(f"[done] best score={global_best_score:.6f}, total time={total_time:.1f}s")
    return global_best_a.tolist()
