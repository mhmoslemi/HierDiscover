"""
Propose a non‑negative step‑function that (hopefully) beats the 1.503 bound.

The algorithm consists of three phases:

1. **Broad analytic sweep** – for several lengths (n) we generate many
   families of candidates (power‑law “product”/“sum”, simple windows,
   triangular, uniform).  The search uses a fairly fine grid on the
   exponent `p` and the offset parameter.  For each candidate we
   evaluate the exact objective
       2·n·max(conv(a,a))   (with ∑a = 1)
   and keep the best one.

2. **Smooth‑max L‑BFGS refinement** – the best analytic distribution is
   re‑parameterised with a soft‑max (`a_i = exp(x_i)/∑exp(x)`).
   A continuation on the smoothing parameter λ (0.5 → 5000) is run
   using SciPy’s L‑BFGS‑B optimiser on the smooth approximation
   Φ_λ(a) = (1/λ)·log ∑_k exp(λ·(a*a)_k).  After each stage the true
   objective is evaluated and the best vector is stored.

3. **Light stochastic local search** – a few thousand tiny mass‑transfer
   moves (`a_i ← a_i−δ, a_j ← a_j+δ`) are tried; every improvement
   updates the incumbent.  The step size is adapted on‑the‑fly.

All work respects the 1000 s time budget; the routine prints progress
and finally returns the best sequence found (as a list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector a (∑a = 1):
        2·n·max(conv(a,a))
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of probability mass from a random source index
    to a random destination index.  The total sum stays exactly 1.
    """
    n = a.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    # keep delta ≤ scale·a[i] so a[i] stays non‑negative
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def product_distribution(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n-1-i+off))^{‑p}  (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= total
    return a


def sum_distribution(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ (i+off)^{‑p} + (n-1-i+off)^{‑p}  (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= total
    return a


def triangular_distribution(n: int) -> np.ndarray:
    """a_i ∝ (i+1)*(n-i)  (normalised)."""
    i = np.arange(n, dtype=float)
    a = (i + 1) * (n - i)
    a /= a.sum()
    return a


def uniform_distribution(n: int) -> np.ndarray:
    return np.ones(n, dtype=float) / n


def hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def kaiser_window(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window using scipy.special.i0; falls back to uniform."""
    try:
        from scipy.special import i0

        i = np.arange(n, dtype=float)
        alpha = (n - 1) / 2.0
        arg = beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)
        w = i0(arg) / i0(beta)
        return w / w.sum()
    except Exception:
        return uniform_distribution(n)


def slepian_window(n: int, NW: float = 2.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, if scipy is present."""
    try:
        from scipy.signal import windows

        w = windows.dpss(n, NW=NW, Kmax=1, sym=False)[0]
        w = np.abs(w)  # ensure non‑negative
        return w / w.sum()
    except Exception:
        return uniform_distribution(n)


# ----------------------------------------------------------------------
# Smoothed max utilities (for L‑BFGS)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smoothed max Φ_λ for a = softmax(x) together with its gradient w.r.t.
    the unconstrained vector x.
    """
    # soft‑max
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)            # length 2n‑1
    max_c = np.max(conv)

    # smooth max (log‑sum‑exp)
    shift = conv - max_c                # for numerical stability
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp              # weights (sum to 1)

    # gradient of Φ_λ w.r.t. a:
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    # slice the part that corresponds to length‑n gradient
    slice_part = conv_wa[n - 1 : 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]     # ∂Φ/∂a

    # map to x‑space via the Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine:
        * performs a dense analytic sweep over several lengths;
        * refines the best candidate with a smoothed‑max L‑BFGS continuation;
        * finishes with a short stochastic mass‑transfer local search;
        * returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Analytic sweep over a modest set of lengths
    # ------------------------------------------------------------------
    # lengths to try – 2000 is the baseline, larger n may improve a tiny bit
    n_candidates = [1500, 2000, 2500, 3000, 3500, 4000, 5000]

    # parameter grids
    p_grid = np.linspace(0.35, 0.75, 9)          # 0.35 … 0.75
    off_grid = np.linspace(0.0, 2.0, 9)          # 0 … 2

    best_score = np.inf
    best_a = None
    best_n = None

    for n in n_candidates:
        if time.time() >= deadline:
            break

        # ---- simple deterministic families ---------------------------------
        families = []

        families.append(uniform_distribution(n))
        families.append(triangular_distribution(n))
        families.append(hann_window(n))
        families.append(hamming_window(n))
        families.append(blackman_window(n))
        families.append(kaiser_window(n, beta=6.0))
        families.append(slepian_window(n, NW=2.5))

        # ---- product power‑law ------------------------------------------------
        for p in p_grid:
            for off in off_grid:
                families.append(product_distribution(n, p, off))

        # ---- sum power‑law ---------------------------------------------------
        for p in p_grid:
            for off in off_grid:
                families.append(sum_distribution(n, p, off))

        # ---- evaluate ---------------------------------------------------------
        for a in families:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_n = n
                elapsed = time.time() - start_time
                print(f"[init n={n}] new best score = {best_score:.6f} (t={elapsed:.1f}s)")

    if best_a is None:               # safety fallback
        best_a = uniform_distribution(2000)
        best_n = 2000
        best_score = compute_score(best_a)

    # ------------------------------------------------------------------
    # 2) Smoothed‑max L‑BFGS refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # soft‑max parameterisation (log‑space)
        eps = 1e-12
        x = np.log(np.maximum(best_a, eps))

        lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0]

        # Try using SciPy's L‑BFGS‑B; if unavailable fall back to a tiny Adam loop.
        try:
            from scipy.optimize import minimize
            have_scipy = True
        except Exception:                     # pragma: no cover
            have_scipy = False
            print("[info] SciPy not available – will use simple Adam fallback.")

        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            if have_scipy:
                # allow more iterations when we have more time left
                remaining = deadline - time.time()
                max_iter = int(min(800, max(30, remaining * 80)))
                res = minimize(
                    compute_phi_and_grad,
                    x,
                    args=(lam, best_n),
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": max_iter, "disp": False},
                )
                x = res.x
            else:
                # very small Adam optimiser (fallback)
                alpha = 0.03
                beta1, beta2 = 0.9, 0.999
                eps_opt = 1e-8
                m = np.zeros_like(x)
                v = np.zeros_like(x)
                t_step = 0
                while time.time() < deadline:
                    t_step += 1
                    phi, grad_x = compute_phi_and_grad(x, lam, best_n)
                    m = beta1 * m + (1 - beta1) * grad_x
                    v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                    m_hat = m / (1 - beta1 ** t_step)
                    v_hat = v / (1 - beta2 ** t_step)
                    step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                    x = x - step
                    if t_step >= 2000:
                        break

            a = np.exp(x - np.max(x))
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Light stochastic local search (mass‑transfer moves)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02  # start with 2 % of the source mass per trial
    while time.time() < deadline:
        cand = random_transfer(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            # shrink step size after a success to fine‑tune
            scale = max(0.001, scale * 0.9)
        else:
            # slowly expand step size again to retain exploration
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
