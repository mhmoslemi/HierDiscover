"""
propose_candidate – hybrid parametric + simulated‑annealing optimizer.

The algorithm consists of three phases:

1️⃣  **Parametric exploration**  
   A family of Beta‑type step functions  
   a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1}   (α,β>0)  
   (including the special cases uniform (α=β=1) and arcsine (α=β=½)).  
   A coarse grid over α,β (and a convex‐combination of uniform & arcsine)
   is evaluated with an FFT‑based convolution.  The best shape found is kept
   as the starting point for the next stage.

2️⃣  **Fine‑grained refinement of the parameters**  
   Around the best (α,β) the grid is repeatedly narrowed (step size ↓) until
   the improvement becomes smaller than 1e‑6.

3️⃣  **Local stochastic refinement (simulated annealing)**  
   Starting from the best parametrised vector, random mass transfers between
   two coordinates are attempted.  Moves are accepted if they improve the
   exact objective or with Metropolis probability exp(‑Δ/T); the temperature
   T is linearly cooled.  Occasionally a small Gaussian perturbation is
   applied if no improvement occurred for several seconds.

All operations respect the non‑negativity constraint; the vector is kept
normalised (sum = 1) because the evaluation function is scale invariant.
The routine respects the supplied time budget (default 1000 s) and finally
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a,a)) using FFT (linear convolution)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def exact_score(n: int, conv_max: float) -> float:
    """Exact value of evaluate_sequence for a normalised vector (sum=1)."""
    return 2.0 * n * conv_max


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Return a discretised Beta(α,β) density on {0,…,n‑1},
    i.e. a_i ∝ ((i+½)/n)^{α-1} (1-(i+½)/n)^{β-1}.
    """
    # avoid log(0) – the (i+0.5) shift guarantees positivity.
    i = np.arange(n, dtype=np.float64) + 0.5
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i)
    # stabilise the exponentials
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0.0:                     # should never happen for α,β>0
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (α=β=½)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    a /= a.sum()
    return a


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector (α=β=1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def blended_shape(n: int, w: float) -> np.ndarray:
    """
    Linear blend between uniform (w=0) and arcsine (w=1).
    The result is normalised.
    """
    u = uniform_shape(n)
    a = arcsine_shape(n)
    blended = (1.0 - w) * u + w * a
    blended /= blended.sum()
    return blended


def compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the exact evaluation‑function score for a (assumed normalised).
    """
    conv_max = max_convolution(a, fft_len)
    return exact_score(n, conv_max)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine first explores a parametrised Beta family, then refines the
    best candidate with simulated annealing.  The length of the sequence can be
    set via the `n` keyword argument (default 4000).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0) Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))                         # default length
    fft_len = next_pow2(2 * n - 1)                         # for a*a convolution

    # ------------------------------------------------------------------
    # 1) Parametric exploration (≈15 % of the budget)
    # ------------------------------------------------------------------
    stage1_deadline = start_time + 0.15 * budget_s
    best_score = np.inf
    best_a = None

    # ---- uniform & arcsine (baseline) ---------------------------------
    for name, shape in [("uniform", uniform_shape(n)),
                        ("arcsine", arcsine_shape(n))]:
        sc = compute_score(shape, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()
            print(f"[param] {name} score = {sc:.6f}")

    # ---- blend uniform–arcsine ----------------------------------------
    for w in np.linspace(0.0, 1.0, 21):
        a = blended_shape(n, w)
        sc = compute_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[param] blend w={w:.2f} score = {sc:.6f}")

    # ---- symmetric Beta grid -------------------------------------------
    alphas = np.linspace(0.25, 1.00, 16)   # 0.25 … 1.00
    for alpha in alphas:
        a = beta_shape(n, alpha, alpha)   # symmetric
        sc = compute_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[param] symmetric β={alpha:.3f} score = {sc:.6f}")

    # ---- fine‑grained refinement around current best (α≈β) ---------------
    # we assume the optimum is near symmetry; search a small neighbourhood.
    # The loop stops when the step size falls below 1e‑4 or time runs out.
    if best_a is not None:
        # recover the α,β that produced best_a (approximate by fitting)
        # Instead of fitting, just repeat a local brute‑force search.
        cur_alpha = 0.5
        # crude initial guess: use the α that gave the best symmetric result.
        # (we stored none, but we can reuse the last used alpha if needed)
        # Here we start from 0.5 and explore a small interval.
        step = 0.1
        while time.time() < stage1_deadline and step > 1e-4:
            improved = False
            candidates = np.arange(max(0.1, cur_alpha - 2 * step),
                                   min(2.0, cur_alpha + 2 * step) + 1e-12,
                                   step)
            for a_val in candidates:
                a = beta_shape(n, a_val, a_val)
                sc = compute_score(a, fft_len, n)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    cur_alpha = a_val
                    improved = True
                    print(f"[refine] α={a_val:.4f} score={sc:.6f}")
            if not improved:
                step *= 0.5   # shrink the search window

    # ---- asymmetric Beta exploration (α ≠ β) ---------------------------
    # Use the best symmetric α as centre, sweep a modest rectangle.
    if best_a is not None:
        # Determine a good symmetric α (store from above if possible)
        # We can approximate it by solving a simple fit: take middle index.
        center = n // 2
        # Roughly estimate exponent from a[center] (optional – skipped)
        # Instead we just try a grid around 0.5‑1.0.
        base_alpha = cur_alpha if 'cur_alpha' in locals() else 0.5
        for d_alpha in np.linspace(-0.3, 0.3, 13):
            for d_beta in np.linspace(-0.3, 0.3, 13):
                a_val = max(0.05, base_alpha + d_alpha)
                b_val = max(0.05, base_alpha + d_beta)
                a = beta_shape(n, a_val, b_val)
                sc = compute_score(a, fft_len, n)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    print(f"[asymm] α={a_val:.3f} β={b_val:.3f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing local refinement (≈85 % of the budget)
    # ------------------------------------------------------------------
    # Remaining time reserve for the stochastic phase.
    remaining_deadline = overall_deadline
    if time.time() >= remaining_deadline:
        # budget exhausted during the parametric phase
        final_a = best_a if best_a is not None else uniform_shape(n)
        final_a = np.maximum(final_a, 0.0)
        final_a /= final_a.sum()
        return final_a.tolist()

    # initialise local search state
    a = best_a.copy()
    conv_max = max_convolution(a, fft_len)
    best_score = exact_score(n, conv_max)
    best_a = a.copy()
    last_improve = time.time()
    it = 0

    # temperature schedule (linear cooling)
    T_start = best_score * 0.02          # 2 % of the current score
    T_end = best_score * 1e-6

    while time.time() < remaining_deadline:
        # progress fraction for temperature
        frac = (time.time() - (remaining_deadline - (budget_s * 0.85))) / (budget_s * 0.85)
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a small fraction (up to 5 %) of the mass at i
        max_move = min(a[i], 0.05 * a[i] + 1e-12)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate exact objective for the proposal
        new_conv_max = max_convolution(a_new, fft_len)
        new_score = exact_score(n, new_conv_max)

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                last_improve = time.time()
                print(f"[SA {it:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

        it += 1

        # occasional restart if stagnation persists (>12 s)
        if it % 5000 == 0 and (time.time() - last_improve) > 12.0:
            # perturb the best solution slightly and continue
            perturb = rng.normal(scale=0.02, size=n)
            a = np.maximum(best_a + perturb, 0.0)
            a /= a.sum()
            last_improve = time.time()
            print(f"[restart] perturbation at it={it}")

    # ------------------------------------------------------------------
    # 3) Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = uniform_shape(n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] total time={elapsed:.1f}s, best score={best_score:.6f}, length={n}")

    return best_a.tolist()
