"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
* 1) Build a few analytic “power‑law” families, keep the best.
* 2) Refine the best start with a continuation L‑BFGS on a smooth
  log‑sum‑exp approximation of the max‑convolution (as in the
  previous baseline).
* 3) Perform a projected sub‑gradient descent on the *exact* max‑convolution
  objective.  The sub‑gradient of max(conv) is cheap: for the active shift
  k* it is g_i = 2·a_{k*‑i} (zero outside the support).  After each step we
  project back onto the simplex and enforce symmetry (a_i = a_{n‑1‑i}).
* 4) Finish with a short random‑mass‑transfer local search for escaping
  possible plateaus.

The routine respects a 1000 s budget, prints progress information and
returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth max of the convolution (log‑sum‑exp) and its gradient w.r.t. x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a (linear)
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp smoothing
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                  # weights (sum to 1)

    # gradient ∂Φ/∂a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]          # length n
    grad_a = 2.0 * slice_part[::-1]                # ∂Φ/∂a

    # map to x‑space (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x

# Sub‑gradient utilities for the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_max_conv(a: np.ndarray, tol: float = 1e-12):
    """
    Compute a sub‑gradient of the function
        f(a) = max_k (a ⋆ a)[k]
    and return also the current maximal convolution value.
    The gradient (for each active shift k*) is g_i = 2·a_{k*‑i}
    (zero when the index is out of bounds).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)                # length 2n‑1
    max_b = np.max(conv)
    # treat all shifts that are within tolerance of the maximum as active
    active = np.where(conv >= max_b - max(tol, tol * max_b))[0]

    grad = np.zeros_like(a)
    for k in active:
        # valid i such that 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)               # inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i_range                     # correspondingly valid
        grad[i_range] += 2.0 * a[j]
    return grad, max_b


def projected_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.5,
    decay: float = 0.5,
    symmetrize: bool = True,
):
    """
    Simple projected sub‑gradient descent on the exact max‑convolution
    objective.  The iterate is kept on the probability simplex and,
    if `symmetrize` is True, forced to be mirror‑symmetric.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    step = init_step
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_b = subgradient_max_conv(a)
        # step size schedule – diminishing like 1/√t
        eta = step / math.sqrt(it)

        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        if symmetrize:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = project_simplex(a_new)   # re‑project after symmetrisation

        cur_score = 2.0 * n * max_b          # exact objective (no re‑norm needed)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # be a bit aggressive: restore step size a little
            step = min(step / decay, 1.0)
            # optional logging
            print(f"[subgrad] iter={it} new best={best_score:.6f}")
        else:
            # slowly shrink the step if we are not improving
            step *= decay

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that minimises
    the evaluation function.  The routine:

    1. builds a small grid of analytic families and keeps the best;
    2. refines it with a continuation L‑BFGS minimisation of a smooth
       max‑convolution surrogate;
    3. runs projected sub‑gradient descent on the true max‑convolution;
    4. performs a short random‑mass‑transfer local search;
    5. returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search – provides a solid starting point
    # ------------------------------------------------------------------
    candidates = []

    # power‑law families (product and sum modes)
    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55):
            for off in (0.0, 0.2, 0.5, 1.0):
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, p, off_eff, mode)
                candidates.append(cand)

    # uniform distribution
    candidates.append(np.ones(n) / n)

    # include global fallback if available
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the smoothed max‑convolution
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0]

    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:                     # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – will use a simple Adam optimizer instead.")

    for lam in lam_vals:
        if time.time() >= deadline:
            break

        if have_scipy:
            remaining = deadline - time.time()
            max_iter = int(min(800, max(30, remaining * 60)))  # generous per‑stage budget
            res = minimize(
                compute_phi_and_grad,
                x,
                args=(lam, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'disp': False}
            )
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # simple Adam optimiser (fallback)
            alpha = 0.05
            beta1, beta2 = 0.9, 0.999
            eps_opt = 1e-8
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t = 0
            while time.time() < deadline:
                t += 1
                phi, grad_x = compute_phi_and_grad(x, lam, n)
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t >= 2500:                      # safety stop for this λ
                    break

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the *exact* max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = deadline - 0.1   # leave a tiny margin for the final phase
        best_a, subgrad_score = projected_subgradient_descent(
            best_a,
            subgrad_deadline,
            init_step=0.6,
            decay=0.9,
            symmetrize=True,
        )
        if subgrad_score < best_score:
            best_score = subgrad_score
            print(f"[subgrad] improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search (escape possible plateaus)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand.copy()
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
