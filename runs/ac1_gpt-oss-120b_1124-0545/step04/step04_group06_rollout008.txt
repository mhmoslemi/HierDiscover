"""
Propose a non‑negative step‑function (a list of floats) that minimizes the
``evaluate_sequence`` objective.

The method has three stages:

1. **Parameter sweep** – generate many quick candidates from simple families
   (uniform, power‑law, exponential, sine‑squared).  The best of these seeds
   the expensive optimisation.

2. **Smoothed max‑convolution descent** – minimise the quantity
   `M = max_k (a*a)[k]` using a differentiable
   log‑sum‑exp approximation of the max.  The gradient is obtained analytically
   (a cross‑correlation) and the update is performed with Adam, followed by a
   Euclidean projection onto the probability simplex (∑a = 1, a_i ≥ 0).

3. **Multiple lengths & restarts** – the optimisation is run for several
   sequence lengths (up to ≈ 20000).  Each run gets a time slice from the
   global budget; the overall best sequence is returned.

The implementation uses only NumPy, works on a single CPU and respects the
1000 s time budget (a 5 s safety margin is kept).  Progress is printed to
stdout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implementation of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all would become negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when sum(a)=1:
    2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α)  (then normalised to sum = 1)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ r^{i-offset} for i ≥ offset (then normalised)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Simple decreasing linear shape (triangular) with optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sin2_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin²(π·(i+½)/n)  (non‑negative, normalised)."""
    idx = np.arange(n, dtype=float)
    a = np.sin(np.pi * (idx + 0.5) / n) ** 2
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a

def _optimize_from_init(
    a0: np.ndarray,
    deadline: float,
    target_score: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam‑style optimisation on the smoothed max‑convolution objective.
    Returns the best sequence found (sum = 1) and its objective value.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.05                # base learning rate for Adam
    t = 0

    # smoothing parameter for log‑sum‑exp (starts soft, becomes sharp)
    smooth_beta = 1.0
    smooth_inc = 0.0015

    # counters for escaping plateaus
    no_improve = 0

    while time.time() < deadline:
        t += 1

        # ----- convolution a*a -------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_c = conv.max()

        # ----- smooth approximation of the max (log‑sum‑exp) --------------------
        # w_k = exp(β·c_k) / Σ exp(β·c_k)
        exp_vals = np.exp(smooth_beta * (conv - max_c))   # shift for stability
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the smoothed objective -----------------------------
        # g_i = 2·∑_k w_k·a_{k‑i}   (cross‑correlation)
        w_rev = w[::-1]                              # reversed soft‑max vector
        conv2 = _conv_fft(a, w_rev)                  # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)             # (2n‑2‑i) for i=0..n‑1
        grad = 2.0 * conv2[idx]                      # shape (n,)

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- keep feasibility (simplex) ------------------------------------
        a = _project_onto_simplex(a)

        # ----- anneal the smoothing parameter ---------------------------------
        smooth_beta = min(smooth_beta + smooth_inc, 200.0)

        # ----- periodic evaluation --------------------------------------------

            if cur_score < best_score:

                if best_score < target_score:
                    break
            else:
                no_improve += 1

        # ----- escape plateaus -------------------------------------------------

                a[j] += delta

            a = a + 0.001 * (np.random.rand(n) - 0.5)
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)
            no_improve = 0

    return best_a, best_score
