"""
Propose a non‑negative step function (list of floats) that minimizes
the ``evaluate_sequence`` objective within a fixed time budget.

Algorithm
---------
1.  **Broad family sweep** – generate a handful of simple parametric
    sequences (power‑law, exponential, triangular, arcsine‑like) for a
    few candidate lengths.  Keep the best few as starting points.

2.  **Active‑set projected sub‑gradient descent** – for each seed run a
    long optimisation loop.  At each iteration the full auto‑convolution
    is computed, all indices whose value is within a tiny tolerance of the
    maximum are collected (the *active set*) and an averaged sub‑gradient
    of the max‑convolution is formed.  The step is taken with a momentum
    term and then projected onto the probability simplex.

    * Every 100 iterations a **multiplicative** update (exponential
      weights) is also tried and the better of the two candidates is kept.
    * Every 200 iterations a tiny random mass‑swap is performed to help
      escape plateaus.
    * Learning rate decays slowly; if no improvement is seen for a while
      it is reduced more aggressively.

3.  **Best‑of‑many** – the best sequence encountered over all lengths and
    seeds is returned.  The routine stops early as soon as the objective
    drops below the target 1.503.

The code uses only NumPy (FFT for fast full convolution) and runs on a
single thread; it respects the 1000 s time limit by checking the clock
every iteration.  All helper functions are top‑level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would become negative – uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using an FFT (fast for n≥~1 k)."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


# ----------------------------------------------------------------------
# Parametric families (all returned already normalised to sum=1)
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ r^{i‑offset} for i≥offset (0<r<1)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Decreasing linear (triangular) shape, optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
    a_i ∝ 1 / sqrt((i+0.5)*(n‑i‑0.5)).
    """
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective / gradient utilities
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` for sum(a)=1."""
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


def _max_convolution(a: np.ndarray):
    """Return (max value, list of indices achieving (nearly) the max)."""
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance accounts for floating‑point noise
    eps = max_val * 1e-12 + 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    return max_val, conv, idxs


def _average_subgradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv(a,a)) over the active index set.
    For a single index k the sub‑gradient is g_j = 2·a_{k‑j}
    (zero when the partner index is out of range).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active_idxs:
        idx = k - np.arange(n)           # partner index for each component
        mask = (idx >= 0) & (idx < n)
        grad[mask] += 2.0 * a[idx[mask]]
    grad /= max(1, active_idxs.size)     # avoid division by zero
    return grad

# Main optimisation routine
# ----------------------------------------------------------------------

            seed_pool.append(a)

        for a in top_seeds:
            n = a.size
            momentum = np.zeros_like(a)
            lr = 0.8
            lr_min = 5e-7
            lr_decay = 0.9999
            last_improve_iter = 0
            iter_cnt = 0

            while time.time() < deadline:
                iter_cnt += 1

                # full auto‑convolution and active set
                max_val, conv, active = _max_convolution(a)

                # sub‑gradient averaged over the active set
                grad = _average_subgradient(a, active)

                # momentum (exponential moving average) and additive step
                momentum = 0.9 * momentum + grad
                a_add = a - lr * momentum
                a_add = _project_onto_simplex(a_add)

                # occasional multiplicative (exponential‑weights) step
                if iter_cnt % 100 == 0:
                    eta = lr * 0.5
                    a_mul = a * np.exp(-eta * grad)
                    a_mul = a_mul / a_mul.sum()
                    # evaluate both candidates (re‑use conv for a_add later)
                    score_add = 2.0 * n * _conv_fft(a_add, a_add).max()
                    score_mul = 2.0 * n * _conv_fft(a_mul, a_mul).max()
                    if score_mul < score_add:
                        a = a_mul
                        cur_score = score_mul
                    else:
                        a = a_add
                        cur_score = score_add
                else:
                    a = a_add
                    cur_score = 2.0 * n * _conv_fft(a, a).max()   # cheap: reuse after step

                # keep track of global best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    last_improve_iter = iter_cnt
                    print(f"[improve] n={n} iter={iter_cnt} score={best_score:.6f}")

                # early stop if we have already beaten the target bound
                if best_score < 1.503:
                    print("[target reached]")
                    break

                # --------- random mass‑swap to help escape plateaus ---------
                if iter_cnt % 200 == 0:
                    i_src, i_dst = np.random.choice(n, 2, replace=False)
                    # move at most 25 % of the source mass (bounded by step size)
                    delta = min(a[i_src] * 0.25, lr * 0.2)
                    if delta > 1e-12:
                        a[i_src] -= delta
                        a[i_dst] += delta
                        a = _project_onto_simplex(a)

                # --------- learning‑rate schedule -------------------------
                lr = max(lr * lr_decay, lr_min)

                # if we haven't improved for a long while, shrink the step more
                if iter_cnt - last_improve_iter > 200_000:
                    lr = max(lr * 0.5, lr_min)
                    last_improve_iter = iter_cnt   # reset counter to avoid endless shrinkage

                # very generous safeguard against infinite loops
                if iter_cnt > 5_000_000:
                    break

            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.   Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:                      # fallback – should never happen
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    print(f"[final] best_score={best_score:.6f}  length={best_seq.size}")
    return best_seq.tolist()
