```
    min_tau = 1e-8

    it = 0

        # Compute convolution
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # Update best
        if cur_score < best_score:

            best_seq = a.copy()

        tau = max(tau * tau_decay, min_tau)

        # Softmax gradient
        # w = exp((conv_full - max_b)/tau)
        w = np.exp((conv_full - max_b) / tau)
```

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \((a*a)_k\) is the linear convolution of the vector `a` with itself.  
If we normalise the vector to `sum(a)=1` the problem reduces to minimising  

\[
2\,n\;\max_k (a*a)_k .
\]

The best known construction is a discretised Cantor‑measure; it already beats the uniform distribution but still leaves room for improvement.  
The new search follows three ideas:

1. **Higher‑resolution Cantor start** – we use a much finer Cantor histogram (`L=18` → 262 144 atoms) which gives a flatter initial self‑convolution.
2. **Progressive smooth‑max optimisation** – a soft‑max (log‑sum‑exp) weight is used at the beginning, its temperature is decreased quickly so the algorithm focuses on the current worst peak.
3. **Hard‑max fine tuning** – once the temperature is tiny we switch to the exact sub‑gradient of the true max, i.e. a simple shift of the current vector.
4. **Momentum & adaptive step‑size** – Nesterov‑style momentum stabilises the descent, while the step size decays as \(1/\sqrt{t}\).

The whole routine is a pure NumPy implementation, runs entirely in RAM and respects the 1000 s budget.  
The best sequence found during the run is returned.

```python
"""
Propose a step‑function sequence with a very low value of
evaluate_sequence().  The algorithm

1. builds a high‑resolution Cantor histogram,
2. repeatedly minimises a smooth approximation of the maximum of a*a,
3. finishes with a hard‑max sub‑gradient step,
4. uses Nesterov momentum and an adaptive step size,
5. keeps the best sequence seen.

All helper functions are top‑level; no closures or lambdas are used.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    """
    Linear convolution of u and v using FFT.
    ``size`` must be ≥ len(u)+len(v)-1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the full linear convolution a*a."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the middle‑third Cantor measure on a grid of size ``n``.
    ``L`` is the depth of the construction (2**L atoms).  The result
    sums to one and contains only non‑negative entries.
    """
    N = 1 << L                                 # number of atoms
    # bit pattern of the atom indices (shape (N, L))
    idx = np.arange(N, dtype=np.uint32)[:, None]   # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)

    # position of each atom in [0,1)
    powers = 3.0 ** np.arange(1, L + 1)                # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)          # (N,)

    # histogram on the desired grid
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)            # normalise → sum == 1
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed (reproducibility).
    budget_s : int
        Time budget.
    kwargs
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from a high‑resolution Cantor histogram
    # ------------------------------------------------------------------
    # Use a depth that gives many atoms but still fits comfortably in RAM.
    L = max(18, int(np.ceil(np.log2(4 * n))))   # 2**L >> n
    try:
        a = _cantor_histogram(n, L)
    except Exception:                           # fallback on a lighter histogram
        L = max(14, int(np.ceil(np.log2(2 * n))))
        a = _cantor_histogram(n, L)

    # Safety: if the histogram is all zeros (should not happen), fall back to uniform.
    if a.sum() < 1e-3:
        a = np.ones(n, dtype=float) / n

    # ------------------------------------------------------------------
    # Pre‑compute FFT sizes (they stay constant)
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)      # for a*a
    size2 = _next_pow2(3 * n - 2)      # for a_rev * weight

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b                      # sum(a)=1
    best_seq = a.copy()
    best_iter = 0

    print(f"[init] n={n}, L={L}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    base_step = 0.9                # controls the absolute step size
    momentum = 0.85                # Nesterov‑type momentum
    v = np.zeros_like(a)           # momentum accumulator

    # temperature for the soft‑max (log‑sum‑exp) approximation of max
    tau = max_b * 0.5
    tau_decay = 0.99995            # decay per iteration
    min_tau = 1e-12

    it = 0
    last_improve = start

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and score
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            best_iter = it
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ------------------------------------------------------------------
        # Choose the gradient: soft‑max while the temperature is still > 1e‑5,
        # otherwise fall back to the exact sub‑gradient of the true max.
        # ------------------------------------------------------------------
        if tau > 1e-5:
            # ------- soft‑max (smooth max) -------
            # w_i ∝ exp((conv_i - max)/tau) , normalised to sum 1
            w = np.exp((conv_full - max_b) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # gradient = -2 * (a_rev convolved with w)
            g_full = _conv_fft(a_rev, w, size2)
            g = g_full[n - 1 : n - 1 + n]          # central n entries
            d = -2.0 * g
        else:
            # ------- hard max (true sub‑gradient) -------
            max_idx = int(conv_full.argmax())
            shift = max_idx - (n - 1)               # how far the delta is shifted
            d = np.zeros_like(a)

            if shift >= 0:
                # indices 0 … n‑shift‑1 receive a[shift … n‑1]
                d[: n - shift] = -2.0 * a[shift:]
            else:
                # shift < 0  →  indices -shift … n‑1 receive a[0 … n+shift‑1]
                d[-shift:] = -2.0 * a[: n + shift]

        # keep the total mass unchanged (project onto Σd = 0)
        d -= d.mean()

        # Nesterov momentum update
        v = momentum * v + (1.0 - momentum) * d

        # adaptive step size (decreases like 1/√t)
        step = base_step / np.sqrt(it + 1.0)
        a = a + step * v

        # projection onto the feasible set: non‑negative, sum = 1
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= s

        # keep the reversed copy up‑to‑date
        a_rev = a[::-1].copy()

        # --------------------------------------------------------------
        # temperature annealing
        # --------------------------------------------------------------
        tau = max(tau * tau_decay, min_tau)

        # --------------------------------------------------------------
        # occasional random perturbations to escape plateaus
        # --------------------------------------------------------------
        if it % 250000 == 0 and it > 0:
            # add a tiny zero‑mean Gaussian noise and re‑normalise
            noise = np.random.randn(n) * 0.001
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            a_rev = a[::-1].copy()

        it += 1

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, "
          f"best_iter={best_iter}, time={total:.1f}s")
    return best_seq.tolist()
```
