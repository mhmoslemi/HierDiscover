def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_fft(u, v, size):
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[:len(u) + len(v) - 1]

def _full_convolution(a, size):
    return _conv_fft(a, a, size)

def _cantor_histogram(n, L=14):
    N = 1 << L
    # bits of shape (N, L)
    idx = np.arange(N, dtype=np.uint32)[:, None]  # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)
    powers = 3.0 ** np.arange(1, L + 1) # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)  # (N,)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)
    return a

      (a*a).  The gradient is obtained analytically via a single FFT.
  2️⃣  After a fixed number of iterations the update switches to an
      exponentiated‑gradient (multiplicative) step, which respects the
      simplex constraints and often yields finer reductions.
Random jitter and occasional strong perturbations are injected when the
search stalls.  The algorithm starts from a high‑resolution Cantor‑measure
histogram (L=18) or from a previously found solution if available and
keeps the best sequence encountered.  It runs for at most ``budget_s``
seconds and returns the best list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Produce a length‑`n` histogram approximating the middle‑third Cantor
    (1/3) measure.  The histogram is normalised to have total mass 1.
    """
    N = 1 << L  # number of atoms = 2**L
    # binary expansion of the integers 0 … N‑1
    idx = np.arange(N, dtype=np.uint32)[:, None]          # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N,L)
    # positions of atoms in [0,1)
    powers = 3.0 ** np.arange(1, L + 1)                    # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)             # (N,)
    # histogram on the required grid
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)               # normalise → sum == 1
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The optimisation routine is described in the
    module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin for the harness
    deadline = start + budget_s - 0.2

    n = int(kwargs.get("n", 6000))

    # ------------------- 1️⃣  Initialise ------------------------------------
    # Try to reuse an already discovered sequence if it exists.
    a0 = None
    if "height_sequence_1" in globals():
        try:
            seq = globals()["height_sequence_1"]
            a_tmp = np.asarray(seq, dtype=np.float64)
            if a_tmp.shape[0] != n:
                # Resample by linear interpolation.
                x_old = np.linspace(0.0, 1.0, a_tmp.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a_tmp = np.interp(x_new, x_old, a_tmp)
            a_tmp = np.clip(a_tmp, 0.0, None)
            s = a_tmp.sum()
            if s >= 0.01:
                a0 = a_tmp / s
        except Exception:
            a0 = None

    # Cantor histogram with a higher resolution (L=18)
    a_cantor = _cantor_histogram(n, L=18)
    a_cantor = np.clip(a_cantor, 0.0, None)
    a_cantor = a_cantor / max(a_cantor.sum(), 1e-12)

    # Helper to compute the objective without calling ``evaluate_sequence``.
    size1 = _next_pow2(2 * n - 1)
    size2 = _next_pow2(3 * n - 2)

    def _objective_val(a: np.ndarray) -> float:
        conv = _full_convolution(a, size1)
        maxb = conv.max()
        sum_a = a.sum()
        return float(2.0 * n * maxb / (sum_a ** 2))

    # Choose the better of the two initialisations (if both exist).
    if a0 is not None:
        obj0 = _objective_val(a0)
        objc = _objective_val(a_cantor)
        if obj0 < objc:
            a = a0.copy()
            print(f"[init] Using previous sequence, score={obj0:.6f}")
        else:
            a = a_cantor.copy()
            print(f"[init] Using Cantor L=18, score={objc:.6f}")
    else:
        a = a_cantor.copy()
        print(f"[init] Using Cantor L=18, score={_objective_val(a):.6f}")

    # Safety normalisation (should already be fine).
    if a.sum() < 0.01:
        a = _cantor_histogram(n, L=18)
        a = a / max(a.sum(), 1e-12)

    # ------------------- bookkeeping ---------------------------------------
    a_rev = a[::-1].copy()
    best_a = a.copy()
    best_score = _objective_val(a)
    print(f"[init] n={n}, initial best score={best_score:.6f}")

    # ------------------- hyper‑parameters ----------------------------------
    # Additive (Nesterov) phase
    base_step = 0.9
    momentum = 0.85
    v = np.zeros_like(a)

    # Smooth‑max temperature parameters
    tau_factor = 0.5
    tau_decay = 0.99995

    # Switch to multiplicative (exponentiated‑gradient) after this many iterations
    switch_iter = int(2e5)  # 200 k iterations

    # Multiplicative step size
    base_eta = 0.4
    eta_decay = 0.99997

    it = 0
    last_improve = start

    # ------------------- main optimisation loop ----------------------------
    while time.time() < deadline:
        # Full linear convolution a*a (used for both the objective and the soft‑max)
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b          # sum(a) is always 1 after each projection
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:8d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # Adaptive temperature for the smooth‑max
        tau = max(tau_factor * max_b, 1e-14)
        tau_factor *= tau_decay

        # Soft‑max weights (numerically stable)
        w = np.exp((conv_full - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # Gradient of the smooth‑max (central n entries)
        g_full = _conv_fft(a_rev, w, size2)
        g = g_full[n - 1 : n - 1 + n]
        d = -2.0 * g
        # Zero‑mean to keep the ℓ₁‑norm unchanged
        d -= d.mean()

        # --------------------------------------------------------------
        # Update step (additive → multiplicative transition)
        # --------------------------------------------------------------
        if it < switch_iter:
            # Additive Nesterov‑type step
            step = base_step / np.sqrt(it + 1.0)
            v = momentum * v + (1.0 - momentum) * d
            a_new = a + step * v
        else:
            # Exponentiated‑gradient (multiplicative) step
            eta = base_eta / np.sqrt(it - switch_iter + 1.0)
            exp_arg = -eta * d
            exp_arg -= exp_arg.max()           # avoid overflow
            a_new = a * np.exp(exp_arg)

        # Projection onto the probability simplex (non‑negative, Σ=1)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s

        # --------------------------------------------------------------
        # Exploration helpers
        # --------------------------------------------------------------
        # Small jitter every 50 k iterations
        if it % 50000 == 0 and it > 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02          # ±1 %
            a_new = a_new * (1.0 + jitter)
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / max(a_new.sum(), 1e-12)

        # If we have not improved for 30 s, inject a stronger random perturbation
        if time.time() - last_improve > 30.0 and it > 1000:
            print(f"[{it}] stagnation – applying strong perturbation")
            perturb = np.random.rand(n)
            a_new = a * 0.9 + 0.1 * perturb
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / max(a_new.sum(), 1e-12)
            v = np.zeros_like(a)           # reset momentum
            last_improve = time.time()

        # --------------------------------------------------------------
        # Prepare for next iteration
        # --------------------------------------------------------------
        a = a_new
        a_rev = a[::-1].copy()
        it += 1

        # Decay multiplicative step size (harmless during additive phase)
        base_eta *= eta_decay

    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_a.tolist()
