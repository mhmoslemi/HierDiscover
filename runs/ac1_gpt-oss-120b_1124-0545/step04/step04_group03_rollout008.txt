"""
Propose a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Build a fairly dense grid of analytic “beta‑type’’ candidates
   a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}  (product) or a_i ∝ (i+off)^{‑p}+(n‑1‑i+off)^{‑p}
   for many values of p ∈ [0.30,0.80] and several offsets.  Keep the K best
   candidates (K≈5).

2. For each of those starts run a smoothed‑max optimisation using the
   continuation on λ (λ from 0.5 up to 2000).  The smooth objective is

        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)

   Its gradient w.r.t. the unconstrained log‑it vector x (a = softmax(x))
   is computed analytically with FFTs.  An Adam optimiser (no external
   optimiser needed) performs many updates per λ (≈2000), constantly
   evaluating the true objective and remembering the best distribution.

3. After the continuation a short multiplicative‑mirror‑descent phase
   (sub‑gradient on the exact max‑convolution) is applied to remove any
   remaining bias.

4. Finally a lightweight random‑mass‑transfer local search (simulated
   annealing style) explores the neighbourhood of the best found vector.

All steps respect a 1000 s wall‑clock budget; the best sequence seen
so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low–level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a) - 1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (sum does not matter because
    the function is scale‑invariant).  Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode='product' : a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}
    mode='sum'     : a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another and re‑project."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # project back onto the simplex (simple scaling works because only two entries changed)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth max of the self‑convolution.
    Returns Φ_λ(a) and ∇_x Φ_λ where a = softmax(x).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # conv = a * a
    conv = conv_fft_self(a)                     # length 2n‑1
    max_c = np.max(conv)

    # smooth max via log‑sum‑exp
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                      # weights, sum to 1

    # gradient of Φ_λ w.r.t. a:
    # dΦ/da = 2 * (w ↔ a)  where ↔ denotes convolution with reversed w
    w_rev = w[::-1]                             # reverse weights
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    # keep the n entries that correspond to the gradient
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad_a = 2.0 * slice_part[::-1]             # ∂Φ/∂a

    # map gradient to x‑space via soft‑max Jacobian
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                 # ∇_x Φ
    return phi, grad_x


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def adam_optimise(x0: np.ndarray,
                  lam_vals,
                  n,
                  deadline,
                  max_iter_per_lambda=2000,
                  alpha0=0.05):
    """
    Adam optimiser on the smooth Φ_λ objective.
    Returns the best distribution (as a vector) and its exact score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    t = 0

    best_a = softmax(x)
    best_score = compute_score(best_a)

    # adapt learning‑rate a bit with λ
    for lam in lam_vals:
        # a slightly larger step for early (smooth) λ
        alpha = alpha0 * (0.5 + 0.5 * np.log1p(lam) / np.log1p(lam_vals[-1]))
        for _ in range(max_iter_per_lambda):
            if time.time() >= deadline:
                return best_a, best_score
            phi, grad_x = compute_phi_and_grad(x, lam, n)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
            x = x - step

            # occasional exact evaluation (every 20 steps)
            if (_ + 1) % 20 == 0:
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        # after each λ, tighten the learning‑rate a little
        alpha0 *= 0.95
    return best_a, best_score


def mirror_subgradient_descent(a0,
                               max_steps=2000,
                               eta=0.02,
                               deadline=None):
    """
    Multiplicative (exponentiated) mirror descent on the exact
    max‑convolution objective using a sub‑gradient of the max term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]

    for step in range(max_steps):
        if deadline is not None and time.time() >= deadline:
            break
        conv = conv_fft_self(a)
        k = int(np.argmax(conv))
        # sub‑gradient of max conv_k w.r.t. a: g_i = a_i + a_{k-i} (if index in range)
        g = a.copy()
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g[mask] += a[idx[mask]]

        # exponentiated update
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # evaluate each 10 steps
        if step % 10 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # slowly shrink η after improvement
                eta = max(0.001, eta * 0.9)
            else:
                # otherwise gently increase η to keep exploring
                eta = min(0.05, eta * 1.001)
    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises the
    evaluation function.  The routine builds a fine grid of analytic candidates,
    runs a multi‑start smoothed‑max Adam optimisation, refines with mirror descent
    and finishes with a short random‑move local search.  The best sequence found
    before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Rich analytic grid (store the few best candidates)
    # ------------------------------------------------------------------
    top_k = 5
    candidates = []    # list of (score, vector)

    # product / sum families, finer grid in p and offset
    p_vals = np.linspace(0.30, 0.80, 11)  # 0.30 … 0.80
    offsets = [0.0, 0.1, 0.3, 0.5, 1.0, 2.0]

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in offsets:
                cand = analytic_candidate(n, float(p), float(off), mode)
                sc = compute_score(cand)
                candidates.append((sc, cand))

    # uniform (baseline)
    uni = np.ones(n) / n
    candidates.append((compute_score(uni), uni))

    # possibly use the globally‑provided construction
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            if ref.sum() > 0.0:
                ref = ref / ref.sum()
                candidates.append((compute_score(ref), ref))

    # keep the best few
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:top_k]

    best_a = top_candidates[0][1].copy()
    best_score = top_candidates[0][0]
    print(f"[init] best analytic score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑start Adam optimisation on the smoothed max
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                    50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

    for idx, (s0, a0) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # initialise log‑it vector
        eps = 1e-12
        x0 = np.log(np.maximum(a0, eps))

        # run Adam (allow it to consume a reasonable chunk of the budget)
        a_opt, sc_opt = adam_optimise(
            x0,
            lam_vals=lam_schedule,
            n=n,
            deadline=deadline,
            max_iter_per_lambda=2000,
            alpha0=0.05,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            elapsed = time.time() - start_time
            print(f"[adam start {idx}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement on the exact max‑convolution
    # ------------------------------------------------------------------
    a_refined, sc_refined = mirror_subgradient_descent(
        best_a,
        max_steps=3000,
        eta=0.02,
        deadline=deadline,
    )
    if sc_refined < best_score:
        best_score, best_a = sc_refined, a_refined.copy()
        print(f"[mirror] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search (simulated‑annealing flavour)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02  # relative amount moved per trial
    temp = 1.0    # temperature for occasional uphill moves

    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            # tighten step size a bit
            scale = max(0.001, scale * 0.9)
            temp = max(0.01, temp * 0.9)
        else:
            # accept uphill moves with small probability (simulated annealing)
            if np.random.rand() < np.exp(-(sc - best_score) / temp):
                a = cand
            # slowly increase step size and temperature to keep exploration alive
            scale = min(0.05, scale * 1.001)
            temp = min(2.0, temp * 1.001)

    total = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total:.1f}s")
    # Return a plain Python list (any positive scaling works – keep sum≈1)
    return best_a.tolist()
