    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    length_candidates = [2000, 3000, 4000, 5000, 8000, 12000]

    for n in length_candidates:
        # generate some param families
        # exponential family: r powers from 0.93 to 0.99
        for r in np.linspace(0.93, 0.99, 7):
            a = _exp_sequence(n, r, offset=0)
            score = _objective(a)
            if score < best_score:
                best_score = score; best_seq = a.copy()
        # power family
        for alpha in [0.3,0.5,0.7]:
            a = _power_sequence(n, alpha, offset=0)
            score = _objective(a)
            if score < best_score:
                best_score = score; best_seq = a.copy()
        # linear
        a = _linear_sequence(n, offset=0)
        # ...
    # Also use global height_sequence_1 if available:
    try: use...
    except: pass
    
    # after sweep, run refinement using annealed Adam with softmax
    # we may try for each length (top few) but for simplicity use the best length from sweep.
    a = best_seq.copy()
    n = a.size
    # ensure sum=1
    a = a / a.sum()
    
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.5   # base learning rate
    eps_min = 1e-12
    
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv and max
        conv = _conv_fft(a, a)
        max conv = conv.max()
        # compute softmax weights
        eps = max(conv)*0.2  # fudge factor
        # ensure eps not too small
        eps = max(eps, 1e-8)
        w = np.exp((conv - max(conv)) / eps)
        w_sum = w.sum()
        w = w / w_sum
        # gradient via cross-correlation
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_min)
        a_new = a - step
        a = _project_onto_simplex(a_new)
        # occasionally recompute epsilon schedule (maybe decay)
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score; best_seq = a.copy()

            # check early exit target

                print("[target reached by anneal]")
                break
        # random mass swap occasionally
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.2, lr * 0.1)
            a[i] -= delta; a[j] += delta
            a = _project_onto_simplex(a)
    
    # Return best_seq as list
    print(f"[final] best_score={best_score:.6f} n={best_seq.size} iter={iteration}")
    return best_seq.tolist()

"""
Annealed Adam optimizer for the max‑convolution objective.

The objective after normalising a sequence `a` (∑a=1) is

    obj(a) = 2·n·max_k (a∗a)[k] .

We minimise it by a smooth approximation of the hard max:
    F_ε(a) = ε·log∑_k exp((a∗a)[k]/ε) .

The gradient of F_ε is a weighted sum of the sub‑gradients of every
convolution entry.  Using a temperature ε that is gradually reduced
(annealing) gives a stable descent direction that can escape the plateaus
which trap a naïve sub‑gradient method.  An Adam step is taken, the iterate
is projected onto the probability simplex and a few random mass‑swaps are
performed to add stochasticity.

The routine performs a cheap parametric sweep (exponential / power‑law
families) to obtain a good start, then refines the best candidate with the
annealed Adam optimiser.  It respects a global time‑budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum = 1).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using FFT (fast for lengths ≳ 1 000).

    Parameters
    ----------
    x, y : np.ndarray
        Real 1‑D input arrays.

    Returns
    -------
    np.ndarray
        Full convolution (length = len(x)+len(y)-1).
    """
    size = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """
    Normalised power‑law: a_i ∝ (i+1+offset)^(-α).

    Parameters
    ----------
    n      : int   – length of the sequence.
    alpha  : float – exponent (α>0).
    offset : int   – leading zeroes before the law starts.

    Returns
    -------
    np.ndarray of length `n` with non‑negative entries and sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """
    Normalised exponential decay: a_i ∝ r^{i‑offset} for i≥offset,
    zeros before ``offset``.  ``0 < r < 1``.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """
    Simple decreasing linear (triangular) shape: a_i = n‑i for i≥offset,
    else 0.  Normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the vector is
    normalised to sum = 1:
        2·n·max(conv(a,a))
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()
