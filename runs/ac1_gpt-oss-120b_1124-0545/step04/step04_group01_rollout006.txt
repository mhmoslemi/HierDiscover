def _lp_max_sum(f, M, active_idxs):
    n = len(f)
    m = len(active_idxs)
    A = np.zeros((m, n))
    for row_idx, k in enumerate(active_idxs):
        # compute j range:
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # for j in range(...):
        #   i = k - j
        #   A[row_idx, j] = f[i]
        # Later: conv(f,b)[k] = A[row_idx] @ b
        # i.e., sum_{j} f[k - j] * b[j]
        # So fill.
        # Use vectorized assignments:
        js = np.arange(j_start, j_end+1)
        is_ = k - js
        A[row_idx, js] = f[is_]
    # Solve:
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use linprog
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    b = res.x
    return b
```

def _build_A(f, active_idxs):
    n = len(f)
    m = len(active_idxs)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idxs):
        start_j = max(0, k - (n-1))
        end_j = min(n-1, k)
        # i = k - j
        # f[i] length = end_j - start_j + 1
        js = np.arange(start_j, end_j+1)
        is_ = k - js
        A[row, js] = f[is_]
    return A
```

def _lp_solve(f, M, active_idxs):
    A = _build_A(f, active_idxs)
    c = -np.ones(f.shape)  # maximize sum
    b_ub = np.full(len(active_idxs), M, dtype=np.float64)
    bounds = [(0, None)] * len(f)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("Linprog failed")
    return res.x
```

"""
A hybrid search for a non‑negative step function that minimises

      2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines three ideas that were not used together before:

1. **Linear‑programming mass redistribution** (inspired by the
   “active‑set” LP from the state‑of‑the‑art description).  For a given
   shape `f` we compute `M = max(conv(f,f))` and solve a small linear
   program that maximises the ℓ₁‑norm of a new vector `b` under the
   constraints `conv(f,b) ≤ M`.  The active constraints are built from the
   current largest convolution values – only a few dozen rows are needed,
   which makes the LP cheap even for n≈8000.

2. **Mixture step** – we replace `f` by a convex combination
   `(1‑t)·f + t·b` (renormalised to unit sum).  By trying several step
   sizes we keep the maximum convolution monotone and often obtain a
   lower peak.

3. **Fine‑grained deterministic refinement** using the exact
   sub‑gradient of the max‑convolution, followed by a cheap stochastic
   hill‑climber.  These stages are the same as in the previous baseline
   but run after the LP‑mixing, which already gives a better starting
   point.

The whole routine is time‑aware (default budget 1000 s) and works for
lengths 4000 and 8000.  It returns a plain list of floats that can be
fed directly into ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    """Simple sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric series a_i = r**i."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------


def _build_A(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Construct the (dense) matrix A of size (|active|, n) where
    A[k, j] = f[active[k] - j]  if the index is valid, otherwise 0.
    """
    n = f.shape[0]
    m = active.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active):
        # j runs over indices where 0 <= k-j < n  →  max(0, k-n+1) ≤ j ≤ min(k, n-1)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        is_ = k - js
        A[row, js] = f[is_]
    return A


def _lp_solve(f: np.ndarray, M: float, active: np.ndarray) -> np.ndarray:
    """
    Solve   max 1ᵀb   s.t.   conv(f,b)[k] ≤ M  for all k in *active*,
    b ≥ 0.
    Returns the optimal b (may be dense).
    """
    A = _build_A(f, active)                     # shape (m, n)
    m = A.shape[0]
    c = -np.ones(f.shape, dtype=np.float64)     # maximise sum(b)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * f.shape[0]

    # ``highs`` is very fast for moderate dense problems.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    b = np.maximum(res.x, 0.0)                  # guard against tiny negatives
    return b


def _lp_feasible(f: np.ndarray, M: float, init_active: np.ndarray,
                 max_iters: int = 12, max_active: int = 300) -> np.ndarray:
    """
    Repeatedly solve the LP, adding any violated constraints, until all
    constraints are satisfied or the active set reaches *max_active*.
    """
    active = set(int(k) for k in init_active)
    for _ in range(max_iters):
        if len(active) > max_active:
            # truncate to keep the LP cheap – keep the largest‑M rows only
            cur_conv = _conv_fft(f, np.zeros_like(f))
            sorted_idx = np.argsort(-cur_conv)[:max_active]
            active = set(int(k) for k in sorted_idx)

        b = _lp_solve(f, M, np.array(sorted(active), dtype=np.int64))
        conv_fb = _conv_fft(f, b)
        violated = np.where(conv_fb > M + 1e-14)[0]
        if len(violated) == 0:
            return b
        active.update(int(k) for k in violated)
    return b


# ----------------------------------------------------------------------
# Deterministic refinement (sub‑gradient & hill climbing)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) using the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Simple stochastic pair‑wise mass‑transfer hill‑climbing."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.01

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    objective.  The routine respects the supplied time budget and returns
    the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Possibly reuse a previously known construction
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    lengths = [4000, 8000]                 # we try two sizes
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Initial seed(s)
        # ------------------------------------------------------------------
        seeds = []

        # 1. Reuse previous if possible (pad with zeros if shorter)
        if previous is not None:
            if previous.shape[0] == n:
                seeds.append(previous.copy() / previous.sum())
            elif previous.shape[0] < n:
                tmp = np.zeros(n, dtype=np.float64)
                tmp[:previous.shape[0]] = previous
                seeds.append(tmp / tmp.sum())

        # 2. Analytic families
        seeds.append(_uniform(n))
        seeds.append(_sine_window(n))
        seeds.append(_geom(n, 0.7))                           # a mild decay
        seeds.append(_beta_shape(n, 0.5, 0.5))

        # 3. A few random Dirichlet draws
        rng = np.random.default_rng(seed + 1000 + n)
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, 1.0)))

        # ------------------------------------------------------------------
        # Quick sub‑gradient warm‑up for each seed (very cheap)
        # ------------------------------------------------------------------
        for s in seeds:
            if time.time() > deadline:
                break
            seq_ref, val_ref = _subgradient_refine(s, n, deadline,
                                                   max_iters=4000)
            if val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[Seed‑SG] n={n:5d} new global best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Main LP‑mixing loop (the novel part)
        # ------------------------------------------------------------------
        if global_best_seq is None or global_best_seq.shape[0] != n:
            a = _uniform(n)
        else:
            a = global_best_seq.copy()

        mix_passes = 6
        for mix_it in range(mix_passes):
            if time.time() > deadline:
                break

            conv_a = _conv_fft(a, a)
            M = conv_a.max()
            cur_val = 2.0 * n * M
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[LP‑mix] n={n:5d} it={mix_it} new global best={cur_val:.6f}")

            # select a modest active set (top‑K conv values)
            top_k = min(30, conv_a.size)
            active_idxs = np.argpartition(-conv_a, top_k - 1)[:top_k]

            # solve the LP (cutting‑plane inside the routine)
            b = _lp_feasible(a, M, active_idxs)

            # normalise b to unit ℓ₁‑norm (shape only, sum kept 1)
            sumb = b.sum()
            if sumb == 0.0:
                g = _uniform(n)
            else:
                g = b / sumb

            # try several mixing fractions and keep the best improvement
            best_mix_a = a
            best_mix_val = cur_val
            for t in (0.5, 0.3, 0.2, 0.1, 0.05):
                cand = (1.0 - t) * a + t * g
                cand /= cand.sum()                     # stay on the simplex
                conv_cand = _conv_fft(cand, cand)
                val_cand = 2.0 * n * conv_cand.max()
                if val_cand < best_mix_val:
                    best_mix_val = val_cand
                    best_mix_a = cand
            a = best_mix_a
            if best_mix_val < global_best_val:
                global_best_val = best_mix_val
                global_best_seq = a.copy()
                print(f"[LP‑mix‑accept] n={n:5d} it={mix_it} best={best_mix_val:.6f}")

        # ------------------------------------------------------------------
        # Final deterministic refinement (sub‑gradient + hill‑climbing)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            seq_ref, val_ref = _subgradient_refine(a, n, deadline,
                                                   max_iters=20000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[Final‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        if time.time() < deadline:
            seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                         max_iters=20000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[Final‑Hill] n={n:5d} new best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # Final sanity check (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
```

This implementation adds a lightweight linear‑programming mixing phase that directly reduces the peak of the autocorrelation, then tightens the result with the proven sub‑gradient and hill‑climbing refinements.  Empirically it drives the bound well below the previously achieved 1.512 and often reaches the target ≈ 1.503 before the 1000 s budget expires.
