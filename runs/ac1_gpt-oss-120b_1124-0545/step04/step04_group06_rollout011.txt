def _make_basis(n):
    exponents = np.arange(0.2,0.9+0.05,0.1)   # 0.2,...0.9
    offsets = [0,5,10,20,30,50,100,200]
    exp_rs = [0.90,0.94,0.97,0.99]

    bases = []
    for offset in offsets:
        for alpha in exponents:
            b = _power_sequence(n, alpha, offset)
            bases.append(b)
    for offset in exp_offsets:
        for r in exp_rs:
            b = _exp_sequence(n, r, offset)
            bases.append(b)

   and a few linear shapes).  Any feasible distribution can be written as
   a convex combination of these bases:
       a = B @ w ,   w ≥ 0 , Σ w = 1 .
   Optimise the weight vector `w` with projected gradient descent on the
   exact max‑convolution objective.  Because the dimension of `w` is only
   a few dozen, many thousand iterations fit easily into the time budget.

3) **Fine‑grained sub‑gradient refinement** – start from the best mixture
   obtained above and run a few hundred thousand projected sub‑gradient
   steps on the full‑length vector (the same update used in the previous
   state‑of‑the‑art code).  Random mass‑swaps are injected occasionally to
   escape plateaus.

All helper routines are top‑level; the code respects the global time
budget (default 1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n) algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using FFT (fast for lengths ≥ ~1 000)."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """
    Normalised power‑law: a_i ∝ (i+1+offset)^(-α), i ≥ offset.
    Result sums to 1 and is non‑negative.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """
    Normalised exponential decay: a_i ∝ r^{i‑offset} for i ≥ offset,
    with 0 < r < 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """
    Simple decreasing linear (triangular) shape: a_i = n‑i for i≥offset.
    Normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when ``sum(a) = 1``:
    2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


def _build_basis(n: int) -> np.ndarray:
    """
    Construct a modest basis of candidate shapes.
    Returns a matrix B of shape (n, K) where each column is a
    normalised distribution (sum = 1).  The column count K is
    intentionally kept small (≈70‑80) so that weight optimisation is cheap.
    """
    bases = []

    # power‑law family, many exponents & offsets
    exponents = np.arange(0.20, 0.91, 0.10)          # 0.20 … 0.90
    offsets = [0, 5, 10, 20, 30, 50, 100, 200]     # leading zeros
    for off in offsets:
        for al in exponents:
            bases.append(_power_sequence(n, al, off))

    # exponential decays, a few rates and offsets
    rs = [0.90, 0.94, 0.97, 0.99]
    exp_offsets = [0, 5, 20, 50]
    for off in exp_offsets:
        for r in rs:
            bases.append(_exp_sequence(n, r, off))

    # linear decreasing with a few offsets
    for off in [0, 10, 30, 100]:
        bases.append(_linear_sequence(n, off))

    # global variable, if present – useful as a seed
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if init.size == n:
            b = np.maximum(init, 0.0)
            if b.sum() == 0.0:
                b = np.full(n, 1.0 / n)
            else:
                b /= b.sum()
            bases.append(b)
    except Exception:
        pass

    B = np.column_stack(bases)   # shape (n, K)
    return B


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimizes ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for finalisation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (cheap but surprisingly effective)
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

    # candidate lengths (keep ≤ 8000 for speed)
    length_candidates = [4000, 5000, 6000, 8000]

    # grids for families
    power_alphas = np.arange(0.20, 0.91, 0.10)
    power_offsets = [0, 5, 10, 20, 30, 50, 100, 200]
    exp_rs = [0.90, 0.94, 0.97, 0.99]
    exp_offsets = [0, 5, 20, 50]
    linear_offsets = [0, 10, 30, 100]

    for n in length_candidates:
        if time.time() > deadline:
            break

        # ---- Power‑law -------------------------------------------------
        for off in power_offsets:
            if off >= n:
                continue
            for al in power_alphas:
                a = _power_sequence(n, al, off)
                sc = _objective(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[sweep] n={n} power α={al:.2f} off={off} score={sc:.6f}")

        # ---- Exponential ------------------------------------------------
        for off in exp_offsets:
            if off >= n:
                continue
            for r in exp_rs:
                a = _exp_sequence(n, r, off)
                sc = _objective(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[sweep] n={n} exp r={r:.2f} off={off} score={sc:.6f}")

        # ---- Linear ----------------------------------------------------
        for off in linear_offsets:
            if off >= n:
                continue
            a = _linear_sequence(n, off)
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[sweep] n={n} linear off={off} score={sc:.6f}")

        # ---- Random Dirichlet seeds ------------------------------------
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[sweep] n={n} random dirichlet score={sc:.6f}")

        # ---- Use global variable if size matches -----------------------
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                a = np.maximum(init, 0.0)
                if a.sum() == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a.sum()
                sc = _objective(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[sweep] n={n} global init score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Low‑dimensional mixture optimisation (weights on a basis)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform sequence
        n = 5000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    n = best_seq.size
    B = _build_basis(n)                # shape (n, K)
    K = B.shape[1]

    # initialise weight vector – start from uniform simplex
    w = np.full(K, 1.0 / K)
    # optionally project the starting vector onto the simplex (already is)
    w = _project_onto_simplex(w)

    # momentum for weights
    w_mom = np.zeros(K, dtype=float)

    # learning‑rate schedule for the weight space
    lr_w = 0.2
    lr_decay_w = 0.9997
    lr_min_w = 1e-5

    # tracking best found in this phase
    best_score_w = best_score
    best_seq_w = best_seq.copy()

    it = 0
    while time.time() < deadline:
        it += 1

        # current mixture
        a = B @ w                     # length‑n vector
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # ---- gradient of the max‑convolution w.r.t. the weights ----------
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        v = np.zeros_like(a)
        v[mask] = a[idx[mask]]               # v_j = a_{k*‑j}
        grad_w = 2.0 * (B.T @ v)             # ∇_w max ≈ 2 Bᵀ v

        # momentum + projection
        w_mom = 0.9 * w_mom + grad_w
        w = w - lr_w * w_mom
        w = _project_onto_simplex(w)

        # decay learning rate
        lr_w = max(lr_w * lr_decay_w, lr_min_w)

        # occasional random jitter to escape flat plateaus
        if it % 150 == 0:
            i, j = np.random.choice(K, 2, replace=False)
            delta = min(w[i] * 0.2, lr_w * 0.5)
            if delta > 1e-12:
                w[i] -= delta
                w[j] += delta
                w = _project_onto_simplex(w)

        # evaluate periodically
        if it % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score_w:
                best_score_w = cur_score
                best_seq_w = a.copy()
                print(f"[mix] iter={it} n={n} score={cur_score:.6f}")
            # early exit if we already beat the target
            if best_score_w < 1.503:
                print("[mix] target reached – breaking")
                break

    # ------------------------------------------------------------------
    # 3. Fine‑grained sub‑gradient descent on the best full‑vector
    # ------------------------------------------------------------------
    a = best_seq_w.copy()
    best_seq_final = a.copy()
    best_score_final = best_score_w

    # momentum for the full‑vector updates
    mom = np.zeros_like(a)

    lr = 0.8
    lr_decay = 0.9999
    lr_min = 1e-6

    iter2 = 0
    while time.time() < deadline:
        iter2 += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # sub‑gradient (as used in the reference implementation)
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum + projection
        mom = 0.9 * mom + grad
        a = a - lr * mom
        a = _project_onto_simplex(a)

        lr = max(lr * lr_decay, lr_min)

        # random mass‑swap every few hundred steps
        if iter2 % 70 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.3, lr * 0.5)
            if delta > 1e-12:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # periodic exact evaluation
        if iter2 % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score_final:
                best_score_final = cur_score
                best_seq_final = a.copy()
                print(f"[refine] iter={iter2} n={n} score={cur_score:.6f}")

            if best_score_final < 1.503:
                print("[refine] target reached – breaking")
                break

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score_final:.6f} (n={best_seq_final.size}) "
          f"mix_iter={it} refine_iter={iter2}")

    # The harness will call evaluate_sequence on the returned list;
    # we keep the vector normalised (sum ≈ 1) which satisfies all checks.
    return best_seq_final.tolist()
