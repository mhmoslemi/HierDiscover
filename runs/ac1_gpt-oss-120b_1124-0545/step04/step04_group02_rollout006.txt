"""
Smooth‐max gradient descent for the step‑function problem.

The code starts from the best known construction (the global variable
`height_sequence_1` if it exists) or from a simple analytic U‑shaped
family, normalises the vector to sum 1 and then runs a projected gradient
descent on the smooth approximation of the objective  

    f(a) = (1/τ)·log Σ_k exp(τ·(a⋆a)_k) ,

which converges to the true max when τ→∞.  The gradient is obtained
efficiently by a single FFT‑based correlation.  The step size and the
smoothing parameter τ follow linear schedules, a Metropolis acceptance
criterion adds occasional uphill moves, and jitter / random restarts
prevent stagnation.  In the last few seconds a fine‑grained Gaussian
refinement is performed.  The routine respects a wall‑time budget and
returns the best sequence (a plain list of non‑negative floats) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a
    normalised to sum 1.  Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ ((i+off)*(n‑1‑i+off))^{-p} (U‑shaped, symmetric).
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def correlation_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_j w_{i+j} a_j  (i = 0 … n‑1) using FFT.
    This is the gradient of the smooth max objective w.r.t. a.
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)        # linear correlation length
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_w * np.conj(fft_a), n=L)
    # linear correlation gives array of length len(w)+len(a)-1 = 3n‑2 ;
    # the needed slice is the first n entries reversed.
    return 2.0 * corr[:n][::-1]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm:
        1. Initialise from a known good construction or a simple analytic family.
        2. Run projected gradient descent on a smooth approximation of the max.
           – τ (smoothness) grows from τ_min to τ_max.
           – Step size decays like 1/√iteration.
           – Metropolis acceptance with a temperature schedule.
           – Stagnation triggers a small jitter, occasional random Dirichlet restarts.
        3. Final fine‑grained Gaussian refinement.
    All numbers are kept non‑negative and projected onto the simplex (sum = 1),
    making the objective scale‑invariant.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for the harness

    # ------------------------------------------------------------------
    # 1) Initialisation
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # Prefer the globally supplied reference if its length matches n
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            a0 = np.maximum(ref, 0.0)
            a0 = a0 / a0.sum()
            print("[init] Using global reference as start.")
        else:
            a0 = None
    else:
        a0 = None

    # Fallback: simple U‑shaped analytic family (p≈0.6 gives good start)
    if a0 is None:
        a0 = analytic_product(n, p=0.6, offset=0.1)
        print("[init] Using analytic product family as start.")

    # Normalise to sum 1 (project_simplex also guarantees this)
    a = project_simplex(a0.astype(np.float64))

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed max gradient descent
    # ------------------------------------------------------------------
    iteration = 0
    last_improve = start_time

    # Schedules
    tau_min, tau_max = 0.5, 50.0          # smoothness schedule
    temp_start, temp_end = 0.5, 0.0      # Metropolis temperature schedule
    stagnation_time = 15.0               # seconds without improvement → jitter

    while time.time() < deadline:
        # --- current convolution ---
        conv = conv_fft(a)                # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:07d}] new best = {best_score:.6f}")

        # --- smooth max (softmax) ---
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)   # keep a few seconds for refinement
        tau = tau_min + (tau_max - tau_min) * progress
        # stable softmax
        conv_max = conv.max()
        exp_vals = np.exp(tau * (conv - conv_max))
        w = exp_vals / exp_vals.sum()            # shape (2n‑1,)

        # --- gradient = 2 * correlation(w, a) ---
        grad = correlation_w_a(w, a)

        # --- step size schedule (1/√iteration) ---
        step = 0.5 / (math.sqrt(iteration + 1) + 1.0)

        # --- proposal and projection ---
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        # --- evaluate candidate (exact objective) ---
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # --- Metropolis acceptance ---
        temp = temp_start * (1.0 - progress) + temp_end * progress
        accept = False
        if cand_score <= cur_score:
            accept = True
        elif temp > 1e-12:
            prob = math.exp(-(cand_score - cur_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand

        # --- occasional random mass‑transfer move (exploration) ---
        if np.random.rand() < 0.02:
            a = random_move(a, step * 0.5)

        # --- rare Dirichlet restart (global jump) ---
        if np.random.rand() < 0.005:
            a = np.random.rand(n)
            a = project_simplex(a)
            print(f"[{iteration:07d}] random Dirichlet restart")

        # --- jitter restart after stagnation ---
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # 3) Fine‑grained stochastic refinement (last ~4 s)
    # ------------------------------------------------------------------
    refine_deadline = start_time + budget_s - 2.0
    while time.time() < refine_deadline:
        pert = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + pert
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
