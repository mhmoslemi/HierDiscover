"""
Propose a low‑discrepancy step‑function using a multi‑scale
optimization pipeline.

The pipeline works as follows (all time‑aware, stops at the deadline):

1. **Coarse scale search** – enumerate a few small lengths (500, 1000,
   2000).  For each length we generate a rich collection of analytic
   seeds (uniform, geometric, power‑law, sine‑window, beta‑like,
   Dirichlet draws and the previously best construction).  Every seed
   is immediately evaluated and the best one is kept.

2. **Smooth‑max Adam / L‑BFGS refinement** – each promising seed is
   turned into log‑softmax parameters `v` (so the coefficients lie on
   the probability simplex).  We minimise the smooth approximation
   `L_β(a)= (1/β)·log Σ exp(β·conv(a,a))` of the max‑convolution.
   The optimisation first runs Adam with an increasing smoothing
   coefficient (β grows from 2 up to ≈100).  Optionally, a short L‑BFGS
   run (scipy) is performed for a more accurate local step.

3. **Exact sub‑gradient mirror descent** – starting from the best
   Adam‑result we perform exponentiated‑gradient (mirror descent) steps
   using the true sub‑gradient of the max‑convolution.  The step size
   decays as 1/√t.  After each update we enforce *symmetry* (`a←(a+a[::-1])/2`)
   which reduces the search space and matches the known optimal shape
   symmetry.

4. **Stochastic hill‑climbing with annealed step‑size** – a cheap
   neighbourhood move transfers a tiny amount of mass between two
   random positions; the move is kept if it improves the true
   objective `2·n·max(conv)`.  Accepted states are again symmetrised.
   The transfer magnitude decays slowly.

5. **Scale‑up** – the best shape for a given length is linearly
   interpolated to the next larger length and used as a seed, so the
   optimisation of larger `n` starts from a high‑quality approximation.

The whole routine respects the supplied time budget (default 1000 s)
and returns the best sequence found as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s != 0 else np.full_like(v, 1.0 / len(v))


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i]=a[-i-1] while preserving the total sum."""
    a_sym = (a + a[::-1]) * 0.5
    # Re‑normalise to avoid tiny drift due to rounding.
    s = a_sym.sum()
    if s == 0:
        return a_sym
    return a_sym / s

# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max of the auto‑convolution.
    Returns the best (probability) vector and its true objective.
    """
    # log‑softmax parameters
    v = np.log(seed + 1e-30)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters (tuned for this problem)
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # smoothing coefficient (starts low, grows to make the loss sharper)
    beta = 2.0
    beta_mul = 1.04

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} val={cur_val:.6f}")

        # --- smooth‑max weights ---
        beta *= beta_mul
        shift = np.max(conv)               # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)        # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # ensure symmetry before returning (helps later stages)
    if best_seq is not None:
        best_seq = _symmetrize(best_seq)

    return best_seq, best_val


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule: list[float] = (1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0),
                    iters_per_stage: int = 250) -> tuple[np.ndarray, float]:
    """
    Optional L‑BFGS refinement on the smooth max.
    Uses the same soft‑max parametrisation as Adam.
    """
    try:
        from scipy.optimize import minimize
    except Exception as exc:                     # scipy not available – fall back
        print("[LBFGS] SciPy not available:", exc)
        return None, float("inf")

    v = np.log(seed + 1e-30)
    best_seq = None
    best_val = float("inf")

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def loss_and_grad(v_flat):
            a = _softmax(v_flat)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # Smooth max: (1/β)·log ∑ exp(β·conv)
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            loss = shift + np.log(w_sum) / beta

            # Gradient w.r.t a
            w_norm = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / len(w))
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w_norm)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Gradient w.r.t v (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return loss, grad_v

        # L‑BFGS‑B – we only bound v implicitly (no explicit bounds)
        res = minimize(fun=lambda v_: loss_and_grad(v_)[0],
                       x0=v,
                       jac=lambda v_: loss_and_grad(v_)[1],
                       method='L-BFGS-B',
                       options={'maxiter': iters_per_stage, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False})
        v = res.x
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LBFGS] n={n:5d} β={beta:.1f} val={cur_val:.6f}")

    if best_seq is not None:
        best_seq = _symmetrize(best_seq)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement with the exact sub‑gradient of max(conv).
    Uses exponentiated‑gradient (multiplicative) updates and enforces symmetry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} val={cur_val:.6f}")

        # sub‑gradient of the (unique) maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step with decaying step size
        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        # numerical stability
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # enforce symmetry
        a = _symmetrize(a)

    if best_seq is not None:
        best_seq = _symmetrize(best_seq)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a geometrically decaying step size.
    Maintains symmetry after each accepted move.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    # initial relative step (fraction of the donor's mass)
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # pick donor & receiver uniformly
        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        # re‑normalise and symmetrise (maintains sum=1)
        new /= new.sum()
        new = _symmetrize(new)

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} val={val:.6f}")

        # decay step size slowly
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    return best_seq, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:

    seeds: list[np.ndarray] = []

    # elementary families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws

        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered (or up‑scaled) construction
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Ensure symmetry for all seeds (helps the later stages)
    seeds = [_symmetrize(s) for s in seeds]

    return seeds


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a larger length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    # Interpolate on the interval [0, 1]
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    # Re‑normalise and symmetrise
    up /= up.sum()
    up = _symmetrize(up)
    return up

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Re‑use a globally stored best construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Multi‑scale lengths – start small, finish with a long one
    lengths = [500, 1000, 2000, 4000, 8000]

    rng = np.random.default_rng(seed + 9999)   # independant generator for seeds

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # Upscale the current best (if any) to serve as a strong seed
        upscaled_prev = None
        if global_best_seq is not None:
            upscaled_prev = _upscale(global_best_seq, n)

        seeds = _make_seeds(n, upscaled_prev, rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimiser with smooth‑max
        # ------------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

            # Occasionally run a short L‑BFGS refinement on the very best seed
            # (only once per length to keep the cost modest)
            if global_best_seq is not None and np.random.rand() < 0.07:
                seq2, val2 = _lbfgs_optimize(global_best_seq, n, deadline)
                if seq2 is not None and val2 < global_best_val:
                    global_best_val = val2
                    global_best_seq = seq2.copy()
                    print(f"[overall‑LBFGS] n={n:5d} new global best={val2:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (mirror descent)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=300_000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Stochastic hill‑climbing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                         max_iters=150_000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid result
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Final sanity check (the harness will also call evaluate_sequence)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
