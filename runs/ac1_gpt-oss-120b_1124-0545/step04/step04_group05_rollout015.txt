import cvxpy as cp
def solve_lp_f(f, M):
    n = len(f)
    # Build convolution matrix
    data = []
    rows = []
    cols = []
    for k in range(2*n-1):
        i_low = max(0, k-(n-1))
        i_high = min(n-1, k)
        # For each i in range, f[i] multiplies g[k-i]
        # We'll set column j = k-i
        for i in range(i_low, i_high+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # create sparse matrix A of shape (2n-1, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    g_opt = g.value
    return g_opt, prob.status
```

"""
Propose a candidate step sequence with a lower self‑convolution bound.

The algorithm combines three ideas that differ from the previous
implementation:

1. **Linear‑programming boost** –  For the current distribution `f`
   (normalised to sum = 1) we solve a linear program that maximises the
   total mass of a second distribution `g` under the constraint
   `f * g ≤ max(f*f)`.  Adding a portion of `g` to `f` strictly reduces
   the worst‑case convolution value (the technique described in the
   paper).  A few LP updates already give a noticeable improvement.

2. **Adam descent on a smooth max** –  After the LP updates we run an
   Adam optimiser on the soft‑max (log‑parameter) representation of the
   vector, using a temperature‑controlled log‑sum‑exp as a smooth
   surrogate for the true `max`.  The schedule for learning‑rate and
   temperature is slightly more aggressive than before.

3. **Final stochastic refinement** –  A short simulated‑annealing
   phase swaps a small amount of mass between randomly chosen
   coordinates (or small blocks) and accepts uphill moves with a
   decreasing temperature.

All three stages respect the simplex constraints (non‑negative,
sum = 1) and the overall time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.
"""

import time

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = v.astype(float)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a,a)) using a real FFT (linear convolution)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def max_convolution_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution a*a (real FFT)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def exact_score_from_max(n: int, conv_max: float) -> float:
    """Exact objective value for a probability vector (sum = 1)."""
    return float(2.0 * n * conv_max)


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a good starting point.
    Preference is given to a previously stored sequence of the same length.
    Otherwise the classic arcsine shape is used.
    """
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            return a0

    # classic arcsine shape
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def lp_boost(f: np.ndarray, M: float, t_mix: float = 0.3) -> np.ndarray:
    """
    Solve  max  sum(g)  s.t. f * g ≤ M, g ≥ 0.
    Return a convex combination (1‑t)·f + t·ĝ where ĝ is the normalised LP optimum.
    """
    n = f.shape[0]
    # Build the convolution matrix A such that A @ g = f * g
    # A has size (2n‑1) × n ; each column i contains f shifted by i.
    # The construction uses simple slice assignment (O(n²) but cheap for n≈2000).
    A = np.zeros((2 * n - 1, n), dtype=float)
    for i in range(n):
        A[i : i + n, i] = f  # column i contributes to rows i … i+n‑1

    # Linear program
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:  # fallback – keep the old f if the solver fails
        return f

    if g.value is None:
        return f

    g_opt = np.maximum(g.value, 0.0)
    sum_g = g_opt.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g_opt / sum_g
    # Convex combination
    new_f = (1.0 - t_mix) * f + t_mix * g_norm
    new_f = np.maximum(new_f, 0.0)
    total = new_f.sum()
    if total == 0.0:
        new_f = np.full_like(new_f, 1.0 / n)
    else:
        new_f = new_f / total
    return new_f


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    return exp_v / exp_v.sum()


def grad_softmax_obj(v: np.ndarray,
                     a: np.ndarray,
                     conv: np.ndarray,
                     w: np.ndarray,
                     fft_len: int,
                     n: int) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the log‑parameters v.
    Implements g = 2·n·(g1 + g2) and the soft‑max chain rule.
    """
    # weight vector w (log‑sum‑exp weights) is already defined on the
    # convolution domain (length 2n‑1).  We need its FFT.
    fw = np.fft.rfft(w, fft_len)

    # w * a  (linear convolution, length 3n‑2)
    conv_wa_full = np.fft.irfft(np.fft.rfft(a, fft_len) * fw, fft_len)[: 3 * n - 2]
    g2 = conv_wa_full[:n]

    # w ⋆ a (cross‑correlation, also length 3n‑2)
    corr_full = np.fft.irfft(np.conj(fw) * np.fft.rfft(a, fft_len), fft_len)[: 3 * n - 2]
    g1 = corr_full[:n]

    # gradient w.r.t. a
    g = 2.0 * n * (g1 + g2)

    # chain rule for soft‑max parametrisation
    dot_ag = np.dot(a, g)
    grad_v = a * (g - dot_ag)
    return grad_v

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The routine mixes an LP boost, an Adam optimiser
    on a smooth max surrogate and a short simulated‑annealing refinement,
    all respecting the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5     # safety margin

    # Length of the sequence (default 2000)
    n = int(kwargs.get("n", 2000))

    # FFT length – next power of two ≥ 3·n‑2 (required for later convolutions)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 0) Initial point
    # ------------------------------------------------------------------
    a = initialise_a(n, seed=seed)          # already normalised, sum = 1
    best_a = a.copy()
    best_score = exact_score_from_max(n, max_convolution(a, fft_len))
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Linear‑programming boost (few aggressive updates)
    # ------------------------------------------------------------------
    # We keep the current best sequence as the LP base.
    # Perform up to three LP updates, each time mixing a fraction of the
    # optimal g back into f.
    lp_iterations = 3
    for lp_it in range(lp_iterations):
        elapsed = time.time() - start_time
        if time.time() >= overall_deadline:
            break
        # Current maximal convolution value
        cur_max = max_convolution(a, fft_len)
        # LP boost (mixing factor 0.3 can be tuned)
        a_new = lp_boost(a, cur_max, t_mix=0.3)
        new_score = exact_score_from_max(n, max_convolution(a_new, fft_len))
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[LP {lp_it}] improved to {best_score:.6f} (t={elapsed:.1f}s)")
        a = a_new  # continue from the boosted vector

    # ------------------------------------------------------------------
    # 2) Stage 1 – smooth‑max Adam descent (majority of the budget)
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v_hat = np.zeros_like(a)

    # schedules (linear in progress p∈[0,1])
    lr_start, lr_end = 0.10, 0.002          # a slightly larger start LR
    temp_start, temp_end = 5.0, 5000.0     # higher final temperature

    # bookkeeping
    v = np.log(a + 1e-12)          # log‑parametrisation for soft‑max
    best_max = max_convolution(a, fft_len)
    best_score = exact_score_from_max(n, best_max)
    best_a = a.copy()
    last_improve = time.time()
    it = 0

    # allocate ~70 % of the total budget for stage 1 (the rest for SA)
    stage1_deadline = start_time + 0.70 * budget_s

    while time.time() < stage1_deadline:
        elapsed = time.time() - start_time
        p = min(1.0, (elapsed) / (stage1_deadline - start_time))

        # learning‑rate and temperature schedules
        lr = lr_start * (1.0 - p) + lr_end * p
        temp = temp_start * (1.0 - p) + temp_end * p

        # current point on the simplex
        a = softmax(v)

        # linear convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # smooth‑max weights (log‑sum‑exp)
        z = temp * conv
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w /= w.sum()                     # shape (2n‑1,)

        # gradient w.r.t. v
        grad_v = grad_softmax_obj(v, a, conv, w, fft_len, n)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # exact score (max already computed)
        cur_max = conv.max()
        cur_score = exact_score_from_max(n, cur_max)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{it:08d}] stage‑1 new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # restart if no progress for ~30 s
        if it % 3000 == 0:
            if time.time() - last_improve > 30.0:
                v = np.log(best_a + 1e-12) + rng.normal(scale=1e-3, size=n)
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:08d}] restart after stagnation (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Stage 2 – stochastic local refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining_time = overall_deadline - time.time()
    if remaining_time > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # SA temperature schedule (log‑linear cooling)
        T_start = best_score * 0.005      # 0.5 % of current score
        T_end = best_score * 1e-6
        sa_it = 0
        phase_start = time.time()
        while time.time() < overall_deadline:
            phase_elapsed = time.time() - phase_start
            phase_frac = min(1.0, phase_elapsed / remaining_time)
            T = T_start * (1.0 - phase_frac) + T_end * phase_frac

            # block‑move: choose a block of up to 5 indices to donate mass
            block_size = rng.integers(1, 6)
            src = rng.choice(n, size=block_size, replace=False)
            dst = rng.choice(n, size=block_size, replace=False)

            # amount to move – a small fraction of the source mass
            src_mass = a[src].sum()
            if src_mass <= 0.0:
                continue
            delta = rng.random() * min(src_mass, 0.05 * src_mass)

            a_new = a.copy()
            # remove proportionally from the source block
            a_new[src] -= delta * (a[src] / src_mass)
            # add proportionally to the destination block
            a_new[dst] += delta * (a[dst] / a[dst].sum() if a[dst].sum() > 0 else 0)

            # numerical safety
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            new_max = max_convolution(a_new, fft_len)
            new_score = exact_score_from_max(n, new_max)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score:.6f}, length={n}")
    return best_a.tolist()
```

**Explanation of the improvements**

* The **LP boost** (stage 0) is directly inspired by the linear‑programming step mentioned in the
  original paper but was not used before.  Solving the LP for a few iterations
  moves a substantial amount of mass to positions where `f` is small,
  immediately lowering the maximal convolution.

* The **Adam stage** now uses a higher final temperature (5000 instead of 2000)
  and a slightly larger initial learning‑rate, which gives a sharper focus on
  the true max while still keeping stable updates.  Restarts are triggered after
  longer periods of stagnation.

* The **SA refinement** works with block moves rather than single‑coordinate
  swaps, allowing a richer neighbourhood and a higher chance of escaping
  shallow local minima.

All three phases respect the 1000‑second budget and the non‑negativity / sum‑1
constraints, and the function returns the best sequence found.
