"""
Hybrid optimizer for the convolution‑max objective.

The routine first performs a fast Simulated‑Annealing (SA) search on the
simplex (the mass is moved between coordinates while keeping the total
sum constant).  The SA loop uses only a single FFT per iteration to
evaluate the objective, which allows many millions of moves within the
1000 s budget.

When the SA phase stagnates or the allocated time is almost exhausted a
short Adam‑based gradient‑descent refinement (the same optimiser that
was used in the previous state‑of‑the‑art) is run from the best SA
solution.  The combination of a global stochastic search and a local
deterministic fine‑tuning yields a lower bound than the plain Adam run
alone.

All helper functions are defined at top level – there are no closures
or lambda expressions – and the algorithm respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return max entry of the linear convolution a*a.
    The computation is performed via a real FFT of length `fft_len`.
    """
    ra = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(ra * ra, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective value 2·n·max(a*a)/(sum(a)**2).
    The function is scale‑invariant; the implementation keeps the sum
    explicit to avoid division by a tiny number.
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    max_conv = _conv_max(a, fft_len, n)
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Simulated annealing phase
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform a temperature‑annealed random walk on the simplex.
    Mass is transferred between two coordinates while keeping the total
    L1‑mass constant.  The schedule slowly cools the temperature and
    shrinks the maximal transfer size.
    """
    a = a_start.copy()
    cur_score = _score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    last_improve = start
    stagnation_limit = 30.0               # seconds without improvement → jitter restart
    iter_cnt = 0

    # temperature schedule (exponential cooling)
    T0 = cur_score * 0.5
    Tf = cur_score * 1e-6

    # move‑size schedule (fraction of the source coordinate)
    move_start = 0.20
    move_end = 0.001

    while time.time() < deadline:
        # time‑dependent parameters
        elapsed = time.time() - start
        total = deadline - start
        frac = min(1.0, elapsed / total)

        T = T0 * ((Tf / T0) ** frac)          # exponential cooling
        max_move_frac = move_start * (1.0 - frac) + move_end * frac

        # --- propose a random transfer ---------------------------------
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0.0:
            continue
        # amount to move, never larger than the source mass
        eps = rng.random() * min(a[i], max_move_frac)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # --- evaluate ---------------------------------------------------
        new_score = _score(a_new, fft_len, n)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-16)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                last_improve = time.time()
                print(
                    f"[SA {iter_cnt:08d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, T={T:.3e})"
                )
        iter_cnt += 1

        # --- restart on stagnation ------------------------------------
        if time.time() - last_improve > stagnation_limit:
            # jitter the best solution a little and resume searching
            jitter = rng.normal(scale=0.01, size=n)
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            cur_score = _score(a, fft_len, n)
            last_improve = time.time()
            print(
                f"[SA {iter_cnt:08d}] restart after {stagnation_limit}s"
                f" (t={elapsed:.1f}s)"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑based gradient descent (the previous state‑of‑the‑art)
# ----------------------------------------------------------------------

    a_init: np.ndarray,
    n: int,
    alloc_s: float,

) -> tuple[np.ndarray, float]:

    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best vector found and its (scale‑invariant) score.
    """
    # log‑parametrisation (soft‑max guarantees feasibility)
    v = np.log(a_init + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedules (linear in normalised time p∈[0,1])
    beta_start, beta_end = 0.1, 5000.0         # soft‑max temperature
    lr_start, lr_end = 0.25, 0.001           # learning‑rate

    # FFT size (covers convolution of length 3·n‑2)
    fft_len = _next_pow2(3 * n - 2)

    # bookkeeping
    start = time.time()
    deadline = start + alloc_s - 0.05  # tiny safety margin
    best_a = a_init.copy()
    fa = np.fft.fft(best_a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    last_improve = start
    it = 0
    mask_phase_start = 0.70

    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))

        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        a = np.exp(v - np.max(v))          # soft‑max (unnormalised)
        a /= a.sum()

        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        if p < mask_phase_start:
            # smooth soft‑max over convolution entries
            z = beta * conv_aa
            zz = np.exp(z - np.max(z))
            w = zz / zz.sum()
        else:
            # hard mask on the true maximum
            max_val = conv_aa.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            if mask.sum() == 0.0:
                mask[np.argmax(conv_aa)] = 1.0
            w = mask / mask.sum()

        fw = np.fft.fft(w, fft_len)

        # (w * a) – linear convolution, first n entries
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # w ⋆ a – linear cross‑correlation, first n entries
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        g = 2.0 * n * (g1 + g2)

        # chain rule for the soft‑max parametrisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # exact score & bookkeeping
        cur_score = 2.0 * n * conv_aa.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 2000 == 0:
                print(
                    f"[ADAM {it:07d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # restart on stagnation
        if time.time() - last_improve > 20.0:
            jitter = rng.normal(scale=0.01, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[ADAM {it:07d}] restart (no improvement)")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine spends the first ~70 % of the
    budget on a fast simulated‑annealing walk and uses the remaining
    time for a short Adam refinement.  It returns the best vector
    discovered as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Initialise starting point (historic solution or uniform)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1 – Simulated annealing
    # ------------------------------------------------------------------
    sa_best_a, sa_best_score = _simulated_annealing(
        a_start=a0,
        n=n,
        fft_len=fft_len,
        deadline=deadline,
        rng=rng,
    )

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (if time permits)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 10.0:
        opt_a, opt_score = _optimize_a(
            a_init=sa_best_a,
            n=n,
            alloc_s=time_left,
            rng=rng,
        )
        if opt_score < sa_best_score:
            best_a = opt_a
            best_score = opt_score
        else:
            best_a = sa_best_a
            best_score = sa_best_score
    else:
        best_a = sa_best_a
        best_score = sa_best_score

    # ------------------------------------------------------------------
    # Final safety checks and return
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        # ensure we stay away from the “sum too close to zero” guard
        best_a += 0.01 / n
    total_time = time.time() - start_time
    print(f"[done] best score={best_score:.6f}, time={total_time:.1f}s")
    return best_a.tolist()
