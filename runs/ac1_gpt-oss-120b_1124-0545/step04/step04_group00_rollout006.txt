    cur = a.copy()
    cur_val = _objective(cur)
    best = cur.copy()

    decay = 0.9996
    it = 0

        it += 1
        T = T0 * (decay ** it)

        conv = _conv_fft(cur, cur)
        k_max = int(np.argmax(conv))

        # contributions
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        rev = k_max - idx
        contrib = cur[idx] * cur[rev]

1.  Build a diverse set of seeds (uniform, geometric, power‑law,
    symmetric windows, random soft‑max vectors).  Each seed is also
    enriched by its reversal and a random permutation.

2.  Smoothed‑max Adam optimisation – the loss is a log‑sum‑exp
    approximation of `max(conv(a,a))`.  The parameters are logits of a
    soft‑max, guaranteeing non‑negativity and unit sum (simplex).

3.  Exact sub‑gradient mirror‑descent – uses the true sub‑gradient of
    the maximum convolution entry and an exponentiated‑gradient update.

4.  Simple Metropolis annealing (mass‑transfer moves) – as in the
    previous baseline.

5.  **New “max‑reduce” annealer** – repeatedly identifies the lag at
    which the convolution attains its maximum, then moves a small
    amount of mass away from the most responsible index to a low‑mass
    index.  The move is accepted by a Metropolis rule with a cooling
    schedule.  Periodically a global Gaussian perturbation (or a random
    permutation) is tried.  This targeted move helps flatten the
    auto‑correlation more aggressively.

All phases stop as soon as the global deadline is reached.  The best
sequence discovered over all examined lengths is returned (as a plain
list of floats).  The sequence is stored in the global variable
`height_sequence_1` for possible reuse by a subsequent run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple objective (identical to evaluate_sequence up to clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²), seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Helper: numerically stable soft‑max (log‑space parametrisation)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families used as seeds
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of diverse, normalised seeds for length *n*."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # optionally reuse a previously discovered construction
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))
    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp smoothing of max(conv). Returns best seq."""
    v = np.log(seed + 1e-30)          # initialise logits

    m = np.zeros_like(v)              # first moment
    v2 = np.zeros_like(v)             # second moment

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule (beta grows → sharper)
    beta = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement using the true sub‑gradient of
    max(conv).  Uses exponentiated‑gradient updates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)                 # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()                           # projection onto simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (mass‑transfer moves)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that moves a tiny amount of mass between two
    random positions.  Temperature decays exponentially.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional “jump” to escape plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# New phase – targeted max‑reduction annealer
# ----------------------------------------------------------------------


def _max_reduce_anneal(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Aggressive annealer that repeatedly attacks the lag where the
    convolution is maximal.  It moves a tiny portion of mass away from the
    index that contributes most to that lag, accepting moves with a
    Metropolis probability.  Periodic global perturbations (Gaussian
    noise + symmetrisation) are also attempted.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9996

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        conv = _conv_fft(cur, cur)
        k_max = int(np.argmax(conv))

        # contributions to the maximal lag
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k_max - idx
        contrib = cur[idx] * cur[rev]

        # index that contributes most to the peak
        i_max = idx[np.argmax(contrib)]

        # pick a target index with small mass
        cand = rng.integers(0, n, size=5)
        cand = cand[cand != i_max]
        if cand.size == 0:
            j = (i_max + 1) % n
        else:
            j = cand[np.argmin(cur[cand])]

        # move a tiny fraction of mass from i_max to j
        delta = cur[i_max] * rng.random() * 0.02
        if delta <= 0.0:
            continue
        new_seq = cur.copy()
        new_seq[i_max] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val

            # enforce symmetry and renormalise
            cur = (cur + cur[::-1]) / 2.0
            cur /= cur.sum()
            cur_val = _objective(cur)

            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
                print(f"[MaxReduce] n={n} it={it:6d} best={cur_val:.6f}")

        # every few thousand iterations try a larger, global perturbation
        if it % 2000 == 0:
            # Gaussian perturbation + symmetrisation
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_seq = (new_seq + new_seq[::-1]) / 2.0
            new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[MaxReduce] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point (called by the grader)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0     # safety margin

    # reuse a previously computed construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to explore – start with 4000 and grow if time permits
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1) generate a modest but diverse seed set
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # quick evaluation of raw seeds
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(7, len(seed_vals))]]

        # augment the top seeds with reversed & randomly permuted copies
        rng = np.random.default_rng()
        augmented = []
        for s in top_seeds:
            augmented.append(s)                         # original
            augmented.append(s[::-1])                   # reversed
            perm = rng.permutation(n)
            augmented.append(s[perm])                   # shuffled
        top_seeds = augmented

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Adam optimisation on the smoothed max
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=8000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # --------------------------------------------------------------
        if (global_best_seq is not None and
                global_best_seq.shape[0] == n and
                time.time() < deadline):
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                          max_iters=60000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Simple annealing (mass‑transfer)
        # --------------------------------------------------------------
        if (global_best_seq is not None and
                global_best_seq.shape[0] == n and
                time.time() < deadline):
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Targeted max‑reduction annealer (new)
        # --------------------------------------------------------------
        if (global_best_seq is not None and
                global_best_seq.shape[0] == n and
                time.time() < deadline):
            seq, val = _max_reduce_anneal(global_best_seq, n, deadline,
                                          max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall MaxReduce] n={n} best={val:.6f}")

        # store the best of this length for possible reuse later
        if global_best_seq is not None:
            previous = global_best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # expose the best sequence for the next run
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
