The evaluation function is  

\[
\Phi(a)=\frac{2\,n\;\max_k\;(a*a)_k}{\Bigl(\sum_i a_i\Bigr)^2},
\qquad a_i\ge 0 .
\]

Because the expression is homogeneous we may **normalise** every candidate to a
probability vector (`∑a = 1`).  Then  

\[
\Phi(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is to find a non‑negative vector whose *self‑convolution* is as flat
as possible, i.e. minimise the ℓ∞‑norm of the sum‑distribution of two i.i.d.
copies of the discrete random variable defined by `a`.

The search consists of three phases

1. **Analytic families** – a cheap grid over several parametric families
   (`U`‑shape, power‑law, exponential tail, cosine‑power, Beta‑shape,
   Gaussian‑mixture).  
   The best vector from this stage is the start point for the optimisation.

2. **Smoothed max optimisation** – minimise a smooth approximation of the
   ℓ∞‑norm:

   \[
   L_\gamma(a)=\frac{1}{\gamma}\log\!\Bigl(\sum_k e^{\gamma\,(a*a)_k}\Bigr)
   \xrightarrow[\gamma\to\infty]{}\max_k (a*a)_k .
   \]

   The gradient of \(L_\gamma\) is
   \[
   \nabla L_\gamma = 2\;{\rm corr}(a,\;w),
   \qquad
   w_k=\frac{e^{\gamma (a*a)_k}}{\sum_j e^{\gamma (a*a)_j}} .
   \]

   `corr` denotes the linear cross‑correlation, which is obtained by a single
   FFT (linear convolution of `w` with the reversed `a`).  
   An Adam optimiser (first‑order momentum) together with a projection onto the
   probability simplex produces fast, stable updates.  The temperature parameter
   `γ` is increased logarithmically from a small start value to a large final
   value, gradually turning the smooth loss into the true max‑loss.

3. **Fine stochastic refinement** – after the main loop a short
   Gaussian‑perturbation stage tries very small local moves and keeps any
   improvement.

The algorithm works for any length `n`.  A short list of candidate lengths
(`1200 … 4000`) is examined; the one that produces the smallest score after the
analytic phase is taken for the intensive Adam optimisation.  All helper
functions are top‑level, there are no closures, and the code respects the 1000 s
wall‑time budget.

"""
Proposes a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of:
1. an analytic grid over several families;
2. a smoothed‑max Adam optimisation using FFT‑based gradients;
3. a final short Gaussian refinement.

All vectors are kept on the probability simplex (sum = 1, non‑negative) so
the objective is scale‑invariant.  The routine respects a 1000 s time budget.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary vectors a and b (zero‑padded)
    using FFT.  Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    total = a.sum()
    a = a / total
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(π i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    total = a.sum()
    a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α} * (n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    total = a.sum()
    a = a / total
    return a


def analytic_gaussian_mix(n: int, mu: float, sigma: float) -> np.ndarray:
    """A single Gaussian centred at mu with width sigma, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    total = a.sum()
    a = a / total
    return a

def smooth_max_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Returns the gradient of the smoothed max loss

        Lγ(a) = (1/γ) log Σ exp(γ·(a*a)_k)

    with respect to a.  Both a and the returned gradient are length n
    arrays that sum to zero (the gradient is later used inside Adam).
    """
    n = a.shape[0]

    # self‑convolution
    conv = conv_fft_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (avoid overflow)
    exp_arg = gamma * (conv - max_c)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2 * correlation(a, w)
    a_rev = a[::-1]
    conv_wr = conv_fft_two(w, a_rev)           # length 3n‑2
    start = n - 1                               # slice that aligns with a
    g = 2.0 * conv_wr[start:start + n]
    return g


def objective_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine respects a 1000 s wall‑time budget and returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # 1) Analytic grid search over several lengths
    # ------------------------------------------------------------------
    length_candidates = [1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]
    best_overall_seq = None
    best_overall_score = np.inf

    for n in length_candidates:
        if time.time() >= deadline - 5:
            break  # time is almost up

        # ---- candidate holder for this length -----------------------
        best_seq_n = None
        best_score_n = np.inf

        # uniform (baseline)
        cand = np.ones(n) / n
        sc = objective_score(cand)
        if sc < best_score_n:
            best_score_n, best_seq_n = sc, cand.copy()

        # product family
        for p in (0.5, 1.0, 1.5, 2.0):
            for off in (0.5, 1.0, 2.0):
                cand = analytic_product(n, p, off)
                sc = objective_score(cand)
                if sc < best_score_n:
                    best_score_n, best_seq_n = sc, cand.copy()

        # sum family
        for p in (0.5, 1.0, 1.5):
            for off in (0.5, 1.0, 2.0):
                cand = analytic_sum(n, p, off)
                sc = objective_score(cand)
                if sc < best_score_n:
                    best_score_n, best_seq_n = sc, cand.copy()

        # exponential tails
        for lam in (0.01, 0.1, 0.5, 1.0):
            cand = analytic_exponential(n, lam)
            sc = objective_score(cand)
            if sc < best_score_n:
                best_score_n, best_seq_n = sc, cand.copy()

        # cosine‑power family
        for p in (0.5, 1.0, 2.0, 3.0, 4.0):
            cand = analytic_cosine(n, p)
            sc = objective_score(cand)
            if sc < best_score_n:
                best_score_n, best_seq_n = sc, cand.copy()

        # beta shape family
        for a in (-0.5, 0.0, 0.5, 1.0):
            for b in (-0.5, 0.0, 0.5, 1.0):
                cand = analytic_beta(n, a, b)
                sc = objective_score(cand)
                if sc < best_score_n:
                    best_score_n, best_seq_n = sc, cand.copy()

        # Gaussian mixtures (centred near the edges)
        for mu in (0.0, n - 1):
            for sigma in (n * 0.08, n * 0.15):
                cand = analytic_gaussian_mix(n, mu, sigma)
                sc = objective_score(cand)
                if sc < best_score_n:
                    best_score_n, best_seq_n = sc, cand.copy()

        # incorporate the global reference if present
        if "height_sequence_1" in globals():
            ref = np.asarray(globals()["height_sequence_1"], dtype=float)
            if ref.shape[0] == n:
                ref = np.maximum(ref, 0.0)
                ref = ref / ref.sum()
                sc = objective_score(ref)
                if sc < best_score_n:
                    best_score_n, best_seq_n = sc, ref.copy()

        print(f"[init n={n}] best analytic score = {best_score_n:.6f}")

        # ----------------------------------------------------------------
        # 2) Smoothed‑max Adam optimisation for this length
        # ----------------------------------------------------------------
        a = best_seq_n.copy()                     # currently on simplex
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        t_iter = 0
        last_improve = time.time()
        stagnation_limit = 12.0                    # seconds without improvement

        # schedule parameters
        gamma_start, gamma_end = 0.5, 500.0        # log‑linear increase
        lr_start, lr_end = 0.6, 5e-5              # linear decay

        while time.time() < deadline - 5:
            t_iter += 1
            elapsed = time.time() - start_time
            progress = min(elapsed / budget_s, 1.0)

            # learning‑rate and γ schedules
            lr = lr_start * (1.0 - progress) + lr_end * progress
            gamma = gamma_start * (gamma_end / gamma_start) ** progress

            # current objective
            conv = conv_fft_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_overall_score - 1e-12:
                best_overall_score = cur_score
                best_overall_seq = a.copy()
                last_improve = time.time()
                print(f"[{t_iter:07d}] n={n} new best = {best_overall_score:.6f}")

            # gradient of the smoothed max
            g = smooth_max_gradient(a, gamma)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = project_simplex(a_new)

            # occasional random mass‑transfer move to improve exploration
            if np.random.rand() < 0.01:
                a_new = random_move(a_new, scale=lr)

            a = a_new

            # jitter restart after stagnation
            if time.time() - last_improve > stagnation_limit:
                jitter = np.random.uniform(-1e-4, 1e-4, size=n)
                a = best_overall_seq + jitter
                a = project_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()
                print(f"[{t_iter:07d}] n={n} jitter restart (stagnation)")

            # early exit if we already reached the target bound
            if best_overall_score <= 1.503:
                print("[early] target reached")
                break

        # store the best of this length
        if best_overall_score < best_overall_score:
            best_overall_score = best_overall_score
            best_overall_seq = best_seq_n.copy()

        # break the outer length loop if target met
        if best_overall_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3) Final fine‑grained stochastic refinement (last few seconds)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – should never happen, but keep a safe default
        best_overall_seq = np.ones(2000) / 2000.0
        best_overall_score = objective_score(best_overall_seq)

    n_final = len(best_overall_seq)
    refine_deadline = start_time + budget_s - 1.0
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=5e-4, size=n_final)
        cand = best_overall_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cur_score = objective_score(cand)
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = cand.copy()
            print(f"[refine] improved to {best_overall_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_overall_score:.6f}, length = {len(best_overall_seq)}, time = {total_time:.1f}s")
    return best_overall_seq.tolist()
