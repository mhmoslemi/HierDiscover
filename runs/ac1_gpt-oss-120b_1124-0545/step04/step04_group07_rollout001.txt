def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = kwargs.get('n', 6000)
    # initial a
    a = _cantor_histogram(n, L=16)   # maybe L higher

    # add jitter
    a += np.random.rand(n) * 1e-4
    a = np.clip(a, 0, None)
    a /= a.sum()

    # initialize z = np.log(a + 1e-12)
    z = np.log(a + 1e-12)

    size1 = _next_pow2(2*n -1)
    size2 = _next_pow2(3*n -2)
    a_rev = None # compute later
    # ADAM parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # schedule
    step0 = 0.1 (learning rate for z)
    tau = None # set after first conv

    it = 0
    best_score = np.inf
    best_a = None

    while time.time() < deadline:
        # compute a from z
        a = np.exp(z - np.max(z))  # to avoid overflow
        a_sum = a.sum()
        a = a / a_sum

        # compute convolution
        conv_full = _full_convolution(a, size1)   # returns length 2n-1
        max_c = conv_full.max()
        # true objective
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{it}] new best = {best_score:.6f} time={time.time() - start:.1f}s")

        # temperature
        if tau is None:
            tau = 0.5 * max_c
        else:
            tau = max(tau * 0.9995, max_c * 1e-4) # gradually lower

        # compute softmax weights w
        # shift to stability
        w = np.exp((conv_full - max_c) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # compute gradient g = 2 * conv_rev(a, w)
        a_rev = a[::-1]
        g_full = _conv_fft(a_rev, w, size2)
        g = g_full[n-1 : n-1 + n]   # central n entries
        grad_a = 2.0 * g    # gradient of smooth max wrt a

        # gradient wrt z
        avg = np.dot(grad_a, a)
        grad_z = a * (grad_a - avg)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        z = z - step0 * m_hat / (np.sqrt(v_hat) + eps)

   # after loop, return best_a as list.
   return best_a.tolist()

"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. **Initial guess** – Approximate the Cantor (1/3) measure with a high‑resolution
   histogram (2**L atoms, L=16) and normalise to a probability vector.
2. **Smooth‑max optimisation** – Minimise a differentiable surrogate of the
   maximum of the self‑convolution:
        fτ(a) = τ·log∑exp((a*a−max(a*a))/τ) + max(a*a) .
   The gradient of fτ is 2·(a_rev ∗ w) where w is the soft‑max weight
   w∝exp((a*a−max)/τ).  Using the “softmax” weighting yields a smooth
   approximation that becomes tighter as τ→0.
3. **Exponentiated‑gradient (Adam) on the simplex** – Encode a as the soft‑max
   of an unconstrained vector `z` (a_i = exp(z_i)/∑exp(z)).  The gradient w.r.t.
   `z` is a_i·(∂f/∂a_i − ⟨a,∂f/∂a⟩).  Adam adapts step‑sizes per coordinate.
4. **Annealing** – τ starts at ½·max(a*a) and decays geometrically
   (≈0.9995 per iteration) until it reaches a tiny fraction of the current
   maximum.  The annealing gradually sharpens the surrogate toward the true
   max.
5. **Best‑ever tracking & occasional perturbation** – Whenever the true
   objective 2·n·max(a*a) improves, we store the sequence.  If no improvement
   occurs for several seconds we inject a small random perturbation to help
   escape plateaus.
6. The loop runs until the time budget (default 1000 s) is exhausted and
   returns the best sequence discovered.

The whole pipeline uses only NumPy and a few cheap FFTs, runs in
O(n log n) per iteration, and respects the non‑negativity / simplex
constraints required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 16) -> np.ndarray:
    """
    Produce a length‑`n` histogram approximating the Cantor (1/3) measure.
    `L` determines the number of atomic points (2**L).  The result sums to 1.
    """
    N = 1 << L                      # number of atoms, power of two
    # Powers of 3 used for the ternary digit scaling
    powers = 3.0 ** np.arange(1, L + 1)          # (L,)
    # Binary representation for all atoms
    idx = np.arange(N, dtype=np.uint32)[:, None]   # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)
    # Position of each atom in [0,1)
    pos = (bits * (2.0 / powers)).sum(axis=1)      # (N,)
    # Histogram on the required grid
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)        # normalise → sum == 1
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for finalisation

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # 1️⃣  Initialise from a high‑resolution Cantor histogram
    # --------------------------------------------------------------
    try:
        # Use the global fallback if it exists (might be from a previous run)
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            a = np.asarray(seq0, dtype=float)
            if a.size != n:
                # Resample to the requested length.
                x_old = np.linspace(0, 1, a.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 0.01:
                raise ValueError
            a /= s
        else:
            raise KeyError
    except Exception:
        # Default Cantor histogram (L=16 gives 65536 atoms → fine granularity)
        a = _cantor_histogram(n, L=16)

    # Small random jitter to avoid stagnation from exact symmetry
    a += np.random.rand(n) * 1e-8
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # ------------------------------------------------------------------
    # 2️⃣  Optimisation set‑up (FFT sizes, Adam parameters, annealing)
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * w

    # Parameterisation a = softmax(z) → ensures positivity & Σa = 1
    eps = 1e-12
    z = np.log(a + eps)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Learning‑rate for the Adam update (applied to z)
    lr = 0.5

    # Annealing for the smooth‑max temperature τ
    tau = None                 # will be initialised after first conv
    tau_decay = 0.9995
    # we keep τ ≥ 1e‑6·max_c to avoid numerical underflow
    min_tau_factor = 1e-6

    # Book‑keeping of the best solution seen
    best_score = np.inf
    best_seq = a.copy()
    it = 0
    last_improve = start

    # ------------------------------------------------------------------
    # 3️⃣  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- recover a from z (soft‑max) -----
        # subtract max for numerical stability
        z_shift = z - np.max(z)
        a = np.exp(z_shift)
        a_sum = a.sum()
        a = a / a_sum                       # Σa = 1, a ≥ 0

        # ----- compute self‑convolution -----
        conv_full = _full_convolution(a, size1)   # length 2n‑1
        max_c = conv_full.max()

        # ----- true objective (no smoothing) -----
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- temperature (smooth‑max) handling -----
        if tau is None:
            tau = 0.5 * max_c
        else:
            # geometric decay, but never below a tiny fraction of max_c
            tau = max(tau * tau_decay, max_c * min_tau_factor)

        # ----- soft‑max weights w ∝ exp((c−max)/τ) -----
        # shift by max_c for stability
        w_raw = np.exp((conv_full - max_c) / tau)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.ones_like(w_raw) / w_raw.size
        else:
            w = w_raw / w_sum

        # ----- gradient of smooth‑max w.r.t. a (2·(a_rev ∗ w)) -----
        a_rev = a[::-1]
        g_full = _conv_fft(a_rev, w, size2)         # length 3n‑2
        # central n entries correspond to gradient components
        g = g_full[n - 1 : n - 1 + n]
        grad_a = 2.0 * g                            # ∂fτ/∂a

        # ----- convert to gradient w.r.t. z (soft‑max chain rule) -----
        avg = np.dot(grad_a, a)                     # scalar = ⟨a,∂f/∂a⟩
        grad_z = a * (grad_a - avg)                 # ∂fτ/∂z

        # ----- Adam update on z -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- occasional random perturbation if stuck -----
        if it % 5000 == 0:
            # If we haven't improved for >30 s, add a tiny jitter
            if time.time() - last_improve > 30.0:
                perturb = np.random.randn(n) * 0.02   # 2 % Gaussian noise
                # Add directly to z (log‑space) and keep a normalized afterwards
                z += perturb
                # Reset Adam momentum to avoid stale directions
                m.fill(0.0)
                v.fill(0.0)
                # Reset temperature to be a bit larger (helps re‑exploration)
                tau = max(tau, 0.5 * max_c)

        # Safety break in case of numerical explosion (should not happen)
        if np.isnan(z).any() or np.isinf(z).any():
            print("[warning] numerical issue encountered – reinitialising z")
            z = np.log(a + eps)

    # ------------------------------------------------------------------
    # 4️⃣  Final reporting and return
    # ------------------------------------------------------------------
    total_time = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Return the best sequence as plain Python floats (list of float)
    return best_seq.tolist()
