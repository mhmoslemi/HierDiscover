"""
Search for a non‑negative step function (list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases

1. **Dense analytic grid** – a fine sweep over several parametric families
   (inverse‑product, inverse‑sum, edge‑exponential and edge‑Gaussian).  The
   best candidate of this grid supplies a very strong starting point.

2. **Multi‑start smoothed‑max L‑BFGS** – the objective
   Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) is a smooth approximation of the
   true max‑convolution.  Starting from the analytic best we run several
   random restarts of an L‑BFGS optimiser, gradually increasing λ from
   0.5 up to ≈10 000 (geometric schedule).  The optimiser works in the
   unconstrained soft‑max space (x → a = softmax(x)), guaranteeing
   non‑negativity and unit L₁‑norm.

3. **Simulated‑annealing style local search** – a cheap random mass‑transfer
   move is repeatedly applied while time permits; occasionally a worse
   move is accepted (temperature gradually cooling) to escape shallow
   plateaus.

The whole routine respects the 1000 s budget, prints intermediate
scores and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using an FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation used for bookkeeping.
    a is assumed non‑negative; it is normalised to sum = 1.
    Returns the quantity 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index i
    to a random destination j, then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, off: float) -> np.ndarray:
    """a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, p: float) -> np.ndarray:
    """a_i ∝ exp(‑p·i) + exp(‑p·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-p * i) + np.exp(-p * (n - 1 - i))
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i)²/(2σ²)) + exp(-(n‑1‑i)²/(2σ²))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i ** 2) / (2.0 * sigma ** 2)) + np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smoothed max and its gradient (in soft‑max space)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given parameter vector x (soft‑max logits) and smoothing λ,
    return (Φ_λ, ∇_x Φ_λ).  The function is suitable for SciPy's L‑BFGS‑B
    when `jac=True` (it returns a tuple).
    """
    # --- a = softmax(x) ---
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # --- convolution a*a ---
    conv = conv_fft_self(a)                 # length 2n‑1
    max_c = np.max(conv)

    # --- smooth max via log‑sum‑exp ---
    shift = conv - max_c                     # ≤0, keeps numbers moderate
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam      # Φ_λ(a)

    # weights for each convolution entry
    w = exp_term / sum_exp                   # sum to 1, shape (2n‑1,)

    # --- gradient w.r.t. a ---
    # dΦ/da = Σ_k w_k ∂(a*a)_k/∂a = 2 * (w_rev * a)   (convolution)
    w_rev = w[::-1]                          # reverse
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]    # length n
    grad_a = 2.0 * slice_part[::-1]          # ∂Φ/∂a

    # --- map gradient to x‑space via the Jacobian of soft‑max ---
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)              # ∇_x Φ = J^T ∇_a Φ

    return phi, grad_x

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the provided evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # spare half a second for cleanup

    # Length of the step function.  Larger n gives a smoother problem
    # at the price of a few extra FFT operations.
    n = int(kwargs.get("n", 3000))

    # ------------------------------------------------------------------
    # Phase 1 – dense analytic grid search
    # ------------------------------------------------------------------
    candidates = []

    # Parameter grids
    p_vals = np.arange(0.35, 0.78, 0.02)       # 0.35 … 0.77
    off_vals = np.arange(0.0, 1.6, 0.15)       # 0 … 1.45

    for p in p_vals:
        for off in off_vals:
            # product family (symmetric, high at the edges)
            candidates.append(analytic_product(n, float(p), float(off) + 1e-8))
            # sum family (two tails, no singularity at centre)
            candidates.append(analytic_sum(n, float(p), float(off) + 1e-8))

    # exponential tails
    for p in np.arange(0.02, 0.55, 0.04):
        candidates.append(analytic_exponential(n, float(p)))

    # edge‑Gaussian shapes
    for sigma in np.arange(5.0, 51.0, 5.0):
        candidates.append(analytic_gaussian(n, float(sigma)))

    # uniform baseline
    candidates.append(np.ones(n) / n)

    # If a reference construction is present, add it (normalised)
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            if ref.sum() > 0:
                candidates.append(ref / ref.sum())

    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[grid] best analytic score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑start smoothed‑max L‑BFGS
    # ------------------------------------------------------------------
    # Convert best analytic start into soft‑max parameters.
    eps = 1e-12
    x0 = np.log(np.maximum(best_a, eps))

    # λ schedule – geometric progression up to a large value.
    lam_vals = [0.5]
    while lam_vals[-1] < 10000:
        lam_vals.append(lam_vals[-1] * 5.0)

    # Detect SciPy availability for a fast L‑BFGS implementation.
    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:                     # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – falling back to a simple Adam optimiser.")

    # Number of random restarts (including the deterministic start)
    n_restarts = 4

    for restart in range(n_restarts):
        # Random perturbation of the log‑probabilities for diversity.
        if restart == 0:
            x = x0.copy()
        else:
            noise = np.random.randn(n) * 0.3
            x = x0 + noise

        # Lambda loop – each step re‑optimises from the previous solution.
        for lam in lam_vals:
            if time.time() >= deadline:
                break

            # Decide a modest iteration budget for this λ.
            remaining = deadline - time.time()
            # Roughly allocate a few hundred iterations per λ, but never exceed the time left.
            max_iter = int(min(350, max(30, remaining * 70)))  # cheap heuristic

            if have_scipy:
                # SciPy's L‑BFGS‑B expects the function to return (value, grad) when jac=True.
                res = minimize(
                    compute_phi_and_grad,
                    x,
                    args=(lam, n),
                    method='L-BFGS-B',
                    jac=True,
                    options={'maxiter': max_iter, 'disp': False}
                )
                x = res.x
            else:
                # Simple Adam optimiser (fallback).
                alpha = 0.03
                beta1, beta2 = 0.9, 0.999
                m = np.zeros_like(x)
                v = np.zeros_like(x)
                t = 0
                while t < max_iter and time.time() < deadline:
                    t += 1
                    phi, grad_x = compute_phi_and_grad(x, lam, n)
                    m = beta1 * m + (1 - beta1) * grad_x
                    v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                    m_hat = m / (1 - beta1 ** t)
                    v_hat = v / (1 - beta2 ** t)
                    step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
                    x = x - step

            # Evaluate true (non‑smoothed) score after this λ‑stage.
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[restart {restart+1}, λ={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Phase 3 – light simulated‑annealing local search
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.03   # initial relative mass‑transfer magnitude
    temperature = 1.0
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)

        # Accept improvements always; accept worse ones with a tiny probability to escape.
        if sc < best_score or np.random.rand() < math.exp(-(sc - best_score) / temperature):
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                elapsed = time.time() - start_time
                print(f"[anneal] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            a = cand
            # slowly tighten the step size and temperature after successes
            scale = max(0.0005, scale * 0.985)
            temperature = max(0.001, temperature * 0.995)
        else:
            # If move rejected, gently increase exploration radius.
            scale = min(0.05, scale * 1.005)
            temperature = min(1.0, temperature * 1.002)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Convert to plain Python list for the harness.
    return best_a.tolist()
