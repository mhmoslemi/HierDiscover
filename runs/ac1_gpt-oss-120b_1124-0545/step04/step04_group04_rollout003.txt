"""
Propose a high‑quality non‑negative step‑function (probability vector)
that minimizes  E(a)=2·n·max(conv(a,a)) / (sum(a)**2).

Algorithm
---------

1. **Multiple stochastic starts** – a handful of diverse initial vectors
   (Dirichlet draws, half‑sine, triangular, historic solution) are each
   optimised for a bounded portion of the time budget using an Adam optimiser
   on the soft‑max parametrisation (guaranteeing the simplex constraint).

2. **Final refinement** – the best vector from step 1 is further polished by a
   projected sub‑gradient method that works directly on the simplex.
   The refinement begins with a smooth soft‑max approximation of the max‑operator
   and switches to a hard‑max mask halfway through the remaining time.
   A simple ∓​project‑onto‑simplex step enforces non‑negativity and the
   ∑a = 1 constraint after every update.

3. The routine respects the supplied time budget (default 1000 s) and returns
   the best sequence it discovered.  All heavy work is done with FFTs
   (O(n log n)) to keep the optimisation fast for vectors of length 2000–5000.

The code is self‑contained, uses only NumPy and the supplied ``evaluate_sequence``,
and prints occasional progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    return exp_v / exp_v.sum()


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # find the rho value
    rho_arr = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(rho_arr):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(rho_arr)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(a: np.ndarray) -> float:
    """Return max(a*a) using an FFT of appropriate length."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv.max()


def gradient_max_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (smoothed) max‑convolution objective with respect to a,
    given a weight vector w (length 2·n‑1) satisfying Σ w = 1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    fa = np.fft.fft(a, fft_len)
    fw = np.fft.fft(w, fft_len)

    # (w * a) – linear convolution, length 3·n‑2
    conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
    g2 = conv_wa[:n]

    # w ⋆ a – linear cross‑correlation, length 3·n‑2
    corr = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
    g1 = corr[:n]

    # full gradient of 2·n·max(conv) w.r.t. a
    return 2.0 * n * (g1 + g2)


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _optimize_a(
    a_init: np.ndarray,
    n: int,
    alloc_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a single start vector ``a_init`` for at most
    ``alloc_s`` seconds.  The optimisation works on log‑parameters (soft‑max)
    to keep the iterate on the simplex.
    Returns the best vector found (as a probability vector) and its exact score.
    """
    # initialise log‑parametrisation (soft‑max guarantees feasibility)
    v = np.log(a_init + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedule parameters (linear in normalised time p∈[0,1])
    beta_start, beta_end = 0.1, 5000.0        # soft‑max temperature
    lr_start, lr_end = 0.25, 0.001           # learning‑rate

    # FFT size sufficient for both convolution and w‑convolution
    fft_len = _next_pow2(3 * n - 2)

    # bookkeeping
    start = time.time()
    deadline = start + alloc_s - 0.05  # safety margin
    best_a = a_init.copy()
    # exact score for a probability vector is 2·n·max(conv)
    best_score = 2.0 * n * conv_max(best_a)
    last_improve = start
    it = 0

    # after this fraction we switch to the true hard‑max mask
    mask_phase_start = 0.70

    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))

        # temperature and learning‑rate schedules
        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        # ----- current probabilities on the simplex -----
        a = softmax(v)               # Σa = 1, a_i ≥ 0

        # ----- convolution a * a -----
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # ----- weight vector w (smooth or hard mask) -----
        if p < mask_phase_start:
            # smooth soft‑max over the convolution entries
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:
            # hard mask on the true maximum (sub‑gradient of max)
            max_val = conv_aa.max()
            eps_mask = max_val * 1e-12 + 1e-14
            w = (conv_aa >= max_val - eps_mask).astype(float)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv_aa)
                w[np.argmax(conv_aa)] = 1.0
                w_sum = 1.0
            w = w / w_sum

        # ----- gradient wrt a (via FFT) -----
        g = gradient_max_conv(a, w)          # shape (n,)

        # ----- chain rule for the soft‑max parametrisation -----
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)            # ∂f/∂v

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # ----- exact score & bookkeeping -----
        cur_score = 2.0 * n * conv_aa.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            # occasional progress report
            if it % 2000 == 0:
                print(
                    f"[{it:07d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # ----- restart on stagnation -----
        if time.time() - last_improve > 20.0:
            jitter = rng.normal(scale=0.01, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[{it:07d}] restart after {elapsed:.1f}s (no improvement)")

    return best_a, best_score


def refine_solution(a0: np.ndarray, n: int, budget_s: float) -> tuple[np.ndarray, float]:
    """
    Fine‑tune a probability vector using a projected sub‑gradient method.
    The first half of the allocated time uses a smooth soft‑max approximation,
    the second half uses a hard‐max mask.
    Returns the refined vector and its exact score.
    """
    start = time.time()
    deadline = start + budget_s
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * conv_max(a)

    it = 0
    # pre‑compute RHS lengths for FFTs
    fft_len_conv = _next_pow2(2 * n - 1)          # for plain convolution (scoring)
    fft_len_grad = _next_pow2(3 * n - 2)          # for gradient (w * a)

    while time.time() < deadline:
        elapsed = time.time() - start
        p = elapsed / budget_s                      # progress fraction ∈[0,1]

        # ----- current convolution -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # ----- weight vector w -----
        if p < 0.5:   # smooth phase
            # temperature ramps up exponentially; avoid overflow
            beta = 0.5 + 5000.0 * p   # a simple increasing schedule
            z = beta * conv
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:        # hard mask phase
            max_val = conv.max()
            eps_mask = max_val * 1e-12 + 1e-14
            w = (conv >= max_val - eps_mask).astype(float)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            w = w / w_sum

        # ----- gradient -----
        # Need a FFT of length grad_len (covers w (2n‑1) and a (n))
        fa_full = np.fft.fft(a, fft_len_grad)
        fw = np.fft.fft(w, fft_len_grad)

        conv_wa = np.real(np.fft.ifft(fw * fa_full))[: 3 * n - 2]
        g2 = conv_wa[:n]

        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa_full))[: 3 * n - 2]
        g1 = corr_full[:n]

        g = 2.0 * n * (g1 + g2)

        # ----- step size (classic diminishing step) -----
        eta = 0.6 / np.sqrt(it + 1.0)

        # ----- projected update onto the simplex -----
        a_new = a - eta * g
        a = project_onto_simplex(a_new)

        # ----- bookkeeping -----
        cur_score = 2.0 * n * conv_max(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and returns the
    best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to guarantee return before the hard deadline
    deadline = start_time + max(0.0, budget_s - 0.5)

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Independent stochastic starts (including historic one)
    # ------------------------------------------------------------------
    attempts = 6
    global_best_a = None
    global_best_score = float("inf")
    rng_global = np.random.default_rng(seed)

    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 12.0:
            break
        remaining = attempts - attempt
        # allocate ~70 % of the currently available time to the remaining starts
        alloc = time_left * 0.70 / remaining

        # ---- initialise a start vector ----
        if attempt == 0 and "height_sequence_1" in globals():
            a0 = np.array(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            mod = attempt % 4
            if mod == 0:
                # Dirichlet with moderate sparsity
                a0 = rng_global.dirichlet(np.full(n, 0.5))
            elif mod == 1:
                # Dirichlet with stronger sparsity
                a0 = rng_global.dirichlet(np.full(n, 0.2))
            elif mod == 2:
                # half‑sine shape (zero at ends, peak in centre)
                idx = np.arange(n, dtype=float)
                a0 = np.sin(np.pi * idx / (n - 1))
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / a0.sum()
            else:
                # triangular shape (linear rise then fall)
                half = n // 2
                left = np.arange(1, half + 1)
                right = np.arange(half, 0, -1) if n % 2 == 0 else np.arange(half, 0, -1)
                a0 = np.concatenate([left, right]).astype(float)
                a0 = a0 / a0.sum()

        # ---- optimise this start ----
        a_opt, score_opt = _optimize_a(
            a_init=a0,
            n=n,
            alloc_s=alloc,
            rng=np.random.default_rng(seed + attempt + 12345),
        )
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {attempt}] new global best = {global_best_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 2) Final high‑precision refinement on the best candidate
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 8.0 and global_best_a is not None:
        print(f"[refine] allocating {time_left:.1f}s for final refinement")
        refined_a, refined_score = refine_solution(global_best_a, n, time_left)
        if refined_score < global_best_score:
            global_best_a = refined_a
            global_best_score = refined_score

    # ------------------------------------------------------------------
    # 3) Safety fallback
    # ------------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)

    # final sanity: ensure non‑negativity and exact simplex normalisation
    global_best_a = np.maximum(global_best_a, 0.0)
    global_best_a = global_best_a / global_best_a.sum()

    total_time = time.time() - start_time
    print(f"[done] best score = {global_best_score:.6f}, total time = {total_time:.1f}s")
    return global_best_a.tolist()
