def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))

def _adam_optimize(seed: np.ndarray,

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a tiny amount of mass between
    two random positions and keep changes that improve the objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()

    step_scale = 0.01  # initial relative step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a small amount from i towards j
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer  # always positive
        # apply transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        # check non‑negativity (should hold) and evaluate
        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val

        # slowly reduce the step scale
        if it % 1000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val
