"""
Propose a non‑negative step‑function (list of floats) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Broad parametric sweep** – generate many simple families
   (power‑law, exponential, linear, Beta‑like) for a handful of lengths.
   Keep the best sequence (lowest exact objective 2 n·max(conv)).

2. **Annealed soft‑max gradient descent** – minimise the smooth
   approximation of the max‑convolution:

        f_c(a) = Σ_k  w_k(a)·(a∗a)[k]   with   w_k ∝ exp(c·(a∗a)[k]).

   The gradient of f_c is the correlation of a with the weight vector w.
   For each iteration we
   * compute the convolution a∗a,
   * form the soft‑max weights (c is increased slowly),
   * compute the gradient via an FFT‑based correlation,
   * take an Adam step,
   * project back onto the probability simplex.

   The soft‑max parameter `c` starts small (≈0.1) – a smooth landscape –
   and is multiplied by a factor every few hundred iterations, so that
   the method gradually turns into the sub‑gradient descent used in the
   previous baseline.

3. **Escaping plateaus** – occasional random mass exchanges are performed
   and the best sequence seen so far is tracked.  The loop stops when the
   time budget (default 1000 s) is exhausted or when the target bound
   1.503 is reached.

The final sequence is returned as a plain Python list; ``evaluate_sequence``
will clamp it and compute the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Algorithm from Duchi et al., ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-α}, normalised to sum = 1."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ r^{i‑offset} for i≥offset, otherwise 0; normalised."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Simple decreasing linear (triangular) shape, normalised."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_sequence(n: int, alpha: float, beta: float, offset: int = 0) -> np.ndarray:
    """
    Discrete Beta‑like shape on the interval [offset, n-1].

    a_i ∝ (t)^{α‑1} (1‑t)^{β‑1} where t ranges uniformly over (0,1).
    The vector is normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n - 1:
        length = n - offset
        # t ∈ (0,1) strictly, avoid exact 0 or 1 for α,β<1
        t = (np.arange(length, dtype=float) + 0.5) / length
        vals = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
        a[offset:] = vals
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` when Σa = 1:
    2·n·max(a∗a).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimisation – annealed soft‑max gradient descent
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, c: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k ∝ exp(c·conv_k) shifted by max(conv) for stability.
    """
    maxv = conv.max()
    # shift to avoid overflow
    exp_term = np.exp(c * (conv - maxv))
    w = exp_term / exp_term.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)[k] = 2·corr(a, w) using FFT.
    The correlation is obtained by convolving a with the reversed weight vector.
    """
    # reverse w
    w_rev = w[::-1]
    corr = _conv_fft(a, w_rev)                 # length = 3·n‑2
    # correlation at lag j is corr[ (len(w)-1) - j]; extract via reverse slice
    grad = 2.0 * corr[::-1][:a.size]
    return grad

    # beta‑distribution parameters
    beta_alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    beta_betas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    beta_offsets = [0, 10, 50, 200]

    for n in length_candidates:
        if time.time() > deadline:
            break

        # ----- Power law -----
        for offset in power_offsets:
            if offset >= n:
                continue
            for alpha in power_alphas:
                a = _power_sequence(n, alpha, offset)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} power α={alpha:.2f} off={offset}  score={score:.6f}")

        # ----- Exponential -----
        for offset in exp_offsets:
            if offset >= n:
                continue
            for r in exp_rs:
                a = _exp_sequence(n, r, offset)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} exp r={r:.3f} off={offset}  score={score:.6f}")

        # ----- Linear -----
        for offset in linear_offsets:
            if offset >= n:
                continue
            a = _linear_sequence(n, offset)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} linear off={offset}  score={score:.6f}")

        # ----- Beta‑like -----
        for offset in beta_offsets:
            if offset >= n:
                continue
            for a_alpha in beta_alphas:
                for a_beta in beta_betas:
                    a = _beta_sequence(n, a_alpha, a_beta, offset)
                    score = _objective(a)
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"[sweep] n={n} beta α={a_alpha:.2f} β={a_beta:.2f} off={offset}  score={score:.6f}")

        # ----- Random Dirichlet seeds -----
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} random seed  score={score:.6f}")

        # ----- Global variable if present -----
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                a = np.maximum(init, 0.0)
                if a.sum() == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} global init  score={score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Annealed soft‑max / Adam descent starting from the best sweep
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform vector of moderate size
        n = 4000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    a = best_seq.copy()
    n = a.size

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.12          # base learning rate
    lr_min = 5e-5

    # soft‑max temperature (c)
    c = 0.1
    c_factor = 1.5          # increase every 500 iterations
    c_max = 800.0

    iteration = 0
    eval_every = 30

    while time.time() < deadline:
        iteration += 1

        # --- convolution and soft‑max weights ---
        conv = _conv_fft(a, a)               # length 2n‑1
        max_conv = conv.max()
        w = _softmax_weights(conv, c)

        # --- gradient (correlation) ---
        grad = _gradient_from_weights(a, w)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step
        a = _project_onto_simplex(a)

        # --- learning‑rate decay (slow) ---
        lr = max(lr * 0.9998, lr_min)

        # --- increase soft‑max temperature occasionally ---
        if iteration % 500 == 0:
            c = min(c * c_factor, c_max)

        # --- occasional random mass exchange to break symmetries ---
        if iteration % 73 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.25, lr * 0.4)
            if delta > 1e-12:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # --- periodic exact evaluation ---
        if iteration % eval_every == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] iter={iteration} n={n} score={best_score:.6f}")

            if best_score < 1.503:
                print("[target reached]")
                break

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} (n={best_seq.size}) "
          f"iterations={iteration}")
    return best_seq.tolist()
