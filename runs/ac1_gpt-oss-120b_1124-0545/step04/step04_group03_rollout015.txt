"""
Propose a non‑negative step‑function (list of floats) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Analytic grid** – generate a large family of power‑law, mixed,
   and exponential candidates (including the provided `height_sequence_1`);
   keep the best one.

2. **Smooth‑max optimisation** – minimise a log‑sum‑exp surrogate of
   `max(conv)` using Adam with a continuation on the smoothing parameter λ.
   The parameters are the soft‑max logits `x` (so the simplex constraint is
   satisfied automatically).  The surrogate becomes the exact max for very
   large λ.

3. **L₂‑norm surrogate** – minimise the squared L₂ norm of the convolution,
   ∑_k conv(k)², whose gradient can be obtained by a second convolution.
   This encourages a flatter autoconvolution and helps to lower the true
   maximum.

4. **Exact max sub‑gradient descent** – perform multiplicative (exponentiated)
   gradient descent on the genuine `max(conv)` using its sub‑gradient
   (∂/∂a_i max(conv) = 2·a_{k−i} for a maximising index k).

5. **Random mass‑transfer local search** – final hill‑climbing by moving a
   tiny amount of mass between random positions.

All steps respect the 1000 s wall‑clock budget; the best sequence found is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def exponential_candidate(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped exponential family: a_i ∝ exp(-p·(i+offset)) + exp(-p·(n-1-i+offset))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-p * (i + offset)) + np.exp(-p * (n - 1 - i + offset))
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smoothed max Φ_λ(a) and its gradient w.r.t. the soft‑max logits x.
    a = softmax(x) is the underlying probability vector.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max via log‑sum‑exp
    shift = conv - max_c                    # for numerical stability
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam     # Φ_λ

    w = exp_term / sum_exp                  # weights (sum to 1)

    # gradient of Φ_λ w.r.t. a
    w_rev = w[::-1]                        # reversed weights
    size = _next_pow2(len(w_rev) + n - 1)   # safe length for FFT
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]  # length n
    grad_a = 2.0 * slice_part[::-1]

    # map gradient to x‑space via Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def compute_l2_and_grad(a: np.ndarray):
    """
    L₂ surrogate: minimise sum(conv(k)²).
    Returns the loss value and its gradient w.r.t. a.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)                     # length 2n‑1
    loss = np.sum(conv * conv)

    # gradient = 4 * (conv_rev * a)  (appropriate slice)
    conv_rev = conv[::-1]
    size = _next_pow2(len(conv_rev) + n - 1)    # length 3n‑2 after conv
    fft_cr = np.fft.rfft(conv_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    h = np.fft.irfft(fft_cr * fft_a, n=size)    # full conv of conv_rev and a

    idx_start = len(conv_rev) - 1               # = 2n‑2
    grad = 4.0 * h[idx_start - np.arange(n)]
    return loss, grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the given evaluation function.
    The routine builds analytic candidates, refines them with smooth‑max
    Adam optimisation, followed by L₂‑norm descent, exact‑max sub‑gradient
    descent and a short random local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a small safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1) Analytic grid – diverse families + mixtures
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Uniform
    uniform = np.ones(n) / n
    candidates = [uniform]

    # Power‑law families (product & sum)
    p_vals = np.linspace(0.30, 0.90, 13)          # 13 values
    off_vals = [0.0, 0.2, 0.5, 1.0, 2.0]
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                cand = analytic_candidate(n, p, off, mode)
                candidates.append(cand)

    # Exponential families
    exp_p_vals = np.linspace(0.001, 0.02, 10)    # very shallow exponentials
    for p in exp_p_vals:
        for off in off_vals:
            cand = exponential_candidate(n, p, off)
            candidates.append(cand)

    # Mixtures of two power‑law product shapes
    mix_ps = [0.35, 0.45, 0.55, 0.65]
    mix_ws = [0.25, 0.5, 0.75]
    for p1 in mix_ps:
        for p2 in mix_ps:
            for w in mix_ws:
                for off in off_vals:
                    i = np.arange(n, dtype=np.float64)
                    a1 = ((i + off) * (n - 1 - i + off)) ** (-p1)
                    a2 = ((i + off) * (n - 1 - i + off)) ** (-p2)
                    a = w * a1 + (1 - w) * a2
                    a = np.where(np.isfinite(a), a, 0.0)
                    total = a.sum()
                    if total > 0:
                        a = a / total
                    else:
                        a = uniform
                    candidates.append(a)

    # Include globally‑provided construction if present and same length
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    # Evaluate all candidates
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smooth‑max optimisation with Adam (continuation on λ)
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_vals = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    step_size = 0.05    # base learning rate for Adam

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0

    for lam in lam_vals:
        if time.time() >= deadline:
            break
        # allocate a modest number of Adam steps for each λ
        max_steps = 2000
        for _ in range(max_steps):
            if time.time() >= deadline:
                break
            t += 1
            phi, grad_x = compute_phi_and_grad(x, lam, n)
            # Adam update on x
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = step_size * m_hat / (np.sqrt(v_hat) + eps_opt)
            x = x - step

            # evaluate the exact objective
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[adam lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # --------------------------------------------------------------
    # 3) L₂‑norm surrogate descent (helps flatten the convolution)
    # --------------------------------------------------------------
    a = best_a.copy()
    base_eta_l2 = 1.0
    l2_iter = 0
    while time.time() < deadline and l2_iter < 3000:

        l2_iter += 1
        eta = base_eta_l2 / math.sqrt(l2_iter)
        # multiplicative update keeps us on the simplex automatically
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[l2 iter={l2_iter}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # --------------------------------------------------------------
    # 4) Exact‑max sub‑gradient descent (exponentiated update)
    # --------------------------------------------------------------
    a = best_a.copy()
    base_eta_max = 0.5
    max_iter = 0
    while time.time() < deadline and max_iter < 5000:
        max_iter += 1
        conv = conv_fft_self(a)
        max_val = np.max(conv)
        k = int(np.argmax(conv))                     # a maximising index
        # gradient of the max entry
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        grad[idx_i] = 2.0 * a[idx_j]

        eta = base_eta_max / math.sqrt(max_iter)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[subgrad iter={max_iter}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # --------------------------------------------------------------
    # 5) Light random‑mass‑transfer local search (fine‑tuning)
    # --------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)
        a = best_a  # keep proposing around the current best

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
