def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 12000))

    # Initialize a
    try:
        seq0 = globals().get('height_sequence_1')
        if seq0 is not None:
            a0 = np.asarray(seq0, dtype=float)
            if a0.size != n:
                # resample using linear interpolation
                x_old = np.linspace(0, 1, a0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a0)
            else:
                a = a0
        else:
            raise KeyError
    except Exception:
        a = _cantor_histogram(n, L=15)

    a = np.clip(a, 0.0, None)
    if a.sum() < 0.01:
        a = _cantor_histogram(n, L=15)
    a /= a.sum()
    
    # Precompute sizes
    size_conv = _next_pow2(2*n-1)
    size_grad = _next_pow2(3*n-2)

    a_rev = a[::-1].copy()
    # Adam states:
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    # hyperparameters
    lr = 0.2   # Learning rate base (maybe 0.5)
    # Temperature schedule
    tau = None  # initially set later
    tau_decay = 0.9995  # per iteration
    min_tau = 1e-10

    best_seq = a.copy()
    # Evaluate current best via evaluate_sequence (though scaling doesn't matter)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, init score={best_score:.6f}")

    it = 0
    last_improve_time = start

    # main loop
    while time.time() < deadline:
        # conv
        conv_full = _full_convolution(a, size_conv)
        max_conv = conv_full.max()

        if tau is None:
            tau = max_conv * 0.5
        else:
            tau = max(tau * tau_decay, min_tau)

        # Softmax
        w = np.exp((conv_full - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n-1:n-1+n]  # central n
        grad = -2.0 * g  # descent direction

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # update a
        a = a + step
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a = a / s

        a_rev = a[::-1].copy()

        # evaluate occasionally (every few iterations)
        if (it % 200 == 0):
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve_time = time.time()
                print(f"[{it:7d}] new best = {best_score:.6f}  time={last_improve_time-start:.1f}s")
            # optional: check stagnation
            if time.time() - last_improve_time > 30.0 and it > 1000:
                # add noise to escape
                noise = np.random.normal(scale=0.01, size=n)
                a = a + noise
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                a_rev = a[::-1].copy()
                m[:] = 0.0
                v[:] = 0.0
                last_improve_time = time.time()
                print(f"[{it:7d}] restart noise injection, score={cur_score:.6f}")

        it += 1

    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    # Save globally for later runs
    globals()['height_sequence_1'] = best_seq.tolist()
    return best_seq.tolist()

"""
Adam‑based smoothed‑max gradient descent for the step‑function problem.

The objective is                2·n·max(conv(a,a)) / (sum(a)^2)
and is homogeneous, so we keep sum(a)=1 throughout the search.
We start from a Cantor‑measure histogram (L=15) or from the previous
solution stored in ``height_sequence_1``.  The algorithm repeatedly:

1. Computes the full self‑convolution a*a via FFT.
2. Forms a soft‑max weighting w that approximates the max of the
   convolution (temperature τ slowly anneals).
3. Obtains the gradient of the smoothed maximum:
      ∇ ϕ(a) = -2·(a_rev * w)               (linear convolution)
4. Updates a with the Adam optimiser; after each step we clip to
   non‑negative values and renormalise to Σa=1 (projection onto the
   probability simplex).
5. Tracks the best sequence seen.  When progress stalls we inject a
   small random perturbation and reset the Adam moments to escape
   local minima.

The whole loop runs for the allotted time budget (default 1000 s).
The returned list of floats is the best candidate found.
"""

import numpy as np

# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT.

    Parameters
    ----------
    u, v : np.ndarray
        Real‑valued input arrays.
    size : int
        Length of the zero‑padded FFT; must be ≥ len(u)+len(v)-1.

    Returns
    -------
    np.ndarray
        The full linear convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the full convolution a*a using the pre‑allocated FFT size."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 15) -> np.ndarray:
    """
    Approximate the (1/3)-Cantor measure on [0,1] with 2**L atoms
    and bin the atoms into ``n`` equally spaced bins.

    The result is a non‑negative vector of length ``n`` that sums to 1.
    """
    N = 1 << L  # number of atoms
    # Compute ternary positions of the atoms
    # Each atom corresponds to a binary string of length L; the ternary
    # expansion uses digits 0 and 2.
    idx = np.arange(N, dtype=np.uint32)[:, None]               # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(
        np.float64)                                            # (N,L)
    powers = 3.0 ** np.arange(1, L + 1)                        # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)                  # (N,)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)                    # normalise → sum = 1
    return a

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Full self‑convolution of the current a
        # --------------------------------------------------------------
        conv_full = _full_convolution(a, size_conv)
        max_conv = conv_full.max()

        # --------------------------------------------------------------
        # 2) Update temperature (soft‑max smoothing)
        # --------------------------------------------------------------
        tau = max(tau * tau_decay, min_tau)

        # --------------------------------------------------------------
        # 3) Soft‑max weights (stable by shifting with max_conv)
        # --------------------------------------------------------------
        w = np.exp((conv_full - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 4) Gradient of the smooth‑max (via FFT)
        # --------------------------------------------------------------
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n - 1 : n - 1 + n]          # central n entries
        grad = -2.0 * g                        # descent direction

        # --------------------------------------------------------------
        # 5) Adam update (bias‑corrected)
        # --------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # 6) Project onto the simplex (non‑negative, sum=1)
        # --------------------------------------------------------------
        a = a + step
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= s

        a_rev = a[::-1].copy()

        # --------------------------------------------------------------
        # 7) Track best solution
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_conv
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # --------------------------------------------------------------
        # 8) Stagnation handling – occasional random shake‑up
        # --------------------------------------------------------------
        if (time.time() - last_improve > 30.0) and (it > 1000):
            # Add a tiny random perturbation and reset Adam moments.
            noise = np.random.normal(scale=0.002, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.ones_like(a) / n
            else:
                a /= a_sum
            a_rev = a[::-1].copy()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] stagnation – noise injection, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning with a very low temperature (almost exact max)
    # ------------------------------------------------------------------
    fine_iters = 3000
    for _ in range(fine_iters):
        if time.time() >= deadline:
            break
        conv_full = _full_convolution(a, size_conv)
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[fine] new best = {best_score:.6f}  time={elapsed:.1f}s")
        # Use an almost‑zero temperature – the soft‑max becomes a hard‑max.
        tau_fine = max(min_tau, max_conv * 0.01)
        w = np.exp((conv_full - max_conv) / tau_fine)
        w_sum = w.sum()
        w = np.full_like(w, 1.0 / w.size) if w_sum == 0.0 else w / w_sum
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n - 1 : n - 1 + n]
        grad = -2.0 * g
        # Simple gradient step with a small fixed LR
        a = a - 0.02 * grad
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= a_sum
        a_rev = a[::-1].copy()

    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Store the best sequence globally for possible reuse.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
