"""
Propose a step‑function (non‑negative list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.
The algorithm follows a multi‑phase strategy:

* **Diverse analytic seeds** – uniform, sine‑window, geometric,
  power‑law, symmetric, discrete Gaussian, Laplace, Dirichlet, etc.
* **Simulated‑Annealing on the probability simplex** – the vector is
  stored as log‑probabilities (soft‑max) and perturbed with a
  temperature‑controlled Gaussian walk.  The exact objective
  (using an FFT for the auto‑convolution) is evaluated, and Metropolis
  acceptance is used.  The temperature is linearly cooled.
* **Exact sub‑gradient mirror descent** – after the stochastic phase we
  run the exponentiated‑gradient refinement that uses the true
  sub‑gradient of the max‑convolution.
* **Fine‑grained hill‑climbing** – tiny random mass transfers between
  two positions are accepted if they improve the objective.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for the lengths used)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of the evaluation function (sum(a) assumed ≈1)."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _gaussian(n: int, sigma_ratio: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, sigma = sigma_ratio·n."""
    mu = (n - 1) / 2.0
    sigma = sigma_ratio * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _laplace(n: int, beta_ratio: float) -> np.ndarray:
    """Discrete Laplace (double‑exponential) centred, scale = beta_ratio·n."""
    mu = (n - 1) / 2.0
    beta = beta_ratio * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(x - mu) / beta)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent refinement (same as the reference)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Mirror‑descent refinement using the true sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step
        lr = 0.6 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)       # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (uses pairwise mass transfer)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.01               # relative amount of mass moved

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor],
                           step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = _objective(new, n)

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing on the simplex (log‑softmax representation)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk on the probability simplex.
    The state is stored as logits (log‑softmax) to keep the sum = 1.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)          # initialise from a feasible point
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq, n)

    best_seq = cur_seq.copy()
    best_val = cur_val

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # Linear cooling from T0 to T_end
        frac = it / max_iters
        T = (0.5 * (1.0 - frac) + 0.001 * frac)

        # Step size is proportional to temperature
        proposal_logit = logit + rng.normal(scale=0.2 * T, size=logit.shape)
        proposal_seq = _softmax(proposal_logit)
        prop_val = _objective(proposal_seq, n)

        delta = prop_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-delta / (T + 1e-12)):
                accept = True

        if accept:
            logit = proposal_logit
            cur_seq = proposal_seq
            cur_val = prop_val
            if prop_val < best_val:
                best_val = prop_val
                best_seq = proposal_seq.copy()
                print(f"[SA] n={n:5d} it={it:5d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed construction (diverse analytical families)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalised vectors."""
    seeds: list[np.ndarray] = []

    rng = np.random.default_rng(2023)

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # discrete Gaussian (different widths)
    for sigma_ratio in np.linspace(0.08, 0.45, 8):
        seeds.append(_gaussian(n, sigma_ratio))

    # Laplace (different scales)
    for beta_ratio in np.linspace(0.04, 0.20, 5):
        seeds.append(_laplace(n, beta_ratio))

    # random Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors (logits from a normal distribution)
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse a previously discovered construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # guard margin

    # Allow reuse of a previously discovered construction if present
    previous = globals().get("height_sequence_1")

            previous = None

        # --------------------------------------------------------------
        # Phase 1 – Simulated annealing (global stochastic search)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            sa_seq, sa_val = _simulated_annealing(global_best_seq, n,
                                                  deadline, max_iters=200000)
            if sa_val < global_best_val:
                global_best_val = sa_val
                global_best_seq = sa_seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={sa_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (mirror descent)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            ref_seq, ref_val = _subgradient_refine(global_best_seq, n,
                                                   deadline, max_iters=40000)
            if ref_val < global_best_val:
                global_best_val = ref_val
                global_best_seq = ref_seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={ref_val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Stochastic hill climbing (fine‑grained local search)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            hc_seq, hc_val = _hill_climb(global_best_seq, n,
                                         deadline, max_iters=20000)
            if hc_val < global_best_val:
                global_best_val = hc_val
                global_best_seq = hc_seq.copy()
                print(f"[overall‑Hill] n={n:5d} new best={hc_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be required
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq, n)

    # Final sanity check using the provided external evaluator (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
