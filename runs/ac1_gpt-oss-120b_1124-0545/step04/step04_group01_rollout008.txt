def _solve_max_sum_lp(f, M):
    n = len(f)
    A = _conv_matrix(f)  # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum => maximize sum
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None, 0.0
    b = res.x
    return b, b.sum()

def _objective(a):
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    return 2.0 * n * max_conv / (sum_a**2)

def _lp_refine(a, deadline, max_iters=5):
    n = len(a)
    best_a = a.copy()
    best_val = _objective(a)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        a = best_a  # use current best
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # compute conv matrix
        A = diags([np.full(n, ai) for ai in a], np.arange(n), shape=(2*n-1, n), format='csr')
        # Solve LP: maximize sum(b) -> minimize -1^T b
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print("[LP] solve failed", res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            break
        # Scale to sum=1
        a_new = b / sum_b  # new candidate
        val_new = _objective(a_new)
        print(f"[LP] it={it} sum_b={sum_b:.6f} new_val={val_new:.6f}")
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        else:
            # If not improve, stop? maybe continue anyway.
            best_a = a_new.copy()
        # optional break if improvement less than threshold
    return best_a, best_val
```

            seeds.append(_geom(n,r))
    for p in np.linspace(0.0,2.0,9):
        seeds.append(_power(n,p))
    for c in np.linspace(0.5,4.0,5):
        r = 1.0 - c/n
        if r>0:
            seeds.append(_sym_geom(n,r))
    for a in (0.3,0.5,0.7):
        for b in (0.3,0.5,0.7):
            seeds.append(_beta_shape(n,a,b))
    rng = np.random.default_rng(2023)
    for conc in (0.3,1.0,5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    return seeds

def _objective(a):
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a**2)

def _adam_optimize(seed, n, deadline, max_iters=8000):
    v = np.log(seed + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr0=0.8
    beta1=0.9
    beta2=0.999
    eps=1e-12
    beta = 5.0
    beta_mul = 1.03
    best_seq = None
    best_val = float('inf')
    t = 0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        m = beta1 * m + (1-beta1) * grad_v
        v2 = beta2 * v2 + (1-beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v2_hat = v2 / (1 - beta2**t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val

def _subgradient_refine(start_seq, n, deadline, max_iters=40000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (np.sum(a)**2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # EG step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s
    return best_seq, best_val

def _hill_climb(start_seq, n, deadline, max_iters=20000):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.01
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = _objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")
        if it % 1000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)
    return best_seq, best_val

def _lp_refine(a, deadline, max_iters=5):
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # compute current max convolution
        conv = _conv_fft(best_a, best_a)
        M = np.max(conv)
        # Build convolution constraint matrix A (2n-1) x n
        # Using sparse diagonal representation
        diagonals = [np.full(n, ai) for ai in best_a]
        offsets = np.arange(n, dtype=int)
        A = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
        c = -np.ones(n)  # maximize sum(b)
        b_ub = np.full(2*n-1, M)
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible
            break
        # Scale to unit sum
        a_new = b / sum_b
        val_new = _objective(a_new)
        print(f"[LP] it={it} sum_b={sum_b:.6f} new_val={val_new:.6f}")
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        else:
            # still accept? we can keep the new a for next iteration
            best_a = a_new.copy()
    return best_a, best_val

"""
Propose a non‑negative step function minimizing
    2·n·max(conv(a,a)) / (sum(a)²)

The routine follows a multi‑phase pipeline (time‑aware):

1. Generate a diverse set of analytic seeds (uniform, geometric,
   power‑law, beta‑like, sine‑window, random Dirichlet, etc.).
2. Fast evaluation of all seeds → keep a few best.
3. Adam optimisation on a smooth log‑sum‑exp approximation of the
   max–convolution (variables are the log‑softmax of the coefficients).
4. Exact sub‑gradient mirror‑descent refinement.
5. Linear‑programming improvement:
   - maximise Σb_j subject to b≥0 and (a∗b)_k ≤ M (M = max(a∗a)).
   - obtain a new candidate g = b/Σb and mix  f←(1‑t)f + t·g  (t=0.5).
   This step typically yields a noticeable drop of the objective.
6. Stochastic hill‑climbing pair‑wise mass transfers for final polishing.

The algorithm respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest set of diverse initial vectors (ℓ₁‑normalised)."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return the exact evaluate_sequence objective (sum(a) assumed ≠0)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a ** 2)


# ----------------------------------------------------------------------
# Optimisation primitives (Adam, sub‑gradient, hill climbing)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on the smoothed max of the auto‑convolution."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    vv = np.zeros_like(v)

    lr0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        vv = beta2 * vv + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        vv_hat = vv / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(vv_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Mirror‑descent refinement with the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing with pair‑wise mass transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               deadline: float,
               max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    LP‑based improvement.

    maximise Σb_j   subject to   b≥0 , (a∗b)_k ≤ M  (M = max(a∗a)).
    The solution b typically has Σb_j > 1; we scale it to unit sum
    and mix it with the current a (t=0.5).  Several iterations are performed
    until no further gain is observed or time runs out.
    """
    n = len(a)
    a = a / np.sum(a)                     # ensure ℓ₁‑norm = 1
    best_a = a.copy()
    best_val = _objective(best_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(best_a, best_a)
        M = np.max(conv)

        # build the (2n‑1)×n convolution constraint matrix (sparse)
        diagonals = [np.full(n, ai) for ai in best_a]
        offsets = np.arange(n, dtype=int)
        A = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")

        c = -np.ones(n)                                 # maximise Σb_j
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()

        if sum_b <= 1.0 + 1e-12:
            # No slack → cannot improve further
            break

        g = b / sum_b          # unit‑sum candidate
        t = 0.5                 # mixing coefficient
        a_new = (1 - t) * best_a + t * g
        val_new = _objective(a_new)

        print(f"[LP] it={it} sum_b={sum_b:.6f} new_val={val_new:.6f}")

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        else:
            # accept the mixed point anyway (it may help the next LP)
            best_a = a_new.copy()

    return best_a, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function of non‑negative heights that minimises
    the ``evaluate_sequence`` objective.  Returns the best sequence found
    as a plain list of floats.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Optional reuse of a previously discovered construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # We focus on a single, reasonably large length; extend if time permits.
    lengths = [4000]
    global_best_seq = None
    global_best_val = float("inf")
    global_best_n = None

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                global_best_n = n
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Keep a handful of promising seeds for deeper work
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [sv[0] for sv in seed_vals[:5]]

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max
        # ------------------------------------------------------------------
        for s in top_seeds:
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_optimize(s, n, deadline, max_iters=6000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                global_best_n = n
                print(f"[overall‑Adam] n={n:5d} new global best={val_adam:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_sub, val_sub = _subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=20000)
            if val_sub < global_best_val:
                global_best_val = val_sub
                global_best_seq = seq_sub.copy()
                global_best_n = n
                print(f"[overall‑SubGrad] n={n:5d} new global best={val_sub:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Linear‑programming improvement
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_lp, val_lp = _lp_refine(global_best_seq, deadline, max_iters=6)
            if val_lp < global_best_val:
                global_best_val = val_lp
                global_best_seq = seq_lp.copy()
                global_best_n = n
                print(f"[overall‑LP] n={n:5d} new global best={val_lp:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Stochastic hill‑climbing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                         max_iters=40000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                global_best_n = n
                print(f"[overall‑Hill] n={n:5d} new global best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Normalise explicitly (objective is scale‑invariant)
    total = np.sum(global_best_seq)
    if total == 0.0:
        global_best_seq = _uniform(global_best_n or 4000)
    else:
        global_best_seq = np.asarray(global_best_seq) / total

    # Optional verification via the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
