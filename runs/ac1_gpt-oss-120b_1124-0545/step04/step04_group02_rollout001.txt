"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The approach consists of three phases:

1. **Large analytic grid** – we enumerate many parametric families
   (U‑shaped product/sum power laws, exponential tails, cosine‑powers,
   Beta‑shaped and discrete Gaussian).  The grid is much finer than the
   reference implementation, and the best candidate is kept.

2. **Adam‑based sub‑gradient descent** – the active index of the
   max‑convolution gives a sub‑gradient; we apply the Adam optimiser
   (momentum + adaptive step‑size) and after each update project onto
   the probability simplex.  The learning‑rate is linearly annealed.
   Random “mass‑transfer” moves and occasional Dirichlet restarts help
   to escape plateaus.  The loop runs until the time budget is almost
   exhausted.

3. **Fine stochastic refinement** – a short Gaussian jitter phase
   explores the neighbourhood of the best vector found.

All numbers are kept non‑negative, summed to one (scale‑invariant), and
the routine respects the 1000 s wall‑time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution a*a using FFT.
    Output length = 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a probability vector a.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Symmetric sum family a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tail a_i ∝ exp(-lam·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power family a_i ∝ |cos(π·i/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient utilities
# ----------------------------------------------------------------------


def subgradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Return a sub‑gradient of the objective at vector a.
    The active set consists of all convolution indices whose value
    lies within `tol` of the maximum.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    max_val = conv.max()
    # treat indices within a tiny absolute tolerance as active
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    # accumulate contributions from every active index
    for k in active:
        rev = k - np.arange(n)
        mask = (rev >= 0) & (rev < n)
        g[mask] += 2.0 * a[rev[mask]]
    if active.size > 0:
        g /= float(active.size)          # average over the active set
    return g


# ----------------------------------------------------------------------
# Adam optimiser step (projected onto simplex)
# ----------------------------------------------------------------------


def adam_step(a: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t: int,
              lr: float,
              beta1: float,
              beta2: float,
              eps: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam update using the sub‑gradient of the max‑convolution.
    The result is projected onto the probability simplex.
    """
    g = subgradient_max_conv(a)
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * (g * g)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation metric.
    The method combines a very fine analytic grid, an Adam‑based
    sub‑gradient descent, and a short stochastic refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.7          # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  Larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3000))
    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # 1) Very fine analytic grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ----- product family ------------------------------------------------
    p_vals = np.linspace(0.05, 2.0, 40)          # finer than before
    off_vals = np.linspace(0.01, 1.5, 30)
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # ----- sum family ----------------------------------------------------
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # ----- exponential tails ---------------------------------------------
    lam_vals = np.logspace(-4, 2, 35)           # 0.0001 … 100
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # ----- cosine‑power --------------------------------------------------
    p_cos_vals = np.linspace(0.3, 8.0, 30)
    for p in p_cos_vals:
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # ----- beta‑shape ----------------------------------------------------
    alpha_vals = np.linspace(-0.5, 3.0, 30)
    beta_vals = np.linspace(-0.5, 3.0, 30)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # ----- Gaussian -------------------------------------------------------
    sigma_vals = np.linspace(0.05 * n, 0.5 * n, 30)
    for sigma in sigma_vals:
        cand = analytic_gaussian(n, sigma)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 2) If a reference sequence is provided, evaluate it as well
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam‑based sub‑gradient descent (projected onto simplex)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters (will be annealed manually)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.4
    lr_end = 5e-5

    iteration = 0
    last_improve = time.time()
    stagnation_limit = 15.0          # seconds without improvement → jitter

    while time.time() < deadline:
        # linearly anneal learning rate
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 5.0))
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # Adam step
        a, m, v = adam_step(a, m, v, iteration + 1, lr, beta1, beta2, eps)

        # occasional random mass‑transfer move (helps escaping flat regions)
        if np.random.rand() < 0.015:
            a = random_move(a, scale=lr * 0.6)

        # very rare Dirichlet restart – keep if it improves globally
        if np.random.rand() < 0.004:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            sc_rand = compute_score(rand_vec)
            if sc_rand < best_score:
                best_score = sc_rand
                best_seq = rand_vec.copy()
                a = rand_vec.copy()
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()
                print(f"[{iteration:06d}] Dirichlet restart improved to {best_score:.6f}")

        # evaluate current vector
        cur_score = compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # jitter restart if we have been stuck for a while
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # 4) Fine‑grained stochastic refinement (last few seconds)
    # ------------------------------------------------------------------
    refine_deadline = start_time + budget_s - 1.0
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=3e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")

    # Return a plain Python list (required by the harness)
    return best_seq.tolist()
