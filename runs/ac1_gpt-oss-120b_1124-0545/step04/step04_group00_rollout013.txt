"""
Low‑dimensional search for the minimum of `evaluate_sequence`.

The method expands the previous three‑stage pipeline:

1. A richer collection of analytic seeds (geometric, power‑law, 
   rational‑power, shifted‑logistic, sine, Gaussian, and a few random
   soft‑max vectors).  The seed set also contains a finely‑tuned power‑law
   family around the empirically good exponent p≈0.5.

2. Adam optimisation of a *smoothed* max loss.  The smoothing (log‑sum‑exp)
   uses a temperature β that is increased slowly (β←β·β_mul) so that the
   algorithm first sees a smooth landscape and then concentrates on the
   true ∞‑norm.  More iterations and a larger initial learning rate are
   allowed than in the reference code.

3. Mirror‑descent (exponentiated‑gradient) sub‑gradient refinement.
   Instead of using a single most‑violating shift, we use the top‑K
   shifts (default K=5) to form a richer sub‑gradient that drives the
   whole autocorrelation envelope toward flatness.

4. A longer simulated‑annealing phase that occasionally performs larger
   “jumps” (Gaussian perturbations) and uses an adaptive temperature
   schedule.

All phases are time‑aware – they stop as soon as the global deadline is
reached.  The best sequence found (as a plain list) is returned.
"""

import time
import numpy as np
from math import log, exp, sqrt, pi

# ----------------------------------------------------------------------
# FFT utilities (convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT.  Returns full length."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / sum(seq)² for non‑negative seq (sum≈1)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric families used as seeds
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """r‑geometric times power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, a: float, b: float) -> np.ndarray:
    """Logistic / sigmoidal shape: a / (1+exp(b*(i - n/2)))."""
    i = np.arange(n, dtype=np.float64)
    a_i = a / (1.0 + np.exp(b * (i - n / 2.0)))
    s = a_i.sum()
    return a_i / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirrored)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of start vectors, already normalised to sum=1."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # Very simple baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.15))

    # Geometric families  (r = 1 - c/n)
    for c in np.linspace(0.3, 2.5, 13):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families (p around the known sweet spot)
    for p in np.linspace(0.1, 1.0, 19):
        seeds.append(_power(n, p))

    # Geometric × power mixes
    for c in np.linspace(0.3, 2.0, 7):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.2, 7):
            seeds.append(_geom_power(n, r, p))

    # Logistic / sigmoidal shapes (steeper transitions)
    for b in (0.3, 0.6, 1.0, 1.5):
        seeds.append(_logistic(n, a=1.0, b=b))

    # Random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # Re‑use a previously discovered construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(np.abs(previous).astype(np.float64))
        seeds[-1] /= seeds[-1].sum()

    # Guard against empty pool
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   beta0: float = 1.0,
                   beta_mul: float = 1.02) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp approximation of max(conv)."""
    # logits for the simplex (log‑softmax parametrisation)
    v = np.log(seed + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = beta0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = np.exp(v - np.max(v))
        a /= a.sum()
        conv = _conv_fft(a, a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weighting -----
        beta = beta * beta_mul
        shift = np.max(conv)                     # numerical shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        # full convolution length = 3n‑2
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient wrt logits (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000,
                        top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style refinement using the sub‑gradient formed by the
    *top_k* largest convolution entries (instead of only the single maximum).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # Keep track of the best objective seen
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:6d} best={cur_val:.6f}")

        # Identify the top‑k indices (ties are fine)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            # argpartition gives unsorted top‑k; we sort them for determinism
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            idxs = idxs[np.argsort(-conv[idxs])]

        # Build sub‑gradient as a sum over the selected shifts
        g = np.zeros_like(a)
        for k_idx in idxs:
            # contribution of shift k_idx:
            i_start = max(0, k_idx - (n - 1))
            i_end = min(n - 1, k_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k_idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # Exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)          # a slightly larger step‑size than before
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that moves a tiny amount of mass between two
    random positions, with an exponentially decaying temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a modest fraction of the current objective
    T0 = max(1e-7, cur_val * 0.04)
    decay = 0.9993

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a small fraction (up to 2 %) from i to j
        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional larger “jumps” – whole‑vector perturbation
        if it % 1200 == 0:
            noise = rng.normal(scale=0.001, size=n)
            pert = cur + noise
            pert = np.clip(pert, 0.0, None)
            if pert.sum() == 0.0:
                pert = _uniform(n)
            else:
                pert /= pert.sum()
            new_val = _objective(pert)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = pert
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = pert.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start moderate and grow if time permits
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000]   # a mix of odd/even sizes

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) generate diverse seeds
        seeds = _make_seeds(n, previous)

        # 2) quick evaluate all seeds, keep a few promising ones
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(7, len(seed_vals))]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on the smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=18000,
                                      beta0=1.0,
                                      beta_mul=1.015)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (using top‑k active shifts)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                          max_iters=120000,
                                          top_k=5)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # expose the best sequence for possible reuse in the next run
    globals()["height_sequence_1"] = np.array(global_best_seq, dtype=np.float64)
    return global_best_seq.tolist()
