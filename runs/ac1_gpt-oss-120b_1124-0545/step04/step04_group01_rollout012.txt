    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)

    temp = 0.01
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        donor = i
        receiver = j
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0: continue

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        # compute new conv
        new_conv = _conv_fft(new, new)
        new_val = 2.0 * n * np.max(new_conv)
        # accept if improvement or according to temperature
        accept = False
        if new_val < cur_val:

        else:
            prob = np.exp(-(new_val - cur_val) / temp)

                best_val = new_val
                best_seq = new.copy()

Propose a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑phase search:

1.  Diverse analytic and random seeds (uniform, geometric, power‑law,
    beta‑like, sine window, Dirichlet draws and the previous best).
2.  Adam optimisation on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The smoothing parameter is increased gradually but
    capped to stay numerically stable.
3.  Exact sub‑gradient descent (exponentiated‑gradient / mirror descent)
    on the true max‑convolution objective, with a long schedule of
    decreasing steps.
4.  A simple simulated‑annealing hill‑climber that moves a small amount
    of mass between two random positions and accepts occasional
    deteriorations to escape shallow local minima.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()

    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _random_logits(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:

    auto‑convolution.  Returns best sequence found and its objective.

        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement using the exact
    sub‑gradient of the max‑auto‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.8 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                    # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _hill_climb_sa(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing with simulated‑annealing acceptance.
    Mass is transferred between two random positions; occasional
    worsening moves are accepted according to a decreasing temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_val = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.05
    temperature = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_conv = _conv_fft(new, new)
        new_val = 2.0 * n * np.max(new_conv)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = new_val
            cur_conv = new_conv
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:6d} best={new_val:.6f}")

        # schedule decay
        if it % 500 == 0:
            temperature *= 0.97
            step_scale *= 0.97
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function of non‑negative heights that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence it discovered as a plain
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # reuse a previously discovered construction, if present
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # try a couple of lengths – larger n gives a finer search space
    lengths = [8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on a smoothed max
        # ------------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=150000)
            if val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing hill climbing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb_sa(global_best_seq, n,
                                            deadline, max_iters=60000)
            if val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑SA] n={n:5d} new best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = 2.0 * 4000 * np.max(_conv_fft(global_best_seq,
                                                        global_best_seq))
        print("[fallback] using uniform distribution")

    # Optional verification (ignore any exception – evaluate_sequence may
    # not be imported yet in the sandbox).
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
