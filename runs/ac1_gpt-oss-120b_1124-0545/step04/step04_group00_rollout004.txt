"""
Propose a highly tuned step‑function.

The search proceeds in three length scales (default 4000 → 8000 → 12000)
and uses a rich pool of parametric seeds (uniform, geometric, power‑law,
Gaussian, sine, cosine, exponential, beta‑pdf, arcsine, random sparse,
mixtures, etc.).  Each seed is quickly evaluated, the best few are
optimised by

1. Adam on a smooth approximation of the max‑convolution (log‑sum‑exp)
   – the logits are kept on the simplex via the softmax Jacobian.
2. Exact sub‑gradient mirror‑descent (exponentiated‑gradient) that
   directly minimises the true max.
3. A cheap simulated‑annealing walk that swaps tiny mass amounts between
   positions.

All phases are time‑aware: they stop as soon as the global deadline
(1000 s − 1 s safety) is reached.  The best sequence found over all
lengths is returned as a plain list of non‑negative floats (already
normalised to sum = 1, which is invariant for the evaluation function).
"""

import time
import numpy as np
from scipy.special import beta as beta_func  # for Beta‑pdf

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns the full result of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The sequence is assumed to be non‑negative and the sum ≠ 0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 0.5 * (1.0 + np.cos(np.pi * (i - centre) / centre))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_pdf(n: int, a: float, b: float) -> np.ndarray:
    """Beta distribution on the grid i/(n-1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # pdf ∝ x^{a‑1} (1‑x)^{b‑1}
    a_vals = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x) - np.log(beta_func(a, b)))
    a_vals = np.nan_to_num(a_vals, nan=0.0, posinf=0.0, neginf=0.0)
    s = a_vals.sum()
    return a_vals / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly random positions with Dirichlet‑distributed weights."""
    idx = rng.choice(n, size=min(k, n), replace=False)
    w = rng.random(len(idx))
    w = w / w.sum()
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of starting points, already normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_cosine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_exponential_decay(n, c=0.003))
    seeds.append(_exponential_decay(n, c=0.008))
    seeds.append(_arcsine(n))

    # beta‑pdf families
    for _ in range(5):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        seeds.append(_beta_pdf(n, a, b))

    # random sparse seeds
    for k_factor in [0.015, 0.03, 0.05]:
        k = max(2, int(k_factor * n))
        seeds.append(_random_sparse(n, k, rng))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # mixture of uniform and gaussian
    for mix in np.linspace(0.2, 0.8, 4):
        u = _uniform(n)
        g = _gaussian(n, sigma_factor=0.12)
        a = mix * u + (1 - mix) * g
        seeds.append(a / a.sum())

    # reuse previous construction if size matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guard against empty collection (should never happen)
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a log‑sum‑exp approximation of the max
    auto‑convolution. Returns the best sequence discovered and its true
    (unsmoothed) objective.
    """
    v = np.log(seed + 1e-30)  # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 2.0          # smoothing strength
    beta_mul = 1.015    # gradual sharpening

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement using the exact sub‑gradient of the true
    max of the auto‑convolution. The update is an exponentiated‑gradient
    step that respects the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass between
    two random positions, with an exponentially decaying temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # Re‑use a previously discovered construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore – start small, then increase if time permits
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) generate a rich seed set
        seeds = _make_seeds(n, previous)

        # 2) quick evaluation of raw seeds; keep the most promising
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on the smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=15000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=80000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

        # store the current best as possible `previous` for larger n
        previous = global_best_seq.copy()

        # early exit if we already beat the target
        if global_best_val <= 1.503:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # expose the final construction for possible reuse in a later run
    globals()["height_sequence_2"] = global_best_seq.copy()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
