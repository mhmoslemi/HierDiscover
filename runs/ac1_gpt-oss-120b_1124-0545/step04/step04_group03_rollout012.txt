def propose_candidate(seed=..., budget_s=..., **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    
    n = kwargs.get("n", 2000)
    eps = 1e-12
    
    # initial a
    a = analytic_candidate(n, p=0.5, offset=0.0, mode='product')
    a = np.maximum(a, eps)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # helper function to evaluate and update best
    def update_if_better(candidate):
        nonlocal best_score, best_a
        sc = compute_score(candidate)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = candidate.copy()
            print(f"[update] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")
    
    # main LP refinement loop
    t = 0.5  # mixing param
    iter_num = 0
    while time.time() < deadline:
        # compute convolution and max bound M
        conv_a = conv_fft_self(a)
        M = conv_a.max()
        # solve LP
        col = np.concatenate([a, np.zeros(n-1)])
        row = np.concatenate([[a[0]], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
        c = -np.ones(n)  # maximize sum(b)
        bounds = [(0, None)] * n
        
        # We can optionally set high time limit per LP
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        except Exception as e:
            print(f"[LP] exception: {e}")
            break
        if not res.success:
            print(f"[LP] failed: {res.message}")
            break
        b = res.x
        # Normalize g to sum 1
        sum_b = b.sum()
        if sum_b == 0:
            # fallback to random perturb
            a = random_move(a, scale=0.01)
            continue
        g = b / sum_b
        # mix
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, eps)
        a_new = a_new / a_new.sum()
        # evaluate
        sc_new = compute_score(a_new)
        # Accept if improvement, else reduce t
        if sc_new < best_score:
            a = a_new
            update_if_better(a_new)
            # maybe increase t slowly
            t = min(0.9, t * 1.05)
        else:
            # try smaller t
            t = max(0.01, t * 0.5)
            # optional: keep a unchanged and maybe random perturb
            # apply small random move to possibly escape
            a = random_move(a, scale=0.01)
        iter_num += 1
        if iter_num % 10 == 0:
            # small local search
            cand = random_move(a, scale=0.02)
            update_if_better(cand)
    # final random refinement
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(best_a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[final] new best = {best_score:.6f} (t={time.time() - start:.1f}s)")
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)
    print(f"[done] best score = {best_score:.6f}")
    return best_a.tolist()

"""
Hybrid optimisation for the step‑function bound.

The programme builds a good analytic start (product / sum power‑law families)
and then refines it by minimising a smooth approximation of the max‑convolution
using a continuation on the smoothing parameter λ.  The smoother is much more
gradual than in the reference solution (more λ‑levels and many more L‑BFGS
iterations) which gives the optimiser time to move out of shallow local minima.
After the continuation a short random‑mass‑transfer hill‑climb finishes the run.

All numbers are kept non‑negative and the vector is always normalised to sum 1,
so the exact objective is simply 2·n·max(conv(a,a)).
"""
import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two not less than x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself via FFT.
    Result length = 2*len(a)-1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex.
    The function normalises a to sum 1 internally.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised analytic candidate (sum = 1).

    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index i to a random index j
    (both uniformly chosen).  The resulting vector is projected back onto the
    probability simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # Euclidean projection onto the simplex
    u = np.sort(a_new)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u - cssv / (np.arange(n) + 1.0)
    rho_idx = np.where(rho > 0)[0]
    if rho_idx.size == 0:
        theta = 0.0
    else:
        rho = rho_idx[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(a_new - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given soft‑max parameter vector x and smoothing λ,
    return the smooth max Φ_λ(a) and its gradient ∇_x Φ_λ,
    where a = softmax(x) and Φ_λ is the log‑sum‑exp smoothing of max(conv).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a (full)
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max: Φ_λ = max_c + (1/λ)·log Σ_k exp(λ·(conv_k‑max_c))
    shift = conv - max_c                    # ≤ 0, stabilises exp
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                  # soft‑max weights, sum to 1

    # gradient wrt a:
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]          # length n
    grad_a = 2.0 * slice_part[::-1]                # ∂Φ/∂a

    # map to x‑space via soft‑max Jacobian J^T∇_aΦ
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine combines a small analytic grid, a long continuation‑L‑BFGS
    optimisation on a smooth max‑convolution surrogate, and a final random
    mass‑transfer local search.  It respects the time budget and always returns
    the best sequence found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid – provides a solid starting point
    # ------------------------------------------------------------------
    analytic_cands = []

    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55, 0.65):
            for off in (0.0, 0.2, 0.5, 1.0):
                # ensure we never divide by zero in the product mode
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, p, off_eff, mode)
                analytic_cands.append(cand)

    analytic_cands.append(np.ones(n) / n)  # uniform baseline

    # If a previous construction is available, include it
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            if ref.sum() > 0:
                analytic_cands.append(ref / ref.sum())

    best_a = None
    best_score = np.inf
    for a0 in analytic_cands:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the smoothed max‑convolution
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # A denser λ schedule (smooth → hard max)
    lam_vals = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0
    ]

    # check for scipy.optimize.minimize availability
    have_scipy = True
    try:
        from scipy.optimize import minimize
    except Exception:                     # pragma: no cover
        have_scipy = False
        print("[info] SciPy minimize unavailable – will skip L‑BFGS stage.")

    if have_scipy:
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            # allocate a generous number of iterations while respecting the budget
            remaining = deadline - time.time()
            max_iter = int(min(2000, max(50, remaining * 80)))   # approx 80 L‑BFGS steps per sec
            res = minimize(
                compute_phi_and_grad,
                x,
                args=(lam, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'disp': False}
            )
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
    else:
        # fallback: simple Adam optimiser (very rare to be needed)
        alpha = 0.05
        beta1, beta2 = 0.9, 0.999
        eps_opt = 1e-8
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        t_counter = 0
        for lam in lam_vals:
            if time.time() >= deadline:
                break
            while time.time() < deadline:
                t_counter += 1
                phi, grad_x = compute_phi_and_grad(x, lam, n)
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t_counter)
                v_hat = v / (1 - beta2 ** t_counter)
                step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t_counter >= 3000:
                    break

    # ------------------------------------------------------------------
    # 3) Light random‑mass‑transfer local search (escape plateaus)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand.copy()
            a = cand
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            # after an improvement shrink the step size a little
            scale = max(0.001, scale * 0.9)
        else:
            # slowly increase step size again to keep exploration alive
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Convert to plain Python list as required by the harness
    return best_a.tolist()
