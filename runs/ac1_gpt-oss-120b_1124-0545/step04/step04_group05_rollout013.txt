def gradient_step(a, η, k):
    # Compute subgradient g: 2 * a shifted
    n = len(a)
    g = np.zeros_like(a)
    low = max(0, k - (n-1))
    high = min(k, n-1)
    # i in [low, high]
    # For i from low to high:
    # g[i] = 2 * a[k-i]
    idx = np.arange(low, high+1)
    g[idx] = 2 * a[k - idx]
    # gradient descent:
    a_new = a - η * g
    # clip negative, renorm
    a_new = np.maximum(a_new, 0)
    total = a_new.sum()
    if total == 0:
        a_new = np.full_like(a, 1.0/n)
    else:
        a_new /= total
    return a_new

import time
import numpy as np

def proj_simplex(v):
    """Project an array v onto the probability simplex (nonnegative, sum = 1)."""
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def shape_from_p(p, n):
    """Return a normalised non‑negative step function for exponent p.
    The unnormalised values are ((i+0.5)*(n-i-0.5))^{-p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (-p)   # equivalent to ((i+0.5)*(n-i-0.5))^{-p}
    # avoid infinities (should not happen for positive p)
    a = vals
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def max_convolution_fft(a, fft_len):
    """Return max of the linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2 * a.size - 1]
    return conv.max(), conv

def exact_score(n, conv_max):
    """Exact value of evaluate_sequence for a normalised vector."""
    return float(2.0 * n * conv_max)

def subgradient_descent(a, n, fft_len, max_iters, deadline, start_time):
    """Projected sub‑gradient descent on max(a*a) with simplex projection."""
    best_a = a.copy()
    conv_max, _ = max_convolution_fft(a, fft_len)
    best_score = exact_score(n, conv_max)
    cur_a = a.copy()
    cur_score = best_score

    eta0 = 0.5   # initial step size
    it = 0
    while time.time() < deadline - 0.2 and it < max_iters:
        it += 1
        # compute current convolution and argmax
        conv_max, conv = max_convolution_fft(cur_a, fft_len)
        k = int(np.argmax(conv))

        # sub‑gradient at position k
        g = np.zeros_like(cur_a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1)
        g[idx] = 2.0 * cur_a[k - idx]

        # step size decay
        eta = eta0 * (1.0 - it / max_iters)
        if eta <= 0:
            break

        # gradient step and projection onto simplex
        cur_a = cur_a - eta * g
        cur_a = np.maximum(cur_a, 0)
        # renormalise (simple projection)
        s = cur_a.sum()
        if s == 0:
            cur_a = np.full(n, 1.0 / n)
        else:
            cur_a /= s

        # evaluate new objective
        conv_max_new, _ = max_convolution_fft(cur_a, fft_len)
        cur_score = exact_score(n, conv_max_new)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[subgrad {it:05d}] improved score = {best_score:.6f}")

    return best_a, best_score

def simulated_annealing(a, n, fft_len, start_score, deadline, start_time, rng):
    """Local stochastic refinement using simulated annealing."""
    best_a = a.copy()
    best_score = start_score
    cur_a = a.copy()
    cur_score = start_score

    total_time = deadline - time.time()
    if total_time < 5.0:
        return best_a, best_score

    # temperature schedule (linear cooling)
    T_start = best_score * 0.01   # 1 % of current score
    T_end = best_score * 1e-8

    it = 0
    while time.time() < deadline - 0.1:
        elapsed = time.time() - start_time
        frac = min(1.0, (time.time() - (deadline - total_time)) / total_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount of mass to transfer
        max_move = cur_a[i]
        if max_move == 0.0:
            continue
        # allow occasional large jumps
        if rng.random() < 0.01:
            delta = rng.random() * max_move   # up to full mass
        else:
            delta = rng.random() * max_move * 0.05  # up to 5 %

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        conv_max_new, _ = max_convolution_fft(a_new, fft_len)
        new_score = exact_score(n, conv_max_new)

        # Metropolis accept
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[SA {it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine consists of three stages:
    1.  One‑dimensional sweep over the exponent p of the family
        a_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p} (uniform at p=0, arcsine at p≈0.5).
        A coarse grid is evaluated, followed by a finer local grid to locate
        the best exponent.
    2.  A short projected sub‑gradient descent that directly minimises the
        exact max‑convolution (using the true argmax instead of a soft‑max).
    3.  A simulated‑annealing style local search that performs random mass
        transfers between two coordinates while the temperature is cooled
        linearly.  All stages respect the total time budget.
    The final sequence is normalised to sum = 1 and all entries are
    non‑negative, which is sufficient for the scoring function.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5

    # Length of the sequence – a default that works well with the FFT code.
    n = int(kwargs.get("n", 2000))

    # FFT length – next power of two ≥ 2·n‑1.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # 0)  Parameter sweep over exponent p
    # ------------------------------------------------------------
    # Coarse sweep
    p_vals = np.linspace(0.0, 1.0, 41)   # step of 0.025
    best_p = None
    best_score = np.inf
    best_seq = None

    for p in p_vals:
        seq = shape_from_p(p, n)
        conv_max, _ = max_convolution_fft(seq, fft_len)
        score = exact_score(n, conv_max)
        if score < best_score:
            best_score = score
            best_p = p
            best_seq = seq.copy()
        # early exit if we run out of time
        if time.time() >= overall_deadline - 5.0:
            break

    # Refine around the best p – use a finer grid.
    if best_p is not None:
        low = max(0.0, best_p - 0.05)
        high = min(1.0, best_p + 0.05)
        p_fine = np.linspace(low, high, 81)   # step ≈0.00125
        for p in p_fine:
            seq = shape_from_p(p, n)
            conv_max, _ = max_convolution_fft(seq, fft_len)
            score = exact_score(n, conv_max)
            if score < best_score:
                best_score = score
                best_p = p
                best_seq = seq.copy()
            if time.time() >= overall_deadline - 5.0:
                break

    # Ensure we have a valid start.
    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(n, 1.0 / n)
        conv_max, _ = max_convolution_fft(best_seq, fft_len)
        best_score = exact_score(n, conv_max)
        best_p = 0.0

    print(f"[init] best p = {best_p:.5f}, score = {best_score:.6f}")

    # ------------------------------------------------------------
    # 1) Projected sub‑gradient descent on exact max‑convolution
    # ------------------------------------------------------------
    subgrad_deadline = start_time + 0.15 * budget_s
    if time.time() < subgrad_deadline - 2.0:
        best_seq, best_score = subgradient_descent(
            best_seq, n, fft_len,
            max_iters=3000,
            deadline=subgrad_deadline,
            start_time=start_time,
        )
        print(f"[subgrad] after descent score = {best_score:.6f}")

    # ------------------------------------------------------------
    # 2) Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------
    final_seq, final_score = simulated_annealing(
        best_seq, n, fft_len, best_score,
        deadline=overall_deadline,
        start_time=start_time,
        rng=rng,
    )
    print(f"[done] total time = {time.time() - start_time:.1f}s, best score = {final_score:.6f}")

    # Normalise once more – guards against tiny numerical drift.
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total == 0.0:
        final_seq = np.full(n, 1.0 / n)
    else:
        final_seq = final_seq / total

    return final_seq.tolist()

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project the vector *v* onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of the algorithm by Duchi et al. (O(n log n)).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):               # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def shape_from_p(p: float, n: int) -> np.ndarray:
    """
    Produce a normalised sequence belonging to the one‑parameter family

        a_i ∝ ((i+0.5)*(n‑i‑0.5))^{-p}   ,   i = 0,…,n‑1

    p = 0 gives the uniform distribution,
    p ≈ 0.5 reproduces the classic arcsine shape.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    # ((i+0.5)*(n‑i‑0.5)) ≈ i·(n‑i); both are > 0 for i∈{0.5,…,n‑0.5}
    vals = (i * (n - i)) ** (-p)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Linear convolution a*a using a real FFT and return (max, full_conv).
    The length of the returned convolution is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max()), conv


def exact_score(n: int, conv_max: float) -> float:
    """
    Exact value of the evaluation function for a normalised vector
    (sum = 1).  Because the function is scale‑invariant,
    this is exactly `evaluate_sequence` when the vector is already
    non‑negative.
    """
    return float(2.0 * n * conv_max)


# ----------------------------------------------------------------------
# Stage 1 – 1‑D parameter sweep (exponent p)
# ----------------------------------------------------------------------


def sweep_exponent(n: int, fft_len: int, deadline: float) -> tuple[np.ndarray, float, float]:
    """Coarse → fine search over the exponent p."""
    best_seq = None
    best_score = np.inf
    best_p = None

    # ------ coarse grid -------------------------------------------------
    p_vals = np.linspace(0.0, 1.0, 41)          # step ≈ 0.025
    for p in p_vals:
        seq = shape_from_p(p, n)
        conv_max, _ = max_convolution_fft(seq, fft_len)
        score = exact_score(n, conv_max)
        if score < best_score:
            best_score, best_seq, best_p = score, seq.copy(), p
        if time.time() > deadline - 5.0:
            break

    # ------ fine grid around the best coarse point ----------------------
    if best_p is not None:
        lo = max(0.0, best_p - 0.05)
        hi = min(1.0, best_p + 0.05)
        p_fine = np.linspace(lo, hi, 81)      # step ≈ 0.00125
        for p in p_fine:
            seq = shape_from_p(p, n)
            conv_max, _ = max_convolution_fft(seq, fft_len)
            score = exact_score(n, conv_max)
            if score < best_score:
                best_score, best_seq, best_p = score, seq.copy(), p
            if time.time() > deadline - 5.0:
                break

    # fallback – uniform (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        conv_max, _ = max_convolution_fft(best_seq, fft_len)
        best_score = exact_score(n, conv_max)
        best_p = 0.0

    print(f"[sweep] best p = {best_p:.5f}, score = {best_score:.6f}")
    return best_seq, best_score, best_p


# ----------------------------------------------------------------------
# Stage 2 – Projected sub‑gradient descent on the exact max‑conv.
# ----------------------------------------------------------------------


def subgradient_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient method:
        - compute the exact arg‑max of the convolution,
        - use the sub‑gradient g_i = 2·a_{k‑i},
        - take a diminishing step and project back onto the simplex.
    """
    best_a = a.copy()
    conv_max, _ = max_convolution_fft(a, fft_len)
    best_score = exact_score(n, conv_max)

    cur_a = a.copy()
    cur_score = best_score
    eta0 = 0.5
    it = 0

    while time.time() < deadline - 0.2 and it < max_iters:
        it += 1
        # current convolution and its maximal index
        conv_max, conv = max_convolution_fft(cur_a, fft_len)
        k = int(np.argmax(conv))

        # sub‑gradient for the arg‑max entry
        g = np.zeros_like(cur_a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1)
        g[idx] = 2.0 * cur_a[k - idx]

        # step (linear decay)
        eta = eta0 * (1.0 - it / max_iters)
        if eta <= 0.0:
            break

        # gradient step + simplex projection (cheap: clip + renorm)
        cur_a = cur_a - eta * g
        cur_a = np.maximum(cur_a, 0.0)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full(n, 1.0 / n)
        else:
            cur_a /= s

        # evaluate new objective
        conv_max_new, _ = max_convolution_fft(cur_a, fft_len)
        cur_score = exact_score(n, conv_max_new)

        if cur_score < best_score:
            best_score, best_a = cur_score, cur_a.copy()
            print(f"[subgrad {it:05d}] improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – Simulated‑annealing local refinement
# ----------------------------------------------------------------------


def simulated_annealing(
    start_a: np.ndarray,
    n: int,
    fft_len: int,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Classic SA: repeatedly transfer a random amount of mass from i → j.
    Acceptance follows the Metropolis rule with a linearly cooled temperature.
    """
    best_a = start_a.copy()
    best_score = start_score
    cur_a = start_a.copy()
    cur_score = start_score

    total_time = deadline - time.time()
    if total_time < 5.0:
        return best_a, best_score

    # temperature schedule (linear cooling)
    T_start = best_score * 0.01      # 1 % of the current score
    T_end = best_score * 1e-8

    it = 0
    while time.time() < deadline - 0.1:
        # linear cooling based on elapsed proportion
        elapsed = time.time() - (deadline - total_time)
        frac = min(1.0, elapsed / total_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # random mass transfer, sometimes large, usually modest
        max_move = cur_a[i]
        if max_move <= 0.0:
            continue
        if rng.random() < 0.01:                     # occasional large jump
            delta = rng.random() * max_move
        else:                                        # typical small move
            delta = rng.random() * max_move * 0.05

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate proposal
        conv_max_new, _ = max_convolution_fft(a_new, fft_len)
        new_score = exact_score(n, conv_max_new)

        # Metropolis acceptance
        if (new_score < cur_score) or (rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12))):
            cur_a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
                print(f"[SA {it:07d}] new best = {best_score:.6f} (t={time.time() - (deadline - total_time):.1f}s)")

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method consists of three stages:

    1. **1‑D sweep** over the exponent *p* of the family
       a_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑p}.  Coarse then fine grids locate
       a promising shape (uniform at p=0, arcsine at p≈0.5).
    2. **Projected sub‑gradient descent** that directly reduces the exact
       max‑convolution (using the true arg‑max as sub‑gradient).  The iterate
       is kept on the probability simplex.
    3. **Simulated‑annealing refinement** performing random mass transfers
       between two coordinates while the temperature is cooled linearly.
       All stages respect the allotted 1000 s budget.

    The final sequence is normalised to sum = 1 and all entries are
    non‑negative – sufficient for the scoring function.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5

    # length of the candidate (default 2000, a power‑of‑two FFT size works well)
    n = int(kwargs.get("n", 2000))

    # FFT length – next power of two ≥ 2·n‑1.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 0)  Try to reuse a previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=float)
        if cand.size == n and np.isfinite(cand).all() and cand.sum() > 0.0:
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            conv_max, _ = max_convolution_fft(cand, fft_len)
            score = exact_score(n, conv_max)
            best_seq, best_score, best_p = cand, score, None
            print(f"[reuse] imported sequence, score = {best_score:.6f}")
        else:
            best_seq, best_score, best_p = None, np.inf, None
    else:
        best_seq, best_score, best_p = None, np.inf, None

    # ------------------------------------------------------------------
    # 1)  1‑D sweep over exponent p (if we have no good seed)
    # ------------------------------------------------------------------
    if best_seq is None:
        sweep_deadline = start_time + 0.10 * budget_s   # 10 % of total time
        best_seq, best_score, best_p = sweep_exponent(n, fft_len, sweep_deadline)

    # ------------------------------------------------------------------
    # 2)  Projected sub‑gradient descent (≈15 % of total time)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.25 * budget_s
    if time.time() < subgrad_deadline - 1.0:
        best_seq, best_score = subgradient_descent(
            best_seq,
            n,
            fft_len,
            max_iters=3000,
            deadline=subgrad_deadline,
        )
        print(f"[subgrad] after descent score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Simulated‑annealing fine tuning (remaining budget)
    # ------------------------------------------------------------------
    final_seq, final_score = simulated_annealing(
        best_seq,
        n,
        fft_len,
        best_score,
        deadline=overall_deadline,
        rng=rng,
    )
    print(f"[done] total time = {time.time() - start_time:.1f}s, best score = {final_score:.6f}")

    # final safety normalisation
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total == 0.0:
        final_seq = np.full(n, 1.0 / n)
    else:
        final_seq = final_seq / total

    return final_seq.tolist()
