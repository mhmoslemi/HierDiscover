def frank_wolfe_step(a, n, max_iter=5, time_budget=...):
    # a sum = 1
    # returns improved a after steps
    start = time.time()
    for _ in range(max_iter):
        if time.time() - start >= time_budget:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        # Build Toeplitz matrix A (2n-1, n)
        col = np.concatenate([a, np.zeros(n-1)])
        row = np.concatenate([a[:1], np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)
        # Solve LP
        c = -np.ones(n)  # maximize sum => minimize -sum
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        if not res.success:
            break
        b = res.x
        if b.sum() <= 1e-12:
            break
        # Scale b to have sum 1 (same as a)
        g = b / b.sum()
        # Interpolation factor - we can compute t that minimizes max(conv((1-t)a + t g, ...))

"""
Propose a non‑negative step function (probability vector) that minimises

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three complementary phases
------------------------------------------------
1. Adam optimiser on the log‑softmax parametrisation.
   • A temperature‑annealed soft‑max approximates the max‑operator.
   • Adam updates move the distribution towards lower convolution peaks.
2. A tiny deterministic hill‑climber that directly reduces the current
   maximum convolution entry by moving a little mass from the most
   responsible coordinate to a low‑mass coordinate.
3. (Optional) short simulated‑annealing style random perturbations
   to escape shallow local minima.

Multiple independent starts (including the historic `height_sequence_1`
if present) are tried within the total time budget.  The best sequence
found is returned as a plain ``list[float]``.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    return exp_v / exp_v.sum()


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _score_from_a(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    conv = np.convolve(a, a)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _adam_optimize(a_init: np.ndarray, n: int, alloc_s: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a soft‑max temperature schedule.
    Returns a (still normalised) probability vector and its exact score.
    """
    # log‑parametrisation (soft‑max guarantees feasibility)
    v = np.log(a_init + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedules (linear in normalised time p∈[0,1])
    beta_start, beta_end = 0.1, 3000.0      # soft‑max temperature
    lr_start, lr_end = 0.25, 0.001         # learning‑rate

    # FFT length (covers convolution of length 3·n‑2)
    fft_len = _next_pow2(3 * n - 2)

    # bookkeeping
    start = time.time()
    deadline = start + max(0.0, alloc_s - 0.05)  # safety margin
    best_a = a_init.copy()
    # exact score for a probability vector is 2·n·max(conv)
    best_score = _score_from_a(best_a, n)
    last_improve = start
    it = 0

    # after this fraction of the allocated time we switch to the hard‑max mask
    mask_phase_start = 0.70

    while time.time() < deadline:
        elapsed = time.time() - start
        # progress fraction p ∈ [0,1]
        p = min(1.0, elapsed / (alloc_s - 5.0 if alloc_s > 5 else 1.0))

        # temperature and learning‑rate schedules
        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        # ------- current probabilities (simplex) -----
        a = softmax(v)               # Σa = 1, a_i ≥ 0

        # ------- convolution a * a -------------------
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # ------- weight vector w  --------------------
        if p < mask_phase_start:
            # smooth soft‑max over the convolution entries
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
        else:
            # hard mask on the true maximum (sub‑gradient of max)
            max_val = conv_aa.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv_aa)
                mask[np.argmax(conv_aa)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # ------- gradient wrt a (via FFT) -------------
        fw = np.fft.fft(w, fft_len)

        # (w * a) – linear convolution, first n entries
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # w ⋆ a – linear cross‑correlation, first n entries
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        # gradient of the (smoothed) max‑objective
        g = 2.0 * n * (g1 + g2)

        # chain rule for the soft‑max parametrisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # --------------- Adam update ----------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # --------------- exact score & bookkeeping -------
        cur_score = _score_from_a(a, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            # occasional progress report
            if it % 2000 == 0:
                print(
                    f"[{it:07d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # --------------- restart on stagnation ----------
        if time.time() - last_improve > 20.0:
            jitter = rng.normal(scale=0.01, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(
                f"[{it:07d}] restart after {elapsed:.1f}s (no improvement)"
            )

    return best_a, best_score


def _hill_climb(a: np.ndarray, n: int, time_allow: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climber that focuses on the current maximal
    convolution entry.  It repeatedly moves a small amount of mass from
    the most responsible coordinate to a low‑mass coordinate.
    """
    start = time.time()
    best_a = a.copy()
    best_score = _score_from_a(best_a, n)

    while time.time() - start < time_allow:
        # current convolution and its global maximum
        conv = np.convolve(best_a, best_a)[: 2 * n - 1]
        max_val = conv.max()
        k = int(np.argmax(conv))                   # index of a peak

        # contribution of each i to this peak (a_i * a_{k-i})
        i_idx = np.arange(n)
        j_idx = k - i_idx
        valid = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(best_a)
        contrib[valid] = best_a[i_idx[valid]] * best_a[j_idx[valid]]

        # source index = the one that participates most heavily
        src = int(np.argmax(contrib))
        if best_a[src] <= 1e-15:
            break  # nothing left to move

        # target index = a currently light coordinate (avoids creating a new peak)
        tgt = int(np.argmin(best_a))

        # move only a tiny fraction – aggressive moves may overshoot
        eps = min(best_a[src] * 0.05, 1e-3)   # ≤5 % of src weight, capped
        if eps <= 0:
            break

        cand = best_a.copy()
        cand[src] -= eps
        cand[tgt] += eps

        cand_score = _score_from_a(cand, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            # occasional log
            print(f"[hill‑climb] improved to {best_score:.6f}")
        else:
            # if no improvement, try a random tiny perturbation
            i, j = np.random.randint(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            eps2 = min(best_a[i] * 0.02, 5e-4)
            cand2 = best_a.copy()
            cand2[i] -= eps2
            cand2[j] += eps2
            cand2_score = _score_from_a(cand2, n)
            if cand2_score < best_score - 1e-12:
                best_a = cand2
                best_score = cand2_score
                print(f"[hill‑climb] random tweak improved to {best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    rng_global = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Prepare several independent start vectors (including historic)
    # ------------------------------------------------------------------
    init_vectors = []

    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        init_vectors.append(a0)

    # uniform baseline
    init_vectors.append(np.full(n, 1.0 / n))

    # a few random Dirichlet starts
    for _ in range(4):
        a = rng_global.dirichlet(np.full(n, 0.5))
        init_vectors.append(a)

    global_best = None
    global_best_score = float("inf")

    attempts = len(init_vectors)

    for att, a_start in enumerate(init_vectors):
        time_left = deadline - time.time()
        if time_left < 30.0:
            break
        # allocate a fair share of the remaining time to this attempt
        remaining_attempts = attempts - att
        alloc = time_left * 0.90 / remaining_attempts

        # ------------------------------------------------------------------
        # Phase A – Adam optimisation (soft‑max + annealing)
        # ------------------------------------------------------------------
        a_opt, score_opt = _adam_optimize(
            a_init=a_start,
            n=n,
            alloc_s=alloc * 0.55,                     # ~55 % of this attempt
            rng=np.random.default_rng(seed + att + 1000),
        )
        # ------------------------------------------------------------------
        # Phase B – deterministic hill climbing
        # ------------------------------------------------------------------
        a_opt, score_opt = _hill_climb(
            a_opt,
            n,
            time_allow=alloc * 0.40,                 # ~40 % of this attempt
        )
        # ------------------------------------------------------------------
        # Phase C – brief random tweak (simulated annealing style)
        # ------------------------------------------------------------------
        end_time = time.time()
        remaining = deadline - end_time
        if remaining > 5.0:
            # cheap random walk – each step recomputes the exact score
            a_cur = a_opt.copy()
            cur_score = score_opt
            t_start = time.time()
            while time.time() - t_start < min(remaining, 5.0):
                i, j = rng_global.integers(0, n, size=2)
                if i == j or a_cur[i] <= 1e-15:
                    continue
                eps = rng_global.random() * min(a_cur[i] * 0.03, 5e-4)
                a_new = a_cur.copy()
                a_new[i] -= eps
                a_new[j] += eps
                new_score = _score_from_a(a_new, n)
                if new_score < cur_score:
                    a_cur, cur_score = a_new, new_score
            a_opt, score_opt = a_cur, cur_score

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {att}] new global best = {global_best_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 2) Fallback to uniform if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if global_best is None:
        global_best = np.full(n, 1.0 / n)

    # final safety – enforce non‑negativity and normalisation
    global_best = np.maximum(global_best, 0.0)
    global_best = global_best / global_best.sum()

    total_time = time.time() - start_time
    print(f"[done] best score = {global_best_score:.6f}, time = {total_time:.1f}s")
    return global_best.tolist()
