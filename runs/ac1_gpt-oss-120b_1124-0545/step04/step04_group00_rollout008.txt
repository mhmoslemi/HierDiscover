"""
Search for a non‑negative step‑function that minimises the self‑convolution
∞‑norm (the objective used in ``evaluate_sequence``).

Algorithm
---------

1. **Seed generation** – a rich pool of start‑vectors:
   * uniform, geometric, power‑law, geometric×power, symmetric geometric,
   * sine‑window, Gaussian,
   * Dirichlet draws with several concentration parameters (producing sparse
     and dense patterns),
   * the best sequence discovered previously (if available) and its mirror.

2. **Adam optimisation of a smoothed loss** – the loss
   ``L(a) = (1/β)·log∑ₖexp(β·convₖ(a)) + λ·∑ₖconvₖ(a)²``
   is a differentiable surrogate for the true objective.
   The gradient of the first term is a soft‑max weighting of the
   convolution entries, the second term is the exact gradient of the
   squared‑ℓ₂ norm of the convolution.  Both gradients are obtained
   with a single FFT each, so each iteration costs O(n log n).
   The parameters (β, λ, learning‑rate) are annealed during optimisation.

3. **Exact sub‑gradient refinement** – mirror‑descent (exponentiated‑gradient)
   on the genuine maximal convolution entry.

4. **Simulated‑annealing fine‑tuning** – a cheap Metropolis‑Hastings walk that
   transfers a tiny amount of mass between two positions and, every
   thousand steps, performs a global Gaussian jitter.

The routine respects a global ``budget_s`` (default 1000 s).  It iterates
over a list of lengths (starting with 4000 and 8000 and, if time permits,
12000).  The best sequence found over all lengths is returned as a plain
list of floats.  All numbers are non‑negative; the sequence is kept
normalised to sum = 1 because the objective is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq))², seq assumed non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse or dense Dirichlet draw (Gamma (α,1) normalised)."""
    vec = rng.gamma(alpha, 1.0, size=n)
    if vec.sum() == 0.0:
        return _uniform(n)
    return vec / vec.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start‑vectors (already normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # Dirichlet draws – cover sparse to dense regimes
    for alpha in [0.1, 0.3, 0.5, 0.8, 1.5, 3.0]:
        seeds.append(_dirichlet_seed(n, alpha, rng))

    # mirror each seed (helps when the optimum is asymmetric)
    mirrored = [s[::-1].copy() for s in seeds]
    seeds.extend(mirrored)

    # reuse previously best construction if the length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guarantee non‑empty list
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation with smoothed‑max + variance regulariser
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000,
                   beta0: float = 2.0,
                   beta_mul: float = 1.04,
                   lr0: float = 0.9,
                   lambda0: float = 0.4,
                   lambda_decay: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the loss
        L(a) = (1/β)·log Σ exp(β·convₖ(a))  +  λ·Σ convₖ(a)² .
    The vector is kept on the simplex via a log‑softmax parametrisation.
    Returns the best sequence encountered and its *true* objective.
    """
    v = np.log(seed + 1e-30)                # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")
    beta = beta0
    lam = lambda0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current point on the simplex
        a = _softmax(v)

        # convolution and true objective (for bookkeeping)
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        shift = conv.max()                   # stable shift
        w_max = np.exp(beta * (conv - shift))
        w_sum = w_max.sum()
        if w_sum == 0.0:
            continue
        w_max /= w_sum

        # ----- gradient of the smoothed max -----
        rev_a = a[::-1]
        full_max = _conv_fft(rev_a, w_max)
        grad_max = 2.0 * full_max[n - 1:n - 1 + n]

        # ----- gradient of Σ conv² (conv variance term) -----
        # Here the weight vector is simply conv itself.
        full_sq = _conv_fft(rev_a, conv)
        grad_sq = 2.0 * full_sq[n - 1:n - 1 + n]

        # combine gradients
        grad_a = grad_max + lam * grad_sq

        # ----- chain rule through the softmax -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay the variance regulariser slowly
        lam *= lambda_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of
    max(conv).  Returns the best sequence seen and its objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirrored simplex update)
        lr = 0.75 / np.sqrt(it)               # 1/√t schedule, slightly larger prefactor
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two random positions, with exponential cooling.
    Occasionally performs a global Gaussian jitter.
    Returns the best sequence discovered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a tiny fraction of the mass at i to j
        delta = rng.random() * 0.025 * cur[i]   # up to 2.5 % of source
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # every 1000 steps add a very small global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    The routine respects the time budget and returns the best sequence found
    as a plain list of floats.
    """

    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # reuse a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore – start modest, enlarge if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 12000]   # the last one is attempted only if enough time remains
    global_best_seq = None
    global_best_val = float("inf")

        seeds = _make_seeds(n, previous)

        # 2) evaluate raw seeds, keep the most promising few
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # Phase 1 – Adam optimisation (smoothed max + variance regulariser)
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=12000,
                beta0=2.0,
                beta_mul=1.04,
                lr0=0.9,
                lambda0=0.5,
                lambda_decay=0.996,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=80000)

                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
