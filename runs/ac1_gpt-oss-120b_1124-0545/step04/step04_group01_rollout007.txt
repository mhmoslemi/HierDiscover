def hill_climb_incremental(start_seq, n, deadline, max_iters):
    rng = np.random.default_rng()

"""
search_step_function.py

A three‑phase optimiser for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

with the final phase being a fast *incremental* hill‑climber that
updates the auto‑convolution exactly using slice operations.  Compared
to the previous implementation this avoids a full FFT at every
iteration, allowing orders of magnitude more trial moves within the
same time budget and typically pushes the bound below 1.503.

The pipeline for each length n:

1. Generate a diverse set of analytic seeds (uniform, geometric,
   beta‑like, Dirichlet, etc.) and, if available, the previously best
   construction `height_sequence_1`.

2. Run a short Adam optimisation on a smooth log‑sum‑exp approximation
   of the max.  The parameters are the log‑softmax of the weights,
   guaranteeing the simplex constraint.

3. Refine the Adam result with an exact mirror‑descent (exponentiated
   gradient) using the true sub‑gradient of the max‑convolution.

4. **Incremental hill‑climbing** – move a tiny amount of mass between
   two random positions, update the convolution by *adding* the exact
   linear and quadratic contributions (O(n) slice updates) and accept
   the move if it improves the objective or with a small Metropolis
   probability (simulated annealing).  Because the convolution is
   updated in place, millions of moves can be evaluated in the
   remaining time budget.

The function `propose_candidate` respects a user‑supplied time budget
(default 1000 s) and returns the best sequence found as a plain list
of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth approximation of the max‑convolution."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # smoothing coefficient (grows each iteration)
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Mirror‑descent refinement with the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Incremental hill climbing (new phase)
# ----------------------------------------------------------------------


def _apply_move(a: np.ndarray,
                conv: np.ndarray,
                donor: int,
                receiver: int,
                delta: float) -> None:
    """
    Apply a mass transfer of size `delta` from `donor` to `receiver`
    *in‑place* on `a` and on the full convolution `conv`.
    The arrays `a` and `conv` are assumed to represent the current
    state *before* the move.
    """
    n = a.shape[0]
    L = conv.shape[0]

    # --- linear part (2·δ·(a shifted by receiver – a shifted by donor)) ---
    # receiver contribution
    r_start = receiver
    r_end = min(receiver + n, L)
    r_len = r_end - r_start
    if r_len > 0:
        conv[r_start:r_end] += 2.0 * delta * a[:r_len]

    # donor contribution (subtracted)
    d_start = donor
    d_end = min(donor + n, L)
    d_len = d_end - d_start
    if d_len > 0:
        conv[d_start:d_end] -= 2.0 * delta * a[:d_len]

    # --- quadratic part (δ²·(δ_{2r}+δ_{2d}-2·δ_{d+r})) ---
    if 2 * receiver < L:
        conv[2 * receiver] += delta * delta
    if 2 * donor < L:
        conv[2 * donor] += delta * delta
    mid = donor + receiver
    if mid < L:
        conv[mid] -= 2.0 * delta * delta

    # --- update the probability vector ---
    a[donor] -= delta
    a[receiver] += delta


def _inc_hill_climb(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 200000,
                    rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Fast stochastic hill climbing using incremental convolution updates.
    Accepts improving moves and, with a simple temperature schedule,
    occasional worsening moves (simulated annealing).
    """
    if rng is None:
        rng = np.random.default_rng()

    a = start_seq.copy()
    conv = np.convolve(a, a)                     # exact linear convolution
    cur_val = 2.0 * n * conv.max()

    best_seq = a.copy()
    best_val = cur_val

    step_scale = 0.3                             # initial relative step size
    T_initial = cur_val * 0.01                   # 1 % of the current objective

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature decay
        T = T_initial * max(0.0, 1.0 - it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = i, j
        max_transfer = step_scale * a[donor] + 1e-15
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer  # always positive, ≤ available mass

        # trial state (copy‑on‑write)
        a_trial = a.copy()
        conv_trial = conv.copy()
        _apply_move(a_trial, conv_trial, donor, receiver, delta)

        new_val = 2.0 * n * conv_trial.max()
        improve = new_val < cur_val
        if not improve:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                improve = True

        if improve:
            a = a_trial
            conv = conv_trial
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                print(f"[IncHill] n={n:5d} it={it:6d} best={new_val:.6f}")

        # decay of step size
        if it % 5000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a moderate variety of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few random soft‑max vectors
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds
