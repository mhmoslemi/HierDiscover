"""
Propose a candidate step‑function using a two‑phase algorithm:

Phase 1 – Projected (sub)gradient descent on the exact ℓ∞‑norm of the
self‑convolution.  The objective is
    F(a) = 2·n·max(conv(a,a))          (the sum of a is kept equal to 1)
A subgradient is obtained from the index where the convolution attains
its maximum; the update is performed on the probability simplex by
projecting onto the ℓ¹‑simplex.  The step size decays like 1/√t and
restarts are triggered after a stagnation period.

Phase 2 – Simulated‑annealing style local refinement.  Starting from the
best vector found by phase 1, random mass transfers between two
coordinates are accepted if they improve the exact objective or with a
Metropolis probability that decays linearly in the remaining time.

Both phases respect the 1000 s budget, use only NumPy (no external
solvers) and return a non‑negative list that can be fed directly to
`evaluate_sequence`.
"""

import time

def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (kept for compatibility)."""
    shift = v - np.max(v)
    exp_v = np.exp(shift)
    return exp_v / exp_v.sum()


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a non‑negative probability vector of length *n*.
    Preference is given to a previously found global variable
    `height_sequence_1`; otherwise the classic arcsine (inverse‑sqrt)
    shape is used and normalised to sum to 1.
    """
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 /= a0.sum()
            return a0

    # classic arcsine shape  (inverse‑sqrt)
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 /= a0.sum()
    return a0


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return the exact maximum entry of the linear self‑convolution a*a."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("project_onto_simplex works on 1‑D arrays only")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def exact_score_from_max(n: int, conv_max: float) -> float:
    """Exact score for a probability vector (sum = 1)."""
    return float(2.0 * n * conv_max)

            T = T_start * (1.0 - frac) + T_end * frac

            # pick a donor (large gradient) and a recipient (small gradient)
            fa = np.fft.fft(a, fft_len)
            conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            idx_max = int(np.argmax(conv_full))

            # compute the sub‑gradient at the current maximum
            g = np.zeros_like(a)
            ks = max(0, idx_max - n + 1)
            ke = min(n, idx_max + 1)
            rev = idx_max - np.arange(ks, ke)
            g[ks:ke] = 2.0 * a[rev]

            donor = int(np.argmax(g))
            # pick a recipient among indices with a low gradient value
            low_candidates = np.where(g <= np.percentile(g, 20))[0]
            if low_candidates.size == 0:
                recipient = rng.integers(0, n)
            else:
                recipient = int(rng.choice(low_candidates))
            if donor == recipient:
                recipient = (donor + rng.integers(1, n)) % n

            # move a small fraction of mass from donor to recipient
            max_move = min(a[donor], 0.05 * a[donor] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[donor] -= delta
            a_new[recipient] += delta

            # exact evaluation of the proposal
            new_max = max_convolution(a_new, fft_len)
            new_score = exact_score_from_max(n, new_max)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA {sa_iter:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_iter += 1

    # ------------------------------------------------------------------
    # 5) Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score:.6f}, length={n}")
    return best_a.tolist()
```
