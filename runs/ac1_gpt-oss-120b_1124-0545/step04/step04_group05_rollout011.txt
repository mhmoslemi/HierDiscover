def simplex_projection(v):
    # Project onto probability simplex
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative list of heights) that minimises
`evaluate_sequence`.  

The routine works in three phases:

1. **Parametric sweep** – a few analytically inspired families
   (inverse‑power law, Beta(α,α), smoothed arcsine and a mixture with the
   uniform distribution) are sampled.  Each candidate is normalised to sum
   to 1, its exact score `2·n·max(conv(a,a))` is computed with an FFT
   (O(n log n)) and the best candidate is kept.

2. **Exact sub‑gradient descent** – starting from the best sequence of
   phase 1 we minimise the true L∞‑norm of the convolution.
   For the current `a` we compute `M = max(conv(a,a))` and the set
   of active indices `I = {k | conv[k] = M}`.
   The sub‑gradient of `M` w.r.t. `a` is
       g_i = 2·∑_{k∈I} a_{k‑i}   (indices out of range are ignored)
   which can be obtained efficiently by one FFT.  After each step we
   project back onto the probability simplex.  A decreasing step size
   and simple restart logic keep the search active.

3. **Local simulated‑annealing** – a short stochastic refinement that
   moves a small fraction of mass between two random coordinates, accepting
   downhill moves or uphill moves with probability `exp(−Δ/T)`.  The temperature
   drops linearly.

All phases respect the 1000 s total budget.  The final sequence is
normalised, clipped to non‑negative values and returned as a plain list.
"""

import time
import numpy as np

# Helper utilities (no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(w, 1.0 / n)
    return w


def max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) using an FFT (linear convolution)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.fft(a, L)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(conv.max())


def evaluate_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        score = 2 * n * max(conv(a,a))
    """
    total = a.sum()
    if total == 0.0:
        return float("inf")
    a = a / total
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def arcsine_shape(n: int) -> np.ndarray:
    """Classic inverse‑sqrt (arcsine) shape, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


def power_shape(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """a_i = ((i+0.5)*(n-i-0.5) + eps)^{-p}, normalised."""
    i = np.arange(n, dtype=float)
    a = ((i + 0.5) * (n - i - 0.5) + eps) ** (-p)
    a /= a.sum()
    return a


def beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a[np.isnan(a)] = 0.0
    a[np.isinf(a)] = 0.0
    a += 1e-12          # avoid exact zeros (helps numerical stability)
    a /= a.sum()
    return a

def subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞‑norm of the convolution."""
    n = a0.shape[0]
    L = 1 << ((3 * n - 2).bit_length())   # length for a (n) * w (2n‑1)
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_score(a)
    step0 = 0.5
    it = 0
    # tolerance for active set
    atol = 1e-12

    while time.time() < start_time + time_limit:
        it += 1
        # convolution a*a
        fa = np.fft.fft(a, L)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        # active indices (allow tiny tolerance)
        active = np.where(conv >= M - atol)[0]

        # build weight vector w with ones on the active set
        w = np.zeros_like(conv)
        w[active] = 1.0

        # compute sub‑gradient g_i = 2 * sum_{k∈active} a_{k-i}
        # using one FFT with reversed w
        fw_rev = np.fft.fft(w[::-1], L)
        g_full = 2.0 * np.real(np.fft.ifft(fa * fw_rev))
        # g_i is entry 2*n-2-i of g_full
        grad = g_full[2 * n - 2 - np.arange(n)]

        # step‑size schedule (classic 1/√t)
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        score = evaluate_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[subgrad {it:07d}] improved to {best_score:.6f}")

        # simple accept‑if‑better policy (occasionally keep worse move)
        if score <= best_score or np.random.rand() < 0.01:
            a = a_new

        # occasional restart if no improvement for a while
        if it % 2000 == 0:
            # re‑centre around current best with tiny noise
            a = best_a + np.random.default_rng().normal(scale=1e-4, size=n)
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: stochastic local refinement (simulated annealing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Simple SA that moves a small fraction of mass between two coordinates."""
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_score(a)
    rng = np.random.default_rng()
    # temperature schedule
    T_start = best_score * 0.01
    T_end = best_score * 1e-5
    it = 0

    while time.time() < start_time + time_limit:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / time_limit)
        T = T_start * (1.0 - frac) + T_end * frac

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move up to 5 % of the mass at i
        max_move = a[i] * 0.05
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        score = evaluate_score(a_new)
        if score < best_score or rng.random() < np.exp(-(score - best_score) / (T + 1e-14)):
            a = a_new
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                print(f"[SA {it:07d}] improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The algorithm mixes a parametric sweep,
    exact sub‑gradient descent on the max‑convolution and a short simulated‑annealing
    refinement, all under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin before the external watchdog kills the process
    deadline = start_time + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to try a new candidate and keep the best one seen so far.
    # ------------------------------------------------------------------
    def consider(candidate: np.ndarray, tag: str) -> None:
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0)
        if candidate.sum() == 0.0:
            return
        candidate = candidate / candidate.sum()
        score = evaluate_score(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[{tag}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0) Start from any previously discovered sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        consider(np.asarray(globals()["height_sequence_1"], dtype=float), "global start")

    # classic arcsine shape
    consider(arcsine_shape(n), "arcsine")

    # ------------------------------------------------------------------
    # 1) Parametric families
    # ------------------------------------------------------------------
    # a) Inverse‑power‑law family   a_i = ((i+0.5)*(n-i-0.5)+eps)^{-p}
    for p in np.linspace(0.15, 0.80, 14):
        consider(power_shape(n, p, eps=0.0), f"power p={p:.3f}")
        if time.time() > deadline:
            break
    # b) Symmetric Beta(α,α) family
    for alpha in np.linspace(0.30, 2.00, 18):
        consider(beta_shape(n, alpha), f"beta α={alpha:.3f}")
        if time.time() > deadline:
            break
    # c) Smoothed arcsine (add a constant inside the sqrt)
    for eps in np.geomspace(1e-4, 1e2, 12):
        consider(power_shape(n, p=0.5, eps=eps), f"smooth eps={eps:.1e}")
        if time.time() > deadline:
            break
    # d) Convex mixture of arcsine and the uniform distribution
    uniform = np.full(n, 1.0 / n)
    arcsine = arcsine_shape(n)
    for lam in np.linspace(0.0, 1.0, 11):
        consider(lam * arcsine + (1.0 - lam) * uniform, f"mix λ={lam:.2f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Sub‑gradient descent on the exact max‑convolution objective
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        # give roughly 60 % of the remaining budget to this phase
        sub_time = remaining * 0.60
        seq_sgd, score_sgd = subgradient_descent(best_seq, sub_time, time.time())
        if score_sgd < best_score:
            best_score = score_sgd
            best_seq = seq_sgd.copy()
            print(f"[subgrad final] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Short simulated‑annealing refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        seq_sa, score_sa = simulated_annealing(best_seq, remaining, time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = seq_sa.copy()
            print(f"[SA final] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_score(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    best_seq = best_seq / total if total > 0 else np.full(n, 1.0 / n)

    elapsed = time.time() - start_time
    print(f"[done] time={elapsed:.1f}s, best score={best_score:.6f}, length={n}")
    return best_seq.tolist()
