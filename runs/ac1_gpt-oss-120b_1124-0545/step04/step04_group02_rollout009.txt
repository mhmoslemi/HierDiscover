def gradient_softmax(a, alpha):
    conv = conv_fft(a)
    # shift conv for numerical stability:
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute gradient = 2* corr(a, w)
    size = _next_pow2(3*a.shape[0] - 2)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(A * np.conj(W), n=size)
    g = 2 * corr[:a.shape[0]]
    return g
```

"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is split into three phases:

1️⃣  Analytic families (product, sum‑power, exponential, cosine‑power,
    beta‑shape and a simple edge‑plateau) are explored on a base length
    `n0` (default 2000).  The best vector (normalised to sum = 1) is kept.

2️⃣  A smooth‑max gradient descent on the exact convolution is performed.
    The true max is approximated by a soft‑max
        f(a)=log‑sum‑exp(α·(a*a))/α .
    The gradient is a (cross‑)correlation that can be obtained with a
    single FFT.  An exponentiated‑gradient (mirror) step is applied,
    followed by a projection onto the probability simplex.
    α and the learning rate are increased/decreased geometrically.
    Random jitter, occasional Dirichlet restarts and a stagnation‑reset
    keep the search from getting trapped.

3️⃣  The resulting distribution is up‑sampled to a larger size (`2·n0`),
    refined again with the same smooth‑max optimiser and finally polished
    with tiny Gaussian perturbations.

All steps respect the global wall‑time budget (default 1000 s).  The
returned list contains non‑negative floats; its sum is positive so the
evaluation function is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (zero‑padded) via FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_j a[j]·w[i+j] for i=0…n‑1.
    Implemented by a single FFT of size ≥ 3·n‑2.
    """
    n = a.shape[0]
    size = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)
    return corr_full[:n]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector on the simplex.
    Returns 2·n·max(conv(a,a)) because sum(a)=1.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families for the initial search
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, off: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= s
    return a


def analytic_sum(n: int, p: float, off: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= s
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a /= a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a /= a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a /= a.sum()
    return a


def analytic_plateau(n: int, k: int) -> np.ndarray:
    """k points on each side receive equal mass (edges only)."""
    a = np.zeros(n, dtype=np.float64)
    if k == 0:
        return a
    w = 1.0 / (2 * k)
    a[:k] = w
    a[-k:] = w
    return a

# Core optimisation loops
# ----------------------------------------------------------------------


def analytic_grid_search(n: int, init_seq: np.ndarray | None = None):
    """Exhaustively sweep a few analytic families, keep the best."""
    best_seq = None
    best_score = np.inf

    # product family
    for p in np.arange(0.2, 2.01, 0.2):
        for off in np.arange(0.01, 2.01, 0.2):
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # sum power‑law family
    for p in np.arange(0.2, 2.01, 0.2):
        for off in np.arange(0.01, 2.01, 0.2):
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # exponential tails
    for lam in np.logspace(-3, 1, 15):
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # cosine‑power
    for p in np.arange(0.5, 6.1, 0.5):
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # beta‑shaped
    for a in np.arange(-0.5, 2.01, 0.5):
        for b in np.arange(-0.5, 2.01, 0.5):
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # simple edge‑plateau
    step = max(1, n // 100)          # ~100 different plateau widths
    for k in range(1, n // 2 + 1, step):
        cand = analytic_plateau(n, k)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # optionally keep a supplied initial sequence
    if init_seq is not None:
        cand = np.asarray(init_seq, dtype=np.float64)
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0:
            cand = cand / cand.sum()
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    return best_seq, best_score


def smoothed_descent(
    a_init: np.ndarray,
    deadline: float,
    *,
    alpha_start: float = 0.1,
    alpha_end: float = 60.0,
    lr_start: float = 0.5,
    lr_end: float = 1e-6,
    temp_start: float = 0.1,
    temp_end: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max gradient descent on the simplex.
    Returns the best vector seen and its score.
    """
    n = a_init.shape[0]
    a = a_init.copy()
    best_seq = a.copy()
    best_score = compute_score(a)

    stage_start = time.time()
    last_improve = stage_start
    stagnation_limit = 30.0
    iteration = 0

    while time.time() < deadline:
        # --- schedule parameters -------------------------------------------------
        elapsed = time.time() - stage_start
        total = deadline - stage_start
        progress = min(1.0, elapsed / total)

        # geometric schedule for α and learning‑rate, linear for temperature
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # --- current convolution & soft‑max weights -------------------------------
        conv = conv_fft(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # soft‑max over convolution entries (numerically stable)
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()                     # length = 2·n‑1

        # --- gradient of the smoothed objective ---------------------------------
        grad = 2.0 * correlation_fft(a, w)                # shape (n,)

        # --- exponentiated‑gradient step (mirror descent) -----------------------
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-16)                 # avoid exact zeros
        a_new = a_new / a_new.sum()

        # --- occasional random mass‑transfer move --------------------------------
        if np.random.rand() < 0.02:
            a_new = random_move(a_new, scale=lr * 0.5)

        # --- compute new score ---------------------------------------------------
        conv_new = conv_fft(a_new)
        max_c_new = conv_new.max()
        cand_score = 2.0 * n * max_c_new

        # --- Metropolis‑type acceptance -----------------------------------------
        accept = False
        if cand_score <= cur_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new

        # --- rare Dirichlet restart ------------------------------------------------
        if np.random.rand() < 0.004:
            a_rand = np.random.gamma(1.0, 1.0, size=n)
            a_rand /= a_rand.sum()
            conv_r = conv_fft(a_rand)
            score_r = 2.0 * n * conv_r.max()
            if score_r < best_score:
                best_score = score_r
                best_seq = a_rand.copy()
                a = a_rand.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] random restart improved to {best_score:.6f}")

        # --- stagnation handling -------------------------------------------------
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        # --- record best ---------------------------------------------------------
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best {best_score:.6f}")

        iteration += 1

    return best_seq, best_score


def upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a probability vector from size n_small to n_big."""
    n_small = a_small.shape[0]
    xs = np.linspace(0.0, 1.0, n_small)
    xb = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(xb, xs, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    return a_big


def final_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """Tiny Gaussian jitter around the best vector until the time runs out."""
    best = a.copy()
    best_score = compute_score(best)
    n = a.shape[0]
    while time.time() < deadline:
        pert = np.random.normal(scale=5e-4, size=n)
        cand = best + pert
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")
    return best


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point used by the harness.
    Returns a list of non‑negative floats (the step‑function heights).
    """
    np.random.seed(seed)
    global_start = time.time()
    total_deadline = global_start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Analytic grid search on a moderate length
    # ------------------------------------------------------------------
    n0 = int(kwargs.get("n", 2000))
    print(f"[init] Analytic grid search on n = {n0}")
    best_seq, best_score = analytic_grid_search(n0)
    print(f"[init] best analytic score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent on the same length
    # ------------------------------------------------------------------
    stage_deadline = global_start + 0.55 * budget_s
    best_seq, best_score = smoothed_descent(best_seq, stage_deadline)
    print(f"[phase1] best after descent (n={n0}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Upsample to a larger length and repeat the descent
    # ------------------------------------------------------------------
    n_big = n0 * 2
    print(f"[upsample] enlarging to n = {n_big}")
    best_big = upsample(best_seq, n_big)

    # a quick analytic sweep on the larger grid (cheap)
    best_big, _ = analytic_grid_search(n_big, init_seq=best_big)

    # second descent on the larger vector
    stage_deadline2 = global_start + 0.95 * budget_s
    best_big, best_score = smoothed_descent(best_big, stage_deadline2)
    print(f"[phase2] best after large descent (n={n_big}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic refinement (few seconds left)
    # ------------------------------------------------------------------
    final_deadline = total_deadline
    best_big = final_refine(best_big, final_deadline)
    print(f"[done] final score = {compute_score(best_big):.6f}, length = {len(best_big)}")

    return best_big.tolist()
