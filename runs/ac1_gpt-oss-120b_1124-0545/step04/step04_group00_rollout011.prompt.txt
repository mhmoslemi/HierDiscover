def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns the full result of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _objective(seq: np.ndarray) -> float:

    Compute 2·n·max(conv(seq,seq)) / sum(seq)².
    The sequence is assumed to be non‑negative.

    conv = _conv_fft(seq, seq)

def _uniform(n: int) -> np.ndarray:

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()

def _sym_geom(n: int, r: float) -> np.ndarray:

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed maximum of the auto‑convolution.
    The sequence is represented by log‑softmax logits `v`; the loss is a
    log‑sum‑exp approximation of the true max.  Returns the best sequence
    discovered and its (unsmoothed) objective value.
    """
    v = np.log(seed + 1e-30)          # initialise logits

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule – start gentle and sharpen slowly
    beta = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(conv)                # safe shift for exponentials
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,

            best_val = cur_val

            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)                 # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()                           # re‑project onto simplex

    return best_seq, best_val

    cur_val = _objective(cur)
    best = cur.copy()

    decay = 0.9995                     # exponential cooling

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices

        while i == j:
            j = rng.integers(0, n)

        # propose moving a small fraction of the mass at i to j
