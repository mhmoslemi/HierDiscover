"""
Optimise a non‑negative step‑function (probability vector) for the
objective  E(a)=2·n·max(conv(a,a)) / (sum(a)**2).

The optimiser works in three stages:

  1.  A quick “window sweep” examines a collection of analytic windows
      (uniform, Hann, Hamming, Blackman, Bartlett and Kaiser windows
      with a range of β‑values, plus a few sinusoidal power shapes).
      The best among these simple candidates provides a good initial
      point.

  2.  A mirror‑descent / exponentiated‑gradient phase refines the
      candidate.  The (soft) max‑operator is approximated by a
      temperature‑controlled soft‑max over the convolution entries; the
      temperature is increased from a very smooth regime (β≈0.1) to an
      almost hard max (β≈8000).  In each iteration the sub‑gradient of
      the smoothed max is computed via FFTs, an exponentiated update
      is performed and the vector is projected back onto the probability
      simplex.

  3.  If the progress stalls for a few seconds, a small random
      multiplicative jitter is applied to the current best point
      (re‑initialisation) and the optimisation continues.

The routine respects the supplied time budget (default 1000 s) and
returns the best vector discovered as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the exact (scale‑invariant) score
    2·n·max(conv(a,a)) for a probability vector ``a``.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries."""
    z = beta * conv
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _hard_mask_weights(conv: np.ndarray) -> np.ndarray:
    """Hard mask (sub‑gradient of exact max)."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-14
    mask = (conv >= max_val - eps).astype(float)
    mask_sum = mask.sum()
    if mask_sum == 0.0:
        mask[np.argmax(conv)] = 1.0
        mask_sum = 1.0
    return mask / mask_sum


def _gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of 2·n·max(conv) w.r.t. ``a`` using the weight vector ``w``.
    Both ``a`` (length n) and ``w`` (length 2·n‑1) are real.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)

    fw = np.fft.rfft(w, fft_len)

    # (w * a) – linear convolution, first 3·n‑2 entries
    conv_wa = np.fft.irfft(fw * fa, fft_len)[: 3 * n - 2]
    g2 = conv_wa[:n]

    # w ⋆ a – cross‑correlation (equivalent to convolution with reversed w)
    corr = np.fft.irfft(np.conj(fw) * fa, fft_len)[: 3 * n - 2]
    g1 = corr[:n]

    # gradient of 2·n·max(conv)
    return 2.0 * n * (g1 + g2)


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a modest library of analytic windows and a few random Dirichlet draws.
    All candidates are normalised to sum to 1.
    """
    cand = []

    # ----- simple analytic windows ------------------------------------
    # uniform
    cand.append(np.full(n, 1.0 / n))

    # Hann, Hamming, Blackman, Bartlett (numpy defines them for n>=2)
    for fn in (np.hanning, np.hamming, np.blackman, np.bartlett):
        w = fn(n)
        if w.sum() > 0:
            cand.append(w / w.sum())

    # Kaiser windows with a sweep of β
    for beta in np.linspace(0.0, 20.0, 41):
        w = np.kaiser(n, beta)
        if w.sum() > 0:
            cand.append(w / w.sum())

    # Sine‑power shapes  (sin(π (i+0.5)/n))**p
    i = np.arange(n, dtype=float) + 0.5
    base = np.sin(np.pi * i / n)
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        w = base ** p
        if w.sum() > 0:
            cand.append(w / w.sum())

    # Random Dirichlet draws – provide additional diversity
    alpha = np.full(n, 0.5)
    for _ in range(4):
        a = rng.dirichlet(alpha)
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time.

    Parameters
    ----------
    seed : int
        Base random seed – different runs use different seeds.
    budget_s : int
        Total time budget in seconds.
    kwargs : dict
        Optional ``n`` (length of the sequence).  Defaults to 2000.

    Returns
    -------
    list[float]
        The best coefficient list discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)  # tiny safety margin

    n = int(kwargs.get("n", 2000))
    fft_len = _next_pow2(3 * n - 2)          # covers all needed convolutions

    # ------------------------------------------------------------------
    # 1) Candidate sweep – cheap analytic windows
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)

    # if a historic solution exists, include it (may have different length)
    if "height_sequence_1" in globals():
        hist = np.array(globals()["height_sequence_1"], dtype=float)
        if hist.size != n:

            else:

        hist = np.maximum(hist, 0.0)
        hist_sum = hist.sum()
        if hist_sum > 0:
            hist = hist / hist_sum
        candidates.append(hist)

    best_a = None
    best_score = float("inf")

    for a in candidates:
        if time.time() > deadline:
            break
        sc = _score_from_a(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.time() - start
            print(
                f"[window sweep] new best {best_score:.6f} (t={elapsed:.1f}s, norm={np.linalg.norm(a):.3f})"
            )

    # ------------------------------------------------------------------
    # 2) Mirror‑descent refinement (exponentiated gradient)
    # ------------------------------------------------------------------
    if best_a is None:
        # safety net – uniform vector
        best_a = np.full(n, 1.0 / n)

    a = best_a.copy()
    last_improve = time.time()
    it = 0

    # schedule parameters (tuned a little compared to the previous baseline)
    beta_start, beta_end = 0.05, 8000.0          # soft‑max temperature schedule
    lr_start, lr_end = 0.30, 0.001              # learning‑rate schedule
    mask_phase_start = 0.60                    # when to switch to hard mask

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        p = min(1.0, elapsed / total)            # normalised progress

        # adaptive temperature and learning rate
        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        # --------------------------------------------------------------
        # convolution a * a
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # --------------------------------------------------------------
        # choose weighting for the max (soft or hard)
        # --------------------------------------------------------------
        if p < mask_phase_start:
            w = _softmax_weights(conv, beta)
        else:
            w = _hard_mask_weights(conv)

        # --------------------------------------------------------------
        # gradient of the (smoothed) max‑objective
        # --------------------------------------------------------------
        grad = _gradient(a, w, fft_len)

        # --------------------------------------------------------------
        # exponentiated‑gradient update (mirror descent on simplex)
        # --------------------------------------------------------------
        exp_term = -lr * grad
        # shift for numerical stability
        exp_term -= np.max(exp_term)
        a = a * np.exp(exp_term)
        sum_a = a.sum()
        if sum_a == 0.0 or np.isnan(sum_a):
            # fallback to jitter around the current best
            jitter = rng.normal(scale=0.005, size=n)
            a = best_a * np.exp(jitter)
            sum_a = a.sum()
        a = a / sum_a

        it += 1

        # --------------------------------------------------------------
        # evaluate current score (exact, no smoothing)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 500 == 0:
                print(
                    f"[iter {it:07d}] new best {best_score:.6f} (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # --------------------------------------------------------------
        # restart if stagnation (>30 s without improvement)
        # --------------------------------------------------------------
        if time.time() - last_improve > 30.0:

            last_improve = time.time()

    # ------------------------------------------------------------------
    if best_a is None:

    best_a = np.maximum(best_a, 0.0)

    total_time = time.time() - start
