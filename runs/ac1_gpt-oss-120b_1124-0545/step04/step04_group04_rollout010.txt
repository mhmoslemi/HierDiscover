def project_onto_simplex(v):
    # returns projection onto simplex {x >= 0, sum(x) = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _optimize(initial_a, n, alloc_s, rng):
    a = initial_a.copy()
    best_a = a.copy()
    best_score = compute_score(a, n)
    start = time.time()
    # hyperparameters
    lr_start = 0.4
    lr_end = 0.001
    beta_start = 0.1
    beta_end = 2000.0
    mask_phase = 0.6
    # momentum
    m = np.zeros_like(a)
    beta_mom = 0.9
    # FFT length
    fft_len = 1 << ((3 * n - 2).bit_length())
    last_imp = start
    while time.time() - start < alloc_s - 0.02:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))
        lr = lr_start * (1-p) + lr_end * p
        beta = beta_start * (1-p) + beta_end * p

        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

        max_val = conv.max()
        if p < mask_phase:
            z = beta * conv
            # stable softmax
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w /= w.sum()
        else:
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask[np.argmax(conv)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # gradient
        fw = np.fft.fft(w, fft_len)
        conv_wa = np.real(np.fft.ifft(fw * fa))[:3*n-2]
        g2 = conv_wa[:n]
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[:3*n-2]
        g1 = corr_full[:n]
        g = 2.0 * n * (g1 + g2)

        # momentum update
        m = beta_mom * m + (1 - beta_mom) * g

        # step
        a_new = a - lr * m

        # project onto simplex
        a = _project_onto_simplex(a_new)

        # evaluate score
        conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)))[:2*n-1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_imp = time.time()

        # stagnation restart

"""
Optimization of a non‑negative step‑function for the objective
E(a)=2·n·max(conv(a,a))/(sum a)^2.

Algorithm:
- Initialise several diverse probability vectors (uniform, sinusoidal,
  Dirichlet draws; optionally re‑use the historic `height_sequence_1`).
- For each start run a projected‑gradient descent loop.
  The gradient of the (soft‑max or hard‑max) objective is obtained
  efficiently by FFTs.  Momentum (Adam‑style) and a linearly decaying
  learning‑rate are employed.  After every gradient step the iterate is
  projected onto the probability simplex.
- If no improvement is observed for a while the iterate is jittered
  around the current best point to escape stagnation.
- After the allocated time a short “pair‑exchange” refinement is performed
  that moves a small amount of mass between two coordinates and keeps
  any accepted improvement.
- The best vector found over all attempts is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x≥0, sum(x)=1}."""
    # Implementation of the algorithm from
    # Duchi et al., “Efficient Projections onto the l1‑Ball for
    # Learning in High Dimensions”, ICML 2008.
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT, returns length 2·n‑1 array."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return conv


def _gradient_max_conv(a: np.ndarray, fft_len: int,
                       beta: float, use_hard_mask: bool) -> np.ndarray:
    """Gradient of 2·n·max(conv) w.r.t. a.
    If ``use_hard_mask`` is False a soft‑max with temperature ``beta`` is used."""
    n = a.shape[0]
    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    if use_hard_mask:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(float)
        mask_sum = mask.sum()
        if mask_sum == 0:
            mask = np.zeros_like(conv)
            mask[np.argmax(conv)] = 1.0
            mask_sum = 1.0
        w = mask / mask_sum
    else:
        # soft‑max over the convolution entries
        z = beta * conv
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

    # gradient of weighted sum of convolutions
    fw = np.fft.fft(w, fft_len)

    # (w * a) – linear convolution, keep first 3·n‑2 terms,
    # then take the first n entries (= the part that depends on a)
    conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
    g2 = conv_wa[:n]

    # w ⋆ a – linear cross‑correlation (a convolved with reversed w)
    corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
    g1 = corr_full[:n]

    # final gradient of the objective 2·n·max(conv)
    g = 2.0 * n * (g1 + g2)
    return g, conv


def _optimize_start(a_init: np.ndarray,
                    n: int,
                    alloc_s: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Optimise a single start vector ``a_init`` for at most ``alloc_s`` seconds.
    Returns the best vector found and its exact (scale‑invariant) score."""
    # ensure feasibility
    a = _project_onto_simplex(a_init.astype(float))

    # Adam‑style momentum buffers
    m = np.zeros_like(a)

    # hyper‑parameters (tuned heuristically)
    beta1 = 0.9          # momentum decay
    beta2 = 0.999        # not used – kept for compatibility
    eps = 1e-8

    # schedules (linear in normalised time p∈[0,1])
    lr_start, lr_end = 0.4, 0.001          # learning‑rate
    beta_start, beta_end = 0.1, 2000.0    # temperature for soft‑max
    mask_phase_start = 0.65               # switch to hard mask

    # FFT length – must cover convolution of length 3·n‑2
    fft_len = _next_pow2(3 * n - 2)

    start = time.time()
    deadline = start + alloc_s - 0.05  # tiny safety margin

    # bookkeeping
    best_a = a.copy()
    conv = _compute_conv(best_a, fft_len)
    best_score = 2.0 * n * conv.max()
    last_improve = start
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))

        # learning‑rate and temperature schedules
        lr = lr_start * (1.0 - p) + lr_end * p
        beta = beta_start * (1.0 - p) + beta_end * p
        use_hard = p >= mask_phase_start

        # gradient of the (smoothed) max‑convolution
        g, conv = _gradient_max_conv(a, fft_len, beta, use_hard)

        # Adam‑style update (we only keep the first moment)
        m = beta1 * m + (1.0 - beta1) * g
        a_tmp = a - lr * m

        # projection onto simplex (guarantees ∑a = 1, a ≥ 0)
        a = _project_onto_simplex(a_tmp)

        # current exact score (hard max, no smoothing)
        conv_hard = _compute_conv(a, fft_len)
        cur_score = 2.0 * n * conv_hard.max()

        # bookkeeping of the best point
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 5000 == 0:
                print(
                    f"[{it:07d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, lr={lr:.4f})"
                )

        # restart if we have been stagnant for a while
        if time.time() - last_improve > 18.0:
            # jitter around the current best point
            jitter = rng.normal(scale=0.008, size=n)
            a = _project_onto_simplex(best_a + jitter)
            m.fill(0.0)
            last_improve = time.time()
            print(f"[{it:07d}] restart after {elapsed:.1f}s (stagnation)")

    return best_a, best_score


def _local_refine(a: np.ndarray,
                  n: int,
                  time_end: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple pair‑exchange hill climbing using the remaining time."""
    fft_len = _next_pow2(3 * n - 2)

    # current best
    best_a = a.copy()
    conv = _compute_conv(best_a, fft_len)
    best_score = 2.0 * n * conv.max()

    while time.time() < time_end - 0.02:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of the mass from i to j
        delta = rng.random() * min(best_a[i], 0.02)  # at most 2 % of a_i
        if delta == 0.0:
            continue
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # projection is cheap – the operation above keeps the sum,
        # only a_i may become slightly negative, correct it.
        a_new = _project_onto_simplex(a_new)

        conv = _compute_conv(a_new, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a_new
            # occasional progress report
            if rng.integers(0, 5000) == 0:
                print(f"[refine] improved to {best_score:.6f}")

    return best_a, best_score

    The routine respects the time budget (default 1000 s) and returns
    the best sequence found as a plain ``list[float]``.
    """

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    attempts = 8
    best_global_a = None
    best_global_score = float("inf")
    rng_global = np.random.default_rng(seed)

    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 12.0:
            break
        # allocate a fair share of the remaining time to this attempt
        remaining = attempts - attempt
        alloc = time_left * 0.90 / remaining

        # ----- initialise -------------------------------------------------
        if attempt == 0 and "height_sequence_1" in globals():
            a0 = np.array(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            a0 = np.maximum(a0, 0.0)
        elif attempt == 1:

            idx = np.arange(n, dtype=float)
            a0 = np.sin(np.pi * idx / (n - 1))
        elif attempt == 2:

            idx = np.arange(n, dtype=float)
            a0 = (1 - np.cos(2 * np.pi * idx / (n - 1))) / 2.0
        else:
            # random Dirichlet draw, concentration 0.5 (diverse starts)
            alpha = np.full(n, 0.5)
            a0 = rng_global.dirichlet(alpha)

        # ----- run the optimiser for this start -------------------------
        a_opt, score_opt = _optimize_start(
            a_init=a0,
            n=n,
            alloc_s=alloc,
            rng=np.random.default_rng(seed + attempt + 10007),
        )

        if score_opt < best_global_score:
            best_global_score = score_opt
            best_global_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {attempt}] new global best = {best_global_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 2) Final local refinement using the remaining budget
    # ------------------------------------------------------------------
    if best_global_a is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:

                best_global_a,
                n,
                deadline,

    # safety clamps (already non‑negative and sum ≈ 1)
    best_global_a = np.maximum(best_global_a, 0.0)

    total_time = time.time() - start_time
    print(f"[done] best score≈{best_global_score:.6f}, time={total_time:.1f}s")

    best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Multiple independent starts (including historic vector if present)
    # ------------------------------------------------------------------
    attempts = 8
    best_global_a = None
    best_global_score = float("inf")
    rng_global = np.random.default_rng(seed)

    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 12.0:
            break
        remaining = attempts - attempt
        alloc = time_left * 0.90 / remaining

        # ----- initialise a start vector ---------------------------------
        if attempt == 0 and "height_sequence_1" in globals():
            a0 = np.array(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            a0 = np.maximum(a0, 0.0)
        elif attempt == 1:
            # smooth half‑wave sine (positive everywhere)
            idx = np.arange(n, dtype=float)
            a0 = np.sin(np.pi * idx / (n - 1))
        elif attempt == 2:
            # cosine‑squared shape (another smooth, positive window)
            idx = np.arange(n, dtype=float)
            a0 = (1 - np.cos(2 * np.pi * idx / (n - 1))) / 2.0
        else:
            # random Dirichlet draw (concentration 0.5 → diverse supports)
            alpha = np.full(n, 0.5)
            a0 = rng_global.dirichlet(alpha)

        # ----- optimise this start ----------------------------------------
        a_opt, score_opt = _optimize_start(
            a_init=a0,
            n=n,
            alloc_s=alloc,
            rng=np.random.default_rng(seed + attempt + 10007),
        )

        if score_opt < best_global_score:
            best_global_score = score_opt
            best_global_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {attempt}] new global best = {best_global_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 2) Final refinement with remaining budget (pair‑exchange hill‑climb)
    # ------------------------------------------------------------------
    if best_global_a is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            best_global_a, best_global_score = _local_refine(
                best_global_a,
                n,
                deadline,
                rng=np.random.default_rng(seed + 99999),
            )

    # ------------------------------------------------------------------
    # 3) Fallback to a uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_global_a is None:
        best_global_a = np.full(n, 1.0 / n)

    # Ensure non‑negativity (already guaranteed) and return as plain list
    best_global_a = np.maximum(best_global_a, 0.0)

    total_time = time.time() - start_time
    print(f"[done] best score≈{best_global_score:.6f}, time={total_time:.1f}s")
    return best_global_a.tolist()
