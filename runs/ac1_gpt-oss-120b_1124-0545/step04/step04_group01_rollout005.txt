"""
Hybrid global+local optimisation for the “max‑auto‑convolution” problem.

* Global search – a small‑dimensional Differential Evolution (DE) works on
  a piece‑wise‐linear parametrisation of the step function.  The parameter
  vector (size ≈ √n) lives on the simplex; each candidate is up‑sampled by
  linear interpolation to length *n* and evaluated by the exact objective
  2·n·max(conv(a,a)) (the sum of the coefficients is forced to 1, which
  makes the evaluation scale‑invariant).

* Local polishing – the best DE candidate is fed to three deterministic
  refinements that were already used in the reference code:
    – Adam optimisation on a smooth approximation of the max,
    – mirror‑descent with the exact sub‑gradient,
    – a cheap stochastic hill‑climb that transfers a small amount of mass
      between random positions.

All phases respect a global time‑budget (default 1000 s) and keep the
best sequence found so far.  The function ``propose_candidate`` returns the
final sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse normalized start vectors."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max (same as reference code)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑softmax parametrisation, using a smooth‑max surrogate."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smoothing coefficient
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Random pair‑wise mass transfer, keeping improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑dimensional Differential Evolution (global search)
# ----------------------------------------------------------------------


def _eval_param(p: np.ndarray, knots: np.ndarray, n: int) -> tuple[float, np.ndarray]:
    """
    Interpolate a low‑dimensional simplex vector `p` (sum=1)
    to a full length‑`n` sequence, normalise, and evaluate the exact objective.
    Returns (objective, full_sequence).
    """
    # linear interpolation on the integer grid
    a = np.interp(np.arange(n, dtype=np.float64), knots, p)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
        s = a.sum()
    a = a / s
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)      # sum(a)=1 ⇒ objective = 2·n·max(conv)
    return val, a


def _de_optimize(n: int,
                 deadline: float,
                 popsize: int = 30,
                 K: int | None = None,
                 max_gens: int = 200) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on a piece‑wise‑linear representation.
    The population size, number of generations, and knot count `K`
    are chosen to respect the time budget.
    """
    rng = np.random.default_rng()

    if K is None:
        K = max(20, int(np.sqrt(n)))      # ≈ √n knots
    knots = np.linspace(0, n - 1, K)

    # initialise population with Dirichlet draws (already on simplex)
    pop = [rng.dirichlet(np.ones(K)) for _ in range(popsize)]

    pop_vals = np.empty(popsize, dtype=np.float64)
    pop_seq = [None] * popsize
    for i in range(popsize):
        val, seq = _eval_param(pop[i], knots, n)
        pop_vals[i] = val
        pop_seq[i] = seq

    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_seq = pop_seq[best_idx].copy()
    best_p = pop[best_idx].copy()

    F = 0.8
    CR = 0.9
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)
            ds = donor.sum()
            if ds == 0.0:
                donor = _uniform(K)
            else:
                donor = donor / ds

            # binomial crossover
            trial = pop[i].copy()
            jrand = rng.integers(K)
            for j in range(K):
                if rng.random() < CR or j == jrand:
                    trial[j] = donor[j]

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = _uniform(K)
            else:
                trial = trial / ts

            val_trial, seq_trial = _eval_param(trial, knots, n)

            if val_trial < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val_trial
                pop_seq[i] = seq_trial
                if val_trial < best_val:
                    best_val = val_trial
                    best_seq = seq_trial.copy()
                    best_p = trial.copy()

        if gen % 20 == 0:
            print(f"[DE] n={n:5d} gen={gen:3d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines a global DE stage with
    deterministic refinements and respects the given time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # previous best (if any) – stored in the global namespace under this name
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to try – larger n gives a larger design space but costs more time
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1. Quick scan of a diverse set of analytic seeds
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 2. Global search via Differential Evolution
        # ------------------------------------------------------------------
        seq_de, val_de = _de_optimize(n, deadline, popsize=30,
                                      K=None, max_gens=250)
        if val_de < global_best_val:
            global_best_val = val_de
            global_best_seq = seq_de.copy()
            print(f"[DE] n={n:5d} new global best={val_de:.6f}")

        # ------------------------------------------------------------------
        # 3. Adam refinement (more iterations than the reference code)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_adam, val_adam = _adam_optimize(global_best_seq, n,
                                                deadline, max_iters=12000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val_adam:.6f}")

        # ------------------------------------------------------------------
        # 4. Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sg, val_sg = _subgradient_refine(global_best_seq, n,
                                                 deadline, max_iters=60000)
            if seq_sg is not None and val_sg < global_best_val:
                global_best_val = val_sg
                global_best_seq = seq_sg.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val_sg:.6f}")

        # ------------------------------------------------------------------
        # 5. Stochastic hill‑climbing (lightweight)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n,
                                         deadline, max_iters=40000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new global best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be required)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = 2.0 * n * np.max(_conv_fft(global_best_seq,
                                                    global_best_seq))
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Optional verification with the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
