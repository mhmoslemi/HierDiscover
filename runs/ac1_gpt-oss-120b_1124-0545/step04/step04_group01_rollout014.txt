```
def _lp_refine(start_seq, n, deadline, max_iters=10, t=0.5):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # Build LP
        A = _conv_matrix(a) # shape (2n-1,n)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)]*n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("[LP] iteration {} failed: {}".format(it, e))
            break
        if not res.success:
            print("[LP] iteration {} linear program failed: {}".format(it, res.message))
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b * (1.0 / sum_b)  # rescale to sum 1
        # mix
        a = (1-t) * a + t * g
        # optionally adjust t schedule
        # evaluate new objective
        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n} it={it} best={cur_val:.6f}")
        # decrease t
        t = max(0.1, t * 0.9)
    return best_seq, best_val
```

def _lp_refine(start_seq, n, deadline, max_iters=10, t_init=0.5):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    t = t_init
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = np.max(conv)
        # build matrix A
        A = diags([np.full(n, a[d]) for d in range(n)], offsets=np.arange(n), shape=(2*n-1, n), format='csr')
        # linear program
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # To avoid too much time, maybe limit solver via max_iter, but not needed
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # rescale to sum 1
        g = b / sum_b
        # mix with exponent t
        a = (1 - t) * a + t * g
        # evaluate
        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n:5d} it={it:5d} best={cur_val:.6f}")
        # reduce t (optional)
        t = max(0.01, t * 0.85)
    return best_seq, best_val

The objective to minimise is  

\[
\text{score}(a)=\frac{2n\;\max\bigl(\text{conv}(a,a)\bigr)}{\bigl(\sum a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous we may normalise the vector to `∑a_i = 1`.  
After normalisation the score is simply `2 n · max(conv(a,a))`.  
The program below searches for a vector with a very small maximal auto‑convolution.

The search is divided into four phases

1. **Seed generation** – a large collection of analytic shapes (uniform,
   geometric, power‑law, beta‑like, sine‑window, linear, quadratic,
   exponential, random Dirichlet etc.) together with the previously found
   construction, all normalised to a unit ℓ₁‑norm.

2. **Adam optimisation on a smooth approximation of `max`** – the vector is
   parametrised by a log‑softmax `v`; Adam updates are performed on the
   smoothed objective (`log‑sum‑exp`).  The smoothing coefficient is
   increased gradually.

3. **Exact sub‑gradient refinement** – from the best Adam result a mirror‑
   descent (exponentiated‑gradient) step using the true sub‑gradient of the
   max‑convolution is applied many times.

4. **Linear‑program (LP) refinement** – for the current vector `f` we solve  

   \[
   \max\_{b\ge0}\;\mathbf{1}^\top b
   \qquad\text{s.t.}\qquad
   f * b \le M\mathbf{1},
   \]

   where `M = max(conv(f,f))`.  The LP is solved with HiGHS (via
   `scipy.optimize.linprog`).  The optimal `b` is rescaled to have the same
   ℓ₁‑norm as `f` and mixed with `f` (`f←(1‑t)f + t b`).  Repeating this step
   drives the maximal convolution down while keeping the ℓ₁‑norm fixed.

5. **Stochastic hill‑climbing** – a cheap random pair‑wise mass‑transfer
   neighbourhood search finishes the optimisation.

All phases respect the supplied time budget (default 1000 s).  The best
vector found is returned as a plain list of floats.

```python
"""
search_step_function.py

Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑phase strategy:
    1. diverse analytic seeds,
    2. Adam optimisation on a smoothed max,
    3. exact sub‑gradient (exponentiated‑gradient) refinement,
    4. linear‑program refinement (maximise ℓ₁‑norm under the same max‑conv bound),
    5. cheap stochastic hill‑climbing.

The routine respects a total time budget (default 1000 s) and returns the
best sequence found as a list of floats.
"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _linear_decreasing(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _linear_increasing(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _quadratic_decreasing(n: int, p: float = 2.0) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=np.float64)) ** p
    return a / a.sum()


def _quadratic_increasing(n: int, p: float = 2.0) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64)) ** p
    return a / a.sum()


def _exp_decay(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of ℓ₁‑normalised seed vectors."""
    rng = np.random.default_rng(2023)
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # linear / quadratic families
    seeds.append(_linear_decreasing(n))
    seeds.append(_linear_increasing(n))
    seeds.append(_quadratic_decreasing(n, p=2.0))
    seeds.append(_quadratic_increasing(n, p=2.0))

    # exponential decay
    for r in (0.990, 0.995, 0.997, 0.999, 0.9995):
        seeds.append(_exp_decay(n, r))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on the smooth approximation of the max‑convolution."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # starting smoothing coefficient
    beta_mul = 1.02       # growth per iteration

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)                     # safe shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) refinement using the exact sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # keep numbers moderate
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing using tiny pair‑wise mass transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.01  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val


def _conv_matrix(f: np.ndarray) -> diags:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
    A @ b == np.convolve(f, b) for any non‑negative vector b.
    """
    n = len(f)
    # each diagonal d (0 ≤ d < n) contains the constant value f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    return diags(data, offsets, shape=(2 * n - 1, n), format="csr")


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 12,
               t_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Linear‑program refinement:
    maximise ℓ₁‑norm of b under the bound f∗b ≤ M,
    then mix b back into f.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    t = t_init

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = np.max(conv)

        # Build constraint matrix A (conv(f, b) ≤ M)
        A = _conv_matrix(a)

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c,
                      A_ub=A, b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Rescale to maintain unit ℓ₁‑norm
        g = b / sum_b

        # Convex combination
        a = (1.0 - t) * a + t * g

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # slowly shrink the mixing factor
        t = max(0.01, t * 0.85)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of non‑negative heights that minimises the
    provided ``evaluate_sequence`` objective.  Returns the best sequence
    found as a plain list of floats.  The routine respects the given time
    budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # reuse a previous construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # we concentrate on a single length that gave the best known result
    lengths = [4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max
        # ------------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=6000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=30000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Linear‑program refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_lp, val_lp = _lp_refine(global_best_seq, n, deadline,
                                        max_iters=12, t_init=0.5)
            if seq_lp is not None and val_lp < global_best_val:
                global_best_val = val_lp
                global_best_seq = seq_lp.copy()
                print(f"[overall‑LP] n={n:5d} new best={val_lp:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Stochastic hill‑climbing (lightweight)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                         max_iters=20000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new best={val_hc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Optional sanity check with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
```
