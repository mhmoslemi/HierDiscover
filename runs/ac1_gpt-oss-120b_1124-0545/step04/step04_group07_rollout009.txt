"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. **Fractal‑measure families** – For a set of removal fractions ``r`` we
   generate a histogram of the corresponding Cantor‑type measure
   (two‑branch equal weight cascade).  Each histogram is a feasible
   non‑negative vector that sums to 1.  The objective value
   ``2·n·max(conv(a,a))`` is computed cheaply with a single FFT; the best
   few candidates are kept.

2. **Hard‑max projected gradient descent** – Starting from each kept
   candidate we minimise the peak of the self‑convolution.
   Instead of the smooth soft‑max used previously we use a *hard* max:
   a weight vector ``w`` is the uniform distribution over all indices
   where the current convolution attains (or almost attains) its maximum.
   The sub‑gradient of the peak is then ``‑2·(a_rev ∗ w)``.
   A Nesterov‑style momentum term, a decreasing step size and occasional
   small random perturbations keep the iterates away from shallow
   minima.

3. **Time‑aware multi‑restart loop** – The whole search runs until the
   given time budget (default 1000 s) is exhausted, trying several
   values of ``r`` and re‑initialising from the current best solution
   whenever progress stalls.

The final best sequence (as a plain ``list[float]``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (real inputs).
    ``size`` must be ≥ len(u)+len(v)‑1.  Returns the full convolution.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Return the full linear convolution a*a."""
    full = _conv_fft(a, a, size1)
    return full[: 2 * a.shape[0] - 1]  # cut to exact length


def _fractal_histogram(
    n: int, L: int = 14, r: float = 1.0 / 3.0, seed: int | None = None
) -> np.ndarray:
    """
    Histogram approximation of a two‑branch Cantor‑type measure.

    Parameters
    ----------
    n : int
        Desired histogram length.
    L : int
        Number of cascade levels (2**L atoms).
    r : float
        Fraction removed in each iteration (0 < r < 1).
    seed : int | None
        For reproducible shuffling of atoms (optional).

    Returns
    -------
    a : np.ndarray, shape (n,)
        Non‑negative entries that sum to 1.
    """
    # scaling factor for the retained intervals
    s = (1.0 - r) / 2.0                      # length of each kept sub‑interval
    base = (1.0 + r) / 2.0                   # offset contributed by a right‑branch

    # offsets for each binary digit (level)
    offsets = base * (s ** np.arange(L))      # shape (L,)

    N = 1 << L                               # number of atoms = 2**L
    # binary matrix (N × L) – each row is the binary expansion of the atom index
    idx = np.arange(N, dtype=np.uint32)[:, None]          # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)

    # atom positions in [0,1)
    pos = bits @ offsets                     # (N,)

    # optional shuffling to avoid any systematic binning bias
    if seed is not None:
        rng = np.random.RandomState(seed)
        rng.shuffle(pos)

    # histogram on a regular grid with n bins
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)   # normalise → sum == 1
    return a


def _hard_max_weights(conv_full: np.ndarray, rel_tol: float = 1e-12) -> np.ndarray:
    """
    Build the weight vector for the sub‑gradient of ``max(conv)``.
    All indices whose value is within ``rel_tol`` of the maximum receive
    equal weight; the vector is then normalised to sum = 1.
    """
    max_b = conv_full.max()
    if max_b == 0.0:
        # degenerate – fall back to a uniform weight
        w = np.ones_like(conv_full) / conv_full.size
        return w
    mask = conv_full >= max_b * (1.0 - rel_tol)
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        # numeric safety: fall back to a single max index
        w = np.zeros_like(conv_full)
        w[conv_full.argmax()] = 1.0
    else:
        w = np.zeros_like(conv_full)
        w[mask] = 1.0 / float(cnt)
    return w


def _gradient_step(
    a: np.ndarray,
    a_rev: np.ndarray,
    size2: int,
    w: np.ndarray,
) -> np.ndarray:
    """
    Compute the (negative) sub‑gradient direction for the hard‑max.
    Returns a vector of shape ``a.shape``.
    """
    # convolution a_rev * w (full length 3n‑2)
    full = _conv_fft(a_rev, w, size2)
    n = a.shape[0]
    # central n entries correspond to the gradient components
    g = full[n - 1 : n - 1 + n]
    d = -2.0 * g
    d -= d.mean()                     # keep Σd = 0   (preserves total mass)
    return d


def _project_and_normalise(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Clip to non‑negative, then renormalise to sum = 1 (if possible)."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < eps:
        # avoid division by zero – fall back to uniform distribution
        a = np.ones_like(a) / a.shape[0]
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Core optimiser – hard‑max gradient descent with occasional noise
# ----------------------------------------------------------------------


def _run_hardmax_descent(
    a0: np.ndarray,
    n: int,
    size1: int,
    size2: int,
    deadline: float,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the hard‑max objective.

    Returns
    -------
    best_a : np.ndarray
        Sequence that achieved the lowest objective value.
    best_score : float
        Corresponding value of ``2·n·max(conv)`` (i.e. the objective for
        a normalised sequence).
    """
    a = a0.copy()
    a_rev = a[::-1].copy()
    v = np.zeros_like(a)                # momentum buffer
    momentum = 0.9
    base_step = 0.8

    # bookkeeping
    conv_full = _full_convolution(a, size1)
    best_score = 2.0 * n * conv_full.max()
    best_a = a.copy()
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        # current convolution and max
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[gd] it={it:7d}  new best={best_score:.6f}")

        # construct hard‑max weights
        w = _hard_max_weights(conv_full, rel_tol=1e-12)

        # gradient step
        d = _gradient_step(a, a_rev, size2, w)

        # Nesterov‑type momentum
        v = momentum * v + (1.0 - momentum) * d

        # step size schedule
        step = base_step / np.sqrt(it + 1.0)

        a = a + step * v
        a = _project_and_normalise(a)

        # update reverse view for next iteration
        a_rev = a[::-1].copy()

        # occasional tiny random jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.normal(scale=0.001, size=n)
            a = _project_and_normalise(a + jitter)
            a_rev = a[::-1].copy()

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function will return before the
        budget expires; otherwise the harness records a failure.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (plain Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # pre‑compute FFT sizes (constant for the whole run)
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)               # for a*a
    size2 = _next_pow2(3 * n - 2)               # for a_rev * w

    # --------------------------------------------------------------
    # 1️⃣  Scan a family of Cantor‑type histograms (different removal r)
    # --------------------------------------------------------------
    r_candidates = np.linspace(0.20, 0.60, 9)   # 0.20,0.25,…,0.60
    L = 14                                      # depth of the cascade
    prelim = []

    print("[scan] evaluating fractal histograms")
    for r in r_candidates:
        a = _fractal_histogram(n, L=L, r=r, seed=seed)
        conv = _full_convolution(a, size1)
        score = 2.0 * n * conv.max()
        prelim.append((score, a, r))
        print(f"   r={r: .3f}  score={score:.6f}")

    # sort by score, keep the top few
    prelim.sort(key=lambda x: x[0])
    top_k = 3
    top_candidates = prelim[:top_k]

    # --------------------------------------------------------------
    # 2️⃣  Run hard‑max projected gradient descent from each seed
    # --------------------------------------------------------------
    best_overall_a = None
    best_overall_score = np.inf

    # allocate remaining time proportionally among the runs
    remaining_time = deadline - time.time()
    time_per_run = remaining_time / max(len(top_candidates), 1)

    for idx, (init_score, a_start, r_val) in enumerate(top_candidates):
        run_deadline = time.time() + time_per_run
        print(f"[run {idx+1}/{len(top_candidates)}] r={r_val:.3f} start_score={init_score:.6f}")

        a_opt, score_opt = _run_hardmax_descent(
            a_start,
            n=n,
            size1=size1,
            size2=size2,
            deadline=run_deadline,
        )
        print(f"[run {idx+1}] final_score={score_opt:.6f}")

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()

    # --------------------------------------------------------------
    # 3️⃣  Fallback – if nothing succeeded, use the default Cantor histogram
    # --------------------------------------------------------------
    if best_overall_a is None:
        print("[fallback] using plain Cantor histogram (r=1/3)")
        best_overall_a = _fractal_histogram(n, L=14, r=1.0 / 3.0, seed=seed)

    total_time = time.time() - start
    print(f"[done] best_score={best_overall_score:.6f} time={total_time:.1f}s")
    # store globally so that later runs can reuse it (optional)
    globals()["height_sequence_2"] = best_overall_a.tolist()
    return best_overall_a.tolist()
