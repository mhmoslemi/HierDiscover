Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initial seed** – reuse the previously found construction (global
   ``height_sequence_1``) or fall back to a sine‑window.
2. **Linear‑programming improvement** – for a fixed length `n` we solve the
   LP

        max   sum(b)
        s.t.  (f * b)[i] ≤ M   ∀ i,   b ≥ 0

   where `f` is the current sequence, `M = max(f*f)`.  The LP pushes mass to
   positions where the convolution coefficients are smallest, thereby reducing
   the maximal autocorrelation.  The solution `b` is normalised to a probability
   vector `g` and either replaces `f` (if it improves the objective) or is
   mixed with `f` (`f←(1‑t)f + t·g`).  A few dozen LP steps are performed.
3. **Fine tuning** – a short Adam optimisation of a smooth max‑loss,
   followed by exact sub‑gradient mirror‑descent and a quick simulated‑annealing
   walk (the same primitives used in the previous best code).
4. The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned as a plain list of floats.

The whole routine is time‑aware and stops before the given budget (default
1000 s).  All numbers are non‑negative and the sequence is kept normalised to
sum = 1, which is invariant for the evaluation function.

import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real inputs)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a non‑negative sequence."""
    return _conv_fft(seq, seq).max()


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Linear‑programming optimiser (single LP object reused)
# ----------------------------------------------------------------------


class LPImprover:
    """
    Given a current distribution `f` (sum = 1) and its maximal autocorrelation
    `M = max(f*f)`, the LP

        max   1ᵀb
        s.t.  conv(f, b) ≤ M   (elementwise)
              b ≥ 0

    is solved.  The optimiser keeps the cvxpy objects alive and only updates the
    parameter values on each call.
    """

    def __init__(self, n: int):
        self.n = n
        self.b = cp.Variable(n)                       # decision variable
        self.f_param = cp.Parameter(n, nonneg=True)   # current f
        self.M_param = cp.Parameter(nonneg=True)      # current bound M
        # Linear convolution (f * b) – linear in b, constant in f
        conv_expr = cp.conv(self.f_param, self.b)     # length 2n‑1
        constraints = [conv_expr <= self.M_param, self.b >= 0]
        self.prob = cp.Problem(cp.Maximize(cp.sum(self.b)), constraints)
        # solver settings – ECOS works well for pure LPs
        self.solver_opts = {"max_iters": 5000, "abstol": 1e-9, "reltol": 1e-9}

    def solve(self, f: np.ndarray, M: float) -> np.ndarray | None:
        """Solve the LP for the supplied `f` and `M`.  Returns the optimal `b`."""
        self.f_param.value = f
        self.M_param.value = M
        try:
            self.prob.solve(solver=cp.ECOS, warm_start=True, **self.solver_opts)
        except Exception:  # solver may raise if infeasible
            return None
        if self.b.value is None:
            return None
        return np.array(self.b.value, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss (same as previous code)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000) -> tuple[np.ndarray, float]:
    v = np.log(seed + 1e-30)          # logits – stay on simplex via softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta = 2.0
    beta_mul = 1.02
    best_seq = None
    best_val = float("inf")
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:4d} best={cur_val:.6f}")
        # smoothed‑max weights
        beta *= beta_mul
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]
        # gradient w.r.t logits v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * _max_conv(a) / (a.sum() ** 2)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")
        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (small random mass moves)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * _max_conv(cur) / (cur.sum() ** 2)
    best = cur.copy()
    best_val = cur_val
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = 2.0 * n * _max_conv(new_seq) / (new_seq.sum() ** 2)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:5d} best={new_val:.6f}")
        if it % 1000 == 0:
            # occasional Gaussian jitter
            noise = rng.normal(scale=0.001, size=n)
            jump = cur + noise
            jump = np.clip(jump, 0.0, None)
            s = jump.sum()
            if s == 0.0:
                jump = _uniform(n)
            else:
                jump /= s
            jump_val = 2.0 * n * _max_conv(jump) / (jump.sum() ** 2)
            accept = jump_val < cur_val or rng.random() < np.exp(-(jump_val - cur_val) / (T + 1e-12))
            if accept:
                cur = jump
                cur_val = jump_val
                if jump_val < best_val:
                    best_val = jump_val
                    best = jump.copy()
                    print(f"[Anneal] it={it:5d} (jump) best={jump_val:.6f}")
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Length to use – 4000 gave the best known bound; keep it fixed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from previous global or fall back to a sine window.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, (list, tuple, np.ndarray)) and len(previous) == n:
        init = np.asarray(previous, dtype=np.float64)
    else:
        init = _sine_window(n)

    init = np.clip(init, 0.0, None)
    s = init.sum()
    if s == 0.0:
        init = _uniform(n)
    else:
        init = init / s

    best_seq = init.copy()
    best_val = 2.0 * n * _max_conv(best_seq) / (best_seq.sum() ** 2)
    print(f"[Init] n={n} obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP driven improvement
    # ------------------------------------------------------------------
    lp = LPImprover(n)
    max_lp_iters = 15          # generous but cheap
    t_mix = 0.5                # mixing weight if full replacement fails
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break
        M = _max_conv(best_seq)          # current bound
        b_sol = lp.solve(best_seq, M)
        if b_sol is None:
            print("[LP] solver failed – exiting LP phase")
            break
        sum_b = b_sol.sum()
        if sum_b <= 0:
            print("[LP] trivial solution – exiting LP phase")
            break
        g = b_sol / sum_b                # normalised candidate

        # Try full replacement first
        cand_seq = g
        cand_val = 2.0 * n * _max_conv(cand_seq) / (cand_seq.sum() ** 2)
        if cand_val < best_val:
            best_seq = cand_seq
            best_val = cand_val
            print(f"[LP] it={lp_it:2d} replace obj={cand_val:.6f}")
            continue

        # Otherwise mix with current sequence
        mixed = (1 - t_mix) * best_seq + t_mix * g
        mixed = np.clip(mixed, 0.0, None)
        mixed = mixed / mixed.sum()
        mixed_val = 2.0 * n * _max_conv(mixed) / (mixed.sum() ** 2)
        if mixed_val < best_val:
            best_seq = mixed
            best_val = mixed_val
            print(f"[LP] it={lp_it:2d} mix obj={mixed_val:.6f}")
            # gradually increase mixing strength to accelerate later steps
            t_mix = min(0.9, t_mix * 1.15)
        else:
            # no progress – shrink mixing step
            t_mix = max(0.05, t_mix * 0.6)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_adam, val_adam = _adam_optimize(best_seq, n, deadline, max_iters=5000)
        if seq_adam is not None and val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Overall Adam] best={val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sub, val_sub = _subgradient_refine(best_seq, n, deadline, max_iters=25000)
        if seq_sub is not None and val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"[Overall SubGrad] best={val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if seq_ann is not None and val_ann < best_val:
            best_seq, best_val = seq_ann, val_ann
            print(f"[Overall Anneal] best={val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final reporting and global export
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)          # safety
    s = best_seq.sum()
    if s == 0.0:
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / s

    # The grader expects a plain list of floats
    final_list = best_seq.tolist()
    globals()["height_sequence_1"] = final_list  # make it reusable
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {n})")
    return final_list
