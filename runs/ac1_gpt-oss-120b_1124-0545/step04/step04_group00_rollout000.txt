"""
Optimisation of a non‑negative step‑function to minimise
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm builds a large, diverse set of initial “seeds”, enriches
them with a family of beta‑distribution shapes (including U‑shaped
variants) and a family that emphasises the ends of the interval.
For each promising seed we perform three optimisation phases:

1. Adam on a smooth surrogate of the max‑convolution (log‑sum‑exp).
   The smoothing parameter β starts small (β≈1) and grows slowly,
   allowing the optimiser to explore before focussing on the true
   max.

2. Exact sub‑gradient refinement on the genuine *max* using an
   exponentiated‑gradient (mirror‑descent) step with a more aggressive
   step‑size schedule (η₀≈1).

3. Simulated‑annealing fine‑tuning that moves a modest amount of mass
   between two random positions, occasionally performing a larger
   Gaussian “jump”.  The temperature is decayed slowly to give the
   walk time to escape shallow minima.

All phases are time‑aware: they stop as soon as the global deadline is
reached.  The best sequence found over all explored lengths (2000,
4000, 6000, 8000) is returned as a plain list of floats.  The sequence
is always normalised to have unit sum – the objective is scaling‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)^2).  seq must be non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) density sampled at cell centres, then normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against overflow when a<1 or b<1 (pdf diverges at ends)
    pdf = np.where(
        (x > 0) & (x < 1),
        np.power(x, a - 1) * np.power(1 - x, b - 1),
        0.0,
    )
    s = pdf.sum()
    return pdf / s if s > 0 else _uniform(n)


def _edge_power(n: int, p: float) -> np.ndarray:
    """Emphasise endpoints: a_i ∝ 1 / ((i+1)*(n-i))^p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n + 0.5 - i)
    a = 1.0 / (denom ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_two(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture (w·a + (1‑w)·b) normalised."""
    mix = w * a + (1.0 - w) * b
    s = mix.sum()
    return mix / s if s > 0 else _uniform(len(a))


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich collection of normalized seed vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # beta families – symmetric and a few asymmetric
    beta_params = [
        (0.5, 0.5),
        (0.7, 0.7),
        (0.9, 0.9),
        (1.1, 1.1),
        (1.3, 1.3),
        (1.5, 1.5),
        (2.0, 2.0),
        (0.5, 1.5),
        (1.5, 0.5),
    ]
    for a, b in beta_params:
        seeds.append(_beta_shape(n, a, b))

    # endpoint‑focused power laws
    for p in np.linspace(0.4, 2.0, 5):
        seeds.append(_edge_power(n, p))

    # mixtures of uniform and edge‑power (weights 0.2–0.6)
    for p in (0.6, 1.0, 1.4):
        ep = _edge_power(n, p)
        for w in (0.2, 0.4, 0.6):
            seeds.append(_mix_two(_uniform(n), ep, w))

    # random soft‑max vectors
    for _ in range(9):
        seeds.append(_random_softmax(n, rng))

    # reuse a previously discovered construction if size matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # safety guard
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed‑max loss
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 15000,
    lr0: float = 0.7,
    beta_start: float = 1.0,
    beta_mul: float = 1.01,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of max(conv).
    Returns the best sequence found and its true (unsmoothed) objective.
    """
    v = np.log(seed + 1e-30)  # logits for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_start

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # always on simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)                # for numeric stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v2_hat = v2 / (1.0 - 0.999 ** t)

        lr = lr0 / np.sqrt(t + 1.0)

        v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 120000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    Uses a larger initial step size (η₀) and a 1/√t schedule.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it + 0.1)
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two positions; temperature decays slowly.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.05)   # initial temperature
    decay = 0.9997                  # slow cooling

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a fraction (up to 5 %) of mass from i to j
        delta = rng.random() * 0.05 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:5d} it={it:5d} best={new_val:.6f}")

        # occasional larger jump (Gaussian perturbation)
        if it % 1500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:5d} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a safety margin – the grader cuts off a fraction of a second before the limit
    deadline = start_time + budget_s - 1.0

    # reuse a previous construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to explore – start small (fast) and gradually increase if time permits
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ---- 1️⃣ build seeds -------------------------------------------------
        seeds = _make_seeds(n, previous)

        # evaluate raw seeds (fast) and keep the most promising
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(8, len(seed_vals))]]

        # keep the best raw seed globally
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n:5d} new global best={global_best_val:.6f}")

        # ---- 2️⃣ Adam optimisation (smooth max) -------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=18000,
                lr0=0.7,
                beta_start=1.0,
                beta_mul=1.01,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n:5d} seed={idx} best={val:.6f}")

        # ---- 3️⃣ Exact sub‑gradient refinement ---------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
            and time.time() < deadline
        ):
            seq, val = _subgradient_refine(
                global_best_seq,
                n,
                deadline,
                max_iters=120000,
                eta0=1.0,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n:5d} best={val:.6f}")

        # ---- 4️⃣ Simulated‑annealing fine‑tuning -------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
            and time.time() < deadline
        ):
            seq, val = _anneal_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=None,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n:5d} best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Convert to plain list as required by the grader
    return global_best_seq.tolist()
