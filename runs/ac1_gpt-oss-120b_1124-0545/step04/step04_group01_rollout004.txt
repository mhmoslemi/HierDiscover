"""
Search for a non‑negative step‑function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in several phases while respecting a 1000 s time
budget:

1. **Seed generation** – a large, diverse family of analytical shapes
   (uniform, sine‑window, geometric, power‑law, symmetric geometric,
   Gaussian, raised‑cosine, triangular) together with random Dirichlet
   draws and the previously discovered construction (if any).  All seeds
   are ℓ₁‑normalised to sum = 1.

2. **Quick scan** – evaluate every seed with a single convolution and
   keep the few best candidates.

3. **Adam optimisation** – start from each promising seed and run a
   limited‑iteration Adam optimiser on a log‑softmax parametrisation.
   The loss is a smoothed max (log‑sum‑exp) of the auto‑convolution.
   The best iterate is symmetrised (a_i = a_{n‑1‑i}) and renormalised.

4. **Exact sub‑gradient refinement** – mirror‑descent (exponentiated
   gradient) with the true sub‑gradient of the max‑convolution.  After
   each update the vector is projected onto the symmetric subspace.

5. **Stochastic hill‑climbing** – a cheap random pair‑wise mass‑transfer
   neighbourhood search that also enforces symmetry.  Improvements are
   kept and the step size decays slowly.

All phases stop as soon as the global deadline is reached.  The best
sequence encountered is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} and renormalise."""
    b = (a + a[::-1]) * 0.5
    s = b.sum()
    if s == 0.0:
        return np.full_like(b, 1.0 / len(b))
    return b / s


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian, normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - mid) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power_shape(n: int, p: float, w_frac: float) -> np.ndarray:
    """
    Symmetric compact support shape:
        a_i = (1 - |i-mid|/W)^p   for |i-mid| ≤ W, else 0.
    The half‑width W = w_frac * (n‑1)/2.
    """
    mid = (n - 1) / 2.0
    W = w_frac * (n - 1) / 2.0
    if W <= 0:
        return _uniform(n)
    idx = np.arange(n, dtype=np.float64) - mid
    mask = np.abs(idx) <= W
    a = np.zeros_like(idx)
    a[mask] = (1.0 - np.abs(idx[mask]) / W) ** p
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _cosine_shape(n: int, w_frac: float) -> np.ndarray:
    """
    Symmetric raised‑cosine window with half‑width W = w_frac * (n‑1)/2.
    a_i = 0.5·(1+cos(π·|i‑mid|/W)) for |i‑mid| ≤ W, else 0.
    """
    mid = (n - 1) / 2.0
    W = w_frac * (n - 1) / 2.0
    if W <= 0:
        return _uniform(n)
    idx = np.arange(n, dtype=np.float64) - mid
    mask = np.abs(idx) <= W
    a = np.zeros_like(idx)
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * np.abs(idx[mask]) / W))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth approximation of max(conv)."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.04  # slightly faster smoothing increase

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        # enforce symmetry at each iteration – it only helps the objective
        a = _symmetrize(a)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # final symmetry enforce (just in case)
    if best_seq is not None:
        best_seq = _symmetrize(best_seq)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """Mirror‑descent refinement using the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # enforce symmetry after each update
        a = _symmetrize(a)

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """Cheap stochastic hill‑climbing with symmetric mass transfers."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # start with a few‑percent transfer

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        # enforce symmetry and renormalise
        new = _symmetrize(new)
        new_sum = new.sum()
        if new_sum == 0.0:
            new = _uniform(n)
        else:
            new /= new_sum

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale *= 0.85
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of normalised seed vectors."""
    rng = np.random.default_rng(2023 + n)  # deterministic per length
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian family (sigma = factor * n)
    for sigma_fac in np.linspace(0.12, 0.48, 9):
        sigma = sigma_fac * n
        seeds.append(_gaussian_shape(n, sigma))

    # Power‑shape (compact support) family
    for w_frac in np.linspace(0.3, 0.9, 7):
        for p in np.linspace(0.5, 8.0, 7):
            seeds.append(_power_shape(n, p, w_frac))

    # Raised‑cosine family
    for w_frac in np.linspace(0.3, 0.9, 5):
        seeds.append(_cosine_shape(n, w_frac))

    # Triangular (special case of power‑shape with p=1)
    for w_frac in np.linspace(0.3, 0.9, 5):
        seeds.append(_power_shape(n, 1.0, w_frac))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors
    for _ in range(5):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function of non‑negative heights that minimises
    the ``evaluate_sequence`` objective.  Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Possibly reuse a previously known construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore – larger n gives more freedom but costs more
    lengths = [2000, 4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # --------------------------------------------------------------
        # Phase 0 – quick scan of raw seeds, keep the top few
        # --------------------------------------------------------------
        seed_vals = []
        for s in seeds:
            if time.time() > deadline:
                break
            # All seeds are already ℓ₁‑normalised, so denominator = 1
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            seed_vals.append((val, s.copy()))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # keep only a handful of promising seeds
        seed_vals.sort(key=lambda t: t[0])
        top_seeds = [sv[1] for sv in seed_vals[:5]]

        # --------------------------------------------------------------
        # Phase 1 – Adam optimisation for each promising seed
        # --------------------------------------------------------------
        for seed_vec in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=5000)
            if seq is not None:
                seq = _symmetrize(seq)          # enforce symmetry before refinement
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=25000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Stochastic hill‑climbing
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                         max_iters=15000)
            if seq_hc is not None and val_hc < global_best_val:
                global_best_val = val_hc
                global_best_seq = seq_hc.copy()
                print(f"[overall‑Hill] n={n:5d} new best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Verify with the provided evaluate_sequence (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
