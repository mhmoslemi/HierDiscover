def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The method consists of three stages:

1.  A cheap parametric sweep over a few analytically‑motivated families
    (inverse‑product, sum‑power, exponential, mixtures with the uniform
    distribution).  Each candidate is normalised to sum = 1 and its exact
    objective (``2·n·max(conv)``) is computed with an FFT – this yields a
    strong starting point in a fraction of a second.

2.  Starting from the best candidate, a simple projected‑subgradient
    refinement is performed.  The sub‑gradient of the hard‑max objective
    is obtained from the active mask of the convolution, and mass is
    shifted from a location with a large positive gradient to one with a
    large negative gradient.  After every move the vector is projected onto
    the probability simplex (non‑negative, sum = 1).  The step size is
    reduced like ``η / sqrt(iter)`` and the loop stops when the time budget
    is exhausted.

3.  The final sequence (as a plain ``list[float]``) is returned.

The entire routine respects the supplied time budget (default 1000 s)
and never performs any file or network I/O.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        score = 2·n·max(conv(a,a))
    The convolution is performed by FFT for speed.
    """
    n = a.shape[0]
    if n == 0:
        return float("inf")
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s                     # enforce sum = 1 (normally already true)
    L = _next_pow2(2 * n - 1)
    fa = np.fft.fft(a, L)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    maxc = conv.max()
    return float(2.0 * n * maxc)


def _subgradient(a: np.ndarray, L: int) -> Tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of the hard‑max objective
    2·n·max(conv(a,a)) and return the current maximum convolution value.
    The implementation follows the derivation used in the reference code.
    """
    n = a.shape[0]
    ft = np.fft.fft(a, L)
    conv = np.real(np.fft.ifft(ft * ft))[: 2 * n - 1]
    max_val = conv.max()
    # mask of active indices (allow tiny numerical tolerance)
    eps_mask = max_val * 1e-12 + 1e-14
    mask = (conv >= max_val - eps_mask).astype(float)
    fw = np.fft.fft(mask, L)

    # linear convolution part (a * mask)
    conv_wa = np.real(np.fft.ifft(fw * ft))[: 3 * n - 2]
    g2 = conv_wa[:n]

    # cross‑correlation part (a ⋆ mask)
    corr_full = np.real(np.fft.ifft(np.conj(fw) * ft))[: 3 * n - 2]
    g1 = corr_full[:n]

    # sub‑gradient of 2·n·max(conv)
    g = 2.0 * n * (g1 + g2)
    return g, max_val


def _refine_sequence(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient refinement.  Mass is shifted from the
    index with the largest positive gradient to the one with the most
    negative gradient; the step size decays as 1/√iteration.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score_of(a)
    it = 0

    while time.time() < deadline:
        g, _ = _subgradient(a, L)
        # indices that most increase / decrease the objective
        i_dep = int(np.argmax(g))      # remove mass here
        i_inc = int(np.argmin(g))      # add mass here
        if a[i_dep] <= 1e-12:
            # if the chosen donor is empty, pick a random donor
            i_dep = rng.choice(np.flatnonzero(a > 1e-12))
            if i_dep is None:
                break
        # step size – start with a modest fraction of the available mass
        it += 1
        eta = 0.2 / np.sqrt(it)        # 0.2 is a hand‑tuned constant
        delta = min(a[i_dep] * eta, 0.001)  # never move more than 0.001 total mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_dep] -= delta
        a_new[i_inc] += delta
        # project back onto the simplex (preserves sum)
        a_new = _project_simplex(a_new)

        cur_score = _score_of(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
            # occasional re‑normalisation to avoid drift
            a = a / a.sum()
        else:
            # occasional random jitter to escape plateaus
            if it % 500 == 0:
                jitter = rng.normal(scale=0.001, size=n)
                a = _project_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------


def _family_product(n: int, p: float, eps: float) -> np.ndarray:
    """a_i ∝ ((i+eps)*(n-1-i+eps))^{-p}   (inverse product shape)."""
    i = np.arange(n, dtype=float)
    base = (i + eps) * (n - 1 - i + eps) + 1e-12
    w = base ** (-p)
    return w


def _family_sum(n: int, p: float, eps: float) -> np.ndarray:
    """a_i ∝ (i+eps)^{-p} + (n-1-i+eps)^{-p}   (sum of two power‑laws)."""
    i = np.arange(n, dtype=float)
    w = (i + eps) ** (-p) + (n - 1 - i + eps) ** (-p)
    return w


def _family_exp(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (simple exponential decay)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    return w


def _mixture(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination α·a + (1‑α)·b."""
    return alpha * a + (1.0 - alpha) * b

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget (default 1000 s) and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Gather a few simple baselines (uniform, historic if present)
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    # historic sequence (if defined)
    if "height_sequence_1" in globals():
        hist = np.array(globals()["height_sequence_1"], dtype=float)
        if hist.size != n:
            if hist.size > n:
                hist = hist[:n]
            else:
                hist = np.pad(hist, (0, n - hist.size))
        hist = np.maximum(hist, 0.0)
        hist = hist / max(hist.sum(), 1e-12)
        candidates.append(hist)

    # ------------------------------------------------------------------
    # 2) Parametric sweep over analytically‑motivated families
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    # ----- product family (inverse product) -----
    for p in np.linspace(0.5, 2.0, 16):
        for eps in (0.5, 1.0, 2.0):
            w = _family_product(n, p, eps)
            a = w / w.sum()
            candidates.append(a)

    # ----- sum‑power family -----
    for p in np.linspace(0.5, 2.0, 12):
        for eps in (0.5, 1.0, 2.0):
            w = _family_sum(n, p, eps)
            a = w / w.sum()
            candidates.append(a)

    # ----- exponential family -----
    for lam in np.linspace(0.0005, 0.005, 15):
        w = _family_exp(n, lam)
        a = w / w.sum()
        candidates.append(a)

    # ----- mixtures with the uniform distribution -----
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        # mix product‑shape with uniform
        w = _family_product(n, 0.8, 1.0)
        prod = w / w.sum()
        mix = _mixture(prod, uniform, alpha)
        candidates.append(mix / mix.sum())

    # deduplicate (optional – cheap for modest list sizes)
    # ------------------------------------------------------------------
    # 3) Evaluate all candidates – keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # cache FFT length for speed
    fft_len = _next_pow2(2 * n - 1)

    for a in candidates:
        if time.time() >= deadline:
            break
        # normalise (just in case)
        a = np.maximum(a, 0.0)
        a = a / max(a.sum(), 1e-12)
        cur_score = _score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            elapsed = time.time() - start_time
            print(
                f"[grid] new best = {best_score:.6f} (elapsed {elapsed:.1f}s, "
                f"type={a.shape})"
            )

    # ------------------------------------------------------------------
    # 4) Local refinement from the best candidate (projected sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate the remaining time (≈ 30‑40 % of total) to refinement
        refine_deadline = deadline
        refined_seq, refined_score = _refine_sequence(
            best_seq, refine_deadline, rng
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(
                f"[refine] improved to {best_score:.6f} (elapsed {time.time() - start_time:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 5) Fallback – uniform if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # enforce final non‑negativity and normalisation (just for safety)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = best_seq / max(best_seq.sum(), 1e-12)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, time = {total_time:.1f}s")
    return best_seq.tolist()
