def project_simplex(v):
   # from literature: sort v descending, find rho etc.
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

"""
Two‑phase optimisation for the self‑convolution bound.

Phase 0 – cheap parametric sweep over a few analytically‑motivated families
        (inverse‑square‑root, power‑law, symmetric exponential).  The best
        shape (lowest exact objective) is kept.

Phase 1 – long simulated‑annealing run on the simplex.  Each move transfers a
        random fraction of mass between two coordinates; the exact
        convolution max is obtained via a real FFT (O(n log n)).  The
        temperature is linearly cooled; after a period without improvement a
        restart from the current best with a small Gaussian jitter is performed.

All numbers stay non‑negative and are normalised to sum to 1, so the returned
list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def conv_max_linear(a: np.ndarray, fft_len: int) -> float:
    """
    Linear self‑convolution of a (length n) using an FFT of size ``fft_len``.
    Returns the maximum entry of the convolution.
    """
    # real FFT is about twice as fast as the complex version
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # linear convolution length = 2*n-1; the rest of the zero‑padded tail is zero
    return float(conv[: 2 * a.size - 1].max())


def normalise_positive(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1 (handles tiny numerical drift)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fallback – uniform distribution
        n = v.shape[0]
        return np.full(n, 1.0 / n)
    return v / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (default 2000)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    fft_len = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3·n‑2

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    best_max = None

    # helper to evaluate a candidate and possibly update the incumbent
    def try_candidate(a: np.ndarray, label: str = None):
        nonlocal best_a, best_score, best_max
        a = normalise_positive(a)
        max_conv = conv_max_linear(a, fft_len)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_max = max_conv
            if label is not None:
                print(f"[grid] new best ({label}) = {best_score:.6f}")

    # 0.1 – any previously discovered construction
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=float)
        if arr.size == n and arr.sum() > 0.0:
            try_candidate(arr, "global_start")

    # 0.2 – classic arcsine (inverse‑sqrt) shape
    i = np.arange(n, dtype=float)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    try_candidate(arcsine, "arcsine")

    # 0.3 – power‑law family  a_i ∝ ((i+½)(n‑i‑½))^{-p}
    p_vals = np.linspace(0.30, 2.40, 20)          # coarse sweep
    for p in p_vals:
        a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        try_candidate(a, f"power p={p:.2f}")

    # 0.4 – symmetric exponential family  a_i ∝ e^{-λ i}+e^{-λ (n‑1‑i)}
    lam_vals = np.linspace(0.001, 0.020, 20)
    for lam in lam_vals:
        a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        try_candidate(a, f"exp λ={lam:.4f}")

    # ------------------------------------------------------------------
    # Phase 1 – simulated annealing (exact objective)
    # ------------------------------------------------------------------
    # If the sweep produced nothing (should not happen) fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    cur_a = best_a.copy()
    cur_max = best_max if best_max is not None else conv_max_linear(cur_a, fft_len)
    cur_score = 2.0 * n * cur_max

    # SA schedule parameters
    T_start = best_score * 0.01          # start temperature (≈ 1 % of current score)
    T_end   = best_score * 1e-7          # end temperature (practically zero)
    phase_start = time.time()
    last_improve = phase_start

    sa_iters = 0
    while time.time() < overall_deadline:
        now = time.time()
        # linear cooling based on elapsed time inside this phase
        elapsed_phase = now - phase_start
        phase_frac = min(1.0, elapsed_phase / (overall_deadline - phase_start))
        T = T_start * (1.0 - phase_frac) + T_end * phase_frac

        # ---- propose a mass‑transfer move ---------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1                     # guarantee i ≠ j

        # move at most a fraction of the source mass (large early, tiny later)
        max_move = cur_a[i]
        if max_move <= 0.0:
            continue
        move_fraction = 0.10 * (1.0 - phase_frac) + 0.01   # 10 % → 1 %
        delta = rng.random() * max_move * move_fraction

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # ---- exact evaluation --------------------------------------------
        new_max = conv_max_linear(a_new, fft_len)
        new_score = 2.0 * n * new_max

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            cur_max = new_max
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                last_improve = time.time()
                print(f"[SA {sa_iters:07d}] improvement → {best_score:.6f}")

        # ---- occasional restart to escape stagnation -----------------------
        if time.time() - last_improve > 30.0:
            # small Gaussian jitter around the best known point
            cur_a = best_a + 0.001 * rng.standard_normal(n)
            cur_a = normalise_positive(cur_a)
            cur_max = conv_max_linear(cur_a, fft_len)
            cur_score = 2.0 * n * cur_max
            last_improve = time.time()
            print(f"[restart] current score reset to {cur_score:.6f}")

        sa_iters += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = normalise_positive(best_a)
    elapsed = time.time() - start_time
    print(f"[done] total time={elapsed:.1f}s, best score={best_score:.6f}, length={n}")

    return best_a.tolist()
