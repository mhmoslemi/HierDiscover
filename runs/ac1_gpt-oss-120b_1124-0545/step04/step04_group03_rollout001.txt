"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. **Analytic grid search** – generate many power‑law and exponential families
   (product / sum modes) on the probability simplex and keep the best one.

2. **Subgradient descent on the true max‑convolution** – the objective after
   normalising is 2·n·max(conv).  For the current peak index k* the subgradient
   w.r.t. a_i is 2·a_{k*‑i} (zero if the partner index is out of range).  We
   perform a multiplicative update `a <- a * exp(-η·g)` and re‑normalise, with a
   diminishing step size η_t = η0 / sqrt(t).  When no improvement is seen for a
   while we inject a random mass‑transfer move to escape plateaus.

3. **Optional L‑BFGS refinement** – if SciPy is available and a few seconds remain,
   run a short L‑BFGS optimisation of a smooth log‑sum‑exp approximation of the
   max‑convolution (large λ) starting from the best point found.

The routine respects a 1000 s budget, prints progress, and returns the best
sequence discovered (as a Python list of floats).  All elements are non‑negative
and the sequence length defaults to 2000 but can be overridden via the ``n``
keyword argument.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum = 1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'     → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def exponential_candidate(n: int, c: float) -> np.ndarray:
    """
    Symmetric double‑exponential candidate:
    a_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    a = a / a.sum()
    return a


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (assumed on the simplex).
    Returns 2·n·max(conv(a,a)).
    """
    # a is assumed to sum to 1; we normalise just in case.
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def max_convolution_gradient(a: np.ndarray, conv: np.ndarray):
    """
    Given a (sum‑1) and its full self‑convolution conv,
    return the subgradient of max(conv) w.r.t. a,
    together with the index where the max is attained.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))         # index of the current maximum
    # g_i = 2 * a_{k-i} for all i where 0 ≤ k-i < n, else 0
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j = k - i_range                 # partner indices
    g = np.zeros_like(a)
    g[i_range] = 2.0 * a[j]
    return g, k


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of probability mass from one index to another.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Smooth max utilities for the optional L‑BFGS refinement
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth approximation Φ_λ(a) of the max‑convolution and its gradient w.r.t.
    the soft‑max parameters x (a = softmax(x)).
    Returns (Φ_λ, ∇_x Φ_λ).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max via log‑sum‑exp
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                 # weights, sum to 1

    # gradient w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]  # length n
    grad_a = 2.0 * slice_part[::-1]

    # map gradient to x‑space (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that minimises
    the evaluation function.  The routine:

    * builds an extensive grid of analytic families and keeps the best;
    * performs a large‑scale subgradient descent on the true max‑convolution;
    * occasionally injects random mass‑transfer moves to escape plateaus;
    * finishes with an optional L‑BFGS refinement (if SciPy is available);
    * returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search
    # ------------------------------------------------------------------
    candidates = []

    # Power‑law families (product and sum)
    for mode in ("product", "sum"):
        for p in np.arange(0.30, 0.95, 0.05):
            for off in np.arange(0.0, 2.05, 0.20):
                # avoid a zero factor in the product mode
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, float(p), float(off_eff), mode)
                candidates.append(cand)

    # Double‑exponential families
    for c in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        cand = exponential_candidate(n, float(c))
        candidates.append(cand)

    # Uniform distribution (baseline)
    candidates.append(np.ones(n) / n)

    # Global construction if available and of matching size
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    best_a = None
    best_score = np.inf

    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Subgradient descent on max(conv)
    # ------------------------------------------------------------------
    a = best_a.copy()
    conv = conv_fft_self(a)
    current_score = 2.0 * n * np.max(conv)
    best_score = current_score
    best_a = a.copy()

    iters = 0
    improve_counter = 0
    stagnation_counter = 0
    step0 = 0.7                           # initial step size coefficient

    while time.time() < deadline:
        iters += 1

        # gradient of the current max‑convolution entry
        g, k_max = max_convolution_gradient(a, conv)

        # diminishing step size
        eta = step0 / math.sqrt(iters)

        # multiplicative update (maintains positivity)
        a = a * np.exp(-eta * g)
        a = a / a.sum()                    # project onto simplex

        # recompute convolution and objective for the new point
        conv = conv_fft_self(a)
        max_val = np.max(conv)
        current_score = 2.0 * n * max_val

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            improve_counter += 1
            stagnation_counter = 0
            if improve_counter % 10 == 0:
                elapsed = time.time() - start_time
                print(f"[subgrad] iter {iters:6d}  new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            stagnation_counter += 1

        # if we have been stagnant for a while, inject a random move
        if stagnation_counter >= 5000:
            a = random_move(a, scale=0.05)
            a = a / a.sum()
            conv = conv_fft_self(a)
            stagnation_counter = 0

    elapsed = time.time() - start_time
    print(f"[subgrad] finished after {iters} iterations, best = {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Optional L‑BFGS refinement (smooth max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        try:
            from scipy.optimize import minimize
            have_scipy = True
        except Exception:                 # pragma: no cover
            have_scipy = False

        if have_scipy:
            eps = 1e-12
            x = np.log(np.maximum(best_a, eps))   # initialise soft‑max params
            lam = 5000.0                           # large λ ⇒ close to hard max
            # allocate iterations proportional to remaining time
            max_iter = int(min(2000, max(20, remaining * 150)))
            print(f"[lbfgs] starting L‑BFGS (λ={lam:.0f}, maxiter={max_iter})")
            res = minimize(
                compute_phi_and_grad,
                x,
                args=(lam, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'disp': False}
            )
            a_opt = softmax(res.x)
            sc_opt = compute_score(a_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[lbfgs] improved to {best_score:.6f}")
        else:
            print("[lbfgs] SciPy not available – skipping refinement")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Return a Python list of floats (evaluate_sequence will further clip/normalise)
    return best_a.tolist()
