def _beta_histogram(n, alpha):
    x = (np.arange(n)+0.5)/n
    pdf = x**(alpha-1)*(1-x)**(alpha-1)
    # Normalization constant B(alpha,alpha) = gamma(alpha)**2 / gamma(2*alpha)
    from math import gamma
    B = gamma(alpha)*gamma(alpha)/gamma(2*alpha)
    pdf /= B
    a = pdf / n   # integrate pdf over [0,1] approximated by sum(pdf)/n = 1, but compute exactly:
    a = pdf / n  # Since integral approximated by sum(pdf)*(1/n) we need sum(pdf)/n=1 => pdf sum = n. Actually we need a_i = pdf(x_i)*(1/n). So sum a_i approximates 1.
    return a

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 6000))

    # Precompute FFT size for conv.
    size_c = _next_pow2(2*n-1)
    size_g = _next_pow2(3*n-2)

    # Phase 1: simple candidates.

    alphas = np.array([0.2,0.3,0.35,0.4,0.45,0.5,0.55,0.6,0.7,0.8,0.9,1.0,1.2,1.5,2.0,3.0])
    best_score = np.inf
    best_seq = None
    for alpha in alphas:
        a = _beta_histogram(n, alpha)
        score = _score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    # also include Cantor init if present
    try:
        seq0 = globals().get('height_sequence_1')
        if seq0 is not None:
            a0 = np.asarray(seq0, dtype=float)
            if a0.size != n:
                # resample
                x_old = np.linspace(0,1,a0.size,endpoint=False)
                x_new = np.linspace(0,1,n,endpoint=False)
                a0 = np.interp(x_new, x_old, a0)
            a0 = np.clip(a0,0, None)
            s = a0.sum()
            if s < 0.01:
                a0 = _beta_histogram(n,0.5)
            else:
                a0 = a0 / s
            score = _score(a0)
            if score < best_score:
                best_score = score
                best_seq = a0.copy()
    except Exception:
        pass

    print(f"[phase1] n={n}, best initial score={best_score:.6f}")

    # Phase 2: gradient descent
    a = best_seq.copy()
    a_rev = a[::-1].copy()
    best_score = _score(a)  # recompute
    best_seq = a.copy()
    # hyperparameters
    base_step = 0.8
    tau_factor = 0.5
    tau_decay = 0.9995
    momentum = 0.85
    v = np.zeros_like(a)
    it = 0
    last_improve = start
    while time.time() < deadline:
        conv = _full_convolution(a, size_c)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum()**2)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best {best_score:.6f} time={elapsed:.1f}s")
        # compute smooth max weights
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay
        # softmax
        w = np.exp((conv - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum
        # gradient via conv of a_rev and w
        g_full = _conv_fft(a_rev, w, size_g)
        # take central n entries (indices n-1 .. n-1+n-1)
        start_idx = n - 1
        g = g_full[start_idx:start_idx + n]
        d = -2.0 * g
        # enforce zero-mean to keep sum unchanged
        d -= d.mean()
        # momentum update
        v = momentum * v + (1.0 - momentum) * d
        step = base_step / np.sqrt(it + 1.0)
        a = a + step * v

        s = a.sum()
        if s == 0:

        else:
            a /= s
        # update reversed copy

    total = time.time() - start
    print(f"[done] iterations={it}, best_score={best_score:.6f}, time={total:.1f}s")

"""
Propose a high‑quality non‑negative step‑function.

Algorithm
---------
1. **Diversified seeding** – Evaluate a handful of analytically‑defined
   probability densities (Beta(α,α) for several α, the classic Cantor
   histogram and the previously‑found best sequence).  Keep the one with
   the smallest self‑convolution peak.

2. **Smooth‑max projected gradient descent** – Repeatedly minimise a
   differentiable surrogate of  max(a⋆a) using a soft‑max weighting.
   The gradient is obtained by one extra FFT‑convolution
   (a_rev ∗ weights).  After each step the vector is projected onto the
   probability simplex (non‑negative, Σa=1).

3. **Escapes** – If no improvement has been seen for a while a small
   random perturbation (jitter) is added to the current best solution,
   the momentum buffer is cleared and the descent continues.  This
   helps to leave shallow local minima.

All operations are O(n log n) per iteration, so the routine fits easily
into the 1000 s budget even for n≈8000.  The best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np
import math
from scipy.special import gamma  # for Beta normalisation

# ----------------------------------------------------------------------
# Helper utilities – must be defined at top level (no closures / lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution by FFT.
    ``size`` must be ≥ len(u)+len(v)-1.
    Returns the *full* convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the full self‑convolution a⋆a (length 2·len(a)-1)."""
    return _conv_fft(a, a, size)


def _beta_histogram(n: int, alpha: float) -> np.ndarray:
    """
    Approximate a symmetric Beta(α,α) density on [0,1] with a histogram
    of length ``n``.  The result sums to 1 (up to floating‑point noise)
    and is non‑negative.
    """
    # bin centres
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Beta pdf without normalisation constant
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Normalisation constant B(α,α) = Γ(α)² / Γ(2α)
    B = gamma(alpha) * gamma(alpha) / gamma(2.0 * alpha)
    pdf /= B
    # Convert to a histogram (multiply by bin width = 1/n)
    a = pdf / n
    # Numerical guard – ensure non‑negative and normalised
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Histogram of the middle‑third Cantor measure with ``2**L`` atoms,
    binned into ``n`` equal intervals.
    """
    N = 1 << L
    powers = 3.0 ** np.arange(1, L + 1)
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    pos = (bits * (2.0 / powers)).sum(axis=1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _score(a: np.ndarray, size_conv: int) -> float:
    """
    Compute the objective used by ``evaluate_sequence`` (without the final
    clipping performed there).  The vector ``a`` is assumed to be
    non‑negative; its sum is used in the denominator.
    """
    n = a.shape[0]
    conv = _full_convolution(a, size_conv)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    size_grad = _next_pow2(3 * n - 2)   # for a_rev ⋆ weights

    except Exception:

    tau_decay = 0.9997         # exponential decay of temperature

    v = np.zeros_like(a)       # momentum buffer

            best_val = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start

            a = np.clip(a, 0.0, None)
            a_sum = a.sum()

            else:
                a /= a_sum
            a_rev = a[::-1].copy()

        tau = max(tau_factor * max_b, 1e-14)     # avoid division by zero
        tau_factor *= tau_decay                  # slowly anneal

        w_sum = w.sum()

        v = momentum * v + (1.0 - momentum) * d

        else:
            a /= a_sum

        a_rev = a[::-1].copy()
        it += 1

    total = time.time() - start
