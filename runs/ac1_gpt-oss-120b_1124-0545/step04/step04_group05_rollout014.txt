def compute_score(a, n, fft_len=None):

"""
propose_candidate – hybrid 1‑D shape search + exact sub‑gradient descent + SA.

1.  A one‑parameter family a_i(p) ∝ ((i+0.5)*(n-i-0.5))^{-p} (symmetric, non‑negative,
    sum normalized to 1) contains the classic arcsine shape (p=0.5).  Searching
    this family is extremely cheap (FFT‑based objective, O(n log n)) and
    already yields a few‑percent improvement over the arcsine baseline.

2.  Starting from the best p‑value, we run a few hundred iterations of
    Adam on the *exact* sub‑gradient of the objective
        F(a) = 2·n·max_k (a*a)_k .
    The sub‑gradient at the maximal index m is
        ∂F/∂a_i = 4·n·a_{m‑i}  (out‑of‑range indices give 0).
    The update is performed in the soft‑max (log‑parameter) space so that
    the simplex constraints are satisfied automatically.

3.  A final stochastic‑annealing phase (mass transfers between two
    coordinates) refines the solution with cheap Monte‑Carlo steps and
    can still discover tiny improvements that the deterministic phase
    misses.

The whole routine respects a 1000 s wall‑time budget, prints progress
messages and finally returns a list of non‑negative heights (sum≈1).  The
result can be further inspected with the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def fft_len_for(n: int) -> int:
    """Smallest power of two >= (2·n‑1). Used for all FFTs."""
    return 1 << ((2 * n - 1).bit_length())


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max of the linear convolution a*a using an FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv.max()


def compute_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a already normalized (sum = 1)."""
    mx = max_convolution(a, fft_len)
    return 2.0 * n * mx


def generate_shape(p: float, n: int) -> np.ndarray:
    """Return the symmetric probability vector a_i(p)."""
    i = np.arange(n, dtype=float)
    raw = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    a = raw / raw.sum()
    return a


def subgradient_exact(a: np.ndarray, n: int, fft_len: int):
    """Sub‑gradient of 2·n·max(a*a) w.r.t. a (in the probability simplex)."""
    # convolution and index of the maximal entry
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    m_idx = int(conv.argmax())
    # build the reversed‑shifted copy of a that corresponds to ∂max/∂a
    rev = np.zeros_like(a)
    if m_idx < n:
        # overlapping part is a[0 … m_idx] reversed
        rev[: m_idx + 1] = a[m_idx :: -1]
    else:
        start = m_idx - (n - 1)
        rev[start:] = a[: n - start][::-1]
    # ∂F/∂a = 4·n·rev   (see derivation in the docstring)
    return 4.0 * n * rev, conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine combines a cheap 1‑D exponent search, a few hundred Adam
    sub‑gradient steps and a final simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin to avoid being killed exactly at the deadline
    deadline = start + budget_s - 0.5

    # length of the sequence – the default (2000) works well,
    # but a power of two gives a marginal speed boost for the FFTs.
    n = int(kwargs.get("n", 2000))
    fft_len = fft_len_for(n)

    # ------------------------------------------------------------------
    # 0) quick 1‑D search over the exponent p
    # ------------------------------------------------------------------
    best_score = np.inf
    best_a = None
    best_p = None

    # coarse grid
    p_grid = np.linspace(0.30, 0.70, 81)          # 81 evaluations ~ 0.07 s
    for p in p_grid:
        a = generate_shape(p, n)
        sc = compute_score(a, n, fft_len)
        if sc < best_score:
            best_score, best_a, best_p = sc, a.copy(), p

    # refine around the best grid point using a golden‑section search
    # (the objective is smooth enough for this 1‑D optimisation)
    phi = (1 + np.sqrt(5.0)) / 2.0
    invphi = 1.0 / phi
    # search interval = ±grid step around the current best
    step = p_grid[1] - p_grid[0]
    lo = max(0.01, best_p - 2 * step)
    hi = min(2.0, best_p + 2 * step)

    def score_of(p_):
        a_ = generate_shape(p_, n)
        return compute_score(a_, n, fft_len)

    # golden‑section loop (stop when interval is tiny or time runs out)
    c = hi - (hi - lo) * invphi
    d = lo + (hi - lo) * invphi
    fc = score_of(c)
    fd = score_of(d)

    while time.time() < deadline - 2.0 and (hi - lo) > 1e-6:
        if fc < fd:
            hi, fd = d, fc
            d = c
            c = hi - (hi - lo) * invphi
            fc = score_of(c)
        else:
            lo, fc = c, fd
            c = d
            d = lo + (hi - lo) * invphi
            fd = score_of(d)

    # final check of the three candidates inside the last interval
    for p in (lo, hi, (lo + hi) / 2.0):
        a = generate_shape(p, n)
        sc = compute_score(a, n, fft_len)
        if sc < best_score:
            best_score, best_a, best_p = sc, a.copy(), p

    print(f"[init] best p={best_p:.6f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Exact sub‑gradient descent (soft‑max + Adam) – ~30 % of the budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_budget = min(remaining * 0.35, 250.0)          # at most 250 s for this stage
    if gd_budget > 5.0:
        a = best_a.copy()
        v = np.log(a + 1e-12)                         # log‑softmax parameters
        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        it = 0
        gd_deadline = time.time() + gd_budget

        while time.time() < gd_deadline:
            a = softmax(v)                            # stay on the simplex
            g, conv_max = subgradient_exact(a, n, fft_len)

            # Adam on the unconstrained v‑variables
            it += 1
            m = beta1 * m + (1.0 - beta1) * g
            v_hat = beta2 * v_hat + (1.0 - beta2) * (g ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat_corr = v_hat / (1.0 - beta2 ** it)

            # simple decreasing learning‑rate schedule (starts at 0.1)
            lr = 0.12 * (1.0 / np.sqrt(it))
            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # occasional exact evaluation for logging / best‑track
            if it % 500 == 0:
                cur_score = 2.0 * n * conv_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[gd {it:06d}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing refinement – use the rest of the time budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_score = compute_score(a, n, fft_len)
        best_a = a.copy()

        # SA temperature schedule (linear cooling)
        T_start = best_score * 0.02          # start relatively hot (2 % of score)
        T_end = best_score * 1e-7

        sa_it = 0
        sa_start = time.time()
        while time.time() < deadline:
            # linear cooling
            phase = (time.time() - sa_start) / remaining
            T = T_start * (1.0 - phase) + T_end * phase

            # propose a tiny mass transfer between two random indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move up to 8 % of a[i] (or the whole amount if a[i] is tiny)
            if a[i] <= 0.0:
                continue
            delta = rng.random() * min(a[i], 0.08 * a[i])

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact objective for the proposal
            new_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_max

            # Metropolis accept / reject
            if (new_score < best_score) or (rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12))):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} (t={time.time() - start:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    # expose the solution for possible reuse in later runs
    globals()["height_sequence_1"] = best_a.tolist()

    elapsed = time.time() - start
    print(f"[done] time={elapsed:.1f}s, final score={best_score:.6f}, n={n}, p={best_p:.6f}")
    return best_a.tolist()
