    n = len(a)

    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))

            rows.append(k)

            data.append(a[i])
    # But this loops O(n^2) per call; for n=2000 it's 4 million; not too big.
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # objective: maximize sum(g) => minimize -sum(g)

        return g
    else:
        return None

def _lp_improve(a, deadline, max_time=30.0):
    # a is normalized sum=1
    n = len(a)
    M = _conv_fft(a, a).max()
    # Build constraint matrix A of shape (2n-1, n)
    # Use sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    # Fill loop
    for j in range(n):
        i_start = 0
        i_end = n-1
        # k = i + j
        # i from 0 to n-1 => k from j to j+n-1
        k_min = j
        k_max = j + n - 1
        # Coeff for each k: a[i] where i = k - j
        # So a_slice = a[0:n]
        # We can loop k
        for k in range(k_min, k_max+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # Setup LP:
    from scipy.optimize import linprog
    c = -np.ones(n)  # maximize sum(g)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    # Solve using HiGHS
    # Option for time limit: maybe not needed
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a
    g_norm = g / sum_g
    # Mixing step: use t small (maybe 0.1)
    t = 0.2
    a_new = (1 - t) * a + t * g_norm
    a_new = a_new / a_new.sum()
    return a_new

"""
Low‑dimensional search for a step‑function minimizing the max auto‑convolution.
The algorithm

1. Generates many diverse seeds (uniform, geometric, power‑law, symmetric,
   sine, Gaussian, beta‑family, exponential decay, random soft‑max).
2. Optimises the smoothed maximum of the auto‑convolution using Adam
   on log‑softmax parameters.
3. Refines the result with mirror‑descent (exponentiated‑gradient) on the
   true sub‑gradient of the maximum convolution entry.
4. Performs an aggressive simulated‑annealing walk that moves a variable
   fraction of mass between random positions (including occasional swaps).
5. Optionally runs a single linear‑programming step (primal) that maximises
   the total mass under the current convolution bound; the resulting vector
   is mixed back into the current solution.
6. The whole pipeline is first executed on a modest size (n≈2000) to obtain a
   high‑quality shape, then up‑sampled to larger lengths (4000, 8000, 12000)
   and refined again.  The best sequence found before the deadline is
   returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns the full result of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / sum(seq)².
    The sequence is assumed to be non‑negative and not all zero.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

# Parametric families for seeds

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution like shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Exponential decay from the left (lam > 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a collection of diverse starting points, normalised to sum = 1.
    """
    rng = np.random.default_rng()
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # beta‑family shapes
    for a in [0.5, 1.0, 1.5, 2.0]:
        for b in [0.5, 1.0, 1.5, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay
    for lam in [0.001, 0.002, 0.005, 0.01]:
        seeds.append(_exp_decay(n, lam))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse a previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guard against empty collection (should never happen)
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

# Adam optimisation (smooth max)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed approximation to max(conv(a,a)).
    Returns the best sequence found and its (exact) objective value.
    """
    v = np.log(seed + 1e-30)  # initialise logits

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule – start moderate and sharpen slowly
    beta = 5.0
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(conv)                # safe shift for exponentials
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style refinement using the exact sub‑gradient of the true
    max of the auto‑convolution. Exponentiated‑gradient steps keep the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 1.0 / np.sqrt(it)                 # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()                           # re‑project onto simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (enhanced)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk that moves a fraction of the mass between two
    random positions, occasionally performs swaps or global perturbations,
    and accepts moves with a temperature‑controlled probability.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.08)   # larger than before for more exploration
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # 1) standard mass transfer between two positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # propose moving a variable fraction of mass at i to j
        delta = rng.random() * 0.1 * cur[i]   # up to 10 % of the source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # 2) occasional swap move (every 500 iterations)
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal‑Swap] n={n} it={it:5d} best={new_val:.6f}")

        # 3) occasional global perturbation (every 1000 iterations)
        if it % 1000 == 0:
            noise = rng.normal(scale=0.002, size=n)  # slightly stronger noise
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = (new_val < cur_val or
                      rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal‑Jump] n={n} it={it:5d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Optional linear‑programming improvement (primal)
# ----------------------------------------------------------------------


def _lp_improve(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Solve the linear program
        maximize   sum(g_j)
        subject to conv(a,g) ≤ M   (element‑wise)
                   g ≥ 0
    where M = max(conv(a,a)).  The solution g is mixed back with the
    current vector a.  If SciPy is unavailable or time is low, the function
    simply returns the input unchanged.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        return a  # fallback: no improvement possible

    n = a.shape[0]
    # compute bound M
    conv_aa = _conv_fft(a, a)
    M = conv_aa.max()

    # time check before building a (potentially large) matrix
    if time.time() > deadline - 5.0:
        return a

    # Build sparse constraint matrix A of shape (2n-1, n)
    # Row k, column j has coefficient a[k-j] whenever 0 ≤ k-j < n
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        # indices of rows where column j participates
        # coefficient a[k - j] = a[0..n-1] respectively
        # we can append vectorized
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear program: minimize -1·g  → maximize sum(g)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    # Use HiGHS solver, with a modest time limit (if supported)
    # SciPy's linprog does not expose direct time limit – we rely on external check.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a
    g_norm = g / sum_g

    # Mix the new direction back into a
    t = 0.2
    a_new = (1.0 - t) * a + t * g_norm
    a_new = a_new / a_new.sum()
    return a_new

   into the current vector.
6. The whole pipeline runs first on a modest size (≈2000) to obtain a good
   shape, then up‑samples to larger lengths (4000, 8000, 12000) and refines
   again.  The best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader:
        2·n·max(conv(seq,seq)) / sum(seq)²
    seq must be non‑negative and not all zero.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑like distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Exponential decay from the left (lam>0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of starting vectors, already normalised to sum = 1.
    """
    rng = np.random.default_rng()
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # beta‑family shapes
    for a in [0.5, 1.0, 1.5, 2.0]:
        for b in [0.5, 1.0, 1.5, 2.0]:
            seeds.append(_beta_shape(n, a, b))

    # exponential decay
    for lam in [0.001, 0.002, 0.005, 0.01]:
        seeds.append(_exp_decay(n, lam))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse a previously saved construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth approximation to max(conv(a,a)).
    Returns the best sequence discovered and its exact objective.
    """
    v = np.log(seed + 1e-30)               # log‑softmax logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 5.0                # start with moderate smoothing
    beta_mul = 1.001

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ---- smoothed‑max weighting ----
        beta = beta * beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of weighted sum w.r.t a ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- gradient w.r.t logits (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement using the true sub‑gradient of max(conv).
    Exponentiated‑gradient updates keep the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 1.0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (enhanced)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis‑Hastings walk with mass transfers, swaps and occasional
    global perturbations.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current objective
    T0 = max(1e-6, cur_val * 0.08)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ----- mass transfer between two indices -----
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.random() * 0.1 * cur[i]            # up to 10 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # ----- occasional swap move (every 500 iters) -----
        if it % 500 == 0:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal‑Swap] n={n} it={it:5d} best={new_val:.6f}")

        # ----- global Gaussian perturbation (every 1000 iters) -----
        if it % 1000 == 0:
            noise = rng.normal(scale=0.002, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = (new_val < cur_val or
                      rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal‑Jump] n={n} it={it:5d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (primal)
# ----------------------------------------------------------------------


def _lp_improve(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    One‑step LP improvement:
        maximise   Σ g_j
        s.t.      conv(a,g) ≤ M   (element‑wise),   g ≥ 0,
    where M = max(conv(a,a)).
    The result is mixed back into a (20 % weight).  If SciPy is missing
    or time is scarce the function returns the input unchanged.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        return a

    n = a.shape[0]
    conv_aa = _conv_fft(a, a)
    M = conv_aa.max()

    # abort if little time left
    if time.time() > deadline - 5.0:
        return a

    # Build sparse constraint matrix A (2n‑1 rows, n cols)
    rows = []
    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise sum(g)  → minimise –sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return a
    g_norm = g / s

    # convex mixing back into a
    t = 0.2
    a_new = (1.0 - t) * a + t * g_norm
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Base optimisation on a modest length (fast, high‑dimensional)
    # ------------------------------------------------------------------
    base_n = 2000
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    print(f"\n=== Base optimisation (n={base_n}) ===")
    seeds = _make_seeds(base_n, previous)

    # quick evaluation of seeds; keep the most promising
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_vals[:min(7, len(seed_vals))]]

    global_best_seq = None
    global_best_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, base_n, deadline, max_iters=12000)
        if seq is None:
            continue
        seq, val = _subgradient_refine(seq, base_n, deadline, max_iters=40000)
        seq, val = _anneal_optimize(seq, base_n, deadline)

        # optional LP tweak (only if time permits)
        if time.time() < deadline - 5.0:
            seq = _lp_improve(seq, deadline)
            val = _objective(seq)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Base] seed={idx} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to larger lengths and polish further
    # ------------------------------------------------------------------
    for n in [4000, 8000, 12000]:
        if time.time() > deadline:
            break
        if global_best_seq is None:
            break
        print(f"\n=== Upscaling to length {n} ===")
        # linear interpolation of the current best shape
        small = global_best_seq
        x_small = np.linspace(0.0, 1.0, small.shape[0])
        x_big = np.linspace(0.0, 1.0, n)
        seq_big = np.interp(x_big, x_small, small)
        seq_big = np.maximum(seq_big, 0.0)
        seq_big /= seq_big.sum()

        # brief optimisation rounds
        seq_big, val = _adam_optimize(seq_big, n, deadline, max_iters=8000)
        if seq_big is None:
            continue
        seq_big, val = _subgradient_refine(seq_big, n, deadline, max_iters=20000)
        seq_big, val = _anneal_optimize(seq_big, n, deadline)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq_big.copy()
            print(f"[Upscale] n={n} improved global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(2000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # expose the best sequence for possible reuse
    globals()["height_sequence_1"] = global_best_seq.copy()
    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
