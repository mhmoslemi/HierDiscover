"""
Low‑autocorrelation step‑function search.

The algorithm works in four time‑aware stages for each tested length *n*:

1. **Seed generation** – a rich collection of deterministic families
   (uniform, geometric, power‑law, mixed, symmetric, sine‑window,
   Gaussian) together with a handful of random Dirichlet vectors.
   All seeds are normalised to sum = 1.

2. **Adam on a smoothed max loss** – the sequence is parametrised by
   log‑softmax logits.  The loss is a log‑sum‑exp approximation of
   `max(conv(a,a))`.  The smoothing factor is increased slowly; Adam
   updates keep the vector on the simplex.  This phase runs many
   thousands of iterations per seed.

3. **Multi‑peak sub‑gradient mirror‑descent** – from the best Adam
   result we repeatedly move mass away from the *K* largest
   autocorrelation entries (K=5 by default).  The update is an
   exponentiated‑gradient step that automatically respects non‑negativity
   and the simplex constraint.

4. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk that
   moves a tiny fraction of probability mass between two random
   positions; occasional Gaussian “jumps’’ help escape shallow plateaus.

All stages stop as soon as the global deadline (`budget_s` seconds) is
reached.  The globally best sequence found (normalised to sum = 1) is
returned as a plain list of floats – the format expected by the grader.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns the full result of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / sum(seq)²  (sum is assumed positive)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric families used as seeds
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    """Centered Gaussian window; sigma expressed as a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a collection of diverse starting points, already normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # a few random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # random Dirichlet‑like vectors (different concentration)
    for alpha in (0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(3):
            vec = rng.gamma(alpha, 1.0, size=n)
            if vec.sum() > 0:
                seeds.append(vec / vec.sum())

    # reuse a previously discovered construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed‑max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp approximation of max(conv).
    Returns the best sequence discovered and its exact objective value.
    """
    # initialise logits; add a small epsilon for safety
    v = np.log(seed + 1e-30)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule
    beta = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        shift = np.max(conv)                # safe shift for exponentials
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient w.r.t logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient refinement (mirror‑descent)
# ----------------------------------------------------------------------


def _subgradient_refine_multi(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 80000,
                              topk: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style refinement using the exact sub‑gradient of the
    *topk* largest autocorrelation entries.  This reduces all large peaks
    simultaneously rather than only the single maximum.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)

        # pick the top‑k indices (ties broken arbitrarily)
        if topk >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # argpartition for efficiency
            part = np.argpartition(-conv, topk - 1)
            top_idx = part[:topk]

        # weight vector on the convolution domain (length 2n‑1)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[top_idx] = 1.0 / topk

        # gradient of the weighted sum of conv entries w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)                    # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)                    # keep values moderate
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass between
    two random positions and occasionally adds a small Gaussian perturbation.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995                     # exponential cooling

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a tiny fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional larger “jumps’’ to increase exploration
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we never exceed the budget
    deadline = start_time + budget_s - 1.0

    # reuse a previously computed construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to explore – start small, increase if time permits
    lengths = [4000, 6000, 8000, 10000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) generate seeds
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # 2) quick evaluation of all seeds, keep the most promising few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on the smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=20000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} val={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Multi‑peak sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine_multi(global_best_seq, n, deadline,
                                                max_iters=80000, topk=5)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} val={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
