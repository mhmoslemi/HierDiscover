"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Build a modest set of analytic families (power‑law, Gaussian, triangular,
   uniform) and keep the best few.
2. Generate several random Dirichlet starts.
3. For each start run a continuation optimisation on the smooth
   log‑sum‑exp approximation of the max‑convolution.
   * Optimiser = Adam (no external dependencies) on the
     log‑softmax parameters x (a_i = softmax(x)_i).
   * λ (smoothness) schedule: [0.5, 2, 10, 30, 100, 300, 1000, 3000].
   * Up to a few thousand Adam steps per λ (time‑budget aware).
   * After each step compute the exact max‑convolution and keep the best
     sequence found.
4. Refine the best sequence with a mirror‑descent sub‑gradient method
   directly on the true L∞‑objective (sub‑gradient = 2·shifted a).
5. Return the best sequence (normalised to sum = 1).  All operations respect
   the 1000 s budget and print progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:                           # pragma: no cover
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def gaussian_candidate(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the indices, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def triangular_candidate(n: int, mode: str) -> np.ndarray:
    """
    Triangular shape, normalised.
    mode = 'increasing' : a_i ∝ i+1
    mode = 'decreasing' : a_i ∝ (n-i)
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "increasing":
        a = i + 1.0
    elif mode == "decreasing":
        a = n - i
    else:                           # pragma: no cover
        raise ValueError("mode must be 'increasing' or 'decreasing'")
    a = a / a.sum()
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth approximation of max conv(a,a) and its gradient w.r.t. x,
    where a = softmax(x).
    Returns (phi, grad_x, max_c) where max_c is the true maximum of the
    convolution (used for exact evaluation).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max: Φ_λ = max_c + log Σ exp(λ·(conv - max_c)) / λ
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                # weights, sum to 1
    w_rev = w[::-1]

    # gradient dΦ/da = 2 * (w_rev ∗ a) reversed (as derived in the description)
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]        # length n
    grad_a = 2.0 * slice_part[::-1]               # ∂Φ/∂a

    # map gradient to x‑space via soft‑max Jacobian
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)                    # ∇_x Φ

    return phi, grad_x, max_c


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55):
            for off in (0.0, 0.2, 0.5, 1.0):
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                candidates.append(analytic_candidate(n, p, off_eff, mode))

    # Gaussian with a few sigma values
    for sigma in (n / 10.0, n / 6.0, n / 4.0):
        candidates.append(gaussian_candidate(n, sigma))

    # triangular shapes
    candidates.append(triangular_candidate(n, "increasing"))
    candidates.append(triangular_candidate(n, "decreasing"))

    # uniform
    candidates.append(np.ones(n) / n)

    # include the globally‑provided construct if it matches the length
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            total = ref.sum()
            if total > 0:
                candidates.append(ref / total)

    # ------------------------------------------------------------------
    # 2) Evaluate analytics and pick the best few as seeds
    # ------------------------------------------------------------------
    best_score = np.inf
    best_a = None
    for a in candidates:
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    print(f"[init] best analytic score = {best_score:.9f}")

    # keep the top‑3 analytic starts
    scores_analytic = [(compute_score(a), a) for a in candidates]
    scores_analytic.sort(key=lambda t: t[0])
    start_vectors = [pair[1] for pair in scores_analytic[:3]]

    # ------------------------------------------------------------------
    # 3) Random Dirichlet starts (both sparse and dense)
    # ------------------------------------------------------------------
    for alpha in (0.5, 2.0):
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, alpha))
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
            start_vectors.append(a)

    # ------------------------------------------------------------------
    # 4) Continuation Adam optimisation on smooth max‑convolution
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 2.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    eps = 1e-12
    alpha_adam = 0.05          # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8

    for idx_start, a_start in enumerate(start_vectors):
        if time.time() >= deadline:
            break
        # initialise log‑softmax parameters
        x = np.log(np.maximum(a_start, eps))

        print(f"[seed {idx_start+1}/{len(start_vectors)}] start score = {compute_score(softmax(x)):.9f}")

        for lam in lam_schedule:
            if time.time() >= deadline:
                break

            # Adam state
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t = 0

            # decide an iteration budget for this λ (time‑aware)
            # aim for up to 4000 iterations but never use more than remaining/len(lam_schedule)
            remaining = max(0.0, deadline - time.time())
            max_iter = min(4000, int(remaining * 300.0 / len(lam_schedule)))
            max_iter = max(200, max_iter)  # ensure some progress

            for it in range(max_iter):
                if time.time() >= deadline:
                    break
                t += 1
                phi, grad_x, max_c = compute_phi_and_grad(x, lam, n)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = alpha_adam * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step

                # Evaluate exact objective every 25 steps
                if (it % 25) == 0 or it == max_iter - 1:
                    a = softmax(x)
                    cur_score = compute_score(a)
                    if cur_score < best_score:
                        best_score, best_a = cur_score, a.copy()
                        elapsed = time.time() - start_time
                        print(f"[seed {idx_start+1} lam={lam:.0f} it={it} ] new best = {best_score:.9f} (t={elapsed:.1f}s)")

            # after finishing a λ, re‑initialise Adam moments for the next λ
            # (helps stability when λ changes dramatically)

    # ------------------------------------------------------------------
    # 5) Mirror‑descent sub‑gradient refinement on the true L∞‑objective
    # ------------------------------------------------------------------
    if best_a is None:                 # pragma: no cover
        best_a = np.ones(n) / n
        best_score = compute_score(best_a)

    a = best_a.copy()
    eta0 = 0.5                         # initial step size for mirror descent

    sub_iter = 0
    while time.time() < deadline:
        sub_iter += 1
        conv = conv_fft_self(a)
        max_c = np.max(conv)
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            elapsed = time.time() - start_time
            print(f"[subgrad] it={sub_iter} new best = {best_score:.9f} (t={elapsed:.1f}s)")

        # sub‑gradient of the max‑convolution entry
        k = np.argmax(conv)                     # index where the max occurs
        idx = np.arange(n)
        src = k - idx
        mask = (src >= 0) & (src < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[src[mask]]

        # Mirror step (exponential weights) on the simplex
        eta = eta0 / math.sqrt(sub_iter)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:                # numerical safeguard
            a = np.ones_like(a) / n
        else:
            a = a / s

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.9f}, total time = {total_time:.1f}s")
    # assure a non‑negative list; the evaluator will clip anyway
    return best_a.tolist()
