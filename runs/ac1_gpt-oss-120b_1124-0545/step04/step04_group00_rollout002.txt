def _lbfgs_optimize(seed: np.ndarray, n: int, deadline: float,
                    betas: list[float] = [2.0, 5.0, 15.0, 40.0],
                    maxiter_per_beta: int = 1000) -> tuple[np.ndarray,float]:
    v = np.log(seed + 1e-30)  # logits
    best_seq = None
    best_val = float('inf')
    # define callback class
    class TimeOutException(RuntimeError):
        pass

    def callback(xk):
        if time.time() > deadline:
            raise TimeOutException()
        return False

    for beta in betas:
        # define objective
        def fun_grad(v_vec):
            a = _softmax(v_vec)
            conv = _conv_fft(a, a)
            # compute loss
            shift = conv.max()
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            loss = shift + (1.0 / beta) * np.log(sum_exp)
            # weight w
            w = exp_term / sum_exp
            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient wrt v (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return loss, grad_v

        # run optimization
        import scipy.optimize
        try:
            res = scipy.optimize.minimize(fun_grad, v, method='L-BFGS-B',
                                          jac=True,
                                          callback=callback,
                                          options={'maxiter': maxiter_per_beta,
                                                   'disp': False})
        except TimeOutException:
            break

        # update v to result.x
        v = res.x
        # evaluate actual unsmoothed objective
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L-BFGS] n={n} beta={beta} best={cur_val:.6f}")
        # continue to higher beta
    return best_seq, best_val

"""
Improved step‑function search.

The algorithm explores two lengths (8000 and 4000).  For each length it
   * builds diverse seeds (geometric, power‑law, Gaussian, previous
     construction, etc.),
   * runs a few stages of L‑BFGS‑B optimisation on a smooth approximation
     of `max(conv(a,a))` (the smoothness parameter β is increased
     gradually, the variable is the log‑softmax logits so the simplex
     constraint is satisfied automatically),
   * refines the best result with a mirror‑descent (exponentiated‑gradient)
     sub‑gradient step for the true max,
   * finishes with a temperature‑controlled simulated‑annealing walk.

All intermediate vectors are projected onto the simplex and symmetrised
(a_i = a_{n‑1‑i}) – symmetry is known to reduce the peak of the
auto‑convolution.  The routine respects the given time budget and keeps
the best sequence found.  The final sequence is stored in the global
variable ``height_sequence_1`` for possible reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    return np.fft.irfft(fa * fb, n=N)[:L]


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Simple objective (exact, same as evaluate_sequence up to clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / sum(seq)².
    The sequence must be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}, clip negatives and renormalise."""
    seq = (seq + seq[::-1]) / 2.0
    seq = np.clip(seq, 0.0, None)
    total = seq.sum()
    if total > 0:
        seq /= total
    return seq


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a modest list of diverse normalised seeds."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # random vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if dimensions match
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))

    return seeds

# L‑BFGS optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _lbfgs_optimize(seed: np.ndarray,
                    n: int,
                    deadline: float,
                    betas: list[float] = [2.0, 5.0, 15.0, 40.0],
                    maxiter_per_beta: int = 800) -> tuple[np.ndarray, float]:
    """
    Optimise the smoothed maximum of the auto‑convolution using L‑BFGS‑B.
    The variable is the log‑softmax logits to keep the sequence on the simplex.
    The smoothing parameter β is increased gradually.
    Returns the best (exact) objective value and the associated sequence.
    """
    # Safety: SciPy may be unavailable – fall back to a simple pass‑through.
    try:
        import scipy.optimize
    except Exception:  # pragma: no cover
        return None, np.inf

    v = np.log(seed + 1e-30)          # initialise logits

    best_seq = None
    best_val = np.inf

    class _TimeOut(RuntimeError):
        pass

    def _callback(_xk):
        if time.time() > deadline:
            raise _TimeOut()
        return False

    for beta in betas:
        # if we have already run out of time, abort the outer loop
        if time.time() > deadline:
            break

        def fun_and_grad(v_vec):
            """Return (loss, grad) for the current β."""
            a = _softmax(v_vec)                     # stays on simplex
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp (numerically stable)
            shift = conv.max()
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            loss = shift + (1.0 / beta) * np.log(sum_exp)

            # weights for the gradient (softmax over β·conv)
            w = exp_term / sum_exp

            # ∂L/∂a = 2 * (w ⋆ rev(a))
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # ∂L/∂v via the softmax Jacobian
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return loss, grad_v

        try:
            res = scipy.optimize.minimize(fun_and_grad,
                                          v,
                                          method='L-BFGS-B',
                                          jac=True,
                                          callback=_callback,
                                          options={'maxiter': maxiter_per_beta,
                                                   'disp': False})
        except _TimeOut:
            break

        # Update v, evaluate the true (unsmoothed) objective
        v = res.x
        a = _softmax(v)
        a = _symmetrize(a)               # enforce symmetry after each stage
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] n={n} β={beta:.1f} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement using the exact sub‑gradient of the true max.
    The update is an exponentiated‑gradient step (automatically respects the simplex).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)                # a slightly larger schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        a = _symmetrize(a)                    # enforce symmetry during descent

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that nudges a tiny amount of mass between
    two random positions.  Temperature decays exponentially.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a small fraction of the mass at i to j
        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of the source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_seq = _symmetrize(new_seq)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point expected by the grader

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # Retrieve a previously found construction, if any.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.ndim != 1:
                previous = None
        except Exception:
            previous = None

    # We first try the larger length; if time runs out we fall back to 4000.
    lengths = [8000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Seed construction (including an up‑sampled previous solution)
        # ------------------------------------------------------------------
        seeds = []
        if previous is not None and previous.shape[0] == n:
            seeds.append(previous / previous.sum())
        elif previous is not None and n % previous.shape[0] == 0:
            factor = n // previous.shape[0]
            up = np.repeat(previous, factor) / factor
            up = up / up.sum()
            seeds.append(up)

        # Fill the rest with the generic diverse set
        seeds.extend(_make_seeds(n, None))

        # ------------------------------------------------------------------
        # 2) Quick evaluation of raw seeds; keep the most promising few
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(5, len(seed_vals))]]

        # keep the absolute best raw seed
        if seed_vals and seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) L‑BFGS optimisation on smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _lbfgs_optimize(seed_vec, n, deadline)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall L‑BFGS] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (global_best_seq is not None) and (global_best_seq.shape[0] == n):
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=60000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if (global_best_seq is not None) and (global_best_seq.shape[0] == n):
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Store for potential reuse in a later call.
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
