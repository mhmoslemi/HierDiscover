"""
Propose a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Build a modest grid of analytic families (power‑law, product/sum modes)
   and keep the best (exact objective, not the smoothed proxy).

2. Refine the best start with a continuation of the smoothed‑max
   objective Φ_λ(a) = (1/λ)·log Σ exp(λ·(a*a)_k) using L‑BFGS‑B on the
   soft‑max parametrisation.  The λ‑schedule ends at a very large value
   (8000) so the optimiser sees the true max.

3. Perform a dedicated *sub‑gradient* descent on the exact objective:
   * compute the convolution, locate the index k where it attains its
     maximum;
   * the (sub‑)gradient of max(conv) w.r.t. a_i is 2·a_{k‑i} (zero outside
     the support).  Moving probability mass from indices with large
     gradient to indices with small gradient reduces the objective.
   * Transfer a fraction of the source mass, accept the move only if the
     true objective improves; otherwise shrink the step size.
   * Occasionally (1 % of iterations) perform a random mass‑transfer
     to escape plateaus.

4. With any remaining time run a short random‑move local search (the
   routine used in the previous state‑of‑the‑art) as a final polishing
   step.

The whole routine respects the 1000 s budget, prints progress, and
returns the best sequence found (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex (sum=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # projection onto the simplex is unnecessary – we keep the sum=1
    # and avoid negative values by construction.
    return a_new


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smoothed max Φ_λ and its gradient w.r.t. the soft‑max parameters x.
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # smooth max
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                  # weights, sum to 1

    # gradient of Φ w.r.t. a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]  # length n
    grad_a = 2.0 * slice_part[::-1]        # ∂Φ/∂a

    # map gradient to x‑space using soft‑max Jacobian
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return phi, grad_x

    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))          # length of the step function
    # ------------------------------------------------------------------
    # 1) Analytic grid search – yields a solid deterministic start
    # ------------------------------------------------------------------
    candidates = []

    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55):
            for off in (0.0, 0.2, 0.5, 1.0):
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, p, off_eff, mode)
                candidates.append(cand)

    candidates.append(np.ones(n) / n)  # uniform baseline

    # Include the globally‑provided construction if its length matches
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max continuation with L‑BFGS‑B (or Adam fallback)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0, 8000.0]

    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:                 # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – will use a simple Adam optimizer.")

    for lam in lam_vals:
        if time.time() >= deadline:
            break

        if have_scipy:
            # Number of iterations adapts to the remaining time budget
            remaining = deadline - time.time()
            max_iter = int(min(400, max(20, remaining * 70)))
            res = minimize(
                compute_phi_and_grad,
                x,
                args=(lam, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'disp': False}
            )
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.0f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # Simple Adam optimiser (fallback)
            alpha = 0.05
            beta1, beta2 = 0.9, 0.999
            eps_opt = 1e-8
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t = 0
            while time.time() < deadline:
                t += 1
                phi, grad_x = compute_phi_and_grad(x, lam, n)
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam lam={lam:.0f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t >= 3000:
                    break

    # ------------------------------------------------------------------
    # 3) Sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()               # keep the best vector from stage 2
    step_frac = 0.12                # start moving about 12 % of the source mass
    min_step = 1e-5
    max_step = 0.5
    no_improve_cnt = 0
    while time.time() < deadline:
        conv = conv_fft_self(a)
        max_val = np.max(conv)
        k = int(np.argmax(conv))    # index where the max is attained

        # Build the gradient w.r.t. a (sub‑gradient of max convolution)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad_vec = np.zeros_like(a)
        grad_vec[mask] = a[j[mask]]          #  a_{k‑i} for feasible i

        # Select source (largest gradient) and destination (smallest)
        src = int(np.argmax(grad_vec))
        dst = int(np.argmin(grad_vec))

        # Occasionally perform a pure random move to escape flats
        if np.random.rand() < 0.01:
            cand = random_move(a, scale=step_frac)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                a = cand
                print(f"[rand] new best = {best_score:.6f} (t={time.time()-start_time:.1f}s)")
                continue
            else:
                # no improvement – continue with directed move
                pass

        if src == dst:
            # nothing to move, break out
            break

        # Amount to transfer
        delta = step_frac * a[src]
        if delta < 1e-12:
            break

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_score = compute_score(a_new)

        if new_score < best_score - 1e-12:
            # Accept improvement
            a = a_new
            best_score, best_a = new_score, a.copy()
            no_improve_cnt = 0
            step_frac = min(step_frac * 1.05, max_step)
            elapsed = time.time() - start_time
            print(f"[sgd] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # Reject – shrink step size
            no_improve_cnt += 1
            step_frac = max(step_frac * 0.7, min_step)
            if no_improve_cnt >= 200:
                # stagnation – perform a tiny random perturbation
                a = random_move(a, scale=step_frac * 0.5)
                no_improve_cnt = 0

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search (final polishing)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            scale = max(0.001, scale * 0.9)
        else:
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # Return as plain Python list (evaluation harness expects a list)
    return best_a.tolist()
