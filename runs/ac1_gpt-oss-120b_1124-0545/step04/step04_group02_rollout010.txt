"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1) Analytic exploration – a modest grid over several classic families
   (product, sum‑power, exponential, cosine‑power, beta) and a short
   random sweep of the same families with wider parameter ranges.  The
   best normalised vector found is kept as a warm‑start.

2) Mirror‑descent on the probability simplex – we store the log‑weights
   `theta` (so that a = softmax(theta)).  In each iteration we
   * evaluate the convolution (FFT, O(n log n)),
   * compute the sub‑gradient of the active max‑convolution term,
   * transform it to a gradient on `theta` (∇θ = a·(g‑⟨a,g⟩)),
   * apply an Adam update (momentum + adaptive steps) and
   * let the soft‑max automatically enforce non‑negativity and Σa=1.
   A learning‑rate schedule monotonically decays from 3·10⁻¹ to 10⁻⁶.
   Stagnation is broken by a tiny random jitter on `theta`.

3) Final stochastic refinement – for the last few seconds a
   small Gaussian perturbation around the best distribution is tried
   many times; any improvement is kept.

All helper functions are top‑level (no closures) and the routine
respectfully stops before the given wall‑time budget.  The returned
list is a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector ``a``.
    The vector is assumed to be on the simplex (sum = 1); a tiny
    normalisation is performed for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)  # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}  (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(pi * i / (n‑1))|^{p} (ends peaked)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α} * (n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


def analytic_combined(n: int, alpha: float, beta: float, lam: float) -> np.ndarray:
    """product of Beta‑shape and exponential tail."""
    a = analytic_beta(n, alpha, beta) * analytic_exponential(n, lam)
    a = a / a.sum()
    return a

    v = np.zeros_like(theta)          # second moment
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 3e-1, 1e-6    # learning‑rate schedule
    iteration = 0
    last_improve = time.time()

    # Helper for sub‑gradient of the active max convolution term
    def subgrad_max(a_vec: np.ndarray, max_k: int) -> np.ndarray:
        """Sub‑gradient of max_k (a*a)_k (single active index)."""
        n_loc = a_vec.shape[0]
        rev_idx = max_k - np.arange(n_loc)
        mask = (rev_idx >= 0) & (rev_idx < n_loc)
        g_loc = np.zeros_like(a_vec)
        g_loc[mask] = 2.0 * a_vec[rev_idx[mask]]
        return g_loc

    while time.time() < deadline - 5.0:      # reserve ~5 s for final refinement
        # recover probability vector from theta
        exp_theta = np.exp(theta - np.max(theta))
        a = exp_theta / exp_theta.sum()       # guarantees simplex

        # convolution and current objective value
        conv = conv_fft(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:07d}] new best = {best_score:.6f}")

        # sub‑gradient of the active max term
        g = subgrad_max(a, max_idx)

        # transform to gradient on theta (mirror‑descent on simplex)
        mu = np.dot(a, g)                 # scalar ⟨a,g⟩
        grad_theta = a * (g - mu)         # ∇θ = diag(a)·(g‑⟨a,g⟩·1)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)

        iteration += 1
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate schedule (linear decay)
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 5.0))
        lr = lr_start * (1.0 - progress) + lr_end * progress

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        theta -= step

        # occasional tiny random jitter to escape plateaus
        if np.random.rand() < 0.001:
            theta += np.random.randn(n) * 1e-3

        # stagnation handling: if no improvement for ~30 s, inject noise
        if time.time() - last_improve > 30.0:
            theta += np.random.randn(n) * 5e-3
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

    # ------------------------------------------------------------------
    # 4) Fine‑grained stochastic refinement (last few seconds)
    # ------------------------------------------------------------------
    refine_deadline = start_time + budget_s - 0.2
    while time.time() < refine_deadline:
        # Gaussian perturbation around the best vector (perturb theta)
        theta_pert = theta + np.random.normal(scale=5e-4, size=n)
        exp_theta = np.exp(theta_pert - np.max(theta_pert))
        a_pert = exp_theta / exp_theta.sum()
        conv = conv_fft(a_pert)
        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_pert.copy()
            theta = theta_pert                 # keep the improved direction
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    # Return a plain Python list (the harness expects list[float]).
    return best_seq.tolist()
