def project_to_simplex(v):
    """Project a vector onto the probability simplex (sum=1, nonnegative)."""
    # Using algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    # find rho = max i s.t. u_i - cssv_i/(i+1) > 0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries zero, return uniform?
        # but sum=1 required; set uniform.
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_max_conv(a):

"""
Search for a non‑negative step‑function (probability vector) that minimizes

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in two phases:

1.  Projected sub‑gradient descent on the exact nonsmooth objective.
    The sub‑gradient of   max(conv)   is obtained by putting equal weight
    on all indices where the convolution attains its maximum.  After each
    gradient step the iterate is projected back onto the probability simplex
    (clip‑to‑zero and renormalise).

2.  A short simulated‑annealing “hill‑climb” that perturbes the best
    solution found by moving a small amount of mass between two random
    coordinates.  This helps to escape shallow local minima.

The routine respects the supplied time budget (default 1000 s) and returns
the best coefficient list discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex Σx=1, x≥0.
    This cheap version clips negative entries to zero and renormalises.
    It always yields a feasible point (though not the Euclidean projection).
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _max_conv(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return max(conv(a,a)) using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(conv.max())


def _grad_and_max(a: np.ndarray, n: int, fft_len: int):
    """
    Compute the sub‑gradient of 2·n·max(conv(a,a)) and the current maximum
    convolution value.
    """
    # FFT of a
    fa = np.fft.rfft(a, fft_len)

    # linear convolution a*a  (length 2n‑1)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_val = float(conv.max())

    # mask of the (possibly multiple) maximal entries
    eps = max_val * 1e-12 + 1e-14
    mask = (conv >= max_val - eps).astype(float)
    cnt = mask.sum()
    if cnt == 0.0:  # should not happen, but be safe
        mask = np.zeros_like(conv)
        mask[conv.argmax()] = 1.0
        cnt = 1.0
    w = mask / cnt                     # weight vector – a probability distribution

    # FFT of the mask
    fw = np.fft.rfft(w, fft_len)

    # linear convolution w * a   (length 3n‑2)
    conv_wa = np.fft.irfft(fw * fa, fft_len)[: 3 * n - 2]
    g2 = conv_wa[:n]

    # linear cross‑correlation w ⋆ a (length 3n‑2)
    corr = np.fft.irfft(np.conj(fw) * fa, fft_len)[: 3 * n - 2]
    g1 = corr[:n]

    # sub‑gradient of 2·n·max(conv)
    grad = 2.0 * n * (g1 + g2)

    return grad, max_val


def _subgradient_optimize(a0: np.ndarray, n: int, time_budget: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for at most `time_budget` seconds.
    Returns the best vector found and its score (2·n·max(conv)).
    """
    start = time.time()
    deadline = start + time_budget

    # FFT length – same for all calls inside this run
    fft_len = _next_pow2(3 * n - 2)

    # current iterate – ensure it lies on the simplex
    a = _simplex_projection(a0.astype(float))

    best_a = a.copy()
    best_score = 2.0 * n * _max_conv(a, n, fft_len)

    # momentum buffer (Nesterov‑style exponential moving average)
    momentum = np.zeros_like(a)
    mu = 0.9

    it = 0
    last_improvement = start

    while time.time() < deadline:
        it += 1

        # --- sub‑gradient of the objective ---------------------------------
        grad, max_val = _grad_and_max(a, n, fft_len)

        # momentum update
        momentum = mu * momentum + (1.0 - mu) * grad

        # step‑size schedule (large at the beginning, 1/√t decay)
        lr = 0.15 / np.sqrt(it + 10)

        # normalise the step so that its Euclidean length equals `lr`
        step = lr * momentum / (np.linalg.norm(momentum) + 1e-12)

        # projected gradient step
        a = _simplex_projection(a - step)

        # current (exact) score
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improvement = time.time()
            if it % 2000 == 0:
                elapsed = time.time() - start
                print(f"[sgd {it:07d}] new best = {best_score:.6f}  (t={elapsed:.1f}s)")

        # occasional cheap restart if stagnation > 15 s
        if time.time() - last_improvement > 15.0:
            # jitter the current best and continue
            jitter = rng.normal(scale=0.005, size=n)
            a = _simplex_projection(best_a + jitter)
            momentum.fill(0.0)
            last_improvement = time.time()
            print(f"[sgd {it:07d}] restart (no improvement)")

    return best_a, best_score


def _anneal_refine(a_start: np.ndarray, n: int, time_budget: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing refinement around `a_start`.
    Only the objective value is needed (no gradient).
    """
    start = time.time()
    deadline = start + time_budget

    fft_len = _next_pow2(3 * n - 2)

    a = a_start.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _max_conv(a, n, fft_len)

    # initial temperature – a small fraction of the current score
    T0 = best_score * 0.01
    it = 0

    while time.time() < deadline:
        it += 1
        # linear cooling
        T = T0 * (1.0 - (time.time() - start) / time_budget)

        # propose a local move: shift a small amount of mass from i to j
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.random() * min(a[i], 0.005)   # move at most 0.5 % of the total mass
        if delta == 0.0:
            continue
        b = a.copy()
        b[i] -= delta
        b[j] += delta

        # evaluate
        cur_score = 2.0 * n * _max_conv(b, n, fft_len)

        # Metropolis acceptance
        if cur_score < best_score:
            a = b
            best_a = b
            best_score = cur_score
            print(f"[anneal {it}] improvement -> {best_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) / (T + 1e-12))
            if rng.random() < prob:
                a = b

    return best_a, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 2000,
                     **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Searches for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best coefficient list found.
    """
    # reproducibility
    rng_global = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Several independent sub‑gradient runs (different initialisations)
    # ------------------------------------------------------------------
    attempts = 10                                 # number of independent starts
    global_best_a = None
    global_best_score = float("inf")

    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        # split the remaining budget roughly equally among the remaining attempts
        remaining = attempts - attempt
        alloc = time_left * 0.85 / remaining

        # ----- initialise a starting vector --------------------------------
        if attempt == 0 and "height_sequence_1" in globals():
            a0 = np.array(globals()["height_sequence_1"], dtype=float)
            if a0.shape[0] != n:
                if a0.shape[0] > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.shape[0]))
            a0 = np.maximum(a0, 0.0)
        else:
            # Dirichlet draws – concentration 0.3 gives moderately sparse starts
            alpha = np.full(n, 0.3)
            a0 = rng_global.dirichlet(alpha)

        # ----- run projected sub‑gradient descent -------------------------
        a_opt, score_opt = _subgradient_optimize(a0, n, alloc, rng_global)

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(f"[attempt {attempt}] new global best = {global_best_score:.6f}"
                  f" (elapsed {elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 2) Optional simulated‑annealing polishing on the best candidate
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0 and global_best_a is not None:
        a_refined, score_refined = _anneal_refine(global_best_a, n,
                                                  time_left, rng_global)
        if score_refined < global_best_score:
            global_best_score = score_refined
            global_best_a = a_refined.copy()
            elapsed = time.time() - start_time
            print(f"[anneal] improved global best = {global_best_score:.6f}"
                  f" (elapsed {elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Fallback – uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)

    # final sanity check (non‑negative, sums to 1)
    global_best_a = _simplex_projection(global_best_a)

    total_time = time.time() - start_time
    print(f"[done] best score = {global_best_score:.6f}, total time = {total_time:.1f}s")
    return global_best_a.tolist()
