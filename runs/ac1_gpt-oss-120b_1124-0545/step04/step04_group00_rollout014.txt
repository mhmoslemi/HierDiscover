"""
Propose a highly‑optimized step‑function.

The method works in three stages:
1. **Rich seed generation** – many parametric families (geometric,
   power‑law, sine‑window, beta‑shaped, piecewise‑constant, double‑exponential,
   linear ramps, random soft‑max).  All seeds are normalised to sum = 1.
2. **Fast smoothed‑max optimisation** – log‑softmax parametrisation of the
   sequence (automatically stays on the simplex).  The loss is a
   log‑sum‑exp approximation of `max(conv(a,a))`.  Adam updates drive the
   sequence towards a lower max‑convolution.  The temperature `beta`
   is increased linearly from 1 to a configurable `beta_max`.
3. **Exact sub‑gradient refinement + tiny annealing** – mirror‑descent
   with the true sub‑gradient of the *max* convolution, followed by a
   cheap Metropolis walk that moves a small amount of mass between two
   positions.  The whole pipeline respects the time budget (default 1000 s).

The code is deliberately time‑aware: every heavy loop checks the global
deadline and aborts early if needed.  The best sequence found (as a list
of floats) is returned.

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution, softmax)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence for valid inputs)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / sum(seq)² (seq non‑negative)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric families used as seeds (all return a normalised np.ndarray)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_pow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** p
    return a / a.sum()


def _linear_decr(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, …, 1
    return a / a.sum()


def _linear_incr(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · (n‑i)^q   (Beta‑like shape)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** p) * ((n - i + 1) ** q)
    return a / a.sum()


def _piecewise_const(n: int, alpha: float, w: float) -> np.ndarray:
    """
    Two‑block constant: first α·n entries have weight w,
    the rest have weight 1.
    """
    cut = int(alpha * n)
    a = np.empty(n, dtype=np.float64)
    a[:cut] = w
    a[cut:] = 1.0
    return a / a.sum()


def _double_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """Mixture of two exponentials."""
    i = np.arange(n, dtype=np.float64)
    a = w * (r1 ** i) + (1.0 - w) * (r2 ** i)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of normalised seeds.
    The list is deliberately large – the expensive optimisation later
    will be run only on the most promising ones.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # 1. elementary families
    seeds.append(_uniform(n))
    seeds.append(_linear_decr(n))
    seeds.append(_linear_incr(n))
    seeds.append(_sine_pow(n, 2.0))

    # 2. geometric (r = 1 - c/n)
    for c in np.linspace(0.2, 4.0, 20):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # 3. power‑law decreasing
    for p in np.linspace(0.0, 2.5, 13):
        seeds.append(_power(n, p))

    # 4. sine‑window with various exponents
    for p in np.linspace(0.5, 5.0, 10):
        seeds.append(_sine_pow(n, p))

    # 5. beta‑shaped (two‑parameter family)
    for p in [0.0, 0.5, 1.0, 2.0]:
        for q in [0.0, 0.5, 1.0, 2.0]:
            seeds.append(_beta_shape(n, p, q))

    # 6. piecewise constant blocks
    for alpha in [0.2, 0.4, 0.6, 0.8]:
        for w in [0.2, 0.5, 0.8]:
            seeds.append(_piecewise_const(n, alpha, w))

    # 7. double exponential mixtures
    r_vals = [1 - c / n for c in (0.3, 0.8, 1.5, 2.5)]
    for r1 in r_vals:
        for r2 in r_vals:
            if r1 != r2:
                for w in [0.3, 0.5, 0.7]:
                    seeds.append(_double_exp(n, r1, r2, w))

    # 8. random soft‑max vectors
    for _ in range(10):
        seeds.append(_random_softmax(n, rng))

    # 9. reuse a previously discovered construction (if same length)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Guard against empty collection – should never happen
    if not seeds:
        seeds.append(_uniform(n))

    return seeds

# Smoothed‑max optimisation (Adam on a log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000,
                   beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed maximum of the auto‑convolution.
    The sequence is represented by logits `v` (log‑softmax).  Returns the
    best ever sequence together with its *exact* (unsmoothed) objective.
    """
    v = np.log(seed + 1e-30)          # initialise logits safely

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on the simplex
        conv = _conv_fft(a, a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max (linear schedule for beta) -----
        beta = 1.0 + (beta_max - 1.0) * (it / max_iters)   # goes from 1 to beta_max
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule back to logits -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t + 1.0)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val

# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style refinement using the true sub‑gradient of
    the max of the auto‑convolution.  The update is an exponentiated‑gradient
    step that respects the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)                 # 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # keep numbers moderate
        a = np.exp(log_a)
        a /= a.sum()                           # re‑project onto simplex

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis–Hastings random walk that moves a tiny amount of mass
    between two positions.  The temperature decays exponentially.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9996                     # exponential cooling

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a tiny fraction of the mass at i to j
        delta = rng.random() * 0.015 * cur[i]   # up to 1.5 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional global perturbation (helps escape plateaus)
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _upsample(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from len(seq_small) to n_large, then renormalise."""
    old_x = np.linspace(0.0, 1.0, len(seq_small))
    new_x = np.linspace(0.0, 1.0, n_large)
    up = np.interp(new_x, old_x, seq_small)
    if up.sum() == 0.0:
        up = _uniform(n_large)
    else:
        up = up / up.sum()
    return up

    print(f"\n=== Stage 1: coarse search (n={n_small}) ===")
    seeds = _make_seeds(n_small, previous)

    # quick objective on every seed
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_vals[:min(8, len(seed_vals))]]

    best_seq_small = seed_vals[0][0]
    best_val_small = seed_vals[0][1]
    print(f"[Seed] best raw = {best_val_small:.6f}")

    # Adam optimisation on the best few seeds
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n_small, deadline,
                                  max_iters=12000, beta_max=30.0)
        if seq is not None and val < best_val_small:
            best_val_small = val
            best_seq_small = seq.copy()
            print(f"[Stage1‑Adam] seed={idx} best={val:.6f}")

    # Sub‑gradient refinement (still on the small grid)
    if best_seq_small is not None:
        seq, val = _subgradient_refine(best_seq_small, n_small, deadline,
                                       max_iters=30000)
        if seq is not None and val < best_val_small:
            best_val_small = val
            best_seq_small = seq.copy()
            print(f"[Stage1‑SubGrad] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to a larger grid and perform final polishing
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n=== Stage 2: up‑sample to n={n_large} ===")
    if best_seq_small is not None:
        seq_large = _upsample(best_seq_small, n_large)
    else:
        seq_large = _uniform(n_large)

    # One more round of Adam on the large grid (few thousand iters)
    if time.time() < deadline:
        seq_large, val_large = _adam_optimize(seq_large, n_large, deadline,
                                              max_iters=8000, beta_max=25.0)
        if val_large < best_val_small:
            best_val_small = val_large
            best_seq_small = seq_large.copy()
            print(f"[Stage2‑Adam] improved to {val_large:.6f}")

    # Exact sub‑gradient refinement on the large grid
    if time.time() < deadline:
        seq_large, val_large = _subgradient_refine(seq_large, n_large, deadline,
                                                   max_iters=50000)
        if val_large < best_val_small:
            best_val_small = val_large
            best_seq_small = seq_large.copy()
            print(f"[Stage2‑SubGrad] improved to {val_large:.6f}")

    # Light simulated‑annealing fine‑tuning
    if time.time() < deadline:
        seq_large, val_large = _anneal_optimize(seq_large, n_large, deadline,
                                                max_iters=None)
        if val_large < best_val_small:
            best_val_small = val_large
            best_seq_small = seq_large.copy()
            print(f"[Stage2‑Anneal] improved to {val_large:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    if best_seq_small is None:
        best_seq_small = _uniform(4000)
        print("[Fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val_small:.6f} "
          f"(length {len(best_seq_small)})")
    # store the result in the global name expected by the next call (optional)
    globals()["height_sequence_1"] = best_seq_small.copy()
    return best_seq_small.tolist()
