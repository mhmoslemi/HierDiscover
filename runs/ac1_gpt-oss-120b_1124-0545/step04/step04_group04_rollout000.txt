def refine_sequence(a, steps, max_step=0.02, rng):
    a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    for s in range(steps):
        i, j = rng.integers(0, len(a), size=2)
        if i == j: continue
        delta = (rng.random() - 0.5) * max_step  # maybe negative or positive
        # move mass from i to j
        if delta > 0:
            delta = min(delta, a[i])
        else:
            delta = -min(-delta, a[j])
        a[i] -= delta
        a[j] += delta
        # ensure non-neg
        # renorm to preserve sum (should be preserved exactly)
        # compute objective
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return a, best_score

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm:
  * Work on the probability simplex (sum = 1) via a soft‑max on log‑weights.
  * Use an Adam optimiser with a temperature that slowly turns a
    smooth log‑sum‑exp approximation of the max into the true hard max.
  * Gradient of the smoothed max is obtained efficiently with FFTs.
  * Several diverse initialisations (including the historic solution,
    a sinusoidal start and sparse Dirichlet draws) are tried; each
    optimisation run gets a fair share of the remaining time‑budget.
  * After the main Adam phase a cheap hill‑climbing refinement
    swaps tiny amounts of mass between two coordinates and keeps any
    improvement.

def softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    vmax = np.max(v)
    ev = np.exp(v - vmax)
    return ev / ev.sum()


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _hill_climb(
    a_start: np.ndarray,
    max_steps: int,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple coordinate‑wise hill‑climbing: move a tiny amount of mass from
    one index to another and keep the change if the exact evaluation
    improves.  Stops when the time budget expires.
    """
    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = best_a.size
    start = time.time()

    # Max amount that can be moved in one step (relative to total mass)
    max_delta = 0.02  # 2 % of the whole mass

    for step in range(max_steps):
        if time.time() - start > time_budget:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Propose moving a random fraction of max_delta from i to j
        delta = (rng.random() - 0.5) * max_delta
        # Clip to keep non‑negativity
        delta = np.clip(delta, -best_a[j], best_a[i])

        if abs(delta) < 1e-12:
            continue

        cand[i] -= delta
        cand[j] += delta
        # Renormalise to mitigate floating‑point drift
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            # occasional progress report

                print(f"[hill‑climb] step {step:06d} new best = {best_score:.6f}")
