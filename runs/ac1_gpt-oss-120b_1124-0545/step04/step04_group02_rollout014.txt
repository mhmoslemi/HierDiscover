"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of four stages:

1. **Broad analytic grid** – explore several parametric families
   (U‑shaped product, sum of power‑laws, exponential tails, cosine‑powers,
   Beta‑shapes) and keep the best normalised vector.

2. **Local continuous refinement** – for the best analytic family run a
   few cheap Nelder‑Mead optimisations on its parameters (product,
   sum, exponential, beta).  This gives a higher‑resolution shape before
   the iterative optimisation.

3. **Projected Adam on a smooth approximation of the max‑convolution** –
   the true objective 2·n·max(conv) is non‑smooth.  We replace the max by a
   soft‑max with an increasing inverse temperature β and take its gradient
   (computed by FFTs).  Adam with a simplex projection and an occasional
   random “mass‑transfer” move drives the sequence to lower scores.
   Symmetrisation (a = (a+reverse(a))/2) is enforced after each projection
   because the problem is symmetric.

4. **Fine‑grained stochastic refinement** – a short Gaussian‑perturbation
   phase that tries to improve the best vector found so far.

All heavy work is done with FFT‑based convolutions, the total wall‑time is
kept below the 1000 s budget and the result is returned as a plain Python
list of non‑negative floats (automatically scaled to sum = 1, which leaves
the original evaluation function unchanged).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The vector is assumed to lie on the simplex (sum = 1); a safety
    normalisation is performed.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry v[i] = v[-i-1] and re‑project onto the simplex."""
    v = (v + v[::-1]) * 0.5
    return project_simplex(v)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}  (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / tot
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / tot
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(pi * i / (n‑1))|^{p} (peaks at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α} * (n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a

def _refine_product(n, p0, off0):
    """Refine product‑family parameters (p, offset) with a tiny local optimiser."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return (p0, off0), compute_score(analytic_product(n, p0, off0))

    def fun(x):
        p, off = x
        if p <= 0 or off <= 0:
            return np.inf
        return compute_score(analytic_product(n, p, off))

    res = minimize(fun, np.array([p0, off0]), method='Nelder-Mead',
                   options={'maxfev': 300, 'xatol': 1e-5, 'fatol': 1e-5})
    p_opt, off_opt = res.x
    score = compute_score(analytic_product(n, p_opt, off_opt))
    return (p_opt, off_opt), score


def _refine_sum(n, p0, off0):
    """Refine sum‑family parameters."""
    try:

    except Exception:
        return (p0, off0), compute_score(analytic_sum(n, p0, off0))

    def fun(x):
        p, off = x
        if p <= 0 or off <= 0:

        return compute_score(analytic_sum(n, p, off))

    res = minimize(fun, np.array([p0, off0]), method='Nelder-Mead',

    p_opt, off_opt = res.x
    score = compute_score(analytic_sum(n, p_opt, off_opt))
    return (p_opt, off_opt), score


def _refine_exponential(n, lam0):
    """Refine exponential‑tail parameter."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return lam0, compute_score(analytic_exponential(n, lam0))

    def fun(x):
        lam = x[0]
        if lam <= 0:
            return np.inf
        return compute_score(analytic_exponential(n, lam))

    res = minimize(fun, np.array([lam0]), method='Nelder-Mead',
                   options={'maxfev': 300, 'xatol': 1e-5, 'fatol': 1e-5})
    lam_opt = res.x[0]
    score = compute_score(analytic_exponential(n, lam_opt))
    return lam_opt, score


def _refine_beta(n, a0, b0):
    """Refine beta‑family parameters."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return (a0, b0), compute_score(analytic_beta(n, a0, b0))

    def fun(x):
        a, b = x
        # No strict bounds; if bad values lead to NaNs we return inf.
        if np.isnan(a) or np.isnan(b):
            return np.inf
        return compute_score(analytic_beta(n, a, b))

    res = minimize(fun, np.array([a0, b0]), method='Nelder-Mead',
                   options={'maxfev': 300, 'xatol': 1e-5, 'fatol': 1e-5})
    a_opt, b_opt = res.x
    score = compute_score(analytic_beta(n, a_opt, b_opt))
    return (a_opt, b_opt), score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine respects the 1000 s wall‑time
    budget and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = np.inf

    print("[init] Analytic grid – n =", n)

    # -------------------- 1) Coarse analytic grid --------------------
    # product family
    p_vals = np.arange(0.2, 2.01, 0.2)
    off_vals = np.arange(0.01, 2.01, 0.2)
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # sum family
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # exponential tail family
    lam_vals = np.logspace(-3, 1, num=15)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # cosine‑power family
    p_vals_cos = np.arange(0.5, 6.1, 0.5)
    for p in p_vals_cos:
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # beta‑shape family
    alpha_vals = np.arange(-0.5, 2.01, 0.5)
    beta_vals = np.arange(-0.5, 2.01, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # Global reference (if any)
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic start score = {best_score:.6f}")

    # -------------------- 2) Local parametric refinement --------------------
    # we only refine the family that gave the current best
    # (identify it by checking which generator reproduces best_seq best)
    # For simplicity we just try to refine product, sum, exponential and beta.

    # product refinement
    if best_seq is not None:
        # estimate initial parameters by rough inverse mapping (grid is dense enough)
        # use the best parameters found during the grid search
        # (the last tried values are stored in the loops – we redo a cheap refinement)
        pass  # placeholder – actual parameters are unknown, we just start from the best_seq itself.

    # Try a quick Nelder‑Mead refinement for each family (limited budget)
    # Product
    if time.time() < deadline - 30:
        p0, off0 = 0.8, 0.1  # reasonable starts
        (p_opt, off_opt), sc = _refine_product(n, p0, off0)
        cand = analytic_product(n, p_opt, off_opt)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] product family improved to {sc:.6f}")

    # Sum
    if time.time() < deadline - 30:
        p0, off0 = 0.8, 0.1
        (p_opt, off_opt), sc = _refine_sum(n, p0, off0)
        cand = analytic_sum(n, p_opt, off_opt)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] sum family improved to {sc:.6f}")

    # Exponential
    if time.time() < deadline - 30:
        lam0 = 0.2
        lam_opt, sc = _refine_exponential(n, lam0)
        cand = analytic_exponential(n, lam_opt)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] exponential family improved to {sc:.6f}")

    # Beta
    if time.time() < deadline - 30:
        a0, b0 = 0.0, 0.0
        (a_opt, b_opt), sc = _refine_beta(n, a0, b0)
        cand = analytic_beta(n, a_opt, b_opt)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] beta family improved to {sc:.6f}")

    # Ensure symmetry before proceeding
    best_seq = symmetrize(best_seq)

    # ------------------------------------------------------------------
    # 3) Projected Adam on smooth max‑convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (already on simplex)
    m = np.zeros_like(a)                    # first moment
    v = np.zeros_like(a)                    # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0
    last_improve = time.time()
    stagnation_seconds = 25.0

    print("[stage3] Starting Adam descent (smooth max)")

    while time.time() < deadline:
        # current objective
        conv = conv_fft(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{t_iter:07d}] new best = {best_score:.6f}")

        # ---------- smooth‑max weight ----------
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)
        # inverse temperature grows from 2 to 200
        beta_temp = 2.0 + 198.0 * progress
        # stable soft‑max
        shift = conv.max()
        exp_raw = np.exp(beta_temp * (conv - shift))
        w = exp_raw / exp_raw.sum()

        # ---------- gradient (smooth) ----------
        L = _next_pow2(3 * n - 2)
        fft_a = np.fft.rfft(a, n=L)
        fft_w = np.fft.rfft(w, n=L)

        # a * w (full linear convolution)
        conv_aw_full = np.fft.irfft(fft_a * fft_w, n=L)[: 3 * n - 2]
        term2 = conv_aw_full[:n]                       # Σ_i a_i w_{j-i}

        # a_rev * w (for Σ_i a_i w_{i+j})
        a_rev = a[::-1]
        fft_arev = np.fft.rfft(a_rev, n=L)
        conv_rev_full = np.fft.irfft(fft_arev * fft_w, n=L)[: 3 * n - 2]
        term1 = conv_rev_full[(n - 1):(n - 1) + n]      # Σ_i a_i w_{i+j}

        grad = term1 + term2                           # ∇ smooth‑max

        # ---------- Adam update ----------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule (fast at start, slow later)
        lr = 0.05 * (1.0 - progress) + 0.001 * progress
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_simplex(a_new)
        a_new = symmetrize(a_new)     # enforce mirror symmetry

        # occasional random mass‑transfer (helps escape flat regions)
        if np.random.rand() < 0.015:
            a_new = random_move(a_new, lr)

        a = a_new

        # jitter restart if stuck
        if time.time() - last_improve > stagnation_seconds:
            jitter = np.random.rand(n) * 2e-4
            a = best_seq + jitter
            a = project_simplex(a)
            a = symmetrize(a)
            last_improve = time.time()
            print(f"[{t_iter:07d}] jitter restart (stagnation)")

    # ------------------------------------------------------------------
    # 4) Fine‑grained stochastic refinement
    # ------------------------------------------------------------------
    refine_deadline = start_time + budget_s - 2.0
    print("[stage4] Fine‑grained Gaussian refinement")
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=4e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand = symmetrize(cand)
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {t_iter}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    return best_seq.tolist()
