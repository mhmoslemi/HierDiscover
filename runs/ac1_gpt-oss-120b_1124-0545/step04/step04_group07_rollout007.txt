"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. **Cantor‑measure start** – Generate a high‑resolution histogram of the
   middle‑third Cantor measure (2**L atoms, L≈18) on the interval [0,1]
   and normalise the histogram to total mass = 1.

2. **Projected (Adam) gradient descent on a smooth‑max of the
   self‑convolution** – Repeatedly minimise a differentiable surrogate of
   ``max(a*a)`` using a soft‑max weighting.  The gradient is
   ``‑2·(a_rev ∗ w)`` (computed with FFTs).  Adam‐style per‑coordinate
   adaptive steps are employed, followed by projection onto the simplex
   (non‑negative and sum = 1).  The temperature τ of the soft‑max
   decays exponentially, so the optimisation sharpens towards the true
   max.

3. **Active‑set LP refinement** – Every few thousand iterations an LP
   is solved that, for the current vector ``a``, finds a new distribution
   ``b`` minimising the *peak* of the convolution ``a*b`` under the same
   normalisation constraints.  Only the K largest convolution entries
   (the “active set”) are kept as constraints, making the LP cheap.
   The new vector is mixed with the old one (small mixing factor) and
   the momentum is reset.

4. **Best‑ever tracking** – The lowest objective value seen
   ``2·n·max(a*a)`` (equivalent to ``evaluate_sequence`` for sum = 1) is
   recorded and finally returned.

The whole search respects the supplied time budget (≈ 1000 s) and
requires only standard scientific libraries plus cvxpy (optional – the
algorithm falls back to pure gradient descent if cvxpy is unavailable).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution via FFT (zero‑padded to *size*)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    # keep only the valid (len(u)+len(v)-1) part
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 18) -> np.ndarray:
    """Histogram (length *n*) approximating the Cantor (1/3) measure.
    Returns a non‑negative array that sums to 1.
    """
    N = 1 << L                     # number of atoms = 2**L
    # ternary expansion via binary bits
    idx = np.arange(N, dtype=np.uint32)[:, None]            # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N,L)
    powers = 3.0 ** np.arange(1, L + 1)                     # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)               # position in [0,1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)                  # normalise → sum == 1
    return a


# ----------------------------------------------------------------------
# Optional LP refinement (active‑set)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _cvxpy_available = True
except Exception:  # pragma: no cover
    _cvxpy_available = False


def _lp_optimize_active(a: np.ndarray, active_idx: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve a small linear program:

        minimise   M
        subject to a * b <= M   (only at the indices ``active_idx``)
                   sum(b) == 1
                   b >= 0

    Returns the optimal b (renormalised to sum = 1) and the optimal M.
    If cvxpy is not available or the solver fails, ``(None, None)`` is returned.
    """
    if not _cvxpy_available:
        return None, None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable(nonneg=True)

    # Linear convolution of the *constant* a with the variable b
    conv = cp.conv(cp.Constant(a), b)  # length 2*n-1

    constraints = [cp.sum(b) == 1] + [conv[k] <= M for k in active_idx]
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-9)
    except Exception:  # pragma: no cover
        return None, None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    b_opt = np.asarray(b.value).reshape(-1)
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s > 0:
        b_opt /= s
    else:
        b_opt = np.full_like(a, 1.0 / n)

    return b_opt, float(prob.value)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The returned list is the best sequence found within *budget_s* seconds.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5     # small safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # 1️⃣ Initialise from a high‑resolution Cantor histogram
    # --------------------------------------------------------------
    try:
        # Use a previously stored candidate if it exists (fallback to Cantor).
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None and len(seq0) == n:
            a = np.asarray(seq0, dtype=float)
        else:
            raise KeyError
    except Exception:
        a = _cantor_histogram(n, L=18)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:                     # safety – re‑initialise if something went wrong
        a = _cantor_histogram(n, L=18)
        s = a.sum()
    a /= s                           # normalise to sum = 1

    # --------------------------------------------------------------
    # FFT sizes (remain constant)
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * weight

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv = _full_convolution(a, size1)
    max_b = conv.max()
    best_score = 2.0 * n * max_b          # true objective for sum=1
    best_seq = a.copy()
    print(f"[init] n={n}, initial objective={best_score:.6f}")

    # Adam / momentum parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    v = np.zeros_like(a)                  # momentum (same as earlier code)
    m_adam = np.zeros_like(a)             # first moment for Adam
    v_adam = np.zeros_like(a)             # second moment for Adam

    # Soft‑max temperature schedule
    tau_factor = 0.5               # τ = tau_factor * max_b initially
    tau_decay = 0.9999

    # LP schedule
    lp_interval = 3000             # try an LP every this many iterations
    active_K = 400                 # size of the active set for the LP
    mix_ratio = 0.18               # how much of the LP solution is mixed in

    it = 0
    last_improve = start

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----------------------------------------------------------
        # Evaluate current objective
        # ----------------------------------------------------------
        conv = _full_convolution(a, size1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----------------------------------------------------------
        # Occasionally perform a small LP refinement (active‑set)
        # ----------------------------------------------------------
        time_left = deadline - time.time()
        if (
            _cvxpy_available
            and time_left > 3.0
            and it % lp_interval == 0
            and it > 0
        ):
            # Pick the K largest entries of the convolution as the active set
            K = min(active_K, conv.size)
            active_idx = np.argpartition(-conv, K - 1)[:K]

            b_opt, M_opt = _lp_optimize_active(a, active_idx)
            if b_opt is not None:
                # Mix a little of the LP solution into the current iterate
                a = (1.0 - mix_ratio) * a + mix_ratio * b_opt
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 0:
                    a /= s
                else:
                    a = np.full_like(a, 1.0 / n)
                a_rev = a[::-1].copy()

                # Reset momentum after a discrete jump
                v = np.zeros_like(a)
                m_adam = np.zeros_like(a)
                v_adam = np.zeros_like(a)

                # Go straight to the next iteration (skip a gradient step this round)
                it += 1
                continue

        # ----------------------------------------------------------
        # Gradient‑descent step (soft‑max approximation of max)
        # ----------------------------------------------------------
        # Adaptive temperature
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # Soft‑max weights (stabilised)
        w = np.exp((conv - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # Gradient of the smooth‑max proxy
        g_full = _conv_fft(a_rev, w, size2)
        d = -2.0 * g_full[n - 1 : n - 1 + n]          # descent direction

        # Adam update (gradient = -d because d already points downhill)
        grad = -d
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m_adam / (1.0 - beta1 ** (it + 1))
        v_hat = v_adam / (1.0 - beta2 ** (it + 1))
        lr = 0.5                                          # base learning rate
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)       # Adam step

        # Projection onto the simplex (non‑negative, sum = 1)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        a_rev = a[::-1].copy()
        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best objective={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
