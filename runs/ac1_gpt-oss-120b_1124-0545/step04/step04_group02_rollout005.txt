"""
Continuated Projected Gradient Descent (cPGD)

* The vector is interpreted as a probability distribution (∑a = 1).  
  The true objective therefore reduces to  2·n·max(conv(a,a)).

* Instead of minimising the non‑smooth max directly we minimise the smooth
  ℓ_p‑norm of the convolution
        F_p(a) = ( Σ_k ( (a*a)_k )^p )^{1/p}.
  For p=2 this is a quadratic norm, for p→∞ it converges to the max.
  We start with a small p and increase it linearly (continuation) so that the
  optimisation surface becomes progressively sharper.

* The gradient of F_p can be written in closed form:
        ∂F_p/∂a_i = 2·c· Σ_k (a*a)_k^{p‑1} a_{k‑i},
  where c = ( Σ_k (a*a)_k^p )^{1/p‑1}.  The inner sum is a (reverse)
  convolution and can be evaluated with a single FFT.

* After each gradient step the iterate is projected onto the probability
  simplex (non‑negative, sum = 1).  A Metropolis acceptance rule together
  with occasional random‐mass‑transfer moves, Dirichlet restarts and a
  jitter‑restart when stagnation occurs helps to escape local minima.

* In the final few seconds a tiny Gaussian perturbation around the best
  solution is tried many times (refinement).

The routine fits into the 1000 s wall‑time budget, prints progress and
returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (zero‑padded) using FFT.
    Returns length 2·len(a)‑1 array.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary real vectors using FFT.
    Returns array of length len(x)+len(y)-1.
    """
    size = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: len(x) + len(y) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Objective helpers (sum is assumed to be 1 → denominator = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a that already satisfies
    Σ a = 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm uses a continuation of smooth ℓ_p‑norm surrogate
    minimisation, projected gradient descent, simulated‑annealing style
    acceptance, occasional random restarts and a final Gaussian refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Length of the vector – larger n gives a lower theoretical bound.
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # Initial vector – interpolate the known best if available,
    # otherwise start from uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        old = np.maximum(old, 0.0)
        old = old / old.sum()
        # linear interpolation to the new size
        x_old = np.linspace(0.0, 1.0, len(old))
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, old)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n, dtype=float)

    # ensure feasibility
    a = project_simplex(a)

    best_seq = a.copy()
    best_score = compute_score(best_seq)

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    momentum = np.zeros_like(a)
    beta = 0.9                     # momentum coefficient
    last_improve = start_time
    stagnation_limit = 30.0       # seconds without improvement → jitter
    iteration = 0

    # schedules (linear in wall‑time)
    step_start, step_end = 0.3, 1e-6
    temp_start, temp_end = 0.6, 0.0
    p_start, p_target = 2.0, 30.0   # continuation from ℓ₂ to near max

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and raw score
        conv = conv_fft(a)
        cur_score = 2.0 * n * conv.max()

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # progress‑dependent schedules
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 10.0))

        step = step_start * (1.0 - progress) + step_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress
        p = p_start + progress * (p_target - p_start)

        # ----- smooth surrogate gradient -----
        # avoid numerical noise for extremely small values
        conv = np.maximum(conv, 0.0)
        w = conv ** (p - 1.0)                     # weights ∝ conv^{p‑1}
        # cross‑correlation a ⋆ w (using reverse a)
        a_rev = a[::-1]
        g_full = conv_full_fft(a_rev, w)
        # extract the n entries that correspond to the gradient
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # ----- momentum + projection -----
        momentum = beta * momentum - step * g
        cand = a + momentum
        cand = project_simplex(cand)

        # ----- Metropolis acceptance based on the true objective -----
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand

        # ----- occasional random mass‑transfer move (exploration) -----
        if np.random.rand() < 0.02:
            a = random_move(a, scale=step * 0.5)

        # ----- rare Dirichlet restart (global exploration) -----
        if np.random.rand() < 0.004:
            rnd = np.random.gamma(1.0, 1.0, size=n)
            rnd = rnd / rnd.sum()
            rnd_score = compute_score(rnd)
            if rnd_score < best_score:
                best_score = rnd_score
                best_seq = rnd.copy()
                a = rnd.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] random restart improved to {best_score:.6f}")

        # ----- jitter restart after stagnation -----
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        iteration += 1

    # --------------------------------------------------------------
    # Fine‑grained Gaussian refinement (last ~2 s)
    # --------------------------------------------------------------
    refine_deadline = start_time + budget_s - 2.0
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    return best_seq.tolist()
