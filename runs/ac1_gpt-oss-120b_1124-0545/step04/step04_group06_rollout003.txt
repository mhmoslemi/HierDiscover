def _grad_logsumexp(a, tau):
    conv = _conv_fft(a, a)
    # stable softmax
    max_c = conv.max()
    exp_term = np.exp((conv - max_c) / tau)
    w = exp_term / exp_term.sum()
    # convolution for gradient
    rev = a[::-1]
    full = np.convolve(rev, w, mode='full')
    n = a.size
    # slice indices
    grad = 2.0 * full[n-1:n-1+n]
    return grad

"""
Propose a non‑negative step‑function (list of floats) that minimizes
``evaluate_sequence``.

Algorithm
---------
1.  Choose a few candidate lengths (≤ 10 000).  
2.  For each length generate a handful of diverse initial sequences
    (power‑law, exponential, random Dirichlet, and the historic
    ``height_sequence_1`` if it matches the length).
3.  Refine each candidate by **smooth max‑convolution minimisation**:

    *  The exact objective is 2·n·max(conv(a,a)).  Replace the max by the
       smooth ``τ·logΣexp(conv/τ)`` (log‑sum‑exp) with a temperature τ>0.
    *  The gradient of this smooth objective is

          ∇f_j = 2·∑_i a_i·w_{i+j}      with   w = softmax(conv/τ).

       This can be obtained by a single FFT‑based convolution of the
       reversed vector a_rev and the weight vector w.
    *  Perform **exponentiated‑gradient (mirror‑descent) updates** on the
       probability simplex:

          a ← a·exp(−η·∇f)   ;   a ← a / Σa

       This keeps all entries non‑negative and automatically enforces
       the sum‑to‑one constraint.
    *  Decrease the temperature τ (and the step size η) geometrically so
       that the optimisation first sees a very smooth landscape and
       gradually focuses on the true max‑convolution.
    *  Periodically evaluate the true objective (the same as
       ``evaluate_sequence`` for a normalised vector) and keep the best
       sequence seen.
    *  Simple random mass‑swaps are applied every few hundred iterations
       to escape plateaus.

4.  The best sequence found before the time budget expires is returned
    as a plain Python list.

The routine is fully self‑contained, uses only NumPy and the standard
library, respects the global time budget, and prints occasional progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using the FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    fz = fx * fy
    conv = np.fft.irfft(fz, n=N)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used inside ``evaluate_sequence`` when Σa = 1:
        2·n·max(conv(a, a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """Normalized power‑law a_i ∝ (i+1+offset)^{-α} (zeros before offset)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """Normalized exponential decay a_i ∝ r^{i‑offset} for i≥offset."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Simple decreasing linear (triangular) shape with optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _grad_logsumexp(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   τ·log Σ_k exp( (a*a)[k] / τ )   w.r.t. a.
    Returns a vector of length len(a).
    """
    n = a.size
    # full self‑convolution
    conv = _conv_fft(a, a)                      # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    exps = np.exp((conv - max_c) / tau)
    w = exps / exps.sum()                       # Σ w = 1, length 2n‑1

    # grad_j = 2·∑_i a_i·w_{i+j}
    rev = a[::-1]                               # reversed a
    full = _conv_fft(rev, w)                    # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_score_global: float,
    best_seq_global: np.ndarray,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Run the smooth log‑sum‑exp descent starting from ``a0``.
    Returns the best sequence and its score encountered before ``deadline``.
    """
    a = a0.copy()
    n = a.size
    best_seq = best_seq_global.copy()
    best_score = best_score_global

    # hyper‑parameters
    tau = 0.5           # initial temperature
    eta = 0.15          # initial step size for exponentiated update
    tau_decay = 0.9995
    eta_decay = 0.9999
    min_tau = 1e-4
    min_eta = 5e-5
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # gradient of the smooth approximation
        grad = _grad_logsumexp(a, tau)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)
        # avoid numerical underflow
        a = np.maximum(a, 1e-12)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # schedule decay
        tau = max(tau * tau_decay, min_tau)
        eta = max(eta * eta_decay, min_eta)

        # occasional random mass exchange to leave plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.2, eta * 0.5)
            if delta > 1e-12:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # periodic exact evaluation
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[descent] n={n} iter={iteration} score={best_score:.6f} "
                    f"tau={tau:.6f} eta={eta:.6f}"
                )
            if best_score < target:
                print("[descent] target reached – exiting early")
                break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # 1. Choose a few promising lengths (still cheap enough for FFT)
    # ------------------------------------------------------------------
    length_candidates = [3000, 5000, 8000]   # up to ~8k keeps FFT cheap

    # ------------------------------------------------------------------
    # 2. For each length generate a small pool of initial vectors
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() > deadline:
            break

        init_pool = []

        # (a) historic global sequence if it matches the length
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                init = np.maximum(init, 0.0)
                if init.sum() == 0.0:
                    init = np.full(n, 1.0 / n)
                else:
                    init = init / init.sum()
                init_pool.append(init)
                print(f"[init] using global height_sequence_1 of length {n}")
        except Exception:
            pass

        # (b) power‑law candidates
        for alpha in (0.35, 0.45, 0.55):
            for offset in (0, 5, 20):
                if offset >= n:
                    continue
                init_pool.append(_power_sequence(n, alpha, offset))

        # (c) exponential‑decay candidates
        for r in (0.96, 0.97, 0.98):
            for offset in (0, 5, 20):
                if offset >= n:
                    continue
                init_pool.append(_exp_sequence(n, r, offset))

        # (d) simple linear (triangular) shape
        for offset in (0, 10, 30):
            if offset >= n:
                continue
            init_pool.append(_linear_sequence(n, offset))

        # (e) a couple of random Dirichlet draws
        for _ in range(2):
            rnd = np.random.rand(n)
            rnd /= rnd.sum()
            init_pool.append(rnd)

        # ------------------------------------------------------------------
        # 3. Refine each initial vector by smooth descent
        # ------------------------------------------------------------------
        for a0 in init_pool:
            if time.time() > deadline:
                break
            # Evaluate the exact objective for the raw start (cheap)
            start_score = _objective(a0)
            if start_score < best_score:
                best_score = start_score
                best_seq = a0.copy()
                print(f"[start] n={n} raw score={best_score:.6f}")

            # Run the smoother optimisation
            best_seq, best_score = _smooth_descent(
                a0,
                deadline,
                best_score,
                best_seq,
                target=1.503,
            )

            # early exit if we already beat the target
            if best_score < 1.503:
                print("[overall] target achieved – finishing")
                print(f"[final] best_score={best_score:.6f} (n={best_seq.size})")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest uniform distribution
        n = 4000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[final] best_score={best_score:.6f} (n={best_seq.size})")
    return best_seq.tolist()
