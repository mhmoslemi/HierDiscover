    a = start_seq.copy()
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    best_a = a.copy()
    # parameters
    T = cur_val * 0.001
    decay = 0.9999
    step_factor = 0.1
    step_decay = 0.99995
    rng = np.random.default_rng()
    L = 2*n - 1

        # compute max conv index
        max_idx = int(np.argmax(conv))
        # subgradient contributions positions i that affect max:
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # choose donor weighted by contribution a[i] * a[max_idx - i]
        idxs = np.arange(i_start, i_end+1)
        contrib = a[idxs] * a[max_idx - idxs]

        else:
            probs = contrib / contrib.sum()
            donor = rng.choice(idxs, p=probs)
        # choose receiver different from donor
        receiver = rng.integers(0, n-1)
        while receiver == donor:
            receiver = rng.integers(0, n-1)
        # compute delta (move from donor to receiver)
        max_transfer = a[donor] * step_factor
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        # apply incremental conv update
        conv_new = conv.copy()

        # but after removal we should consider original a before change
        # use a_original = a (still unchanged at this point)

        conv_new[receiver:receiver+n] += 2.0 * delta * a

    # initial conv
    conv = np.convolve(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_val = cur_val
    best_seq = a.copy()
    print("[Start] initial objective", cur_val)

    decay_T = 0.99995
    step_factor = 0.2  # initial fraction of donor's mass to move
    decay_step = 0.9999

    L = 2 * n - 1

    it = 0
    while time.time() < deadline:
        it += 1
        # locate max conv index
        max_idx = int(conv.argmax())
        # compute donor distribution based on contributions to max
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        # get contributions a[i] * a[max_idx - i]
        contrib = a[idx_range] * a[max_idx - idx_range]
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            # fall back to random donor with positive mass
            donor_candidates = np.where(a > 1e-12)[0]
            if donor_candidates.size == 0:
                break
            donor = rng.choice(donor_candidates)
        else:
            probs = contrib / total_contrib
            donor = rng.choice(idx_range, p=probs)

        # choose receiver randomly (different index)
        receiver = rng.integers(0, n)
        while receiver == donor:

"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)²).

Algorithm
---------

1.  Initialise from the previously discovered construction
    (`height_sequence_1`).  If it is not present fall back to a uniform
    distribution of length 4000.

2.  Normalise the vector to have ℓ₁‑norm 1 (the objective is
    scale‑invariant).

3.  Compute the full linear convolution once (`conv = np.convolve(a, a)`).

4.  Simulated‑annealing / exponential‑gradient hill‑climbing:
    *   The current maximal convolution index `k` is identified.
    *   A “donor’’ index is sampled proportionally to the contribution
        `a[i]·a[k‑i]` (the sub‑gradient of the max‑convolution).
    *   A different “receiver’’ index is chosen uniformly.
    *   A small amount `Δ` (a fraction of the donor’s mass) is transferred.
        The convolution array is updated **in‑place** using the exact
        linearised formula
            conv ← conv – 2·Δ·shift(i) + 2·Δ·shift(j)
            + Δ²·e_{2i}+Δ²·e_{2j}–2·Δ²·e_{i+j},
        which costs O(n) per move.
    *   The new objective `2·n·max(conv)` is computed.
        Moves that improve the objective are always kept; otherwise they
        are accepted with Metropolis probability `exp(-(Δobj)/T)`.
    *   Temperature `T` and the transfer fraction decay slowly,
        allowing a long‑range exploration early on and fine‑grained
        adjustments later.

5.  The best sequence seen during the whole run is returned.

The whole routine respects the supplied time budget (default 1000 s) and
uses only NumPy – no external solvers are required.  The incremental
convolution update makes it possible to perform many‑hundred‑thousand
iterations within the budget, enough to shave the objective a few
‑10⁻³ below the previous best.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Incremental convolution utilities
# ----------------------------------------------------------------------


def _initial_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (size 2·len(a)‑1)."""
    return np.convolve(a, a)


def _apply_move(conv: np.ndarray, a: np.ndarray,
                donor: int, receiver: int, delta: float) -> None:
    """
    In‑place update of ``conv = a*a`` after moving ``delta`` mass from
    ``donor`` to ``receiver`` (using the original ``a``).
    """
    n = a.shape[0]
    L = conv.shape[0]

    # linear part: ±2·Δ·shift
    conv[donor: donor + n] -= 2.0 * delta * a
    conv[receiver: receiver + n] += 2.0 * delta * a

    # quadratic part: Δ² contributions
    two_i = donor + donor
    if two_i < L:
        conv[two_i] += delta * delta
    two_j = receiver + receiver
    if two_j < L:
        conv[two_j] += delta * delta
    ij = donor + receiver
    if ij < L:
        conv[ij] -= 2.0 * delta * delta


def _revert_move(conv: np.ndarray, a: np.ndarray,
                 donor: int, receiver: int, delta: float) -> None:
    """
    Undo the in‑place convolution update performed by ``_apply_move``.
    The vector ``a`` must be the *pre‑move* version.
    """
    n = a.shape[0]
    L = conv.shape[0]

    conv[donor: donor + n] += 2.0 * delta * a
    conv[receiver: receiver + n] -= 2.0 * delta * a

    two_i = donor + donor
    if two_i < L:
        conv[two_i] -= delta * delta
    two_j = receiver + receiver
    if two_j < L:
        conv[two_j] -= delta * delta
    ij = donor + receiver
    if ij < L:
        conv[ij] += 2.0 * delta * delta


# ----------------------------------------------------------------------
# Simulated‑annealing local optimiser
# ----------------------------------------------------------------------


def _simulated_annealing(start_a: np.ndarray,
                         deadline: float,
                         verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Perform a long‑run stochastic hill‑climb with Metropolis acceptance.
    ``start_a`` must have ℓ₁‑norm 1 and be non‑negative.
    Returns the best sequence and its objective value.
    """
    a = start_a.copy()
    n = a.shape[0]
    conv = _initial_convolution(a)
    cur_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T = cur_val * 0.005
    decay_T = 0.99995

    # initial transfer fraction (max fraction of donor's mass moved)
    step_factor = 0.2
    decay_step = 0.9999

    L = 2 * n - 1
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- identify the index of the maximal convolution entry -----
        max_idx = int(conv.argmax())
        # donor candidates are those that actually contribute to that entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        contrib = a[idx_range] * a[max_idx - idx_range]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            donor = rng.choice(idx_range, p=probs)
        else:
            # fallback: any index with positive mass
            pos = np.where(a > 1e-12)[0]
            if pos.size == 0:
                break
            donor = rng.choice(pos)

        # pick a distinct receiver uniformly at random
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        # amount to move – a fraction of the donor's current mass
        max_transfer = a[donor] * step_factor
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        # ----- apply the move on the convolution (in‑place) -----
        _apply_move(conv, a, donor, receiver, delta)

        new_max = conv.max()
        new_val = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability exp(-(Δ)/T)
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move in the probability simplex
            a[donor] -= delta
            a[receiver] += delta
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                if verbose:
                    print(f"[SA] it={it:6d} new best={best_val:.6f}")
        else:
            # revert the convolution update
            _revert_move(conv, a, donor, receiver, delta)

        # ----- schedule update -----
        T *= decay_T
        step_factor *= decay_step
        if step_factor < 1e-6:
            step_factor = 1e-6
        if it % 10000 == 0:
            # occasional gentle reheating to keep the walk alive
            T = max(T, cur_val * 0.001)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
