   random Dirichlet, etc.) plus the previously discovered best sequence.

2. **Adam optimisation on a smooth approximation of the max** – the vector
   is parametrised by soft‑max logits ``v`` (so it stays on the simplex).
   The loss is a log‑sum‑exp of the auto‑convolution, the smoothing factor
   is increased exponentially, and a standard Adam update is applied.

3. **Exact sub‑gradient refinements** – after Adam, an exponentiated‑gradient
   (mirror) descent step using the true sub‑gradient of the (non‑smooth) max
   is performed for many iterations.

4. **Simulated‑annealing fine‑tuning** – a cheap stochastic “mass‑transfer”
   neighbourhood search is run with a temperature schedule that accepts
   occasional uphill moves, allowing the walk to escape shallow local minima.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs → real output)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(seq: np.ndarray, n: int) -> float:
    """
    Return 2·n·max(conv(seq,seq)).
    The sequence is assumed to be non‑negative with sum≈1
    (the scaling cancels out in the true evaluation function).
    """
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    b = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a = a + b
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s


def _raised_cosine(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) ** power
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate of the max
    of the auto‑convolution.  Returns best sequence (sum ≈ 1) and its
    true objective value.
    """
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # initial smooth‑max temperature
    beta_mul = 1.04       # grow a bit faster than the reference code

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        shift = np.max(conv)                     # numeric stabiliser
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through the softmax -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement using the true
    sub‑gradient of the unsmoothed max of the auto‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient at the maximising index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update on the simplex
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numbers well‑scaled
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with a temperature schedule.
    Mass is transferred between a random pair of positions.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature as a small fraction of the current objective
    T0 = max(0.01, cur_val * 0.015)

    # maximum relative transfer at the start of the schedule
    max_step = 0.25

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling: fraction goes from 1 → 0
        frac = max(0.0, (deadline - time.time()) / (deadline - (deadline - 1000)))  # rough scaling
        T = T0 * frac
        step = max_step * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)

        max_transfer = cur[donor] * step
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_val = _objective(new, n)
        diff = new_val - cur_val

        if diff < 0 or rng.random() < math.exp(-diff / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA   ] n={n:5d} it={it:5d} best={new_val:.6f}")

        # occasional large random re‑initialisation to avoid dead‑ends
        if it % 20000 == 0:
            if rng.random() < 0.3:
                new = _random_dirichlet(n, conc=1.0, rng=rng)
                new_val = _objective(new, n)
                if new_val < cur_val:
                    cur, cur_val = new, new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a varied set of initial probability vectors."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (including symmetric variant)
    for c in np.linspace(0.5, 8.0, 16):           # extended range
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
            seeds.append(_sym_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta‑like shapes (dense grid)
    for a in np.linspace(0.2, 2.0, 10):
        for b in np.linspace(0.2, 2.0, 10):
            seeds.append(_beta_shape(n, a, b))

    # raised‑cosine shapes (different powers)
    for pw in np.linspace(0.5, 3.0, 7):
        seeds.append(_raised_cosine(n, pw))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.2, 0.7, 1.5, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # soft‑max random logits
    for _ in range(8):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation function.  The function respects the given time budget.
    Returns the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # retrieve an older construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Try a couple of lengths – longer vectors tend to give lower bounds,
    # but the computational cost rises roughly linearly with n.
    lengths = [4000, 8000]   # feel free to add 12000 if time permits
    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Phase 0 – cheap scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on a smoothed max
        # ------------------------------------------------------------------
        for seed_vec in seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=50000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall‑SubGrad] n={n:5d} new best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                                 deadline,
                                                 max_iters=200000)
            if seq_sa is not None and val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[overall‑SA] n={n:5d} new best={val_sa:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Optional verification using the supplied evaluate_sequence (if available)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
