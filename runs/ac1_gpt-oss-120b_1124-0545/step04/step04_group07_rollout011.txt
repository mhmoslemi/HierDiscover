"""
Propose a low‑value step‑function.

Algorithm
---------
1. **Generate a few well‑chosen seeds** – the Cantor histogram,
   its sorted‐decreasing version (large values never meet large values
   after reversal), a simple interleaving of the sorted values,
   and a random permutation.  The best of these seeds is taken as the
   starting point.
2. **Adam‑based smooth‑max descent** – minimise a differentiable
   approximation of `max(a*a)`:
        * Compute the full convolution `c = a*a`.
        * Form soft‑max weights `w ∝ exp((c‑max(c))/τ)`.
        * The gradient of the smooth max is `‑2·(a_rev ∗ w)`.
        * Apply the Adam optimiser (per‑coordinate adaptive step sizes),
          then project back onto the simplex (clip ≥0 and renormalise).
        * τ (temperature) is annealed geometrically.
   The loop runs until the time budget is exhausted, keeping the best
   sequence seen.
3. **Return the best sequence** (as a plain Python list).

The code uses only NumPy and runs comfortably within the 1000 s limit.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)-1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Compute the full linear convolution a*a (length 2·len(a)-1)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the (1/3) Cantor measure with 2**L atoms and
    bin it into ``n`` equal‑width bins.  The result sums to 1.
    """
    N = 1 << L                      # number of atoms
    # positions of the atoms (ternary expansion with digits 0/2)
    powers = 3.0 ** np.arange(1, L + 1)          # (L,)
    idx = np.arange(N, dtype=np.uint32)[:, None]   # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)
    pos = (bits * (2.0 / powers)).sum(axis=1)      # (N,)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)        # normalise → sum == 1
    return a


def _project_to_simplex(x: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and renormalise to sum 1 (the simplex)."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s == 0.0:
        # fallback to uniform
        return np.ones_like(x) / x.size
    return x / s

            last_improve = time.time()
            print(f"[{it:7d}] new best = {best_score:.6f}  time={last_improve-start:.1f}s")

        # ---- temperature schedule ---------------------------------
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # ---- soft‑max weights (smooth approximation of max) -----
        w = np.exp((conv_full - max_b) / tau)      # shift by max_b for stability
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth max ---------------------------
        # g_full = a_rev ∗ w  (full length 3·n‑2)
        g_full = _conv_fft(a_rev, w, size2)
        # central n entries correspond to ∂/∂a_i
        g = g_full[n - 1 : n - 1 + n]
        grad = -2.0 * g                              # descent direction

        # ---- Adam update ------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_to_simplex(a_new)

        # ---- bookkeeping for the next iteration -------------------
        a = a_new
        a_rev = a[::-1].copy()

        # occasional very small random perturbation to escape plateaus
        if it % 50000 == 0:
            noise = np.random.normal(scale=0.001, size=n)
            a = _project_to_simplex(a + noise)
            a_rev = a[::-1].copy()

    # --------------------------------------------------------------
    # Reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
