    conv = _conv_fft(a, a)
    k_star = conv.argmax()
    grad = np.zeros_like(a)
    idx = k_star - np.arange(a.size)
    mask = (idx >= 0) & (idx < a.size)
    grad[mask] = 2*a[idx[mask]]
    # exponentiated update
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new)
    return a_new, conv.max()

"""
Propose a non‑negative step‑function (a list of floats) that minimizes
the provided ``evaluate_sequence`` objective.

Algorithm
---------

1. **Fast parametric sweep** – generate many simple families
   (power‑law, exponential decay, decreasing linear, U‑shaped
   power‑law and exponential).  For several lengths (up to 8000) the
   best candidate is kept.  This gives a solid starting point close to
   the known optimum.

2. **Multi‑start exponentiated‑gradient descent** – from the best sweep
   candidate and a few random Dirichlet seeds a projected exponentiated
   gradient (mirror descent on the probability simplex) is performed.
   The sub‑gradient of the max‑convolution is
   ``g_i = 2·a_{k*‑i}`` where ``k*`` is the index attaining the current
   maximum.  After each step the vector is projected back onto the
   simplex using the Euclidean projection of Duchi et al.  A small
   momentum term and occasional random mass swaps help escaping plateaus.

3. **Time‑aware loop** – the whole routine respects the global budget
   (default 1000 s).  Progress and intermediate scores are printed.
   The best sequence found is returned as a plain Python list.

The implementation uses only NumPy and the standard library, runs on a
single CPU and stays well within the allowed time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Duchi et al., ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    # next power of two for speed
    N = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α)  (zero before offset)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ r^{i‑offset} for i≥offset, zero before offset."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Decreasing linear (triangular) shape: a_i = n‑i (with leading zeros)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _u_power_sequence(n: int, alpha: float) -> np.ndarray:
    """U‑shaped power law: a_i ∝ (i+1)^α + (n‑i)^α."""
    idx = np.arange(n, dtype=float) + 1.0               # 1 … n
    a = idx ** alpha + (n - idx + 1) ** alpha
    s = a.sum()
    a /= s
    return a


def _u_exp_sequence(n: int, r: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ r^{i} + r^{n‑1‑i},  r>1."""
    idx = np.arange(n, dtype=float)
    a = r ** idx + r ** (n - 1 - idx)
    s = a.sum()
    a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """Exact objective used in optimisation (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative numbers that minimizes the
    ``evaluate_sequence`` upper bound.  The function returns the best
    sequence (as a plain list) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

    # lengths – stay below ~8000 so that FFTs stay cheap
    length_candidates = [1500, 2000, 3000, 4000, 5000, 6000, 8000]

    # families and grids
    power_alphas = [0.2, 0.35, 0.5, 0.65, 0.8]
    power_offsets = [0, 5, 10, 30, 100]

    exp_rs = [0.90, 0.94, 0.97, 0.99]
    exp_offsets = [0, 5, 20, 50]

    linear_offsets = [0, 10, 30, 100]

    u_power_alphas = [0.5, 1.0, 1.5]
    u_exp_rs = [1.1, 1.2, 1.5, 2.0]

    for n in length_candidates:
        # abort sweep early if we run out of time
        if time.time() > deadline:
            break

        # ----- Power‑law family ------------------------------------------------
        for offset in power_offsets:
            if offset >= n:
                continue
            for alpha in power_alphas:
                a = _power_sequence(n, alpha, offset)
                score = _objective(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[sweep] n={n} power α={alpha:.3f} off={offset} score={score:.6f}")

        # ----- Exponential family ---------------------------------------------
        for offset in exp_offsets:
            if offset >= n:
                continue
            for r in exp_rs:
                a = _exp_sequence(n, r, offset)
                score = _objective(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[sweep] n={n} exp r={r:.3f} off={offset} score={score:.6f}")

        # ----- Linear decreasing (triangular) ---------------------------------
        for offset in linear_offsets:
            if offset >= n:
                continue
            a = _linear_sequence(n, offset)
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n} linear off={offset} score={score:.6f}")

        # ----- U‑shaped power ---------------------------------------------------
        for alpha in u_power_alphas:
            a = _u_power_sequence(n, alpha)
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n} u‑power α={alpha:.3f} score={score:.6f}")

        # ----- U‑shaped exponential ---------------------------------------------
        for r in u_exp_rs:
            a = _u_exp_sequence(n, r)
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n} u‑exp r={r:.2f} score={score:.6f}")

        # ----- Random Dirichlet seeds -------------------------------------------
        for _ in range(2):
            a = rng.random(n)
            a /= a.sum()
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[sweep] n={n} random dirichlet score={score:.6f}")

        # ----- Global variable ``height_sequence_1`` if present -------------------
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                a = np.maximum(init, 0.0)
                if a.sum() == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[sweep] n={n} global init score={score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Multi‑start exponentiated‑gradient / sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    n = best_seq.size
    # parameters for the gradient loop
    init_lr = 0.6
    lr_decay = 0.9995
    lr_min = 1e-6
    eval_every = 40        # evaluate exact objective every N iterations
    swap_every = 70        # occasional random mass swaps
    max_restarts = 5       # number of random restarts (incl. best sweep)

    iteration_total = 0
    for restart in range(max_restarts):
        # initialise sequence for this restart
        if restart == 0:
            a = best_seq.copy()
        else:
            a = rng.random(n)
            a /= a.sum()
        lr = init_lr
        momentum = np.zeros_like(a)

        it = 0
        while time.time() < deadline:
            it += 1
            iteration_total += 1

            # convolution and max location
            conv = _conv_fft(a, a)
            max_val = conv.max()
            k_star = int(conv.argmax())

            # sub‑gradient of the max‑convolution (Euclidean)
            idx = k_star - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

            # momentum (optional)
            momentum = 0.9 * momentum + grad

            # exponentiated‑gradient style update (keeps positivity)
            a = a * np.exp(-lr * momentum)
            a = _project_onto_simplex(a)

            # decay learning rate
            lr = max(lr * lr_decay, lr_min)

            # ----- random mass exchange to escape plateaus -----------------
            if it % swap_every == 0:
                i, j = rng.integers(0, n, size=2, endpoint=False)
                delta = min(a[i] * 0.3, lr * 0.5)
                if delta > 1e-12:
                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)

            # ----- periodic exact evaluation --------------------------------
            if it % eval_every == 0:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[improve] restart={restart} it={iteration_total} n={n} "
                          f"score={best_score:.6f}")
                # stop early if target already reached
                if best_score < 1.503:
                    print("[target reached]")
                    break
        if best_score < 1.503:
            break  # no need for further restarts

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} (n={best_seq.size}) "
          f"iterations={iteration_total}")
    # the evaluation routine will normalise/cap as needed – we just return the list
    return best_seq.tolist()
