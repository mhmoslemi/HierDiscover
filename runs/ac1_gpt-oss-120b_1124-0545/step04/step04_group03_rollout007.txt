def lp_step(f, M, t):
    n = len(f)
    A = build_conv_matrix(f)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum(g) -> minimize -sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': 60})
    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_norm = g / sum_g
    # combine
    f_new = (1 - t) * f + t * g_norm
    # Normalize to sum=1 if some rounding error
    f_new = f_new / f_new.sum()
    return f_new
```

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Build a modest library of analytic candidates (power‑law families) and a
   handful of random Dirichlet draws. Keep the best one as a start.
2. For each start run a continuation of a smoothed‐max objective
   Φ_λ(a) = (1/λ)·log∑_k exp(λ·(a*a)_k) using L‑BFGS‑B on the soft‑max
   parametrisation.  The λ‑schedule goes from 0.5 up to 5000, with a larger
   budget of iterations than the previous baseline.
3. Refine the best distribution with a linear‑programming step that
   maximises the total mass of a new distribution g under the linear
   constraints   (f * g)(k) ≤ M   (M = max conv(f,f)).  The LP has n variables
   and 2n‑1 constraints and is solved with SciPy’s HiGHS interior‑point
   method.  The obtained g is mixed with the current f, and the mixing ratio
   is chosen by an inexpensive line‑search on the exact max‑convolution.
4. A short random‑mass‑transfer local search is performed as a final polish.
5. The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) convolution of a with itself using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b (both length n) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    fft_b = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fft_a * fft_b, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """Exact objective (sum normalised to 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Tiny mass transfer between two random indices (projected onto simplex)."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """Generate a normalised analytic family."""
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smoothed max Φ_λ and gradient w.r.t. the soft‑max parameters x.
    Returns (phi, grad_x).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # convolution a*a
    conv = conv_fft_self(a)
    max_c = np.max(conv)

    # log‑sum‑exp smooth max
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                     # weights (sum to 1)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]     # length n
    grad_a = 2.0 * slice_part[::-1]            # ∂Φ/∂a

    # Jacobian of soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1) × n matrix A with A[k,i] = f_i if 0≤k‑i<n else 0."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_step(f: np.ndarray, M: float):
    """
    Solve   max   Σ g_i   s.t.   (f * g)_k ≤ M   (∀k),  g_i ≥ 0.
    Returns the optimal g normalised to sum = 1, or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –sum(g)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    return g / s                         # normalised distribution


def best_mix(f: np.ndarray, g: np.ndarray, M_current: float):
    """
    Choose a mixing coefficient t ∈ [0,1] that minimises the exact
    max‑convolution of (1‑t)f + t g.  Returns (f_new, new_max).
    """
    conv_f = conv_fft_self(f)
    conv_g = conv_fft_self(g)
    conv_fg = conv_fft(f, g)

    # sample a modest grid of t values
    ts = np.linspace(0.0, 1.0, 21)
    best_t = 0.0
    best_max = M_current
    for t in ts:
        conv_mix = ((1 - t) ** 2) * conv_f + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_g
        cur_max = conv_mix.max()
        if cur_max < best_max:
            best_max = cur_max
            best_t = t
    f_new = (1 - best_t) * f + best_t * g
    f_new = f_new / f_new.sum()
    return f_new, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that minimises
    the evaluation function.  The routine builds analytic candidates,
    runs a smooth‑max L‑BFGS continuation from several starts, performs an
    LP‑based refinement, and finishes with a tiny random‑mass‑transfer
    local search.  The best sequence seen before the time budget expires is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Assemble a modest pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # power‑law families (product and sum)
    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55, 0.6):
            for off in (0.0, 0.2, 0.5, 1.0):
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, p, off_eff, mode)
                candidates.append(cand)

    # uniform
    candidates.append(np.ones(n) / n)

    # a few random Dirichlet draws (more diversity)
    for _ in range(8):
        alpha = np.random.uniform(0.15, 1.0)        # concentration
        g = np.random.gamma(alpha, 1.0, size=n)
        candidates.append(g / g.sum())

    # incorporate the globally available construction if length matches
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    # ------------------------------------------------------------------
    # Evaluate all candidates, keep the best
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max L‑BFGS continuation from several starts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0]
    restarts = 3

    for rst in range(restarts):
        if time.time() >= deadline:
            break

        # pick a start: first restart uses the current best, others random
        if rst == 0:
            a_start = best_a.copy()
        else:
            alpha = np.random.uniform(0.1, 1.0)
            g = np.random.gamma(alpha, 1.0, size=n)
            a_start = g / g.sum()

        # soft‑max parameters
        eps = 1e-12
        x = np.log(np.maximum(a_start, eps))

        # continuation over λ
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # allocate iterations adaptively (more when we have time)
            max_iter = int(min(1800, max(30, remaining * 120)))
            res = minimize(compute_phi_and_grad, x,
                           args=(lam, n),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter, 'disp': False})
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[restart {rst} λ={lam:.0f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # short Adam fine‑tuning after the λ‑schedule (if time permits)
        if time.time() < deadline:
            alpha_adam = 0.04
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t_adam = 0
            while time.time() < deadline:
                t_adam += 1
                phi, grad_x = compute_phi_and_grad(x, lam_schedule[-1], n)
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t_adam)
                v_hat = v / (1 - beta2 ** t_adam)
                step = alpha_adam * m_hat / (np.sqrt(v_hat) + eps_adam)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam rs={rst}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t_adam >= 2500:
                    break

    # ------------------------------------------------------------------
    # 3) Linear‑programming refinement (mixing step with optimal t)
    # ------------------------------------------------------------------
    f = best_a.copy()
    no_improve_lp = 0
    while time.time() < deadline and no_improve_lp < 20:
        M = np.max(conv_fft_self(f))               # current worst convolution value
        g = lp_step(f, M)
        if g is None:
            break
        f_new, new_max = best_mix(f, g, M)
        new_score = float(2 * n * new_max)
        if new_score < best_score - 1e-9:
            best_score, best_a = new_score, f_new.copy()
            f = f_new
            no_improve_lp = 0
            elapsed = time.time() - start_time
            print(f"[LP] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            no_improve_lp += 1
            # if LP fails to improve, shrink the mixing aggressiveness
            # (by terminating early after several failed attempts)
            if no_improve_lp >= 10:
                break

    # ------------------------------------------------------------------
    # 4) Light random‑mass‑transfer local search (final polish)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            scale = max(0.001, scale * 0.9)   # shrink step size after improvement
        else:
            scale = min(0.05, scale * 1.001)  # slowly re‑expand to keep exploring

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # The evaluator rescales anyway, but we keep sum≈1 for numerical stability.
    return best_a.tolist()
