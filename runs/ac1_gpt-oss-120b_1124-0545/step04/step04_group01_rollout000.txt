def _simulated_anneal(start_seq, n, deadline, max_iters=20000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    rng = np.random.default_rng()
    T0 = (best_val) * 0.01 # maybe 1% of value
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        T = T0 * (0.99 ** (it // 100)) # exponential decay

        i,j = rng.integers(0, n, size=2)
        if i == j: continue
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i
        # transfer mass
        max_transfer = min(a[donor], 0.1 * a[donor] + 1e-12) # step size maybe 0.05?
        if max_transfer <= 0: continue
        delta = rng.random() * max_transfer
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        new_val = _objective(a_new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T+1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
                print(f"[SA] n={n} it={it} best={new_val:.6f}")
    return best_seq, best_val

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines many analytic seeds, Adam optimisation on a
smooth approximation of the max‑convolution, a long mirror‑descent
sub‑gradient phase, and a simulated‑annealing hill‑climber.  It is
time‑aware: the global deadline is respected and the routine returns
the best sequence found before the budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Objective and simple utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation of the optimisation objective."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_conv = np.max(conv)
    s = seq.sum()
    # Guard against divide‑by‑zero (should never happen)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Seed generation (much richer than the reference)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised starting vectors."""
    rng = np.random.default_rng(2024)
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.99, 12):
        seeds.append(_geom(n, r))

    # symmetric geometric
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 12):
        seeds.append(_power(n, p))

    # beta‑like shapes (dense grid)
    alphas = np.linspace(0.1, 2.5, 9)
    betas = np.linspace(0.1, 2.5, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.3, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors
    for _ in range(8):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # perturbations of the currently best known sequence (if any)
    if previous is not None and previous.shape[0] == n:
        for sigma in (1e-3, 5e-3, 1e-2, 5e-2):
            for _ in range(4):
                pert = previous + rng.normal(scale=sigma, size=n)
                pert = np.clip(pert, 0.0, None)
                if pert.sum() == 0:
                    pert = _uniform(n)
                else:
                    pert /= pert.sum()
                seeds.append(pert)

    # reuse the exact previous construction if it matches the length
    if previous is not None and previous.shape[0] == n:
        seeds.append(previous / previous.sum())

    # shuffle to avoid any ordering bias
    rng.shuffle(seeds)
    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smoothed max of the auto‑convolution.
    Returns best sequence (ℓ₁‑normalised) and its objective value.
    """
    v = np.log(seed + 1e-30)               # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 5.0          # smoothing coefficient (≈ temperature)
    beta_mul = 1.02     # increase per iteration (slow growth)

    best_seq = seed.copy()
    best_val = _objective(best_seq)

    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                         # stays on simplex, sum = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)        # sum = 1 ⇒ denominator = 1

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ---------- smooth‑max weights ----------
        beta *= beta_mul
        shift = np.max(conv)                     # numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---------- gradient wrt a ----------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # ---------- convert to gradient wrt v (softmax Jacobian) ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement using the exact
    sub‑gradient of the max‑convolution.  Long run to squeeze out every
    possible improvement.
    """
    a = start_seq.copy()
    # Ensure strict positivity (required for log‑step)
    eps = 1e-30
    a = np.clip(a, eps, None)
    a /= a.sum()

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)          # slowly decreasing
        log_a = np.log(a) - lr * g
        log_a -= np.max(log_a)          # numeric stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_anneal(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Simple pair‑wise mass‑transfer SA.  Allows uphill moves with a
    decreasing temperature, often escaping shallow local minima.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # Initial temperature roughly 1 % of current objective
    T0 = max(cur_val * 0.01, 1e-4)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling schedule
        T = T0 * (0.9995 ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # choose donor / receiver
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        # amount to move – a fraction of donor's mass, shrinking with temperature
        max_transfer = cur[donor]
        if max_transfer <= 0.0:
            continue
        frac = min(0.2, 0.5 * T / (T0 + 1e-12))   # larger moves early, tiny later
        delta = rng.random() * max_transfer * frac

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        # keep numbers non‑negative (tiny numerical negatives are clipped)
        new = np.clip(new, 0.0, None)
        new_sum = new.sum()
        if new_sum == 0.0:
            continue
        new /= new_sum

        new_val = _objective(new)
        delta_val = new_val - cur_val

        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function of non‑negative heights that minimises the
    ``evaluate_sequence`` objective.  Returns the best sequence found as a
    plain list of floats.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # Retrieve a previously discovered construction, if it exists.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to try – larger n tends to give a tighter asymptotic bound,
    # but also costs more time per iteration.
    lengths = [3000, 5000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Phase 0 – evaluate raw seeds (cheap)
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} val={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max (limited to a few promising seeds)
        # ------------------------------------------------------------------
        # Pick the 7 best seeds according to the raw evaluation
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:7]]

        for idx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                                max_iters=12000)
            if val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall‑Adam] n={n:5d} seed{idx} val={val_adam:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (long run)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sub, val_sub = _subgradient_refine(global_best_seq, n,
                                                   deadline,
                                                   max_iters=130000)
            if val_sub < global_best_val:
                global_best_val = val_sub
                global_best_seq = seq_sub.copy()
                print(f"[overall‑SubGrad] n={n:5d} val={val_sub:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing hill‑climb
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_sa, val_sa = _simulated_anneal(global_best_seq, n,
                                               deadline,
                                               max_iters=90000)
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"[overall‑SA] n={n:5d} val={val_sa:.6f}")

        # keep a copy for the next length (used as a warm‑start)
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Final short sub‑gradient polishing (if time permits)
    # ----------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq_final, val_final = _subgradient_refine(global_best_seq,
                                                   len(global_best_seq),
                                                   deadline,
                                                   max_iters=30000)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final

    # Ensure the returned object is a plain Python list (as required)
    if global_best_seq is None:
        # safety fallback – uniform distribution of length 4000
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    # expose the best sequence under a conventional name for possible reuse
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
