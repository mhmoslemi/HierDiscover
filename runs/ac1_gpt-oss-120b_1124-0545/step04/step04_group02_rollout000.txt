def conv_fft_two(x, y):
    n = len(x)
    m = len(y)
    size = _next_pow2(n + m - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n+m-1]

"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas that differ from the previous solution:

1. **Smooth‑max (log‑sum‑exp) surrogate** – Instead of using the exact
   max‑convolution (which yields a sub‑gradient that is often zero
   on large plateaus), we minimise a differentiable approximation
   ``softmax_alpha(b) = (1/α) log Σ exp(α·b_k)``.  Its gradient is a
   weighted convolution of the current vector with a probability
   distribution over the convolution indices.  By slowly increasing
   α from 1 to a large value (≈200) we first obtain a smooth descent
   and finally converge to the true max‑objective.

2. **Adam‑style adaptive steps** – The surrogate gradient is fed into
   an Adam optimiser with a learning‑rate schedule.  The adaptive
   scaling allows larger moves early on and fine adjustments later.

3. **Symmetry enforcement & stochastic refinement** – The optimal
   shape is known to be symmetric, so after each update we mirror the
   vector and re‑normalise onto the probability simplex.  After the
   main optimisation we perform a short random‑perturbation phase
   (Gaussian jitter) to polish the best candidate.

The routine works for any length *n* (default 2000) and respects the
1000 s wall‑time budget.  All helper functions are top‑level, no
closures or lambdas are used, and the returned list consists of
non‑negative floats that sum to one (the evaluator rescales automatically
if the sum differs from 1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    size = _next_pow2(n + m - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: n + m - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric (v_i = v_{n-1-i}) and renormalise."""
    v_sym = (v + v[::-1]) * 0.5
    s = v_sym.sum()
    if s > 0:
        v_sym /= s
    return v_sym


# ----------------------------------------------------------------------
# Evaluation (scale‑invariant version)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already lies on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises the evaluation function.  The routine follows the
    three‑stage strategy described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Length of the candidate (default matches the previous best)
    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 0) Initialise -------------------------------------------------
    # --------------------------------------------------------------

    # Start from the global reference if it exists, otherwise from a
    # symmetric beta‑shaped distribution.
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            a = np.maximum(ref, 0.0)
            a = a / a.sum()
            a = enforce_symmetry(a)
        else:
            a = None
    else:
        a = None

    if a is None:
        # beta‑shape with modest parameters produces a good U‑shape.
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) ** 0.5 * (n - i) ** 0.5
        a = a / a.sum()
        a = enforce_symmetry(a)

    best_seq = a.copy()
    best_score = compute_score(best_seq)

    # Adam moments
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    # Hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr_start = 0.5
    lr_end = 0.001

    alpha_start = 1.0
    alpha_end = 200.0

    iter_cnt = 0
    last_print = start_time

    # --------------------------------------------------------------
    # 1) Smooth‑max Adam optimisation --------------------------------
    # --------------------------------------------------------------

    while time.time() < deadline:
        # ---- Current objective (exact) ---------------------------------
        conv_self = conv_fft_self(a)
        max_val = conv_self.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iter_cnt:06d}] new best (exact) = {best_score:.6f}")

        # ---- Progress‑dependent schedules ------------------------------
        t = (time.time() - start_time) / (budget_s - 5.0)   # [0,1]
        t = min(max(t, 0.0), 1.0)

        lr = lr_start * (1.0 - t) + lr_end * t
        alpha = alpha_start * (1.0 - t) + alpha_end * t

        # ---- Smooth‑max weights (stable softmax) ----------------------
        # w_k ∝ exp(α·conv_k) ; we subtract max for numerical stability.
        max_conv = max_val
        exp_arg = alpha * (conv_self - max_conv)
        # clip to avoid overflow in exp
        exp_arg = np.clip(exp_arg, -700, 700)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- Gradient of the smooth‑max objective ----------------------
        # grad_i = 2 * Σ_k w_k * a_{k-i}   (indices out of range give 0)
        w_rev = w[::-1]                               # length 2n‑1
        conv_rev = conv_fft_two(a, w_rev)             # length 3n‑2
        idx = (len(w_rev) - 1) - np.arange(n)         # indices for gradient
        g = 2.0 * conv_rev[idx]

        # ---- Adam update -----------------------------------------------
        iter_cnt += 1
        beta1_pow = beta1 ** iter_cnt
        beta2_pow = beta2 ** iter_cnt

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex and enforce symmetry
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_simplex(a_new)
        a_new = enforce_symmetry(a_new)

        a = a_new

        # ---- Occasional stochastic mass‑transfer move -------------------
        if np.random.rand() < 0.01:
            a = random_move(a, scale=0.02)

        # ---- Jitter restart if stagnating --------------------------------
        if iter_cnt % 4000 == 0:
            jitter = np.random.rand(n) * 1e-4
            a = best_seq + jitter
            a = project_simplex(a)
            a = enforce_symmetry(a)
            m.fill(0.0)
            v.fill(0.0)

        # ---- Periodic status output ------------------------------------
        now = time.time()
        if now - last_print > 30.0:
            print(f"[{iter_cnt:06d}] time={now-start_time:.1f}s score={best_score:.6f} α={alpha:.1f} lr={lr:.4g}")
            last_print = now

    # --------------------------------------------------------------
    # 2) Final fine‑grained stochastic refinement --------------------
    # --------------------------------------------------------------

    refine_deadline = start_time + budget_s - 1.0
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=4e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand = enforce_symmetry(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iter_cnt}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    return best_seq.tolist()
