def _optimize_a(a_init, n, alloc_s, rng):
    # maintain sum=1 via softmax
    v = np.log(a_init + 1e-12)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta_start = 0.01
    beta_end = 5000.0
    lr_start = 0.5
    lr_end = 0.001
    beta1=0.9; beta2=0.999; eps=1e-8
    fft_len = _next_pow2(3 * n - 2) # maybe 2 * n big enough but keep 3n
    start = time.time()
    deadline = start + alloc_s - 0.05
    best_a = a_init.copy()
    best_score = fast_score(best_a)  # compute objective
    last_improve = start
    mask_phase_start = 0.75  # after 75% of time switch to hard mask
    
    it=0
    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed/(alloc_s - 5.0))
        beta = beta_start * ((beta_end / beta_start) ** p)
        lr = lr_start * ((lr_end / lr_start) ** p)
        # compute a
        a = np.exp(v)
        a /= a.sum()
        # conv using rfft
        fa = np.fft.rfft(a, fft_len)
        conv_aa = np.fft.irfft(fa * fa, fft_len)[: 2*n-1]
        max_val = conv_aa.max()
        # compute weights
        if p < mask_phase_start:
            z = beta * conv_aa
            w = np.exp(z - np.max(z))
            w /= w.sum()
        else:
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv_aa); mask[np.argmax(conv_aa)] = 1.0; mask_sum=1.0
            w = mask / mask_sum
        
        # compute gradient g
        fw = np.fft.rfft(w, fft_len)
        # convolution of w and a
        conv_wa = np.fft.irfft(fw * fa, fft_len)[: 3*n-2]  # length bigger
        g2 = conv_wa[:n]
        # cross-correlation: conj(fw)*fa
        corr_full = np.fft.irfft(np.conj(fw) * fa, fft_len)[: 3*n-2]
        g1 = corr_full[:n]
        g = 2.0 * n * (g1 + g2)
        # grad wrt v
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)
        it += 1
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat_corr = v_hat / (1 - beta2**it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)
        # evaluate exact
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            # optionally print
            if it % 500 == 0:
                print(f"[{it:07d}] new best {best_score:.6f} (t={elapsed:.1f}s)")
        # restart if no improvement for e.g., 10 seconds
        if time.time() - last_improve > 10.0:
            # jitter
            jitter = rng.normal(scale=0.02, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[{it:07d}] restart after stagnation")
    return best_a, best_score

"""
Non‑negative step‑function optimiser.

The optimiser works in two stages:

1.  **Global random search** – a fast Monte‑Carlo phase that draws many
    Dirichlet vectors with a broad range of concentration parameters and
    a few deterministic Beta‑shaped candidates (U‑shaped, arcsine‑like).
    The objective 2·n·max(conv(a,a)) is evaluated with an FFT‑based
    convolution, which is cheap enough to explore several hundred‑thousand
    candidates in the allocated time.

2.  **Local refinement** – a soft‑max / Adam optimiser (similar to the
    previous state‑of‑the‑art) starting from the best random vector.
    The temperature of the soft‑max is annealed exponentially and the
    learning‑rate follows an exponential schedule.  After 70 % of the
    allocated time the true max is used (hard mask) which yields a genuine
    sub‑gradient of the original non‑smooth objective.

Both stages respect the 1000 s budget, automatically split the time,
log progress and finally return the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector ``a`` does **not** need to be normalised – the function
    normalises it internally, because the objective is scale invariant.
    """
    n = a.shape[0]
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s                     # now Σa = 1
    # length needed for a*a (linear convolution)
    fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    a_init: np.ndarray,
    n: int,

    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:

    Local optimisation of a starting vector ``a_init``.
    Works on the log‑soft‑max parametrisation (v) and uses Adam
    with an exponentially annealed temperature for the smooth max.
    Returns the best vector encountered and its exact (scale‑invariant)
    score.
    """
    # initialise log‑parameters (soft‑max guarantees feasibility)
    v = np.log(a_init + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # annealing schedules (exponential)
    beta_start, beta_end = 0.01, 5000.0          # temperature
    lr_start, lr_end = 0.5, 0.001                # learning‑rate

    # FFT length that safely covers all convolutions we need
    fft_len = _next_pow2(3 * n - 2)

    start = time.time()
    deadline = start + max(0.0, alloc_s - 0.05)   # tiny safety margin

    # best candidate seen so far
    best_a = np.exp(v)
    best_a /= best_a.sum()
    best_score = fast_score(best_a)

    last_improve = start
    it = 0
    # after this fraction of the allocated time we switch to the hard‑max mask
    mask_phase_start = 0.70

    while time.time() < deadline:
        elapsed = time.time() - start
        # normalised progress p∈[0,1]
        p = min(1.0, elapsed / (alloc_s - 5.0))

        # exponential schedules
        beta = beta_start * ((beta_end / beta_start) ** p)
        lr = lr_start * ((lr_end / lr_start) ** p)

        # -------------------- current probabilities --------------------
        a = np.exp(v)
        a /= a.sum()                     # Σa = 1, a_i ≥ 0

        # -------------------- a * a convolution -----------------------
        fa = np.fft.rfft(a, fft_len)
        conv_aa = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv_aa.max()

        # -------------------- weight vector w ------------------------
        if p < mask_phase_start:
            # smooth soft‑max over the convolution entries
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard mask – sub‑gradient of the true max
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0.0:
                mask = np.zeros_like(conv_aa)
                mask[np.argmax(conv_aa)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # -------------------- gradient wrt a (via FFT) --------------
        fw = np.fft.rfft(w, fft_len)

        # (w * a) – linear convolution, first n entries
        conv_wa = np.fft.irfft(fw * fa, fft_len)[: 3 * n - 2]
        g2 = conv_wa[:n]

        # w ⋆ a – cross‑correlation, first n entries
        corr_full = np.fft.irfft(np.conj(fw) * fa, fft_len)[: 3 * n - 2]
        g1 = corr_full[:n]

        # gradient of the (smoothed) max‑objective
        g = 2.0 * n * (g1 + g2)

        # chain rule for the soft‑max parametrisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # -------------------- Adam update -----------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # -------------------- exact score & bookkeeping ---------------
        cur_score = 2.0 * n * max_val            # sum(a) = 1 ⇒ denominator = 1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 2000 == 0:
                print(
                    f"[{it:07d}] new local best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # -------------------- stagnation restart -----------------------
        if time.time() - last_improve > 10.0:
            # jitter around the current best point
            jitter = rng.normal(scale=0.02, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[{it:07d}] restart after stagnation (t={elapsed:.1f}s)")

    return best_a, best_score


def _random_search(
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Monte‑Carlo search for a good start vector.
    Draws Dirichlet samples with a wide range of concentration parameters
    and evaluates a few deterministic Beta‑shaped U‑distributions.
    Returns the best vector and its score seen before ``deadline``.
    """
    start = time.time()
    best_a = None
    best_score = float("inf")

    # occasionally inject the historic construction if present
    if "height_sequence_1" in globals():
        a_hist = np.array(globals()["height_sequence_1"], dtype=float)
        if a_hist.size != n:
            if a_hist.size > n:
                a_hist = a_hist[:n]
            else:
                a_hist = np.pad(a_hist, (0, n - a_hist.size))
        a_hist = np.maximum(a_hist, 0.0)
        a_hist /= a_hist.sum()
        score_hist = fast_score(a_hist)
        best_a, best_score = a_hist, score_hist
        print(f"[init] historic candidate score {best_score:.6f}")

    # parameters for deterministic Beta shapes
    beta_alphas = [0.15, 0.25, 0.35, 0.45]   # all < 1 → U‑shaped

    while time.time() < deadline:
        # 20 % of the time we evaluate a deterministic Beta shape
        if rng.random() < 0.20:
            alpha = rng.choice(beta_alphas)
            x = (np.arange(n) + 0.5) / n                      # mid‑points in (0,1)
            # avoid exact 0/1 for numerical stability
            eps = 1e-12
            x = np.clip(x, eps, 1 - eps)
            a = (x ** (alpha - 1) * (1 - x) ** (alpha - 1))  # Beta(α,α) density (up to const)
            a += 1e-12                                         # prevent zeros
            a /= a.sum()
        else:
            # Dirichlet draw with log‑uniform concentration α∈[0.05,1.5]
            log_alpha = rng.uniform(np.log(0.05), np.log(1.5))
            alpha = np.exp(log_alpha)
            a = rng.dirichlet(np.full(n, alpha))

        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            now = time.time() - start
            print(f"[rand] new best {best_score:.6f} (t={now:.1f}s)")

    return best_a, best_score


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget (default
    1000 s) and returns the best sequence discovered as a plain ``list[float]``.
    """
    rng_global = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose sequence length (default 2000, can be overridden)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 2) Global random search (≈35 % of the total budget)
    # ------------------------------------------------------------------
    time_for_random = (deadline - time.time()) * 0.35
    best_a, best_score = None, float("inf")
    if time_for_random > 5.0:
        best_a, best_score = _random_search(
            n=n,
            deadline=time.time() + time_for_random,
            rng=rng_global,
        )
        print(f"[stage1] best after random search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Local refinement with gradient‑based optimiser
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # we devote the rest of the budget to at most two optimisation runs
        runs = 2
        for run in range(runs):
            alloc = remaining * (1.0 / (runs - run))          # split the rest
            if best_a is None:
                # fall back to a fresh Dirichlet start
                a0 = rng_global.dirichlet(np.full(n, 0.5))
            else:
                a0 = best_a.copy()
                # a tiny jitter before the first local run can help escaping
                if run == 0:
                    a0 = np.clip(a0 + rng_global.normal(scale=0.01, size=n), 0, None)
                    a0 /= a0.sum()
            a_opt, sc_opt = _optimize_a(
                a_init=a0,
                n=n,
                alloc_s=alloc,
                rng=np.random.default_rng(seed + run + 12345),
            )
            if sc_opt < best_score:
                best_score, best_a = sc_opt, a_opt.copy()
                print(
                    f"[stage2 run {run}] new global best = {best_score:.6f}"
                    f" (elapsed {time.time() - start_time:.1f}s)"
                )
            remaining = deadline - time.time()
            if remaining < 5.0:
                break

    # ------------------------------------------------------------------
    # 4) Fallback – uniform vector (should never happen)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a)

    # ------------------------------------------------------------------
    # 5) Final safety / reporting
    # ------------------------------------------------------------------
    total_elapsed = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_elapsed:.1f}s")
    # ensure non‑negativity (already guaranteed) and return as plain list
    best_a = np.maximum(best_a, 0.0)
    return best_a.tolist()
