"""
Two‑stage optimiser for the self‑convolution bound.

* Stage 0 – Initialise a good (possibly already‑optimised) step function.
* Stage 1 – A small Adam/soft‑max warm‑up (fast, cheap, helps the LP step).
* Stage 2 – Frank‑Wolfe style updates realised by a linear programme.
  For the current sequence f we solve

        maximise   Σ b_i
        subject to  (f * b)[k] ≤ M   for all k,
                    b_i ≥ 0,

  where M = max (f*f).  The LP returns a vector b that can carry much more
  ℓ1‑mass than f while respecting the same convolution bound.
  Mixing f and the normalised b (with a line‑search on the mixing ratio)
  strictly reduces the bound unless f is already optimal.

* Stage 3 – A very short simulated‑annealing refinement that moves a tiny
  amount of mass between two coordinates.

The whole routine stops after the supplied budget (default 1000 s) and
returns a list of non‑negative floats whose sum is 1.  The returned list
can be fed to ``evaluate_sequence`` directly.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    v = v - np.max(v)
    e = np.exp(v)
    return e / e.sum()


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Return a normalised starting vector.
    Prefer the global ``height_sequence_1`` if it exists and has the right length.
    Otherwise use the classic arcsine (inverse‑sqrt) shape.
    """
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            return a0

    # arcsine shape – the continuous optimum for the relaxed problem
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def fft_len_for_len(k: int) -> int:
    """
    Return the next power of two that can hold a convolution of length k.
    """
    return 1 << ((k - 1).bit_length())


def max_convolution(a: np.ndarray) -> float:
    """
    Linear self‑convolution a*a and return its maximum entry.
    """
    n = a.size
    L = 2 * n - 1
    Lfft = fft_len_for_len(L)
    fa = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fa * fa, n=Lfft)[:L]
    return float(conv.max())


def exact_score_from_max(n: int, conv_max: float) -> float:
    """Objective value for a probability vector (sum = 1)."""
    return float(2.0 * n * conv_max)


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) such that A @ b = f * b
    (linear convolution).  Row k corresponds to the coefficient of
    (f * b)[k].
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # fill column i with f placed on the i‑th diagonal
    for i in range(n):
        A[i:i + n, i] = f
    return A


def adam_softmax_step(v: np.ndarray, lr: float, beta1: float, beta2: float,
                      m: np.ndarray, v_hat: np.ndarray, it: int,
                      temp: float) -> (np.ndarray, np.ndarray, np.ndarray):
    """
    One Adam update on the log‑parameters v for the smooth‑max objective.
    Returns the updated (v, m, v_hat).
    """
    # a on the simplex
    a = softmax(v)

    # convolution a*a
    n = a.size
    Lfft = fft_len_for_len(2 * n - 1)
    fa = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(fa * fa, n=Lfft)[: 2 * n - 1]

    # smooth‑max weights (log‑sum‑exp)
    z = temp * conv
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w /= w.sum()  # shape (2n‑1,)

    # gradient of the smoothed objective w.r.t. v
    # gradient w.r.t. a (g = 2·n·(g1+g2))
    # g1 = cross‑correlation w ⋆ a  (valid part)
    # g2 = convolution w * a (first n entries)
    nfft = fft_len_for_len(3 * n - 2)
    fw = np.fft.rfft(w, n=nfft)
    fa_full = np.fft.rfft(a, n=nfft)

    conv_wa_full = np.fft.irfft(fw * fa_full, n=nfft)[: 3 * n - 2]
    g2 = conv_wa_full[:n]

    corr_full = np.fft.irfft(np.conj(fw) * fa_full, n=nfft)[: 3 * n - 2]
    g1 = corr_full[:n]

    g = 2.0 * n * (g1 + g2)

    # chain rule for the soft‑max parametrisation
    dot_ag = np.dot(a, g)
    grad_v = a * (g - dot_ag)

    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad_v
    v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat_corr = v_hat / (1.0 - beta2 ** it)

    v = v - lr * m_hat / (np.sqrt(v_hat_corr) + 1e-8)
    return v, m, v_hat


def local_sa(a: np.ndarray, best_score: float, time_deadline: float) -> (np.ndarray,
                                                                       float):
    """
    Very short simulated‑annealing refinement.  It moves a tiny fraction
    of mass between two coordinates and accepts the move if it lowers the
    exact objective (or with a temperature‑controlled probability).
    """
    n = a.size
    rng = np.random.default_rng()
    Lfft = fft_len_for_len(2 * n - 1)

    # parameters for the annealing schedule
    T_start = best_score * 0.005
    T_end = best_score * 1e-6
    start = time.time()
    phase_len = time_deadline - start
    it = 0

    while time.time() < time_deadline:
        # linear temperature schedule
        phase_elapsed = time.time() - start
        frac = min(1.0, phase_elapsed / phase_len)
        T = T_start * (1.0 - frac) + T_end * frac

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a small fraction of a[i] to a[j]
        max_move = min(a[i], 0.05 * a[i] + 1e-12)
        if max_move <= 0.0:
            continue
        delta = rng.random() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        fa = np.fft.rfft(a_new, n=Lfft)
        conv = np.fft.irfft(fa * fa, n=Lfft)[: 2 * n - 1]
        new_max = float(conv.max())
        new_score = exact_score_from_max(n, new_max)

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                # print progress (optional, keep output short)
        it += 1
    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The routine consists of:

    1.  A few Adam iterations on a soft‑max parametrisation (warm‑up).
    2.  Repeated linear‑programming updates (Frank‑Wolfe style) that
        provably decrease the ∞‑norm of the self‑convolution.
    3.  A short stochastic‑annealing finish.

    The function respects the supplied time budget and always returns a
    normalised probability vector (sum = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 0) Choose length and initial point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = init.size
        a = np.maximum(init, 0.0)
        if a.sum() == 0.0:
            a = initialise_a(n, seed=seed)
        else:
            a = a / a.sum()
    else:
        n = int(kwargs.get("n", 2000))
        a = initialise_a(n, seed=seed)

    # keep a copy of the best sequence seen so far
    best_a = a.copy()
    best_max = max_convolution(best_a)
    best_score = exact_score_from_max(n, best_max)

    # ------------------------------------------------------------------
    # 1) Light Adam warm‑up (≈5 % of the total budget)
    # ------------------------------------------------------------------
    warmup_deadline = start_time + 0.05 * budget_s
    v = np.log(a + 1e-12)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    it = 0
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.06, 0.004
    temp_start, temp_end = 4.0, 1500.0

    while time.time() < warmup_deadline:
        elapsed = time.time() - start_time
        p = min(1.0, elapsed / (warmup_deadline - start_time))
        lr = lr_start * (1.0 - p) + lr_end * p
        temp = temp_start * (1.0 - p) + temp_end * p
        it += 1
        v, m, v_hat = adam_softmax_step(v, lr, beta1, beta2, m, v_hat, it, temp)
        a = softmax(v)                     # stay on the simplex
        cur_max = max_convolution(a)
        cur_score = exact_score_from_max(n, cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_max = cur_max

    # ------------------------------------------------------------------
    # 2) Linear‑programming (Frank‑Wolfe) phase
    # ------------------------------------------------------------------
    # allocate the rest of the budget except the final short SA chunk
    lp_deadline = overall_deadline - 10.0    # keep ~10 s for SA

    while time.time() < lp_deadline:
        # current bound
        M = max_convolution(a)

        # build the convolution matrix for the current a
        A = build_conv_matrix(a)                         # shape (2n‑1, n)
        rows = A.shape[0]

        # LP: maximise Σ b_i  subject to A @ b ≤ M , b ≥ 0
        c = -np.ones(n)                                  # minimise -Σb_i  <=> maximise Σb_i
        ub = np.full(rows, M)

        # solve – use HiGHS (fast, handles dense matrices)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None),
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failure – abort the LP phase
            print(f"[LP] failure: {res.message}")
            break

        b = res.x
        b = np.maximum(b, 0.0)               # numerical safety
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # trivial solution – nothing to do
            break

        # normalise to a probability vector
        g = b / sum_b

        # try a few mixing ratios and keep the best improvement
        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 0.9):
            a_mix = (1.0 - t) * a + t * g
            a_mix = a_mix / a_mix.sum()
            cur_max = max_convolution(a_mix)
            cur_score = exact_score_from_max(n, cur_max)

            if cur_score < best_score:
                best_score = cur_score
                best_a = a_mix.copy()
                a = a_mix                     # continue from the improved point
                improved = True
                print(f"[LP] mix t={t:.2f} → score {best_score:.6f} (t={time.time() - start_time:.1f}s)")
                break

        if not improved:
            # No obvious improvement – make a tiny step to escape possible stall
            t = 0.05
            a = (1.0 - t) * a + t * g
            a = a / a.sum()
            # continue loop

        # stop early if we see virtually no progress for a while
        if time.time() - start_time > lp_deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 3) Short simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < overall_deadline - 1.0:
        best_a, best_score = local_sa(best_a, best_score, overall_deadline)

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    elapsed = time.time() - start_time
    print(f"[done] total time={elapsed:.1f}s, best score={best_score:.6f}, n={n}")

    # expose the best sequence for potential reuse in the next run
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
