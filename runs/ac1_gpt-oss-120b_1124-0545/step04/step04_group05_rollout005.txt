    t = beta * conv

    exp_t = np.exp(t_shift)
    Z = exp_t.sum()
    w = exp_t / Z
    # compute smoothed max value:
    smooth_max = (np.log(Z) + np.max(t)) / beta  # because logsumexp = max + log(sum exp(t - max))
    # compute objective: 2*n * smooth_max
    obj = 2.0 * n * smooth_max
    # compute gradient w.r.t a:

    fw = np.fft.fft(w, fft_len)
    fa = np.fft.fft(a, fft_len)

"""
propose_candidate – hybrid continuation L‑BFGS + simulated annealing.

Stage 1 – Continuation L‑BFGS.
    * The objective 2·n·max(conv(a,a)) is approximated by a smooth
      log‑sum‑exp (soft‑max) with a temperature β.
    * β is increased geometrically (5 → 2000).  For each β the
      unconstrained vector v (soft‑max parametrisation a=softmax(v))
      is optimised by L‑BFGS‑B using the exact gradient that can be
      evaluated by FFTs.
    * The best solution of a stage is used as the start for the next β.
    * Several values of β are processed while the global time budget
      permits.

Stage 2 – Fast stochastic refinement (simulated annealing).
    * Starting from the best sequence of Stage 1 a cheap Monte‑Carlo
      search transfers a small random fraction of mass between two
      coordinates.
    * Moves are accepted if they improve the exact objective or with
      probability exp(‑Δ/T) where T is linearly cooled.
    * The schedule is long enough to squeeze out the last few percent.

Both stages respect the 1000 s budget, keep the sequence non‑negative,
and finally return a normalised probability vector (sum = 1).  The
evaluation function is scale‑invariant, therefore the normalisation does
not affect the final score.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def initialise_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise a good starting point.
    1) If a previous construction with the correct length exists, reuse it.
    2) Otherwise search a short family a_i ∝ ((i+½)(n‑i‑½))^(‑p) for a few p.
    """
    # reuse previous result if possible
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        if a0.size == n and np.sum(a0) > 0.0:
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            return a0

    # try a tiny grid of exponents p (p = 0 gives uniform)
    i = np.arange(n, dtype=float)
    base = (i + 0.5) * (n - i - 0.5)          # > 0
    best_a = None
    best_score = math.inf

    # a very quick hand‑crafted grid (covers the classic arcsine p=0.5)
    for p in np.linspace(0.0, 1.5, 13):
        a = base ** (-p)
        a /= a.sum()
        # fast exact score (no clipping needed here)
        score = compute_score(a, n, fft_len_for_n(n))
        if score < best_score:
            best_score = score
            best_a = a

    # add a small random perturbation of the best grid point
    if best_a is not None:
        perturb = rng.normal(scale=1e-3, size=n)
        a = best_a + np.maximum(perturb, -best_a)   # keep non‑negative
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    # fallback: classic arcsine shape (p = 0.5)
    a = 1.0 / np.sqrt(base)
    return a / a.sum()


def fft_len_for_n(n: int) -> int:
    """FFT length needed for convolutions of length n (next power of two ≥ 3n‑2)."""
    return 1 << ((3 * n - 2).bit_length())


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Linear convolution a*a using FFT, return its maximum entry."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv.max()


def compute_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return math.inf
    conv_max = max_convolution(a, fft_len)
    return float(2.0 * n * conv_max / (sum_a * sum_a))


def grad_softmax_obj(v: np.ndarray,
                     a: np.ndarray,
                     conv: np.ndarray,
                     w: np.ndarray,
                     fft_len: int,
                     n: int) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the log‑parameters v.
    Implements g = 2·n·(g1 + g2) and the soft‑max chain rule.
    """
    fw = np.fft.fft(w, fft_len)
    fa = np.fft.fft(a, fft_len)

    # convolution w * a  (full length 3n‑2)
    conv_wa_full = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
    g2 = conv_wa_full[:n]

    # cross‑correlation w ⋆ a (full length 3n‑2)
    corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
    g1 = corr_full[:n]

    g = 2.0 * n * (g1 + g2)                      # gradient w.r.t. a
    dot_ag = np.dot(a, g)
    grad_v = a * (g - dot_ag)                    # chain rule through soft‑max
    return grad_v


def obj_val(v: np.ndarray, beta: float, n: int, fft_len: int) -> float:
    """Smooth approximation of the true objective (log‑sum‑exp with temperature β)."""
    a = softmax(v)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    t = beta * conv
    t_max = np.max(t)
    lse = t_max + np.log(np.sum(np.exp(t - t_max)))   # log‑sum‑exp
    smooth_max = lse / beta
    return float(2.0 * n * smooth_max)


def obj_grad(v: np.ndarray, beta: float, n: int, fft_len: int) -> np.ndarray:
    """Gradient of the smoothed objective (same arguments as obj_val)."""
    a = softmax(v)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    t = beta * conv
    t_max = np.max(t)
    exp_t = np.exp(t - t_max)
    w = exp_t / np.sum(exp_t)                 # soft‑max weights (size 2n‑1)

    return grad_softmax_obj(v, a, conv, w, fft_len, n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  A continuation L‑BFGS‑B phase followed by a
    short simulated‑annealing refinement respects the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # give a small safety margin for the host environment
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # length of the sequence (default 2000)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    fft_len = fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 0) obtain a good initial point (grid over exponent p)
    # ------------------------------------------------------------------
    a = initialise_a(n, rng)
    v = np.log(a + 1e-12)          # log‑parametrisation for soft‑max

    # ------------------------------------------------------------------
    # 1) Stage 1 – continuation L‑BFGS‑B on the smoothed objective
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = compute_score(best_a, n, fft_len)

    # geometric schedule for the temperature β
    betas = [5.0, 10.0, 30.0, 100.0, 300.0, 800.0, 2000.0]

    # spend roughly 80 % of the total budget on the gradient phase
    stage1_deadline = start_time + 0.80 * budget_s

    for beta in betas:
        if time.time() >= stage1_deadline:
            break

        # L‑BFGS‑B options – keep the number of internal iterations modest
        opts = {"maxiter": 200, "ftol": 1e-12, "gtol": 1e-12, "disp": False}

        # minimise the smooth objective for the current β
        res = optimize.minimize(
            fun=obj_val,
            x0=v,
            args=(beta, n, fft_len),
            jac=obj_grad,
            method="L-BFGS-B",
            options=opts,
        )
        # update the working point
        v = res.x
        a = softmax(v)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # evaluate the true (non‑smooth) score
        cur_score = compute_score(a, n, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[beta={beta:6.1f}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

        # occasional small random perturbation if stuck for a while
        if time.time() - stage1_deadline > 0:  # dummy to suppress unused‑var warning
            pass  # (placeholder – can be filled with restarts if desired)

    # ------------------------------------------------------------------
    # 2) Stage 2 – stochastic local refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining_time = overall_deadline - time.time()
    if remaining_time > 5.0:
        a = best_a.copy()
        best_conv_max = max_convolution(a, fft_len)
        best_score = 2.0 * n * best_conv_max / (a.sum() ** 2)

        # linear cooling schedule
        T_start = best_score * 0.01          # start temperature (≈1 % of score)
        T_end = best_score * 1e-6            # final temperature
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            # temperature based on elapsed SA time
            elapsed_sa = time.time() - sa_start
            frac = min(1.0, elapsed_sa / remaining_time)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a small mass transfer between two random indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move up to 5 % of a[i] (or a tiny epsilon if a[i] is small)
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact score for the proposal
            new_conv_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_conv_max / (a_new.sum() ** 2)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < math.exp(-(new_score - best_score) / (T + 1e-16)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score:.6f}, length={n}")

    # return a plain Python list (the harness expects `list[float]`)
    return best_a.tolist()
