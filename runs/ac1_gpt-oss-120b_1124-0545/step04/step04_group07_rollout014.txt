def propose_candidate(seed=42, budget_s=1000, n=None):
    if n is None: n = int(kwargs.get('n', 6000))
    # Setup
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Choose number of restarts based on remaining budget
    restarts = 3
    # allocate time per restart
    restart_deadline = deadline
    # but inside loop check time
    best_seq = None
    best_score = float('inf')
    for restart in range(restarts):
        # time left
        if time.time() >= deadline:
            break
        # allocate sub.time per restart as something relative
        # Choose init method
        if restart == 0:
            a = _cantor_histogram(n, L=14)
        elif restart == 1:
            # Random Dirichlet
            a = np.random.dirichlet(np.ones(n)).astype(np.float64)
        else:
            # Uniform random perturbation of Cantor
            a = _cantor_histogram(n, L=13)
            a = 0.5*a + 0.5*np.random.dirichlet(np.ones(n))
        # Ensure non-neg and sum=1 (should)
        a /= a.sum()
        # Precompute rev
        a_rev = a[::-1].copy()
        size1 = _next_pow2(2*n-1)
        size2 = _next_pow2(3*n-2)
        # Adam parameters
        beta1=0.9; beta2=0.999; eps=1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        # temperature schedule
        # initial tau = max_b maybe
        # we will compute it each iteration
        tau_factor = 1.0  # set tau as tau_factor * max_b
        tau_decay = 0.9995
        # base learning rate for Adam
        alpha0 = 0.05   # moderate learning rate
        it = 0
        while time.time() < deadline:
            # Compute convolution
            conv_full = _full_convolution(a, size1)   # length 2n-1
            max_b = conv_full.max()
            # compute objective
            cur_score = 2.0 * n * max_b
            # track best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[restart {restart} iter {it}] new best = {best_score:.6f}  time={elapsed:.1f}s")
            # softmax weights
            tau = max(tau_factor * max_b, 1e-12)
            # update tau factor
            tau_factor *= tau_decay
            # compute weights
            w = np.exp((conv_full - max_b) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            # gradient of smooth max: grad = 2 * conv(a_rev, w)
            g_full = _conv_fft(a_rev, w, size2)
            # central slice length n
            g = g_full[n-1:n-1+n]
            direction = -2.0 * g   # descent direction
            # optionally subtract mean to keep sum easier
            # direction -= direction.mean()
            # Adam update
            it += 1
            # update moments
            m = beta1 * m + (1 - beta1) * direction
            v = beta2 * v + (1 - beta2) * (direction**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            step = alpha0 / (np.sqrt(it) + 1.0)
            # apply Adam scaling
            a = a + step * (m_hat / (np.sqrt(v_hat) + eps))
            # projection onto non-negative simplex
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.ones_like(a) / n
            else:
                a /= s
            # update reversed version
            a_rev = a[::-1].copy()
        # after time ends for this restart, loop to next
    # final best_seq
    if best_seq is None:
        # fallback: uniform
        best_seq = np.ones(n) / n
    total = time.time() - start
    print(f"[done] best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get('n', 6000))

    # Precompute FFT size
    size1 = _next_pow2(2*n -1)
    size2 = _next_pow2(3*n -2)

    # best tracking
    best_seq = None
    best_score = float('inf')

    # define possible init functions list
    init_fns = [
        lambda: _cantor_histogram(n, L=14),
        lambda: np.random.dirichlet(np.ones(n)),
        lambda: np.array([0.5 if i in (0,n-1) else 0.0 for i in range(n)])  # maybe not good
    ]
    # maybe only first two

    restart = 0
    while time.time() < deadline and restart < len(init_fns):
        # init
        a = init_fns[restart]()
        a = np.clip(a, 0.0, None)
        if a.sum() ==0:
            a = np.ones(n)/n
        else:
            a = a / a.sum()
        # enforce symmetry
        a = (a + a[::-1]) / 2
        a = a / a.sum()

        # optimizer parameters
        beta1=0.9; beta2=0.999; eps=1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        # dynamic temperature factor: start with tau_factor=1.0, decays
        tau_factor = 1.0
        tau_decay = 0.999

        # base learning rate due to Adam
        alpha = 0.1

        it = 0
        last_improve_time = time.time()
        while time.time() < deadline:
            # compute convolution
            conv_full = _full_convolution(a, size1)
            max_b = conv_full.max()
            cur_score = 2.0 * n * max_b
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_improve_time = time.time()
                print(f"[restart {restart} it {it}] new best = {best_score:.6f}")

            # temperature
            tau = max(tau_factor * max_b, 1e-12)
            tau_factor *= tau_decay

            # compute softmax weights
            w = np.exp((conv_full - max_b) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # gradient
            g_full = _conv_fft(a[::-1], w, size2)
            g = g_full[n-1:n-1+n]
            grad = 2.0 * g   # gradient of smooth max

            # Adam moments
            it += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)

            # step
            a = a - alpha * (m_hat / (np.sqrt(v_hat) + eps))

            # projection onto nonnegative simplex + enforce symmetry
            a = np.clip(a, 0.0, None)
            # enforce symmetry before renorm
            a = (a + a[::-1]) / 2.0
            s = a.sum()
            if s == 0.0:
                a = np.ones_like(a) / n
            else:
                a /= s

            # optional random jitter to escape local minima after many iterations
            if it % 20000 == 0:
                # small perturbation
                perturb = np.random.rand(n) * 0.01
                a = a + perturb
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                a = (a + a[::-1]) / 2.0
                a /= a.sum()
            # break condition maybe if too many iterations without improvement?
            if time.time() - last_improve_time > 30.0:   # 30 sec of stagnation
                # restart early
                break
        restart += 1

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = 2.0 * n * (1.0/n)   # 2

    total = time.time() - start
    print(f"[done] total time {total:.1f}s, best_score {best_score:.6f}")
    return best_seq.tolist()

"""
Gradient‑Adam search on a smooth approximation of the ℓ∞‑norm of a*a.

The algorithm
---------------
1. Initialise a probability vector `a` (non‑negative, sum = 1) – either the
   Cantor‑measure histogram or a random Dirichlet draw.
2. Repeatedly minimise the smooth‑max
        fτ(a)=τ·log∑_k exp((a∗a)_k/τ)
   using an Adam optimiser.  The gradient of fτ is
        ∇fτ = 2·conv_rev(w)  with  w∝exp((a∗a−max)/τ) .
3. After each Adam step the vector is projected onto the simplex
   (clip ≥ 0 and renormalise).  Symmetry a_i=a_{n‑1‑i} is enforced – the
   problem is invariant under reversal, so restricting to symmetric
   vectors does not lose optimality.
4. Several short restarts (different seeds) are run within the time
   budget; the best sequence (lowest value of evaluate_sequence) is kept.
5. The final sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution by FFT.
    ``size`` must be ≥ len(u)+len(v)-1.
    The returned array has length len(u)+len(v)-1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Histogram of the middle‑third Cantor measure on ``n`` bins.
    Returns a non‑negative vector that already sums to 1.
    """
    N = 1 << L                    # number of atoms = 2**L
    # ternary positions of each atom
    powers = 3.0 ** np.arange(1, L + 1)                # (L,)
    idx = np.arange(N, dtype=np.uint32)[:, None]       # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N,L)
    pos = (bits * (2.0 / powers)).sum(axis=1)          # (N,)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)            # normalise → sum == 1
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and renormalise to sum = 1 (simple Euclidean projection)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.size
    return v / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin – a tiny bit of time left for finalisation
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # Pre‑compute FFT sizes (fixed for the whole run)
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)      # for a*a
    size2 = _next_pow2(3 * n - 2)      # for a_rev * w

    # --------------------------------------------------------------
    # Tracking of the best solution
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Initialisers (first one is the Cantor measure, the others are random)
    # --------------------------------------------------------------
    init_fns = [
        lambda: _cantor_histogram(n, L=14),
        lambda: np.random.dirichlet(np.ones(n)),
        lambda: _cantor_histogram(n, L=13) * 0.7 + np.random.dirichlet(np.ones(n)) * 0.3,
    ]

    # --------------------------------------------------------------
    # Optimisation loop – one pass per initialiser
    # --------------------------------------------------------------
    for restart, init_fn in enumerate(init_fns):
        if time.time() >= deadline:
            break

        # ----- initialise -------------------------------------------------
        a = init_fn()
        a = _project_simplex(a)
        # enforce symmetry (a_i = a_{n-1-i}) – the problem is symmetric
        a = (a + a[::-1]) / 2.0
        a = _project_simplex(a)

        # Adam state
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow, beta2_pow = 1.0, 1.0

        # temperature handling for the soft‑max
        tau_factor = 1.0          # τ = tau_factor * max_b
        tau_decay = 0.9995

        # learning rate for Adam
        alpha = 0.12

        # iteration counter and stagnation timer
        it = 0
        last_improve = time.time()

        while time.time() < deadline:
            # ----- convolution a*a and objective -----
            conv_full = _full_convolution(a, size1)          # length 2n‑1
            max_b = conv_full.max()
            cur_score = 2.0 * n * max_b                     # sum(a)=1 by construction

            # ----- record best -----
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.time()
                elapsed = last_improve - start
                print(f"[restart {restart} it {it}] new best = {best_score:.6f}  time={elapsed:.1f}s")

            # ----- soft‑max weights (smooth max) -----
            tau = max(tau_factor * max_b, 1e-12)
            tau_factor *= tau_decay
            w = np.exp((conv_full - max_b) / tau)           # stability via shift by max_b
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # ----- gradient of the smooth‑max -----
            a_rev = a[::-1]
            g_full = _conv_fft(a_rev, w, size2)            # length 3n‑2
            # central n entries correspond to derivative of the full convolution
            g = g_full[n - 1 : n - 1 + n]
            grad = 2.0 * g                                 # ∇fτ (positive direction)

            # ----- Adam update -----
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            step = alpha * (m_hat / (np.sqrt(v_hat) + eps))
            a = a - step                                   # descent step

            # ----- projection onto the simplex + symmetry enforcement -----
            a = _project_simplex(a)
            a = (a + a[::-1]) / 2.0                         # keep a symmetric
            a = _project_simplex(a)

            # ----- occasional random jitter to escape plateaus -----
            if it % 20000 == 0:
                jitter = np.random.rand(n) * 0.005
                a = a + jitter
                a = _project_simplex(a)
                a = (a + a[::-1]) / 2.0
                a = _project_simplex(a)

            # ----- early‑stop this restart if we have been idle for a while -----
            if time.time() - last_improve > 30.0:
                # no improvement for 30 s → start a new restart
                break

        # end of this restart
        # (the loop will automatically start the next initialiser)

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = 2.0 * n * (1.0 / n)  # uniform bound = 2.0

    total = time.time() - start
    print(f"[done] total time={total:.1f}s, best score={best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
