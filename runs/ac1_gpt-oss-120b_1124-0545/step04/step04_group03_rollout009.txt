"""
Hybrid optimizer for the step‑function problem.

The algorithm works in four phases:
1️⃣ Analytic grid search on two power‑law families (product and sum) to obtain a
   good starting distribution on the probability simplex.
2️⃣ Smooth‑max optimisation of  Φ_λ(a)= (1/λ)·log Σ_k exp(λ·(a*a)_k)  using a
   continuation on λ and L‑BFGS‑B (or SciPy’s interior‑point if unavailable).
3️⃣ Sub‑gradient descent on the *exact* objective  max(conv(a,a))  (the
   non‑smooth part) with Euclidean projection onto the simplex.
4️⃣ Upsampling of the final distribution to a larger length (default ×2) to
   reduce discretisation error, followed by a short final refinement.

All intermediate candidates are kept; the best one seen before the time
budget expires is returned as a plain list of non‑negative floats (sum≈1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no globals)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    # rfft works on real input, returning the positive frequency part
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability simplex vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector a (sum(a)=1).
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # a is already normalised – no need to re‑scale
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (non‑negative,
    sum = 1).  Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_star = rho[-1]
        theta = cssv[rho_star] / (rho_star + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+δ)*(n‑1‑i+δ))^{‑p}
    mode = 'sum'      → a_i ∝ (i+δ)^{‑p} + (n‑1‑i+δ)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smooth‑max objective Φ_λ(a) and its gradient w.r.t. the unconstrained
    parameter vector x (where a = softmax(x)).
    """
    # a = softmax(x)
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # Convolution a*a
    conv = conv_fft_self(a)                   # length 2n‑1
    max_c = np.max(conv)

    # Smooth max via log‑sum‑exp
    shift = conv - max_c                       # improve numeric stability
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam        # Φ_λ

    # Weights of the soft‑max over convolution entries
    w = exp_term / sum_exp                     # sum(w)=1

    # Gradient of Φ_λ w.r.t. a (∂Φ/∂a).  It is 2 * (a * w) reversed convolution.
    # The reversed convolution implements the matrix product with Toeplitz matrix.
    w_rev = w[::-1]                           # reverse
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    # extract the n entries that correspond to ∂Φ/∂a_i
    slice_part = conv_wa[n - 1: 2 * n - 1]     # length n
    grad_a = 2.0 * slice_part[::-1]

    # Map gradient to x-space via the Jacobian of the soft‑max
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return phi, grad_x


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Tiny random mass transfer between two coordinates, followed by simplex projection."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    Strategy:
    * Analytic grid search over two power‑law families.
    * Smooth‑max L‑BFGS optimisation with a geometric λ‑schedule.
    * Sub‑gradient descent on the true max‑convolution.
    * Optional up‑sampling to a larger length for reduced discretisation.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # User‑controllable parameters
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2000))          # initial length for the search
    final_factor = int(kwargs.get("final_factor", 2))   # up‑sample factor
    final_n = n * final_factor if final_factor > 1 else n

    # --------------------------------------------------------------
    # 1) Analytic grid search – quick way to obtain a solid start
    # --------------------------------------------------------------
    candidates = []

    # product and sum families – wide but cheap grid
    modes = ("product", "sum")
    p_vals = np.arange(0.35, 0.71, 0.05)            # 0.35 .. 0.70
    off_vals = np.arange(0.0, 5.1, 0.5)             # 0 … 5
    for mode in modes:
        for p in p_vals:
            for off in off_vals:
                # for product mode offset=0 is allowed (singular edges become 0)
                cand = analytic_candidate(n, p, off, mode)
                candidates.append(cand)

    # uniform distribution (baseline)
    candidates.append(np.ones(n) / n)

    # use the globally supplied construction if it matches the length
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Continuation L‑BFGS on the smoothed max‑convolution
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # λ schedule – start smooth, end very sharp
    lam_vals = [0.5, 2.0, 5.0, 10.0, 20.0, 50.0,
                100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]

    # Try to use SciPy’s L‑BFGS‑B; fall back to a simple Adam optimiser
    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:                     # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – will use a simple Adam optimizer instead.")

    for lam in lam_vals:
        if time.time() >= deadline:
            break

        if have_scipy:
            # allocate iterations proportionally to the remaining time
            remain = deadline - time.time()
            # each iteration costs O(n log n), we allow up to 3000 iter if time permits
            max_iter = int(min(3000, max(30, remain * 80)))
            res = minimize(compute_phi_and_grad,
                           x,
                           args=(lam, n),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter,
                                    'disp': False,
                                    'gtol': 1e-12})
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.0f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # ------------------------------------------------------------------
            # Simple Adam fallback (much slower per iteration but works)
            # ------------------------------------------------------------------
            alpha = 0.05
            beta1, beta2 = 0.9, 0.999
            eps_opt = 1e-8
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t_counter = 0
            while time.time() < deadline:
                phi, grad_x = compute_phi_and_grad(x, lam, n)
                t_counter += 1
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t_counter)
                v_hat = v / (1 - beta2 ** t_counter)
                step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam lam={lam:.0f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t_counter >= 2000:
                    break

    # --------------------------------------------------------------
    # 3) Sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    a = best_a.copy()
    step = 0.5               # initial step size (will be adapted)
    while time.time() < deadline:
        conv = conv_fft_self(a)
        jstar = int(np.argmax(conv))

        # gradient of (a*a)_{jstar} w.r.t. a:
        grad = np.zeros_like(a)
        # overlapping index range: i such that 0 <= jstar-i < n
        i_start = max(0, jstar - (n - 1))
        i_end = min(n - 1, jstar)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[jstar - idx]

        # gradient step + simplex projection
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            elapsed = time.time() - start_time
            print(f"[subgrad] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            step = max(step * 0.85, 1e-6)   # shrink after improvement
        else:
            # slowly increase step to escape plateaus
            step = min(step * 1.02, 2.0)

        a = a_new

    # --------------------------------------------------------------
    # 4) Upsampling – halve discretisation error
    # --------------------------------------------------------------
    if final_n > n:
        # linear interpolation of the best distribution onto a finer grid
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0, n - 1, final_n, dtype=float)
        a_up = np.interp(x_new, x_old, best_a)
        # re‑normalise to sum=1
        a_up = a_up / a_up.sum()
        # a brief refinement on the larger grid (optional, cheap)
        a = a_up.copy()
        step = 0.2
        # allow a few sub‑gradient steps if time permits
        while time.time() < deadline:
            conv = conv_fft_self(a)
            jstar = int(np.argmax(conv))
            grad = np.zeros_like(a)
            i_start = max(0, jstar - (final_n - 1))
            i_end = min(final_n - 1, jstar)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] = 2.0 * a[jstar - idx]
            a_new = a - step * grad
            a_new = project_simplex(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                elapsed = time.time() - start_time
                print(f"[up‑subgrad] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                step = max(step * 0.85, 1e-8)
            else:
                step = min(step * 1.01, 0.5)
            a = a_new
        # final up‑sampled vector to be returned
        final_seq = best_a.tolist()
    else:
        final_seq = best_a.tolist()

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return final_seq
