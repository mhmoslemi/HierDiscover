    best_seq = a.copy()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2
    tau_factor = 0.8
    tau_decay = 0.9997
    smooth_iters = 200000  # can be higher but limited by time.

    it = 0
    last_improve_time = time.time()

    while time.time() < deadline:
        # compute conv
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2*n*max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_time = time.time()
            print(f"[{it}] new best = {best_score:.6f} time={last_improve_time - start:.1f}s")

        # adapt tau
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # compute weight w
        if it < smooth_iters:
            # softmax
            w = np.exp((conv_full - max_b) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
        else:
            # hard indicator near max
            delta = max_b * 0.001 * (0.5 ** ((it - smooth_iters)//10000))  # decreasing
            mask = conv_full >= max_b - delta
            count = mask.sum()
            if count == 0:
                w = np.ones_like(conv_full) / conv_full.size
            else:
                w = mask.astype(np.float64) / count

        # gradient via convolution
        g_full = _conv_fft(a_rev, w, size2)
        g = g_full[n-1 : n-1 + n]
        grad = -2.0 * g
        # optionally remove mean to preserve sum
        grad -= grad.mean()

        # Adam updates
        it += 1
        lr = base_lr / np.sqrt(it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection

    total = time.time() - start

"""
Propose a low‑value non‑negative step‑function sequence.

Algorithm
---------
1. Initialise the vector from a Cantor‑measure histogram (sum = 1).
2. Optimise the sequence with a two‑phase Adam‑style projected gradient
   descent on a smooth approximation of the L∞‑norm of the self‑convolution:
   * Phase 1 (smooth): use a soft‑max weighting of the convolution entries.
   * Phase 2 (hard): focus on the entries that are within a shrinking
     tolerance of the current maximum (sub‑gradient of the true max).
3. After each gradient step:
   * Apply Adam momentum (β₁=0.9, β₂=0.999).
   * Project onto the probability simplex (clip ≥0, renormalise).
   * Enforce symmetry aᵢ = a_{n‑1‑i} – symmetry reduces the peak of the
     convolution for the same mass distribution.
4. If no improvement has been seen for a while, add a tiny random
   perturbation to escape possible plateaus.
5. Keep the best sequence encountered (according to `evaluate_sequence`).

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT.
    ``size`` must be ≥ len(u)+len(v)-1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a * a (full convolution) using FFT."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the (1/3) Cantor measure on ``n`` bins.
    Returns a non‑negative vector summing to 1.
    """
    N = 1 << L  # number of atoms = 2**L
    # indices of atoms in binary form, shape (N, L)
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    # ternary scaling factors 2/3**k
    powers = 3.0 ** np.arange(1, L + 1)
    pos = (bits * (2.0 / powers)).sum(axis=1)  # positions in [0,1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)
    return a


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the probability simplex.
    Simple method: clip to ≥0 then renormalise (sum must be >0).
    """
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s == 0.0:
        return np.ones_like(x) / x.size
    return x / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # Initialisation (Cantor histogram or fallback)
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            a = np.asarray(seq0, dtype=float)
            if a.size != n:
                # Resample linearly to the desired length.
                x_old = np.linspace(0, 1, a.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, None)
            if a.sum() < 0.01:
                raise ValueError
            a /= a.sum()
        else:
            raise KeyError
    except Exception:
        a = _cantor_histogram(n, L=14)

    # --------------------------------------------------------------
    # Pre‑compute FFT sizes (constant throughout the run)
    # --------------------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)   # for a*a
    size_grad = _next_pow2(3 * n - 2)   # for a_rev * weight

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv_full = _full_convolution(a, size_conv)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b      # because sum(a) == 1
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # --------------------------------------------------------------
    # Hyper‑parameters for the two‑phase schedule
    # --------------------------------------------------------------
    base_lr = 0.2                       # base learning rate
    tau_factor = 0.8                    # τ = tau_factor * max_b
    tau_decay = 0.9997                  # exponential decay of τ
    smooth_iters = 200000               # iterations spent in soft‑max phase
    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- evaluate current objective -----
        conv_full = _full_convolution(a, size_conv)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- temperature for smooth‑max -----
        tau = max(tau_factor * max_b, 1e-14)
        tau_factor *= tau_decay

        # ----- compute weight vector w -----
        if it < smooth_iters:
            # Soft‑max weighting (smooth approximation of max)
            w = np.exp((conv_full - max_b) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
        else:
            # Hard sub‑gradient: focus on entries near the maximum
            # Tolerance shrinks geometrically.
            decay_steps = (it - smooth_iters) // 5000
            delta = max_b * (0.5 ** decay_steps) * 0.001
            mask = conv_full >= (max_b - delta)
            cnt = mask.sum()
            if cnt == 0:
                w = np.ones_like(conv_full) / conv_full.size
            else:
                w = mask.astype(np.float64) / cnt

        # ----- gradient via convolution a_rev * w -----
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n - 1 : n - 1 + n]          # central n entries
        grad = -2.0 * g
        # Remove the mean component to keep Σa unchanged (project onto
        # the hyperplane Σa = 1).
        grad -= grad.mean()

        # ----- Adam update -----
        it += 1
        lr = base_lr / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # Bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_onto_simplex(a_new)

        # ----- enforce symmetry (a_i = a_{n-1-i}) -----
        a_new = (a_new + a_new[::-1]) * 0.5

        a = a_new
        a_rev = a[::-1].copy()

        # ----- occasional random perturbation if stuck -----
        if it % 100000 == 0:
            # If we haven't improved for >30 s, add tiny noise.
            if time.time() - last_improve > 30.0:
                rng = np.random.default_rng()
                pert = rng.normal(scale=1e-3, size=n)
                a = _project_onto_simplex(np.clip(a + pert, 0.0, None))
                a_rev = a[::-1].copy()
                # Reset Adam moments to avoid stale directions.
                m.fill(0.0)
                v.fill(0.0)
                print(f"[{it}] perturbation applied after stagnation")

    # --------------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
