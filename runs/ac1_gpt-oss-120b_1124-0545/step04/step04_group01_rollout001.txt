"""
Propose a step‑function with a very low value of

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines several complementary ideas:

1. **Diverse analytic seeds** – uniform, geometric, power‑law,
   beta‑like, sine‑window and a few random Dirichlet draws.  Seeds are
   l₁‑normalised, so the objective reduces to ``2·n·max(conv)``.

2. **Fast sub‑gradient mirror descent** – an exponentiated‑gradient
   (EG) iteration on the true sub‑gradient of the max‑convolution.
   This quickly drives the max down while staying on the probability
   simplex.

3. **Linear‑programming improvement step** – from the current
   coefficients ``a`` we build a linear model of the convolution
   (first‑order Taylor) and solve a small LP that finds a feasible
   direction ``d`` reducing the maximum convolution value.  A
   back‑tracking line search ensures a genuine improvement.

4. **Stochastic hill‑climbing** – tiny random mass‑transfer moves
   between two positions are accepted only if they improve the
   objective, allowing the optimizer to escape shallow local minima.

All phases respect the global time budget (default 1000 s).  The best
sequence found (as a plain list of floats) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sin_pow(n: int, p: float) -> np.ndarray:
    """sin‑window raised to a power p (p≥1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** p
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the exact sub‑gradient
    of the max‑convolution.
    Returns the best sequence seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 1.0 / np.sqrt(it)          # a bit more aggressive than 0.6/√it
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


def _lp_improve(a: np.ndarray,
                n: int,
                deadline: float,
                topk: int = 200,
                max_trials: int = 5) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    One linear‑programming improvement step.
    Builds a first‑order linear model of the convolution at the
    current worst indices (top‑k) and solves a small LP for a feasible
    direction that reduces the maximal entry.
    Returns a new sequence and its objective if an improvement was found.
    """
    conv = _conv_fft(a, a)
    old_max = np.max(conv)

    # pick indices with the largest convolution values
    idx_desc = np.argsort(-conv)[:topk]
    # make sure the absolute maximum is present
    max_idx = int(np.argmax(conv))
    if max_idx not in idx_desc:
        idx_desc[-1] = max_idx

    # build constraint matrix A (topk × n)
    rows = []
    for i in idx_desc:
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        row = np.zeros(n, dtype=np.float64)
        k = np.arange(start, end + 1, dtype=int)
        row[k] = 2.0 * a[i - k]
        rows.append(row)
    A = np.stack(rows, axis=0)               # shape (topk, n)

    # Linear program:
    #   minimise   t
    #   subject to A @ d - t <= -conv[idx_desc]
    #              sum(d) = 0
    #              d_i >= -a_i
    # variables: d (n) and t (1)
    K = A.shape[0]
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = -conv[idx_desc]

    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([0.0])

    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]

    # objective vector: minimise t
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None

    if not res.success:
        return None, None

    d = res.x[:n]
    # direction may be numerically tiny – skip if useless
    if np.linalg.norm(d) < 1e-12:
        return None, None

    # line‑search on the direction (full step, then halves)
    for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
        a_new = a + alpha * d
        # feasibility is guaranteed by the bounds, but guard against tiny negatives
        if np.any(a_new < -1e-12):
            continue
        # (sum stays 1 because sum(d)=0)
        conv_new = _conv_fft(a_new, a_new)
        new_max = np.max(conv_new)
        if new_max < old_max - 1e-12:
            new_val = 2.0 * n * new_max
            print(f"[LP] n={n:5d} α={alpha:.3f} old={old_max:.6f} new={new_max:.6f}")
            return a_new, new_val

    return None, None


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a tiny amount of mass between
    two random indices and keep the move if it improves the objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.random() < 0.5 else (j, i)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest but diverse set of normalised seeds."""
    rng = np.random.default_rng(2023)
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes – a richer grid (including very U‑shaped)
    for a in (0.15, 0.3, 0.5, 0.7, 1.0):
        for b in (0.15, 0.3, 0.5, 0.7, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # sine‑power windows (sharp edges)
    for p in (1.0, 2.0, 4.0):
        seeds.append(_sin_pow(n, p))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few random soft‑max vectors
    for _ in range(4):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse a previously discovered construction if its length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ensure all seeds are proper probabilities
    final = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        totals = s.sum()
        if totals > 0:
            final.append(s / totals)
    return final

        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # Verify (optional, ignore any error)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
