"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that
minimises ``evaluate_sequence``.  The routine combines three ideas:

1. **Fast parametric sweep** – simple power‑law, exponential and linear
   families are tried quickly; the best candidate (or the global
   ``height_sequence_1`` if present) is taken as a warm‑start.

2. **Deterministic annealing** – minimise a smooth approximation of the
   non‑convex objective

       f(a) = max_k (a∗a)[k]

   by using the soft‑max (log‑sum‑exp) of the convolution values.
   The gradient of the smoothed objective is obtained analytically
   (a convolution of the current vector with the soft‑max weights) and
   an exponentiated‑gradient step keeps the iterate on the probability
   simplex automatically.

3. **Fine‑grained sub‑gradient phase** – once the annealing temperature
   has grown large the smooth gradient becomes the true sub‑gradient of
   the max‑convolution; a few extra Euclidean‑projected sub‑gradient
   steps are performed to polish the solution.

The algorithm respects the time budget (default 1000 s), logs progress
and finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """Normalised decreasing power‑law a_i ∝ (i+1+offset)^(-α)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """Normalised exponential decay a_i ∝ r^{i‑offset} for i ≥ offset."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Simple decreasing linear sequence (triangular shape)."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return the soft‑max (log‑sum‑exp) weights for a convolution vector.
    The maximum entry receives weight 1/Σexp(·), all others smaller.
    """
    m = conv.max()
    # shift by the maximum for numerical stability
    exp_vals = np.exp(beta * (conv - m))
    w = exp_vals / exp_vals.sum()
    return w


def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective  (1/β)·log Σ exp(β·conv)
    with respect to the probability vector a.
    Formula: g_j = 2·Σ_k w_k·a_{k‑j}.
    Implemented via a single FFT‑based convolution.
    """
    n = a.size
    # reverse the weights for the proper linear convolution
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)               # length = 3n‑2
    # pick the slice that corresponds to  Σ_k w_k·a_{k‑j}
    idx = 2 * n - 2 - np.arange(n)               # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]
    return grad


def _objective_from_a(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a normalised vector (sum=1)."""
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()

        max_abs = np.max(np.abs(step))
        if max_abs > 20.0:
            step = step * (20.0 / max_abs)

        a = a * np.exp(step)                      # element‑wise
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- schedules ----------------------------------------------------
        eta = max(eta * eta_decay, eta_min)
        beta = min(beta * beta_mul, beta_max)

        # ---- occasional random mass exchange to help escape plateaus -----
        if iter_cnt % 97 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # transfer at most 25 % of source mass, limited by current step size
            delta = min(a[i] * 0.25, eta * 0.3)
            if delta > 1e-12:
                a[i] -= delta
                a[j] += delta
                a /= a.sum()

    # ------------------------------------------------------------------
    # 3. Optional fine‑grained sub‑gradient polishing (max‑convolution)
    # ------------------------------------------------------------------
    # Switch to the exact sub‑gradient (only the maximal index) for a few
    # thousand extra steps – this often squeezes a few more hundredths.
    if time.time() < deadline:
        lr = 0.8
        lr_decay = 0.9999
        lr_min = 1e-6
        momentum = np.zeros_like(a)

        while time.time() < deadline:
            conv = _conv_fft(a, a)
            k_star = int(conv.argmax())
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[polish] iter={iter_cnt} score={best_score:.6f}")

            if best_score < 1.503:
                print("[target reached during polishing]")
                break

            # sub‑gradient of the max‑convolution:
            idx = k_star - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

            # momentum and projection
            momentum = 0.9 * momentum + grad
            a = a - lr * momentum
            a = _project_onto_simplex(a)

            lr = max(lr * lr_decay, lr_min)

            iter_cnt += 1
            if iter_cnt % 2000 == 0:
                # occasional random perturbation to avoid stalling
                i, j = np.random.choice(n, 2, replace=False)
                delta = min(a[i] * 0.3, lr * 0.5)
                if delta > 1e-12:
                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 4. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} (n={best_seq.size}) "
          f"total_iterations={iter_cnt}")
    return best_seq.tolist()
