def compute_gradient_smooth(a, beta):
    n = len(a)
    conv = conv_fft(a)                # length 2n-1
    maxc = conv.max()
    # compute w as softmax (shifted) 
    exp_term = np.exp(beta * (conv - maxc))
    w = exp_term / exp_term.sum()
    # compute gradient g = 2 * (w * a_rev) convolution
    a_rev = a[::-1]
    # conv of w (len 2n-1) and a_rev (len n)
    size = _next_pow2(len(w) + len(a_rev) - 1)
    fft_w = np.fft.rfft(w, n=size)
    fft_rev = np.fft.rfft(a_rev, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_rev, n=size)[:len(w)+len(a_rev)-1]

    return conv, g

The algorithm consists of:
1. A dense analytic grid of parameterised families (product, sum, exponential,

2. Upsampling the best shape to a larger support (default factor 2) and a
   projected Adam optimisation of a smooth approximation of the max‑convolution.
   The smooth maximum is realised with a temperature‑controlled soft‑max, which
   yields a well‑behaved gradient.
3. A short stochastic refinement with Gaussian perturbations.

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]

    """
    Exact objective for a non‑negative vector a that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

    """a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}  (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:

    return a

    """a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(π·i/(n‑1))|^{p} (peaks at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()

    i = np.arange(n, dtype=np.float64)

    return a / a.sum()

    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a that already sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(π·i/(n‑1))|^{p} (peaks at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point (called by the harness)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine runs an analytic grid, a smooth‑max Adam optimiser on an
    up‑sampled vector and a final stochastic refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # 1) Analytic grid search on a modest size (base_n)
    # --------------------------------------------------------------
    base_n = int(kwargs.get("n", 2000))
    best_seq = None
    best_score = np.inf

    print("[init] Analytic grid search, base_n =", base_n)

    # product family
    for p in np.linspace(0.2, 2.0, 10):
        for off in np.linspace(0.01, 2.0, 10):
            cand = analytic_product(base_n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # sum family
    for p in np.linspace(0.2, 2.0, 10):
        for off in np.linspace(0.01, 2.0, 10):
            cand = analytic_sum(base_n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # exponential tail
    for lam in np.logspace(-3, 1, 12):
        cand = analytic_exponential(base_n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # cosine‑power
    for p in np.arange(0.5, 6.1, 0.5):
        cand = analytic_cosine(base_n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # beta‑shape
    for a in np.arange(-0.5, 2.01, 0.5):
        for b in np.arange(-0.5, 2.01, 0.5):
            cand = analytic_beta(base_n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # optional global reference
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == base_n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()
                print("[init] adopted global reference, score =", sc)

    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Upsample and smooth‑max Adam optimisation
    # --------------------------------------------------------------
    up_factor = 2
    target_n = base_n * up_factor
    # keep size reasonable (avoid >10 000 entries)
    if target_n > 10000:
        target_n = base_n

    if target_n != base_n:
        a = np.repeat(best_seq, up_factor) * (1.0 / up_factor)
        n = target_n
        a = a / a.sum()
    else:
        a = best_seq.copy()
        n = base_n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_seq_full = a.copy()
    best_score_full = compute_score(a)
    last_improve = start_time
    stagnation_time = 12.0                # seconds without improvement → jitter
    iteration = 0

    # linear schedules for step‑size and temperature
    step_start, step_end = 0.4, 1e-5
    temp_start, temp_end = 0.5, 200.0

    while time.time() < deadline:
        # current convolution and objective
        conv = conv_fft(a)                 # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best found so far
        if cur_score < best_score_full - 1e-12:
            best_score_full = cur_score
            best_seq_full = a.copy()
            last_improve = time.time()
            print(f"[{iteration:07d}] new best = {best_score_full:.6f}")

        # progress‑based hyper‑parameters
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 5.0)
        step = step_start * (1.0 - prog) + step_end * prog
        temp = temp_start * (1.0 - prog) + temp_end * prog

        # ---- smooth max (soft‑max) weights ----
        maxc = cur_max
        exp_arg = temp * (conv - maxc)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # ---- gradient of the smooth objective ----
        a_rev = a[::-1]
        size = _next_pow2(len(w) + len(a_rev) - 1)
        fft_w = np.fft.rfft(w, n=size)
        fft_rev = np.fft.rfft(a_rev, n=size)
        conv_wa = np.fft.irfft(fft_w * fft_rev, n=size)[: len(w) + len(a_rev) - 1]
        g = 2.0 * conv_wa[n - 1 : n - 1 + n]   # gradient w.r.t. a

        # ---- Adam update (projected) ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step_vec = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step_vec
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        a = cand

        # occasional tiny random transfer to keep exploration alive
        if np.random.rand() < 0.015:
            a = random_move(a, step * 0.7)

        # jitter restart if we have been stuck for a while
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 2e-4
            a = best_seq_full + jitter
            a = project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # rare full random restart
        if np.random.rand() < 0.004:
            rand_vec = np.random.rand(n)
            rand_vec /= rand_vec.sum()
            a = rand_vec
            m.fill(0.0)
            v.fill(0.0)

    # --------------------------------------------------------------
    # 3) Fine‑grained stochastic refinement (final few seconds)
    # --------------------------------------------------------------
    refine_deadline = start_time + budget_s - 1.5
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=4e-4, size=n)
        cand = best_seq_full + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        conv = conv_fft(cand)
        cand_score = 2.0 * n * np.max(conv)
        if cand_score < best_score_full:
            best_score_full = cand_score
            best_seq_full = cand.copy()
            print(f"[refine] improved to {best_score_full:.6f}")

    total = time.time() - start_time
    print(f"[done] best score = {best_score_full:.6f}, iters = {iteration}, "
          f"time = {total:.1f}s, length = {len(best_seq_full)}")
    return best_seq_full.tolist()
