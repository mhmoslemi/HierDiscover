"""
Propose candidate – projected sub‑gradient descent + simulated annealing.

The evaluation function is scale‑invariant, so we work with a probability
vector (∑a_i = 1, a_i ≥ 0).  The exact objective for a vector a of length n
is

    f(a) = 2·n·max_k (a * a)[k]

where “*” denotes the linear convolution.  The gradient of the
max‑convolution entry is known in closed form:
if the maximal index is k*, then ∂max/∂a_i = 2·a_{k*‑i} (when the index
k*‑i lies in the valid range, otherwise 0).  Using this sub‑gradient we
perform a projected sub‑gradient descent on the simplex (Euclidean
projection).  The step size decays like η_t = η0/√t.  Whenever no
improvement is seen for a while we restart from a fresh Dirichlet
sample, keeping the best vector found so far.

When the time budget for the first stage is exhausted we switch to a
simple stochastic refinement (simulated annealing) that moves a small
fraction of mass between two random coordinates.  The temperature is
cooled linearly; a move is always accepted when it improves the exact
objective and otherwise with the Metropolis probability.

The whole procedure respects the 1000 s budget, prints a few progress
messages and finally returns the best sequence as a plain Python list.
"""
import time
import numpy as np

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, ∑ x_i = 1}.
    O(N log N) implementation from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying u_ρ > (cssv_ρ) / ρ
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max(a * a) using FFT (linear convolution)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv.max()


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) w.r.t. the vector a.
    If the maximum is attained at several indices we average the
    corresponding sub‑gradients.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to collect all maximal entries (relatively tight)
    eps = 1e-14
    k_idxs = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    for k in k_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    grad /= max(1, len(k_idxs))          # average if multiple maxima
    return grad


def initialise_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a reasonable starting probability vector.
    Preference order:
      1. reuse a previously found good sequence (global variable)
      2. classic arcsine shape
      3. random Dirichlet sample (fallback)
    """
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            return a0

    # arcsine shape (inverse‑sqrt)
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0

                print(f"[{it:08d}] restart with fresh Dirichlet (t={time.time()-start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Stage 2 – stochastic local refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear cooling schedule
        T_start = best_score * 0.006      # 0.6 % of current score
        T_end = best_score * 1e-6

        sa_it = 0
        phase_start = time.time()
        while time.time() < overall_deadline:
            # current temperature
            phase_elapsed = time.time() - phase_start
            phase_frac = min(1.0, phase_elapsed / remaining)
            T = T_start * (1.0 - phase_frac) + T_end * phase_frac

            # propose a small mass transfer between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move at most 5 % of the source mass
            max_move = min(a[i], 0.05 * a[i] + 1e-14)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact objective for the proposal
            new_max = compute_conv_max(a_new, fft_len)
            new_score = 2.0 * n * new_max

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-15)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Final sanity checks and output
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        # Should never happen – fall back to uniform distribution
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed = time.time() - start_time
    print(f"[done] total time={elapsed:.1f}s, best score={best_score:.6f}, length={n}")
    # make the result available for future runs
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
