assistantfinal```python
"""
Propose a non‑negative step‑function that (approximately) solves

    min 2·n·max(conv(a,a)) / (sum(a)²)

by a three‑stage hybrid:

1️⃣  Alternating linear‑program updates (the “LP‑step” from the
    reference paper).  For a current vector f we maximise Σg
    subject to the linear constraints (f*g) ≤ M where M = max(f*f).
    The solution g is rescaled to unit sum and mixed with f.
    A few cheap LP iterations already push the shape far below the
    uniform bound.

2️⃣  Continuous optimisation on the probability simplex.
    The vector (now of the target length, e.g. 4000) is optimised
    with Adam on a smooth log‑sum‑exp approximation of the max,
    followed by an exact sub‑gradient (mirror‑descent) refinement.

3️⃣  A light stochastic hill‑climbing finish that swaps tiny
    amounts of mass between random positions.

All steps are time–aware; the routine stops before the supplied
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax on a vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Linear‑program update (LP‑step)
# ----------------------------------------------------------------------


def _build_conv_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A (shape (2n‑1)×n) such that
        (A @ g)[k] = (f * g)[k]    (full linear convolution).
    Each column j contains the vector f (truncated where it would overflow).
    """
    n = f.shape[0]
    m = 2 * n - 1
    rows = []
    cols = []
    data = []

    for j in range(n):
        start = j
        end = min(j + n, m)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        data.extend(f[:length])

    return sparse.csr_matrix((data, (rows, cols)), shape=(m, n))


def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f * g)[k] ≤ M   ∀k,   g ≥ 0.
    Returns the optimiser g (not normalised) or None on failure.
    """
    A = _build_conv_constraint_matrix(f)
    n = f.shape[0]
    c = -np.ones(n)                           # maximise Σg ⇔ minimise –Σg
    b_ub = np.full(A.shape[0], M + 1e-14)     # tiny slack for numeric safety
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Adam optimisation on a smooth max (phase 2)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max(conv)."""
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 5.0                # initial smoothing, will be increased
    beta_mul = 1.04

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # a lives on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a of Σ w_k·conv_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- back‑prop through softmax -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (phase 2)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) using the true sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror‑descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        a /= s if s > 0 else np.ones_like(a) / n

    return best_seq, best_val

# Stochastic hill‑climbing (phase 3)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves; keep improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02  # relative amount of mass moved per step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation to a new length (preserves shape, not sum)."""
    n = a.shape[0]
    if n == target_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_new, x_old, a)
    # re‑normalise to unit ℓ₁‑norm
    up = np.maximum(up, 0.0)
    s = up.sum()
    return up / s if s > 0 else np.full_like(up, 1.0 / target_n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP‑alternation on a modest size (fast, strong progress)
    # ------------------------------------------------------------------
    base_n = 1024                      # medium size for the LP loop
    rng = np.random.default_rng(seed + 1)

    # start from a random Dirichlet vector (strictly positive)
    f = rng.dirichlet(np.ones(base_n))
    f = np.maximum(f, 1e-12)
    f /= f.sum()

    best_f = f.copy()
    best_val = 2.0 * base_n * np.max(_conv_fft(f, f))

    t_mix = 0.5      # initial mixing weight
    no_improve = 0

    print("\n=== LP‑alternation (n=%d) ===" % base_n)

    while time.time() < deadline - 5.0:
        M = np.max(_conv_fft(f, f))
        g = _lp_optimize(f, M)
        if g is None:
            print("[LP] linear programme failed – aborting LP phase")
            break

        g = np.maximum(g, 0.0)
        if g.sum() == 0:
            break
        g /= g.sum()                      # rescale to unit ℓ₁‑norm

        # try a few mixing weights, keep the first that improves the objective
        improved = False
        for t in (t_mix, t_mix * 0.7, t_mix * 0.4, 0.1):
            new_f = (1.0 - t) * f + t * g
            new_f = np.maximum(new_f, 0.0)
            s = new_f.sum()
            if s == 0:
                continue
            new_f /= s
            cur_val = 2.0 * base_n * np.max(_conv_fft(new_f, new_f))
            if cur_val < best_val:
                best_val = cur_val
                best_f = new_f.copy()
                f = new_f
                improved = True
                print(f"[LP] it={int(no_improve)} t={t:.3f} best={cur_val:.6f}")
                break

        if not improved:
            no_improve += 1
            if no_improve >= 6:
                print("[LP] stagnation – leaving LP phase")
                break
        else:
            no_improve = 0

    # ------------------------------------------------------------------
    # Phase 2 – upsample to the target length and continuous optimisation
    # ------------------------------------------------------------------
    target_n = 4000
    a = _upsample(best_f, target_n)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # ------------------------------------------------------------------
    #   2a – Adam on a smoothed max
    # ------------------------------------------------------------------
    print("\n=== Adam optimisation (n=%d) ===" % target_n)
    seq_adam, val_adam = _adam_optimize(a, target_n, deadline, max_iters=15000)
    if seq_adam is not None and val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam.copy()
    else:
        best_seq = a.copy()

    # ------------------------------------------------------------------
    #   2b – Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    print("\n=== Sub‑gradient refinement (n=%d) ===" % target_n)
    seq_sub, val_sub = _subgradient_refine(best_seq, target_n, deadline,
                                           max_iters=80000)
    if val_sub < best_val:
        best_val = val_sub
        best_seq = seq_sub.copy()

    # ------------------------------------------------------------------
    # Phase 3 – Local stochastic hill‑climb
    # ------------------------------------------------------------------
    print("\n=== Hill‑climbing (n=%d) ===" % target_n)
    seq_hc, val_hc = _hill_climb(best_seq, target_n, deadline,
                                 max_iters=50000)
    if val_hc < best_val:
        best_val = val_hc
        best_seq = seq_hc.copy()

    # ------------------------------------------------------------------
    # Final safety check with the provided evaluate_sequence
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
          f"(length {len(best_seq)})")
    # expose the result as a global for possible reuse by a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
