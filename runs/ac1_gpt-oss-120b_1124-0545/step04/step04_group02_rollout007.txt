from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    n = len(f)
    # Build A of shape (2n-1, n)
    # For each k from 0 to 2n-2, the dot product is sum_i f_i * g_{k-i}
    # That's equivalent to (f reversed) convolved with g.
    # We can compute A via for each k: the entries where 0 <= k-i < n, i in [0,n-1].
    # We'll fill a dense matrix.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i runs max(0, k-(n-1)) .. min(n-1, k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # then contribution f[i] * g[k-i]
        # So column j = k-i => i = k-j
        # For each j in column range:
        for i in range(i_min, i_max+1):
            j = k - i
            A[k, j] = f[i]
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a non‑negative step function (vector of floats) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The routine  consists of three phases

1. **Warm‑start** – if a previous construction is available (`height_sequence_1`)
   it is used, otherwise a simple analytic “U‑shaped’’ family is taken.
2. **Adam‑powered sub‑gradient descent** – only the active (max) convolution
   term contributes a sub‑gradient; an Adam optimiser together with a
   projected‑simplex step, occasional random mass‑transfer moves, occasional
   Dirichlet restarts and cheap permutation swaps keep the search from stagnating.
3. **Fine‑grained Gaussian refinement** – in the last few seconds a tiny
   Gaussian jitter is applied many times; any improvement is kept.

All arithmetic is performed on the simplex (non‑negative, sum = 1) so that the
objective is scale‑invariant.  The code respects a 1000 s wall‑time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using a single FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector that already lies on the simplex.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]              # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}   (U‑shaped family)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The algorithm first adopts a warm start,
    then performs Adam‑based sub‑gradient descent on the true max‑convolution,
    and finally carries out a short stochastic fine‑tuning phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Warm start
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=float)
        if a.shape[0] != n:
            # truncate / pad if length differs
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.shape[0]), mode="constant")
        a = np.maximum(a, 0.0)
    else:
        # fallback analytic family
        a = analytic_product(n, p=0.6, offset=0.5)

    a = a / a.sum()          # ensure we are on the simplex
    best_seq = a.copy()
    best_score = compute_score(best_seq)

    print(f"[init] start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑based sub‑gradient descent on the max‑convolution term
    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start, lr_end = 0.5, 1e-4          # linear decay
    temp_start, temp_end = 0.4, 0.0      # Metropolis temperature schedule
    last_improve_time = start_time
    iteration = 0
    stagnation_limit = 25.0               # seconds without improvement → jitter

    while time.time() < deadline:
        # ----- evaluate current state -----
        conv = conv_fft(a)
        max_idx = int(conv.argmax())
        cur_max = conv[max_idx]
        cur_score = 2.0 * n * cur_max

        # ----- record improvement -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_time = time.time()
            print(f"[{iteration:07d}] new best = {best_score:.6f}")

        # ----- sub‑gradient of the active max term -----
        idx = np.arange(n)
        rev = max_idx - idx
        mask = (rev >= 0) & (rev < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev[mask]]          # gradient of (a*a)_{max_idx}

        # ----- Adam update (linear learning‑rate & temperature schedule) -----
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        iteration += 1
        t = iteration
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        # ----- simple Metropolis acceptance -----
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temp > 0.0:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            a = cand

        # ----- occasional diversification moves -----
        # small mass‑transfer
        if np.random.rand() < 0.02:
            a = random_move(a, scale=lr * 0.5)

        # random Dirichlet restart (rare)
        if np.random.rand() < 0.005:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            a = rand_vec

        # random permutation swap (helps escape ordering traps)
        if np.random.rand() < 0.001:
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            perm = a.copy()
            perm[i], perm[j] = perm[j], perm[i]
            a = perm

        # jitter restart if we have been stuck for a while
        if time.time() - last_improve_time > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve_time = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # quit early if only a few seconds remain (reserve them for refinement)
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 3) Fine‑grained stochastic refinement (last few seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(
        f"[done] best score = {best_score:.6f}, iterations ≈ {iteration}, "
        f"time = {total_time:.1f}s, length = {len(best_seq)}"
    )
    # Return plain Python list as required by the harness
    return best_seq.tolist()
