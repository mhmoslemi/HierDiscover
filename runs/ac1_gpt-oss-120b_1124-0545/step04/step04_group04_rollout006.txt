    n = len(a)

        left = a[:half]

def optimize_run(a_init, n, time_alloc):
    start = time.time()
    deadline = start + time_alloc - 0.05
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    last_improve_time = start
    lr = init_lr
    # schedule parameters
    temp_start = 1.0
    temp_end = 2000.0
    # switch fraction maybe 0.8 for hard mask
    while time.time() < deadline:
        t = (time.time() - start) / (time_alloc - 5.0)
        beta = temp_start + t * (temp_end - temp_start)

        # compute conv via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        # compute softmax weights
        if t < mask_phase:
            z = beta * conv
            # stable softmax
            max_z = np.max(z)
            w = np.exp(z - max_z)
            w /= w.sum()
        else:
            max_val = conv.max()
            eps = max_val*1e-12 + 1e-14
            mask = (conv >= max_val - eps).astype(float)
            mask_sum = mask.sum()
            w = mask / mask_sum
        # compute gradient
        fw = np.fft.fft(w, fft_len)
        conv_wa = np.real(np.fft.ifft(fw * fa))[:3*n-2]
        g2 = conv_wa[:n]
        corr = np.real(np.fft.ifft(np.conj(fw) * fa))[:3*n-2]
        g1 = corr[:n]
        grad = 2.0 * n * (g1 + g2)
        # projected gradient step
        step = lr * grad
        a_new = a - step
        a_new = project_simplex(a_new)
        # maybe symmetrization optional:
        # a_new = (a_new + a_new[::-1]) / 2

        a = a_new
        # compute score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_time = time.time()
        # adapt lr maybe
        # if stagnation: jitter
        if time.time() - last_improve_time > stagnate_secs:
            # move to best + gaussian noise
            jitter = rng.normal(scale=jitter_scale, size=n)
            a = best_a + jitter
            a = np.maximum(a, 0)
            a = a / a.sum()
            last_improve_time = time.time()
        # maybe shrink lr linearly
        lr = lr_start * (1 - t) + lr_end * t

def _optimize_pg(a_init, n, alloc_s, rng):
    a = a_init.copy()
    best_a = a.copy()
    # compute initial conv and score
    fft_len = next_pow2(3 * n - 2)
    best_score = _score_of_a(a, n, fft_len)
    last_improve = time.time()
    start = time.time()
    deadline = start + alloc_s - 0.05
    # schedule parameters
    lr_start, lr_end = 0.2, 0.001
    beta_start, beta_end = 0.1, 2000.0
    mask_phase_start = 0.70
    stagnation_time = 15.0
    jitter_scale = 0.02

    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))
        # linear schedule
        lr = lr_start * (1 - p) + lr_end * p
        beta = beta_start * (1 - p) + beta_end * p
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2 * n - 1]
        # weight w
        if p < mask_phase_start:
            z = beta * conv
            # stable softmax
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:
            max_val = conv.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv)
                mask[np.argmax(conv)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # gradient
        fw = np.fft.fft(w, fft_len)
        conv_wa = np.real(np.fft.ifft(fw * fa))[:3 * n - 2]
        g2 = conv_wa[:n]
        corr = np.real(np.fft.ifft(np.conj(fw) * fa))[:3 * n - 2]
        g1 = corr[:n]
        grad = 2.0 * n * (g1 + g2)
        # projected gradient step
        a_next = a - lr * grad
        # project onto simplex
        a_next = _project_simplex(a_next)
        # optional symmetrization (disabled for now)
        a = a_next

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()

                a = a / a.sum()
                last_improve = time.time()
                print(f"[restart] no improvement for {stagnation_time}s, jittering")
    return best_a, best_score

def _project_simplex(v):
    """Project onto simplex: v_i >= 0, sum v_i = 1."""
    if v.ndim != 1:

"""
Propose a step‑function (non‑negative vector on the simplex) that
minimises
    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:
  1. A diverse collection of initial vectors (uniform, geometric decay,
     Dirichlet draws and the historic solution if available).
  2. A projected‑gradient descent on the simplex.  The gradient of a
     smoothed version of the max‑operator is obtained via FFT
     (exactly as in the reference implementation).  The step is
     followed by an Euclidean projection onto the probability simplex.
  3. Simple restart / jitter logic when progress stalls and a final
     optional symmetrisation step.

The time budget (default 1000 s) is split among the different starts;
the best vector found is returned as a plain ``list[float]``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:          # all entries are non‑positive
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_a(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact score for a vector that already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _optimize_pg(
    a_init: np.ndarray,
    n: int,
    alloc_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient optimisation of a single start vector.

    Parameters
    ----------
    a_init : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    n : int
        Length of the sequence.
    alloc_s : float
        Time budget for this run (seconds).
    rng : np.random.Generator
        Random generator for jitter on stagnation.

    Returns
    -------
    best_a : np.ndarray
        Best vector discovered for this start.
    best_score : float
        Corresponding exact score (2·n·max(conv)).
    """
    a = a_init.copy()
    fft_len = _next_pow2(3 * n - 2)

    # bookkeeping
    best_a = a.copy()
    best_score = _score_of_a(a, n, fft_len)
    start = time.time()
    deadline = start + alloc_s - 0.05  # tiny safety margin
    last_improve = start

    # hyper‑parameters (linear schedules)
    lr_start, lr_end = 0.25, 0.001          # learning‑rate
    beta_start, beta_end = 0.1, 2000.0     # soft‑max temperature
    mask_phase_start = 0.70                # when to switch to hard mask
    stagnation_secs = 15.0
    jitter_scale = 0.02

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        # normalised progress p ∈ [0,1]
        p = min(1.0, elapsed / (alloc_s - 5.0))

        # schedule
        lr = lr_start * (1.0 - p) + lr_end * p
        beta = beta_start * (1.0 - p) + beta_end * p

        # ----- convolution a*a (linear) -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]   # length 2n‑1

        # ----- weight w (smooth max or hard mask) -----
        if p < mask_phase_start:
            z = beta * conv
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:
            max_val = conv.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv)
                mask[np.argmax(conv)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # ----- gradient wrt a (linear) -----
        fw = np.fft.fft(w, fft_len)

        # (w * a) linear convolution, first 3n‑2 entries
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # cross‑correlation (w ⋆ a)
        corr = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr[:n]

        grad = 2.0 * n * (g1 + g2)               # gradient of the smoothed max

        # ----- projected gradient step -----
        a_next = a - lr * grad
        a_next = _project_simplex(a_next)        # back to simplex

        a = a_next

        # ----- bookkeeping -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
        else:
            if time.time() - last_improve > stagnation_secs:
                # jitter around the best point
                jitter = rng.normal(scale=jitter_scale, size=n)
                a = best_a + jitter
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                last_improve = time.time()
                print(
                    f"[{it:07d}] restart after {stagnation_secs}s (no improvement)"
                )
        it += 1

    return best_a, best_score


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget (default
    1000 s) and returns the best sequence found as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1) Build a collection of diverse initial vectors
    # --------------------------------------------------------------
    starts = []

    # historic solution if available
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.size != n:
            if hs.size > n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
        hs = np.maximum(hs, 0.0)
        if hs.sum() == 0:
            hs = np.full(n, 1.0 / n)
        else:
            hs = hs / hs.sum()
        starts.append(hs)

    # uniform vector
    starts.append(np.full(n, 1.0 / n))

    # geometric decay families
    for lam in [0.001, 0.003, 0.01, 0.03, 0.1]:
        i = np.arange(n, dtype=float)
        a = np.exp(-lam * i)
        a = a / a.sum()
        starts.append(a)

    # Dirichlet draws (moderately sparse)
    rng_global = np.random.default_rng(seed + 5555)
    for _ in range(4):
        alpha = np.full(n, 0.5)
        a = rng_global.dirichlet(alpha)
        starts.append(a)

    attempts = len(starts)
    global_best_a = None
    global_best_score = float("inf")

    # --------------------------------------------------------------
    # 2) Run a projected‑gradient optimisation for each start,
    #    sharing the remaining time budget.
    # --------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        time_left = deadline - time.time()
        if time_left < 8.0:          # not enough time for a full run
            break
        remaining = attempts - idx
        # keep a small safety margin for later refinement
        alloc = time_left * 0.92 / remaining

        a_opt, score_opt = _optimize_pg(
            a_init=a0,
            n=n,
            alloc_s=alloc,
            rng=np.random.default_rng(seed + idx + 12345),
        )
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {idx}] new global best = {global_best_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # --------------------------------------------------------------
    # 3) Final refinement on the overall best (if any time remains)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 12.0 and global_best_a is not None:
        a_opt, score_opt = _optimize_pg(
            a_init=global_best_a,
            n=n,
            alloc_s=time_left,
            rng=np.random.default_rng(seed + 99999),
        )
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            print(f"[final refine] new best = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Optional symmetrisation (some instances improve)
    # --------------------------------------------------------------
    if global_best_a is not None:
        sym_a = (global_best_a + global_best_a[::-1]) / 2.0
        # keep sum = 1 automatically
        fft_len = _next_pow2(3 * n - 2)
        sym_score = _score_of_a(sym_a, n, fft_len)
        if sym_score < global_best_score:
            global_best_a = sym_a
            global_best_score = sym_score
            print(f"[symmetrised] improved score = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Safety fallback
    # --------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)

    # Ensure final vector respects the very cheap checks of evaluate_sequence
    global_best_a = np.maximum(global_best_a, 0.0)

    total_time = time.time() - start_time
    print(f"[done] best score (raw) = {global_best_score:.6f}, time = {total_time:.1f}s")
    # Final verification using the provided evaluate_sequence (might clip)
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        if np.isfinite(final_score):
            print(f"[final check] evaluate_sequence = {final_score:.6f}")
    except Exception:   # pragma: no cover
        pass

    return global_best_a.tolist()
