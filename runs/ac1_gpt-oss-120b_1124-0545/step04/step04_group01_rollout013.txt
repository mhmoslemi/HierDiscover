def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Linear‑programming matrix (conv(f,·))
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2·n‑1) × n matrix C such that C @ g = conv(f,g) (full linear
    convolution).  The matrix is very structured: each column i contains the
    value f[i] repeatedly over a contiguous block of n rows.
    """
    n = f.shape[0]
    nnz = n * n                     # each column contributes exactly n entries
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    # fill column‑wise – vectorised inside the loop
    for i in range(n):
        start = i
        end = i + n                 # exclusive
        idx = i * n
        rows[idx:idx + n] = np.arange(start, end, dtype=np.int32)
        cols[idx:idx + n] = i
        data[idx:idx + n] = f[i]

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Sub‑gradient refinement (same as reference implementation)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """Mirror‑descent refinement using the exact sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # keep numbers moderate
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (same as reference implementation)
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer moves that keep improvements."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.01

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = 2.0 * n * np.max(_conv_fft(new, new))
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 1000 == 0:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  Returns the best sequence found as a
    plain list of floats.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a length (the earlier best used 4000)
    # ------------------------------------------------------------------
    n = 4000
    sum_target = math.sqrt(2.0 * n)          # the invariant used in the paper

    # ------------------------------------------------------------------
    # Initialise with a sensible seed (previous best if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:                     # pragma: no cover
            previous = None

    if previous is not None and previous.shape == (n,):
        a = previous.copy()
        a = np.maximum(a, 0.0)
    else:
        # uniform distribution on the simplex, then scaled to sum_target
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to the required ℓ₁‑norm (≈√(2n))
    a = a / a.sum() * sum_target

    # ------------------------------------------------------------------
    # Evaluate the starting bound
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)                 # full linear convolution
    M = conv.max()
    best_seq = a.copy()
    best_val = M                               # because evaluate = M when sum=√(2n)
    print(f"[Init] n={n}  initial max(conv) = {M:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming active‑set improvement
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        max_lp_iters = 8                     # a handful of LP solves fits easily
        t = 0.3                               # mixing weight
        for lp_it in range(1, max_lp_iters + 1):
            if time.time() > deadline:
                break

            # Build the sparse matrix representing conv(a,·)
            C = _build_conv_matrix(a)

            # Linear programme: maximise Σ g_i  ⇔ minimise –Σ g_i
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            bounds = [(0, None)] * n

            res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                print(f"[LP] iteration {lp_it} failed: {res.message}")
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                print(f"[LP] iteration {lp_it} gave trivial solution – stop.")
                break

            # Scale g0 up to the target ℓ₁‑norm
            g = g0 * (sum_target / sum_g0)

            # Convex combination
            a = (1.0 - t) * a + t * g

            # Numerical safety – enforce non‑negativity and exact ℓ₁‑norm
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:                     # should never happen
                a = np.full(n, sum_target / n)
            else:
                a *= (sum_target / a_sum)

            # Re‑evaluate bound
            conv = np.convolve(a, a)
            M_new = conv.max()
            if M_new < best_val:
                best_val = M_new
                best_seq = a.copy()
                print(f"[LP] it={lp_it} new best max(conv) = {M_new:.6f}")

            # The theory guarantees M does not increase; keep the tighter one
            if M_new < M:
                M = M_new
            else:
                # if it stalls, shrink the mixing weight a bit
                t = max(t * 0.8, 0.02)

    else:                                       # pragma: no cover
        print("[Info] SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ref, val_ref = _subgradient_refine(best_seq, n, deadline,
                                               max_iters=30000)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"[SubGrad] new best max(conv) = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_hc, val_hc = _hill_climb(best_seq, n, deadline, max_iters=15000)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[Hill] final best max(conv) = {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:                         # pragma: no cover
        pass

    print(f"\n=== Finished ===\nBest max(conv) ≈ {best_val:.6f} "
          f"(length {len(best_seq)})")
    return best_seq.tolist()
