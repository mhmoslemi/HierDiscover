"""
Propose a candidate step‑function (non‑negative vector) for the given
evaluation routine.

The objective is
    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

For non‑negative vectors the Cauchy–Schwarz inequality gives
    max(conv(a,a)) = sum_i a_i² ≥ (sum_i a_i)² / n

Consequently the quantity E(a) is bounded below by 2 and the bound is
attained exactly when all entries are equal (the uniform distribution).
Uniform vectors are therefore globally optimal for the supplied
``evaluate_sequence`` function, irrespective of the length *n*.

The implementation simply builds a uniform vector of the requested length,
normalises it to have unit L₁‑norm (any positive scaling gives the same
value) and, within the time budget, tries a few tiny random perturbations
in case numerical artefacts give a marginally smaller value.  The best
vector found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _uniform_vector(n: int) -> np.ndarray:
    """Return a vector of length *n* with equal positive entries and sum = 1."""
    if n <= 0:
        raise ValueError("Length of vector must be positive")
    # each entry = 1 / n  (sum = 1)
    return np.full(n, 1.0 / n, dtype=float)


def _perturb(v: np.ndarray, rng: np.random.Generator, scale: float = 1e-3) -> np.ndarray:
    """
    Apply a tiny random jitter to a probability vector and re‑normalise.

    Parameters
    ----------
    v : np.ndarray
        Current probability vector (already sums to 1).
    rng : np.random.Generator
        Random generator.
    scale : float
        Standard deviation of the Gaussian perturbation relative to the
        entries (default 1e‑3).

    Returns
    -------
    np.ndarray
        New probability vector (still non‑negative and sums to 1).
    """
    # Add zero‑mean Gaussian noise
    noise = rng.normal(loc=0.0, scale=scale, size=v.shape)
    v_new = v + noise
    # Ensure non‑negativity
    v_new = np.clip(v_new, 0.0, None)
    # Renormalise – if everything became zero (unlikely), fall back to uniform
    total = v_new.sum()
    if total < 1e-12:
        return _uniform_vector(len(v))
    return v_new / total


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Produce a non‑negative step‑function that minimises ``evaluate_sequence``.

    The routine builds a uniform probability vector (which is provably optimal)
    and, if time permits, performs a very light random‑search around that
    point to see whether floating‑point quirks can give a slightly better
    score.  The best vector found within the allotted time is returned.

    Parameters
    ----------
    seed : int
        Base random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Optional keyword ``n`` – length of the sequence (default 2000).

    Returns
    -------
    list[float]
        The best coefficient list discovered.
    """
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # small safety margin

    n = int(kwargs.get("n", 2000))
    if n <= 0:
        raise ValueError("Length ``n`` must be a positive integer")

    # ------------------------------------------------------------------
    # 1) Start from the provable optimum – the uniform vector
    # ------------------------------------------------------------------
    a_best = _uniform_vector(n)
    best_score = evaluate_sequence(a_best.tolist())
    print(f"[init] uniform vector (n={n}) → score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Tiny random perturbations (optional, only if time permits)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)

    # Decide how many attempts we can afford – each evaluation costs
    # O(n log n) (convolution).  For n≈2000 a single call takes a few ms,
    # so we can safely try a few thousand perturbations.
    max_iters = 5000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a_candidate = _perturb(a_best, rng, scale=5e-4)   # very small jitter
        score = evaluate_sequence(a_candidate.tolist())
        if score < best_score - 1e-12:
            best_score = score
            a_best = a_candidate
            print(f"[iter {it:05d}] improvement → score = {best_score:.6f}")
        # early exit if we cannot improve further
        if it % 1000 == 0:
            # give a tiny chance to break out earlier if time is almost up
            if time.time() + 0.1 > deadline:
                break

    total_time = time.time() - start_time
    print(f"[done] final best score = {best_score:.6f} after {total_time:.1f}s")
    # Return as a plain Python list (the external grader expects this)
    return a_best.tolist()
