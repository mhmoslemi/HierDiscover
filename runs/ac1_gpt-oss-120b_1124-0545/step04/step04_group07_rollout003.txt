def project_simplex(v):
    # v is any real vector
    # algorithm from Wang & Carreira-Perpinan (2013)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid L2‑flattening + hard‑max subgradient optimisation.

The objective to minimise is 2·n·max(conv(a,a)), where `conv(a,a)` is the
full linear convolution of a non‑negative vector `a` with itself.
The algorithm works in two phases:

1. **L2‑flattening phase** – minimise the squared L2‑distance between the
   self‑convolution and the perfectly flat target distribution.
   The gradient of this quadratic objective is cheap to compute via FFT.
   After each gradient step the vector is projected onto the probability
   simplex (non‑negative, sum = 1).

2. **Hard‑max sub‑gradient phase** – once the convolution is already fairly
   flat we focus on the remaining peak.  The sub‑gradient of `max(conv)`
   is obtained by putting weight only on the indices where the current
   convolution attains its maximum (a one‑hot “soft‑max”).  A momentum term
   (Nesterov‑style) accelerates convergence.

Both phases share the same FFT‑based convolution kernels and use an
adaptive step size `η / sqrt(iter)`.  The best sequence seen during the
run is kept and finally returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution using FFT (real‑to‑real).
    ``size`` must be ≥ len(u)+len(v)-1.  The returned array is the full
    convolution of length len(u)+len(v)-1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (full linear convolution)."""
    return _conv_fft(a, a, size)


# ----------------------------------------------------------------------
# Heuristic initialisation (Cantor measure)
# ----------------------------------------------------------------------


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the (1/3)‑Cantor measure with ``2**L`` atoms and bin it
    into ``n`` equally spaced bins on [0,1).  The returned vector sums to 1.
    """
    N = 1 << L                                 # number of atoms
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    powers = 3.0 ** np.arange(1, L + 1)
    pos = (bits * (2.0 / powers)).sum(axis=1)   # atom positions
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    return hist.astype(np.float64) / float(N)


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – place all mass on the largest
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: if rounding errors produce a sum ≈0, fallback to uniform
    if w.sum() == 0.0:
        w = np.ones_like(v) / n
    return w

    # ------------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None and len(seq0) == n:
        a = np.asarray(seq0, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 0.01:
            a = _cantor_histogram(n, L=14)
        else:
            a /= s
    else:
        a = _cantor_histogram(n, L=14)

    # ------------------------------------------------------------------
    # Pre‑compute FFT sizes – they stay constant for the whole run.
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)          # for a*a
    size2 = _next_pow2(3 * n - 2)          # for a_rev * (weights)

    # ------------------------------------------------------------------
    # Target flat convolution (uniform distribution)
    # ------------------------------------------------------------------
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv_full = _full_convolution(a, size1)
    best_score = 2.0 * n * conv_full.max()
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Momentum for both phases
    v = np.zeros_like(a)
    beta = 0.9                               # momentum coefficient

    it = 0
    # Define the time at which we change from L2 → hard‑max phase
    phase_switch = start + budget_s * 0.55   # 55 % of the budget

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- Convolution and current score -----
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # ----- Keep the best solution seen so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            elapsed = time.time() - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----------------------------------------------------------------
        # Phase decision
        # ----------------------------------------------------------------
        if time.time() < phase_switch:
            # ------------------------------------------------------------
            # Phase 1 – L2 flattening
            # ------------------------------------------------------------
            diff = conv_full - target                     # shape 2n‑1
            # Gradient of ½‖conv‑target‖² :  grad = 2·a_rev ∗ diff
            grad_full = _conv_fft(a_rev, diff, size2)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # central n entries
            d = -grad                                      # descent direction
            base_lr = 0.6
        else:
            # ------------------------------------------------------------
            # Phase 2 – hard‑max sub‑gradient
            # ------------------------------------------------------------
            # indices where convolution attains the maximum
            max_idx = np.flatnonzero(conv_full == max_b)
            w = np.zeros_like(conv_full, dtype=np.float64)
            w[max_idx] = 1.0 / max_idx.size                # uniform weight over ties
            # Descent direction = -2·(a_rev ∗ w)
            grad_full = _conv_fft(a_rev, w, size2)
            d = -2.0 * grad_full[n - 1 : n - 1 + n]
            # Ensure the direction does not change the total mass
            d -= d.mean()
            base_lr = 0.12

        # ------------------------------------------------------------
        # Momentum and adaptive step size
        # ------------------------------------------------------------
        v = beta * v + (1.0 - beta) * d
        step = base_lr / np.sqrt(it + 1.0)

        # ------------------------------------------------------------
        # Update + projection onto the simplex (non‑negative, sum = 1)
        # ------------------------------------------------------------
        a_candidate = a + step * v
        a = _project_simplex(a_candidate)

        # ------------------------------------------------------------
        # Keep the reversed copy up‑to‑date
        # ------------------------------------------------------------
        a_rev = a[::-1].copy()

        it += 1

    # ------------------------------------------------------------------
    # End of optimisation
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
