Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` objective.

The algorithm combines three ideas:

1. **Initialization from diverse families** – power‑law, exponential,
   linear, a previously found best sequence and a random Dirichlet.
2. **Smoothed max‑convolution optimisation** – instead of using only the
   current maximal convolution entry we minimise a smooth approximation
   `smaxβ(conv) = (1/β)·log Σ exp(β·conv)`.  Its gradient is
   `∇ = 2·(w ∗ a_rev)`, where `w` are the soft‑max weights
   `w_k ∝ exp(β·conv_k)`.  A single FFT evaluates both the convolution
   `a∗a` and the gradient.
3. **Projected Adam on the probability simplex** – after each Adam step
   the vector is projected onto the simplex `{x≥0, Σx=1}`.  The
   smoothing parameter `β` is increased slowly (1 → 5000) so the method
   first moves in a smooth landscape and finally focuses on the true
   supremum.

The search runs for at most `budget_s` seconds, keeps the best
sequence ever seen, and stops early if the target bound `1.503` is
reached.
"""

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum‑1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT (fast for lengths ≥ ~1e3).

    Parameters
    ----------
    x, y : np.ndarray
        Real 1‑D input arrays.

    Returns
    -------
    np.ndarray
        Full convolution (length = len(x)+len(y)-1).
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """
    Normalised power‑law: a_i ∝ (i+1+offset)^(-α).

    Parameters
    ----------
    n      : int
        Length of the sequence.
    alpha  : float
        Exponent (α>0).
    offset : int
        Number of leading zeros before the power‑law starts.

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum = 1 (all entries ≥ 0).
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """
    Normalised exponential decay: a_i ∝ r^{i‑offset} for i≥offset,
    zeros before ``offset``.  0<r<1.

    Parameters
    ----------
    n      : int
    r      : float
    offset : int

    Returns
    -------
    np.ndarray
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """
    Simple decreasing linear (triangular) shape with optional leading zeros.
    a_i = (n‑i) for i≥offset, else 0.  Normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _normalize_to_simplex(arr: np.ndarray) -> np.ndarray:
    """
    Clip to non‑negative and normalise so the sum is 1.
    """
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s == 0.0:
        return np.full_like(arr, 1.0 / arr.size)
    return arr / s
