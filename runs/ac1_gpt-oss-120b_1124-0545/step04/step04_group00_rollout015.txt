def _lp_improve(seq, n, deadline, max_iters=10, t_init=0.4):
    best_seq = seq.copy()
    best_val = _objective(seq)
    t = t_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = _conv_fft(seq, seq).max()
        # build matrix
        A = _build_A(seq)  # shape (2n-1, n)
        # solve LP
        from scipy.optimize import linprog
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] iteration {} failed: {}".format(it, res.message))
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[LP] zero sum solution")
            break
        g = g0 / sum_g0  # normalise to sum 1
        # combine
        seq = (1 - t) * seq + t * g
        # reproject to simplex (sum 1) just in case
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        cur_val = _objective(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.copy()
            print(f"[LP-improve] n={n} iter={it} t={t:.3f} best={cur_val:.6f}")
        # optionally reduce t
        t = max(0.01, t * 0.9)  # decay
    return best_seq, best_val

"""
Propose a non‑negative step function that (approximately) minimizes
evaluate_sequence.  The algorithm combines three ideas:

1.  A diverse set of analytical seeds (uniform, geometric, power‑law,
    sine‑squared, etc.) are generated and the best few are kept.
2.  An LP‑based “active‑set” refinement (inspired by the literature) takes a
    current sequence f (∑f=1) and solves
        max ∑g_j   s.t.   f * g ≤ M,   g ≥ 0,
    where M = max(f*f).  The solution g is mixed with f (convex combination)
    and the process is repeated – each iteration provably does not increase M.
    This drives the maximum auto‑convolution down.
3.  A short sub‑gradient/exponentiated‑gradient polishing step finalises the
    solution, followed by a cheap simulated‑annealing fine‑tune.

The whole search respects the given wall‑time budget (default 1000 s) and
returns the best sequence found (as a plain list of floats).
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT.  Returns full length."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum is assumed >0)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _sin_sq(n: int) -> np.ndarray:
    """Raised‑cosine (sin²) window – the continuous optimum shape."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** 2
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse pool of starting points (all normalised to sum=1)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_sin_sq(n))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse a previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # guard against empty collection
    if not seeds:
        seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “active‑set” step)
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) such that
        A @ g = (f * g)    (linear convolution).
    Row k corresponds to shift k (0‑based).
    """
    n = f.size
    rows = 2 * n - 1
    # c = [f, 0, …, 0]  (length rows)
    # r = [f[0], 0, …, 0] (length n)
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(c, r)[:rows, :]
    return A


def _lp_improve(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 12,
                t0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑based improvement described
    in the problem statement.  Each iteration solves a linear programme
    (maximise L1 norm of g) under the constraint (start_seq * g) ≤ M,
    where M is the current maximum auto‑convolution.
    The new sequence is a convex combination (1‑t)·f + t·g,
    re‑normalised to sum = 1.  The routine stops when the time budget
    expires or no further improvement is observed.
    """
    f = start_seq.copy()
    best_seq = f.copy()
    best_val = _objective(f)
    t = t0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current bound M
        conv_f = _conv_fft(f, f)
        M = conv_f.max()

        # build linear constraints A @ g ≤ M
        A = _build_constraint_matrix(f)

        # linear programme: maximise 1ᵀg  (i.e. minimise –1ᵀg)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve with HiGHS (fast, supports dense matrices)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # if LP fails we abort the refinement
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[LP] iteration {it} produced zero‑mass solution.")
            break

        # scale to keep the L1 norm equal to 1 (same as f)
        g = g0 / sum_g0

        # convex update and re‑normalisation
        f = (1.0 - t) * f + t * g
        f = np.clip(f, 0.0, None)
        f /= f.sum()

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[LP] n={n} it={it:2d} t={t:.3f} best={cur_val:.6f}")

        # slowly shrink the step‑size – early steps move aggressively,
        # later steps fine‑tune.
        t = max(0.01, t * 0.85)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated gradient on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement using the true
    sub‑gradient of max(conv).  The update respects the simplex
    constraints automatically.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = idx - i_range
        g[i_range] = 2.0 * a[rev]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (cheap Metropolis walk)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny fraction of mass
    between two random positions, with an exponentially decaying temperature.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        # occasional global jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= s
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence as
    a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the caller
    deadline = start_time + budget_s - 1.0

    # possibly reuse a previously found construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    lengths = [4000, 8000]                 # try the two standard sizes
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1. generate seeds (including a sin² window)
        seeds = _make_seeds(n, previous)

        # 2. quick evaluation of raw seeds; keep the best few
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:3]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – LP‑based active‑set improvement
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _lp_improve(seed_vec, n, deadline,
                                    max_iters=12, t0=0.5)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                          max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
