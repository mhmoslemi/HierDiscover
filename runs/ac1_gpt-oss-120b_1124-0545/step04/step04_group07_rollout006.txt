import time, numpy as np

def _next_pow2(x):

def _conv_fft(u, v, size):
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu*fv, n=size)
    return w[:len(u)+len(v)-1]

def _full_convolution(a, size):
    return _conv_fft(a, a, size)

def _cantor_histogram(n, L=14):
    N = 1 << L
    idx = np.arange(N, dtype=np.uint32)[:, None]  # shape (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N,L)
    powers = 3.0 ** np.arange(1, L+1)
    pos = (bits * (2.0/powers)).sum(axis=1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0,1.0))
    a = hist.astype(np.float64) / float(N)
    return a

def _project_simplex(v):
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.size
    return v / s

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (full length).
    ``size`` must be ≥ len(u)+len(v)‑1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the full self‑convolution a∗a (length 2·len(a)‑1)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the middle‑third Cantor measure on [0,1] with 2**L atoms
    and bin it into ``n`` equally spaced bins.
    Result sums to 1.
    """
    N = 1 << L  # number of atoms
    # binary representation of the indices (N × L)
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    powers = 3.0 ** np.arange(1, L + 1)          # 3, 3², …, 3ᴸ
    pos = (bits * (2.0 / powers)).sum(axis=1)   # location of each atom
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)     # normalise → sum == 1
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection onto the probability simplex Σv = 1, v ≥ 0.
    Implemented by simple clipping + renormalisation (sufficient here).
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.size
    return v / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function under the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for clean exit

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # 1️⃣  Build a small pool of feasible initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) previous solution, if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        a_prev = np.asarray(prev, dtype=float)
        if a_prev.size != n:
            # linear resampling to the requested length
            x_old = np.linspace(0, 1, a_prev.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            a_prev = np.interp(x_new, x_old, a_prev)
        a_prev = _project_simplex(a_prev)
        candidates.append(a_prev)

    # b) Cantor histograms (coarse & finer)
    candidates.append(_cantor_histogram(n, L=14))
    # a finer approximation – may be a bit heavier but still fine for n≈6000
    try:
        candidates.append(_cantor_histogram(n, L=18))
    except Exception:
        # If memory becomes an issue simply skip the finer version
        pass

    # c) uniform distribution
    candidates.append(np.full(n, 1.0 / n))

    # d) random Dirichlet‑like draw
    rnd = np.random.rand(n)
    candidates.append(rnd / rnd.sum())

    # ------------------------------------------------------------------
    # 2️⃣  Pick the best initial candidate (objective = 2·n·max(conv))
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)        # FFT size for a∗a
    size2 = _next_pow2(3 * n - 2)        # FFT size for a_rev∗w
    best_score = np.inf
    best_seq = None

    for idx, a in enumerate(candidates):
        conv = _full_convolution(a, size1)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        print(f"[init #{idx}] score={cur_score:.6f}")

    a = best_seq.copy()
    a_rev = a[::-1].copy()

    # ------------------------------------------------------------------
    # 3️⃣  Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.05

    # soft‑max temperature schedule
    tau_factor = 0.5          # τ = tau_factor * max_b (initially)
    tau_decay = 0.99995

    # bookkeeping
    m = np.zeros_like(a)      # first moment estimate
    v = np.zeros_like(a)      # second moment estimate
    it = 0
    last_improve = start
    best_score = 2.0 * n * _full_convolution(a, size1).max()
    best_seq = a.copy()

    # perturbation schedule (to escape plateaus)
    plateau_seconds = 15.0
    perturb_eps0 = 0.02
    perturb_decay = 0.999

    # ------------------------------------------------------------------
    # 4️⃣  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- current objective ----
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # ---- keep the best seen so far ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ---- temperature for the soft‑max ----
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # ---- soft‑max weights (numerically stable) ----
        w = np.exp((conv_full - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # ---- gradient of the smooth max (−2·a_rev∗w) ----
        g_full = _conv_fft(a_rev, w, size2)
        g = g_full[n - 1 : n - 1 + n]          # central n entries
        grad = -2.0 * g
        # keep the sum unchanged before projection (helps Adam)
        grad -= grad.mean()

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step_vec = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a + step_vec
        a = _project_simplex(a)
        a_rev = a[::-1].copy()

        # ---- learning‑rate decay (gentle) ----
        learning_rate *= 0.9999

        # ---- plateau detection & random perturbation ----
        if time.time() - last_improve > plateau_seconds:
            eps = perturb_eps0 * (perturb_decay ** (it // 1000))
            a = a + eps * np.random.randn(n)
            a = _project_simplex(a)
            a_rev = a[::-1].copy()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] perturbation applied (ε={eps:.6f})")

        # ---- occasional status report ----
        if it % 5000 == 0:
            elapsed = time.time() - start
            print(f"[{it:7d}] cur={cur_score:.6f} best={best_score:.6f} time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    # 5️⃣  Return the best sequence found
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
