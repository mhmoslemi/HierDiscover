def convolution_matrix(a):
    n = len(a)
    rows = 2*n-1
    # Build using LIL then convert to CSR
    from scipy.sparse import dok_matrix
    data = {}
    # iterate rows
    for i in range(rows):
        j_min = max(0, i-(n-1))
        j_max = min(n-1, i)
        # a_idx = i - j
        for j in range(j_min, j_max+1):
            a_idx = i - j
            data[(i, j)] = a[a_idx]
    # convert
    T = scipy.sparse.csr_matrix((list(data.values()), 
                                 (list(zip(*data.keys()))[0], list(zip(*data.keys()))[1])),
                                 shape=(rows, n))
    return T

def _lp_refine(seq, n, deadline, max_iter=10):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute conv and M
        conv = _conv_fft(a, a)
        M = conv.max()
        # build conv matrix using a
        T = _build_conv_matrix(a)
        # set up LP
        import cvxpy as cp
        b = cp.Variable(n, nonneg=True)
        constraints = [T @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
        except Exception as e:
            print("LP solve failed:", e)
            break
        if b.value is None:
            print("LP no solution at iteration", it)
            break
        b_val = np.array(b.value).flatten()
        sum_b = b_val.sum()
        if sum_b <= 1e-12:
            print("LP sum_b zero")
            break
        a_new = b_val / sum_b
        # compute new value
        new_val = _objective(a_new)
        print(f"[LP] iter {it} sum_b {sum_b:.6f} obj {new_val:.6f}")
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
        # check improvement
        if new_val >= best_val - 1e-9: # maybe no improvement
            # break if not improving
            pass
        a = a_new
    return best_seq, best_val

2. **Smoothed‑max Adam** – the retained seeds are optimised on a
   log‑sum‑exp approximation of `max(conv(a,a))`.  The parameters are the
   logits of a soft‑max, so the simplex constraint is automatic.  A
   moderate number of iterations (≈ 4000) is performed per seed.

3. **Exact sub‑gradient mirror descent** – from the best Adam result we
   run a pure sub‑gradient method on the true max‐convolution, i.e.
   a mirror‑descent on the simplex with the classic `1/√t` step‑size.
   This sharply reduces the objective.

4. **Linear‑programming refinement** – using the current sequence *a* we
   compute `M = max(conv(a,a))`.  Then we solve the linear program

       maximise   1ᵀ b
       subject to  T(a)·b ≤ M ,   b ≥ 0 ,

   where `T(a)` is the (2n‑1) × n convolution matrix of *a*.
   The LP finds a distribution *b* that carries **more total mass**
   without increasing the peak.  After normalising `b` the bound `M`
   shrinks, and we repeat the LP a few times (each LP takes ≤ 1 s).
   Occasionally we invoke a short sub‑gradient polish after an LP step.

All phases are time‑aware; as soon as the global deadline (budget – 2 s)
is reached the routine aborts further work and returns the best sequence
found so far.  The final sequence is a plain list of non‑negative floats
and is stored in the global `height_sequence_1` for possible reuse.

The main entry point required by the grader is `propose_candidate`.
"""

import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to the harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  Assumes seq ≥ 0."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Parametric families used as seeds
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n)
    a = np.clip(a, 0, None)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised seeds."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_cosine_window(n))

    # random soft‑max vectors
    for _ in range(30):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:
        seeds.append(_uniform(n))
    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 4000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp surrogate of max(conv). Returns best seq."""
    v = np.log(seed + 1e-30)  # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # smooth‑max weights
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the true sub‑gradient of max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP on a·b ≤ M)
# ----------------------------------------------------------------------


def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n sparse matrix T such that T@b = conv(a,b)
    (linear convolution, not circular).  Only non‑negative diagonals
    of a are needed because a is non‑negative.
    """
    n = a.shape[0]
    # each diagonal offset d (0 … n‑1) contains the constant a[d]
    data = [a[d] * np.ones(n, dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)  # 0,1,…,n‑1
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return T


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_lp_iters: int = 5,
               subgrad_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Alternating linear programme + short sub‑gradient polish.
    Each LP maximises total mass under the current peak bound M.
    """
    try:
        import cvxpy as cp
    except Exception:
        print("[LP] cvxpy not available – skipping LP stage.")
        return start_seq, _objective(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        T = _build_convolution_matrix(a)

        # LP: maximise sum(b)  s.t.  T @ b <= M, b >= 0
        b = cp.Variable(n, nonneg=True)
        constraints = [T @ b <= M * np.ones(2 * n - 1, dtype=np.float64)]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:
            print(f"[LP] solve failed on iteration {lp_it}: {e}")
            break

        if b.value is None:
            print(f"[LP] no solution on iteration {lp_it}")
            break

        b_val = np.array(b.value).flatten()
        sum_b = b_val.sum()
        if sum_b <= 1e-12:
            print("[LP] degenerate solution (zero total mass).")
            break

        a_new = b_val / sum_b
        new_val = _objective(a_new)
        print(f"[LP] it={lp_it} sum_b={sum_b:.6f} obj={new_val:.6f}")

        # keep the best solution seen so far
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()

        # simple termination heuristic
        if new_val >= best_val - 1e-9:
            # not improving substantially – stop early
            pass

        # a short sub‑gradient polishing after each LP step
        a = a_new
        a, _ = _subgradient_refine(a, n, deadline, max_iters=subgrad_iters)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths to try – start with the proven good length 4000
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) gather a large and diverse seed set
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        # quick evaluation of raw seeds, keep the top few
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(9, len(seed_vals))]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=4000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Linear‑programming refinement (the new ingredient)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _lp_refine(global_best_seq, n, deadline,
                                  max_lp_iters=6, subgrad_iters=1500)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Tiny final annealing (optional – cheap and may catch leftovers)
        # ------------------------------------------------------------------
        # Re‑use the same annealer from the previous solution, but with a
        # reduced iteration budget so we stay within the time limit.
        def _anneal_opt(start_seq, n, deadline, max_iters=2000):
            rng = np.random.default_rng()
            cur = start_seq.copy()
            cur_val = _objective(cur)
            best = cur.copy()
            best_val = cur_val
            T0 = max(1e-6, cur_val * 0.04)
            decay = 0.9995
            it = 0
            while it < max_iters and time.time() < deadline:
                it += 1
                T = T0 * (decay ** it)
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                delta = rng.random() * 0.02 * cur[i]
                if delta > cur[i]:
                    delta = cur[i]
                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                new_val = _objective(new_seq)
                if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                    cur = new_seq
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best = new_seq.copy()
                        print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")
                if it % 500 == 0:
                    # occasional Gaussian jump
                    noise = rng.normal(scale=0.001, size=n)
                    jump = cur + noise
                    jump = np.clip(jump, 0.0, None)
                    if jump.sum() == 0.0:
                        jump = _uniform(n)
                    else:
                        jump /= jump.sum()
                    jump_val = _objective(jump)
                    accept = jump_val < cur_val or rng.random() < np.exp(-(jump_val - cur_val) / (T + 1e-12))
                    if accept:
                        cur = jump
                        cur_val = jump_val
                        if jump_val < best_val:
                            best_val = jump_val
                            best = jump.copy()
                            print(f"[Anneal] n={n} it={it:5d} (jump) best={jump_val:.6f}")
            return best, best_val

        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_opt(global_best_seq, n, deadline, max_iters=3000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (unlikely)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")
    else:
        # store the best sequence globally for possible reuse
        globals()["height_sequence_1"] = global_best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
