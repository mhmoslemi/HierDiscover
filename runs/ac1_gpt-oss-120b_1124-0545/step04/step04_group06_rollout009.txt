def _smooth_gradient_step(a, temperature, lr):
    conv = _conv_fft(a, a)  # length 2n-1
    max_val = conv.max()
    # compute weights using softmax
    # for numerical stability, subtract max_val
    w = np.exp((conv - max_val) / temperature)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # Compute gradient via correlation:
    N = 1 << ((len(a) + len(w) - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    # cross-correlation: inverse FFT of A * conj(W)
    g = 2 * np.fft.irfft(A * np.conj(W), n=N)[:len(a)]
    # gradient step and projection
    a_new = a - lr * g
    a_new = _project_onto_simplex(a_new)
    return a_new, conv.max()

"""
Propose a non‑negative step‑function (a list of floats) that minimizes the
``evaluate_sequence`` objective.

Algorithm – three stages within the time budget
------------------------------------------------
1. **Warm start** – initialise the vector either from the global
   ``height_sequence_1`` (if present) or from a Dirichlet draw.
2. **Smoothed soft‑max gradient descent** – minimise a differentiable
   approximation
   `F(a)=τ·log Σ_k exp((a∗a)[k]/τ)`.  The gradient is
   `∇F = 2·(a ⋆ w)` where the weight vector `w` is the soft‑max of the
   current convolution.  After each step the vector is projected onto
   the probability simplex (non‑negative, sum = 1).  A small learning‑rate
   schedule and a decaying temperature τ drive the optimisation.
3. **Simulated‑annealing fine‑tuning** – once the gradient phase stalls,
   perform random local moves (mass transfers or swaps) accepted with a
   Metropolis rule.  The temperature linearly cools to zero; the best
   sequence seen over the whole run is returned.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`),
so a length of 5000 (the size used in the previous best solution) fits
comfortably into the 1000 s budget while still leaving many thousands of
iterations for refinement.

The function returns a plain Python list; the caller will feed it to the
provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the O(n) algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become negative → uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using FFT.  ``len(x)`` and ``len(y)`` may differ.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when Σa = 1:
    2·n·max(a∗a).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
