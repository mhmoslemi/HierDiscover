def adam_optimize(start_a, loss_and_grad, max_iter, deadline, name):
    a = start_a.copy()
    beta1=0.9; beta2=0.999; lr=0.05; eps=1e-8
    m = np.zeros_like(a); v = np.zeros_like(a)
    best_local_score = compute_score(a)
    best_local_a = a.copy()
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        loss, grad = loss_and_grad(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_simplex(a)
        # evaluate score occasionally maybe every 10 iterations
        if t % 10 == 0:
            sc = compute_score(a)
            if sc < best_local_score:
                best_local_score, best_local_a = sc, a.copy()
                print(f"[{name} iter={t}] new best local = {sc:.6f}")
    return best_local_a, best_local_score

def smooth_max_grad(a, lam):
    conv = conv_fft_self(a)
    max_c = np.max(conv)
    shift = conv - max_c
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # gradient for smooth max wrt a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + len(a) - 1)

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Build a modest set of analytic candidates (product / sum power‑law families,
   uniform, few random Dirichlet draws) and keep the best one.
2. Run a projected Adam optimiser on a *quadratic* surrogate:
      L(a) = Σ_k ( (a*a)_k – 1/(2n‑1) )² .
   This drives the convolution towards a constant – a good proxy for the
   true max‑convolution objective.  Several restarts are performed.
3. Switch to a smoothed max objective
      Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   and continue Adam optimisation while gradually increasing λ, i.e. a
   continuation scheme that tightens the approximation of the true max.
4. Finish with a lightweight random‑mass‑transfer local search to escape
   remaining plateaux.
5. The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector with non‑negative entries summing to 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (with sum normalised to 1).  Returns
        2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    else:  # mode == "sum"
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def uniform_loss_and_grad(a: np.ndarray):
    """
    Quadratic surrogate:
        L(a) = Σ_k (conv_k - avg)^2 ,   avg = 1/(2n-1)
    Returns the loss value and the gradient w.r.t. a (which already respects
    the factor 2 from ∂conv/∂a).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg                           # length 2n‑1
    loss = np.sum(diff * diff)                  # scalar

    # w = 2 * diff  (see derivation in analysis)
    w = 2.0 * diff
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]               # ∂L/∂a
    return loss, grad


def smooth_max_grad(a: np.ndarray, lam: float):
    """
    Gradient of the smooth max Φ_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k) w.r.t. a.
    Returns the convolution vector, its max entry, and ∇_a Φ_λ.
    """
    n = a.shape[0]
    conv = conv_fft_self(a)                     # length 2n‑1
    max_c = np.max(conv)
    shift = conv - max_c                        # ≤ 0
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                      # weights, sum to 1

    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * slice_part[::-1]               # ∂Φ/∂a
    return conv, max_c, grad


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def adam_optimize(start_a, loss_and_grad, max_iter, deadline, name, best_score, best_a):
    """
    Generic Adam optimiser for a differentiable surrogate.
    *loss_and_grad* returns (loss, grad) given the current a.
    The function updates the global best (best_score / best_a) when an
    improvement is observed.
    """
    a = start_a.copy()
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        loss, grad = loss_and_grad(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_simplex(a)

        # occasional true‑objective check
        if t % 10 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[{name}] iter {t} → new global best = {best_score:.6f} (t={elapsed:.1f}s)")
    return best_a, best_score

    """
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidates = []

                cand = analytic_candidate(n, p, off_eff, mode)

    candidates.append(np.ones(n) / n)

    # a few random Dirichlet draws for diversity

    # include the globally‑provided construction if it matches the length

        if ref.shape[0] == n:

    best_a = None
    best_score = np.inf
    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic / random start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Adam on the quadratic surrogate (uniform convolution)

    # Number of restarts and iterations are chosen to stay well within budget.
    restarts = 8

        if r == 0:
