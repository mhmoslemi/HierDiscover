"""
Propose a non‑negative step‑function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣ Analytic grid search – a large collection of parametric families
   (product‑law, sum‑law, exponential, cosine‑power, Beta) is scanned.
   The best vector (normalised to the probability simplex) is kept as a
   warm start.

2️⃣ Smooth‑max Adam optimisation – the max‑convolution is approximated by a
   softmax (temperature β).  The gradient of the smooth objective is
   obtained analytically (a double convolution) and an Adam optimiser
   performs many cheap FFT‑based updates while projecting onto the simplex.
   The temperature β is increased linearly, so early steps move
   ``smoothly'' and later steps focus on the true maximum.

3️⃣ Final stochastic refinement – a short Gaussian‑perturbation loop that
   accepts any improvement.  This stage cleans up the last few seconds.

All helper routines are top‑level, there are no closures or lambdas, and the
search respects a 1000 s wall‑time budget.  The returned list can be fed
directly into the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (zero‑padded) using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector on the simplex.
    The vector is first normalised to sum = 1.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the initial grid search)
# ----------------------------------------------------------------------


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Sum‑law: a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential tail: a_i ∝ exp(-lam·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    a = a / a.sum()
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑power: a_i ∝ |cos(πi/(n‑1))|^{p}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    a = a / a.sum()
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped law: a_i ∝ (i+1)^{α}·(n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective (softmax of the convolution)
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  (1/β)·log Σ_k exp(β·conv_k(a,a)).
    The gradient equals 2·(a_rev ⊛ p) where p is the softmax
    over the convolution values.
    """
    conv = conv_fft(a)                     # length 2n‑1
    max_c = conv.max()
    # softmax probabilities (stable)
    expvals = np.exp(beta * (conv - max_c))
    p = expvals / expvals.sum()            # length 2n‑1, Σ p = 1

    # convolution a_rev * p  (full length 3n‑2)
    a_rev = a[::-1]
    size = _next_pow2(len(a_rev) + len(p) - 1)
    ft_a = np.fft.rfft(a_rev, n=size)
    ft_p = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ft_a * ft_p, n=size)

    start = len(a_rev) - 1                # n‑1
    grad = 2.0 * conv_rev[start:start + len(a)]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine combines an analytic grid, a soft‑max Adam optimiser,
    occasional jitter restarts and a final Gaussian refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # 1) Analytic grid search (very cheap, provides a good warm start)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = np.inf

    print("[init] Analytic grid search – n =", n)

    # product law
    for p in np.arange(0.2, 2.01, 0.2):
        for off in np.arange(0.01, 2.01, 0.2):
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # sum law
    for p in np.arange(0.2, 2.01, 0.2):
        for off in np.arange(0.01, 2.01, 0.2):
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # exponential tails
    for lam in np.logspace(-3, 1, num=15):
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # cosine powers
    for p in np.arange(0.5, 6.1, 0.5):
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # beta‑shaped families
    for a in np.arange(-0.5, 2.01, 0.5):
        for b in np.arange(-0.5, 2.01, 0.5):
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # a few random Dirichlet seeds (quick diversifier)
    for _ in range(200):
        cand = np.random.gamma(1.0, 1.0, size=n)
        cand /= cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # incorporate global reference if present
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score, best_seq = sc, ref.copy()

    print(f"[init] best start score = {best_score:.6f}")

    # 2) Smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (already on simplex)
    m = np.zeros_like(a)                    # first‑moment estimate
    v = np.zeros_like(a)                    # second‑moment estimate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    iteration = 0
    last_improve = start_time
    stagnation_time = 25.0                  # seconds without improvement -> jitter

    # hyper‑parameters for the schedule
    beta_start, beta_end = 0.5, 30.0        # temperature (soft‑max sharpness)
    lr_start, lr_end = 0.5, 5e-4            # learning‑rate (decreases)

    while time.time() < deadline:
        # --- current score (used for tracking) ---
        conv = conv_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # --- temperature and learning‑rate (linear ramps) ---
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / (budget_s - 20.0))
        beta = beta_start * (1.0 - progress) + beta_end * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # --- gradient of the smooth‑max objective ---
        grad = softmax_gradient(a, beta)

        # --- Adam update ---
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        a_new = project_simplex(a_new)

            delta = 0.01 * a_new[i] * np.random.rand()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = project_simplex(a_new)

        a = a_new

            jitter = np.random.rand(n) * 1e-3
            a = project_simplex(best_seq + jitter)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{iteration:06d}] jitter restart (stagnation)")

            a = np.random.gamma(1.0, 1.0, size=n)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{iteration:06d}] full random restart")

    # ------------------------------------------------------------------
    # 3) Final Gaussian refinement (last few seconds)
    # ------------------------------------------------------------------
    refine_deadline = start_time + budget_s - 2.0
    while time.time() < refine_deadline:
        cand = best_seq + np.random.normal(scale=5e-4, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    # Return plain Python list for the evaluation harness
    return best_seq.tolist()
