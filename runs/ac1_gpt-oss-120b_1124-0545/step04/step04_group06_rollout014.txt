"""
Search for a non‑negative step‑function (list of floats) that minimizes
the provided ``evaluate_sequence`` objective.

Algorithm (single‑CPU, deterministic, ≤ 1000 s)

1. **Parameter sweep** – generate many simple families (power‑law,
   exponential, linear) for several lengths and keep the best few
   candidates (by calling ``evaluate_sequence`` directly).

2. **Multi‑start projected simulated annealing** – for each saved
   candidate run a short stochastic optimisation:
   * perturb a random pair of positions (move a small amount of mass)
   * project the resulting vector onto the probability simplex (Euclidean
     projection, Duchi et al. 2008)
   * accept the move if it improves the objective, otherwise accept
     with a temperature‑controlled probability.
   * temperature decays geometrically; the schedule and step size are
     tuned so that many cheap iterations fit into the time budget.

3. **Periodic refinement** – occasional gradient‐step (sub‑gradient of the
   max‑convolution) is applied to escape flat regions.

All helpers are top‑level – no closures or lambdas – and only NumPy,
SciPy and the Python standard library are used.

The returned list is a plain ``list[float]`` (the best sequence found
before the global deadline).  Print statements give a progress trace.
"""

import time
import math
import random
import numpy as np
from scipy import fftpack

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, Σ x_i = 1}.  Implements the O(n) algorithm of
    Duchi, Shalev‑Shwartz, Singer & Chandra (ICML 2008).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution via FFT (real‑to‑real).  Faster than the O(n²)
    ``np.convolve`` for n ≳ 1 000.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sum of the
    sequence is 1 (the usual case throughout the search).  It returns

        2 * n * max(conv(seq, seq))

    where the convolution is performed with the fast FFT routine.
    """
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max()


def _make_power(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """Normalized power‑law a_i ∝ (i+1+offset)^{-α}."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _make_exp(n: int, r: float, offset: int = 0) -> np.ndarray:
    """Normalized exponential decay a_i ∝ r^{i‑offset} for i≥offset."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _make_linear(n: int, offset: int = 0) -> np.ndarray:
    """Decreasing linear (triangular) shape with optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _random_dirichlet(n: int) -> np.ndarray:
    """Uniform Dirichlet(1,…,1) sample (i.e. random point on the simplex)."""
    x = np.random.rand(n)
    x /= x.sum()
    return x


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for the best non‑negative step‑function under the provided
    ``evaluate_sequence`` budget.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0           # safety margin

    # ---------------------------------------------------------------
    # 1. Broad parametric sweep (cheap families)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    length_grid = [1200, 1500, 2000, 2500, 3000, 4000, 5000, 6000]
    power_alphas = [0.2, 0.35, 0.5, 0.65, 0.8]
    power_offsets = [0, 5, 10, 30, 80, 150]
    exp_rs = [0.90, 0.94, 0.97, 0.99]
    exp_offsets = [0, 3, 10, 30, 80]
    linear_offsets = [0, 5, 20, 80]

    print("[sweep] start")

    for n in length_grid:
        if time.time() > deadline:
            break

        # power‑law family
        for off in power_offsets:
            if off >= n:
                continue
            for a in power_alphas:
                seq = _make_power(n, a, off)
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"[sweep] n={n} power α={a:.2f} off={off} "
                          f"score={score:.6f}")

        # exponential family
        for off in exp_offsets:
            if off >= n:
                continue
            for r in exp_rs:
                seq = _make_exp(n, r, off)
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"[sweep] n={n} exp r={r:.2f} off={off} "
                          f"score={score:.6f}")

        # linear decreasing family
        for off in linear_offsets:
            if off >= n:
                continue
            seq = _make_linear(n, off)
            score = _objective(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"[sweep] n={n} linear off={off} score={score:.6f}")

        # few random Dirichlet seeds
        for _ in range(3):
            seq = _random_dirichlet(n)
            score = _objective(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"[sweep] n={n} random dirichlet score={score:.6f}")

        # try to use the globally supplied construction if it matches
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                seq = np.maximum(init, 0.0)
                if seq.sum() == 0.0:
                    seq[:] = 1.0 / n
                else:
                    seq /= seq.sum()
                score = _objective(seq)
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
                    print(f"[sweep] n={n} global init score={score:.6f}")
        except Exception:
            pass

    # ---------------------------------------------------------------
    # 2. Multi‑start projected simulated annealing
    # ---------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest uniform vector
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # keep a pool of promising seeds (the best from sweep plus a few rand)
    seed_pool = [best_seq.copy()]
    for _ in range(4):
        n = best_seq.size
        seed_pool.append(_random_dirichlet(n))

    # annealing parameters
    T0 = 0.5                # initial temperature
    T_min = 1e-6
    decay = 0.9995
    max_iter = 1500000     # generous upper bound; we stop by deadline

    # main loop over seeds
    for start_vec in seed_pool:
        a = start_vec.copy()
        n = a.size
        T = T0
        it = 0

        while time.time() < deadline and T > T_min and it < max_iter:
            it += 1

            # ----- cheap proposal: move a random fraction of mass -----
            i, j = np.random.choice(n, 2, replace=False)
            # pick a step size proportional to current temperature
            delta = min(a[i] * 0.4, T * 0.3)
            if delta <= 0:
                continue
            a[i] -= delta
            a[j] += delta
            # projection back onto simplex (preserves non‑negativity)
            a = _project_onto_simplex(a)

            cur_score = _objective(a)

            # ----- decide acceptance ---------------------------------
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[improve] seed_len={n} iter={it} score={cur_score:.6f}")

            # simulated‑annealing acceptance rule
            if cur_score < best_score:
                accept = True
            else:
                accept = (np.random.rand() < math.exp(-(cur_score - best_score) / T))

            if not accept:
                # revert the move
                a[i] += delta
                a[j] -= delta
                a = _project_onto_simplex(a)

            # temperature schedule
            T = max(T * decay, T_min)

            # ----- occasional gradient‑step to escape flat plateaus ---
            if it % 5000 == 0:
                conv = _conv_fft(a, a)
                k_star = int(conv.argmax())
                idx = k_star - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                grad = np.zeros_like(a)
                grad[mask] = 2.0 * a[idx[mask]]
                # a simple step in the negative gradient direction
                a = a - 0.2 * grad
                a = _project_onto_simplex(a)

    # ---------------------------------------------------------------
    # 3. Final check (optional – re‑evaluate with the official function)
    # ---------------------------------------------------------------
    # The harness will call evaluate_sequence on the returned list, but
    # we perform a final sanity check here to make sure the sequence
    # is well‑formed and to have a final printed score.
    final_score = _objective(best_seq)
    print(f"[final] best_score={final_score:.6f} length={best_seq.size}")

    # return a plain Python list as required
    return best_seq.tolist()
