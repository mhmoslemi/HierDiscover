"""
propose_candidate – two‑phase optimiser for the self‑convolution bound.

Phase 0 – cheap parametric sweep.
    * We restrict the search to a one‑parameter family that mimics the
      classic arcsine shape:
            a_i ∝ ((i+α)*(n‑i‑α))^{-p/2}   ,  i = 0 … n‑1
      with two tunable scalars α>0 (controls the regularisation at the
      ends) and p>0 (controls how strongly the mass concentrates near
      the boundaries).  For each (α,p) we normalise the vector to sum = 1
      and evaluate the exact objective (no smooth‑max approximation).
    * A coarse grid followed by three stages of local refinement quickly
      homes in on a near‑optimal (α,p) pair.  This step costs only a few
      seconds even for n≈2000.

Phase 1 – smooth‑max Adam descent.
    * Starting from the best parametric candidate we run an Adam optimiser
      on the soft‑max parametrisation a = softmax(v).  The objective is
      the temperature‑controlled log‑sum‑exp approximation of the true
      max‑convolution, with the temperature linearly increased from 5 to
      5000.  Learning rates are annealed from 0.08 to 0.001.  Restarts are
      triggered if no improvement is seen for 20 s.
    * The exact score (using the true max) is monitored and the best
      vector seen so far is stored.

Phase 2 – stochastic local refinement (simulated annealing).
    * The remaining time budget is spent on a simple SA that repeatedly
      moves a small fraction of mass from one coordinate to another
      (up to 10 % of the source coefficient) and occasionally performs
      a “splitting” move that distributes a fraction of a coefficient
      among a few random indices.  Proposals are accepted if they improve
      the exact score or with probability exp(−Δ/T) where T cools
      linearly from 0.5 % of the current best score to 1e‑6 % of it.
    * The best vector found is finally normalised (sum = 1) and returned.

All helper functions are defined at top‑level; the implementation respects
the 1000 s time budget and works with any sequence length n (default 2000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a,a)) using FFT (linear convolution)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def exact_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the exact evaluation score:
        2 * n * max(conv(a, a))   (a is already normalised to sum=1)
    """
    max_conv = max_convolution(a, fft_len)
    return 2.0 * n * max_conv


def parametric_shape(n: int, alpha: float, p: float) -> np.ndarray:
    """
    Return a symmetric arcsine‑like shape for given parameters.
    a_i ∝ ((i+α)*(n‑i‑α))^{-p/2}.
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    # product is always positive for the chosen ranges of α and p
    w = ((i + alpha) * (n - i - alpha)) ** (-p / 2.0)
    # protect against underflow – a tiny epsilon keeps the sum >0
    w = np.maximum(w, 1e-300)
    a = w / w.sum()
    return a


def gradient_softmax_obj(v: np.ndarray,
                         a: np.ndarray,
                         conv: np.ndarray,
                         w: np.ndarray,
                         fft_len: int,
                         n: int) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the log‑parameters v.
    Implements g = 2·n·(g1+g2) and applies the soft‑max chain rule.
    """
    # FFT of the soft‑max weights
    fw = np.fft.fft(w, fft_len)

    # convolution w * a
    conv_wa_full = np.real(np.fft.ifft(fw * np.fft.fft(a, fft_len)))[: 3 * n - 2]
    g2 = conv_wa_full[:n]                      # first n entries

    # cross‑correlation w ⋆ a  (valid part)
    corr_full = np.real(np.fft.ifft(np.conj(fw) * np.fft.fft(a, fft_len)))[: 3 * n - 2]
    g1 = corr_full[:n]

    # gradient w.r.t. a
    g = 2.0 * n * (g1 + g2)

    # chain rule for soft‑max parametrisation
    dot_ag = np.dot(a, g)
    grad_v = a * (g - dot_ag)
    return grad_v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine combines a fast parametric grid search, a smooth‑max Adam
    descent, and a simulated‑annealing refinement, all respecting the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to avoid being killed exactly at the deadline
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))          # length of the sequence
    fft_len = 1 << ((3 * n - 2).bit_length())  # power‑of‑two FFT size

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    best_score = np.inf
    best_a = None
    best_alpha = None
    best_p = None

    # Coarse grid
    alphas_coarse = np.linspace(0.1, 2.0, 10)     # α ∈ [0.1, 2.0]
    ps_coarse = np.linspace(0.2, 4.0, 39)        # p ∈ [0.2, 4.0]

    print("[phase0] coarse grid search started")
    for alpha in alphas_coarse:
        for p in ps_coarse:
            # time‑check
            if time.time() >= overall_deadline - 5:
                break
            a = parametric_shape(n, alpha, p)
            score = exact_score(a, n, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_alpha = alpha
                best_p = p
    print(f"[phase0] coarse best: score={best_score:.6f}, α={best_alpha:.4f}, p={best_p:.4f}")

    # Three refinement stages (progressively smaller neighbourhoods)
    for refinement in range(1, 4):
        if time.time() >= overall_deadline - 5:
            break

        # shrink the search window
        delta_alpha = max(0.2 / (2 ** refinement), 0.005)
        delta_p = max(0.4 / (2 ** refinement), 0.01)

        al_min = max(0.01, best_alpha - delta_alpha)
        al_max = best_alpha + delta_alpha
        p_min = max(0.01, best_p - delta_p)
        p_max = best_p + delta_p

        al_grid = np.linspace(al_min, al_max, 7)
        p_grid = np.linspace(p_min, p_max, 7)

        print(f"[phase0] refinement {refinement}: α∈[{al_min:.4f},{al_max:.4f}], "
              f"p∈[{p_min:.4f},{p_max:.4f}]")
        for alpha in al_grid:
            for p in p_grid:
                if time.time() >= overall_deadline - 5:
                    break
                a = parametric_shape(n, alpha, p)
                score = exact_score(a, n, fft_len)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_alpha = alpha
                    best_p = p
        print(f"[phase0] after refinement {refinement}: score={best_score:.6f}, "
              f"α={best_alpha:.4f}, p={best_p:.4f}")

    # If a global seed sequence exists and is better, reuse it
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        if seq.size == n and seq.sum() > 0:
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            seq_score = exact_score(seq, n, fft_len)
            if seq_score < best_score:
                best_score = seq_score
                best_a = seq.copy()
                print(f"[phase0] global seed improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam descent
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 10.0:
        print("[phase1] starting Adam descent")
        a = best_a.copy()
        # initialise log‑parameters for soft‑max
        v = np.log(a + 1e-12)

        # Adam hyper‑parameters
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)

        # schedules (linear in progress)
        lr_start, lr_end = 0.08, 0.001
        temp_start, temp_end = 5.0, 5000.0

        it = 0
        last_improve = time.time()
        # allocate ~45 % of the remaining budget for phase 1
        phase1_deadline = time.time() + 0.45 * remaining

        while time.time() < phase1_deadline:
            # progress ratio for the current phase
            p = min(1.0, (time.time() - (overall_deadline - remaining)) /
                    (phase1_deadline - (overall_deadline - remaining)))

            lr = lr_start * (1.0 - p) + lr_end * p
            temp = temp_start * (1.0 - p) + temp_end * p

            # current simplex point
            a = softmax(v)

            # exact convolution (FFT)
            fa = np.fft.fft(a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

            # smooth‑max weights (log‑sum‑exp)
            z = temp * conv
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w /= w.sum()

            # gradient wrt v
            grad_v = gradient_softmax_obj(v, a, conv, w, fft_len, n)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat_corr = v_hat / (1.0 - beta2 ** it)

            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # evaluate exact score (max already available)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.time()
                print(f"[phase1] iter {it:08d} new best {best_score:.6f}")

            # simple restart policy
            if it % 4000 == 0 and (time.time() - last_improve > 20.0):
                v = np.log(best_a + 1e-12) + rng.normal(scale=1e-3, size=n)
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[phase1] restart at iter {it}")

        # update remaining budget after phase 1
        remaining = overall_deadline - time.time()
        print(f"[phase1] finished, best score={best_score:.6f}, time left={remaining:.1f}s")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic local refinement (simulated annealing)
    # ------------------------------------------------------------------
    if remaining > 5.0:
        print("[phase2] starting simulated annealing")
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # SA temperature schedule (linear cooling)
        T_start = best_score * 0.005   # 0.5 % of current score
        T_end = best_score * 1e-6

        sa_it = 0
        phase_start = time.time()

        while time.time() < overall_deadline:
            # progress fraction of SA phase
            elapsed = time.time() - phase_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # ---- propose a move -------------------------------------------------
            move_type = rng.integers(0, 3)   # 0: simple transfer, 1: split, 2: Gaussian jitter

            if move_type == 0:
                # simple mass transfer between two distinct indices
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_move = min(a[i], 0.10 * a[i] + 1e-12)   # up to 10 % of source
                if max_move <= 0.0:
                    continue
                delta = rng.random() * max_move
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

            elif move_type == 1:
                # split move: take a fraction from i and distribute to k random slots
                i = rng.integers(0, n)
                frac = rng.random() * 0.08   # up to 8 % of a[i]
                delta = a[i] * frac
                if delta <= 0.0:
                    continue
                k = rng.integers(2, 6)        # split among 2‑5 positions
                idxs = rng.choice(n, size=k, replace=False)
                share = delta / k
                a_new = a.copy()
                a_new[i] -= delta
                a_new[idxs] += share

            else:
                # Gaussian jitter on log‑space (small perturbation)
                v = np.log(a + 1e-12)
                v += rng.normal(scale=0.02, size=n)
                a_new = softmax(v)

            # ---- evaluate proposal ---------------------------------------------
            new_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_max

            # ---- Metropolis acceptance -----------------------------------------
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA] iter {sa_it:07d} improved to {best_score:.6f} "
                          f"(t={time.time() - start_time:.1f}s)")

            sa_it += 1

        print(f"[phase2] finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, final score={best_score:.6f}, length={n}")

    # expose the final vector for possible reuse by future runs
    globals()["height_sequence_2"] = best_a.tolist()

    return best_a.tolist()
