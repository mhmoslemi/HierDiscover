"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The evaluation is scale‑invariant; we therefore keep every candidate
normalised to sum = 1.  The optimiser works in three stages:

1. **Projected sub‑gradient descent** on the exact ∞‑norm ‑ we repeatedly
   locate the convolution entry that attains the current maximum,
   compute its sub‑gradient, step in the opposite direction and project
   back onto the probability simplex.  The learning‑rate decays
   linearly.

2. **Linear‑programming refinement** – given the current vector `a`
   (sum = 1) with maximal self‑convolution value *M*, we solve

        maximise   Σ b_j
        subject to (a * b)_k ≤ M   for all k,
                   b_j ≥ 0.

   If the optimum satisfies Σ b > 1 we can increase the L¹‑mass
   while respecting the same max‑convolution bound.  By mixing the
   normalised optimiser `b/Σb` with `a` we strictly decrease *M*.
   The LP is tiny (≈ n variables, 2n‑1 constraints) and is solved with
   HiGHS via ``scipy.optimize.linprog``.  A few mixing steps are enough
   to push the bound beyond what plain gradient descent achieves.

3. **Fine‑grained stochastic refinement** (simulated‑annealing style)
   – random mass transfers between two coordinates are accepted if they
   lower the exact objective, otherwise with a temperature that cools
   linearly.  This stage cleans up the remaining tiny gaps.

The whole search respects the 1000 s budget, tries a few different
sequence lengths (default 2000, 2500, 3000) and always returns the best
vector found.
"""

def arcsine_init(n: int) -> np.ndarray:
    """Classic arcsine (inverse‑sqrt) shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


def max_convolution(a: np.ndarray, fft_len: int = None) -> float:
    """Return max(a*a) using a real FFT (linear convolution)."""
    n = a.size
    if fft_len is None:
        fft_len = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(conv.max())


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Return the full linear convolution a*a (length 2n‑1)."""
    n = a.size
    if fft_len is None:
        fft_len = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (sum=1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def max_conv_subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the k‑th convolution entry
        (a*a)[k] = Σ_i a_i a_{k-i}
    with respect to `a`.  Only indices where k‑i is valid contribute.
    """
    n = a.size
    grad = np.zeros_like(a)
    idx = np.arange(n)
    mask = (k - idx >= 0) & (k - idx < n)
    j = idx[mask]
    grad[mask] = a[k - j] + a[j]
    return grad


def build_A_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) for the linear constraints
    (a * b)_k = Σ_i a_i b_{k-i} ≤ M.
    Row k, column j contains a_{k‑j} if the index is in range, else 0.
    """
    n = a.size
    rows = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (rows >= 0) & (rows < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[rows[mask]]
    return A

        # linear decay of learning rate
        elapsed = time.time() - start
        frac = min(1.0, elapsed / (deadline - start - 5.0))
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # current maximum entry of the convolution
        conv = full_convolution(a, fft_len=fft_len)
        k_max = int(conv.argmax())

        # sub‑gradient of that entry
        g = max_conv_subgradient(a, k_max)

        # gradient step & projection onto simplex
        a = proj_simplex(a - lr * g)

        # track improvement
        cur_score = 2.0 * n * conv.max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{n}] sub‑grad it={it:05d}  score={best_score:.6f}")

        it += 1

    # ----- 2. Linear‑programming refinement ---------------------------------
    # we keep a separate timer for this stage
    while time.time() < deadline - 2.0:
        # current convolution bound
        conv = full_convolution(a, fft_len=fft_len)
        M = conv.max()

        # build constraint matrix A (dense, size (2n‑1)×n)
        A = build_A_matrix(a)

        # LP: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # should not happen, but abort the LP stage if it does
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # no feasible increase of the ℓ¹‑mass → stop
            break

        # Mix the current vector with the normalised LP solution.
        b_norm = b / sum_b
        # a modest mixing factor works well; we keep it fixed.
        t_mix = 0.3
        a_new = (1.0 - t_mix) * a + t_mix * b_norm

        # evaluate new bound
        new_score = 2.0 * n * max_convolution(a_new, fft_len=fft_len)

        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{n}] LP mix  score={best_score:.6f}")

        # accept the mixed point (even if it is not better) – it can still
        # make the next LP feasible for a larger mass.
        a = a_new

    # ----- 3. Stochastic local refinement (simulated annealing) ----------
    remaining = deadline - time.time()
    if remaining > 1.0:
        # temperature schedule (linear cooling)
        T_start = best_score * 0.005
        T_end = best_score * 1e-6
        phase_start = time.time()
        sa_it = 0
        a_sa = best_a.copy()
        while time.time() < deadline:
            progress = min(1.0, (time.time() - phase_start) / remaining)
            T = T_start * (1.0 - progress) + T_end * progress

            # propose a small mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_move = min(a_sa[i], 0.05 * a_sa[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_prop = a_sa.copy()
            a_prop[i] -= delta
            a_prop[j] += delta

            prop_score = 2.0 * n * max_convolution(a_prop, fft_len=fft_len)

            # Metropolis acceptance
            if prop_score < best_score or rng.random() < np.exp(-(prop_score - best_score) / (T + 1e-12)):
                a_sa = a_prop
                if prop_score < best_score:
                    best_score = prop_score
                    best_a = a_prop.copy()
                    print(f"[{n}] SA it={sa_it:07d}  improved to {best_score:.6f}")

            sa_it += 1

    # final sanity check (nan/inf removal, tiny negative round‑off)
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine tries a few sequence lengths,
    runs a deterministic sub‑gradient descent, refines the solution
    by a linear‑programming step and finishes with a short simulated‑
    annealing clean‑up.  The best vector found (normalised to sum = 1) is
    returned as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a set of candidate lengths.
    # If the caller supplies ``n``, we honour it; otherwise we try a few.
    # ------------------------------------------------------------------
    if "n" in kwargs:
        n_list = [int(kwargs["n"])]
    else:
        n_list = [1500, 2000, 2500, 3000]

    # If a previous construction is available, prefer its length as well.
    if "height_sequence_1" in globals():
        glen = len(globals()["height_sequence_1"])
        if glen not in n_list:
            n_list.append(glen)

    best_score_glob = float("inf")
    best_seq_glob = None

    # ------------------------------------------------------------------
    # Optimise each length in the remaining time budget.
    # ------------------------------------------------------------------
    for idx, n in enumerate(sorted(n_list)):
        if time.time() >= overall_deadline:
            break

        # allocate a fair share of the remaining budget to this length
        remaining_lengths = len(n_list) - idx
        sub_deadline = time.time() + (overall_deadline - time.time()) / remaining_lengths

        print(f"[global] optimise length n={n} (deadline {sub_deadline - time.time():.1f}s)")

        seq, score = optimise_one_length(n, sub_deadline, rng)

        if score < best_score_glob:
            best_score_glob = score
            best_seq_glob = seq
            print(f"[global] new best overall: n={n}  score={best_score_glob:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced.
    # ------------------------------------------------------------------
    if best_seq_glob is None:
        n = kwargs.get("n", 1000)
        best_seq_glob = np.full(int(n), 1.0 / int(n))

    # final normalisation (protect against tiny numerical drift)
    best_seq_glob = np.maximum(best_seq_glob, 0.0)
    total = best_seq_glob.sum()
    if total == 0.0:
        best_seq_glob = np.full_like(best_seq_glob, 1.0 / best_seq_glob.size)
    else:
        best_seq_glob = best_seq_glob / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score_glob:.6f}, length={len(best_seq_glob)}")
    return best_seq_glob.tolist()
