"""
Algorithm:
  - Starts from a symmetric beta‑shaped distribution.
  - Uses an annealed soft‑max of the convolution to compute a smooth
    approximation of the max‑convolution.
  - Performs mirror‑descent on the simplex (multiplicative update
    a_i ← a_i exp(−η g_i)) where g is the gradient of the smooth
    objective.  The temperature β and the step size η are increased /
    decreased linearly with time.
  - Enforces symmetry after each update, adds occasional jitter and
    rare random Dirichlet restarts to avoid stagnation.
  - In the last seconds runs a fine‑grained Gaussian perturbation.
  This differs from the previous sub‑gradient method by smoothing the
  max operator and using multiplicative updates, which tend to escape
  shallow plateaus and reach lower max‑convolution values.
"""
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT, returns length 2·len(a)-1."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size)
    return conv_full[: 2 * n - 1]

def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) (sum of a is normalised to 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * np.max(conv))

def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped a_i ∝ (i+1)^α (n‑i)^β, symmetric when α=β."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

"""
Algorithm:
  - Starts from a symmetric beta‑shaped distribution.
  - Uses an annealed soft‑max of the convolution to obtain a smooth
    approximation of the max‑convolution.
  - Performs mirror‑descent on the probability simplex via multiplicative
    updates (a_i ← a_i·exp(−η·g_i)), where g is the gradient of the smooth
    objective.
  - The temperature β and step size η are linearly annealed.
  - Enforces exact symmetry a[i] = a[n‑1‑i] after each update,
    adds occasional jitter and rare Dirichlet restarts to avoid stagnation.
  - In the final seconds a fine‑grained Gaussian perturbation refines the
    best candidate.
This differs from the earlier sub‑gradient method by smoothing the max
operator and using multiplicative (entropy‑mirror) updates, which
tend to escape shallow plateaus and reach lower max‑convolution values.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fft_a * fft_a, n=size)
    return conv_full[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a vector on the simplex.
    The vector is renormalised for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped a_i ∝ (i+1)^α·(n‑i)^β (symmetric when α=β)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising

        2·n·max(conv(a,a)) / (sum(a)**2)

    The procedure combines a symmetric beta‑initialisation,
    annealed soft‑max smoothing of the max‑convolution and
    multiplicative mirror‑descent on the simplex.  Random jitter,
    occasional Dirichlet restarts and a final short stochastic
    refinement are employed to escape plateaus.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3000))

    # ------------------------------------------------------------------
    # 1. Initialise a symmetric distribution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            a = np.maximum(ref, 0.0)
            a = a / a.sum()
        else:
            a = analytic_beta(n, -0.5, -0.5)
    else:
        a = analytic_beta(n, -0.5, -0.5)   # U‑shape (mass at the ends)

    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(best_a)
    print("[init] n =", n, "initial score =", best_score)

    # ------------------------------------------------------------------
    # Pre‑computed FFT sizes
    # ------------------------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)          # for a*a
    size_grad = _next_pow2(3 * n - 2)          # for w * a_rev

    # ------------------------------------------------------------------
    # Annealing / step‑size schedules
    # ------------------------------------------------------------------
    step_start, step_end = 5.0, 0.001
    beta_start, beta_end = 0.5, 200.0
    last_improve = start
    stagnation_time = 30.0   # seconds without improvement → jitter

    iteration = 0

    while time.time() < deadline:
        # ----- convolution and current objective -----
        a_fft = np.fft.rfft(a, n=size_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=size_conv)
        conv = conv_full[: 2 * n - 1]                # linear convolution
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{iteration:07d}] new best = {best_score:.6f}")

        # ----- schedule parameters -----
        elapsed = time.time() - start
        progress = min(1.0, elapsed / (budget_s - 2.0))
        step = step_start * (1.0 - progress) + step_end * progress
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # ----- soft‑max weights -----
        maxc = cur_max
        w_raw = np.exp(beta * (conv - maxc))           # stable exponent
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.shape[0])
        else:
            w = w_raw / w_sum

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=size_grad)
        rev_fft = np.fft.rfft(a_rev, n=size_grad)
        g_full = np.fft.irfft(w_fft * rev_fft, n=size_grad)
        # slice corresponds to indices 0 … n‑1
        g = 2.0 * g_full[(n - 1) : (n - 1 + n)]

        # ----- multiplicative mirror‑descent update (on simplex) -----
        upd = np.exp(-step * g)          # element‑wise
        a = a * upd
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.ones_like(a) / n
        else:
            a = a / a_sum

        # ----- enforce exact symmetry -----
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        # ----- occasional jitter to break tiny plateaus -----
        if np.random.rand() < 0.01:
            jitter = np.random.rand(n) * 1e-4
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()

        # ----- stagnation handling (jitter restart) -----
        if time.time() - last_improve > stagnation_time:
            a = best_a + np.random.randn(n) * 1e-3
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            a = (a + a[::-1]) * 0.5
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # ----- rare random Dirichlet restarts -----
        if np.random.rand() < 0.002:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            # quick evaluation
            r_fft = np.fft.rfft(rand_vec, n=size_conv)
            conv_rand = np.fft.irfft(r_fft * r_fft, n=size_conv)[: 2 * n - 1]
            rand_score = 2.0 * n * conv_rand.max()
            if rand_score < best_score:
                best_score = rand_score
                best_a = rand_vec.copy()
                a = rand_vec.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] random sample improved to {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Fine‑grained stochastic refinement (last ~1 s)
    # ------------------------------------------------------------------
    refine_deadline = start + budget_s - 0.8
    while time.time() < refine_deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_a + perturb
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        # keep symmetry (helps a little)
        cand = (cand + cand[::-1]) * 0.5
        cand = cand / cand.sum()
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total = time.time() - start
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total:.1f}s, length = {len(best_a)}")
    return best_a.tolist()
