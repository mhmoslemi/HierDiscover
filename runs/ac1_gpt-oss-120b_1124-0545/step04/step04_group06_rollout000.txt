"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Parametric init** – try a handful of simple families (power‑law,
   exponential, linear) together with a few random Dirichlet vectors.
   Keep the best one as a starting point.

2. **Smoothed max‑convolution descent** – minimise
   `F(a)=log‑sum‑exp( β·(a∗a) )` with β slowly increased.
   For a given β the gradient is
   `∇F_i = 2·∑_k w_k a_{k−i}` where `w = softmax(β·(a∗a))`.
   The gradient is obtained by a single FFT‑based convolution:
   `∇F = 2·(w * a_rev)[n‑1:2n‑1]`.

   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1) using the Duchi et al. projection.

3. **Momentum & random swaps** – a cheap momentum term (0.9) and occasional
   random mass transfers help escaping plateaus.

4. **Early stopping / multiple lengths** – the routine stops when the
   global time budget (default 1000 s) is exhausted and returns the
   best sequence found.

The procedure works for any length up to a few‑10⁴ (FFT‐based convolutions
are O(n log n)).  All helper routines are top‑level; no closures or file I/O
are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """Normalized power‑law a_i ∝ (i+1+offset)^{‑α}."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """Normalized exponential decay a_i ∝ r^{i‑offset} for i ≥ offset."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Decreasing linear (triangular) shape, optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a vector with
    sum = 1: 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()
