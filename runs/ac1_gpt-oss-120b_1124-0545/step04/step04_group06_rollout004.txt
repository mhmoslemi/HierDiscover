"""
Propose a non‑negative step‑function (a list of floats) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Broad parametric sweep** – explore a few analytic families
   (power‑law, exponential, mixture of two exponentials, simple linear)
   for several lengths.  The best candidate (lowest exact objective)
   becomes the starting point.

2. **Smoothed exponential‑gradient descent** – minimise a smooth
   approximation of the max‑convolution:
        f_β(a) = (1/β)·log( Σ_k exp(β·(a∗a)[k]) )
   Its gradient can be written in closed form using a second convolution.
   We update the probability vector by a multiplicative (exponentiated)
   step and renormalise (mirror descent on the simplex).  The temperature
   β is increased slowly so that the optimisation ultimately targets the
   true max and not the smooth surrogate.

3. **Fine‑grained sub‑gradient descent** – once β is large we switch to
   the exact sub‑gradient of the max‑convolution (the same update that
   the original baseline used) to polish the solution.

The routine respects the global time budget (default 1000 s) and keeps
track of the best sequence found (both the smoothed phase and the final
sub‑gradient phase).  No file I/O is performed; all helpers are defined
at module level so the evaluator can import ``propose_candidate`` directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised vector
    (∑a = 1):  2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α), normalised to sum = 1."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """Geometric decay a_i ∝ r^{i‑offset} for i≥offset, normalised."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Triangular decreasing shape a_i = n‑i for i≥offset, normalised."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _mixture_sequence(n: int, r1: float, r2: float, lam: float,
                      offset: int = 0) -> np.ndarray:
    """
    Mixture of two exponentials:
        a = λ·r1^{i‑offset} + (1‑λ)·r2^{i‑offset},    i≥offset
    Normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = lam * (r1 ** idx) + (1.0 - lam) * (r2 ** idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max–convolution
        f_β(a) = (1/β)·log( Σ_k exp(β·(a∗a)[k]) )
    Returns the vector g where
        g_j = 2·Σ_k w_k·a_{k‑j},
    with w being the soft‑max weights over the convolution entries.
    """
    n = a.size
    conv = _conv_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # softmax weights (numerically stable)
    exp_term = np.exp(beta * (conv - maxc))
    w = exp_term / exp_term.sum()              # Σ w = 1

    # convolution of a with reversed weight vector
    w_rev = w[::-1]
    conv_aw = _conv_fft(a, w_rev)              # length 3n‑2
    idxs = (2 * n - 2) - np.arange(n)          # selects positions needed
    grad = 2.0 * conv_aw[idxs]
    return grad

    (as a plain Python list).  The routine combines a parametric sweep,
    smoothed exponential‑gradient descent, and a final sub‑gradient phase.
    """

    start_time = time.time()

    # 1. Broad parametric sweep (cheap, yet often very effective)

    exp_offsets = [0, 5, 25, 80]

    mix_rs = [(0.96, 0.99), (0.94, 0.985), (0.92, 0.97)]

                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} linear off={offset} score={score:.6f}")

        # --- Mixture of two exponentials --------------------------------------
        for offset in mix_offsets:
            if offset >= n:
                continue
            for (r1, r2) in mix_rs:
                for lam in mix_lams:
                    a = _mixture_sequence(n, r1, r2, lam, offset)
                    score = _objective(a)
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"[sweep] n={n} mix r1={r1:.3f} r2={r2:.3f} "
                              f"lam={lam:.2f} off={offset} score={score:.6f}")

        # --- Random Dirichlet seeds -------------------------------------------
        for _ in range(2):
            a = np.random.rand(n)
            a /= a.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} random dirichlet score={score:.6f}")

        # --- Use the global ``height_sequence_1`` if available -----------------
        try:
            init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            if init.size == n:
                a = np.maximum(init, 0.0)
                if a.sum() == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} global init score={score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Smoothed exponential‑gradient descent (global smoothing)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    a = best_seq.copy()
    n = a.size

    # schedules
    eta = 1.0                     # initial multiplicative step size
    eta_decay = 0.9997
    eta_min = 1e-6

    beta = 5.0                    # start with a very smooth max
    beta_growth = 1.001
    beta_max = 200.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 1) compute the smooth weights
        conv = _conv_fft(a, a)                 # full convolution
        maxc = conv.max()
        exp_term = np.exp(beta * (conv - maxc))
        w = exp_term / exp_term.sum()          # soft‑max over convolution entries

        # 2) gradient of the smooth objective (vector of length n)
        #    using the identity derived in _smooth_gradient
        w_rev = w[::-1]
        conv_aw = _conv_fft(a, w_rev)          # length 3n‑2
        idxs = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv_aw[idxs]

        # 3) exponentiated (multiplicative) update + re‑normalisation
        a = a * np.exp(-eta * grad)
        # avoid complete underflow
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # 4) schedules
        eta = max(eta * eta_decay, eta_min)
        beta = min(beta * beta_growth, beta_max)

        # 5) occasional random mass exchange to escape plateaus
        if iteration % 120 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.25, eta * 0.5)
            if delta > 1e-15:
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()

        # 6) periodic exact evaluation
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[expgrad] iter={iteration} n={n} score={best_score:.6f}")

            # early exit if target already reached
            if best_score < 1.503:
                print("[expgrad] target reached")
                break

    # ------------------------------------------------------------------
    # 3. Fine‑grained sub‑gradient descent on the *exact* max‑convolution
    # ------------------------------------------------------------------
    # Switch to the exact sub‑gradient (as in the previous baseline) to
    # polish the solution.  We keep the same learning‑rate schedule but
    # now the gradient is taken at the actual maximal index.
    momentum = np.zeros_like(a)
    lr = 0.7
    lr_decay = 0.9999
    lr_min = 1e-6

    while time.time() < deadline:
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # sub‑gradient of max‑convolution:
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum + projection onto simplex
        momentum = 0.9 * momentum + grad
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        lr = max(lr * lr_decay, lr_min)

        # occasional random swap (to keep exploration)
        if np.random.rand() < 0.001:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.2, lr * 0.5)
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # periodic exact evaluation
        if np.random.rand() < 0.02:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[subgrad] n={n} score={best_score:.6f}")

            if best_score < 1.503:
                print("[subgrad] target reached")
                break

    # ------------------------------------------------------------------
    # 4. Return best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
