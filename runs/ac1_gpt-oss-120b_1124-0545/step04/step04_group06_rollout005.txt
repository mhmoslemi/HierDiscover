"""
Propose a non‑negative step function (list of floats) that minimizes
``evaluate_sequence``.

Algorithm
---------
1.  Perform a cheap parametric sweep (power‑law, exponential, linear,
    random Dirichlet) for a few candidate lengths and keep the best
    sequence for each length.
2.  Refine the best candidate with a *smooth‑max* optimisation:
    -   The objective  f(a)=max_k (a∗a)[k]  is approximated by the
        log‑sum‑exp smooth max:
        f_μ(a)= (1/μ)·log Σ_k exp(μ·(a∗a)[k]).
    -   Its gradient w.r.t. a is  g = 2·corr(a, w) where
        w_k ∝ exp(μ·(a∗a)[k]) (the soft‑max weights) and
        corr denotes the linear cross‑correlation.
    -   An Adam optimiser is applied to g and after each step the
        vector is projected onto the probability simplex (non‑negative,
        sum = 1).
    -   The smoothing parameter μ is increased slowly (annealing) so
        that early iterations move quickly in a smooth landscape and
        later iterations focus on the true max.
3.  Several lengths (up to ≈10 000) are tried while respecting the
    global time budget (default 1000 s).  The best sequence found is
    returned as a plain Python list.

The implementation uses only NumPy, no external solvers, and runs on a
single CPU (FFT‑based convolutions make each iteration O(n log n)).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper families (normalised to sum = 1)
# ----------------------------------------------------------------------
def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-α), then normalised."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """a_i ∝ r^{i-offset} for i≥offset (0<r<1), otherwise 0."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """Decreasing linear (triangular) shape, optional leading zeros."""
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Fast full convolution using FFT (real -> real)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective used during optimisation (sum = 1 → 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a with Σa=1."""
    n = a.size
    conv = _full_convolution(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------
def _smooth_max_gradient(
    a: np.ndarray,
    mu: float,
    Nfft: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the smoothed max (log‑sum‑exp) and its gradient.

    Returns
    -------
    max_val : float
        True maximum of the convolution (used for reporting).
    grad : np.ndarray
        Gradient of the smooth objective w.r.t. a (before projection).
    """
    # FFT of a (zero‑padded to Nfft)
    Af = np.fft.rfft(a, n=Nfft)

    # Full convolution a*a
    conv_full = np.fft.irfft(Af * Af, n=Nfft)[: 2 * a.size - 1]

    # Soft‑max weights over convolution entries
    cmax = conv_full.max()
    # subtract cmax to avoid overflow
    w = np.exp(mu * (conv_full - cmax))
    w_sum = w.sum()
    p = w / w_sum                       # p_k = soft‑max weight

    # Linear correlation of a with p (gives Σ_k p_k a_{k‑j})
    Pf = np.fft.rfft(p, n=Nfft)
    corr_full = np.fft.irfft(Af * np.conj(Pf), n=Nfft)[: a.size]

    grad = 2.0 * corr_full               # g_j = 2·Σ_k p_k a_{k‑j}
    return cmax, grad


def _adam_step(a, grad, m, v, t, lr, beta1, beta2, eps):
    """
    One Adam update step (no projection). Returns the updated a, m, v.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def _anneal_optimize(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    global_best: float,
):
    """
    Refine a vector with smooth‑max Adam descent + simplex projection.

    Parameters
    ----------
    a_start : np.ndarray
        Starting point (already normalised to sum = 1).
    n : int
        Length of the vector.
    deadline : float
        Unix time after which the routine must stop.
    global_best : float
        Current best objective (used only for early‑exit information).

    Returns
    -------
    best_seq : np.ndarray
        Best sequence discovered for this length.
    best_score : float
        Corresponding objective value.
    """
    a = a_start.copy()
    # FFT size that can hold both a (len n) and p (len 2n‑1) for correlation
    Nfft = 1 << ((3 * n - 2 - 1).bit_length())

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5                # base learning rate (after Adam scaling)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    mu = 10.0               # start smoothing parameter
    mu_factor = 1.008
    mu_max = 2000.0

    best_seq = a.copy()
    best_score = _objective(a)
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- smooth‑max gradient ------------------------------------------------
        _, grad = _smooth_max_gradient(a, mu, Nfft)

        # ----- Adam update --------------------------------------------------------
        a, m, v = _adam_step(a, grad, m, v, t, lr, beta1, beta2, eps)

        # ----- projection onto the probability simplex -------------------------------
        a = _project_onto_simplex(a)

        # ----- annealing of µ -------------------------------------------------------
        mu = min(mu * mu_factor, mu_max)

        # ----- occasional exact evaluation (every ~20 iterations) --------------------
        if t % 20 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[anneal] n={n} iter={t} score={cur_score:.6f}")

            # early stop if we have already crossed the target
            if best_score < 1.503:
                break

        # ----- random perturbation to escape possible plateaus ----------------------
        if t % 500 == 0:
            # add small zero‑mean noise then re‑project
            noise = np.random.randn(n) * 0.001
            a = a + noise
            a = _project_onto_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs at most ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list).  All entries are non‑negative
    and the sum is guaranteed to be ≥ 0.01, satisfying the validator.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds margin for finalisation
    deadline = start_time + budget_s - 5.0

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # 1.  Candidate lengths (keep them moderate – expensive O(n log n) per iter)
    # ------------------------------------------------------------------
    length_candidates = [8000, 6000, 5000, 4000, 3000, 2000, 1000]

    # ------------------------------------------------------------------
    # 2.  Sweep a few parametric families for every length
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() > deadline:
            break

        # Keep the best raw (pre‑optimisation) candidate for this length
        cand_best = None
        cand_score = np.inf

        # Power‑law family
        for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
            for offset in (0, 5, 10, 30, 100):
                if offset >= n:
                    continue
                a = _power_sequence(n, alpha, offset)
                sc = _objective(a)
                if sc < cand_score:
                    cand_score, cand_best = sc, a.copy()
                    print(f"[sweep] n={n} power α={alpha:.2f} off={offset} score={sc:.6f}")

        # Exponential family
        for r in (0.90, 0.94, 0.96, 0.98, 0.99):
            for offset in (0, 5, 20, 50):
                if offset >= n:
                    continue
                a = _exp_sequence(n, r, offset)
                sc = _objective(a)
                if sc < cand_score:
                    cand_score, cand_best = sc, a.copy()
                    print(f"[sweep] n={n} exp r={r:.2f} off={offset} score={sc:.6f}")

        # Linear decreasing (triangular) family
        for offset in (0, 10, 30, 100):
            if offset >= n:
                continue
            a = _linear_sequence(n, offset)
            sc = _objective(a)
            if sc < cand_score:
                cand_score, cand_best = sc, a.copy()
                print(f"[sweep] n={n} linear off={offset} score={sc:.6f}")

        # Few random Dirichlet seeds
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            sc = _objective(a)
            if sc < cand_score:
                cand_score, cand_best = sc, a.copy()
                print(f"[sweep] n={n} random seed score={sc:.6f}")

        # If we already have a global best, try to interpolate it to this length
        if best_seq is not None:
            old_n = len(best_seq)
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, best_seq)
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a[:] = 1.0 / n
            else:
                a /= a.sum()
            sc = _objective(a)
            if sc < cand_score:
                cand_score, cand_best = sc, a.copy()
                print(f"[sweep] n={n} interp-from-best score={sc:.6f}")

        # ------------------------------------------------------------------
        # 3.  Local refinement by smooth‑max Adam descent
        # ------------------------------------------------------------------
        if cand_best is not None:
            seq_opt, score_opt = _anneal_optimize(
                cand_best, n, deadline, best_score
            )
            if score_opt < best_score:
                best_score, best_seq = score_opt, seq_opt.copy()
                print(f"[global] new best n={n} score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4.  Fallback – uniform sequence (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ----------------------------------------------------------------------
    # 5.  Final reporting
    # ----------------------------------------------------------------------
    print(
        f"[final] best_score={best_score:.6f} "
        f"best_len={best_seq.size if best_seq is not None else None}"
    )
    # The harness expects a plain list of floats
    return best_seq.tolist()
