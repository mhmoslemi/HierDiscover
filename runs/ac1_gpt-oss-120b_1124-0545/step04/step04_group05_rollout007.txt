"""
Optimisation of a non‑negative step function for the self‑convolution bound.

Algorithm
---------
0.   A quick grid‑search over symmetric and asymmetric Beta‑type
     densities (a_i ∝ x^{α‑1}(1‑x)^{β‑1}) is performed to obtain a
     good starting shape.  The grid is tiny (≈100 points) and runs in
     a few seconds.

1.   Sub‑gradient mirror‑descent on the exact objective
     f(a)=2·n·max(conv(a,a))  (the sum of the coefficients is kept equal
     to 1 by a soft‑max re‑parameterisation a=softmax(v)).
     The gradient of max(conv) is obtained analytically:
        ∂max/∂a_i = Σ_{k∈K}2·a_{k‑i}  (K = set of indices where the max
        is attained).  The chain rule for the soft‑max gives the update
        for v.  An Adam optimiser with a linearly decreasing learning‑rate
        is used and occasional restarts are triggered after 20 s of
        stagnation.

2.   Simulated‑annealing (SA) refinement.  From the best point of stage 1
     we repeatedly transfer a small fraction of mass from the coordinate
     with the largest sub‑gradient to one with a small sub‑gradient.
     Exact convolution is evaluated after each move; improvements are
     kept and worsening moves are accepted with probability
        exp(‑Δ / T) .
     The temperature T is linearly cooled.

Both stages respect the 1000 s budget.  The final sequence is normalised
to sum = 1 (the objective is scale‑invariant).  The best sequence is also
stored in the global variable `height_sequence_1` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    return exp_v / exp_v.sum()


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """Return a normalised non‑negative start vector.

    If a global `height_sequence_1` exists it is resampled to the target
    length `n`; otherwise a symmetric beta‑distribution is used.
    """
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        if old.size > 0:
            # resample by linear interpolation
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0.0:
                return a0 / a0.sum()

    # default: symmetric beta (α=β=0.5) → arcsine law
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def fft_len_for_n(n: int) -> int:
    """FFT size = next power of two ≥ 3·n‑2 (required for convolutions)."""
    return 1 << ((3 * n - 2).bit_length())


def conv_max_and_grad(a: np.ndarray, fft_len: int):
    """Return (max(conv(a,a)), gradient w.r.t. a of that max).

    The gradient is the sub‑gradient of max(conv) and equals
        g_i = Σ_{k∈K} 2·a_{k‑i}
    where K are the indices at which the maximum is attained.
    """
    n = a.shape[0]
    # linear convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    idx_max = np.where(np.abs(conv - max_val) <= 1e-12)[0]

    grad = np.zeros_like(a)
    for k in idx_max:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over [i_start, i_end]; j = k - i
        i_arr = np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] += 2.0 * a[k - i_arr]

    return max_val, grad, conv


def exact_score_from_max(n: int, max_conv: float) -> float:
    """Exact objective value (sum of coefficients is 1)."""
    return float(2.0 * n * max_conv)


def simplex_project(x: np.ndarray) -> np.ndarray:
    """Euclidean projection of x onto the probability simplex."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    if x.sum() == 1.0 and np.all(x >= 0):
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    Returns a list of floats (sum ≈ 1).  The routine follows the three‑stage
    strategy described in the module docstring and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    fft_len = fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 0) QUICK GRID SEARCH over Beta‑type families
    # ------------------------------------------------------------------
    best_a = initialise_a(n, seed=seed)  # fallback
    best_score = float('inf')
    # Grid limits – we stay inside (0,1] to avoid infinities.
    alphas = np.linspace(0.25, 0.75, 11)   # 0.25 … 0.75
    betas = np.linspace(0.25, 0.75, 11)
    for α in alphas:
        for β in betas:
            # early exit if we run out of time
            if time.time() > overall_deadline - 5.0:
                break
            # discretise the Beta(α,β) density on { (i+0.5)/n }
            x = (np.arange(n, dtype=float) + 0.5) / n
            # x^{α‑1} (1‑x)^{β‑1}
            w = np.power(x, α - 1.0) * np.power(1.0 - x, β - 1.0)
            # Guard against NaNs (should not happen with the chosen range)
            if np.any(np.isnan(w)):
                continue
            a = w
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                continue
            a = a / s
            max_c, _, _ = conv_max_and_grad(a, fft_len)
            score = exact_score_from_max(n, max_c)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[grid] α={α:.3f} β={β:.3f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 1) SUB‑GRADIENT MIRROR DESCENT (Adam on soft‑max logits)
    # ------------------------------------------------------------------
    # initialise logits
    a = best_a.copy()
    v = np.log(a + 1e-12)          # logits for the soft‑max
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    lr_start, lr_end = 0.30, 0.001           # learning‑rate schedule
    stage1_deadline = start_time + 0.70 * budget_s  # 70 % of the total budget

    it = 0
    last_improve = start_time
    best_score = exact_score_from_max(n, conv_max_and_grad(best_a, fft_len)[0])
    best_a = best_a.copy()

    while time.time() < stage1_deadline:
        elapsed = time.time() - start_time
        p = (elapsed) / (stage1_deadline - start_time)   # progression ∈[0,1]
        lr = lr_start * (1.0 - p) + lr_end * p

        # current point on the simplex
        a = softmax(v)

        max_val, grad_a, _ = conv_max_and_grad(a, fft_len)

        # chain‑rule for the soft‑max parametrisation
        dot_ag = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        cur_score = exact_score_from_max(n, max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{it:08d}] stage‑1 new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # restart if stuck for more than 20 s
        if it % 4000 == 0:
            if time.time() - last_improve > 20.0:
                v = np.log(best_a + 1e-12) + rng.normal(scale=1e-3, size=n)
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:08d}] restart after stagnation (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 2) SIMULATED‑ANNEALING REFINEMENT
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_score_sa = best_score
        best_a_sa = a.copy()

        sa_start = time.time()
        T_start = best_score_sa * 0.004    # start temperature = 0.4 % of the score
        T_end = best_score_sa * 1e-6

        while time.time() < overall_deadline:
            # linear temperature decay
            phase = (time.time() - sa_start) / remaining
            T = T_start * (1.0 - phase) + T_end * phase

            # evaluate current gradient (needed for directed mass transfer)
            max_val, grad_a, _ = conv_max_and_grad(a, fft_len)
            cur_score = exact_score_from_max(n, max_val)

            # pick donor (largest gradient) and receiver (smallest gradient)
            i = int(np.argmax(grad_a))
            # Choose a receiver among the 50 smallest‑gradient indices
            low_inds = np.argsort(grad_a)[:50]
            j = int(rng.choice(low_inds))
            if i == j:
                continue

            # move a modest fraction of mass from i to j
            move_frac = min(0.1, T / (T_start + 1e-12))   # cooler → smaller steps
            delta = move_frac * a[i] * rng.random()
            if delta <= 0.0:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate the proposal
            new_max, _, _ = conv_max_and_grad(a_new, fft_len)
            new_score = exact_score_from_max(n, new_max)

            # Metropolis acceptance
            if new_score < best_score_sa or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score_sa:
                    best_score_sa = new_score
                    best_a_sa = a_new.copy()
                    print(f"[SA] improved to {best_score_sa:.6f} (t={time.time() - start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a_sa, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    # expose the best sequence for future runs
    globals()["height_sequence_1"] = best_a.tolist()

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score_sa:.6f}, length={n}")

    return best_a.tolist()
