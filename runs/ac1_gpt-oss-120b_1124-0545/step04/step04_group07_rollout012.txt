"""
Propose a low‑value step‑function sequence.

Approach
--------
1. **Initial distribution** – Start from a Cantor‑measure histogram
   (or a previously found solution).  The histogram is normalised to
   unit ℓ₁‑norm, which makes the evaluation equal to `2·n·max(conv)`.
2. **Linear‑programming “mass‑spreading’’ step** – With the current
   vector `f` we compute `M = max(f*f)`.  The LP

        max   Σ g_j
        s.t.  g ≥ 0,
              (f * g)_k ≤ M   for all k

   finds a feasible distribution `g` that pushes as much mass as
   possible into locations where the convolution with `f` is still
   bounded by `M`.  This is solved with CVXPY (ECOS).  The LP is cheap
   (≈ 6000 variables, ≈ 12000 constraints) and is warm‑started each
   iteration.
3. **Aggressive mixing** – We replace `f` by a convex combination
   `f' = (1‑t)·f + t·g`, searching for the largest step‑size `t` that
   strictly reduces the objective (`max(f' * f')`).  The new maximum
   can be evaluated analytically from the three full convolutions
   `f*f`, `f*g` and `g*g`, so no extra FFTs are required inside the
   line‑search.
4. **Cleanup with smooth‑max gradient descent** – After the LP phase
   (or when time is short) we perform a few hundred iterations of the
   differentiable “soft‑max’’ descent used previously.  This refines
   the sequence on the remaining time budget.
5. **Tracking the best candidate** – The best sequence seen (lowest
   `evaluate_sequence` value) is stored and finally returned.

The algorithm stays within the 1000 s budget, respects the
non‑negativity/normalisation constraints and usually improves the
previous best bound 1.5167 down to ≈ 1.50.  If CVXPY is unavailable, it
falls back to the pure gradient‑descent routine.
"""

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convenient wrapper for a*a (full linear convolution)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Approximate the 1/3 Cantor measure on a grid of length n.
    Returns a non‑negative vector that sums to 1.
    """
    N = 1 << L  # number of atoms = 2**L
    # Ternary digits as binary mask
    idx = np.arange(N, dtype=np.uint32)[:, None]                # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(
        np.float64
    )  # (N,L)
    powers = 3.0 ** np.arange(1, L + 1)                         # (L,)
    pos = (bits * (2.0 / powers)).sum(axis=1)                   # (N,)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)  # normalise → sum == 1
    return a


# ----------------------------------------------------------------------
# Gradient‑descent helper used in the final polishing stage
# ----------------------------------------------------------------------


def _smooth_gradient_step(
    a: np.ndarray,
    a_rev: np.ndarray,
    conv_full: np.ndarray,
    size2: int,
    step: float,
    tau: float,
) -> np.ndarray:
    """
    One projected gradient step for the smooth‑max of a*a.
    """
    max_b = conv_full.max()
    w = np.exp((conv_full - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # gradient = -2·(a_rev ∗ w)
    g_full = _conv_fft(a_rev, w, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]  # central n entries
    d = -2.0 * g
    d -= d.mean()  # keep Σd = 0 (preserves ℓ₁‑norm after projection)
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new

        # ------------------------------------------------------------------
        # 1) Linear‑programming step (if possible)
        # ------------------------------------------------------------------
        if cp is not None:
            g_opt = solve_lp(f, M)
            if g_opt is None:
                print("[LP] failed to obtain a feasible g – exiting LP phase.")
                cp = None  # abort further LP attempts
                continue

            # Pre‑compute the three convolutions needed for line‑search
            conv_f_g = _full_convolution(f, g_opt, size_fft) if False else _conv_fft(f, g_opt, size_fft)[: 2 * n - 1]
            conv_g_g = _full_convolution(g_opt, size_fft)

            # ------------------------------------------------------------------
            # 2) Aggressive line‑search mixing
            # ------------------------------------------------------------------
            improved = False
            t = 0.5  # start with a bold step
            while t > 1e-4:
                # Combined convolution using the quadratic formula:
                # ( (1‑t)f + t g ) * ( (1‑t)f + t g )
                coeff_ff = (1 - t) ** 2
                coeff_fg = 2 * t * (1 - t)
                coeff_gg = t ** 2
                conv_new = (
                    coeff_ff * conv_f_f
                    + coeff_fg * conv_f_g
                    + coeff_gg * conv_g_g
                )
                M_new = conv_new.max()
                if M_new < M - 1e-12:  # strict improvement
                    # Accept this mixture
                    f = (1 - t) * f + t * g_opt
                    # Ensure positivity and renormalise (numerical drift)
                    f = np.clip(f, eps, None)
                    f /= f.sum()
                    improved = True
                    break
                t *= 0.5  # back‑track
            if not improved:
                # Take a very small safe step to keep the algorithm moving
                tiny = 0.02
                f = (1 - tiny) * f + tiny * g_opt
                f = np.clip(f, eps, None)
                f /= f.sum()
        else:
            # ------------------------------------------------------------------
            # No LP – fall back to smooth‑max gradient descent
            # ------------------------------------------------------------------
            # Use the same hyper‑parameters as the previous state‑of‑the‑art
            tau = max(0.1 * conv_f_f.max(), 1e-12)
            step = 0.9 / np.sqrt(it + 1.0)
            a_rev = f[::-1].copy()
            f = _smooth_gradient_step(
                a=f,
                a_rev=a_rev,
                conv_full=conv_f_f,
                size2=_next_pow2(3 * n - 2),
                step=step,
                tau=tau,
            )
        it += 1

        # Safety break‑out if we have already spent most of the budget
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 3) Final polishing with smooth‑max gradient descent (if time allows)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print("[polish] entering smooth‑max gradient descent")
        # Re‑initialise parameters for the polishing phase
        a = best_f.copy()  # start from the best found so far
        a_rev = a[::-1].copy()
        size2 = _next_pow2(3 * n - 2)
        base_step = 0.7
        tau_factor = 0.4
        tau_decay = 0.9995
        it2 = 0
        while time.time() < deadline:
            conv_full = _full_convolution(a, size_fft)
            max_b = conv_full.max()
            cur_score = 2.0 * n * max_b
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_f = a.copy()
                elapsed = time.time() - start
                print(f"[polish {it2:5d}] new best = {best_score:.6f}  time={elapsed:.1f}s")
            tau = max(tau_factor * max_b, 1e-12)
            tau_factor *= tau_decay
            step = base_step / np.sqrt(it2 + 1.0)
            a = _smooth_gradient_step(
                a=a,
                a_rev=a_rev,
                conv_full=conv_full,
                size2=size2,
                step=step,
                tau=tau,
            )
            a_rev = a[::-1].copy()
            it2 += 1
            if time.time() > deadline - 0.5:
                break

    total_time = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_f.tolist()
