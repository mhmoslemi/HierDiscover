"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Method
------
1. Exhaustive grid search over two analytic families (product‑law and sum‑law)
   with a fine scan of the exponent *p* and the offset.  Also test a simple
   exponential shape and the uniform distribution.  Keep the best one.
2. From the best candidate (and a few random Dirichlet seeds) run a
   continuation of smoothed‑max L‑BFGS optimisation.
   The smoothed objective is
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   whose gradient is computed analytically with FFTs.
   The λ‑schedule goes from a very smooth approx (λ≈0.5) up to a hard max
   (λ≈10 000).  If SciPy is unavailable a tiny Adam optimiser is used instead.
3. A few sub‑gradient steps on the exact max‑convolution are applied (the
   sub‑gradient of max(conv) is 2·a shifted by the index of the current
   maximum).  After each step we project back onto the probability simplex.
4. Finally a light random‑mass‑transfer local search (simulated annealing‑like)
   is performed until the time budget expires.

All helper functions are top‑level (no closures) and the code respects a
1000 s time budget.  The best sequence found is returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a (the sum is assumed >0).
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another and project
    back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def exponential_candidate(n: int, p: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-p·i) + exp(-p·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-p * i) + np.exp(-p * (n - 1 - i))
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    Smoothed max Φ_λ and its gradient w.r.t. the soft‑max logits x.
    Returns (phi, grad_x).
    """
    # ----- a = softmax(x) -----
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # ----- convolution a*a -----
    conv = conv_fft_self(a)                 # length 2n‑1
    max_c = np.max(conv)

    # ----- smooth max over convolution entries -----
    shift = conv - max_c                     # stabilise
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam      # Φ_λ

    w = exp_term / sum_exp                   # weights (sum to 1)

    # ----- gradient of Φ_λ w.r.t. a (∂Φ/∂a) -----
    w_rev = w[::-1]                         # reverse weights
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    slice_part = conv_wa[n - 1: 2 * n - 1]   # length n, part needed for gradient
    grad_a = 2.0 * slice_part[::-1]          # ∂Φ/∂a

    # ----- map gradient to x‑space using soft‑max Jacobian -----
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)              # ∇_x Φ = J^T ∇_a Φ
    return phi, grad_x


# ----------------------------------------------------------------------
# Optimisers (top‑level)
# ----------------------------------------------------------------------


def run_lbfgs(x, lam, n, max_iter, have_scipy):
    """
    Run a limited‑memory BFGS step on the smoothed objective.
    Returns the updated x.
    """
    if have_scipy:
        from scipy.optimize import minimize
        res = minimize(
            compute_phi_and_grad,
            x,
            args=(lam, n),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'disp': False}
        )
        return res.x
    else:
        # Small Adam fallback (deterministic, no external deps)
        alpha = 0.05
        beta1, beta2 = 0.9, 0.999
        eps_opt = 1e-8
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        for t in range(1, max_iter + 1):
            phi, grad = compute_phi_and_grad(x, lam, n)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
            x = x - step
        return x


def subgradient_refine(a, n, deadline, best_score, best_a):
    """
    Perform a few sub‑gradient steps on the exact max‑convolution.
    Returns (a_new, best_score, best_a).
    """
    step = 0.2  # initial step size
    while time.time() < deadline and step > 1e-5:
        conv = conv_fft_self(a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        valid = (k - idx >= 0) & (k - idx < n)
        grad = np.zeros_like(a)
        grad[valid] = 2.0 * a[k - idx[valid]]

        a = project_simplex(a - step * grad)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        step *= 0.96
    return a, best_score, best_a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that
    minimises the evaluation function within the given time budget.
    The routine performs a dense analytic grid search, L‑BFGS continuation
    on a smoothed max‑convolution, sub‑gradient polishing and a final
    random local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so that the harness does not kill us exactly at the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search (product, sum, exponential, uniform)
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Uniform baseline
    a_uni = np.ones(n) / n
    sc = compute_score(a_uni)
    if sc < best_score:
        best_score, best_a = sc, a_uni.copy()

    # Global variable if available (may already be a good start)
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            if ref.sum() > 0:
                ref = ref / ref.sum()
                sc = compute_score(ref)
                if sc < best_score:
                    best_score, best_a = sc, ref.copy()

    # Grid for product and sum families
    p_vals = np.arange(0.35, 0.81, 0.025)       # 0.35 … 0.80
    off_vals = np.arange(0.0, 2.01, 0.20)       # 0 … 2
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-9
                cand = analytic_candidate(n, float(p), float(off_eff), mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()

    # Exponential family
    exp_ps = np.arange(0.1, 3.01, 0.2)          # 0.1 … 3.0
    for p in exp_ps:
        cand = exponential_candidate(n, float(p))
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # A few random Dirichlet seeds (lightweight)
    for _ in range(8):
        if time.time() >= deadline:
            break
        a_rand = np.random.rand(n)
        a_rand = a_rand / a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_a = sc, a_rand.copy()

    print(f"[init] best analytic / random start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the smoothed max‑convolution
    # ------------------------------------------------------------------
    # Determine if SciPy is present
    try:
        import scipy.optimize  # noqa: F401
        have_scipy = True
    except Exception:  # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – falling back to Adam optimiser.")

    # Prepare the log‑softmax parameters from the best start
    eps = 1e-12
    x0 = np.log(np.maximum(best_a, eps))

    # λ‑schedule (geometric growth, ending at a hard max)
    lam_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0, 10000.0]

    # Run the continuation for the best start
    x = x0.copy()
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest number of iterations (depends on remaining time)
        max_iter = int(min(400, max(20, remaining * 70)))
        x = run_lbfgs(x, lam, n, max_iter, have_scipy)
        a_opt = softmax(x)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score, best_a = sc, a_opt.copy()
            elapsed = time.time() - start_time
            print(f"[lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 3) Additional restarts from random seeds (lighter optimisation)
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline and restart_counter < 12:
        # generate a fresh random start
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        x = np.log(np.maximum(a0, eps))
        # Run a shortened λ‑schedule (skip the very fine final λ)
        for lam in lam_schedule[:5]:          # 0.5 … 200
            if time.time() >= deadline:
                break
            max_iter = 150
            x = run_lbfgs(x, lam, n, max_iter, have_scipy)
        a_opt = softmax(x)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score, best_a = sc, a_opt.copy()
            elapsed = time.time() - start_time
            print(f"[restart {restart_counter}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        restart_counter += 1

    # ------------------------------------------------------------------
    # 4) Sub‑gradient polishing on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined, best_score, best_a = subgradient_refine(
            best_a.copy(), n, deadline, best_score, best_a.copy()
        )
        # keep the refined version if it helped
        if best_score < compute_score(best_a):
            pass  # best_a already updated inside subgradient_refine

    # ------------------------------------------------------------------
    # 5) Light random‑mass‑transfer local search (simulated‑annealing flavour)
    # ------------------------------------------------------------------
    a_cur = best_a.copy()
    scale = 0.02  # starting relative transfer amount
    while time.time() < deadline:
        cand = random_move(a_cur, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a_cur = cand.copy()
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            scale = max(0.001, scale * 0.9)
        else:
            # slowly vary the step size to keep exploration alive
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    # The evaluation harness will clamp negatives etc.; we already ensure non‑negativity
    return best_a.tolist()
