"""
Advanced step‑function search for minimise `evaluate_sequence`.

Strategy
--------
1. **Seed generation** – creates a rich pool of initial vectors:
   uniform, geometric / power‑law families, smooth windows,
   random soft‑max, sparse random, beta‑distribution and Dirichlet draws.
   Previously discovered best vector (if any) is also injected.

2. **Adam on a smoothed max** – the sequence is parametrised by log‑softmax
   logits, the loss is a log‑sum‑exp approximation of the true max‑convolution.
   Adam updates keep the vector on the simplex.  A relatively long schedule
   (up to 12 k iterations) is run for each promising seed.

3. **Exact sub‑gradient refinement** – mirror‑descent (exponentiated‑gradient)
   using the true sub‑gradient of the unsmoothed max.  Up to 80 k iterations
   improve the solution without any smoothing.

4. **Simulated‑annealing fine‑tuning** – cheap Metropolis random walk that
   shifts a tiny fraction of mass between two positions; temperature
   decays exponentially.

5. **Upsampling** – the best vector found for the base length (default 4000)
   is stretched to longer lengths (×2 and ×3) by inserting zeros between
   entries.  The longer vectors inherit the low‑peak property and are
   re‑optimised (sub‑gradient + anneal).  Larger n reduces the theoretical
   lower bound (≈ 2 n/(2 n‑1) → 1) and therefore helps to push the objective
   well below the previous 1.512 … score.

All phases are time‑aware: they stop as soon as the global deadline
(budget − 1 s safety margin) is reached.  The best sequence discovered is
returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns full length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to the harmless clipping)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / sum(seq)².
    Sequence must be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


def _sparse_random(n: int, rng: np.random.Generator, sparsity: float = 0.02) -> np.ndarray:
    """Select a few random positions, give them equal mass."""
    m = max(2, int(n * sparsity))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _beta_distribution(n: int, rng: np.random.Generator, a: float = 0.5, b: float = 0.5) -> np.ndarray:
    """Beta‑distributed heights, then normalised."""
    vals = rng.beta(a, b, size=n)
    s = vals.sum()
    return vals / s if s > 0 else _uniform(n)


def _dirichlet_seed(n: int, rng: np.random.Generator, conc: float = 0.3) -> np.ndarray:
    """Sparse Dirichlet draw."""
    vals = rng.dirichlet(np.full(n, conc))
    return vals


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a diverse set of initial vectors (already normalised)."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # stochastic families
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))

    for _ in range(3):
        seeds.append(_sparse_random(n, rng, sparsity=0.015))

    for _ in range(2):
        seeds.append(_beta_distribution(n, rng, a=0.4, b=0.6))

    for _ in range(2):
        seeds.append(_dirichlet_seed(n, rng, conc=0.2))

    # re‑use previously discovered construction (if length matches)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # safety
    if not seeds:
        seeds.append(_uniform(n))
    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑sum‑exp approximation of max(conv).
    Returns the best sequence encountered and its true objective value.
    """
    v = np.log(seed + 1e-30)          # initialise logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters
    lr0 = 0.9
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule
    beta = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                 # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # smoothed‑max weights
        beta = beta * beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 80000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the true sub‑gradient of max(conv).
    The update is an exponentiated‑gradient step (automatically respects the simplex).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk: move a small fraction of mass between two random
    positions; temperature decays exponentially.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current objective
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # propose moving a tiny fraction of mass from i to j
        delta = rng.random() * 0.025 * cur[i]   # up to 2.5 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:6d} best={new_val:.6f}")

        # occasional global jitter
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            tmp = cur + noise
            tmp = np.clip(tmp, 0.0, None)
            s = tmp.sum()
            if s > 0.0:
                tmp /= s
                new_val = _objective(tmp)
                accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
                if accept:
                    cur = tmp
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best = tmp.copy()
                        print(f"[Anneal] n={n} it={it:6d} (jitter) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Upsampling helper – spread entries by inserting zeros
# ----------------------------------------------------------------------


def _upsample_with_zeros(seq: np.ndarray, factor: int) -> np.ndarray:
    """
    Return a longer vector where each original entry is placed at positions
    0, factor, 2·factor, … (zeros elsewhere).  Length = factor * len(seq).
    """
    n_old = seq.shape[0]
    n_new = n_old * factor
    up = np.zeros(n_new, dtype=np.float64)
    up[np.arange(n_old) * factor] = seq
    return up

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0     # safety margin

    # try to reuse a previously successful construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Stage 1 – optimise a moderate length (base length)
    # ------------------------------------------------------------------
    base_n = 4000
    global_best_seq = None
    global_best_val = float("inf")

    if time.time() < deadline:
        print(f"\n=== Base optimisation n={base_n} ===")
        seeds = _make_seeds(base_n, previous)

        # quick evaluation of raw seeds
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(12, len(seed_vals))]]

        # record best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] new global best={global_best_val:.6f}")

        # ---- Adam phase ----
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, base_n, deadline, max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] seed={idx} best={val:.6f}")

        # ---- Sub‑gradient phase ----
        if global_best_seq is not None and len(global_best_seq) == base_n:
            seq, val = _subgradient_refine(global_best_seq, base_n, deadline,
                                           max_iters=80000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] best={val:.6f}")

        # ---- Annealing phase ----
        if global_best_seq is not None and len(global_best_seq) == base_n:
            seq, val = _anneal_optimize(global_best_seq, base_n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] best={val:.6f}")

    # Stage 2 – upsample to longer lengths and re‑refine
    # ------------------------------------------------------------------

        if global_best_seq is None:

        n_new = base_n * factor
        if time.time() > deadline:

        print(f"\n=== Upsampling factor={factor} → length {n_new} ===")
        up_seq = _upsample_with_zeros(global_best_seq, factor)

        # sub‑gradient on the upsampled vector (shorter budget)
        seq, val = _subgradient_refine(up_seq, n_new, deadline, max_iters=30000)
        if seq is not None and val < global_best_val:

            global_best_seq = seq.copy()
            print(f"[Upsample SubGrad] factor={factor} best={val:.6f}")

        # annealing refinement

                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Upsample Anneal] factor={factor} best={val:.6f}")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
