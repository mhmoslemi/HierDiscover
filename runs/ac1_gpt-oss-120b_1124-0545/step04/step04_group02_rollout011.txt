import numpy as np
def compute_conv(a):
    n = a.shape[0]
    size = 1 << (2*n-1-1).bit_length()
    conv = np.fft.ifft(np.fft.fft(a, size) * np.fft.fft(a, size)).real
    return conv[:2*n-1]

def gradient_smooth(a, gamma=10):
    conv = compute_conv(a)
    exp = np.exp(gamma * conv - np.max(gamma * conv))
    w = exp / exp.sum()
    # compute correlation using FFT
    n = a.shape[0]
    Lw = conv.shape[0]
    size = 1 << (n + Lw - 1).bit_length()

    corr = np.fft.ifft(fft_a * np.conj(fft_w)).real
    # corr length size; linear correlation for lags >=0: corr[l] = sum_i a_i w_{i+l}
    idx1 = np.arange(n)
    idx2 = 2*(n-1 - np.arange(n))

"""
Propose a non‑negative step function that (approximately) solves

    minimise  2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a deterministic‑annealing Adam optimiser on a smooth
approximation of the max‑convolution:

* a is kept on the probability simplex (sum = 1) – the objective is
  scale‑invariant, so any normalisation works for the final answer.
* At each iteration the convolution b = a*a is computed.
* A soft‑max with temperature γ turns the max into a smooth,
  differentiable loss L = ∑ w_k·b_k (w = soft‑max(γ·b)).
* The exact gradient of L w.r.t. a can be written as a cheap
  correlation involving a and the weight vector w; it is obtained
  with two FFTs (O(n log n)).
* Adam with Euclidean projection onto the simplex updates a.
* γ is increased linearly from 1 to 80, turning the smooth loss into the
  true max‑loss.
* Early on we initialise a from the global reference vector (if
  available) by linear interpolation to the current length,
  otherwise from a simple U‑shaped analytic family.
* In the last seconds a fine‑grained Gaussian jitter refines the best
  found vector.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def conv_full(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of a and b using FFT.
    Returns the first len(a)+len(b)‑1 elements.
    """
    fft_a = np.fft.rfft(a, n=size)
    fft_b = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fft_a * fft_b, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a that already lies on the
    simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another and
    re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step function (list of non‑negative floats) that minimizes
    ``evaluate_sequence``.  The routine uses a deterministic‑annealing Adam
    optimiser on a smooth approximation of the max‑convolution.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # 1) Choose length and initial vector
    # --------------------------------------------------------------

    else:
        # analytic U‑shape (product family) as a quick starter
        i = np.arange(n, dtype=np.float64)
        p0, off0 = 1.0, 0.5
        a = ((i + off0) * (n - 1 - i + off0)) ** (-p0)
        a = np.where(np.isfinite(a), a, 0.0)
        a = project_simplex(a)

    best_seq = a.copy()
    best_score = compute_score(best_seq)

    # --------------------------------------------------------------
    # 2) Deterministic‑annealing Adam optimisation
    # --------------------------------------------------------------
    size_g = _next_pow2(3 * n - 2)          # size for gradient convolution
    m = np.zeros_like(a)                   # Adam first moment
    v = np.zeros_like(a)                   # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_base = 0.1

    gamma_start, gamma_end = 1.0, 80.0
    last_improve = start_time
    iteration = 0
    stagnation_limit = 30.0                # seconds without improvement

    while time.time() < deadline:
        # --- exact loss (for book‑keeping) ---------------------------------
        conv = conv_fft(a)
        cur_max = np.max(conv)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # --- smooth approximation of the max (soft‑max) --------------------
        progress = (time.time() - start_time) / budget_s
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        scaled = gamma * conv
        scaled_max = np.max(scaled)
        w = np.exp(scaled - scaled_max)
        w /= w.sum()                         # w is the soft‑max weight vector

        # --- gradient of the smooth loss ------------------------------------
        w_rev = w[::-1]                      # reverse for correlation
        conv_aw = conv_full(a, w_rev, size_g)   # length = 3·n‑2

        idx1 = (2 * n - 2) - np.arange(n)    # 2n‑2‑j
        idx2 = 2 * np.arange(n)              # 2·j
        g = 2.0 * (conv_aw[idx1] + conv_aw[idx2])

        # --- Adam update ----------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr_base * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # --- occasional random mass‑transfer move ----------------------------
        if np.random.rand() < 0.01:
            a_candidate = random_move(a_candidate, scale=0.02)

        # --- stagnation handling ---------------------------------------------
        if time.time() - last_improve > stagnation_limit:
            jitter = np.random.rand(n) * 1e-3
            a = project_simplex(best_seq + jitter)
            last_improve = time.time()
            continue

        a = a_candidate

    # --------------------------------------------------------------
    # 3) Final fine‑grained stochastic refinement
    # --------------------------------------------------------------
    refine_deadline = start_time + budget_s - 0.2
    while time.time() < refine_deadline:
        cand = best_seq + np.random.normal(scale=5e-4, size=n)
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Scale to the conventional normalisation (sum = sqrt(2·n))
    # --------------------------------------------------------------
    scale = math.sqrt(2.0 * n)
    final_seq = (best_seq * scale).tolist()

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, n = {n}, iterations = {iteration}, "
          f"time = {total_time:.1f}s")
    return final_seq
