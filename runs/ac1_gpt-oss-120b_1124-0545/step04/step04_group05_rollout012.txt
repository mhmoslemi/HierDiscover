def proj_simplex(v):
    # Project onto simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def max_conv(a, fft_len=None):
    n = len(a)
    if fft_len is None:
        fft_len = 1 << ((2*n-1).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
    return conv.max()
```

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.  Implements the classic O(n log n) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically all entries are zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Returns max(conv(a, a)) using an FFT of length `fft_len`
    (linear convolution, i.e. length 2*len(a)-1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return float(conv.max())


def objective_from_max(n: int, conv_max: float) -> float:
    """Exact value of `evaluate_sequence` when the vector is already
    normalised to sum = 1."""
    return 2.0 * n * conv_max


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Exact discretisation of the continuous arcsine density
    f(x)=1/(π√(x(1‑x))) on [0,1] by integrating over each bin
    [i/n,(i+1)/n).  The resulting vector sums to 1.
    """
    i = np.arange(n, dtype=float)
    left = i / n
    right = (i + 1) / n
    a = (np.arcsin(2.0 * right - 1.0) - np.arcsin(2.0 * left - 1.0)) / np.pi
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta(α,β) density on a uniform grid.
    The density is evaluated at the bin centre (i+0.5)/n and then
    normalised to sum = 1.  Works for any α,β>0.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return a / total


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray,
                         max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv(a,a))  w.r.t. a.
    For each index s where the max is attained,
    the contribution is 2·a shifted by s.
    The returned gradient is the average over all maximisers.
    """
    n = a.size
    g = np.zeros_like(a)
    for s in max_idx:
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        # i runs from i_start … i_end inclusive
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = s - i
        g[i] += 2.0 * a[j]
    if max_idx.size > 0:
        g /= float(max_idx.size)
    return g


def evaluate_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the exact score used by `evaluate_sequence` for a normalised
    vector (sum = 1).  Avoids the overhead of calling the external function.
    """
    mx = max_convolution(a, fft_len)
    return objective_from_max(a.size, mx)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm proceeds through a fast
    parametric sweep, a projected sub‑gradient descent, and a final
    stochastic refinement.  It respects the time budget (default 1000 s)
    and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin
    overall_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))
    # FFT size = next power of two ≥ 2*n-1 (required for convolution)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 0) Build a small pool of simple candidates
    # ------------------------------------------------------------------
    candidates = []

    # 0a) reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # 0b) exact arcsine discretisation
    candidates.append(arcsine_distribution(n))

    # 0c) uniform distribution (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # 0d) symmetric Beta families (α = β) – explore a reasonable range
    for alpha in np.linspace(0.25, 2.0, 16):   # 0.25,0.35,…,2.0
        candidates.append(beta_distribution(n, alpha, alpha))

    # 0e) mixtures of arcsine and uniform
    arcsine = candidates[1]   # already computed
    uniform = candidates[2]
    for lam in np.linspace(0.0, 1.0, 11):   # λ·arcsine + (1‑λ)·uniform
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix /= mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # 1) Pick the best among the cheap candidates
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = evaluate_vec(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[init {idx:02d}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    # allocate ~70 % of the total time for this phase
    phase1_deadline = start_time + 0.70 * budget_s
    a = best_a.copy()
    it = 0
    last_improve = time.time()
    while time.time() < phase1_deadline:
        # current convolution and its maximisers
        conv_vals = np.real(np.fft.ifft(np.fft.fft(a, fft_len) *
                                        np.fft.fft(a, fft_len)))[: 2 * n - 1]
        conv_max = float(conv_vals.max())
        max_idxs = np.where(np.abs(conv_vals - conv_max) < 1e-14)[0]

        # sub‑gradient (average over all maximisers)
        g = subgradient_max_conv(a, conv_vals, max_idxs)

        # step size – classic 𝛼 / √t schedule
        it += 1
        lr = 0.1 / np.sqrt(it)

        # gradient step and simplex projection
        a = proj_simplex(a - lr * g)

        # evaluate exact score
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[PGD {it:07d}] improved to {best_score:.6f}")

        # occasional random restart if we are stuck
        if time.time() - last_improve > 30.0:
            # perturb the best vector slightly and project back
            noise = rng.normal(scale=0.02, size=n)
            a = proj_simplex(best_a + noise)
            last_improve = time.time()
            print(f"[PGD {it:07d}] restart after stagnation")

    # ------------------------------------------------------------------
    # 3) Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    remaining = overall_deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # linear temperature schedule (cooling)
        T_start = best_score * 0.004
        T_end = best_score * 1e-6
        sa_start = time.time()
        sa_it = 0

        while time.time() < overall_deadline:
            elapsed = time.time() - sa_start
            frac = min(1.0, elapsed / remaining)
            T = T_start * (1.0 - frac) + T_end * frac

            # propose a random elementary mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # move up to 5 % of a[i] to a[j]
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # exact evaluation of the proposal
            new_conv_max = max_convolution(a_new, fft_len)
            new_score = 2.0 * n * new_conv_max

            # Metropolis rule
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[SA {sa_it:07d}] new best = {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
