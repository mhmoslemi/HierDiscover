"""
Candidate proposer using a linear‑programming (LP) refinement loop.

The method starts from a good seed (the previously discovered best
sequence if it exists, otherwise a set of diverse seeds), then repeatedly
solves a tiny LP:

    maximise   sum(g)
    subject to  g ≥ 0
                (f * g)(k) ≤ M   for all k

where M = max(f * f) and f is the current step‑function.  
If the LP can increase the total mass while keeping the same bound M,
the normalised solution ĝ = g / sum(g) is mixed with the current f.
A simple back‑tracking line‑search on the mixing weight guarantees that
the objective 2·n·max(conv(f,f))/sum(f)² never increases.  After a few
LP updates the sequence is polished with the exact sub‑gradient mirror
descent from the original code and a short annealing phase.

All operations respect the 1000 s time budget, use only non‑negative
numbers, and return a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.fft

# optional imports – the LP step uses CVXPY if it is available
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False

# Convolution / objective utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = scipy.fft.rfft(a, n=N)
    fb = scipy.fft.rfft(b, n=N)
    conv = scipy.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / sum(seq)²  (seq is non‑negative)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 6.0, 13):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 13):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 7):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 7):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 5.0, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.18))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    if not seeds:            # safety net – should never happen
        seeds.append(_uniform(n))

    return seeds

def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve  max sum(g)  s.t. g≥0 and conv(f,g) ≤ M.
    Returns the optimal g (may contain tiny negatives → clipped to 0)."""
    if not _HAS_CVXPY:      # pragma: no cover
        return None

    n = f.size
    g = cp.Variable(n)

    # element‑wise convolution constraint; cp.conv returns a vector
    conv_expr = cp.conv(f, g)
    constraints = [g >= 0, conv_expr <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception:      # pragma: no cover
        return None

    if g.value is None:
        return None

    g_val = np.asarray(g.value, dtype=np.float64)
    g_val[g_val < 0.0] = 0.0
    return g_val


def _lp_refine(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_outer: int = 12) -> tuple[np.ndarray, float]:
    """Iteratively improve a sequence by the LP step described in the paper.
    A simple back‑tracking line search guarantees monotone improvement."""
    f = start_seq.copy()
    best_seq = f.copy()
    best_val = _objective(f)

    # initial mixing coefficient (will be reduced if no improvement)
    mix = 0.5

    for it in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(f, f)
        M = conv.max()

        g = _lp_step(f, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g

        # ----- line‑search on mixing weight -----
        step = mix
        improved = False
        while step > 1e-6:
            new_f = (1.0 - step) * f + step * g_norm
            new_val = _objective(new_f)
            if new_val < best_val - 1e-12:
                # accept
                f = new_f
                best_seq = new_f.copy()
                best_val = new_val
                improved = True
                # keep a generous mixing factor for next iteration
                mix = max(step * 0.9, 0.01)
                print(f"[LP] n={n} it={it:2d} step={step:.4f} best={new_val:.6f}")
                break
            step *= 0.5

        if not improved:
            # no improvement – shrink mixing parameter and try again later
            mix *= 0.5
            if mix < 1e-4:
                break

    return best_seq, best_val

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) refinement using the true sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Tiny Metropolis‑Hastings walk moving a small fraction of mass between two positions."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n} it={it:5d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n} it={it:5d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence
    discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # reuse a previously computed construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:               # pragma: no cover
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start with 4000, then 8000 if time permits
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) generate seeds
        seeds = _make_seeds(n, previous)

        # 2) quick evaluation, keep a few best raw seeds
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda x: x[1])
        top_seeds = [s for s, _ in seed_vals[:min(9, len(seed_vals))]]

        # record the best raw seed
        if seed_vals[0][1] < global_best_val:
            global_best_val = seed_vals[0][1]
            global_best_seq = seed_vals[0][0].copy()
            print(f"[seed] n={n} new global best={global_best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase A – LP refinement (the new ingredient)
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq, val = _lp_refine(seed_vec, n, deadline, max_outer=12)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall LP] n={n} seed={idx} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase B – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                          max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall SubGrad] n={n} best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase C – Simulated‑annealing fine‑tuning
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                        max_iters=None)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Anneal] n={n} best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
