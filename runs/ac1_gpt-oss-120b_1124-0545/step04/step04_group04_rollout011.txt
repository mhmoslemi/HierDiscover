def solve_lp_max_sum(H, M):
    import cvxpy as cp
    n = H.shape[1]
    g_var = cp.Variable(n, nonneg=True)
    constraints = [H @ g_var <= M + 1e-12]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver='ECOS', warm_start=False, max_iters=5000, verbose=False)
    except Exception as e:
        print("LP solve failed:", e)
        return None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        print("LP status:", prob.status)
        return None
    g_val = g_var.value
    if g_val is None:
        return None
    # Clip negative due to numerical issues
    g_val = np.maximum(g_val, 0.0)
    return g_val

"""
Propose a better step‑function sequence.

The optimizer works in two stages.
  1.  A Frank‑Wolfe‑type iteration: given the current
      distribution `a`, we solve a linear programme
      max ∑ g_j   s.t.   a * g ≤ M  (pointwise) and g ≥ 0,
      where M = max(a * a).  The LP tends to increase the
      total mass that can be placed under the same convolution
      bound, after which we move a little towards the LP
      solution (convex combination).  Repeating this step drives
      the peak of the self‑convolution down towards the theoretical
      lower bound 1/(2n‑1).

  2.  A short Adam‑based soft‑max optimisation (the code used in
      the previous run) to fine‑tune the final distribution.

Both phases respect the time budget (default 1000 s).  All
constraints of `evaluate_sequence` (non‑negative, sum‑invariant)
are satisfied automatically.
"""
import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    return exp_v / exp_v.sum()


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] for a real vector a (sum(a) = 1)."""
    n = a.size
    fft_len = _next_pow2(3 * n - 2)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(conv.max())


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Construct the linear mapping `g ↦ a*g` as a sparse matrix.
    For a length‑n vector a this returns a (2n‑1)×n matrix H with
        H[k, j] = a[k‑j]   when 0 ≤ k‑j < n, else 0.
    """
    n = a.size
    # each diagonal offset d (0 … n‑1) is constant a[d]
    diags = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)          # offsets = row - col
    H = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return H


def solve_lp_max_sum(H: sp.csr_matrix, M: float, n: int):
    """
    Solve   max 1ᵀg   s.t.   H g ≤ M ,   g ≥ 0
    Returns the optimal g as a dense np.ndarray (or None on failure).
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[lp] cvxpy not available:", e)
        return None

    g = cp.Variable(n, nonneg=True)
    constraints = [H @ g <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS works well for dense LPs of this size
    try:
        prob.solve(solver="ECOS", warm_start=False, max_iters=5000, verbose=False)
    except Exception as e:
        print("[lp] solving failed:", e)
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        print("[lp] problem status:", prob.status)
        return None

    g_val = g.value
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def frank_wolfe_step(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Perform a single Frank‑Wolfe update:
        * compute current M = max(a*a)
        * solve the LP to obtain a direction g
        * move a ← (1‑lam)·a + lam·(g/∑g)   and renormalise
    Returns the new a and the new M.
    """
    n = a.size
    M = max_convolution(a)

    # build the convolution matrix for the current a
    H = build_conv_matrix(a)

    g_raw = solve_lp_max_sum(H, M, n)
    if g_raw is None or g_raw.sum() == 0.0:
        return a, M

    # normalise g to a probability vector (sum = 1)
    g = g_raw / g_raw.sum()

    # convex combination (lam ∈ (0,1])
    a_new = (1.0 - lam) * a + lam * g
    # re‑normalise to guard against tiny numerical drift
    a_new = a_new / a_new.sum()
    M_new = max_convolution(a_new)
    return a_new, M_new


# ----------------------------------------------------------------------
# Adam optimiser (same idea as the previous state‑of‑the‑art code)
# ----------------------------------------------------------------------


def _optimize_a(
    a_init: np.ndarray,
    n: int,
    alloc_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑parameterisation (soft‑max)."""
    v = np.log(a_init + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta_start, beta_end = 0.1, 3000.0
    lr_start, lr_end = 0.30, 0.001

    fft_len = _next_pow2(3 * n - 2)

    start = time.time()
    deadline = start + max(0.0, alloc_s - 0.10)

    best_a = a_init.copy()
    # exact score for a probability vector
    best_score = 2.0 * n * max_convolution(best_a)
    last_improve = start
    it = 0
    mask_phase_start = 0.68

    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (alloc_s - 5.0))

        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        a = softmax(v)

        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # weight vector for the (smoothed) max
        if p < mask_phase_start:
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)
        else:
            max_val = conv_aa.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv_aa)
                mask[np.argmax(conv_aa)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # gradient of the (smoothed) max – FFT based
        fw = np.fft.fft(w, fft_len)
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]
        g = 2.0 * n * (g1 + g2)

        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        cur_score = 2.0 * n * conv_aa.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 3000 == 0:
                print(f"[ADAM {it:07d}] new best = {best_score:.6f} "
                      f"(t={elapsed:.1f}s, β={beta:.1f})")

        # simple restart if stagnating
        if time.time() - last_improve > 18.0:
            jitter = rng.normal(scale=0.008, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(f"[ADAM {it:07d}] restart after {elapsed:.1f}s (no improve)")

    return best_a, best_score

                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = 2.0 * n * max_convolution(best_a)

    # ------------------------------------------------------------------
    # 1) Frank‑Wolfe LP refinement (takes roughly 60‑70% of the budget)
    # ------------------------------------------------------------------
    lp_time_budget = (budget_s - 5.0) * 0.65   # leave time for Adam
    lp_deadline = start_time + lp_time_budget
    a = a0.copy()
    iter_fw = 0

    while time.time() < lp_deadline:
        # current max‑convolution value
        M = max_convolution(a)

        # try a Frank‑Wolfe step with a modest step size
        lam = 0.35   # 0 < lam ≤ 1
        a_new, M_new = frank_wolfe_step(a, lam)

        # if the step was not improving, shrink lam and retry a few times
        if M_new >= M:
            # back‑off strategy
            for _ in range(4):
                lam *= 0.5
                if lam < 1e-4:
                    break
                a_new, M_new = frank_wolfe_step(a, lam)
                if M_new < M:
                    break

        # accept the best of the candidates
        if M_new < M:
            a = a_new
            M = M_new

        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[FW {iter_fw:04d}] improved score = {best_score:.6f}")

        iter_fw += 1
        # stop after a reasonable number of FW steps to keep time budget
        if iter_fw >= 30:
            break

    # ------------------------------------------------------------------
    # 2) Short Adam fine‑tuning on the best candidate found so far.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 6.0:
        a_opt, score_opt = _optimize_a(
            a_init=best_a,
            n=n,
            alloc_s=time_left,
            rng=np.random.default_rng(seed + 9999),
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[ADAM] final fine‑tune improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence found.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    best_a = best_a / best_a.sum()          # re‑normalise just in case
    total_time = time.time() - start_time
    print(f"[DONE] best score = {best_score:.6f}, time = {total_time:.1f}s")
    return best_a.tolist()
