"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Analytic grid search over several parametric families (product, sum,
   exponential, cosine‑power, beta).  The 10 best candidates are kept.
2. Pairwise convex combinations of the top candidates are examined
   (21 mixing ratios each) – a cheap way to explore a richer family.
3. Fast sub‑gradient descent on the exact max‑convolution (the same
   sub‑gradient used by the reference solution) with Nesterov momentum.
4. A softened‑max (log‑sum‑exp) objective is optimised with Adam.
   The soft‑max uses a temperature schedule (beta) that starts low
   (smooth) and ramps up, so the method first explores globally and
   then focuses on the true maximum.
5. Occasional random mass‑transfer moves, Dirichlet restarts and
   jitter‑restarts keep the search from stagnating.
6. In the final few seconds a fine‑grained Gaussian perturbation is
   applied to the best vector found.

All updates are projected onto the probability simplex (∑a_i = 1, a_i≥0),
so the evaluation function is scale‑invariant.  The routine respects the
wall‑time budget (default 1000 s) and returns the best sequence discovered.
"""
import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using FFT."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale invariant) for a on the simplex."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_self(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    return a / a.sum()


def analytic_cosine(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    return a / a.sum()


def analytic_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta_
    return a / a.sum()

def _add_candidate(cand: np.ndarray,
                   best_list: list[tuple[float, np.ndarray]],
                   max_len: int = 10) -> None:
    """Insert a candidate into the bounded best‑list (sorted by score)."""
    sc = compute_score(cand)
    # keep list sorted increasing (better = lower)
    inserted = False
    for idx, (s, _) in enumerate(best_list):
        if sc < s:
            best_list.insert(idx, (sc, cand.copy()))
            inserted = True
            break
    if not inserted and len(best_list) < max_len:
        best_list.append((sc, cand.copy()))
    # trim
    while len(best_list) > max_len:
        best_list.pop()


def analytic_grid_search(n: int) -> tuple[np.ndarray, list[tuple[float, np.ndarray]]]:
    """
    Exhaustively scan a set of families; return the overall best vector
    and a list of the top‑10 candidates (score, vector).
    """
    top = []   # (score, vector) sorted by score

    # Product family – p in [0.1,3] step 0.15, offset in [1e‑3,5] step 0.3
    p_vals = np.arange(0.1, 3.01, 0.15)
    off_vals = np.concatenate(
        (np.geomspace(1e-3, 0.01, 5), np.arange(0.01, 5.01, 0.3))
    )
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            _add_candidate(cand, top)

    # Sum family – same ranges
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            _add_candidate(cand, top)

    # Exponential tails – lam from 1e‑3 to 10 (logspace)
    lam_vals = np.logspace(-3, 1, num=20)
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        _add_candidate(cand, top)

    # Cosine‑power – p from 0.5 to 8 (step 0.5)
    p_cos = np.arange(0.5, 8.1, 0.5)
    for p in p_cos:
        cand = analytic_cosine(n, p)
        _add_candidate(cand, top)

    # Beta family – α,β from -0.5 to 2.5 step 0.5
    alpha_vals = np.arange(-0.5, 2.51, 0.5)
    beta_vals = np.arange(-0.5, 2.51, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            _add_candidate(cand, top)

    # include any global reference sequence, if present
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            _add_candidate(ref, top)

    # best of the top‑10
    best_seq = min(top, key=lambda x: x[0])[1].copy()
    return best_seq, top


def mix_top_candidates(top: list[tuple[float, np.ndarray]],
                       n: int) -> np.ndarray:
    """
    Try convex combinations of the best few candidates.
    Returns the best mixture found (or the original best if none improve).
    """
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    alphas = np.linspace(0.0, 1.0, 21)  # 0,0.05,…,1
    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            a_i = top[i][1]

            for alpha in alphas:

                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    return best_seq

        a_candidate = project_simplex(a_candidate)

        # occasional random move (mass transfer)
        if np.random.rand() < 0.02:
            a_candidate = random_move(a_candidate, step * 0.5)

        # occasional Dirichlet restart (rare)
        if np.random.rand() < 0.003:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            rand_score = compute_score(rand_vec)
            if rand_score < best_score:
                best_score = rand_score
                best_seq = rand_vec.copy()
                a_candidate = rand_vec.copy()
                print(f"[sg {iter_cnt:07d}] random restart improved to {best_score:.6f}")

        # jitter restart on stagnation
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[sg {iter_cnt:07d}] jitter restart")

        a = a_candidate
        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3) Soft‑max (log‑sum‑exp) optimisation with Adam
    # ------------------------------------------------------------------
    # Parameters for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_adam = 0
    lr = 0.12                     # base learning rate for Adam
    beta_start, beta_end = 0.5, 400.0   # temperature schedule (higher → sharper)

    # keep another copy for jitter handling
    last_improve = time.time()
    stagnation_time = 20.0

    while time.time() < deadline - 10:   # keep ~10 s for fine refinement
        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] new best = {best_score:.6f}")

        # soft‑max weights over all shift indices
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20.0)
        beta = beta_start * (1.0 - progress) + beta_end * progress
        w_raw = np.exp(beta * (conv - max_val))    # max exponent = 0
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # weighted shift = Σ_k w_k * a_{k-i}  (correlation w * a)
        a_rev = a[::-1]
        corr_full = conv_two(w, a_rev)               # length = 3n‑2
        weighted_shift = corr_full[n - 1:n - 1 + n]   # length n

        # gradient of soft‑max objective: a + weighted_shift
        g = a + weighted_shift

        # Adam update
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step_vec
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = project_simplex(a_candidate)

        # occasional jitter / random move to escape shallow plateaus
        if np.random.rand() < 0.015:
            a_candidate = random_move(a_candidate, lr * 0.3)
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 5e-4
            a_candidate = a_candidate + jitter
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_simplex(a_candidate)

        # stagnation handling
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a_candidate = best_seq + jitter
            a_candidate = project_simplex(a_candidate)
            last_improve = time.time()
            print(f"[adam {t_adam:07d}] jitter restart")

        a = a_candidate

    # ------------------------------------------------------------------
    # 4) Fine‑grained random refinement (last few seconds)
    # ------------------------------------------------------------------
    print("[refine] starting final random perturbation phase")
    while time.time() < deadline:
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, length = {n}, time = {total_time:.1f}s")
    return best_seq.tolist()
