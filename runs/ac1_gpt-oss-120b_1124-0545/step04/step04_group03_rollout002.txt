def conv_fft_self(a):
    n = a.shape[0]
    size = _next_pow2(2*n - 1)
    ft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    return conv[:2*n-1]

def compute_score(a):
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a, scale):
    n = a.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n-1)
    if j >= i:
        j += 1
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # ensure nonnegative, then project onto simplex
    a_new = project_simplex(a_new)
    return a_new

"""
Propose a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Method
------
1.  Choose a relatively large length n (default 10000) – the objective
    approaches its asymptotic optimum when n grows, while the cost of an
    FFT based convolution stays modest.

2.  Perform a dense grid search over a two‑parameter symmetric power‑law
    family  

        a_i ∝ ((i+off)·(n−1−i+off))^(−p)          (product mode)

    and over the analogous “sum” family  

        a_i ∝ (i+off)^(−p) + (n−1−i+off)^(−p)

    The parameters p∈[0.1,1.2] and off∈[0,5] are explored on a moderate
    grid.  Every candidate is normalised to the probability simplex and
    its exact score 2·n·max(conv) is evaluated (the sum‑normalisation
    cancels out in the original objective).

3.  Starting from the best grid point, a cheap stochastic refinement on
    the continuous parameters (p,off) is carried out – a few hundred
    small random perturbations are tried, each time keeping an improvement.

4.  Finally a light **random‑mass‑transfer** local search is applied.
    Starting from the best distribution, a tiny amount of probability
    mass is moved from a random index to another, the new vector is
    projected back onto the simplex and accepted only if the true
    objective improves.  The step size is slowly adapted.

The whole routine respects the given time budget (default 1000 s) and
prints progress.  It returns the best sequence found as a plain Python
list of floats.
"""

import time
import numpy as np

# Low‑level utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)−1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector a (sum(a)=1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_fft_self(a)
    return float(2.0 * n * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of probability mass from one index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new


def analytic_product(p: float, offset: float, n: int) -> np.ndarray:
    """
    Symmetric product power‑law candidate:
        a_i ∝ ((i+off)*(n−1−i+off))^(−p)
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / s
    return a


def analytic_sum(p: float, offset: float, n: int) -> np.ndarray:
    """
    Symmetric sum power‑law candidate:
        a_i ∝ (i+off)^(−p) + (n−1−i+off)^(−p)
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s == 0.0:

    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 10000))
    n = max(500, min(20000, n))  # keep n in a reasonable range

    best_score = np.inf
    best_a = None
    best_mode = None
    best_params = None

    # ------------------------------------------------------------------
    # 0b) Optionally seed from a previously supplied construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                ref = ref / s
                sc = compute_score(ref)
                if sc < best_score:
                    best_score = sc
                    best_a = ref.copy()
                    best_mode = "reference"
                    print(f"[init] loaded height_sequence_1 with score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Coarse grid search over (p, offset) for both families
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.1, 1.2, 35)          # 35 values
    off_vals = np.linspace(0.0, 5.0, 31)        # 31 values

    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                if time.time() >= deadline:
                    break
                # avoid exact zero offset for product mode (division by zero)
                offset = off if (mode == "sum" or off > 0.0) else 1e-6
                if mode == "product":
                    a = analytic_product(p, offset, n)
                else:
                    a = analytic_sum(p, offset, n)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    best_mode = mode
                    best_params = (p, offset)
                    elapsed = time.time() - start_time
                    print(f"[grid] mode={mode} p={p:.4f} off={offset:.4f} "
                          f"score={best_score:.6f} t={elapsed:.1f}s")
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Stochastic refinement of the continuous parameters (p,off)
    # ------------------------------------------------------------------
    if best_params is not None:
        p_cur, off_cur = best_params
        for it in range(500):
            if time.time() >= deadline:
                break
            # random perturbation
            p_new = p_cur + (np.random.rand() * 2.0 - 1.0) * 0.02   # ±0.02
            off_new = off_cur + (np.random.rand() * 2.0 - 1.0) * 0.1  # ±0.1

            p_new = np.clip(p_new, 0.05, 1.5)
            off_new = max(off_new, 0.0)
            if best_mode == "product" and off_new == 0.0:
                off_new = 1e-6

            if best_mode == "product":
                a_new = analytic_product(p_new, off_new, n)
            else:
                a_new = analytic_sum(p_new, off_new, n)

            sc_new = compute_score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                best_params = (p_new, off_new)
                p_cur, off_cur = best_params
                elapsed = time.time() - start_time
                print(f"[refine] mode={best_mode} p={p_cur:.4f} off={off_cur:.4f} "
                      f"score={best_score:.6f} t={elapsed:.1f}s")
            # otherwise keep current parameters

    # ------------------------------------------------------------------
    # 3) Light random‑mass‑transfer local search
    # ------------------------------------------------------------------
    if best_a is not None:
        a = best_a.copy()
        scale = 0.02  # initial relative step size
        while time.time() < deadline:
            cand = random_move(a, scale)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                elapsed = time.time() - start_time
                print(f"[local] new best {best_score:.6f} t={elapsed:.1f}s")
                scale = max(0.001, scale * 0.9)   # tighten step size
            else:
                # slowly increase step size again to keep exploration alive
                scale = min(0.05, scale * 1.001)
                a = best_a.copy()

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, n = {n}, time = {total_time:.1f}s")

    # Return a plain Python list of floats (no numpy scalars)
    if best_a is None:
        return [0.0]
    return [float(v) for v in best_a.tolist()]
