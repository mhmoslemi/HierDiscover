def spectral_flatten(a, iters):
    n = len(a)
    size = _next_pow2(2*n - 1)
    for t in range(iters):
        # compute DFT of zero-padded a
        A = np.fft.rfft(a, n=size)
        # compute magnitude
        mag = np.abs(A)
        # average magnitude (excluding zero freq? include)
        avg = mag.mean()
        # replace magnitude with avg, keep phase
        A = avg * np.exp(1j * np.angle(A))
        # inverse transform
        a_new = np.fft.irfft(A, n=size)[:n]
        # clip and renorm
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s == 0:
            a_new = np.ones_like(a) / n
        else:
            a_new /= s
        a = a_new
    return a
```

def _run_gradient_descent(a0, n, size_fft, size_fft2, deadline):
    a = a0.copy()
    a_rev = a[::-1]
    v = np.zeros_like(a)
    base_step = 0.8
    momentum = 0.9
    tau_factor = 0.6
    tau_decay = 0.9995
    best_score = np.inf
    best_a = a.copy()
    it = 0
    while time.time() < deadline - 0.2 and it < 5000:
        conv = _full_convolution(a, size_fft)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b   # sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute softmax weights
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay
        step = base_step / np.sqrt(it + 1.0)
        w = np.exp((conv - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        g_full = _conv_fft(a_rev, w, size_fft2)
        g = g_full[n-1:n-1+n]
        d = -2.0 * g
        d -= d.mean()
        v = momentum * v + (1.0 - momentum) * d
        a = a + step * v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a /= s
        a_rev = a[::-1].copy()
        it += 1
    return best_a, best_score
```

"""
A fast multi‑restart optimiser for the step‑function bound.

The objective to minimise is

    F(a) = 2·n·max_k (a * a)_k     (sum(a) = 1, a_i ≥ 0)

where “*” is the linear convolution.  The algorithm works as follows:

1.  **Initialisation** – several cheap ways to obtain a feasible a:
        * Cantor histogram (the previous strong start)
        * Dirichlet draw (random point on the probability simplex)
        * Random phase signal – a real sequence whose DFT has a
          constant magnitude and a random phase, then projected onto the
          simplex.
        * Re‑use the previously best global sequence if it exists.

2.  **Spectral flattening** – a few iterations of
    “fix the magnitude of the DFT to its mean value”.  This directly
    pushes the convolution of a with itself towards a flat profile
    (the DFT of a*a is the square of the DFT of a).

3.  **Smooth‑max gradient descent** – minimise a smooth approximation
    of the max (soft‑max with a temperature τ) using Nesterov momentum.
    The gradient is obtained with a single FFT‑based convolution.
    τ is exponentially cooled and the step size decays as 1/√t.

4.  **Hard‑max refinement** – after the smooth phase we switch to the
    exact sub‑gradient (w = indicator of the maximal positions) for a
    short final polishing.

5.  **Multi‑restart** – the whole pipeline is executed repeatedly
    (different random seeds, different initialisers) while time permits.
    The best sequence seen (according to the fast FFT based evaluation)
    is kept and finally returned.

All operations are O(n log n) per iteration; with n≈6000 a few ×10⁴
iterations comfortably fit inside the 1000 s budget.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (full length).  ``size`` must be ≥ len(u)+len(v)-1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a * a (full length)."""
    return _conv_fft(a, a, size)


# ----------------------------------------------------------------------
# Cantor histogram (the previous strong start)
# ----------------------------------------------------------------------


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Length‑n histogram approximating the (1/3)‑Cantor measure.
    Returns a non‑negative vector that already sums to 1.
    """
    N = 1 << L
    powers = 3.0 ** np.arange(1, L + 1)
    idx = np.arange(N, dtype=np.uint32)[:, None]
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)
    pos = (bits * (2.0 / powers)).sum(axis=1)
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    return hist.astype(np.float64) / float(N)


# ----------------------------------------------------------------------
# Spectral flattening (equalise magnitude of the DFT)
# ----------------------------------------------------------------------


def _spectral_flatten(a: np.ndarray, size_fft: int, iters: int = 5) -> np.ndarray:
    """
    Replace the magnitude of the FFT of ``a`` by its mean magnitude,
    keep the phase, then project back onto the probability simplex.
    """
    for _ in range(iters):
        A = np.fft.rfft(a, n=size_fft)
        mag = np.abs(A).mean()
        A = mag * np.exp(1j * np.angle(A))
        a = np.fft.irfft(A, n=size_fft)[: a.shape[0]]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = np.ones_like(a) / a.shape[0] if s == 0.0 else a / s
    return a


# ----------------------------------------------------------------------
# Random phase initialiser
# ----------------------------------------------------------------------


def _random_phase_initialiser(n: int, size_fft: int) -> np.ndarray:
    """
    Produce a real vector whose DFT has unit magnitude and completely
    random phase.  Afterwards project onto the simplex.
    """
    # constant magnitude (the mean magnitude of a unit‑mass Dirichlet vector)
    mag = 1.0
    angles = np.random.uniform(0.0, 2.0 * np.pi, size=size_fft // 2 + 1)
    # enforce Hermitian symmetry implicitly satisfied by rfft/irfft
    A = mag * np.exp(1j * angles)
    a = np.fft.irfft(A, n=size_fft)[:n]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = np.ones_like(a) / n if s == 0.0 else a / s
    return a


# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but using FFT)
# ----------------------------------------------------------------------


def _fast_score(a: np.ndarray, n: int, size_conv: int) -> float:
    """
    Returns 2·n·max(a*a) assuming sum(a) == 1.
    Uses an FFT based convolution to obtain the full linear convolution.
    """
    conv = _full_convolution(a, size_conv)
    max_b = conv.max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(
    a0: np.ndarray,
    n: int,
    size_conv: int,
    size_grad: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform the soft‑max based gradient descent described in the
    reference implementation.  Returns the best vector seen and its
    fast score.
    """
    a = a0.copy()
    a_rev = a[::-1].copy()
    v = np.zeros_like(a)

    # hyper‑parameters – tuned for a fast convergence
    base_step = 0.9
    momentum = 0.9
    tau_factor = 0.6          # τ = tau_factor * max_b
    tau_decay = 0.9995

    best_a = a.copy()
    best_score = _fast_score(a, n, size_conv)
    it = 0

    while time.time() < deadline - 0.2 and it < 8000:
        conv = _full_convolution(a, size_conv)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature for the soft‑max
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # step size decays like 1/√t
        step = base_step / np.sqrt(it + 1.0)

        # soft‑max weighting of the convolution entries
        w = np.exp((conv - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient = -2·(a_rev ∗ w)
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n - 1 : n - 1 + n]
        d = -2.0 * g
        d -= d.mean()                     # keep Σd = 0  (mass preservation)

        # Nesterov‑type momentum
        v = momentum * v + (1.0 - momentum) * d
        a = a + step * v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = np.ones_like(a) / n if s == 0.0 else a / s
        a_rev = a[::-1].copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard‑max (exact sub‑gradient) polishing
# ----------------------------------------------------------------------


def _hard_max_refine(
    a0: np.ndarray,
    n: int,
    size_conv: int,
    size_grad: int,
    deadline: float,
) -> np.ndarray:
    """
    Short polishing phase using the exact sub‑gradient of the max:
    the weight vector is the indicator of the maximal convolution entries.
    """
    a = a0.copy()
    a_rev = a[::-1].copy()
    v = np.zeros_like(a)

    base_step = 0.3
    momentum = 0.9
    it = 0

    while time.time() < deadline - 0.1 and it < 4000:
        conv = _full_convolution(a, size_conv)
        max_b = conv.max()
        # indicator of the maximal positions (there may be several)
        w = (conv == max_b).astype(float)
        w_sum = w.sum()
        w /= w_sum

        # gradient
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n - 1 : n - 1 + n]
        d = -2.0 * g
        d -= d.mean()

        step = base_step / np.sqrt(it + 1.0)
        v = momentum * v + (1.0 - momentum) * d
        a = a + step * v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = np.ones_like(a) / n if s == 0.0 else a / s
        a_rev = a[::-1].copy()
        it += 1

    return a


# ----------------------------------------------------------------------
# Core optimisation driver (multiple restarts)
# ----------------------------------------------------------------------


def _optimise_one_restart(
    n: int,
    size_conv: int,
    size_grad: int,
    size_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Execute a single optimisation run: initialise, spectral flatten,
    smooth‑max descent, and a final hard‑max polish.
    Returns the best sequence found and its score.
    """
    # ----- initialise -------------------------------------------------
    method = np.random.choice(["cantor", "dirichlet", "random_phase"])
    if method == "cantor":
        a = _cantor_histogram(n, L=14)
    elif method == "dirichlet":
        alpha = np.random.uniform(0.3, 4.0)      # shape of the Dirichlet
        a = np.random.gamma(alpha, 1.0, size=n)
        a = a / a.sum()
    else:   # random_phase
        a = _random_phase_initialiser(n, size_fft)

    # small random jitter to avoid pathological symmetries
    a += np.random.normal(0.0, 0.003, size=n)
    a = np.clip(a, 0.0, None)
    a = np.ones_like(a) / n if a.sum() == 0.0 else a / a.sum()

    # ----- spectral equalisation (a few cheap iterations) ----------
    a = _spectral_flatten(a, size_fft, iters=5)

    # ----- smooth‑max gradient descent -------------------------------
    a_opt, score_opt = _smooth_max_descent(a, n, size_conv, size_grad, deadline)

    # ----- final hard‑max polishing ---------------------------------
    a_final = _hard_max_refine(a_opt, n, size_conv, size_grad, deadline)

    final_score = _fast_score(a_final, n, size_conv)
    if final_score < score_opt:
        return a_final, final_score
    else:
        return a_opt, score_opt


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # Pre‑compute the FFT sizes that stay constant throughout the run
    size_conv = _next_pow2(2 * n - 1)       # for a*a
    size_grad = _next_pow2(3 * n - 2)       # for a_rev * weight
    size_fft = _next_pow2(2 * n - 1)        # used by the spectral steps

    best_a = None
    best_score = np.inf
    iteration = 0

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        iteration += 1
        a_candidate, sc = _optimise_one_restart(
            n, size_conv, size_grad, size_fft, deadline
        )
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            elapsed = time.time() - start
            print(
                f"[restart {iteration:3d}] new global best = {best_score:.6f}  time={elapsed:.1f}s"
            )
        # tiny safety check – do not start a new restart if not enough time left
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded, use the Cantor construction
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = _cantor_histogram(n, L=14)
        best_score = _fast_score(best_a, n, size_conv)

    # ------------------------------------------------------------------
    # Final sanitisation (non‑negative, normalised)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    s = best_a.sum()
    best_a = np.ones_like(best_a) / n if s == 0.0 else best_a / s

    total_time = time.time() - start
    print(f"[done] best score≈{best_score:.6f}, time={total_time:.1f}s, length={n}")
    return best_a.tolist()
