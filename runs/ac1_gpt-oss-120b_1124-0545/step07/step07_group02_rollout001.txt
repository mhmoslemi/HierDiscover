def _adam_optimize(start_seq, n, deadline, max_iters=None, gamma_initial=5.0, gamma_max=100.0, lr_initial=0.2):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    gamma = gamma_initial
    # schedule for gamma and lr
    gamma_increase_iter = 50000  # increase gamma every 50k it
    # compute total number of it from max_iters or break by time
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        # compute convolution
        conv = _conv_fft(a, a)
        # compute softmax weights with current gamma
        # scale conv for stability
        scaled = conv * gamma
        max_scaled = scaled.max()
        # subtract max to avoid overflow
        exp_vals = np.exp(scaled - max_scaled)
        w = exp_vals / exp_vals.sum()
        # gradient
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]  # shape n

        # Adam update on log_a
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule
        lr = lr_initial / (1 + it / 200000.0)  # decays slowly
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # normalize a
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0:
            a = _uniform(n)
        else:
            a /= sum_a

        # evaluate exact objective occasionally (e.g., every 1000 it)
        if it % 1000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        # gamma schedule: increase slowly

"""
Hybrid optimizer with multi‑resolution Adam + simulated annealing.

The algorithm works in three stages:

1. **Coarse search (n≈2000).**  
   A rich seed pool (classical windows, β‑pdf, geometric families, random
   soft‑max vectors) is built.  The best seed is refined by a
   mirror‑descent variant of Adam that minimizes a *soft‑max* approximation
   of the L∞‑norm of the convolution.  The smoothing parameter γ is
   increased gradually (continuation) which helps escaping poor local minima.

2. **Upsampling to a larger dimension (n≈4000 or 8000).**  
   The coarse solution is interpolated to a finer grid, renormalised and
   refined again by the same Adam‑based optimizer (with larger γ and a
   slightly smaller learning‑rate).

3. **Final polishing.**  
   A simulated‑annealing walk with large mass‑transfer moves finishes the
   search, using all remaining time.

All iterates stay on the probability simplex (non‑negative, sum = 1) by
working in log‑space (exponentiated‑gradient) and normalising after each
update.  The best sequence seen during the run is stored in the global
``height_sequence_1`` variable for possible continuation in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families that return probability vectors (sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – α ∈ [0,1]."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] produces edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of normalised start vectors."""
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic parametric families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic window shapes
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if shape matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0.0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Interpolation / up‑sampling of a sequence
# ----------------------------------------------------------------------


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.clip(new_seq, 0.0, None)
    total = new_seq.sum()
    if total > 0.0:
        new_seq /= total
    else:
        new_seq = _uniform(new_n)
    return new_seq


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def _adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
    gamma_initial: float = 5.0,
    gamma_max: float = 80.0,
    lr_initial: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using Adam on the log‑variables (exponentiated‑gradient).
    The loss is 2·n·softmax_γ(conv) where softmax_γ approximates the L∞‑norm.
    """
    a = start_seq.copy()
    # keep a on the simplex throughout – work in log‑space
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    gamma = gamma_initial
    gamma_step = 50000          # increase γ every this many iterations
    # simple decay of learning rate
    lr = lr_initial

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1

        # --- convolution of the current iterate ---
        conv = _conv_fft(a, a)               # length = 2n-1

        # --- soft‑max weights (smooth approximation of the max) ---
        scaled = conv * gamma
        max_scaled = scaled.max()
        exp_vals = np.exp(scaled - max_scaled)   # stabilise
        w = exp_vals / exp_vals.sum()            # length 2n‑1

        # --- gradient of the smoothed objective ---
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        g = 2.0 * full[n - 1 : n - 1 + n]        # size n

        # --- Adam update in log‑space ---
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr_initial / (1.0 + it / 200000.0)  # very gentle decay
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- project back to the simplex ---
        a = np.exp(log_a)
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = _uniform(n)
        else:
            a /= sum_a

        # --- occasional exact evaluation ---
        if it % 1000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] n={n:4d} it={it:7d} best={cur_val:.6f}")

        # --- continuation on γ ---
        if it % gamma_step == 0 and gamma < gamma_max:
            gamma = min(gamma * 1.5, gamma_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (same as in the reference code)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with occasional global jumps.
    All moves keep the sequence on the simplex.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.06)   # initial temperature
    decay = 0.9992

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move a random fraction (up to 10 %) from i to j
        delta = rng.random() * 0.10 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] n={n:4d} it={it:7d} best={new_val:.6f}")

        # global perturbation every 800 steps
        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= total
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] n={n:4d} it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Stage 1 – coarse optimisation (n≈2000)
    # --------------------------------------------------------------
    n_small = 2000

    # try to obtain a down‑sampled version of a previous result if available
    prior = globals().get("height_sequence_1")
    prior_small = None
    if prior is not None:
        try:
            prior_arr = np.asarray(prior, dtype=np.float64)
            if prior_arr.ndim == 1 and prior_arr.shape[0] > 1:
                # simple linear interpolation to the smaller size
                prior_small = _interpolate_seq(prior_arr, n_small)
        except Exception:
            prior_small = None

    seeds_small = _make_seeds(n_small, prior_small)
    seed_vals_small = [(s, _objective(s)) for s in seeds_small]
    seed_vals_small.sort(key=lambda x: x[1])
    best_seq_small = seed_vals_small[0][0].copy()
    best_val_small = seed_vals_small[0][1]
    print(f"[Stage1 seed] n={n_small} best_val={best_val_small:.6f}")

    # allocate roughly 30 % of the remaining budget (but at least 120 s)
    now = time.time()
    stage1_deadline = now + min(0.3 * (deadline - now), max(120.0, (deadline - now) * 0.3))
    best_seq_small, best_val_small = _adam_optimize(
        best_seq_small,
        n_small,
        stage1_deadline,
        max_iters=None,
        gamma_initial=5.0,
        gamma_max=80.0,
        lr_initial=0.2,
    )
    print(f"[Stage1 Adam] n={n_small} best_val={best_val_small:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – upsample to a larger dimension and refine
    # --------------------------------------------------------------
    # Decide the larger size based on remaining time
    now = time.time()
    remaining = deadline - now
    n_large = 4000 if remaining < 250.0 else 8000

    best_seq_large = _interpolate_seq(best_seq_small, n_large)
    best_seq_large = best_seq_large / best_seq_large.sum()
    best_val_large = _objective(best_seq_large)
    print(f"[Stage2 upsample] n={n_large} init_val={best_val_large:.6f}")

    # allocate the bulk of the remaining time, leaving ~80 s for final anneal
    now = time.time()
    stage2_deadline = now + max(0.0, deadline - now - 80.0)

    best_seq_large, best_val_large = _adam_optimize(
        best_seq_large,
        n_large,
        stage2_deadline,
        max_iters=None,
        gamma_initial=10.0,
        gamma_max=120.0,
        lr_initial=0.15,
    )
    print(f"[Stage2 Adam] n={n_large} best_val={best_val_large:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final simulated‑annealing polishing
    # --------------------------------------------------------------
    now = time.time()
    final_seq = best_seq_large
    final_val = best_val_large
    if deadline - now > 5.0:
        final_seq, final_val = _anneal_optimize(final_seq, n_large, deadline, max_iters=None)
        print(f"[Final anneal] n={n_large} best_val={final_val:.6f}")

    # fallback in the unlikely event that nothing was produced
    if final_seq is None:
        final_seq = _uniform(4000)
        final_val = _objective(final_seq)

    # store for possible subsequent runs
    globals()["height_sequence_1"] = final_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {final_val:.6f} (length {len(final_seq)})")
    return final_seq.tolist()
