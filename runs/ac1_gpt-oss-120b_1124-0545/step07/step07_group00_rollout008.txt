"""
Small‑scale evolutionary optimizer for the step‑function bound.

The algorithm works for a single, relatively large length (default 8000)
and consists of three phases:

1. **Seed generation** – a diverse set of analytically motivated
   families (uniform, sine, Gaussian, Tukey, Dirichlet etc.) is built
   and normalised to ℓ₁‑norm = 1.  The best seed (by the surrogate
   objective 2·n·max(conv)) is kept.

2. **Heavy optimisation** – the chosen seed is refined by a long
   Adam run on the smooth log‑sum‑exp surrogate of the max‑convolution,
   followed by an exact sub‑gradient mirror‑descent (exponentiated‑gradient)
   that works on the true ℓ∞‑objective.  Both loops are stopped
   automatically when the time budget is exhausted.

3. **Fine‑grained stochastic hill‑climbing** – simulated annealing
   repeatedly reshuffles a small random subset of coordinates.
   It can escape shallow minima and often yields a tiny final improvement.

All stages respect a global timeout (default 1000 s).  The function
returns the best sequence found as a plain list of floats, ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast, O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) – assumes ∑seq==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic shape families (already normalised to ℓ₁‑norm = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle, sigma = sigma_frac * n."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window, α∈[0,1]."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                     (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    s = new.sum()
    return new / s if s > 0 else _uniform(new_n)


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Build a diverse set of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic structural families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    for sigma in (0.2, 0.3, 0.4, 0.5):
        seeds.append(_gaussian(n, sigma))
    for a in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, a))

    # random draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns – a handful of points
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (rescaled if length differs)
    if previous is not None:
        seeds.append(_interpolate(previous, n))

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on the log‑sum‑exp (smooth max) surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits (stay on simplex via softmax)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 1.0               # temperature for log‑sum‑exp
    beta_mul = 1.002
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                         # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # raise temperature slowly – it approaches the true max
        beta = min(beta * beta_mul, beta_max)

        # smooth max via log‑sum‑exp on convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # all indices attaining the maximum (might be several)
        max_inds = np.where(np.abs(conv - max_val) < 1e-12)[0]

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # build sub‑gradient: sum of contributions from every active index
        g = np.zeros_like(a)
        for k in max_inds:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx
            g[idx] += 2.0 * a[rev_idx]       # factor 2 from symmetry

        # exponentiated‑gradient step
        lr = 0.8 / np.sqrt(it)                # a slightly larger step
        log_a = np.log(a + eps) - lr * g
        # stabilise
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # occasional “jitter” to avoid getting stuck
        if it % 10000 == 0:
            jitter = 0.001
            noise = rng.dirichlet(np.full(n, 1.0))
            a = (1 - jitter) * a + jitter * noise
            a /= a.sum()

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing that redistributes the mass of
    a random small subset of coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # swap between 2 … 6 positions
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
