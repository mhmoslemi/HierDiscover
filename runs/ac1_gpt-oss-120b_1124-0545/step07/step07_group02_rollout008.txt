"""
Hybrid optimiser with a two‑stage resolution (small → large) and an
expanded seed pool.

The algorithm works as follows:

1. **Small resolution (n≈2000).**  A rich collection of candidate
   vectors is built from classical windows, power / geometric families,
   beta‑pdfs (including very edge‑heavy α∈[0.1,0.9]), random Dirichlet
   draws, etc.  The best few seeds are refined by projected
   exponentiated‑gradient sub‑gradient descent (≈few 10⁵ iterations – fast
   because the vectors are short).

2. **Upscaling.**  The best small‑resolution vector is linearly
   interpolated to a larger length (n≈4000) and re‑normalised.  The
   upscaled shape inherits the low‑maximum‑convolution property of the
   small optimum.

3. **Large‑resolution refinement.**  Starting from the upscaled vector
   the same mirror‑descent sub‑gradient method runs for the remaining
   time budget, now on the full “real” size.

4. **Final simulated‑annealing walk.**  A Metropolis–Hastings walk
   with aggressive mass‑transfer moves (up to 30 % of the source mass)
   finishes the search, using every remaining second.

All vectors live on the probability simplex (non‑negative, sum = 1) –
the evaluation routine `evaluate_sequence` therefore sees exactly the
same objective value (its clipping limits never kick in).  The routine
returns the best sequence found before the timeout; if the budget is
exhausted without a result the grader will assign “‑inf” points.

The code respects the 1000 s budget, prints occasional progress
messages and stores the final sequence in the global name
`height_sequence_1` for possible chaining.
"""

# Low‑level convolution utilities (FFT based, real‑valued)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)²)
    The vector is assumed non‑negative and already normalised (sum = 1).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families – each returns a probability vector on the simplex
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_power(n: int, p: float) -> np.ndarray:
    """U‑shaped power: a_i ∝ (i+1)^‑p + (n‑i)^‑p."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** -p) + ((n - i + 1) ** -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if not SCIPY_AVAILABLE or beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α∈(0,∞).  Small α produces heavy edges.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration (α)."""
    if concentration <= 0.0:
        return _uniform(n)
    sample = rng.gamma(concentration, 1.0, size=n)
    s = sample.sum()
    return sample / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    e = np.exp(logits - logits.max())
    return e / e.sum()


def _make_seeds(n: int, prior: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse start pool (all vectors are normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    # geometric families – explore a fairly wide range
    for r in np.linspace(0.90, 0.9999, 12):
        if r > 0.0:
            seeds.append(_geom(n, r))
            seeds.append(_sym_geom(n, r))

    # power families
    for p in np.linspace(0.0, 5.0, 13):
        seeds.append(_power(n, p))
        seeds.append(_u_power(n, p))

    # mixed geometric‑power
    for r in np.linspace(0.95, 0.999, 5):
        for p in np.linspace(0.0, 2.0, 5):
            seeds.append(_geom_power(n, r, p))

    # classic windows (always useful)
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))
    if SCIPY_AVAILABLE:
        for beta in [2.0, 4.0, 8.0, 12.0, 20.0]:
            seeds.append(_kaiser_window(n, beta))

    # Gaussian sweep
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta‑pdf sweep – include very edge‑heavy α
    for alpha in np.linspace(0.10, 0.95, 18):
        seeds.append(_beta_window(n, alpha))

    # Dirichlet draws (different concentrations)
    for conc in [0.1, 0.3, 0.7, 1.5, 3.0, 7.0]:
        seeds.append(_random_dirichlet(n, conc, rng))

    # random soft‑max vectors
    for _ in range(10):
        seeds.append(_random_softmax(n, rng))

    # reuse a previous construction if supplied
    if prior is not None and prior.shape == (n,):
        seeds.append(prior / prior.sum())

    # clean‑up – clip negatives and renormalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact objective.
    The input vector must lie on the simplex.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the convolution is (almost) maximal
        active = np.where(conv >= max_val - rel_tol)[0]

        # build the sub‑gradient (average over the active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:4d} it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (more aggressive moves)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with larger mass‑transfer moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature – a small fraction of the current value
    T0 = max(1e-8, cur_val * 0.06)
    decay = 0.9993
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        # transfer a random fraction (up to 30 %) of the source mass
        delta = rng.random() * 0.30 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation (every 1000 steps)
        if it % 1000 == 0:
            noise = rng.normal(scale=0.002, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(cur.size)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal (jump)] it={it:7d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Upscaling utilities
# ----------------------------------------------------------------------


def _upscale(seq_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from len(seq_small) → n_large,
    followed by renormalisation to the simplex.
    """
    x_small = np.linspace(0.0, 1.0, seq_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    interpolated = np.interp(x_large, x_small, seq_small)
    interpolated = np.clip(interpolated, 0.0, None)
    total = interpolated.sum()
    if total <= 0.0:
        return _uniform(n_large)
    return interpolated / total


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately)
    minimises ``evaluate_sequence``.  Returns the best sequence as a list.
    The routine respects the supplied time budget (≈ budget_s seconds).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – small resolution (fast, many iterations)
    # ------------------------------------------------------------------
    n_small = 2000
    print(f"[Stage 0] Using small resolution n={n_small}")

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size != n_small:
                previous = None
        except Exception:
            previous = None

    seeds_small = _make_seeds(n_small, previous)
    # quick evaluation of all seeds
    seed_vals = [(s, _objective(s)) for s in seeds_small]
    seed_vals.sort(key=lambda x: x[1])
    best_seq_small, best_val_small = seed_vals[0]
    print(f"[Stage 0] Best seed objective = {best_val_small:.6f}")

    # refine the best small seed
    if time.time() < deadline - 30:
        best_seq_small, best_val_small = _subgradient_refine(
            best_seq_small, deadline, max_iters=500_000
        )
        print(f"[Stage 0] After sub‑grad refinement best = {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscaling to full size
    # ------------------------------------------------------------------
    n_large = 4000
    print(f"[Stage 1] Upscaling to n={n_large}")
    seq_large = _upscale(best_seq_small, n_large)
    best_val_large = _objective(seq_large)
    print(f"[Stage 1] Upscaled objective = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – large‑resolution refinement (sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        seq_large, best_val_large = _subgradient_refine(
            seq_large, deadline, max_iters=600_000
        )
        print(f"[Stage 2] After sub‑grad refinement best = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_large, best_val_large = _anneal_optimize(seq_large, deadline)
        print(f"[Stage 3] After annealing best = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that we have nothing
    # ------------------------------------------------------------------
    if seq_large is None:
        seq_large = _uniform(n_large)
        best_val_large = _objective(seq_large)
        print("[Fallback] Using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val_large:.6f} (length {len(seq_large)})")

    # store for possible chained runs
    globals()["height_sequence_1"] = seq_large.copy()
    return seq_large.tolist()
