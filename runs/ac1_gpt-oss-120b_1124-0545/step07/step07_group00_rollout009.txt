"""
Propose a step‑function with a low evaluate_sequence() value.

The strategy combines three ideas that complement the previous
minimalist implementation:

1. **Very diverse seeds** – analytic families (uniform, triangular,
   geometric, beta, Gaussian, Tukey, sine‑window) together with a few
   random Dirichlet samples.  All seeds are normalised to ℓ₁‑norm = 1
   and the best few are kept.

2. **Linear‑programming refinement** – for a fixed sequence `f` we
   solve the LP
        maximise   Σ bⱼ
        subject to  (f * b)ₖ ≤ M   (k = 0…2n‑2)
                    bⱼ ≥ 0
   where `M = max(f*f)`.  Because `f` is fixed, the convolution constraints
   are linear in `b`.  The LP (solved with SciPy’s HiGHS interior‑point
   solver) yields a “more spread’’ vector `g` that can be mixed into `f`
   (convex combination) while keeping the maximal convolution ≤ M.
   Repeating a few such mixing steps steadily reduces the objective.

3. **Active‑set exponentiated‑gradient descent** – after the LP phase we
   run a fast mirror‑descent on the true max‑convolution objective.
   The sub‑gradient is formed from *all* active constraints (indices
   whose convolution value is within a tiny tolerance of the current
   maximum).  Updates are performed in log‑space, guaranteeing non‑negativity
   and automatic ℓ₁‑normalisation.

4. **Simulated annealing polishing** – a final stochastic hill‑climber
   reshuffles the mass of a small random subset of coordinates.
   This can occasionally escape shallow minima and yields a tiny extra
   improvement.

All phases respect the global time budget (default 1000 s).  The best
sequence found (as a plain list of floats) is stored in the global
variable ``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np

# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) (seq is ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _uniform(n: int) -> np.ndarray:
    """Uniform ℓ₁‑normalised vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geometric(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geometric(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Build a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.5))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geometric(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geometric(n, r))

    # power‑law
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power_law(n, p))

    # beta‑family
    alphas = np.linspace(0.2, 2.0, 8)
    betas = np.linspace(0.2, 5.0, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.2, 0.35, 0.5, 0.65):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _lp_solve(f: np.ndarray, M: float):
    """
    Solve   maximise   Σ bⱼ
            subject to   (f * b)ₖ ≤ M   (k = 0 … 2n‑2)
                         bⱼ ≥ 0
    Returns the optimal b (ℓ₁‑unnormalised) and a success flag.
    """
    import scipy.optimize
    import scipy.sparse as sp

    n = f.shape[0]
    L = 2 * n - 1

    # Build the constraint matrix A (size L × n) where
    #   A[k, j] = f[k - j]  if 0 ≤ k - j < n  else 0
    # This is a very sparse matrix with exactly n² non‑zero entries.
    # We construct it via vectorised broadcasting.
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
    cols = np.broadcast_to(j, (n, n)).ravel()        # shape (n*n,)
    data = np.tile(f, n)                             # each column gets the whole f

    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    # Linear program: minimise -∑b  (i.e. maximise ∑b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            return res.x, True
        else:
            return None, False
    except Exception as exc:  # pragma: no cover – defensive
        print("LP solve failed:", exc)
        return None, False

def _subgradient_active(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    The sub‑gradient is formed from all active constraints (indices
    whose convolution value is within a relative tolerance of the current
    maximum).  Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    no_improve = 0
    # tolerance for the active set (relative)
    rel_tol = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --- active set -------------------------------------------------
        tol = max(rel_tol * max_val, 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # weighting: uniform over the active indices
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # gradient of Σ wₖ·convₖ w.r.t. a  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]          # central n entries

        # exponentiated‑gradient update
        lr = 1.0 / np.sqrt(it)                      # step size ~ 1/√t
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)                      # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        # early termination if stagnant
        if no_improve > 60_000:
            break

    final_val = _objective(best_seq)
    return best_seq, final_val


# ----------------------------------------------------------------------
# Simple simulated annealing hill‑climber
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Random Dirichlet reshuffles of a few coordinates with exponential cooling."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)               # 2 … 6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (as a plain list of floats) found before the
    time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Carry over the previously discovered sequence, if any.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that balances optimisation speed and quality.
    # ------------------------------------------------------------------
    n = 3000  # a moderate size that still leaves room for improvement

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and retain the best few.
    # ------------------------------------------------------------------
    print(f"\n--- Generating {n}-length seeds ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(s, _objective(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, val in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate objectives:")
    for i, (vec, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP based refinement (a few aggressive mixing steps).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- LP refinement run {idx+1}/{len(best_seeds)} ---")
        f = seed_vec.copy()
        f = f / f.sum()
        cur_max = _conv_fft(f, f).max()
        # at most 5 LP mixing iterations per seed
        for lp_iter in range(5):
            if time.time() > deadline:
                break
            b_opt, ok = _lp_solve(f, cur_max)
            if not ok or b_opt is None:
                print("  LP solve failed – skipping further LP steps.")
                break
            sum_b = b_opt.sum()
            if sum_b < 1e-12:
                break
            g = b_opt / sum_b                       # ℓ₁‑normalised
            t = 0.2                                 # mixing weight
            new_f = (1.0 - t) * f + t * g
            new_f = new_f / new_f.sum()
            new_max = _conv_fft(new_f, new_f).max()
            if new_max < cur_max - 1e-12:
                f = new_f
                cur_max = new_max
                surrogate = 2.0 * n * cur_max
                if surrogate < global_best_val:
                    global_best_val = surrogate
                    global_best_seq = f.copy()
                    print(f"  [LP] iter {lp_iter+1} new global best = {surrogate:.6f}")
            else:
                # mixing did not improve – shrink the step size
                t *= 0.5
                if t < 1e-3:
                    break

        # ------------------------------------------------------------------
        # Stage 2b – active‑set exponentiated‑gradient refinement of this f.
        # ------------------------------------------------------------------
        seq_sub, val_sub = _subgradient_active(f, n, deadline, max_iters=400_000)
        if val_sub < global_best_val:
            global_best_val = val_sub
            global_best_seq = seq_sub.copy()
            print(f"  [SubGrad] improved global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic annealing for final polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (calls the real evaluation function).
    # ------------------------------------------------------------------
    try:
        real_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nReal evaluate_sequence result = {real_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning – evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f}"
        f" (length {len(global_best_seq)})"
    )

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
