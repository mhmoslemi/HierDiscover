import numpy as np
import scipy.optimize
import scipy.linalg

def lp_step(f, M, n):
    # Build Toeplitz matrix A
    col = np.concatenate([f, np.zeros(n-1, dtype=np.float64)])
    row = np.concatenate([[f[0]], np.zeros(n-1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
    # constraints: A @ g <= M * np.ones(2n-1)
    b_ub = M * np.ones(2*n - 1)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return None
    g = g0 / sum_g0   # normalize to sum 1
    return g
```

"""Propose a step function minimizing

        2·n·max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear convolution.  The algorithm proceeds in four
stages:

1. **Seed generation** – a large, diverse collection of analytic and random
   vectors is built and the best (on the simplex) is kept as a warm start.
2. **LP‑mixing refinement** – given the current vector `f`, we solve the
   linear programme

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

   where `M = max(conv(f, f))`.  The optimal `g` is renormalised to sum to one
   and mixed with `f` via `f ← (1‑t)·f + t·g` with a small step size `t`.
   Repeating this a few times provably reduces `M`.
3. **Gradient‑based polishing** – the improved vector is refined with a
   sequence of optimisation primitives:
   * Adam on a smooth‑max surrogate,
   * exact sub‑gradient descent on the true max,
   * an L2‑flattening step that drives the auto‑convolution toward a constant,
   * a targeted stochastic hill‑climb.
   Each phase is time‑aware and respects the global 1000 s budget.
4. **Final verification** – the result is checked with the supplied
   `evaluate_sequence`.  The best vector is stored in the global variable
   `height_sequence_1` for warm‑starting future calls.

All numbers are non‑negative; the sequence is normalised to have unit sum,
which makes the objective scale‑invariant.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a, a)) assuming Σa = 1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)

def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # half‑sine window
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # geometric decay
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # power‑law
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # beta‑like
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(3):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear
    for _ in range(3):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal blobs
    for _ in range(3):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # Warm‑start, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate (log‑softmax parametrisation)."""
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0            # initial smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:          # safety
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimises ½‖a*a – μ·1‖² with exponentiated‑gradient,
    driving the auto‑convolution toward a constant.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Targeted stochastic hill‑climb."""
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def _build_conv_matrix(f: np.ndarray) -> diags:
    """
    Build the (2n‑1) × n Toeplitz matrix T such that T @ g = conv(f, g).
    The matrix is returned as a CSR sparse matrix.
    """
    n = f.shape[0]
    # each diagonal d (0 ≤ d < n) contains the constant value f[d]
    data = np.tile(f[:, None], (1, n))          # shape (n, n), row i = f[i] repeated
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_mix_step(f: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform a single LP‑mixing step.
    Returns (new_f, improved) where `improved` signals whether the objective
    decreased.
    """
    conv_ff = _conv_full(f, f)
    M = conv_ff.max()

    # 1. solve LP maximise Σ g_i  subject to T(f) g ≤ M, g ≥ 0
    A = _build_conv_matrix(f)
    c = -np.ones(n)                       # linprog minimises, so negate
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        print("[LP] linprog failed – aborting LP phase")
        return f, False

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        print("[LP] trivial solution – aborting LP phase")
        return f, False

    # renormalise to the simplex
    g = g0 / sum_g0
    g = np.maximum(g, 0.0)

    # 2. try a few step sizes and keep the first improvement
    for t in (0.5, 0.2, 0.1, 0.05, 0.02):
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()                     # re‑normalise (numeric safety)
        val_new = _objective(f_new, n)
        if val_new < _objective(f, n) - 1e-9:
            print(f"[LP‑mix] t={t:.3f} improved objective to {val_new:.6f}")
            return f_new, True

    # no improvement with any tested step size
    return f, False

    """

    deadline = start_time + budget_s - 2.0      # safety margin

    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------

    best_val = float("inf")

        val = _objective(a, n)

            best_val = val
            best_seq = a.copy()
            print(f"[Seed] new best {val:.6f}")

    # ------------------------------------------------------------------
    # LP‑mixing phase – allocate about 30 % of the remaining budget
    # ------------------------------------------------------------------
    lp_phase_deadline = time.time() + (deadline - time.time()) * 0.30
    lp_iters = 0
    while time.time() < lp_phase_deadline and lp_iters < 12:
        best_seq, improved = _lp_mix_step(best_seq, n, deadline)
        if improved:
            best_val = _objective(best_seq, n)
        else:
            break
        lp_iters += 1
    print(f"LP‑mixing finished after {lp_iters} iterations, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Gradient‑based refinement (same schedule as the previous SOTA)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:

        # Phase 1 – Adam on smooth‑max
        phase_deadline = time.time() + remaining * 0.40
        seq, val = _adam_optimize(best_seq, n, phase_deadline,
                                  max_iters=300_000)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Phase‑Adam] best={val:.6f}")

        # Phase 2 – Exact sub‑gradient descent
        remaining = deadline - time.time()
        phase_deadline = time.time() + remaining * 0.30
        seq, val = _subgradient_optimize(best_seq, n, phase_deadline,
                                         max_iters=300_000)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Phase‑SubGrad] best={val:.6f}")

        # Phase 3 – L2‑flattening descent
        remaining = deadline - time.time()
        phase_deadline = time.time() + remaining * 0.20
        seq, val = _l2_flatten_optimize(best_seq, n, phase_deadline,
                                        max_iters=300_000)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Phase‑L2Flat] best={val:.6f}")

        # Phase 4 – Targeted hill‑climb (use whatever time is left)
        remaining = deadline - time.time()
        if remaining > 5.0:
            phase_deadline = time.time() + remaining - 2.0
            seq, val = _hill_climb_optimize(best_seq, n, phase_deadline,
                                            max_iters=300_000)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
