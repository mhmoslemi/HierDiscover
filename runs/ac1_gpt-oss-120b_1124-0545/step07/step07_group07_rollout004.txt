def _adam_optimize(a0, n, deadline, max_iters=200000):
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta = 0.5
    beta_mul = 1.01
    best_seq = a0.copy()
    best_val = _objective(a0, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a = np.exp(v - np.max(v))
        a /= a.sum()
        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it} best={cur_val:.6f}")
        # smooth-max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient of weighted sum conv = 2(a*w reversed)
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]
        # softmax Jacobian
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val
```

def _subgradient_optimize(a0, n, deadline, max_iters=200000):
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)
    it = 0
    eps_zero = 1e-30
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(a, a)
        max_val = np.max(conv)
        # compute all indices close to max (within tolerance)
        # if multiple indices, average gradient
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        g = np.zeros_like(a)
        # accumulate gradients
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        g /= len(max_idxs)
        # exponentiated gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps_zero) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it} best={cur_val:.6f}")
    return best_seq, best_val

def _hill_climb_anneal(a0, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02
    total_time = deadline - time.time()
    start = time.time()
    # temperature schedule: start temp = 0.01
    T0 = 0.02
    while time.time() < deadline and max_iters > 0:
        it = int(time.time() - start)  # not used
        # compute conv
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))
        # compute contributions
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        val = _objective(new, n)
        # acceptance
        if val < cur_val:
            accept = True
        else:
            # temperature decreasing linearly
            elapsed = time.time() - start
            remaining = deadline - time.time()
            # compute temperature linearly decreasing
            T = T0 * (remaining / total_time)
            prob = np.exp(-(val - cur_val) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[HillAnneal] new best {val:.6f}")
        # adjust step_scale slowly
        if rng.random() < 0.001:
            step_scale = max(step_scale * 0.9, 1e-6)
    return best_seq, best_val
```

"""
Improved step‑function optimizer.

The algorithm works as follows

1. Choose a moderate length `n` (5000) – larger than the previous 4000,
   which gives more degrees of freedom while keeping FFT costs modest.
2. Build a large, diverse seed set (analytic windows, Dirichlet draws,
   random spikes, and the warm‑start from the previous run, interpolated
   to the new length).
3. Keep the best seed (objective = 2·n·max(conv(a,a)) with sum(a)=1).
4. Perform four time‑aware optimisation phases:
   • Adam on a smooth‑max surrogate (slow temperature increase for a
     more accurate gradient near the optimum).
   • Exact sub‑gradient descent that averages the gradients of *all*
     active maximal indices (instead of a single arg‑max).
   • L2‑flattening: minimise the squared distance of the auto‑convolution
     to a constant (drives the convolution towards a flat shape).
   • Simulated‑annealing hill‑climb that moves a tiny amount of mass from
     indices that contribute most to the current peaks towards random
     receivers, accepting occasional uphill moves early on.
5. After the time budget the best sequence is verified with the provided
   `evaluate_sequence` and stored in the global variable
   `height_sequence_1` for warm‑starting future calls.

All phases respect the global 1000 s budget; the code is completely
self‑contained, uses only NumPy and the standard library, and avoids
closures as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_full(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Warm‑start handling
# ----------------------------------------------------------------------


def _adjust_warm_start(warm: np.ndarray, n: int) -> np.ndarray:
    """Resize a previous sequence to length ``n`` by linear interpolation."""
    old_n = warm.shape[0]
    if old_n == n:
        arr = warm.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
        arr = np.interp(x_new, x_old, warm)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s > 0.0:
        arr /= s
    else:
        arr = np.full(n, 1.0 / n, dtype=np.float64)
    return arr

# Seed generation (large, diverse family)
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator, warm_arr: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Classic analytic windows
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    i = np.arange(n, dtype=np.float64)
    half_sine = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(half_sine / half_sine.sum())                              # half‑sine

    # 2. Raised cosine (Hann) window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    seeds.append(hann / hann.sum())

    # 3. Geometric families
    for r in np.linspace(0.2, 0.95, 12):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # 4. Power‑law families
    for p in np.linspace(0.0, 4.0, 11):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # 5. Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 6. Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(6):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # 7. Softmax of Gaussian logits
    for _ in range(8):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # 8. Random piecewise‑linear shapes
    for _ in range(8):
        K = max(8, int(np.sqrt(n) * rng.uniform(0.8, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # 9. Log‑normal (Gaussian bell) shapes
    for _ in range(8):
        mu = rng.uniform(-2.0, 2.0) * n / 10.0
        sigma = rng.uniform(0.5, 2.0) * n / 10.0
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 10. Random “spike” patterns (few equally weighted spikes)
    for K in (8, 12, 16, 20, 24):
        pos = rng.choice(n, size=K, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[pos] = 1.0 / K
        seeds.append(a)

    # 11. Warm‑start (interpolated) if available
    if warm_arr is not None:
        seeds.append(warm_arr)

    return seeds

def _adam_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).  The smooth‑max temperature β
    is increased *slowly* (β ← β·β_mul) to obtain a finer approximation
    of the true max as optimisation proceeds.
    """
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 0.5                # start with a relatively smooth surrogate
    beta_mul = 1.01           # increase slowly for stability

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # a = softmax(v)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]   # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the *true* max, averaging over all active indices."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)
    it = 0
    eps0 = 1e-30

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(a, a)
        max_val = np.max(conv)

        # Find *all* indices that are (practically) maximal
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]

        # Accumulate sub‑gradient contributions
        g = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        g /= len(max_idxs)                     # average over active set

        # Exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps0) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖²  (μ = 1/(2n‑1)) using exponentiated‑gradient.
    This drives the auto‑convolution towards a constant vector.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)               # target constant value of each conv entry

    it = 0
    eps0 = 1e-30

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)      # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]      # gradient of ½‖·‖²

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + eps0) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_anneal(a0: np.ndarray, n: int, deadline: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.
    A tiny amount of mass is moved from a donor (chosen proportionally to its
    contribution to the current maximal convolution entry) to a random receiver.
    Early on uphill moves are accepted with a temperature that linearly decays
    to zero.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)

    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02                     # relative size of each transfer
    total_time = deadline - time.time()
    start = time.time()
    T0 = 0.02                             # initial temperature for annealing

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # Contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_val = _objective(new, n)

        # Acceptance test (annealing)
        if new_val < cur_val:
            accept = True
        else:
            elapsed = time.time() - start
            remaining = max(deadline - time.time(), 0.0)
            T = T0 * (remaining / total_time)   # linear cooling
            prob = np.exp(-(new_val - cur_val) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[HillAnneal] new best {new_val:.6f}")

        # Occasionally shrink the step scale to keep changes fine‑grained
        if iteration % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the given time budget,
    returns the best sequence found (as a plain list of floats) and stores
    it in the global variable ``height_sequence_1`` for future warm‑starts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final verification step
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose sequence length (larger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    n = 5000                         # still fast enough for many FFTs

    # Interpolate warm start to the new length, if available
    warm_arr_n = _adjust_warm_start(warm_arr, n) if warm_arr is not None else None

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr_n)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of seeds – keep the best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Time‑aware optimisation phases
    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.35
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline, max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # Phase 2 – Exact sub‑gradient descent (averaged active set)
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline, max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # Phase 3 – L2 flattening (drive convolution to a constant)
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.20
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline, max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # Phase 4 – Simulated‑annealing hill‑climb
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _hill_climb_anneal(global_best_seq, n, phase_deadline, max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform distribution")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
