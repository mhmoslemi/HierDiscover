def gradient_smooth_max(a: np.ndarray, tau: float) -> np.ndarray:
    n = a.shape[0]
    conv = _conv_fft(a, a)  # len 2n-1
    max_val = conv.max()
    # compute softmax weights
    # subtract max for numerical stability
    w = np.exp((conv - max_val) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback uniform
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # rev a
    a_rev = a[::-1]
    # compute convolution w * a_rev
    # need appropriate length for FFT
    prod = _conv_fft(w, a_rev)  # len = (2n-1)+n-1 = 3n-2
    # slice indices n-1..n-1+n-1 (i.e., n entries)
    start = n - 1
    slice_ = prod[start:start + n]
    grad = 2.0 * slice_
    return grad

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of three time‑aware stages:

1. **Deterministic seeds** – uniform, Catalan (coefficients of
   (1‑z)^‑½) and a previously found solution (if available) are
   evaluated for several lengths.

2. **Massive Dirichlet sampling** – while time permits many independent
   Dirichlet(1) vectors of several lengths are drawn and the best
   (smallest) objective is kept.

3. **Smooth‑max mirror descent** – starting from the best vector we run
   an accelerated exponentiated‑gradient method that minimises a smooth
   approximation of the ``max(conv)`` functional:
   * we replace the hard max by a soft‑max with a temperature τ,
   * the (soft) gradient is obtained analytically via an additional FFT,
   * a multiplicative (exponentiated) update is performed and the vector
     is re‑normalised onto the probability simplex.

The temperature is gradually annealed, the learning rate decays as
1/√t, and the whole routine respects the supplied time budget (default
1000 s).  The best sequence found is returned as a plain ``list`` of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` when ``seq`` is already normalised
    to sum = 1: 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^‑½, normalised to sum = 1."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α = 1)
# ----------------------------------------------------------------------


def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector (normalised exponential draws)."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implementation follows the algorithm of [Wang & Carreira‑Perpiñán 2013].
    """
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical guard – if the sum is zero, fallback to uniform
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max over the convolution entries)
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, tau: float) -> tuple[np.ndarray, float]:
    """
    Returns a gradient of the smooth approximation
        f_τ(a) = τ·log Σ_k exp(conv_k / τ)
    and the current (hard) maximum value of the convolution.
    The gradient w.r.t. a is
        g_i = 2·Σ_k w_k·a_{k-i}   with   w_k = softmax(conv_k / τ).
    """
    n = len(a)
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_val = conv.max()
    # soft‑max weights, numerically stable
    shifted = conv - max_val                    # ≤ 0
    w = np.exp(shifted / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient via a second convolution
    a_rev = a[::-1]                            # length n
    grad_full = _conv_fft(w, a_rev)            # length (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]    # keep only n useful entries
    return grad, max_val

# Smooth‑max mirror descent on the simplex
# ----------------------------------------------------------------------


def _refine_smooth(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) minimising the smooth‑max
    functional.  The temperature τ is annealed, the learning rate is
    η_t = η₀ / √t.  The iteration stops when time runs out.
    Returns the best sequence seen and its hard‑max objective.
    """
    a = a_start.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    eta0 = 1.0        # initial learning rate
    tau = 0.01        # initial temperature
    tau_min = 1e-6
    tau_decay = 0.9995

    while time.time() < deadline:

        grad, max_conv = _smooth_max_gradient(a, tau)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex
        a = _project_to_simplex(a)

        # temperature annealing
        tau = max(tau_min, tau * tau_decay)

                best_seq = a.copy()
