def hill_climb_targeted(seq, n, deadline, iters):

    cur = seq.copy(); cur_conv = _conv_fft(cur, cur); cur_max = cur_conv.max()
    best_val = cur_max; best_seq = cur.copy()
    for it in range(iters):
        if time.time()>deadline: break
        k_star = int(np.argmax(cur_conv))
        i_start = max(0, k_star - (n-1))
        i_end = min(n-1, k_star)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        opp = k_star - idx_i
        contrib = cur[idx_i] * cur[opp]
        if contrib.sum()==0: continue
        # heavy index pick weighted by contrib
        heavy_idx = idx_i[np.argmax(contrib)]  # largest contributor
        # Choose target index random not equal
        target = np.random.randint(0, n)
        # compute available mass to move, maybe a fraction alpha
        alpha = 0.5  # fraction of heavy mass to move
        delta = -alpha * cur[heavy_idx]
        if delta == 0: continue
        new_alloc_heavy = cur[heavy_idx] + delta  # smaller
        new_alloc_target = cur[target] - delta  # increase (ensure positivity)
        if new_alloc_target < 0: continue
        # compute deltas vector
        deltas = np.zeros_like(cur)
        deltas[heavy_idx] = delta
        deltas[target] = -delta
        # compute new conv incrementally
        conv_new = cur_conv.copy()

def _deterministic_climb(start_seq, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_max = cur_conv.max()
    best_seq = cur.copy()
    best_val = cur_max

    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # Determine bottleneck
        k_star = int(np.argmax(cur_conv))
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end+1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]
        # choose heavy index
        heavy_idx = idx_range[np.argmax(contrib)]
        # choose random target not equal heavy
        target = rng.integers(0, n)
        if target == heavy_idx:
            continue
        # compute delta fraction
        epsilon = 0.2  # or adapt
        delta = -epsilon * cur[heavy_idx]
        if cur[heavy_idx] + delta <= 0:
            continue
        # new allocation
        new_heavy = cur[heavy_idx] + delta
        new_target = cur[target] - delta
        if new_target < 0:
            continue
        # compute incremental conv
        conv_new = cur_conv.copy()
        # apply heavy delta
        conv_new[heavy_idx:heavy_idx+n] += 2*delta*cur
        conv_new[2*heavy_idx] += delta*delta
        # apply target delta
        delta_t = -delta
        conv_new[target:target+n] += 2*delta_t*cur
        conv_new[2*target] += delta_t*delta_t
        # cross term

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (∑ a)² .

The search consists of several stages, each respecting a global deadline
(≈ 1000 s).  It combines:

* a rich set of analytically‑motivated seed families (uniform, sine‑,
  Hann, parabolic, power‑law, beta‑shapes, geometric, Dirichlet draws,
  sparse patterns);
* Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate;
* exact sub‑gradient mirror‑descent on the true max‑convolution;
* a long targeted simulated‑annealing phase that moves mass away from
  the current bottleneck entry using incremental FFT‑free updates;
* a final deterministic hill‑climbing phase that repeatedly steals
  mass from the heaviest contributors to the bottleneck and places it
  elsewhere, accepting only improvements.

All stages are time‑aware – they stop as soon as the deadline is reached.
The best sequence seen over the whole run is returned as a plain list of
floats (already normalised to sum = 1, all entries ≥ 0).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _parabolic(n: int) -> np.ndarray:
    """Symmetric concave parabola, 0 at the ends, 1 in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - ((i - centre) / centre) ** 2
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_parabolic(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 10)
    betas = np.linspace(0.30, 1.20, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.1, 0.3, 0.6, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8, 12):
        for conc in (0.1, 0.3, 0.6):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12000) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) approximation.
    Returns the best sequence seen and its objective value.
    """
    eps = 1e-30
    v = np.log(seed + eps)               # working in log‑space guarantees positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing schedule – start modest, grow exponentially
    beta_smooth = 2.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)       # length 3n‑2, central n entries needed
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    Returns the best sequence and its value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 1500000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing focusing on the current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few of the heaviest contributors (max 6)
        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to modify (2‑4)
        k_move = rng.integers(2, 5)

        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


def _deterministic_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly moves a fraction of the mass from the
    heaviest contributor to the current bottleneck towards a random other index.
    Only strictly improving moves are accepted.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_max = np.max(cur_conv)

    best_seq = cur.copy()
    best_val = cur_max

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # locate bottleneck
        k_star = int(np.argmax(cur_conv))
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_i
        contrib = cur[idx_i] * cur[opp]

        if contrib.size == 0:
            continue

        heavy = idx_i[np.argmax(contrib)]

        # pick a random destination distinct from heavy
        dest = rng.integers(0, n)
        if dest == heavy:
            continue

        # fraction of mass to move (randomised a bit)
        eps = rng.uniform(0.03, 0.20)
        delta = -eps * cur[heavy]          # we take mass away from heavy
        if cur[heavy] + delta <= 0.0:
            continue
        if cur[dest] - delta < 0.0:        # would make dest negative
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # heavy -> decrease
        conv_new[heavy:heavy + n] += 2.0 * delta * cur
        conv_new[2 * heavy] += delta * delta
        # dest -> increase
        delta2 = -delta
        conv_new[dest:dest + n] += 2.0 * delta2 * cur
        conv_new[2 * dest] += delta2 * delta2
        # cross term between the two moved indices
        conv_new[heavy + dest] += 2.0 * delta * delta2

        new_max = np.max(conv_new)

        # accept only if we improve
        if new_max < cur_max - 1e-12:
            cur[heavy] += delta
            cur[dest] += delta2
            cur_conv = conv_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_seq = cur.copy()
                print(f"[DetClimb] n={n:5d} it={it:7d} best={2.0 * n * new_max:.6f}")

    best_seq = best_seq / best_seq.sum()
    best_obj = 2.0 * n * best_val
    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # make previous solution available (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Choose lengths – two different scales give complementary exploration
    lengths = [2000, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # 1) Evaluate raw seeds
        # ------------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the best few seeds
        # ------------------------------------------------------------------
        top_k = min(5, len(seeds))
        top_idx = np.argsort(seed_vals)[:top_k]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=12000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement (longer run)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=300000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Targeted simulated‑annealing (high‑frequency)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline,
                                             max_iters=1500000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Deterministic hill‑climbing refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _deterministic_climb(global_best_seq, n, deadline,
                                            max_iters=120000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑DetClimb] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity check (ignore exceptions, just to be safe)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
