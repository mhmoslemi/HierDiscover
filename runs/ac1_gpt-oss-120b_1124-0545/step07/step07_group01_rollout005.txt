propose_candidate – a time‑bounded optimiser for the step‑function evaluation
==========================================================================

The optimiser builds on the successful “Adam → sub‑gradient → LP → anneal”
pipeline but allocates the budget more aggressively:

* a very large pool of sparse random supports (size 30 … 400) is generated;
  the best few are kept.
* Adam on a smooth‑max surrogate runs for up to 30 000 iterations per seed.
* An extended exponentiated‑gradient (mirror‑descent) sub‑gradient phase
  performs up to several million iterations (or until the time budget runs
  out).
* If `cvxpy` is available a cheap LP refinement is applied.
* The final stage repeatedly perturbs the current best vector and runs a
  short sub‑gradient refinement – this “re‑start’’ loop extracts a few
  extra percent points while still respecting the 1000 s limit.

All operations stay on the probability simplex (a_i ≥ 0, Σa_i = 1) and the
code is written without closures so that the grader can import it directly.
"""

import time
import math

import sys

# -------------------------------------------------------------------------
# Low‑level FFT based convolution (exact, real‑valued)
# -------------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, length = len(a)+len(b)-1."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The vector is assumed non‑negative; a zero‑sum vector returns np.inf.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# -------------------------------------------------------------------------
# Simple families – used as seeds
# -------------------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    """Full‑support Dirichlet‑like vector."""
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of exactly ``m`` indices."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# -------------------------------------------------------------------------
# Utilities for the simplex
# -------------------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1

        a = _softmax(v)                         # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·(a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Runs until the deadline or the iteration budget is exhausted.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)           # keep numbers bounded
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 5) -> np.ndarray:
    """
    Tiny LP refinement: maximise Σ g_j subject to f * g ≤ M (M is current max).
    Only constraints belonging to the active set are used, making the LP
    extremely cheap.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:               # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that swaps a tiny fraction of mass
    between two random positions and occasionally performs a global Gaussian
    jitter.  Runs for the remainder of the time budget.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# -------------------------------------------------------------------------
# Seed generation (rich and sparse)
# -------------------------------------------------------------------------

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse pool of start vectors (all normalised to Σ=1).
    The pool contains:
      • the previous best (if any),
      • a handful of full‑support random softmax vectors,
      • many sparse random supports with sizes ranging from 30 up to 400.
    """
    seeds: list[np.ndarray] = []

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # a few full‑support random draws
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – the main source of good candidates
    for m in range(30, 420, 40):          # 30,70,…,390
        for _ in range(4):                # a few repetitions per size
            seeds.append(_sparse_random_seed(n, m, rng))

    # clean‑up: enforce non‑negativity and re‑normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# -------------------------------------------------------------------------
# Main entry point expected by the grader
# -------------------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the allotted time budget and returns the best
    sequence found as a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # a small safety margin

    # ------------------------------------------------------------------
    # Length choice – 4000 offers a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Retrieve the previously best construction (if the grader supplied one)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # 1) Build a large seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick screening – keep the top few seeds for expensive optimisation
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] best initial value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on the smoothed‑max surrogate
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=30000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Long sub‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=2_000_000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Optional LP refinement (if cvxpy is present)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and time.time() < deadline - 5:
        remaining = deadline - time.time()
        lp_deadline = time.time() + min(30.0, remaining / 2)
        seq_lp = _lp_refine(global_best_seq, n, lp_deadline,
                            t_mix=0.3, max_lp_iters=5)
        # short sub‑gradient polishing after the LP mix
        seq, val = _subgradient_refine(seq_lp, n, deadline, max_iters=200_000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[post‑LP SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Re‑start loop: perturb + short sub‑gradient
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # small random perturbation of the current best vector
        pert = global_best_seq.copy()
        mask = rng.random(n) < 0.07               # touch ~7 % of entries
        perturbation = rng.normal(scale=0.02, size=mask.sum())
        pert[mask] = np.clip(pert[mask] + perturbation, 0.0, None)
        if pert.sum() == 0.0:
            pert = _uniform(n)
        else:
            pert /= pert.sum()

        # short sub‑gradient run from the perturbed point
        seq, val = _subgradient_refine(pert, n, deadline, max_iters=80_000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[restart] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final simulated‑annealing fine‑tuning for any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be required, but guard against an empty result
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
