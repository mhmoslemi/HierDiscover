"""
search_step_function_v3.py

A time‑limited optimiser that focuses on **sparse support selection**.
The key observation is that for a uniform distribution on a support S
the convolution value at any shift equals

    count(k) / |S|² ,

where count(k) is the number of ordered pairs (i,j)∈S×S with i+j=k.
Hence minimising the objective 2·n·max(conv) is equivalent to
finding a support with a small maximal pair‑count.  The programme:

1.  Performs a fast simulated‑annealing search in the space of supports
    (sizes 120…200) where each move swaps one index of S with a new
    index.  Pair‑counts are maintained incrementally, so millions of
    swaps fit easily into the time budget.

2.  Builds a uniform step‑function on the best support found
    (mass 1/|S| on every selected position).

3.  Refines the mass distribution with the exponentiated‑gradient
    sub‑gradient method from the original script.

4.  (Optionally) runs a tiny LP‑refinement (cvxpy) and a short
    simulated‑annealing fine‑tuning phase.

The whole pipeline respects the supplied time budget (default 1000 s)
and returns the best sequence discovered as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector with sum>0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _init_conv_counts(support: set[int], n: int) -> np.ndarray:
    """
    Compute ordered‑pair counts for a given support.
    Returns an integer array of length 2·n‑1 where entry k holds
    the number of ordered pairs (i,j)∈S×S with i+j=k.
    """
    size = 2 * n - 1
    cnt = np.zeros(size, dtype=np.int32)
    supp = list(support)
    # diagonal contributions (i,i)
    for i in supp:
        cnt[2 * i] += 1
    # off‑diagonal contributions (i,j) and (j,i)
    m = len(supp)
    for idx_i in range(m):
        i = supp[idx_i]
        for idx_j in range(idx_i + 1, m):
            j = supp[idx_j]
            s = i + j
            cnt[s] += 2
    return cnt


def _apply_removal(support: set[int], conv_cnt: np.ndarray, i: int) -> None:
    """
    Remove index i from the support and update ordered‑pair counts.
    """
    for u in support:
        s = i + u
        if u == i:
            conv_cnt[s] -= 1          # diagonal
        else:
            conv_cnt[s] -= 2          # ordered pair (i,u) & (u,i)
    support.remove(i)


def _revert_removal(support: set[int], conv_cnt: np.ndarray, i: int) -> None:
    """
    Undo a previous removal of i (i is currently absent from support).
    """
    for u in support:
        conv_cnt[i + u] += 2
    conv_cnt[2 * i] += 1
    support.add(i)


def _apply_addition(support: set[int], conv_cnt: np.ndarray, j: int) -> None:
    """
    Add index j to the support and update ordered‑pair counts.
    """
    for u in support:
        conv_cnt[j + u] += 2
    conv_cnt[2 * j] += 1
    support.add(j)


def _revert_addition(support: set[int], conv_cnt: np.ndarray, j: int) -> None:
    """
    Undo a previous addition of j (j is currently present in support).
    """
    support.remove(j)
    for u in support:
        conv_cnt[j + u] -= 2
    conv_cnt[2 * j] -= 1


def _local_search_support(n: int,
                          m: int,
                          deadline: float,
                          rng: np.random.Generator,
                          max_swaps: int = 10000) -> tuple[set[int], int]:
    """
    Simulated‑annealing on the support of size m.
    Returns the best support encountered and its maximal ordered‑pair count.
    """
    # start from a random support
    support = set(rng.choice(n, size=m, replace=False))
    conv_cnt = _init_conv_counts(support, n)
    cur_max = int(conv_cnt.max())
    best_support = set(support)
    best_max = cur_max

    # temperature schedule
    T = max(1.0, cur_max * 0.5)
    decay = 0.9995
    swaps = 0

    while time.time() < deadline and swaps < max_swaps:
        swaps += 1

        # propose a swap i (in) -> j (out)
        i = rng.choice(list(support))
        j = rng.integers(0, n)
        while j in support:
            j = rng.integers(0, n)

        # apply swap
        _apply_removal(support, conv_cnt, i)
        _apply_addition(support, conv_cnt, j)
        new_max = int(conv_cnt.max())
        delta = new_max - cur_max

        # acceptance rule
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_support = set(support)
        else:
            # revert
            _revert_addition(support, conv_cnt, j)
            _revert_removal(support, conv_cnt, i)

        # temperature decay
        T *= decay

    return best_support, best_max


def _search_best_support(n: int,
                         time_budget: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly runs the local‑search for different support sizes.
    Returns a probability vector (sum=1) and its exact objective value.
    """
    deadline = time.time() + time_budget
    global_best_seq = None
    global_best_val = float("inf")
    best_m = None
    best_maxcnt = None

    # try a mixture of support sizes
    sizes = list(range(120, 201, 10))   # 120,130,…,200
    while time.time() < deadline:
        m = rng.choice(sizes)
        # give each run at most 0.12 s (enough for a few thousand swaps)
        run_deadline = min(deadline, time.time() + 0.12)
        support, maxcnt = _local_search_support(n, m, run_deadline, rng,
                                                max_swaps=4000)
        # objective for uniform mass on this support
        obj = 2.0 * n * maxcnt / (m * m)

        if obj < global_best_val:
            global_best_val = obj
            best_m = m
            best_maxcnt = maxcnt
            seq = np.zeros(n, dtype=np.float64)
            seq[list(support)] = 1.0 / m
            global_best_seq = seq
            print(f"[Support] m={m} maxcnt={maxcnt} obj={obj:.6f}")

    if global_best_seq is None:
        # fall back to a uniform distribution (should never happen)
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_val = _objective(global_best_seq)

    print(f"[Best support] m={best_m} maxcnt={best_maxcnt} obj={global_best_val:.6f}")
    return global_best_seq, global_best_val

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _geom(n, 0.5)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    v = np.log(seed + 1e-30)  # logits
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 5) -> np.ndarray:
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

    seeds = _make_seeds(n, previous, rng)
    # add the support‑based vector as an extra seed
    seeds.append(seq0)

    # quick evaluation of seeds
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    # update global best from seeds
    if seed_vals[0][1] < global_best_val:
        global_best_val = seed_vals[0][1]
        global_best_seq = seed_vals[0][0].copy()
        print(f"[seed] new global best={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the few elite seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient mirror‑descent (longer run)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=400000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Linear‑programming refinement (if CVXPY present)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and global_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            lp_deadline = time.time() + min(30.0, remaining / 2)
            seq_lp = _lp_refine(global_best_seq, n, lp_deadline,
                                t_mix=0.3, max_lp_iters=5)
            seq, val = _subgradient_refine(seq_lp, n, deadline,
                                           max_iters=100000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[post‑LP SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated‑annealing fine‑tuning (uses all remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
