def _project_simplex(v):
    # standard: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function (list of heights) that minimizes
the objective used by ``evaluate_sequence``.

Algorithm (time‑aware, ~1000 s):
1. **Deterministic seeds** – uniform, Catalan, sine window and a
   symmetric tent for several lengths.  Their exact objective
   ``2·n·max(conv)`` is evaluated; the best per length is kept.
2. **Broad random search** – while time permits we draw many random
   vectors (Dirichlet, power‑law, exponential) of the same lengths.
   The best overall candidate is stored.
3. **Smooth‑max mirror descent** – starting from the best candidates we
   minimise a *smoothed* version of the max‑convolution using a
   Nesterov‑style Adam update on the simplex (exponentiated‑gradient
   is replaced by a projected‑gradient with adaptive step‑size).
   The smoothing parameter ε is gradually reduced, so the method
   converges to the true (non‑smooth) optimum.
4. The best sequence found is returned as a plain ``list`` (the grader
   will call ``evaluate_sequence`` on it).

All heavy work is done with FFT‑based convolutions (O(n log n)).
The routine respects the supplied time budget and never exceeds it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, double‑precision)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return for a normalised
    (sum = 1) sequence: 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i – the series of (1‑z)^‑½.
    After truncation they give a near‑constant auto‑convolution.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _sine_window(n: int) -> np.ndarray:
    """Half‑sine window (positive, sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)  # values in (0,1]
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _tent(n: int) -> np.ndarray:
    """Symmetric tent (linear increase then decrease)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()

def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector (i.e. normalized exponentials)."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with a random exponent p∈[0.1,2]."""
    p = rng.uniform(0.1, 2.0)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ exp(-λ·i) with λ drawn uniformly in a modest range."""
    lam = rng.uniform(0.0005, 0.01)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return a / a.sum()

# Simplex projection

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed max and its gradient on the simplex
# ----------------------------------------------------------------------


def _smooth_max_and_weights(conv: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv) and the corresponding
    soft‑max weights w (sum w = 1).
    """
    maxc = np.max(conv)
    shifted = conv - maxc            # avoid overflow
    exp_shift = np.exp(shifted / eps)
    sum_exp = np.sum(exp_shift)
    w = exp_shift / sum_exp
    smooth_val = maxc + eps * np.log(sum_exp)
    return smooth_val, w


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  $\sum_k w_k \, (a*a)[k]$ w.r.t. a.
    Uses FFT‑based convolution:  g = 2·(reverse(w) * a) restricted to
    the central n entries and reversed.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # length 2n‑1
    N = _next_pow2(n + len(w_rev) - 1)  # size for FFT
    fa = np.fft.rfft(a, n=N)
    fw = np.fft.rfft(w_rev, n=N)
    conv_full = np.fft.irfft(fa * fw, n=N)  # real convolution
    # central slice of length n (indices n‑1 … 2n‑2)
    slice_ = conv_full[n - 1 : n - 1 + n]
    grad = 2.0 * slice_[::-1]  # reverse to align indices
    return grad.real

def _refine_sequence(start_seq: np.ndarray, deadline: float):
    """
    Minimises the true objective by iterating on a smooth surrogate.
    Uses Adam updates and projects back onto the probability simplex.
    Returns the best sequence observed and its exact objective.
    """
    a = start_seq.copy()
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    it = 0
    # maximum number of internal iterations – a safety guard
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        it += 1

        # --------------------------------------------------------------
        # 1) full self‑convolution (exact, needed for objective & smooth)
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)                # length 2n‑1
        actual_val = 2.0 * n * np.max(conv)   # exact objective (sum a = 1)

        if actual_val < best_val:
            best_val = actual_val
            best_seq = a.copy()
            print(f"[Refine] it={it:6d} n={n:5d} best={actual_val:.6f}")

        # --------------------------------------------------------------
        # 2) smoothing parameter (decreases like 1/√it)
        # --------------------------------------------------------------
        eps = max(0.001, 0.02 / np.sqrt(it))

        # --------------------------------------------------------------
        # 3) soft‑max weights for the smooth max
        # --------------------------------------------------------------
        _, w = _smooth_max_and_weights(conv, eps)

        # --------------------------------------------------------------
        # 4) gradient of the smooth surrogate
        # --------------------------------------------------------------
        g = _grad_from_weights(a, w)

        # --------------------------------------------------------------
        # 5) Adam step
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step                         # gradient descend

        # --------------------------------------------------------------
        # 6) project back onto the simplex
        # --------------------------------------------------------------
        a = _project_simplex(a)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence found as a plain ``list`` of ``float``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a tiny safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore (larger n allows a lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]

    # best per length: {n: (seq, value)}
    best_per_len = {}

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    seed_funcs = [_uniform, _catalan_shape, _sine_window, _tent]
    for n in lengths:
        for fn in seed_funcs:
            seq = fn(n)
            val = _objective(seq)
            cur_best = best_per_len.get(n, (None, np.inf))[1]
            if val < cur_best:
                best_per_len[n] = (seq.copy(), val)
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # optionally reuse a previously computed global solution
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, tuple, np.ndarray)):
            try:
                arr = np.asarray(prev, dtype=np.float64)
                if arr.shape == (n,) and np.all(arr >= 0):
                    arr = arr / arr.sum()
                    val = _objective(arr)
                    cur_best = best_per_len.get(n, (None, np.inf))[1]
                    if val < cur_best:
                        best_per_len[n] = (arr.copy(), val)
                        print(f"[Seed] n={n:5d} type=previous   val={val:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Stage 2 – massive random sampling (different families)
    # ------------------------------------------------------------------
    now = time.time()
    sampling_deadline = now + (deadline - now) * 0.4  # 40 % of remaining time

    random_families = [_random_dirichlet_one, _random_powerlaw,
                      _random_exponential]

    while time.time() < sampling_deadline:
        # favour larger n (more degrees of freedom)
        probs = np.array(lengths, dtype=float)
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))

        family = rng.choice(random_families)
        seq = family(n, rng)

        val = _objective(seq)

        cur_best = best_per_len.get(n, (None, np.inf))[1]
        if val < cur_best:
            best_per_len[n] = (seq.copy(), val)
            print(f"[Random] n={n:5d} fam={family.__name__:15s} val={val:.6f}")

        # Early stop if we already beat the target bound
        if val < 1.503:
            break

    # ------------------------------------------------------------------
    # Choose the most promising lengths for refinement
    # ------------------------------------------------------------------
    # sort lengths by current best value (ascending)
    sorted_lengths = sorted(best_per_len.items(),
                           key=lambda kv: kv[1][1])[:2]  # keep up to two

    # ------------------------------------------------------------------
    # Stage 3 – smooth‑max mirror descent on the selected candidates
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 2.0 and sorted_lengths:
        # split the remaining budget equally between the chosen lengths
        per_len_deadline = time.time() + remaining_time / len(sorted_lengths)

        for n, (seq, val) in sorted_lengths:
            # each refinement gets its own slice of the remaining budget
            sub_deadline = min(per_len_deadline, deadline)
            refined_seq, refined_val = _refine_sequence(seq, sub_deadline)

            # update global best if improved
            global_best_val = min(v for _, v in best_per_len.values())
            if refined_val < global_best_val:
                best_per_len[n] = (refined_seq.copy(), refined_val)
                print(f"[Overall‑Refine] n={n:5d} new best={refined_val:.6f}")

    # ------------------------------------------------------------------
    # Pick the overall best sequence among all lengths
    # ------------------------------------------------------------------
    best_n, (best_seq, best_val) = min(best_per_len.items(),
                                       key=lambda kv: kv[1][1])

    # ------------------------------------------------------------------
    # Final sanity check – ensure the grader will accept the output
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {best_n})"
    )
    return best_seq.tolist()
