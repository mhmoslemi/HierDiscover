  thousand iterations.

All heavy work respects the global 1000 s budget; the best vector found
is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a simplex‑normalised vector (∑ a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (0<r<1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _triangular_window(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (cosine‑tapered) window."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.ones_like(i)
    taper = alpha / 2
    mask = i < taper
    a[mask] = 0.5 * (1 + np.cos(np.pi * (i[mask] / taper - 1)))
    mask = i > (1 - taper)
    a[mask] = 0.5 * (1 + np.cos(np.pi * ((i[mask] - 1) / taper + 1)))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    """Exponential decay a[i] = exp(-i/σ)."""
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    """Logistic shape a[i] = 1/(1+exp((i‑offset)/scale))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw (conc < 1 → very sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # deterministic baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_triangular_window(n))
    seeds.append(_tukey_window(n, alpha=0.4))

    # geometric families
    for r in np.linspace(0.2, 0.95, 8):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # Beta‑like shapes – a dense sweep
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2, n):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * rng.uniform(1.0, 3.0)
        seeds.append(_logistic_shape(n, offset, scale))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution if lengths match
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray, n: int, deadline: float, max_iters: int = 25000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of the max."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters (softmax space)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # smoothing parameter for log‑sum‑exp
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)  # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # ---- smooth max weight (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                         # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # Jacobian of softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping to keep updates stable
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float, max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step on simplex
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # for stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


def _sa_optimize(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 5_000_000,
                 init_temp_factor: float = 0.02,
                 min_temp: float = 1e-12,
                 decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated‑annealing with an incremental convolution update.
    Two move types:
        – Dirichlet re‑allocation on 2‑6 randomly chosen indices,
        – Targeted re‑allocation that touches indices contributing to the
          current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # ---------- select move type ----------
        if rng.random() < 0.45:
            # random Dirichlet move on 2‑6 positions
            k = rng.integers(2, 7)                     # 2‑6 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # targeted move based on the current maximal convolution entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # weight each index by its contribution a[i]·a[k_star‑i]
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # pick a second index uniformly elsewhere
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a (both directions)
            conv_new[p:p + n] += 2.0 * delta * cur
            # Δ*Δ term (self)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for i in range(idx.size):
            for j in range(i + 1, idx.size):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj


def _fine_tune(seq: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 200_000,
               epsilon: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climber: repeatedly move a tiny fraction ε of the mass from a
    position that contributes most to the current maximal convolution entry
    to a random other position, accepting only strictly improving moves.
    """
    rng = np.random.default_rng()
    cur = seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        k_star = int(np.argmax(cur_conv))
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        support = np.arange(i_start, i_end + 1, dtype=int)

        # contributions to the peak
        contrib = cur[support] * cur[k_star - support]
        if contrib.sum() == 0.0:
            continue
        i = rng.choice(support, p=contrib / contrib.sum())

        # pick a destination distinct from i
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n - 1)

        move = epsilon * cur[i]          # tiny fraction of mass
        if move <= 0.0:
            continue

        new = cur.copy()
        new[i] -= move
        new[j] += move
        # keep on simplex
        new_sum = new.sum()
        if new_sum == 0.0:
            continue
        new /= new_sum

        # recompute only the part of conv affected by i and j
        conv_new = cur_conv.copy()
        for p, delta in ((i, -move), (j, move)):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross‑terms for the two changed indices
        conv_new[i + j] += 2.0 * (-move) * move

        new_obj = 2.0 * n * np.max(conv_new)

        if new_obj < cur_obj:
            cur = new
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[Fine] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    up[up < 0] = 0.0
    up_sum = up.sum()
    if up_sum == 0.0:
        return _uniform(new_n)
    return up / up_sum

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # previous solution (if any) – may be None
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Progressive lengths – start small, finish with a large n
    lengths = [1024, 2048, 4096, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # obtain a seed of the proper length
        if previous is not None and previous.shape[0] == n:
            init = previous / previous.sum()
        elif previous is not None:
            init = _upsample(previous, n)
        else:
            init = _uniform(n)

        seeds = _make_seeds(n, init, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on smooth max (top‑k seeds only)
        # ------------------------------------------------------------------
        # keep the best few seeds for Adam
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(3, len(seed_vals))
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Adam‑overall] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=500000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[SubGrad‑overall] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Long‑run simulated‑annealing
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            # Rough estimate: ≈20 k SA moves per second for this n
            sa_iters = int(min(12_000_000, remaining * 20_000))
            seq, val = _sa_optimize(global_best_seq, n, deadline,
                                    max_iters=sa_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[SA‑overall] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Final greedy fine‑tuning (very cheap moves)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            ft_iters = int(min(300_000, remaining * 5_000))
            seq, val = _fine_tune(global_best_seq, n, deadline,
                                  max_iters=ft_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Fine‑overall] n={n:5d} new global best={val:.6f}")

        # keep the best sequence so far for the next length
        if global_best_seq is not None:
            previous = global_best_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 1024
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional final verification using the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
