def _proj_simplex(v):
    """Project onto probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero, return uniform
        return np.full_like(v, 1.0/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm overview
------------------
1. **Deterministic seeds** – generate a variety of analytic shapes
   (uniform, Catalan, triangular, half‑sine, Beta‑family) for several
   lengths.  The best among them seeds the search.

2. **Random Dirichlet sampling** – while time permits draw many
   Dirichlet(α) vectors (α is drawn log‑uniformly from 0.1 … 5) for the
   same lengths.  Keep the best sample.

3. **Smoothed gradient descent on the simplex** – starting from the
   current best vector, minimise a smooth approximation of the
   ``max(conv)`` functional.  The gradient of the smoothed max is
   computed via two fast FFT‑convolutions, the update is a projected
   gradient step onto the probability simplex (simplex projection).

4. **Heuristic polishing** – apply random small Gaussian noise,
   symmetrisation and low‑pass smoothing (Gaussian kernels) to the
   best vector found so far.  Each modification is projected back onto
   the simplex and kept if it improves the true objective.

All steps are time‑aware; the routine never exceeds the supplied budget
(and leaves a safety margin).  The returned list is ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Quantity that ``evaluate_sequence`` would return for a *already*
    normalised (sum=1) vector:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Catalan coefficients (1‑z)^‑½, normalised to sum = 1."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangle: 0 at the edges, 1 in the centre, then normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - centre) * 2.0 / (n - 1)
    shape = np.clip(shape, 0.0, None)
    total = shape.sum()
    if total == 0.0:
        return _uniform(n)
    return shape / total


def _half_sine_shape(n: int) -> np.ndarray:
    """Half‑sine wave (positive on (0,π)), normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n)
    shape = np.clip(shape, 0.0, None)
    total = shape.sum()
    if total == 0.0:
        return _uniform(n)
    return shape / total


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density, normalised to sum = 1."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid 0^neg or 1^neg by clipping
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = shape.sum()
    if total == 0.0 or np.isnan(total):
        return _uniform(n)
    return shape / total


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient of max‑convolution
# ----------------------------------------------------------------------


def _smooth_grad(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max functional
        f(seq) = (2·n/β)·log Σ exp(β·conv(seq,seq))
    with respect to `seq`.  The factor 2·n is omitted – it merely
    rescales the step size and is absorbed into the learning‑rate.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)                # length 2n‑1
    scaled = conv * beta
    max_scaled = np.max(scaled)
    # stable soft‑max weights
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # weighted sum of shifted copies of `seq`
    w_rev = w[::-1]
    conv2 = _conv_fft(w_rev, seq)            # length 3n‑2
    start = 2 * n - 2
    grad_slice = conv2[start:start + n]
    grad = 2.0 * grad_slice
    return grad


# ----------------------------------------------------------------------
# Heuristic post‑processing helpers
# ----------------------------------------------------------------------


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (average with its reversal)."""
    return (seq + seq[::-1]) * 0.5


def _smooth_sequence(seq: np.ndarray, kernel_size: int) -> np.ndarray:
    """Low‑pass smooth with a normalized Gaussian kernel (same length)."""
    sigma = kernel_size / 2.0
    x = np.arange(kernel_size) - (kernel_size - 1) / 2.0
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(seq, kernel, mode='same')
    return _project_onto_simplex(smoothed)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  Returns the best sequence found as
    a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Optional previous solution (if one is stored globally)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (larger lengths give smaller averages but also
    # a larger search space)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 0 – deterministic analytic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        # uniform
        seq = _uniform(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] uniform n={n:5d} val={val:.6f}")

        # Catalan
        seq = _catalan_shape(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] Catalan n={n:5d} val={val:.6f}")

        # triangular
        seq = _triangular_shape(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] triangular n={n:5d} val={val:.6f}")

        # half‑sine
        seq = _half_sine_shape(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] half‑sine n={n:5d} val={val:.6f}")

        # a few symmetric Beta families
        for (a, b) in [(2.0, 2.0), (3.0, 3.0), (2.0, 5.0), (5.0, 2.0), (1.5, 1.5)]:
            seq = _beta_shape(n, a, b)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] Beta a={a:.1f} b={b:.1f} n={n:5d} val={val:.6f}")

        # previous solution if it matches the current length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed] previous n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – massive random Dirichlet sampling (α drawn log‑uniform)
    # ------------------------------------------------------------------
    while time.time() < deadline * 0.55:      # use roughly half the budget
        # bias towards larger lengths (more “averaging” power)
        probs = np.array(lengths, dtype=float)
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))

        # draw α from log‑uniform [0.1, 5.0]
        log_alpha = rng.uniform(np.log(0.1), np.log(5.0))
        alpha = np.exp(log_alpha)

        # Dirichlet with common concentration α
        x = rng.gamma(alpha, 1.0, size=n)
        seq = x / x.sum()

        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random] Dirichlet n={n:5d} α={alpha:.3f} val={val:.6f}")

        # early exit if we already cross the target bound
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 2 – smoothed gradient descent on the simplex
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform vector of moderate size
        best_seq = _uniform(2000)

    seq = best_seq.copy()
    n = len(seq)
    it = 0
    gd_deadline = deadline * 0.93   # leave ~7% for polishing

    while time.time() < gd_deadline:
        it += 1
        # gradually increase smoothing (larger β → sharper approximation)
        beta = 4000.0 + (it // 1000) * 2000.0   # 4k,6k,8k,...

        # gradient of the smoothed max
        grad = _smooth_grad(seq, beta)

        # simple step‑size schedule (decays like 1/√t)
        lr = 0.08 / np.sqrt(it)

        # stochastic perturbation (occasionally) to help escape
        if rng.random() < 0.015:
            grad += rng.normal(scale=0.001, size=n)

        # gradient step + projection onto simplex
        new_seq = _project_onto_simplex(seq - lr * grad)

        # evaluate true objective
        val = _objective(new_seq)
        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            print(f"[GD] it={it:6d} β={beta:.0f} val={val:.6f}")

        seq = new_seq   # continue iterating from the latest point

    # ------------------------------------------------------------------
    # Stage 3 – heuristic polishing (symmetrisation & Gaussian smoothing)
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    while time.time() < deadline:
        # 1) random symmetrisation
        if rng.random() < 0.12:
            cand = _symmetrize(seq)
            cand = _project_onto_simplex(cand)
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                seq = cand
                print(f"[Sym] val={val:.6f}")

        # 2) Gaussian smoothing with a randomly chosen kernel size
        k = rng.choice([3, 5, 7, 9, 11])
        cand = _smooth_sequence(seq, kernel_size=k)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            seq = cand
            print(f"[Smooth] k={k:2d} val={val:.6f}")

        # 3) tiny random perturbation (still on the simplex)
        if rng.random() < 0.03:
            noise = rng.normal(scale=0.0005, size=n)
            cand = seq + noise
            cand = _project_onto_simplex(cand)
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                seq = cand
                print(f"[Noise] val={val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
