"""
propose_candidate.py

A time‑bounded optimiser for the “step‑function” challenge.

The algorithm proceeds in several stages, each of which is an
“any‑time” technique:

1️⃣  Build a very diverse pool of start vectors (dense classic families,
    many random sparse supports of various sizes, and a few deterministic
    Sidon‑like constructions).

2️⃣  Keep the best few seeds and run a long Adam optimisation on a
    smooth‑max surrogate of `max(conv(a,a))`.  The temperature β is
    increased gradually but capped to avoid numerical overflow.

3️⃣  Refine the result with exponentiated‑gradient (mirror‑descent) on the
    exact non‑smooth objective.  This phase runs as long as the time
    budget permits.

4️⃣  A cheap linear‑programming step (via CVXPY when available) that
    increases the total mass while keeping the current maximal
    convolution unchanged, followed by a short sub‑gradient pass.

5️⃣  Simulated‑annealing with tiny moves between two random positions and
    occasional global Gaussian jumps.

6️⃣  Final “permutation‑refinement’’ – repeatedly shuffle the positions
    of the (already‑optimised) weights and keep any improvement.  Because
    the convolution depends on the ordering of the coefficients, a
    different placement of the same values can lower the peak autocorrelation
    further.

All steps respect the deadline (`budget_s` seconds).  The best sequence
found is stored in the global ``height_sequence_1`` (so that later runs
can reuse it) and returned as a plain Python list.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Uniform mass on a random support of size `m` (1 ≤ m ≤ n).
    """
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """
    Deterministic “Sidon‑like” support using triangular numbers.
    The positions are i·(i+1)//2 modulo n.
    """
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """
    Positions are i² (mod n).  For a prime n this set is a Sidon set.
    """
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits for a simplex parametrisation

    # Adam first‑order moments
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    # Hyper‑parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Temperature schedule for the smoothed max (capped)
    beta = 1.5
    beta_mul = 1.005
    beta_cap = 50.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = min(beta * beta_mul, beta_cap)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k · (a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam moment update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000,
                        rel_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 5) -> np.ndarray:
    """
    Perform a few cheap linear‑programming refinements.
    For each refinement we:
        * compute the current maximum M of conv(f,f)
        * collect the (few) indices where the bound is attained
        * solve   maximise 1ᵀ g   subject to   f * g ≤ M  and g ≥ 0
        * blend the solution g back into f with weight `t_mix`.
    Returns the refined vector (still on the simplex).
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt /= tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two random positions and occasionally adds a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)   # initial temperature
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # up to 2 % of source mass
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        # occasional larger “jump’’
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


def _permutation_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_trials: int = 2000) -> tuple[np.ndarray, float]:
    """
    Randomly permute the positions of the coefficients.
    Keep any permutation that improves the true objective.
    """
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1
        # generate a random permutation (full shuffle)
        perm = np.arange(n)
        rng.shuffle(perm)
        cand = best_seq[perm]
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand
            print(f"[Perm] trial={trials:4d} new best={cand_val:.6f}")

    return best_seq, best_val

def _make_seeds(n: int,

                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    The pool contains classic families, a large collection of sparse
    random vectors (|support|≈30…400) and a few deterministic Sidon‑like
    constructions.
    """
    seeds: list[np.ndarray] = []

    # ==== Classic dense families ====
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # ==== Sparse random seeds ====
    # support sizes from 30 up to 400 (step 30)
    for m in range(30, 401, 30):
        for _ in range(4):
            seeds.append(_sparse_random_seed(n, m, rng))

    # ==== Deterministic Sidon‑like constructions ====
    for m in (50, 70, 90, 110, 130, 150):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ----- clean‑up -----
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time
    budget and returns the best sequence discovered as a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # attempt to reuse a previously discovered construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Fixed length – 4000 offers a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1) Build a very rich seed pool (including many sparse candidates)
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the top few for the expensive phases
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")

    # record the best raw seed
    if seed_vals[0][1] < global_best_val:
        global_best_val = seed_vals[0][1]
        global_best_seq = seed_vals[0][0].copy()
        print(f"[seed] new global best={global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smoothed‑max (more iterations than the original)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient mirror‑descent (longer run)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=800000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Linear‑programming refinement (tiny LP, very cheap)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and global_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            lp_deadline = time.time() + min(30.0, remaining / 2)
            seq_lp = _lp_refine(global_best_seq, n, lp_deadline,
                                t_mix=0.3, max_lp_iters=5)
            # one short sub‑gradient pass after the LP mix
            seq, val = _subgradient_refine(seq_lp, n, deadline,
                                           max_iters=200000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[post‑LP SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Simulated‑annealing fine‑tuning (uses all remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Random permutation refinement (tiny, cheap, sometimes helpful)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _permutation_refine(global_best_seq, n, deadline,
                                       max_trials=2000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Perm] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
