def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Projects vector v onto the probability simplex (non-negative, sum=1)."""
    # uses algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros? then return uniform?
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function construction (minimising
2·n·max(conv(a,a)) / (sum(a)²))

The optimiser works in several stages:

1.  A very diverse pool of normalised seeds (windows, geometric /
    power families, many random Dirichlet draws and tiny perturbations
    of the best known construction).

2.  Multi‑stage smooth‑max L‑BFGS on the top seeds.  The smooth max is
    the log‑sum‑exp surrogate of the infinity‑norm; an increasingly
    aggressive β‑schedule (up to 2·10⁴) drives the solution towards the
    true max‑objective.

3.  A specialised mirror‑descent (exponentiated‑gradient) on the
    exact nonsmooth objective.  Instead of averaging over *all* active
    indices, we pick a *single* maximising convolution index – this
    provides a stronger descent direction and often pushes the bound
    a little further.

4.  A final high‑β L‑BFGS refinement followed by a very light Metropolis
    annealing (mass swaps) to polish any remaining irregularities.

All phases respect the global 1000 s budget; the best sequence found
is stored in ``height_sequence_1`` for reuse and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (logits → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Window families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _make_seed_pool(n: int,
                    previous: np.ndarray | None,
                    rng: np.random.Generator,
                    extra_random: int = 150) -> list[np.ndarray]:
    """
    Build a large heterogeneous pool of normalised start vectors.
    ``extra_random`` controls how many additional Dirichlet draws are added.
    """
    seeds: list[np.ndarray] = []

    # 1. deterministic families
    seeds.append(_uniform(n))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma in (0.05, 0.10, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma_factor=sigma))

    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # 2. many extra random Dirichlet draws with random concentration
    for _ in range(extra_random):
        alpha = rng.uniform(0.05, 5.0)
        seeds.append(_random_dirichlet(n, alpha, rng))

    # 3. tiny perturbations of already‑good seeds (helps escape plateaus)
    if previous is not None:
        for _ in range(5):
            pert = previous + rng.normal(scale=0.002, size=n)
            pert = np.clip(pert, 0.0, None)
            if pert.sum() > 0:
                seeds.append(pert / pert.sum())

    # 4. include previous construction (if any) – already normalised above
    if previous is not None:
        seeds.append(previous / previous.sum())

    # 5. final clean‑up – clip numerical negatives and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation

        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    ``beta`` > 0. ``v`` are the logits (real numbers).
    """
    a = _softmax(v)                     # probability vector, sum = 1

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)                # numerical stabilisation
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # overall factor 2·n

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    #   d/da Σₖ wₖ·convₖ(a,a) = 2·(rev(a) * w)   (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                 # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 300) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns (best_sequence, best_exact_objective).
    """
    if not SCIPY_OPT:
        # SciPy missing – fall back to the exact objective without optimisation
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_loc):
            return _smooth_obj_and_grad(v_loc, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinements – exponentiated‑gradient (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine_eg(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true non‑smooth objective.
    At each iteration we pick *one* maximising convolution index (instead
    of averaging over all active indices) and perform an exponentiated‑
    gradient update.  The step size decays as 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())               # single maximiser
        # build sub‑gradient for this index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vectorised slice assignment
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]

        lr = 0.9 / math.sqrt(it)             # classic 1/√t schedule
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stabilise exponentiation
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGradEG] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between
    two random coordinates.  Used only if time permits after all
    deterministic phases.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension.  4000 works well, but we keep the option to
    # enlarge it if time permits (the optimiser will automatically adjust).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Grab any previously discovered construction (helps with warm‑starts).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed ^ 0x9E3779B97F4A7C15)

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous, rng,
                                extra_random=200)   # many random Dirichlets

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])            # lower objective is better
    top_k = 8                                   # very few – they will be refined
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 20000.0, num=20)   # finer & higher β
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=400)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient mirror‑descent (single‑max variant).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine_eg(global_best_seq,
                                          n,
                                          deadline,
                                          max_iters=400_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final high‑β L‑BFGS refinement (starting from the latest best).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        # a short aggressive schedule, starting where we left off
        beta_fine = np.geomspace(20000.0, 80000.0, num=10)
        seq, val = _l_bfgs_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    beta_schedule=beta_fine,
                                    maxiter_per_stage=300)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Fine L‑BFGS] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # 6) Very light Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=200_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will call evaluate_sequence on it)
    return global_best_seq.tolist()
