"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is linear convolution.  The search consists of

1.  A very diverse seed pool (including extreme beta‑shapes and
    ultra‑sparse Dirichlet draws).
2.  Time‑aware optimisation phases:
      • Adam on a smooth‑max surrogate (log‑sum‑exp of the convolution)
      • Exponentiated‑gradient sub‑gradient descent on the true max
      • L²‑flattening (drive the auto‑convolution towards a constant)
      • Simulated‑annealing hill‑climb with occasional uphill moves.
   The phases are allocated adaptively to the remaining time budget.
3.  After each phase a cheap “interleaved‑ends” re‑ordering is tried:
    the largest coefficients are placed at opposite ends of the vector,
    the next largest just inside them, etc.  This often reduces the
    maximal convolution entry further.
4.  The final vector is verified with the supplied `evaluate_sequence`
    and stored as `height_sequence_1` for warm‑starting future runs.

All heavy work respects the global 1000 s budget, and the code uses only
standard libraries (numpy, time, math).  No filesystem or network I/O is
performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Return 2·n·max(conv(a,a))  (assumes sum(a)=1).
    The returned value is the numerator of `evaluate_sequence`.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)

def _make_seeds(
    n: int,
    rng: np.random.Generator,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. Half‑sine window (smooth start)
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # 3. Geometric families (both very steep and very flat)
    for r in np.concatenate([np.linspace(0.05, 0.3, 5), np.linspace(0.7, 0.99, 5)]):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # 4. Power‑law families – extend the range towards extreme decay
    for p in np.concatenate([np.linspace(0.0, 2.0, 5), np.linspace(2.5, 5.0, 4)]):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # 5. Beta‑like shapes – include very small alphas/betas (U‑shapes)
    alphas = np.concatenate([np.linspace(0.05, 0.3, 4), np.linspace(0.5, 3.0, 5)])
    betas = np.concatenate([np.linspace(0.05, 0.3, 4), np.linspace(0.5, 3.0, 5)])
    for alpha in alphas:
        for beta in betas:
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 6. Dirichlet draws – include ultra‑sparse concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 1.0, 3.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # 7. Log‑normal bells (different scales)
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 8. Random piecewise‑linear shapes (more control points)
    for _ in range(5):
        K = max(5, int(np.sqrt(n) * rng.uniform(0.8, 1.5)))
        knots = np.linspace(0, n - 1, K)
        heights = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, heights)
        seeds.append(a / a.sum())

    # 9. Warm‑start (if supplied)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

def _interleaved_permutation(a: np.ndarray) -> np.ndarray:
    """Place the largest coefficients alternatingly at the two ends."""
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]  # descending
    out = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out

def _adam_optimize(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Parameterisation: a = softmax(v).  The temperature β is increased
    multiplicatively each iteration.
    """
    v = np.log(a0 + 1e-30)                # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0           # starting smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    while time.time() < deadline:
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        # full convolution and smooth‑max weights
        conv = _conv_full(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            beta *= beta_mul
            continue
        w /= w_sum

        # gradient of weighted sum of convolutions w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]   # central block

        # softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # evaluate true objective occasionally
        if t % 2000 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] it={t:,d} best={cur_val:.6f}")

        beta *= beta_mul

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (mirror‑descent style)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] it={it:,d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step with simple line‑search (three candidates)
        base_lr = 0.6 / np.sqrt(it)
        cand_lrs = (base_lr, 0.5 * base_lr, 1.2 * base_lr)
        best_candidate = a
        best_candidate_val = cur_val

        for lr in cand_lrs:
            log_a = np.log(a + 1e-30) - lr * g
            log_a -= np.max(log_a)
            a_c = np.exp(log_a)
            a_c /= a_c.sum()
            val_c = _objective(a_c, n)
            if val_c < best_candidate_val:
                best_candidate_val = val_c
                best_candidate = a_c

        a = best_candidate
        # keep running best for the outer loop
        if best_candidate_val < best_val:
            best_val = best_candidate_val
            best_seq = a.copy()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖²  (μ = 1/(2n‑1)) using exponentiated‑gradient.
    Drives the auto‑convolution towards a constant.
    """
    a = a0.copy()
    a /= a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        if it % 2500 == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[L2Flat] it={it:,d} best={cur_val:.6f}")

    return best_seq, best_val


def _annealing_hill_climb(
    a0: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing variant of hill‑climbing.
    Moves a tiny amount of mass from a “donor” (chosen proportionally
    to its contribution to the current maximal convolution entry) to a
    random “receiver”.  Worse moves are accepted with probability
    exp(–Δ / T) where the temperature T decays linearly to zero.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    start = time.time()
    step_scale = 0.02

    while time.time() < deadline:
        # linear temperature schedule (starts at ~0.5 * current best)
        elapsed = time.time() - start
        total = deadline - start
        T = max(1e-9, (1.0 - elapsed / total)) * best_val * 0.5

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_val = _objective(new, n)
        delta_val = new_val - cur_val

        # accept if better, otherwise with probability exp(-Δ/T)
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

        # slowly shrink the step size
        if rng.random() < 0.001:
            step_scale = max(step_scale * 0.9, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the supplied time budget
    (≈1000 s) and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length – a larger n gives a lower theoretical bound
    # ------------------------------------------------------------------
    n = 8000  # still comfortably inside the time budget

    # ------------------------------------------------------------------
    # Build a very diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # occasionally try the interleaved permutation on the best seed
    perm = _interleaved_permutation(global_best_seq)
    perm_val = _objective(perm, n)
    if perm_val < global_best_val:
        global_best_val = perm_val
        global_best_seq = perm.copy()
        print(f"[Seed‑perm] improved to {perm_val:.6f}")

    # ------------------------------------------------------------------
    # Phase allocation helper
    # ------------------------------------------------------------------
    def allocate(pct: float) -> float:
        """Return a new deadline consuming `pct` of the remaining time."""
        remaining = deadline - time.time()
        return time.time() + remaining * pct

    # ------------------------------------------------------------------
    # Phase 1 – Adam (smooth‑max surrogate)
    # ------------------------------------------------------------------
    phase_deadline = allocate(0.35)
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ---------------------------------------------------------------
    phase_deadline = allocate(0.30)
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – L2 flattening (drive towards constant convolution)
    # ---------------------------------------------------------------
    phase_deadline = allocate(0.20)
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # ---------------------------------------------------------------
    # Phase 4 – Simulated annealing hill‑climb (final fine‑tuning)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 1.0
        seq, val = _annealing_hill_climb(global_best_seq, n, phase_deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # One last cheap permutation attempt (often yields a tiny gain)
    # ------------------------------------------------------------------
    perm = _interleaved_permutation(global_best_seq)
    perm_val = _objective(perm, n)
    if perm_val < global_best_val:
        global_best_val = perm_val
        global_best_seq = perm.copy()
        print(f"[Final‑perm] improved to {perm_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
