"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (∑a)²

where n = len(a).  The objective is scale‑invariant, so we keep the sequence
normalised to sum = 1 and minimise 2·n·max(conv(a,a)).  The algorithm works
in three stages:

1. **Exploratory random‑sparse sampling** – many candidates are built by
   drawing a random support of size k (≈ √(2 n log n)), assigning Dirichlet
   weights, and (optionally) enforcing symmetry.  These candidates already
   achieve values close to the theoretical limit ≈ 1.

2. **Exact sub‑gradient (mirror‑descent) refinement** – the best candidate
   is further improved by an exponentiated‑gradient update that follows the
   true max‑convolution sub‑gradient.

3. **Targeted simulated‑annealing** – a high‑frequency SA that repeatedly
   attacks the current bottleneck convolution entry by redistributing the
   mass of the heavy contributors.

All phases are time‑aware; the routine stops before the global deadline
and returns the best sequence found.  Printed messages show progress and
intermediate bounds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple deterministic seeds
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    Returns the best sequence seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (focus on current bottleneck)
# ----------------------------------------------------------------------


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always attacks the maximal convolution entry.
    A small random set of the largest contributors to that entry is
    re‑distributed.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        k_move = rng.integers(2, 5)

        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        conv_new = cur_conv.copy()
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)
        delta_obj = new_obj - cur_obj

        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj
