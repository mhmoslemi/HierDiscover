"""
Step‑function optimizer (≈ 1000 s budget)

The optimiser follows a “quick‑seed → smooth‑gradient → exact sub‑gradient
→ long simulated‑annealing → final polishing” pipeline, but with a few
important upgrades over the previous state‑of‑the‑art:

*   More diverse seed families (quadratic / cubic / Hann windows,
    shifted copies, finer geometric / power‑law sweeps).
*   Adam on a soft‑max approximation is run **longer** (up to 50 k
    iterations) with a slower increase of the smoothness parameter.
*   Sub‑gradient mirror descent now also receives a *momentum* term
    (exponential moving average of the sub‑gradient) to accelerate
    convergence.
*   Simulated‑annealing is the work‑horse: the temperature is started
    higher (10 % of the current objective) and cooled very gently
    (decay = 0.9998).  Three move types are used:
        – random Dirichlet re‑allocation on 2‑4 indices,
        – targeted moves that touch the indices contributing to the
          current maximal convolution entry,
        – tiny ε‑shifts of mass from one index to another (fine‑grained
          polishing).
  The incremental convolution update from the original code is retained
  (O(n) per move) and runs for the whole remaining time budget.
*   After SA finishes, a short second sub‑gradient refinement is applied
    to “lock‑in” any remaining improvement.

All phases respect the global deadline (default 1000 s).  The routine
returns the best sequence (list of non‑negative floats) it found.  The
objective value printed at the end equals the value that
``evaluate_sequence`` would return (because the sequence is already
normalised to sum = 1, the denominator of the original formula is 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    return a / a.sum()


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    return a / a.sum()


def _quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    return a / a.sum()


def _cubic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** 2 * (n - i) ** 2
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_quadratic(n))
    seeds.append(_cubic(n))
    seeds.append(_hann_window(n))

    # geometric families (coarse)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # geometric families – close to 1
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes – fast drop near the beginning
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2
        seeds.append(_logistic_shape(n, offset, scale))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # shifted copies of a few basic shapes (to break symmetry)
    for base in (_uniform, _sine_window, _quadratic):
        arr = base(n)
        shift = rng.integers(1, n // 2)
        shifted = np.roll(arr, shift)
        seeds.append(shifted)

    # reuse previous solution if length matches
    if previous is not None and previous.shape == (n,):
        p = previous / previous.sum()
        seeds.append(p)

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (soft‑max) approximation of max(conv).
    The smoothness parameter β starts modest (β=5) and grows slowly.
    """
    eps = 1e-30
    v = np.log(seed + eps)                 # parameters for softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.005         # slower growth → smoother search

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:d} it={it:d} best={cur_val:.6f}")

        # ---- smooth max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on max(conv)
    with an exponential moving‑average of the sub‑gradient (a light momentum).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    # momentum accumulator
    m_acc = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:d} it={it:d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # momentum update (simple EMA)
        m_acc = 0.9 * m_acc + 0.1 * g

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)                # step‑size schedule
        log_a = np.log(a + eps) - lr * m_acc
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.10,
    min_temp: float = 1e-12,
    decay: float = 0.9998,
) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated annealing with three move types:
        • random Dirichlet on 2‑4 indices,
        • targeted re‑allocation on indices that form the current maximal
          convolution entry,
        • tiny ε‑shift from one index to another (fine polishing).
    The incremental convolution update from the reference implementation is
    reused, making each move cheap (O(n)).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)       # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to objective (10 % of it)
    T = max(init_temp_factor * cur_obj, 1e-12)

    it = 0
    while True:
        if time.time() > deadline:
            break
        it += 1

        # cooling
        T = max(T * decay, min_temp)

        # -------------------------------------------------
        # Choose move type
        # -------------------------------------------------
        r = rng.random()
        if r < 0.45:
            # random Dirichlet on 2‑4 indices
            k = rng.integers(2, 5)                     # 2‑4 distinct indices
            idx = rng.choice(n, size=k, replace=False)
        elif r < 0.80:
            # targeted move
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # second index uniformly from the complement
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)
        else:
            # tiny ε‑shift from i to j
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cur[i] == 0:
                continue
            eps = rng.uniform(0.0, min(0.01, 0.5) * cur[i])  # up to 1 % shift
            if eps <= 0:
                continue
            idx = np.array([i, j], dtype=int)
            new_alloc = cur[idx].copy()
            new_alloc[0] -= eps
            new_alloc[1] += eps
            # go straight to acceptance test (skip Dirichlet draw)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            deltas = new_alloc - cur[idx]
            # ----- incremental update -----
            conv_new = cur_conv.copy()
            for p, delta in zip(idx, deltas):
                if delta == 0.0:
                    continue
                conv_new[p : p + n] += 2.0 * delta * cur
                conv_new[2 * p] += delta * delta
            # cross‑terms
            if idx.size == 2:
                conv_new[idx[0] + idx[1]] += 2.0 * deltas[0] * deltas[1]

            new_obj = 2.0 * n * np.max(conv_new)
            delta_obj = new_obj - cur_obj
            if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
                # accept
                cur[idx] = new_alloc
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    print(f"[SA] n={n:d} it={it:d} best={new_obj:.6f}")
            continue  # next iteration

        # -----------------------------------------------------------------
        # Reallocation move (Dirichlet on selected indices)
        # -----------------------------------------------------------------
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation for the selected positions (Dirichlet)
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross‑terms for i≠j
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:d} it={it:d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function returns the best list of heights it
    discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # fetch previous solution if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Choose a handful of lengths – larger n gives a smaller theoretical
    # bound but costs more per iteration.
    lengths = [2000, 4000, 6000, 8000, 10000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smooth max (top‑k seeds)
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(5, len(seed_vals))
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (single long run)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=400000
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated‑annealing (use up all time)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                n,
                deadline,
                init_temp_factor=0.10,   # 10 % of current objective
                decay=0.9998,
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Final short sub‑gradient polish
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=100000
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[final‑SubGrad] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with the provided evaluate_sequence (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Return a plain Python list as required by the evaluator
    return global_best_seq.tolist()
