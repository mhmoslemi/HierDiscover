The goal is to return a non‑negative step‑function `a = (a0,…,a_{n‑1})` that minimises  

```
E(a) = 2·n·max(conv(a,a)) / (∑a)^2                (1)
```

`evaluate_sequence` implements (1) with a few safety checks.  
The objective is *scale invariant* – multiplying the whole vector by a constant does not change the score – therefore we may always normalise the vector so that `∑a = 1`.  

The optimisation problem is a non‑convex quadratic max‑problem.  The
previous submission already combines:

* many analytically motivated **seed families**,
* a **smooth‑max Adam** optimiser,
* a **mirror‑descent sub‑gradient** refinement,
* a **targeted simulated annealing** (SA) phase that attacks the
  current bottleneck entry of the convolution.

That pipeline already reaches `E ≈ 1.5115`.  To push the bound further we
add three ideas

1. **Richer seeds** – monotone, truncated geometric, two‑block and
   half‑uniform shapes, plus random convex combinations of two seeds.
   These families explore highly asymmetric distributions that are very
   effective at breaking the large overlap between the sequence and its
   reversal.

2. **Longer problem size** – we also optimise `n = 8000`.  A larger
   dimension gives the optimiser more freedom; the best shape for a
   larger `n` can be strictly better than the best shape for `n ≤ 6000`.

3. **More aggressive SA** – the SA implementation is rewritten to
   *update the convolution in‑place* (no full copy per move) and to
   allow up to eight moved indices per step.  This makes it cheap enough
   to run many more iterations in the remaining time budget.

All components are written as plain functions (no closures, no lambdas)
and respect the global deadline (`budget_s` seconds).  The routine
returns the best sequence found before the deadline.

```python
#!/usr/bin/env python
"""
propose_candidate – minimise 2·n·max(conv(a,a))/(∑a)²

The algorithm:
* builds many diverse seeds (uniform, geometric, power‑law, beta,
  sine‑window, truncated geometric, monotone linear, half‑uniform,
  two‑block, random convex mixes)
* evaluates the seeds and keeps the best one
* refines the best seed with
    – Adam optimisation on a log‑sum‑exp (smooth‑max) surrogate,
    – exact sub‑gradient mirror descent on the true max,
    – a fast, in‑place, targeted simulated‑annealing that repeatedly
      redistributes mass from the heaviest contributors of the current
      bottleneck convolution entry.
* repeats the above for several lengths (2000, 4000, 6000, 8000) and
  spends any remaining time on a final SA on the globally best vector.
All time‑checks use the supplied deadline; the function returns the
best sequence before the budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (monotone decreasing)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_trunc(n: int, r: float, cut: int) -> np.ndarray:
    """Geometric decay truncated after `cut` entries (rest = 0)."""
    a = np.zeros(n, dtype=np.float64)
    lim = min(cut, n)
    a[:lim] = r ** np.arange(lim, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape:
       a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1},  α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    """Monotonically decreasing linear weights (largest at the start)."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _half_uniform(n: int) -> np.ndarray:
    """Uniform on the first half, zero on the second half."""
    half = n // 2
    a = np.concatenate([np.full(half, 1.0 / half), np.zeros(n - half)])
    return a / a.sum()


def _two_block(n: int, frac: float, ratio: float) -> np.ndarray:
    """
    Two‑block shape:
        first `frac`·n entries receive weight `ratio`,
        the remaining entries receive weight 1.
    After normalisation the vector sums to 1.
    """
    k = max(1, int(frac * n))
    a = np.empty(n, dtype=np.float64)
    a[:k] = ratio
    a[k:] = 1.0
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _mix_two(seeds: list[np.ndarray], rng: np.random.Generator) -> np.ndarray:
    """Random convex combination of two distinct seeds."""
    i, j = rng.choice(len(seeds), size=2, replace=False)
    w = rng.random()
    a = w * seeds[i] + (1.0 - w) * seeds[j]
    a /= a.sum()
    return a


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Create a diverse collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_linear_decrease(n))
    seeds.append(_half_uniform(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # truncated geometric families (different cut‑offs)
    for r in (0.3, 0.5, 0.7, 0.9):
        for cut_frac in (0.25, 0.5, 0.75):
            cut = int(cut_frac * n)
            seeds.append(_geom_trunc(n, r, cut))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 8)
    betas = np.linspace(0.30, 1.20, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # two‑block families (different fractions / ratios)
    for frac in (0.10, 0.20, 0.30, 0.40):
        for ratio in (5.0, 10.0, 20.0):
            seeds.append(_two_block(n, frac, ratio))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8, 12):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # random mixes of the above
    for _ in range(5):
        seeds.append(_mix_two(seeds, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) surrogate.
    Returns best sequence seen and its objective value.
    """
    eps = 1e-30
    v = np.log(seed + eps)               # work in log‑space → positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing: start modest, increase exponentially
    beta_smooth = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # stabilise exponent
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2, keep central n entries
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    Returns the best sequence and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Fast targeted simulated‑annealing.
    The convolution is updated *in‑place* (no full copy per move), allowing
    many millions of cheap iterations.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    # reusable array for incremental updates
    diff = np.zeros_like(cur_conv)

    while time.time() < deadline:
        # cooling
        T = max(T * decay, min_temp)

        # locate the bottleneck index
        k_star = int(np.argmax(cur_conv))

        # indices that truly contribute to the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few of the heaviest contributors (max 8)
        top_k = min(8, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to move (3‑6)
        k_move = rng.integers(3, 7)

        # choose indices: heavy ones first, fill up with random others
        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass among the chosen positions
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update (in‑place) -----
        diff.fill(0.0)                         # reset

        # a·Δ + Δ·a contributions
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            diff[p:p + n] += 2.0 * delta * cur   # old cur (before the move)
            diff[2 * p] += delta * delta          # Δ·Δ self term

        # cross Δ_i·Δ_j for i≠j (appears twice)
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                diff[pos] += 2.0 * deltas[i] * deltas[j]

        new_conv = cur_conv + diff
        new_obj = 2.0 * n * np.max(new_conv)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = new_conv
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} best={new_obj:.6f}")

    return best_seq, best_obj

        # ------------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the most promising seeds (top‑4)
        # ------------------------------------------------------------------
        top_idx = np.argsort(seed_vals)[:4]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            # allow a few more iterations for the larger lengths
            max_adam_iters = 25000 if n >= 6000 else 20000
            seq, val = _adam_optimize(s, n, deadline, max_iters=max_adam_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and len(global_best_seq) == n
        ):
            # allocate more iterations for the largest problem
            sub_max_iters = 500000 if n >= 6000 else 300000
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=sub_max_iters
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Targeted simulated‑annealing (high‑frequency)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and len(global_best_seq) == n
        ):
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

        # keep the best sequence for the next length (may be longer)
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Final‐stage refinement on the globally best vector (if any time left)
    # ----------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        n = len(global_best_seq)
        print("\n=== Final refinement (long SA) ===")
        seq, val = _targeted_sa_optimize(global_best_seq, n, deadline)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[final‑TargetSA] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity‑check with the provided evaluator
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
```
