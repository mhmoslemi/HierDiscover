def _sa_optimize(start_seq, n, deadline):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)
    best_seq = cur.copy()
    best_obj = cur_obj
    # init temperature as fraction of cur_obj
    T = max(0.1 * cur_obj, 1e-12)
    decay = 0.99995 # slower decay

    while time.time() < deadline:
        # optional random cooling reset? Not needed

        # pick move type
        if rng.random() < 0.6:
            # random multi-index reallocation
            k = rng.integers(2, 7)  # 2-6 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # targeted
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)
            # choose additional 1-3 indices randomly (including i)
            extra = rng.integers(0, n, size=rng.integers(1,4), endpoint=False)
            idx = np.unique(np.concatenate(([i], extra)))
        total = cur[idx].sum()
        if total <= 0:
            continue
        # allocate new values via Dirichlet distribution weighted by current values?
        # Use Dirichlet to conserve total:
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0):
            continue
        # incremental conv update
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0:
                continue
            # update shift contributions
            conv_new[p:p+n] += 2.0 * delta * cur
            conv_new[2*p] += delta * delta
        # cross terms
        for i1 in range(idx.size):
            for i2 in range(i1+1, idx.size):
                pos = idx[i1] + idx[i2]
                conv_new[pos] += 2.0 * deltas[i1] * deltas[i2]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:

                best_seq = cur.copy()
                print(f"[SA] n={n} best={new_obj:.6f}")
        # cooling
        T = max(T * decay, 1e-12)
    return best_seq, best_obj

"""
Improved optimiser for the “max‑auto‑correlation” step‑function problem.

Main ideas
~~~~~~~~~~
1. **Broad seed set** – adds Gaussian‑shaped seeds and a few extra
   parametric families to the original diverse collection.
2. **Long Adam phase** – a smooth‑max (log‑sum‑exp) optimiser with a
   larger iteration budget, applied to the best few seeds.
3. **Projected sub‑gradient descent** – uses the exact sub‑gradient of
   `max(conv(a,a))` and projects onto the simplex (mirror‑descent).
4. **Very long Simulated‑Annealing** – runs until the time budget is
   exhausted.  The move set includes multi‑index Dirichlet reallocations
   and a *targeted* move that concentrates on the indices participating
   in the current maximal convolution entry.  Temperature decays slowly
   (`0.99995`) and the initial temperature is a generous fraction of the
   current objective, giving the SA phase many chances to improve the
   solution.

The optimiser focuses all remaining time on a single length
(`n = 4000` by default) – this yields more SA moves than cycling over
several lengths.  The final best sequence (normalised to sum = 1) is
returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# Seed families (already normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float, mu: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw (conc < 1 → very sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""

    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2
        seeds.append(_logistic_shape(n, offset, scale))

    # Gaussian families (different widths and centres)
    for sigma in (n / 10, n / 6, n / 4):
        for mu in (n / 4, n / 2, 3 * n / 4):
            seeds.append(_gaussian(n, sigma, mu))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters of a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothing, grows geometrically
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:d} best={cur_val:.6f}")

        # ---------- smooth max ----------
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  (conv = a*a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Long simulated‑annealing loop (runs until the time budget expires).
    Two move kinds are used:

    * Random Dirichlet re‑allocation on 2‑6 randomly chosen indices.
    * Targeted re‑allocation that touches indices contributing to the
      current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)            # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Initial temperature – a noticeable fraction of the current objective.
    T = max(0.02 * cur_obj, 1e-12)
    decay = 0.99995                           # very slow cooling

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- choose move type -----
        if rng.random() < 0.6:
            # random multi‑index Dirichlet move
            k = rng.integers(2, 7)                # 2‑6 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # targeted move – focus on the maximal convolution entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # weight each index by its contribution to the current max
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # pick a second (or third) random partner – we always keep the
            # targeted index in the move set
            extra = rng.integers(0, n, size=rng.integers(1, 4), endpoint=False)
            idx = np.unique(np.concatenate(([i], extra)))

        # ----- propose re‑allocation -----
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a  (commutative)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ self‑term
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j (appear twice)
        for i1 in range(idx.size):
            for i2 in range(i1 + 1, idx.size):
                pos = idx[i1] + idx[i2]
                conv_new[pos] += 2.0 * deltas[i1] * deltas[i2]

        new_obj = 2.0 * n * np.max(conv_new)

        # ----- Metropolis acceptance -----
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] it={it:d} best={new_obj:.6f}")

        # ----- temperature decay -----
        T = max(T * decay, 1e-12)

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  Returns the best sequence found as
    a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final conversion / return
    deadline = start_time + budget_s - 1.5

    # try to reuse a previously‑found construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a single length – larger gives finer discretisation,
    # but we keep n moderate to allow many SA moves.
    # ------------------------------------------------------------------
    n = 4000

    print(f"\n=== Optimising length {n} ===")
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # Phase 0 – evaluate raw seeds (cheap)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a few promising seeds
    # ------------------------------------------------------------------
    # Sort seeds by their raw objective and keep the top‑k
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda t: t[1])
    top_k = min(3, len(seed_vals))
    for s, _ in seed_vals[:top_k]:
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑Adam] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=600000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SubGrad] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long simulated‑annealing (runs until time runs out)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _sa_optimize(global_best_seq, n, deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SA] final best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional verification with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # Return a plain Python list as required by the arena
    return global_best_seq.tolist()
