"""
genetic‑annealing optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Coarse global search** – a small‑scale genetic algorithm
   (population of Dirichlet samples plus a few analytic seeds)
   is run on a modest length (n≈2000).  The GA rapidly explores the
   non‑convex landscape and returns a promising shape.

2. **Upscaling & smooth‑max descent** – the best coarse shape is
   interpolated to a larger length (default n=8000).  Starting from
   this point we run an Adam optimiser on a smoothed‐maximum
   surrogate (log‑sum‑exp of the auto‑convolution) with a slowly
   increasing temperature.  This phase drives the maximum of the
   convolution down dramatically.

3. **Exact sub‑gradient mirror descent + short annealing** –
   using exponentiated‑gradient updates we minimise the true
   ‑∞‑norm of the convolution.  A final simulated‑annealing
   hill‑climber refines the last few digits.

All time is bounded by the supplied budget (default 1000 s).  The
function returns the best sequence (as a plain list of floats) that
fits the constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT, objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for a normalised sequence (sum == 1):
        2·n·max(conv(seq,seq))
    This equals the value returned by ``evaluate_sequence`` after
    normalisation, but avoids the extra sum‑square division.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised, sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _inverse_sqrt(n: int, power: float = 0.5) -> np.ndarray:
    """Heavy tails: a_i ∝ 1 / sqrt(i·(n‑i+1))^power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (np.sqrt(i * (n - i + 1)) ** power)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Genetic algorithm (coarse search on a small problem)
# ----------------------------------------------------------------------


def _ga_run(
    n: int,
    pop_size: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Simple GA on the surrogate objective.
    Returns the best normalised vector found.
    """
    # ----- initial population -------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _sine_window(n),
        _gaussian(n, 0.2),
        _gaussian(n, 0.4),
        _gaussian(n, 0.6),
        _inverse_sqrt(n, 0.2),
        _inverse_sqrt(n, 0.5),
        _inverse_sqrt(n, 0.8),
    ]

    # add a few random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # fill up to pop_size with Dirichlet samples
    while len(seeds) < pop_size:
        seeds.append(_random_dirichlet(n, 0.3, rng))

    population = np.stack(seeds[:pop_size])
    best_seq = None
    best_val = float("inf")
    generation = 0
    elite_frac = 0.15   # fraction kept unchanged
    mutate_rate = 0.02  # how much fresh Dirichlet noise is injected

    # ----- evolution loop ----------------------------------------------------
    while time.time() < deadline:
        generation += 1

        # evaluate
        vals = np.apply_along_axis(_objective, 1, population)
        # keep elite
        elite_num = max(1, int(elite_frac * pop_size))
        elite_idx = np.argpartition(vals, elite_num - 1)[:elite_num]
        elite = population[elite_idx]
        elite_vals = vals[elite_idx]

        # record global best
        gen_best_idx = elite_idx[np.argmin(elite_vals)]
        gen_best_val = vals[gen_best_idx]
        if gen_best_val < best_val:
            best_val = gen_best_val
            best_seq = population[gen_best_idx].copy()
            print(f"[GA] gen {generation:4d} new best = {best_val:.6f}")

        # create offspring
        new_pop = [elite[i % elite_num] for i in range(elite_num)]  # keep elites
        while len(new_pop) < pop_size:
            # select two parents uniformly from elite set
            p1, p2 = elite[rng.integers(elite_num, size=2)]
            alpha = rng.random()
            child = alpha * p1 + (1.0 - alpha) * p2
            # mutation: blend a small amount of fresh Dirichlet noise
            if rng.random() < mutate_rate:
                noise = _random_dirichlet(n, 0.5, rng)
                child = 0.9 * child + 0.1 * noise
            child /= child.sum()  # re‑normalise to sum==1
            new_pop.append(child)

        population = np.stack(new_pop)

        # stop early if we are close to the deadline
        if time.time() > deadline - 2.0:
            break

    return best_seq


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp smoothed maximum of the
    auto‑convolution.  The temperature ``beta`` is increased slowly.
    Returns the best sequence encountered and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    n = len(start_vec)
    beta = 0.5
    beta_mul = 1.001
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = np.exp(v - np.max(v))
        a /= a.sum()

        # surrogate objective (smooth max)
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        w /= w_sum
        cur_val = 2.0 * n * np.sum(w * conv)     # smooth max value

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} beta={beta:.2f} best={best_val:.6f}")

        # increase temperature
        beta = min(beta * beta_mul, beta_max)

        # gradient of the smooth objective
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # bring gradient to log‑softmax space
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)

        # clip for stability
        norm = np.linalg.norm(grad_v)
        if norm > 10.0:
            grad_v = grad_v * (10.0 / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Minimises the true max‑convolution using exponentiated‑gradient
    updates.  The gradient is a weighted sum of gradients at the
    indices that are close to the maximum (soft‑max with large beta).
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    beta = 250.0                     # fixed high temperature → near‑max

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # soft‑max weighting – almost all weight on the true max but
        # stable for ties.
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of weighted sum of convolutions
        full = _conv_fft(a[::-1], w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)          # decreasing step size
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Random Dirichlet reshuffles of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(0.02 * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, 1e-12)

        k = rng.integers(2, 7)                 # 2 … 6 indices
        idx = rng.choice(len(cur), size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the time budget and returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the evaluator
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Coarse genetic search on a modest problem size.
    # ------------------------------------------------------------------
    n_small = 2000
    ga_deadline = start_time + budget_s * 0.30   # ~30 % of total time
    print(f"\n--- GA on n={n_small} (deadline {ga_deadline - start_time:.1f}s) ---")
    best_small = _ga_run(n_small, pop_size=80, rng=rng, deadline=ga_deadline)
    print(f"[GA] best surrogate on size {n_small} = {_objective(best_small):.6f}")

    # ------------------------------------------------------------------
    # 2. Upscale to a larger length and perform smooth‑max descent.
    # ------------------------------------------------------------------
    n_large = 8000
    print(f"\n--- Upscaling to n={n_large} ---")
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    best_large = np.interp(x_large, x_small, best_small)
    best_large = np.maximum(best_large, 0.0)
    best_large /= best_large.sum()

    # Adam phase (remaining budget ≈ 60 %)
    adam_deadline = start_time + budget_s * 0.80
    print(f"\n--- Adam optimisation (deadline {adam_deadline - start_time:.1f}s) ---")
    seq_adam, val_adam = _adam_optimize(best_large, deadline=adam_deadline)
    if seq_adam is None:
        seq_adam = best_large
        val_adam = _objective(best_large)
    print(f"[Adam] final surrogate = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient refinement (use whatever time is left).
    # ------------------------------------------------------------------
    subgrad_deadline = deadline
    print(f"\n--- Sub‑gradient refinement (deadline {subgrad_deadline - start_time:.1f}s) ---")
    seq_sub, val_sub = _subgradient_refine(seq_adam, deadline=subgrad_deadline)
    print(f"[SubGrad] final surrogate = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional simulated‑annealing polishing (only if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(seq_sub, deadline=deadline)
        if val_sa < val_sub:
            seq_sub = seq_sa
            val_sub = val_sa
            print(f"[SA] final surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the true evaluator.
    # ------------------------------------------------------------------
    try:
        # the evaluator will internally renormalise if necessary;
        # we pass a plain list of floats.
        evaluate_sequence(seq_sub.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {val_sub:.6f} (length {len(seq_sub)})"
    )

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = seq_sub.tolist()
    return seq_sub.tolist()
