"""
LP‑guided search for a non‑negative step function with a low
evaluate_sequence() value.

The algorithm starts from the best known sequence (if present) or from a
uniform distribution (∑a = 1).  In each outer iteration we

1. evaluate M = max(conv(f,f)).
2. solve the linear program   maximise   1ᵀb
   subject to  conv(f,b) ≤ M   and   b ≥ 0.
   The matrix that maps b → conv(f,b) is built as a sparse Toeplitz
   matrix using ``scipy.sparse.diags``.
3. scale the LP solution to the same ℓ₁‑norm as f, obtain g,
   and replace f by a convex combination (1‑t)f + t g.
   The mixing weight t is increased when the move is successful,
   otherwise it is decreased.
4. keep the best sequence seen so far (according to the exact
   evaluate_sequence objective).

After a few LP steps we run a cheap exponentiated‑gradient sub‑gradient
refinement to polish the result and finally a short simulated‑annealing
hill‑climber for a last push.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns the best list of floats found.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective: 2·n·max(conv(seq,seq)) .
    Assumes seq is non‑negative and ∑seq = 1 (otherwise the true
    evaluate_sequence value is 2·n·max(conv)/∑seq²).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Linear‑programming primitive (maximise sum of coefficients
# while keeping the convolution under the current peak)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A such that A @ b = conv(f,b) (full linear
    convolution).  Shape of A is (2·n‑1, n).  The matrix is Toeplitz
    with constant diagonals equal to the entries of f.
    """
    n = f.shape[0]
    # each diagonal k = –d (d = 0…n‑1) contains the constant value f[d]
    offsets = -np.arange(n, dtype=int)                # 0, -1, …, -(n‑1)
    # data is an (n, n) array where row i is filled with f[i]
    data = np.tile(f[:, None], (1, n))                # shape (n, n)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_max_sum(f: np.ndarray, M: float, time_left: float):
    """
    Solve  maximise 1ᵀb  subject to  conv(f,b) ≤ M,  b ≥ 0.
    Returns the optimal b (or None on failure) and the solver status.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    # right‑hand side of the inequality is the same M for every row
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise 1ᵀb  →  minimise –1ᵀb
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # SciPy's HiGHS solver honours a time limit via the generic options
    # dictionary (available from SciPy 1.6).  We set a generous limit; the
    # outer code will abort the whole run if the global deadline is hit.
    options = {"presolve": True, "time_limit": max(0.5, time_left)}

    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method="highs", options=options)
    if not res.success:
        return None, res.message
    return res.x, "ok"


# ----------------------------------------------------------------------
# Fine‑grained exponentiated‑gradient refinement (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  Operates directly on the simplex to keep the sequence
    non‑negative with ℓ₁‑norm = 1.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq


# ----------------------------------------------------------------------
# Light simulated‑annealing hill‑climber (optional final polish)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000) -> np.ndarray:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(0.02 * cur_val, 1e-12)   # initial temperature

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, 1e-12)

        # pick a small random subset and reshuffle its mass
        k = rng.integers(2, 7)                     # 2 … 6 indices
        idx = rng.choice(len(cur), size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.ones(k)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list
    of floats.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Load previous solution (if any) and enforce the desired length.
    # ------------------------------------------------------------------
    n = 4000                                 # length of the discretisation
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is not None and prev_arr.shape == (n,):
        # normalise to ℓ₁‑norm 1 (objective is scale‑invariant)
        total = prev_arr.sum()
        if total > 0:
            f = prev_arr / total
        else:
            f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # fall back to a uniform start
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # bookkeeping of the best sequence found so far
    best_seq = f.copy()
    best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Main LP‑guided loop
    # ------------------------------------------------------------------
    mix_t = 0.3               # initial mixing weight
    min_t = 0.04
    max_t = 0.7
    lp_iter = 0

    while time.time() < deadline and lp_iter < 20:
        lp_iter += 1
        # current peak of the autocorrelation
        conv_f = _conv_fft(f, f)
        M = np.max(conv_f)

        # remaining time for the LP solve
        time_left = deadline - time.time()
        if time_left < 1.0:
            break

        b, status = _lp_max_sum(f, M, time_left)
        if b is None:
            print(f"[LP {lp_iter}] solver failed: {status}")
            break

        # scale b to the same ℓ₁‑norm as f (which is 1)
        b_sum = b.sum()
        if b_sum == 0.0:
            print(f"[LP {lp_iter}] zero‑sum LP solution – abort")
            break
        g = b / b_sum

        # convex combination
        cand = (1.0 - mix_t) * f + mix_t * g
        cand = np.maximum(cand, 0.0)          # numerical safety
        cand /= cand.sum()                    # keep ℓ₁ = 1

        cand_val = _objective(cand)

        print(f"[LP {lp_iter:02d}] obj={cand_val:.6f}  mix_t={mix_t:.3f}")

        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            # when we make progress we are a little more aggressive
            mix_t = min(mix_t * 1.1, max_t)
        else:
            # otherwise be more cautious
            mix_t = max(mix_t * 0.7, min_t)

        f = cand  # continue iterations from the new point

    # ------------------------------------------------------------------
    # Short exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined = _subgradient_refine(best_seq, deadline, max_iters=250_000)
        refined_val = _objective(refined)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined.copy()
            print(f"[Refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing (cheap, stochastic)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final = _simulated_annealing(best_seq, deadline, max_iters=150_000)
        final_val = _objective(final)
        if final_val < best_val:
            best_val = final_val
            best_seq = final.copy()
            print(f"[SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Convert to the exact format expected by the judge and store
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    # sanity check (the actual benchmark function may apply its own caps)
    try:
        evaluate_sequence(result)  # pylint: disable=undefined-variable
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    print(f"\n=== DONE ===\nBest surrogate objective ≈ {best_val:.6f}")
    globals()["height_sequence_1"] = result
    return result
