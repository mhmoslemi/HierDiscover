def _proj_simplex(v):
    # from Duchi et al (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_proj(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # gradient (subgradient) of the max index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))

        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # update step size
        lr = 0.1 / np.sqrt(it)  # could start larger
        a = a - lr * g
        # project onto simplex
        a = _proj_simplex(a)
        # enforce non-neg (proj ensures)
        # optionally ensure at least small epsilon to avoid zero (but proj may produce zeros)
        # Python keep as float
    return best_a, best_val

"""
Propose a step‑function with a low value of

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm consists of three stages:

1. **Diverse seeds** – a large collection of analytically‑motivated families
   (uniform, sine‑window, Tukey, Gaussian, geometric, power‑law, beta)
   together with a few random Dirichlet draws.  All candidates are normalised
   to ℓ₁‑norm = 1 and the best few (according to the cheap surrogate
   ``2·n·max(conv)``) are kept.

2. **Heavy continuous optimisation** – each selected seed is refined by
   *Adam* on a smoothed‑max surrogate (log‑sum‑exp) followed by an *exact
   sub‑gradient* phase that works directly on the true max‑convolution.
   The sub‑gradient step is an *additive* gradient descent with a Euclidean
   projection onto the probability simplex (the faster “mirror‑descent”
   variant of the original code).  Both parts run until the global time
   budget is exhausted.

3. **Final stochastic polishing** – a simple simulated‑annealing on a
   random Dirichlet reshuffle of a handful of coordinates, to escape shallow
   minima.

All stages respect the overall 1000 s limit.  The routine returns a plain
list of floats ready for ``evaluate_sequence`` and stores the best vector in
the global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)) (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ------------------------------------------------------------
# Analytic shape families (already normalised to Σ = 1)
# ------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Seed collection

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 10):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 15):
        seeds.append(_power(n, p))

    # Beta family
    alphas = np.linspace(0.3, 2.0, 6)
    betas = np.linspace(0.3, 4.0, 6)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.15, 0.25, 0.35, 0.45):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits (stay on simplex via softmax)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.1            # start temperature (very smooth)
    beta_mul = 1.005
    beta_max = 300.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probabilities, sum = 1
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple clipping for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.7
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses an additive update with projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # additive step (decrease direction) and simplex projection
        lr = 0.2 / np.sqrt(it)      # a modest step; can be tuned
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 12)          # 2 … 11 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    # deterministic reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length – 4000 works well (good trade‑off between quality
    # and speed).  The algorithm works for any length.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and keep the most promising.
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds for length {n} ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 3                     # a handful of good seeds for heavy work
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam + exact projected sub‑gradient)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # Adam phase – runs until the global deadline is reached
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=1_000_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Exact projected sub‑gradient phase – also runs until deadline
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
