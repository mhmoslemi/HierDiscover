"""
Propose a step‑function (non‑negative vector) that (approximately) minimises

    2·n·max(convolve(a,a)) / (sum(a)²)

The algorithm is a 3‑stage search:

1. **Rich seed pool** – many classic windows (Hann, Blackman, Kaiser,
   Slepian/DPSS), geometric / power‑law families and random Dirichlet draws.
   The arcsine‑type (Catalan) seed is also added for completeness.

2. **Smooth‑max L‑BFGS** – the non‑smooth objective is approximated by a
   log‑sum‑exp (soft‑max) with a geometric schedule of the smoothing
   parameter β (β grows from 0.5 to 1e6).  Each seed is optimised with
   L‑BFGS while staying inside the probability simplex (the variables are
   logits fed to a softmax).  The best result over all seeds is kept.

3. **Exact max‑subgradient descent** – starting from the best L‑BFGS
   solution we minimise the true objective with an additive projected‑gradient
   method.  The sub‑gradient of the max‑convolution is computed analytically
   (averaged over all active indices).  Steps are taken with a diminishing
   step‑size  ηₜ = 0.5/√t and each iterate is projected onto the simplex.
   Occasionally a tiny Gaussian perturbation is added to escape possible
   stagnation.

4. **Very light Metropolis polishing** – if time permits a cheap Metropolis
   walk moves a small fraction of mass between two random coordinates.

All stages respect the overall 1000 s budget, keep the current best
objective and finally return the best vector as a plain Python list.

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – a flexible spectral window."""
    if SCIPY_SIG:
        a = windows.kaiser(n, beta)
        return a / a.sum()
    # fallback: use analytic definition based on modified Bessel I0
    from scipy.special import i0
    n_arr = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    a = i0(beta * np.sqrt(1 - ((n_arr - alpha) / alpha) ** 2))
    return a / a.sum()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """First Slepian (DPSS) window for concentration parameter NW."""
    if SCIPY_SIG:
        a = windows.dpss(n, nw, Kmax=1, sym=False)[0]
        a = np.abs(a)          # ensure non‑negative (should already be)
        return a / a.sum()
    # fallback – use Hann as a crude stand‑in
    return _hann_window(n)


def _catalan_trunc(n: int) -> np.ndarray:
    """Truncated Catalan (binom(2i,i)/4^i) sequence, normalised to sum=1."""
    i = np.arange(n, dtype=np.float64)
    # central binomial coefficient / 4^i = comb(2i,i) / 4**i
    # Use recurrence to avoid overflow:
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * (2 * k - 1)) / (k + 1) / 4.0
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))

    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# -------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective & gradient for L‑BFGS
# -------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).
    The gradient is w.r.t. the logits `v`.
    """
    a = _softmax(v)                     # probability vector (sum = 1)

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # outer factor 2·n

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    #   d/da Σₖ wₖ·convₖ(a,a) = 2·(rev(a) * w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                 # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 400) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS optimisation on the smooth‑max surrogate.
    Returns the best probability vector found together with its exact objective.
    """
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)      # initialise logits

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12,
                                         'disp': False})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1e}   best={cur_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------
# Exact max‑subgradient descent (additive projected gradient)
# -------------------------------------------------------------
def _max_subgradient_descent(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent on the true objective.
    Uses Euclidean projection onto the simplex after every step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build weight vector w for averaging over active indices
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        # gradient of the max‑convolution (averaged)
        grad = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

        # diminishing step size – a simple schedule
        step = 0.5 / math.sqrt(it)

        a_next = a - step * grad
        a_next = _project_simplex(a_next)

        cur_val = _objective(a_next)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_next.copy()
            print(f"[SubGrad‑Add] it={it:6d}  best={cur_val:.6f}")

        a = a_next

        # occasional tiny Gaussian perturbation to escape flat regions
        if it % 20000 == 0:
            noise = np.random.default_rng().normal(scale=0.001, size=n)
            a_pert = a + noise
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = _project_simplex(a_pert)
            val_pert = _objective(a_pert)
            if val_pert < best_val:
                best_val = val_pert
                best_seq = a_pert.copy()
                print(f"[Perturb] it={it:6d} new best={val_pert:.6f}")
            a = a_pert

    return best_seq, best_val


# -------------------------------------------------------------
# Tiny Metropolis annealing (final polish)
# -------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Small Metropolis walk that swaps a tiny fraction of mass between two random
    coordinates.  Used only if time is still left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.015 * cur[i]  # ≤ 1.5 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = _project_simplex(new_seq)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# -------------------------------------------------------------
# Main entry point required by the grader
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a few thousand works well and fits comfortably
    # in memory while giving enough degrees of freedom.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load possible previous construction.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])            # low is good
    top_k = 12                                 # a few good starting points
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(0.5, 1e6, num=15)   # gradual sharpening
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=300)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact projected sub‑gradient descent refinement.
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _max_subgradient_descent(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=600_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad‑Add] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=400_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be hit.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will call evaluate_sequence on it)
    return global_best_seq.tolist()
