"""
Improved search for a non‑negative step function with a low objective.

The algorithm consists of three stages:

1. **Initialisation** – start from the best known sequence (if any) or from a
   uniform distribution.  The sequence is kept on the simplex (sum = 1).

2. **Linear‑programming updates** – given the current vector `f` and its
   maximal self‑convolution `M = max(conv(f,f))`,
   solve the linear programme

        maximise   1ᵀb
        subject to  conv(f, b) ≤ M   (for every lag)
                    b ≥ 0

   The solution `b` has the largest ℓ₁‑mass that can be added without
   increasing the convolution bound.  After normalising `b` to unit sum we
   move the current point a small step towards it:
        f ← (1‑t)·f + t·(b/∑b)
   This update never increases the maximal convolution and quickly flattens
   the autocorrelation curve.

3. **Fine‑grained exponentiated‑gradient refinement** – a fast mirror‑descent
   (exponentiated‑gradient) on the true max‑convolution objective finishes the
   optimisation.  The routine stops automatically when the global time budget
   (default 1000 s) is exhausted.

All helpers are top‑level, use only NumPy, SciPy and the supplied
`evaluate_sequence`.  The returned list is ready for the final evaluation.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – assumes Σseq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) refinement
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the
    true max‑convolution objective.  Stops early when the time budget expires.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the max at index max_idx:
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)          # step‑size – larger than the
                                        # default to accelerate convergence
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming update
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2n‑1)×n matrix A such that A @ b = conv(f, b).
    The matrix is built once per iteration – it is extremely sparse
    (n² non‑zero entries) but the construction is cheap for n≈4000.
    """
    n = f.shape[0]
    # column index repeated n times
    col = np.repeat(np.arange(n), n)
    # row index = column + offset (offset runs 0…n‑1 for each column)
    row = col + np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))


def _lp_step(f: np.ndarray, M: float, t: float) -> tuple[np.ndarray, float]:
    """
    One LP‑driven update.
    Returns (new_f, new_M).  If the LP fails the original f is returned.
    """
    n = f.shape[0]

    # build the constraint matrix A_ub @ b ≤ M
    A_ub = _build_convolution_matrix(f)

    # maximise Σb  → minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    # right‑hand side
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # bounds b_i ≥ 0
    bounds = [(0, None)] * n

    # solve the LP (HiGHS is the default and handles sparse matrices)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

    if not res.success:
        # LP failed – fall back to the current point
        return f, M

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return f, M

    # normalise to the simplex
    g = b / sum_b

    # convex combination with step size t (t∈(0,1))
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(0.0, f_new)
    f_new /= f_new.sum()

    # compute the new maximal self‑convolution
    conv_new = _conv_fft(f_new, f_new)
    M_new = np.max(conv_new)

    return f_new, M_new
