"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

where `conv` is a linear convolution.  

The new search method differs from the previous one in two important ways:

1. **Richer analytical seed set** – we add the exact “integrated arcsine”
   distribution (the continuous solution whose self‑convolution is uniform)
   and a dense grid of Beta‑shaped PDFs with shape parameters
   `α,β ∈ [0.1, 1.5]`.  Seeds with very small Dirichlet concentration are also
   included.  This gives many candidates that are already close to the
   theoretical lower bound.

2. **Time‑aware optimisation pipeline** – after picking the best seed we run
   three short but well‑tuned phases:
        * Adam on a smooth‑max surrogate (log‑sum‑exp of the convolution)
        * Exact sub‑gradient descent on the true max (exponentiated‑gradient)
        * L₂‑flattening (drive the auto‑convolution towards a constant)
   Finally a stochastic hill‑climb “polishes” the result.
   All phases respect the global 1000 s budget.

The implementation uses FFT‑based convolutions (≈ O(n log n) per iteration)
and works for `n` up to several‑thousand without exceeding the time limit.
The final sequence is stored in the global variable `height_sequence_1`
so that future calls can warm‑start from it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full length) using a real FFT.
    The result length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution of a (full length) using a single FFT."""
    L = 2 * len(a) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (sum(a) = 1  ⇒  obj = 2·n·max(conv))
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return the objective value for a probability vector (sum == 1)."""
    conv = _conv_self(a)
    return 2.0 * len(a) * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation (analytical families + random)
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator, warm: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # ---- Integrated arcsine (exact discretisation of the continuous optimum) ----
    left = np.arange(0, n, dtype=np.float64) / n          # i/n
    right = np.arange(1, n + 1, dtype=np.float64) / n   # (i+1)/n
    arcsine = (2.0 / np.pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    arcsine /= arcsine.sum()
    seeds.append(arcsine)

    # ---- Beta‑shaped PDFs (α,β ∈ [0.1, 1.5]) ----
    alphas = np.linspace(0.1, 1.5, 15)
    betas = np.linspace(0.1, 1.5, 15)
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    for a_par in alphas:
        for b_par in betas:
            shape = (idx ** (a_par - 1.0)) * ((n - idx + 1.0) ** (b_par - 1.0))
            shape /= shape.sum()
            seeds.append(shape)

    # ---- Dirichlet draws with very low concentration (spiky seeds) ----
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a)

    # ---- Warm‑start (if available and compatible) ----
    if warm is not None and warm.shape == (n,):
        w = np.maximum(warm, 0.0)
        s = w.sum()
        if s > 0.0:
            seeds.append(w / s)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).  The smooth‑max uses a temperature β
    that grows multiplicatively.
    """
    v = np.log(a0 + 1e-30)                # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 1.0            # start with a smooth bound
    beta_mul = 1.015

    best_seq = a0.copy()
    best_val = _objective(a0)
    t = 0

    while True:
        if time.time() > deadline or t >= max_iters:
            break
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_self(a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:5d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float,
                          max_iters: int = 20_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        conv = _conv_self(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)     # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – µ·1‖²,  µ = 1/(2n‑1).
    Driving the auto‑convolution towards a constant.
    """
    a = a0.copy()
    a /= a.sum()
    best_seq = a.copy()
    best_val = _objective(a)

    mu = 1.0 / (2 * n - 1)

    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        conv = _conv_self(a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away from the
    contributors to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    it = 0
    while True:
        if time.time() > deadline or it >= max_iters:
            break
        it += 1

        conv = _conv_self(cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:5d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val
