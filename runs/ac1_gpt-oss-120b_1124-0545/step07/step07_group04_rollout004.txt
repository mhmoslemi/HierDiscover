def _subgradient_refine(start_seq, n, deadline, max_iters=600_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active indices within rel_tol
        active = np.where(conv >= max_val - rel_tol)[0]
        # construct weight vector w of length 2n-1
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        rev_a = a[::-1]
        # convolution rev_a * w
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n-1:n-1+n]   # gradient of average max
        # exponentiated update
        lr = 0.9 / math.sqrt(it)   # maybe adjust factor
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it} best={cur_val:.6f}")
    return best_seq, best_val

    SCIPY_OPT = True
except Exception:   # pragma: no cover
    SCIPY_OPT = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _linear_decay(n: int) -> np.ndarray:
    """Linearly decreasing from left to right."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    return a / a.sum()


def _double_exponential(n: int, r: float = 0.95) -> np.ndarray:
    """Symmetric sum of a left and a right geometric decay."""
    left = r ** np.arange(n, dtype=np.float64)
    right = left[::-1]
    a = left + right
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _make_seed_pool(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Construct a heterogeneous pool of normalised start vectors.
    """
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # ----- deterministic families -----
    seeds.append(_uniform(n))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))

    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    seeds.append(_gaussian(n, sigma_factor=0.05))
    seeds.append(_gaussian(n, sigma_factor=0.10))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_gaussian(n, sigma_factor=0.30))

    # ----- custom shapes -----
    seeds.append(_linear_decay(n))
    seeds.append(_double_exponential(n, r=0.96))
    seeds.append(_sine_window(n))

    # ----- random Dirichlet draws -----
    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ----- previous construction if compatible -----
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ----- normalise and clip -----
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Return (value, gradient) of the smooth approximation of the max‑conv objective.
    """
    a = _softmax(v)                     # probability vector, sum = 1
    conv = _conv_fft(a, a)              # length 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta   # = (1/β) log Σ exp(β·convₖ)
    obj = 2.0 * n * smooth_val

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t a: 2 * rev(a) * w (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through softmax (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 500) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector found and its exact objective.
    """
    if not SCIPY_OPT:                     # no SciPy → skip L‑BFGS
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)           # logits for the soft‑max

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f} → best={cur_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent refinement
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Uses a vectorised sub‑gradient computed via one FFT per iteration.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build weight vector w = average of active indicator functions
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, len(active))

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        g = 2.0 * full[n - 1:n - 1 + n]          # sub‑gradient for each a_i

        # exponentiated‑gradient update (mirror descent on the simplex)
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                   # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between two random
    coordinates, plus occasional Gaussian jumps.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]          # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
        if accept:
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Respects the given time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem dimension – 4000 proved effective in earlier runs.
    n = 4000

    # Load a previous construction if present.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])            # low is good
    top_k = 12                                 # a few good starting points
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 20000.0, num=15)   # finer schedule, larger final β
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=500)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS {idx}] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient mirror‑descent refinement.
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=500_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Stochastic restarts – perturb current best and re‑refine.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # small Gaussian perturbation of the current best
        rng = np.random.default_rng()
        perturb = rng.normal(scale=0.015, size=n)
        new_start = global_best_seq + perturb
        new_start = np.clip(new_start, 0.0, None)
        tot = new_start.sum()
        if tot == 0.0:
            break
        new_start /= tot

        # short mirror‑descent run
        seq, val = _subgradient_refine(new_start, n, deadline, max_iters=120_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[restart] improved to {val:.6f}")

        # stop early if we are within a few seconds of the deadline
        if time.time() > deadline - 5:
            break

    # ------------------------------------------------------------------
    # 6) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be hit.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list
    return global_best_seq.tolist()
