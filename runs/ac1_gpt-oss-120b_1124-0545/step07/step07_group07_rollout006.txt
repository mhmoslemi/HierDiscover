"""
Propose a non‑negative step‑function (list of floats) that minimizes

    2·n·max(conv(a, a)) / (sum(a)^2)

where `conv` is the linear convolution.  

The algorithm consists of three stages:

1. **Rich seeding** – generate a large, diverse set of analytically
   motivated candidates (geometric, power‑law, beta‑type, sine,
   Gaussian, random Dirichlet, …).  All candidates are ℓ₁‑normalised.

2. **Iterated local optimisation** – starting from the best seed,
   repeatedly apply cheap but effective optimisation primitives:
   * Adam on a smooth‑max surrogate (softmax parametrisation)
   * Exact sub‑gradient descent on the true max (exponentiated‑gradient)
   * L₂‑flattening (drive the auto‑convolution towards a constant)
   * Targeted stochastic hill‑climb
   Each primitive runs for a short, time‑aware slice; the whole loop
   repeats until the overall budget (≈ 1000 s) is exhausted, allowing many
   restarts and refinements.

3. **Verification** – the obtained vector is finally passed to
   `evaluate_sequence` to guarantee it is accepted.  The best vector is
   stored in the global variable `height_sequence_1` for warm‑starting
   future calls.

The code uses only NumPy and the provided `evaluate_sequence` function,
runs on a single thread (≤ 2 CPU cores) and respects the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers (FFT based, reuse the power‑of‑2 trick)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)).  Assumes sum(a)=1 (normalised)."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation – a wide collection of analytic families + random draws
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Simple deterministic families
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                # uniform
    i = np.arange(n, dtype=np.float64)
    seeds.append(np.sin(np.pi * (i + 1) / (n + 1)) /                         # half‑sine
                 np.sin(np.pi * (i + 1) / (n + 1)).sum())
    # Hann window (cosine bell)
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    seeds.append(hann / hann.sum())
    # Hamming window
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    seeds.append(hamming / hamming.sum())

    # 2. Geometric families (r ∈ (0,1))
    for r in np.linspace(0.70, 0.99, 30):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # 3. Power‑law families (p ≥ 0)
    for p in np.linspace(0.0, 6.0, 31):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p + 1e-12)
        seeds.append(a / a.sum())

    # 4. Beta‑like shapes (α,β > 0) – moderately fine grid
    alphas = np.linspace(0.3, 3.0, 7)
    betas = np.linspace(0.3, 3.0, 7)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for alpha in alphas:
        for beta in betas:
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            if a.sum() > 0:
                seeds.append(a / a.sum())

    # 5. Random Dirichlet draws with many concentration values
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a)                     # already sums to 1

    # 6. Log‑normal / Gaussian bell shapes
    for _ in range(12):
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.3 * n, 0.8 * n)   # wide bells
        a = np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 7. Softmax of random logits
    for _ in range(12):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # 8. Warm‑start (if supplied)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

    v = np.log(a0 + 1e-30)                     # logits

    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0            # initial smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² (μ = 1/(2n‑1)) using exponentiated‑gradient.
    Drives the auto‑convolution towards a uniform vector.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away
    from the contributors to the current maximal convolution coefficient.
    Accepts only improving moves (pure ascent).
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:                 # only accept improvements
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine respects the given time budget
    and returns the best vector found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small guard so the caller does not kill us while printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Problem size (4000 works comfortably and was proven good)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a large, diverse seed collection
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a short optimisation phase on the current best.
    # Allocates a fraction of the remaining time and a modest iter limit.
    # ------------------------------------------------------------------
    def _run_phase(opt_func, cur_seq, phase_name, frac, max_iters):
        nonlocal global_best_seq, global_best_val
        remain = deadline - time.time()
        if remain < 5.0:
            return cur_seq
        phase_deadline = time.time() + max(1.0, remain * frac)
        seq, val = opt_func(cur_seq, n, phase_deadline, max_iters=max_iters)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[{phase_name}] best={val:.6f}")
        return seq

    # ------------------------------------------------------------------
    # Initial deep refinement (single pass similar to the reference code)
    # ------------------------------------------------------------------
    cur = global_best_seq
    cur = _run_phase(_adam_optimize, cur, "Adam", 0.30, 120_000)
    cur = _run_phase(_subgradient_optimize, cur, "SubGrad", 0.25, 120_000)
    cur = _run_phase(_l2_flatten_optimize, cur, "L2Flat", 0.20, 120_000)
    cur = _run_phase(_hill_climb_optimize, cur, "Hill", 0.15, 120_000)

    # ------------------------------------------------------------------
    # Repeated short cycles (fast restarts) until time runs out.
    # Each cycle tries a fresh random start around the current best.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # 1‑step random perturbation to create a new starting point
        jitter = np.clip(
            cur + 0.02 * cur * np.random.standard_normal(n), 0.0, np.inf)
        jitter /= jitter.sum()                     # stay on the simplex

        # Run a lightweight sub‑gradient pass
        cur = _run_phase(_subgradient_optimize, jitter,
                         "SG‑Restart", 0.10, 6_000)

        # Follow by a brief Adam pass
        cur = _run_phase(_adam_optimize, cur,
                         "Adam‑Restart", 0.10, 6_000)

        # Finally a few hill‑climb steps
        cur = _run_phase(_hill_climb_optimize, cur,
                         "Hill‑Restart", 0.05, 6_000)

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
