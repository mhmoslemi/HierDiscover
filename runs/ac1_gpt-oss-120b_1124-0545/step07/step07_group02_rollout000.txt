"""
Hybrid optimiser – smooth‑max gradient descent + annealing

Approach
--------
* Work on the probability simplex by representing the vector as a soft‑max
  of free log‑variables `z`.  This automatically guarantees non‑negativity
  and unit sum.
* Approximate the non‑smooth objective    max(conv(a,a))
  with a smoothed version  Σ w_k·C_k where the weights w_k are a
  temperature‑controlled soft‑max (implemented via a power‑law to avoid
  overflow).  The gradient of this surrogate is cheap: it is a convolution
  of the reversed probability vector with the weight vector.
* Optimise the surrogate by Adam (adaptive mirror descent).  The temperature
  (or power) is increased linearly so that the optimisation starts smooth
  and finishes close to the true max.
* After the gradient phase a short simulated‑annealing walk (large mass‑
  transfer moves) is performed to escape possible shallow local minima.
* The whole routine respects the 1000 s time budget, automatically falls
  back to a uniform distribution and stores the best result in the global
  name ``height_sequence_1`` for possible chaining.

The code below is self‑contained and defines the entry point
``propose_candidate`` required by the grader.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution (real FFT, full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families – used for initialisation / fall‑back
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Monotone geometric a_i ∝ r**i, normalised to sum = 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (identical to the one from the reference)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with larger mass‑transfer moves."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.10 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] (jump) it={it:7d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length of the step‑function – a compromise between quality and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initial vector (reuse a previous construction if available)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, (list, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                init_seq = np.clip(prev_arr, 0.0, None)
                if init_seq.sum() > 0:
                    init_seq = init_seq / init_seq.sum()
                else:
                    init_seq = _uniform(n)
            else:
                init_seq = _geom(n, 1.0 - 3.0 / n)
        except Exception:
            init_seq = _geom(n, 1.0 - 3.0 / n)
    else:
        init_seq = _geom(n, 1.0 - 3.0 / n)   # a modestly decaying geometric start

    # --------------------------------------------------------------
    # Mirror‑descent on a smooth approximation of the max‑conv term
    # --------------------------------------------------------------
    # Log‑variables for the simplex (soft‑max parametrisation)
    z = np.log(init_seq + 1e-30)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_seq = init_seq.copy()
    best_val = _objective(best_seq)

    max_iters = 300_000               # generous upper bound – actual count limited by time
    p_max = 30.0                      # final “power” for the soft‑max weighting
    lr0 = 0.12                        # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- current probability vector (stable soft‑max) ---
        z_shift = z - np.max(z)                 # avoid overflow
        a = np.exp(z_shift)
        a /= a.sum()                            # ∑ a_i = 1

        # --- evaluate true objective (for bookkeeping) ---
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv / (a.sum() ** 2)   # a.sum() == 1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Grad] it={it:7d} best={best_val:.6f}")

        # --- temperature / power schedule (gradually sharpen) ---
        p = 1.0 + (p_max - 1.0) * (it / max_iters)

        # --- compute smooth‑max weights w_k ∝ (C_k)^p  (stable via logs) ---
        # ensure positivity
        eps = 1e-30
        log_c = np.log(conv + eps)
        log_w = p * log_c
        log_w -= np.max(log_w)                   # stabilise
        w_raw = np.exp(log_w)
        w = w_raw / w_raw.sum()                  # Σ w_k = 1

        # --- gradient of the surrogate with respect to a ---
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        g = 2.0 * full[n - 1:n - 1 + n]           # ∂/∂a_i Σ w_k C_k

        # --- map gradient onto the simplex (soft‑max Jacobian) ---
        dot = np.dot(g, a)
        grad_z = a * (g - dot)                   # ∂/∂z_i of the surrogate

        # --- Adam update (mirrored onto the simplex) ---
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)                 # simple decay
        z -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # keep occasional prints for very long runs
        if it % 50000 == 0:
            elapsed = time.time() - start_time
            print(f"[Grad] it={it:7d} elapsed={elapsed:.1f}s best={best_val:.6f}")

    # --------------------------------------------------------------
    # Final local search – simulated annealing (uses remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Overall Anneal] best={best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback to uniform if everything failed (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[Fallback] used uniform distribution")

    # --------------------------------------------------------------
    # Final clean‑up – ensure non‑negativity and normalisation
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total == 0.0:
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / total

    # store for possible chained runs
    globals()["height_sequence_1"] = best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {n})")
    return best_seq.tolist()
