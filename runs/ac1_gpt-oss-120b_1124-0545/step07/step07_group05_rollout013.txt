def _proj_simplex(v):
    # ensure that v sums to 1 and non-negative.
    # Duchi et al:
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (∑ a)²

Algorithm
---------
1. **Rich seed pool** – generate many normalised candidates:
   uniform, sine‑window, geometric, exponential, power‑law,
   beta‑like and fine‑gridded families (σ for exponential, p for
   power‑law, and a combined (exponential·power) family).  Re‑use the
   previous best sequence if it exists.

2. **Coarse selection** – evaluate all seeds with a cheap FFT
   convolution and keep the best one.

3. **Projected sub‑gradient descent** – treat  max(conv) as a
   convex‑piecewise function and iteratively move opposite to a
   sub‑gradient (the active convolution entry) while projecting onto
   the probability simplex.  A diminishing step‑size η=α/√t (α≈0.5)
   is used; the loop stops when the time budget is exhausted.

4. **High‑frequency Simulated Annealing** – from the refined vector run
   millions of cheap moves that update the full convolution
   incrementally.  Two move types are used:
   * random Dirichlet re‑allocation on 2‑6 indices,
   * targeted re‑allocation that preferentially touches indices that
     contribute most to the current maximal convolution entry.
   The temperature decays slowly (factor 0.9999) so exploration lasts
   long.  The best sequence seen during SA is kept.

All phases respect the global 1000 s budget; the routine returns the
best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """
    Objective for a simplex‑normalised vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero; return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (all returns are normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    """a[i] = exp(-i/sigma)."""
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (coarse)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    # geometric families – gentle decay
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law families (coarse)
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (wide grid)
    for sigma in np.linspace(n / 30, n / 1.5, 12):
        seeds.append(_exp_decay(n, sigma))

    # power‑law fine grid
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power(n, p))

    # combined exponential·power families
    for sigma in (n / 30, n / 10, n / 5):
        for p in (0.5, 1.0, 1.5, 2.0):
            a = _exp_decay(n, sigma) * _power(n, p)
            a = a / a.sum()
            seeds.append(a)

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution (if shape matches)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # remove duplicates (very cheap)
    uniq = []
    seen = set()
    for s in seeds:
        h = hash(s.tobytes())
        if h not in seen:
            seen.add(h)
            uniq.append(s)
    return uniq

def _projected_gradient_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the true max‑convolution.
    The iterate stays on the probability simplex.
    """
    a = seed.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and the active shift
        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[k_star]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[PGD] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k_star - idx
        g[idx] = 2.0 * a[rev]

        # diminishing step size
        eta = alpha / np.sqrt(it)

        # gradient step and projection onto simplex
        a = _proj_simplex(a - eta * g)

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 6_000_000,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with cheap incremental convolution updates.
    Two move types:
        • random Dirichlet redistribution on 2‑6 indices,
        • targeted move that draws a high‑contribution index from the
          current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # choose move type
        if rng.random() < 0.5:
            # ---------- random Dirichlet reallocation ----------
            k = rng.integers(2, 7)                  # 2‑6 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # ---------- targeted move ----------
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)
            # second index drawn uniformly from the complement
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation for the selected positions (Dirichlet)
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for i in range(idx.size):
            for j in range(i + 1, idx.size):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin – leaving a couple of seconds for final I/O
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Grab a previous construction if it exists (global variable)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a single length (larger n gives more granularity but costs
    # more per iteration).  For the current budget 4000 is a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # 1️⃣ Seed generation (rich family, fine grids)
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    # 2️⃣ Quick evaluation of all seeds – keep the best one
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient refinement (full‑conv gradient steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_pg, val_pg = _projected_gradient_optimize(
            global_best_seq, n, deadline, max_iters=200_000, alpha=0.5
        )
        if val_pg < global_best_val:
            global_best_val = val_pg
            global_best_seq = seq_pg.copy()
            print(f"[overall‑PGD] new global best={val_pg:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Simulated‑annealing phase (dominant time consumer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Estimate a generous iteration budget – each iteration costs ~8e‑5 s
        # for n≈4000.  We aim for as many as the remaining time allows.
        target_iters = int(min(6_000_000, remaining * 12_000))
        seq_sa, val_sa = _sa_incremental_optimize(
            global_best_seq,
            n,
            deadline,
            init_temp_factor=0.02,
            min_temp=1e-9,
            max_iters=target_iters,
            decay=0.9999,
        )
        if val_sa < global_best_val:
            global_best_val = val_sa
            global_best_seq = seq_sa.copy()
            print(f"[overall‑SA] new global best={val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
