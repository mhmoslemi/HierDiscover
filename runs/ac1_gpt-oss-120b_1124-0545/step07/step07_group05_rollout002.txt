"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm concentrates the whole 1000 s budget on a single length
(4000) – the best length found in earlier runs.  It consists of four
stages:

1️⃣  Seed generation – a very diverse catalogue (uniform, sine, beta‑like,
    quadratic, cosine‑square, exponential, random Dirichlet, sparse,
    mixtures of previous seeds …) all normalised to sum = 1.

2️⃣  Adam on a smooth‑max – each of the 5 best seeds is refined by Adam
    on a log‑sum‑exp approximation of the max‑operator.  The smoothing
    parameter β is increased slowly; 40 000 iterations per seed give a
    high‑quality starting point.

3️⃣  Exact sub‑gradient mirror descent – exponentiated‑gradient (mirror)
    steps on the true sub‑gradient of the max‑convolution are performed
    for up to 1 200 000 iterations (or until the time budget expires).

4️⃣  Aggressive simulated‑annealing – an incremental‑update SA with two
    move families (random Dirichlet re‑allocation on 2‑6 indices and a
    targeted move that touches the current maximal‑convolution support).
    The temperature decays very slowly (factor 0.9998) and the loop runs
    for as many iterations as the remaining time permits (up to a few
    million).  Improvements are printed only when a new best value is
    found.

All phases respect a global deadline; the best sequence encountered is
returned as a plain ``list[float]``.
"""

import time

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for many calls)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _objective(a: np.ndarray) -> float:
    """
    Objective for a simplex‑normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = -(i - (n - 1) / 2.0) ** 2 + ((n - 1) / 2.0) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _cosine_sq(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / n) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_exp(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - (n - 1) / 2.0) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))

def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

def _make_seeds_extended(n: int,
                         previous: np.ndarray | None,
                         rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse, enriched set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (including U‑shaped)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2
        seeds.append(_logistic_shape(n, offset, scale))

    # extra analytic families
    seeds.append(_quadratic(n))
    seeds.append(_cosine_sq(n))
    seeds.append(_double_exp(n, sigma=n / 8.0))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # mixture of two random seeds (helps diversity)
    for _ in range(4):
        s1 = rng.choice(seeds)
        s2 = rng.choice(seeds)
        t = rng.random()
        mixed = t * s1 + (1.0 - t) * s2
        mixed /= mixed.sum()
        seeds.append(mixed)

    # reuse previous solution (if same length)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 40000) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)               # parameters of a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                        # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:d} it={it:d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a  (conv = a*a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent (exponentiated‑gradient)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:d} it={it:d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) at k_star
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                     # keep numerics stable
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize_v2(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               init_temp_factor: float = 0.05,
                               min_temp: float = 1e-10,
                               max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    High‑frequency simulated‑annealing with incremental convolution updates.
    Two move families:
        • random Dirichlet re‑allocation on 2‑6 indices,
        • targeted re‑allocation on the support of the current maximal convolution.
    Temperature decays slowly (factor 0.9998) to allow thorough exploration.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9998

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        # ----- select move type -----
        r = rng.random()

            k = rng.integers(2, 7)                # 2‑6 positions
            idx = rng.choice(n, size=k, replace=False)

        elif r < 0.85:                             # targeted move on current max‑conv support
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # choose two distinct indices inside the support,
            # weighted by their contribution a[i]*a[k_star-i]
            contrib = cur[support] * cur[k_star - support]
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i = rng.choice(support, p=probs)
                j = rng.choice(support, p=probs)
                while j == i:
                    j = rng.choice(support, p=probs)
                idx = np.array([i, j], dtype=int)
            else:
                idx = rng.choice(support, size=2, replace=False)

        else:                                      # tiny exchange between two random positions
            i, j = rng.choice(n, size=2, replace=False)
            total = cur[i] + cur[j]
            if total == 0:
                continue
            # move a random fraction (≤10 %) from i to j
            eps = rng.uniform(0.0, 0.1) * cur[i]
            new_i = cur[i] - eps
            new_j = cur[j] + eps
            if new_i < 0:
                continue
            idx = np.array([i, j], dtype=int)
            new_alloc = np.array([new_i, new_j], dtype=np.float64)

            # incremental convolution update for this tiny move
            deltas = new_alloc - cur[idx]
            if np.allclose(deltas, 0.0):
                continue
            conv_new = cur_conv.copy()
            for p, delta in zip(idx, deltas):
                if delta == 0.0:
                    continue
                conv_new[p:p + n] += 2.0 * delta * cur
                conv_new[2 * p] += delta * delta
            for ii in range(idx.size):
                for jj in range(ii + 1, idx.size):
                    pos = idx[ii] + idx[jj]
                    conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

            new_obj = 2.0 * n * np.max(conv_new)
            delta_obj = new_obj - cur_obj
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
                cur[idx] = new_alloc
                cur_conv = conv_new
                cur_obj = new_obj
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = cur.copy()
                    if it % 50000 == 0:
                        print(f"[SA v2] n={n:d} it={it:d} best={new_obj:.6f}")
            continue   # tiny‑exchange handled – go to next iteration

        # ----- generic Dirichlet re‑allocation on `idx` -----
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # incremental convolution update
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)
        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                if it % 50000 == 0:
                    print(f"[SA v2] n={n:d} it={it:d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # previous solution (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if np.any(previous < 0):
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ----------------------------------------------------------
    # Work on a single, well‑studied length (4000)
    # ----------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")
    seeds = _make_seeds_extended(n, previous, rng)

    # ---- Phase 0 : quick evaluation of raw seeds ----
    best_seq = None
    best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ---- Phase 1 : Adam on the most promising seeds ----
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda t: t[1])
    top_k = min(5, len(seed_vals))
    for s, _ in seed_vals[:top_k]:
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=40000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall‑Adam] new global best={val:.6f}")

    # ---- Phase 2 : Exact sub‑gradient mirror descent ----
    if best_seq is not None and best_seq.shape[0] == n:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=1_200_000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall‑SubGrad] new global best={val:.6f}")

    # ---- Phase 3 : High‑frequency simulated‑annealing (longest phase) ----
    if best_seq is not None and best_seq.shape[0] == n:
        remaining = deadline - time.time()
        # Estimate max iterations we can afford (≈5 µs per iteration for n≈4000)
        target_iters = int(min(3_000_000, remaining * 5000))
        seq, val = _sa_incremental_optimize_v2(best_seq, n, deadline,
                                               max_iters=target_iters)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall‑SA] new global best={val:.6f}")

    # ----------------------------------------------------------
    # Fallback (should never trigger)
    # ----------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(2000)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    # optional verification (ignore errors – the evaluator may reject)
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
