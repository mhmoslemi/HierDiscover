def solve_lp_max_sum(a, M):
    import cvxpy as cp
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build constraint: conv(a,b) <= M
    # Use list comprehension to create constraints
    constraints = []
    # For each k produce linear sum a[i]*b[k-i]
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # Build expression: sum a[i] * b[k-i] for i_start..i_end
        expr = 0
        # vectorized using cp.sum?
        expr = cp.sum(cp.multiply(a[i_start:i_end+1], b[k - np.arange(i_start, i_end+1)]))
        constraints.append(expr <= M)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in ("optimal","optimal_inaccurate"):
        return None
    return b.value
```

def upsample(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (∑ a)²

The optimiser works in several stages:

1️⃣ **Seed generation** – a few diverse simplex‑normalised vectors (uniform,
   sine‑window, geometric, power‑law, random Dirichlet).  The best seed is
   kept as a starting point.

2️⃣ **Sub‑gradient mirror descent** – exponentiated‑gradient updates using the
   exact sub‑gradient of the true max‑convolution.  This quickly drives the
   peak of the autocorrelation toward the average value.

3️⃣ **High‑frequency simulated‑annealing** – cheap incremental updates of the
   full convolution allow millions of moves.  Two move types are used:
   *random Dirichlet re‑allocation on 2‑4 indices* and a *targeted move* that
   touches indices participating in the current maximal convolution entry.

4️⃣ **Linear‑programming refinement** – given the current vector `a` and its
   maximal convolution value `M`, we solve the LP

        maximise   1ᵀb
        subject to  conv(a,b) ≤ M ,   b ≥ 0

   (conv(a,b) is linear in `b`).  If the optimum satisfies Σb > 1 we can
   move a little towards the normalised `b` and strictly decrease the
   objective.  This step is repeated a few times.

5️⃣ **Upsampling & final SA** – the best vector for a modest length (2000)
   is linearly interpolated to a larger length (4000) and refined once more
   with simulated‑annealing.

All stages respect a global time budget (default 1000 s).  The routine
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
import scipy.sparse as sp

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    a /= a.sum()
    return a


def _make_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (coarse)
    for r in np.linspace(0.2, 0.9, 5):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 5):
        seeds.append(_power(n, p))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    return seeds

def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        a /= a.sum()
    return best, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with incremental convolution updates.
    Two move types are used:
      * random Dirichlet re‑allocation on 2‑4 positions,
      * targeted re‑allocation touching indices responsible for the
        current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * decay, min_temp)

        if rng.random() < 0.5:
            # random Dirichlet on 2‑4 indices
            k = rng.integers(2, 5)               # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # targeted move – indices from the current maximal entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[support] * cur[k_star - support]
            if contrib.sum() > 0:
                i = rng.choice(support, p=contrib / contrib.sum())
            else:
                i = rng.choice(support)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur     # a*Δ + Δ*a
            conv_new[2 * p] += delta * delta           # Δ*Δ diagonal term

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)
        delta_obj = new_obj - cur_obj

        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best, best_obj

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix B such that B @ b = conv(a,b).
    Row k contains the coefficients a[i] for every pair i + j = k.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # each column j (b_j) contributes a[i] to row i+j for i=0..n-1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    idx = 0
    for i in range(n):
        rows[idx:idx + n] = np.arange(i, i + n)    # rows i … i+n‑1
        cols[idx:idx + n] = np.arange(n)          # columns 0 … n‑1
        data[idx:idx + n] = a[i]                  # coefficient a[i] repeated
        idx += n

    B = sp.csr_matrix((data, (rows, cols)), shape=(L, n))
    return B


def _solve_lp_max_sum(a: np.ndarray, M: float, time_limit: float = 5.0) -> np.ndarray | None:
    """
    Solve   maximise ∑b_i   subject to   conv(a,b) ≤ M , b ≥ 0 .
    Returns the optimal b as a dense ndarray, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)

    B = _build_conv_matrix(a)
    # Add a modest upper bound to avoid unboundedness when a has very tiny entries.
    ub = cp.Parameter(value=10.0, nonneg=True)

    constraints = [B @ b <= M, b <= ub]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, warm_start=True, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    if b.value is None:
        return None
    return np.array(b.value).flatten()


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_n points, renormalised to sum = 1."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new[a_new < 0] = 0.0
    a_new /= a_new.sum()
    return a_new

    # ------------------------------------------------------------------
    if time.time() < deadline:

        # allocate up to a couple of million SA moves (adjusted to remaining time)
        target_iters = int(min(2_000_000, remaining * 12_000))
        seq, val = _sa_incremental_optimize(best_seq, n_small, deadline,
                                            max_iters=target_iters)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[overall‑SA] new global best={val:.6f}")

    # ------------------------------------------------------------------

        best_val = _objective(best_seq)

    if time.time() < deadline:
        remaining = deadline - time.time()
        target_iters = int(min(1_500_000, remaining * 12_000))
        large_seq, large_val = _sa_incremental_optimize(large_seq, n_large,
                                                        deadline,
                                                        max_iters=target_iters)
        if large_val < best_val:
            best_val, best_seq = large_val, large_seq.copy()
            print(f"[final‑SA] new global best={large_val:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence (optional)
    # ----------------------------------------------------------------------
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
