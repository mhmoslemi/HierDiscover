   logistic, arcsine (Beta(½,½)), semicircle, random Dirichlet,
   sparse Dirichlet, and a copy of the previous best sequence.

2️⃣ **Adam on a smooth‐max** – each seed is refined by Adam on a
   log‑sum‑exp approximation of the `max` operator.  The smoothness
   parameter β is increased slowly, so the optimiser gradually
   focuses on the true max.

3️⃣ **Exact sub‑gradient mirror descent** – from the best Adam result
   we perform exponentiated‑gradient steps using the true sub‑gradient
   of `max(conv)`.  The step‑size schedule η=0.7/√t gives fast
   convergence.

4️⃣ **Multi‑restart simulated annealing** – a cheap incremental update
   of the full convolution makes it possible to perform many moves.
   Two move types are used:
        • random Dirichlet re‑allocation on 2‑5 indices,
        • targeted re‑allocation that touches indices participating in
          the current maximal convolution entry.
   The temperature decays slowly and several independent restarts
   are performed, each limited by the remaining time budget.

All phases respect the global 1000 s budget; the best sequence found
is stored in the global variable ``height_sequence_1`` and returned as
a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    a /= a.sum()
    return a


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    a /= a.sum()
    return a


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    a /= a.sum()
    return a


def _arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – arcsine‑like shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    a /= a.sum()
    return a


def _semicircle(n: int) -> np.ndarray:
    """Discrete semicircle: a[i] ∝ √(i·(n‑i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sqrt(i * (n - i + 1.0))
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw (conc < 1 → very sparse)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # Simple baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # Geometric families
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # Beta‑like (including arcsine)
    for a in (0.2, 0.5, 1.0, 2.0):
        for b in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # Logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2.0
        seeds.append(_logistic_shape(n, offset, scale))

    # Arcsine and semicircle
    seeds.append(_arcsine(n))
    seeds.append(_semicircle(n))

    # Random Dirichlet (various concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # Re‑use previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        # Normalise to the simplex (might have been scaled)
        p = previous.astype(np.float64)
        if p.sum() > 0.0:
            seeds.append(p / p.sum())
        else:
            seeds.append(_uniform(n))

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)          # parameters of a softmax (stay on simplex)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = np.exp(v - np.max(v))          # same as softmax but a tiny bit faster
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1: n - 1 + n]

        # convert to gradient w.r.t. v (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_000_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap incremental convolution updates.
    Two move types are used:
        – Dirichlet re‑allocation on 2‑5 random indices,
        – Targeted re‑allocation that touches indices involved in the
          currently maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # temperature schedule
        T = max(T * decay, min_temp)

        # choose move type
        if rng.random() < 0.5:
            # Random Dirichlet on 2‑5 indices
            k = rng.integers(2, 6)                # 2‑5 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # Targeted move based on current maximal convolution entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # weight each index by its contribution a[i]·a[k_star‑i]
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0.0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # second index uniformly chosen (different from i)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # draw a new Dirichlet allocation for the selected positions
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta   # the self‑term

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            # accept move
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final verification step
    deadline = start_time + budget_s - 2.0

    # grab the previously discovered construction, if any
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to explore; we start small and increase while time permits
    candidate_lengths = [2000, 4000, 6000, 8000, 10000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in candidate_lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on smooth max (top‑k seeds only)
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(3, len(seed_vals))          # keep only a few promising seeds
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=500000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Multi‑restart Simulated Annealing
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            # Decide how many SA restarts we can afford with the remaining budget
            remaining = deadline - time.time()
            if remaining > 10.0:                     # need a little time to be worthwhile
                # Rough estimate: ~0.00007 s per SA iteration for n≈6000
                est_iter = int(min(2_000_000, remaining * 14000))
                restarts = max(1, int(remaining // 30))  # at least 30 s per restart
                it_per_restart = max(1, est_iter // restarts)

                for r in range(restarts):
                    if time.time() > deadline:
                        break
                    print(f"[SA‑restart] n={n} restart={r+1}/{restarts} it={it_per_restart}")
                    seq, val = _sa_incremental_optimize(
                        global_best_seq,
                        n,
                        deadline,
                        max_iters=it_per_restart,
                        init_temp_factor=0.04,   # a bit hotter to escape deeper wells
                    )
                    if seq is not None and val < global_best_val:
                        global_best_val = val
                        global_best_seq = seq.copy()
                        print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

                # final short sub‑gradient polishing after SA
                seq, val = _subgradient_refine(
                    global_best_seq, n, deadline, max_iters=200000
                )
                if seq is not None and val < global_best_val:
                    global_best_val = val
                    global_best_seq = seq.copy()
                    print(f"[final‑SubGrad] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # store for the next run (makes the optimiser stateful across calls)
    globals()["height_sequence_1"] = global_best_seq.tolist()

    # optional sanity check with the provided evaluate_sequence
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
