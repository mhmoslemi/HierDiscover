def _project_simplex(v):
    # v is 1D np.ndarray
    # Use algorithm from https://doi.org/10.1007/978-3-540-69617-5_31
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        g = np.zeros_like(a)

        # projection onto simplex
        a = _project_simplex(a)
        
        # re-normalize to avoid numerical drift
        # (project already ensures sum=1)
    return best_seq, best_val

def _adam_optimize(a0, n, deadline, beta_start=1.0, beta_mul=1.02, max_beta=5000, max_iters=500_000):
    v = np.log(a0 + 1e-30)   # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta = beta_start
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        t += 1
        # compute a
        a = np.exp(v - np.max(v))
        a = a / a.sum()
        # conv and objective
        conv = _conv_full(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # update beta
        if beta < max_beta:
            beta = min(max_beta, beta * beta_mul)

        shift = cur_max
        exp_beta = np.exp(beta * (conv - shift))
        w_sum = exp_beta.sum()
        if w_sum == 0.0:
            continue
        w = exp_beta / w_sum

        # gradient wrt a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # gradient wrt logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val

"""
Propose a non‑negative height sequence that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

The search proceeds in four time‑aware phases:

1. Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with an
   exponentially increasing temperature β, keeping the simplex constraint
   via the soft‑max parameterisation.

2. Adam optimisation of an L2‑flattening objective, i.e. minimise
   ‖a*a – µ·1‖² where µ = 1/(2n‑1).  This drives the auto‑convolution
   towards a constant (the theoretical average).

3. Projected sub‑gradient descent directly on the true max‑convolution.
   The sub‑gradient is the exact gradient of the active convolution entry.
   After each step the iterate is projected onto the probability simplex.

4. A stochastic hill‑climb with simulated‑annealing acceptance,
   moving a tiny amount of mass away from indices that contribute most
   to the current maximal convolution coefficient.

All phases respect a global 1000 s time budget.  The best sequence found
is stored in the global variable ``height_sequence_1`` for warm‑starting
future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm from
    # https://doi.org/10.1007/978-3-540-69617-5_31
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed generation (diverse analytic and random families)
# ----------------------------------------------------------------------


def _make_seeds(
    n: int, rng: np.random.Generator, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Half‑sine window (already used in the reference)
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # Raised‑cosine (Hann) and Hamming windows
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    seeds.append(hann / hann.sum())
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    seeds.append(hamming / hamming.sum())

    # Geometric families
    for r in np.linspace(0.2, 0.95, 8):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # Power‑law families
    for p in np.linspace(0.0, 4.0, 7):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 4):
        for beta in np.linspace(0.5, 2.0, 4):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(3):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(3):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal (Gaussian bell) shapes
    for _ in range(3):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # Warm‑start from the previous global best, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    beta_start: float = 1.0,
    beta_mul: float = 1.02,
    max_beta: float = 5000.0,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate.
    Parameterisation: a = softmax(v); the smooth‑max uses a temperature β.
    """
    v = np.log(a0 + 1e-30)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta = beta_start

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # a = softmax(v)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # raise temperature (but never exceed max_beta)
        if beta < max_beta:
            beta = min(max_beta, beta * beta_mul)

        shift = cur_max
        exp_beta = np.exp(beta * (conv - shift))
        w_sum = exp_beta.sum()
        if w_sum == 0.0:
            continue
        w = exp_beta / w_sum  # softmax weights

        # gradient of smoothed max w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _adam_optimize_l2(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000

    """
    Adam optimisation of the L2‑flattening objective
    (½‖a*a – µ·1‖² with µ = 1/(2n‑1)).
    """
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    mu = 1.0 / (2 * n - 1)

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[L2Adam] it={it:6d} best={cur_val:.6f}")

        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)  # length = 3n‑2
        grad_a = 4.0 * grad_full[n - 1 : n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _proj_subgrad_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true max.
    After each step the iterate is projected onto the probability simplex.
    """
    a = a0.copy()
    a /= a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[ProjSub] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # simple diminishing step size
        lr = 1.0 / np.sqrt(it)
        a = a - lr * g
        a = _project_simplex(a)

    return best_seq, best_val


def _hill_climb_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with simulated‑annealing acceptance.
    Mass is moved from donors (chosen proportionally to their contribution
    to the current maximal convolution entry) to random receivers.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02
    init_temp = cur_val * 0.01
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        # pick donor proportionally to contribution
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        # receiver: prefer low‑mass indices (to keep distribution spread)
        # sample a few candidates and pick the lightest
        cand = rng.integers(0, n, size=5)
        cand = cand[cand != donor]
        receiver = cand[np.argmin(cur[cand])]

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        # simulated‑annealing acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / (init_temp * (1 - it / max_iters) + 1e-12)):
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        # slowly shrink the transfer scale
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-7)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence as a list of floats and stores it in the global
    variable `height_sequence_1` for warm‑starting future calls.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length (larger n gives more flexibility but costs more)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max (most time‑consuming phase)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.35
        seq, val = _adam_optimize(
            global_best_seq,
            n,
            phase_deadline,
            beta_start=1.0,
            beta_mul=1.02,
            max_beta=5000.0,
            max_iters=500_000,
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – L2‑flattening via Adam
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.30
        seq, val = _adam_optimize_l2(
            global_best_seq, n, phase_deadline, max_iters=500_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑L2] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.20
        seq, val = _proj_subgrad_optimize(
            global_best_seq, n, phase_deadline, max_iters=300_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑ProjSub] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted hill‑climb (use whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _hill_climb_optimize(
            global_best_seq, n, phase_deadline, max_iters=500_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
