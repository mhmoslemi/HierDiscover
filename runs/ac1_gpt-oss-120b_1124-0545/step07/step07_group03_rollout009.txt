def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (efficient for large arrays)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Objective for a sequence already normalised to sum = 1:
    2 * n * max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    """Flat probability vector (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)

def _catalan_shape(n: int) -> np.ndarray:
    """Catalan-like coefficients (truncated series of (1‑z)^‑½)."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total <= 0.0:
        return _uniform(n)
    return coeffs / total

def _sin_shape(n: int) -> np.ndarray:
    """Half‑period sine shape (non‑negative, symmetric, sum=1)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)  # all positive
    total = vals.sum()
    return vals / total

def _random_dirichlet(alpha: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(alpha,…,alpha) vector (size n) and normalise."""
    if alpha <= 0.0:
        # fallback to uniform
        return _uniform(n)

    s = x.sum()

def _softmax(arr: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max of `beta * arr` – numerically stable."""
    mx = np.max(arr)
    exp = np.exp(beta * (arr - mx))
    s = np.sum(exp)
    if s == 0.0:
        # should not happen, but guard against
        return np.full_like(arr, 1.0 / arr.size)
    return exp / s

def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) for all integer lags.
    Returns an array of length len(w)+len(a)-1.
    """
    n = len(a)
    a_rev = a[::-1]
    L = len(w) + n - 1
    N = _next_pow2(L)
    fw = np.fft.rfft(w, n=N)
    fa_rev = np.fft.rfft(a_rev, n=N)

    # g_i = Σ_k w_k a_{k-i} => slice indices (n-1) .. (n-1+n-1)
    start = n - 1
    end = start + n
    return conv[start:end]

def _refine_exponential(start_seq: np.ndarray, deadline: float,
                        beta_start: float = 5.0,
                        beta_max: float = 500.0,
                        lr_base: float = 0.2) -> tuple[np.ndarray, float]:

    Exponentiated‑gradient descent on the simplex.
    Uses a soft‑max relaxation of the `max` in the objective.
    Returns the best sequence and its objective value encountered.
    """
    a = np.clip(start_seq, 0.0, None)

    else:

    it = 0

    while time.time() < deadline:
        it += 1

        # full auto‑convolution
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        w = _softmax(conv, beta)

        # gradient of the smoothed objective w.r.t. a (∂L/∂a)
        grad_a = 2.0 * _cross_corr(w, a)

        # step‑size schedule – diminishing like 1/√t

        else:

        beta = min(beta_max, beta * 1.001)

    z = np.sum(exp)
    if z == 0.0:
        return np.full_like(arr, 1.0 / arr.size)
    return exp / z


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) for all integer lags.
    Returns the central slice of length ``len(a)``:
        g_i = Σ_k w_k·a_{k‑i}
    """
    n = len(a)
    a_rev = a[::-1]
    L = len(w) + n - 1
    N = _next_pow2(L)
    fw = np.fft.rfft(w, n=N)
    fa_rev = np.fft.rfft(a_rev, n=N)
    conv = np.fft.irfft(fw * fa_rev, n=N)[:L]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (soft‑max relaxation)
# ----------------------------------------------------------------------


def _refine_exponential(start_seq: np.ndarray, deadline: float,
                        beta_start: float = 5.0,
                        beta_max: float = 400.0,
                        lr_base: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using a soft‑max approximation of ``max``.
    Returns the best sequence encountered and its objective value.
    """
    a = np.clip(start_seq, 0.0, None)
    s = a.sum()
    a = a / s if s > 0.0 else _uniform(len(a))

    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    beta = beta_start
    while time.time() < deadline:
        it += 1

        # full auto‑convolution and objective value
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} val={cur_val:.6f}")

        # soft‑max weights for a smoothed max
        w = _softmax(conv, beta)

        # gradient of the smoothed objective w.r.t. a
        grad_a = 2.0 * _cross_corr(w, a)

        # diminishing learning‑rate
        lr = lr_base / np.sqrt(it)

        # exponentiated‑gradient update (multiplicative)
        delta = -lr * grad_a
        # clip to avoid overflow in exp()
        delta = np.clip(delta, -50.0, 50.0)
        a = a * np.exp(delta)

        # renormalise to the simplex
        s = a.sum()
        a = a / s if s > 0.0 else _uniform(n)

        # slowly increase the temperature → sharper soft‑max
        beta = min(beta_max, beta * 1.001)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Optional previous solution (if one is stored globally)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths we will explore – larger n yields a lower theoretical bound
    # ------------------------------------------------------------------
    lengths = [8000, 12000, 16000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        for seed_vec in (_catalan_shape(n), _uniform(n), _sin_shape(n)):
            val = _objective(seed_vec)
            if val < best_val:
                best_val = val
                best_seq = seed_vec.copy()
                print(f"[Seed] n={n:5d} type=deterministic val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – random Dirichlet sampling (≈15 % of total budget)
    # ------------------------------------------------------------------
    now = time.time()
    sampling_deadline = now + 0.15 * (deadline - now)

    alphas = [0.1, 0.3, 0.5, 0.7, 1.0, 2.0, 5.0]

    while time.time() < sampling_deadline:
        # bias toward larger n (more expressive power)
        n = int(rng.choice(lengths, p=np.array(lengths) / np.sum(lengths)))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(alpha, n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        if best_val < 1.503:  # reached target early
            break

    # ------------------------------------------------------------------
    # Stage 3 – smooth exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        refined_seq, refined_val = _refine_exponential(
            best_seq,
            deadline,
            beta_start=5.0,
            beta_max=400.0,
            lr_base=0.3,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq
            print(f"[Overall‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also invoke evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
