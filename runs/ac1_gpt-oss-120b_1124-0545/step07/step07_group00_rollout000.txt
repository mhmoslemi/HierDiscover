def _subgradient_refine2(start_seq, n, deadline, max_iters=1000000):
    a = start_seq.copy()
    best_seq, best_val = a.copy(), _objective(a)
    eps = 1e-30
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SubGrad2] it={it} best={cur_val:.6f}")
        # mask near max
        mask = (conv >= maxc - 1e-12)  # could use relative tolerance
        # compute weight vector w (0/1)
        w = mask.astype(float) / mask.sum() if mask.sum()>0 else mask.astype(float) # maybe uniform
        # gradient = 2 * conv_rev(a_rev, w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n-1:n-1+n]
        # mirror descent step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val
```

    exponential, refined power‑law) and sweeping their parameters more
    densely,
  * up‑sampling the previously found sequence (if any) to the new length,
  * refining the sub‑gradient step: instead of a single active constraint
    we use *all* indices whose convolution value is within a tiny tolerance
    of the maximum (a true sub‑gradient of the max‑operator),
  * performing a final stochastic hill‑climb with a slightly slower cooling
    schedule,
  * polishing the result with a cheap local “mass‑transfer” greedy pass.

All heavy work respects the global time budget (default 1000 s) and the
search stops gracefully before the deadline.  The function returns the
best sequence as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import i0  # for Kaiser window

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)).
    The sequence is assumed to be normalised to sum = 1, therefore the
    returned value equals the true objective after any scaling.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    if not (0 < width_fraction < 1):
        return _uniform(n)
    half_flat = int(np.round(width_fraction * n / 2.0))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), symmetrised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    # make it symmetric if desired – we simply keep the one‑sided version
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay a_i = exp(-lam*|i-centre|)."""
    if lam < 0.0:
        return _uniform(n)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle, sigma = sigma_frac * n."""
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window, alpha∈[0,1]."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i_idx in range(n):
        if i_idx < per:
            a[i_idx] = 0.5 * (1 + np.cos(np.pi * (2 * i_idx / (alpha * (n - 1)) - 1)))
        elif i_idx <= (n - 1) * (1 - alpha / 2):
            a[i_idx] = 1.0
        else:
            a[i_idx] = 0.5 * (1 + np.cos(np.pi * (2 * i_idx / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta_param: float) -> np.ndarray:
    """Kaiser window with parameter beta (≥0)."""
    if beta_param < 0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = (i - alpha) / alpha
    a = i0(beta_param * np.sqrt(1 - arg ** 2)) / i0(beta_param)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Build a large, diverse set of normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.2))
    seeds.append(_sine_window(n))
    seeds.append(_inv_sqrt(n))

    # geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_sym_geom(n, r))

    # power‑law family (dense sweep)
    for p in np.linspace(0.0, 5.0, 31):
        seeds.append(_power(n, p))

    # exponential decay (symmetrised)
    for lam in np.linspace(0.0005, 0.02, 12):
        seeds.append(_exponential_decay(n, lam))

    # beta‑family (broad grid)
    alphas = np.linspace(0.2, 2.5, 9)
    betas = np.linspace(0.2, 6.0, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.15, 0.25, 0.35, 0.45, 0.55):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Kaiser windows
    for beta_param in (2.0, 4.0, 6.0, 8.0, 12.0):
        seeds.append(_kaiser_window(n, beta_param))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10, 20, 40):
        for conc in (0.1, 0.3):
            seeds.append(_random_sparse(n, k, conc, rng))

    # up‑sample previous solution if available
    if previous is not None and previous.size > 0:
        old_n = previous.shape[0]
        if old_n != n:
            # linear interpolation onto the new grid
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, previous)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                seeds.append(up / up.sum())
        else:
            seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 60000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smoothed‑max (log‑sum‑exp) surrogate.
    The parameters have been tuned for the larger size (n≈8000).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits on the simplex

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # start with a modest temperature and increase slowly
    beta = 0.3
    beta_mul = 1.002
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # smooth‑max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping (stability)
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400000,
    eps_mask: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true
    max‑convolution objective.  All constraints that are *active* (within
    eps_mask of the maximal value) are used simultaneously, giving a
    true sub‑gradient of the max‑operator.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # Determine the set of near‑max indices
        mask = conv >= (maxc - eps_mask)
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum          # average sub‑gradient over the active set

        # gradient = 2 * conv_rev(a_rev, w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)      # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing that redistributes mass among a random
    subset of coordinates.  The cooling schedule is deliberately slow
    to allow thorough exploration of the remaining time budget.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9997, min_temp)

        # pick a small random subset
        k = rng.integers(3, 9)          # 3 … 8 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # draw a fresh Dirichlet allocation on the same total mass
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] n={n:5d} it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _local_mass_transfer(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    passes: int = 5,
    inner_iters: int = 2000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy local improvement: repeatedly pick two indices and
    move a tiny fraction of mass from one to the other if the objective
    improves.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_val = _objective(a)

    for p in range(passes):
        if time.time() > deadline:
            break
        for _ in range(inner_iters):
            if time.time() > deadline:
                break
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0.0:
                continue
            delta = step_frac * a[i] * (rng.random() - 0.5)  # signed small move
            new_i = max(a[i] - delta, 0.0)
            new_j = a[j] + delta
            if new_j < 0.0:
                continue
            # renormalise – the move conserves total mass, so simple replace
            a_candidate = a.copy()
            a_candidate[i] = new_i
            a_candidate[j] = new_j
            cur_val = _objective(a_candidate)
            if cur_val < best_val:
                a = a_candidate
                best_val = cur_val
                # early exit to continue from improved point
                # (no break – keep exploring)
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    # ------------------------------------------------------------------
    # Load previous solution (if any) and possibly up‑sample it later.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that balances expressiveness and speed.
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n=== Starting search: n = {n}, budget = {budget_s}s ===")

    # ------------------------------------------------------------------
    # Stage 1 – generate a large, diverse seed set.
    # ------------------------------------------------------------------
    print("\n--- Generating seeds ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate all seeds using the cheap surrogate objective.
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    best_seeds = [vec for vec, val in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # Initialise global best.
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation (Adam + Sub‑gradient) for each top seed.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")
        # Adam phase (smooth surrogate)
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=60000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # Exact sub‑gradient (true max) phase
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=400000
        )
        if seq_sub is not None and val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing for final polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=500000
        )
        if seq_sa is not None and val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap local mass‑transfer greedy refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local mass‑transfer refinement ---")
        seq_local, val_local = _local_mass_transfer(
            global_best_seq, n, deadline, passes=8, inner_iters=3000
        )
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[overall‑Local] new global best = {val_local:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the result for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
