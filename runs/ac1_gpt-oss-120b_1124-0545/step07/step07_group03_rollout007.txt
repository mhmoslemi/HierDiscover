def _refine_smooth(a, deadline):
    best = a.copy()
    best_val = _objective(a)
    it=0
    lam=1.0
    lam_inc = 5.0
    lam_max = 50.0
    while time.time() < deadline:
        it+=1
        # schedule lambda: increase every certain iterations
        if it % 2000 == 0 and lam < lam_max:
            lam = min(lam_max, lam * lam_inc)
        conv = _conv_fft(a, a)
        # compute objective
        cur_val = 2.0 * len(a) * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print("[SmoothRefine] it={} n={} val={:.6f} lam={:.2f}".format(it, len(a), cur_val, lam))
        # compute smoothed gradient
        w = np.exp(lam * conv)
        w_norm = w / w.sum()
        # compute gradient via convolution with reversed w_norm
        w_rev = w_norm[::-1]
        conv_rev = _conv_fft(a, w_rev)
        slice_part = conv_rev[len(a)-1:2*len(a)-1]
        g = 2.0 * slice_part[::-1]
        # step size
        eta = 0.5 / np.sqrt(it)
        # exponentiated gradient update
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best, best_val

"""
Propose a step‑function (non‑negative list of heights) that minimises
the ``evaluate_sequence`` objective.

Algorithm
---------
1. **Deterministic seeds** – For each length in a modest set
   (2000,4000,8000,16000) a few analytically motivated sequences are
   generated:
     * Uniform vector.
     * Truncated Catalan coefficients (coeffs of (1‑z)^‑½).
     * Power‑law transforms of the Catalan shape (c_i**β).
     * Simple power‑law shapes (i+1)^‑p.

   All are normalised to sum = 1 and immediately evaluated.  The best
   candidate per length is stored.

2. **Massive random Dirichlet sampling** – While time permits, many
   random vectors are drawn from Dirichlet(α) with several α values
   (0.5, 1.0, 2.0).  The best vector seen for each length is kept and
   a global best is tracked.

3. **Smoothed exponential‑gradient refinement** –  The current global
   best (and the best found for each length) are refined on the simplex.
   We minimise the smoothed max of the convolution using
   ``f_λ(a)= (1/λ)·log Σ_k exp(λ·(a*a)_k)``.
   Its gradient can be written as a correlation of ``a`` with a
   probability vector derived from the exponential weights.
   An exponentiated‑gradient (KL‑mirror‑descent) step with a diminishing
   stepsize updates the distribution while keeping it non‑negative and
   normalised.

   The temperature λ is increased geometrically (1→5→…→≈30) so that the
   optimisation first follows a soft average and gradually focuses on
   the true maximum.

4. **Final sanity check** – the obtained sequence is passed once to
   ``evaluate_sequence`` (only to catch unexpected errors).

The routine is fully time‑aware (default budget = 1000 s) and returns
the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (both inputs are real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would compute when the input
    sequence is already normalised to sum=1:
        2·n·max(conv(seq,seq))

    Parameters
    ----------
    seq : np.ndarray
        Normalised (sum = 1) non‑negative vector.

    Returns
    -------
    float
        The objective value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^‑½ truncated to length n,
    then normalised to sum = 1."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _catalan_power(n: int, beta: float) -> np.ndarray:
    """c_i^beta, where c_i are Catalan coefficients,
    then normalised."""
    base = _catalan_shape(n) * (np.sum(_catalan_shape(n)))  # raw (non‑normalised) Catalan
    seq = np.power(base, beta, where=base > 0, out=np.zeros_like(base))
    s = seq.sum()
    if s == 0.0:
        return _uniform(n)
    return seq / s


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Simple power‑law a_i ∝ (i+1)^(-p), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(idx, -p, where=idx > 0, out=np.zeros_like(idx))
    s = seq.sum()
    if s == 0.0:
        return _uniform(n)
    return seq / s


# ----------------------------------------------------------------------
# Random Dirichlet sampling (different α)
# ----------------------------------------------------------------------


def _random_dirichlet_one(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector (α may be ≠1)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        # Gamma draws with shape α and scale 1
        x = rng.gamma(shape=alpha, scale=1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient refinement (KL‑mirror descent)
# ----------------------------------------------------------------------


def _refine_smooth(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector on the simplex using a smoothed version
    of the max‑convolution objective.

    The smoothing parameter λ is increased geometrically (1 → … → λ_max).
    At each iteration an exponentiated‑gradient (mirror descent with KL)
    step is taken.

    Parameters
    ----------
    start_seq : np.ndarray
        Normalised (sum = 1) non‑negative start vector.
    deadline : float
        Absolute ``time.time()`` after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        The best vector seen (lowest true objective).
    best_val : float
        Corresponding objective value.
    """
    a = np.copy(start_seq)
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30           # avoid log(0)
    it = 0
    lam = 1.0
    lam_max = 32.0
    lam_mul = 1.5         # increase factor for λ
    next_lam_inc = 1000   # increase λ every ``next_lam_inc`` iterations

    while time.time() < deadline:
        it += 1

        # Compute current convolution and true objective
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SmoothRefine] it={it:7d} n={n:5d} val={cur_val:.6f} λ={lam:.2f}")

        # ----- smoothed gradient -----
        # weights proportional to exp(λ·conv)
        w = np.exp(lam * conv)
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient g_j = 2 Σ_i a_i * w_norm_{i+j}
        # implement via correlation using FFT:
        w_rev = w_norm[::-1]                     # reverse order
        conv_rev = _conv_fft(a, w_rev)           # length 3n‑2
        # take slice (indices n‑1 … 2n‑2) and reverse to obtain g
        slice_part = conv_rev[n - 1 : 2 * n - 1]
        g = 2.0 * slice_part[::-1]

        # exponentiated‑gradient step with diminishing step size
        eta = 0.6 / np.sqrt(it)                  # a modest learning rate
        log_a = np.log(a + eps) - eta * g
        # stabilise numerically
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # schedule λ increase
        if it % next_lam_inc == 0 and lam < lam_max:
            lam = min(lam_max, lam * lam_mul)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget
    and returns the best sequence (plain ``list`` of ``float``).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths we will explore
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]

    # store best per length
    best_per_len = {}
    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds (including power‑law variants)
    # ------------------------------------------------------------------
    betas = [0.5, 0.7, 0.9, 1.0, 1.2, 1.4, 1.6]
    p_exps = [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80]

    for n in lengths:
        candidates = []

        # uniform and Catalan
        candidates.append(_uniform(n))
        candidates.append(_catalan_shape(n))

        # Catalan raised to various powers
        for beta in betas:
            candidates.append(_catalan_power(n, beta))

        # simple power‑law families
        for p in p_exps:
            candidates.append(_powerlaw_shape(n, p))

        # optional previous solution (if it matches length)
        previous = globals().get("height_sequence_1")
        if isinstance(previous, (list, tuple, np.ndarray)):
            try:
                prev_arr = np.asarray(previous, dtype=np.float64)
                if prev_arr.size == n and np.all(prev_arr >= 0):
                    candidates.append(prev_arr / prev_arr.sum())
            except Exception:
                pass

        # evaluate all deterministic candidates
        for seq in candidates:
            val = _objective(seq)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type=deterministic val={val:.6f}")
            # keep best per length
            cur_best_val_len, cur_best_seq_len = best_per_len.get(n, (float("inf"), None))
            if val < cur_best_val_len:
                best_per_len[n] = (val, seq.copy())

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling (several α values)
    # ------------------------------------------------------------------
    # allocate roughly half of remaining time to sampling
    now = time.time()
    time_for_sampling = (deadline - now) * 0.45
    sample_deadline = now + time_for_sampling

    alphas = [0.5, 1.0, 2.0]          # different concentration parameters
    # keep a small pool of the overall top candidates (max 6)
    top_pool = []

    while time.time() < sample_deadline:
        # bias towards larger n (more potential gain)
        probs = np.array(lengths, dtype=float)
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet_one(n, rng, alpha)
        val = _objective(seq)

        # update global best
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        # update per‑length best
        cur_val_len, cur_seq_len = best_per_len.get(n, (float("inf"), None))
        if val < cur_val_len:
            best_per_len[n] = (val, seq.copy())

        # maintain a small pool of the very best overall candidates
        top_pool.append((val, seq.copy()))
        top_pool.sort(key=lambda x: x[0])
        if len(top_pool) > 6:
            top_pool = top_pool[:6]

        # early exit if we already beat the target comfortably
        if global_best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – smoothed exponential‑gradient refinement
    # ------------------------------------------------------------------
    # pick a few promising candidates (global best + per‑length bests)
    refinement_candidates = [global_best_seq]
    for n, (val, seq) in best_per_len.items():
        refinement_candidates.append(seq)
    # also add the top few from the random pool
    for val, seq in top_pool:
        refinement_candidates.append(seq)

    # Ensure uniqueness (by id) and reasonable size
    uniq_candidates = []
    seen_ids = set()
    for seq in refinement_candidates:
        uid = id(seq)
        if uid not in seen_ids:
            seen_ids.add(uid)
            uniq_candidates.append(seq)

    # Run refinement on each candidate until time runs out
    for idx, cand in enumerate(uniq_candidates):
        if time.time() >= deadline:
            break
        n = len(cand)
        # give each candidate a slice of the remaining budget
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)
        refined_seq, refined_val = _refine_smooth(cand, sub_deadline)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[Overall‑Smooth‑Refine] candidate={idx} improved best to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (grader will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
