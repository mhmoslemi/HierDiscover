"""
Propose a step‑function minimising

    2·n·max(conv(a,a)) / (sum(a)²)

using a rich seed pool, a continuation‑L‑BFGS on a smooth‑max
surrogate, a fast mirror‑descent on the exact non‑smooth objective,
and a final lightweight Metropolis annealing.  The search works
with up to 8000 coefficients (adjustable) and respects a 1000 s
budget.  All helper functions are top‑level (no lambdas or closures)
and the code falls back to sub‑gradient only if SciPy is unavailable.
"""

import time
import math
import numpy as np

# Optional SciPy import – used for L‑BFGS
try:

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    n = seq.size
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _parabolic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - ((i - centre) / centre) ** 2
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _cosine_half(n: int) -> np.ndarray:
    """Cosine bump supported on the whole interval."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 0.5 * (1.0 + np.cos(np.pi * (i - centre) / centre))
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _mix_exponential(n: int, r_left: float, r_right: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = r_left ** i
    right = r_right ** (n - 1 - i)
    a = left + right
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    # Classical windows
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma in (0.05, 0.10, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma_factor=sigma))

    # Simple shapes
    seeds.append(_tent(n))
    seeds.append(_parabolic(n))
    seeds.append(_cosine_half(n))

    # Two‑sided exponential mixtures
    for r1, r2 in ((0.3, 0.3), (0.5, 0.2), (0.2, 0.5), (0.9, 0.1)):
        seeds.append(_mix_exponential(n, r1, r2))

    # Dirichlet draws
    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Previous construction (if any)
    if previous is not None and previous.shape == (n,):
        seeds.append(np.clip(previous, 0.0, None))
        if seeds[-1].sum() > 0.0:
            seeds[-1] = seeds[-1] / seeds[-1].sum()

    # Normalise and remove pathological entries
    clean = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            clean.append(s / tot)
    return clean


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation

        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v)))

    The gradient is w.r.t. the logits ``v``.
    """
    a = _softmax(v)                     # probability vector (∑a = 1)

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_val

    # weights for the weighted sum of conv entries
    w = exp_shifted / sum_exp

    # gradient w.r.t. a: d/da Σ wₖ·convₖ = 2·(rev(a) * w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through soft‑max
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                       # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 250) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector and its exact objective.
    """
    if not SCIPY_OPT:            # fallback – just return the input
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12,
                                         'disp': False})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient (mirror‑descent) refinement
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Stops early if time runs out.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        if active.size == 0:
            break

        # weight vector for active indices (average)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # exponentiated‑gradient style step
        lr = 0.6 / math.sqrt(it)          # diminishing learning rate
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)            # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk moving a small percentage of mass between two
    random coordinates, plus occasional Gaussian jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]          # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # Gaussian jump every 1000 steps
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    and returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem dimension.
    # Larger n gives more flexibility but costs more per iteration.
    # 6000–8000 is a good trade‑off under the 1000 s budget.
    # --------------------------------------------------------------
    n = 7000

    # --------------------------------------------------------------
    # Load previously discovered construction (if any).
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # --------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous)

    # --------------------------------------------------------------
    # 2) Quick screening – keep the most promising raw seeds.
    # --------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])        # lower is better
    top_k = 10
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # --------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 2e5, num=15)   # continuation schedule
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=300)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # --------------------------------------------------------------
    # 4) Exact sub‑gradient mirror‑descent refinement.
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=180_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # --------------------------------------------------------------
    # 5) Final Metropolis annealing (if any time left).
    # --------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=350_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # --------------------------------------------------------------
    # Fallback – should never be hit, but ensures a valid return.
    # --------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will evaluate it)
    return global_best_seq.tolist()
