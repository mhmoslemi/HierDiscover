"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Deterministic seeds** – a uniform vector, a Catalan‑shape (coefficients
   of (1‑z)^‑½) and a discretised Beta(½,½) (the arcsine law) are built for a
   few promising lengths.  Their objectives are evaluated and the best one
   is kept.

2. **Massive Dirichlet sampling** – while time permits we draw many
   Dirichlet(α) vectors for several lengths (2000 … 12000) and several
   concentration parameters α∈{0.2,0.5,0.8,1.0,2.0}.  The best sample is stored.

3. **Smoothed mirror‑descent refinement** – starting from the best vector,
   we minimise a *smoothed* version of the max‑convolution:
       fσ(a) = σ·log Σi exp(conv_i/σ)
   whose gradient is
       g = 2·(w * a)  with  w_i ∝ exp(conv_i/σ) .
   An exponentiated‑gradient (KL‑mirror descent) step
       log a ← log a – η·g
   is performed, with a diminishing stepsize η = η₀/√t and a temperature
   σ that shrinks proportionally to the current spread of the convolution.
   The iteration stops when the allocated time budget expires.

4. **Optional final sub‑gradient pass** – a few thousand iterations of the
   classic max‑convolution sub‑gradient (as used in the reference code) are
   run to polish the solution.

All operations are FFT‑based, so each iteration is O(n log n) and the
routine works comfortably within the 1000 s budget.  The final vector is
returned as a plain list of floats; it is already normalised (sum = 1)
and therefore scale‑invariant for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for moderate sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Returns the quantity that ``evaluate_sequence`` would compute once the
    vector is normalised to sum = 1:

        2·n·max(conv(seq,seq))

    where n = len(seq).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Truncated coefficients of (1‑z)^‑½:
        c_i = binom(2i,i) / 4**i
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _beta_05_05_shape(n: int) -> np.ndarray:
    """
    Discretised Beta(½,½) density (= arcsine law) on the midpoint grid.
    The density is proportional to 1/√(x·(1‑x)).
    """
    # mid‑points of the intervals (avoid the exact 0 and 1)
    i = np.arange(n) + 0.5
    x = i / n
    # pdf = 1/(π·√(x·(1‑x)))
    pdf = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    total = pdf.sum()
    if total == 0.0:
        return _uniform(n)
    return pdf / total


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet_alpha(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector (already normalised)."""
    if alpha <= 0.0:
        # fallback to uniform
        return _uniform(n)
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed mirror‑descent on the simplex
# ----------------------------------------------------------------------


def _smooth_refine(start_seq: np.ndarray, deadline: float,
                   eta0: float = 0.8, sigma_factor: float = 0.25,
                   eval_every: int = 2000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed surrogate of max(conv) using exponentiated‑gradient.
    Parameters
    ----------
    start_seq : np.ndarray
        Normalised (sum=1) start vector (non‑negative).
    deadline : float
        Absolute time after which the routine must stop.
    eta0 : float
        Initial stepsize multiplier (learning rate = eta0 / sqrt(t)).
    sigma_factor : float
        Controls the temperature: σ = sigma_factor·(max‑min of conv).
    eval_every : int
        How often to evaluate the true objective (for logging).
    Returns
    -------
    best_seq, best_val
        The best vector seen and its objective (2·n·max(conv)).
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1

        # --- convolution of a with itself ---------------------------------
        conv = _conv_fft(a, a)
        c_max = conv.max()
        c_min = conv.min()
        spread = c_max - c_min
        sigma = max(sigma_factor * spread, 1e-12)

        # --- softmax weights ----------------------------------------------
        # shift by the max to avoid overflow
        shifted = (conv - c_max) / sigma
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed objective ---------------------------
        # g = 2 * (w * a) ; we need the middle n entries of the full convolution
        conv_w_a = _conv_fft(w, a)                # length = 3n‑2
        start_idx = n - 1                         # position of index 0 of a in full conv
        g = 2.0 * conv_w_a[start_idx:start_idx + n]   # length n

        # --- exponentiated‑gradient update --------------------------------
        lr = eta0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        # numerical stabilisation
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # --- occasional true‑objective evaluation -------------------------
        if it % eval_every == 0 or time.time() + 0.5 > deadline:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[SmoothRefine] it={it:7d} n={n:5d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Classic (max‑only) sub‑gradient refinement (fallback)
# ----------------------------------------------------------------------


def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient sub‑gradient descent that only uses the
    currently active (maximum) convolution entry.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val
