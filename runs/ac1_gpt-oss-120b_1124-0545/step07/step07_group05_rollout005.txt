"""
Propose a step‑function (list of non‑negative heights) that minimizes

        2·n·max(conv(a,a)) / (∑ a)²

The algorithm is a lightweight “multi‑phase” optimiser:

1️⃣  Diverse seeds (uniform, sine‑window, geometric, power‑law,
    beta‑like, exponential, logistic, random Dirichlet, sparse, and a few
    random Gaussian‑mixture shapes).  All seeds are normalised to sum‑1.

2️⃣  Adam optimiser on a smooth (log‑sum‑exp) approximation of the
    ‑∞‑norm of the convolution.  The smoothing parameter β is increased
    slowly; the optimiser works on the soft‑max parameters v so that
    a = softmax(v) stays on the simplex.

3️⃣  Exact sub‑gradient mirror descent (exponentiated‑gradient) on the
    true max‑convolution objective.

4️⃣  A long simulated‑annealing (SA) phase with **incremental**
    convolution updates.  Two move types are used:
        – Dirichlet re‑allocation on 2‑4 random positions (large moves).
        – Small “ε‑transfer’’ from the index that contributes most to the
          current maximal convolution entry to a random target (fine‑grained
          moves).
    The temperature schedule is very slow (decay ≈ 0.9999) and the SA
    runs until the time budget is exhausted.

5️⃣  (Optional) Upsample the best 4 000‑point solution to a longer
    length (12 000) and repeat steps 3‑4 on the finer grid – this gives
    the optimiser extra degrees of freedom.

All phases respect the global 1000 s budget; the best sequence found
throughout the run is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _gaussian_mixture(n: int, rng: np.random.Generator,
                      components: int = 3) -> np.ndarray:
    """Random mixture of a few Gaussians (positive, smooth)."""
    xs = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(components):
        mu = rng.uniform(0, n)
        sigma = rng.uniform(n / 30, n / 5)
        amp = rng.uniform(0.1, 1.0)
        a += amp * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # systematic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (coarse)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # gentle geometric
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2.0
        seeds.append(_logistic_shape(n, offset, scale))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(1):
                seeds.append(_random_sparse(n, k, conc, rng))

    # random Gaussian mixtures
    for _ in range(3):
        seeds.append(_gaussian_mixture(n, rng, components=rng.integers(2, 5)))

    # reuse previous solution if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters -> softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0          # initial smoothness
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                 # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) at k_star
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)                # step‑size schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # avoid overflow
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    High‑frequency Simulated Annealing with incremental convolution updates.
    Two move types:
        • Dirichlet re‑allocation on 2‑4 random indices (large moves).
        • Small ε‑transfer from a high‑contribution index to a random target.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)       # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Temperature schedule
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9999

    it = 0
    while time.time() < deadline:
        it += 1
        # cooling
        T = max(T * decay, min_temp)

        # ----- choose move type -----
        move_type = rng.random()
        if move_type < 0.5:
            # ---- large Dirichlet move on 2‑4 indices ----
            k = rng.integers(2, 5)                     # 2‑4 positions
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            deltas = new_alloc - cur[idx]
        else:
            # ---- small ε‑transfer (targeted) ----
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            # contributions a[i]*a[k_star-i]
            contrib = cur[support] * cur[k_star - support]
            if contrib.sum() == 0.0:
                src = rng.choice(support)
            else:
                src = rng.choice(support, p=contrib / contrib.sum())
            tgt = rng.integers(0, n - 1)
            while tgt == src:
                tgt = rng.integers(0, n - 1)

            eps = min(cur[src] * 0.2, cur[src] * rng.uniform(0.05, 0.2))
            if eps <= 0.0:
                continue
            idx = np.array([src, tgt], dtype=int)
            new_alloc = cur[idx].copy()
            new_alloc[0] -= eps
            new_alloc[1] += eps
            deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ term
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for i in range(idx.size):
            for j in range(i + 1, idx.size):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, followed by normalisation."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    # make sure we stay non‑negative and renormalise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    return a_new / s if s > 0 else _uniform(new_n)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # try to reuse a previous solution if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev = None
    else:
        prev = None

    # ------------------------------------------------------------------
    # Phase 0 : optimise a base length (4000) – this is where most of
    #            the time is spent.
    # ------------------------------------------------------------------
    base_n = 4000
    print(f"\n=== Base optimisation length {base_n} ===")
    seeds = _make_seeds(base_n, prev, rng)

    # quick evaluation of raw seeds
    global_best_seq = None
    global_best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # Adam optimisation on the most promising seeds
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda t: t[1])
    top_k = min(4, len(seed_vals))

    for s, _ in seed_vals[:top_k]:
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, base_n, deadline, max_iters=30000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑Adam] new global best={val:.6f}")

    # Exact sub‑gradient refinement
    if global_best_seq is not None:
        seq, val = _subgradient_refine(global_best_seq, base_n, deadline,
                                       max_iters=800000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SubGrad] new global best={val:.6f}")

    # High‑frequency Simulated Annealing (runs until time is almost up)
    if global_best_seq is not None:
        # allocate remaining time for SA on the base length
        seq, val = _sa_incremental_optimize(
            global_best_seq,
            base_n,
            deadline,
            init_temp_factor=0.02
        )
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SA] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 : optional up‑sample to a larger length and fine‑tune.
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:   # keep at least a few seconds for final check
        larger_n = 12000
        print(f"\n=== Upsample to length {larger_n} and fine‑tune ===")
        up_seq = _upsample(global_best_seq, larger_n)

        # sub‑gradient on the larger grid (shorter budget)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.4)
        seq, val = _subgradient_refine(up_seq, larger_n, sub_deadline,
                                       max_iters=400000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[up‑SubGrad] new global best={val:.6f}")

        # SA on the larger grid (uses whatever time remains)
        if time.time() < deadline:
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                larger_n,
                deadline,
                init_temp_factor=0.03
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[up‑SA] new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Final verification (optional) and return
    # ----------------------------------------------------------------------
    # Ensure the returned sequence is a plain Python list of floats.
    final_seq = global_best_seq.tolist()
    try:
        _ = evaluate_sequence(final_seq)   # sanity check – ignore the result
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return final_seq
