def evaluate_sequence(sequence):
    # security checks, convert to float, clamp to [0,1000]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence) # full convolution length 2n-1
    max_b = max(b_sequence)

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

The algorithm builds a large, diverse set of seeds (analytic families,
random binary patterns, sparse spikes, warm‑start).  It then performs a
sequence of time‑aware optimisation phases:

1. Adam on a smooth‑max surrogate (log‑softmax parametrisation) with a
   rapidly increasing temperature to focus on the true max.
2. Exact sub‑gradient descent (mirror‑descent with KL‑projection)
   on the true infinity‑norm.
3. L2‑flattening (driving the auto‑convolution towards a constant).
4. Targeted stochastic hill‑climb.
5. A final simulated‑annealing fine‑tuner that perturbs the log‑vector
   with a temperature schedule.

All phases respect a global 1000 s budget.  The best vector found is
stored in the global variable ``height_sequence_1`` for warm‑starting
future calls.
"""

import time
import math
import numpy as np

# Low‑level helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using an FFT of power‑of‑two length."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Objective 2·n·max(conv(a, a)) assuming sum(a) == 1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)

def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # --- Analytic families ------------------------------------------------
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())                                 # half‑sine

    # geometric
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # power‑law
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # beta‑like
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(5):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal (Gaussian bell) shapes
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # --- Additional random families --------------------------------------
    # random binary masks with various densities
    for dens in (0.2, 0.3, 0.5, 0.7):
        mask = (rng.random(n) < dens).astype(np.float64)
        if mask.sum() > 0:
            seeds.append(mask / mask.sum())

    # sparse spikes (few delta‑like masses)
    for _ in range(5):
        k = max(2, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        pos = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[pos] = 1.0
        seeds.append(a / a.sum())

    # random walk‑style profiles (cumulative sum of flipped signs)
    for _ in range(5):
        steps = rng.choice([-1, 1], size=n)
        a = np.cumsum(steps).astype(np.float64)
        a -= a.min()
        a += 1e-12
        seeds.append(a / a.sum())

    # warm‑start (if compatible)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam with a smooth‑max surrogate.
    Logits (v) parametrize a = softmax(v).  The temperature β starts larger
    and is multiplied by 1.03 each iteration, quickly sharpening the
    surrogate toward the true max.
    """
    v = np.log(a0 + 1e-30)        # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 10.0            # initial smoothness (larger than in the reference)
    beta_mul = 1.03

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # a = softmax(v)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (mirror‑descent in log‑space)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on the simplex)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Drive a*a towards a constant vector (target μ = 1/(2n‑1)).
    Performs exponentiated‑gradient on the L2 objective.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny fraction of mass from a
    donor (chosen proportionally to its contribution to the current
    maximal convolution entry) to a random receiver.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


def _simulated_annealing(a0: np.ndarray, n: int,
                         budget: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained stochastic optimiser.  Work in log‑space,
    perturb with a Gaussian of decreasing scale, and accept
    uphill moves with a Boltzmann probability.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    end = start + budget

    # temperature / perturbation schedule
    T_start = 0.05
    T_end = 1e-5

    while time.time() < end:
        t_frac = (time.time() - start) / budget
        # exponential decay of temperature
        T = T_start * (T_end / T_start) ** t_frac
        sigma = max(T, 1e-5)  # perturbation magnitude

        # propose a new point by multiplicative log‑Gaussian noise
        proposal = cur * np.exp(sigma * rng.normal(size=n))
        proposal = np.maximum(proposal, 0.0)
        s = proposal.sum()
        if s == 0.0:
            continue
        proposal /= s

        val = _objective(proposal, n)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur = proposal
            cur_val = val
            if val < best_val:
                best_val = val
                best = proposal.copy()
                print(f"[SA] new best {best_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  Returns the best sequence found as a plain list
    of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length (larger n tightens the theoretical bound)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase_deadline = time.time() + remaining * 0.40
        seq, val = _adam_optimize(global_best_seq, n, phase_deadline,
                                  max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase_deadline = time.time() + remaining * 0.30
        seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline,
                                         max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L2‑flattening descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.20
        seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline,
                                        max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑L2Flat] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.10
        seq, val = _hill_climb_optimize(global_best_seq, n, phase_deadline,
                                        max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (fine‑tuning)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, val = _simulated_annealing(global_best_seq, n, remaining)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑SA] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
