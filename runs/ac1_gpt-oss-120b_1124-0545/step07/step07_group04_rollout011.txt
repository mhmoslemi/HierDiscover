def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose problem dimension
    n = 6000
    # previous sequence if present
    prev = globals().get('height_sequence_1')
    if prev is not None:
        prev = np.asarray(prev, dtype=np.float64)
    # build seeds for this n
    seeds = _make_seed_pool(n, prev)   # we can adapt _make_seed_pool to produce many more random seeds.
    # optionally augment with random Dirichlet draws many times
    rng = np.random.default_rng(seed+123)
    for _ in range(30):
        seeds.append(rng.dirichlet(np.full(n, 0.5)))
    # also add low-entropy sparse seeds: choose k nonzeros
    for k in [10, 20, 30, 50, 100]:
        a = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.random(k)
        a /= a.sum()
        seeds.append(a)
    
    # evaluate raw seeds and keep top few
    scores = []
    for s in seeds:
        val = _objective(s)
        scores.append((s, val))
    scores.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [s for s, _ in scores[:top_k]]
    best_seq = scores[0][0].copy()
    best_val = scores[0][1]
    print(f"[seed] best raw: {best_val:.6f}")

    beta_sched = np.geomspace(1.0, 20000.0, num=16)
    # L-BFGS
    for seed_vec in top_seeds:
        if time.time() > deadline: break
        seq, val = _l_bfgs_optimize(seed_vec, n, deadline, beta_sched, maxiter_per_stage=400)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[L‑BFGS] new best {val:.6f}")

    # now maybe run a second L-BFGS with higher beta from best_seq
    if time.time() < deadline:
        # high beta schedule
        beta_sched2 = np.geomspace(20000.0, 200000.0, num=8)
        seq, val = _l_bfgs_optimize(best_seq, n, deadline, beta_sched2, maxiter_per_stage=400)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[L‑BFGS2] new best {val:.6f}")

    # subgradient refinement
    if time.time() < deadline:
        seq, val = _subgradient_refine(best_seq, n, deadline, max_iters=500_000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] refined best {val:.6f}")

    # random perturbation restarts
    # Run small perturbation loops if time remains:
    while time.time() < deadline - 0.2:
        # create a perturbed copy
        pert = best_seq.copy()
        # move some mass randomly
        rng = np.random.default_rng()
        # choose fraction of entries to perturb
        frac = 0.05
        idxs = rng.choice(n, size=int(frac*n), replace=False)
        # random reallocate mass among them
        mass = pert[idxs].sum()
        pert[idxs] = rng.random(idxs.shape[0])
        pert[idxs] /= pert[idxs].sum()
        pert[idxs] *= mass
        pert /= pert.sum()
        # refine perturbed via subgradient (short)
        seq2, val2 = _subgradient_refine(pert, n, deadline, max_iters=100_000)
        if val2 < best_val:
            best_val = val2
            best_seq = seq2.copy()
            print(f"[Perturb] new best {val2:.6f}")

    # final annealing
    if time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=300_000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Anneal] final best {val:.6f}")

    # final info
    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()

"""
Utility to search for a non‑negative step‑function that minimises
2·n·max(conv(a,a))/(sum(a)²).

The strategy:
  1. Build a diverse pool of normalised seed vectors (classic windows,
     geometric/power families, random Dirichlet draws and a few sparse
     constructions).
  2. Keep the K best raw seeds (by the exact objective) and run a
     multi‑stage smooth‑max L‑BFGS optimisation on each of them.
  3. Refine the current best with a mirror‑descent exponentiated‑gradient
     sub‑gradient method that works on the exact non‑smooth objective.
  4. Apply a short Metropolis‑type annealing walk to escape possible
     local minima.
  5. A final high‑β L‑BFGS stage is performed if time permits.
All stages respect the supplied time budget (default 1000 s) and the
function finally returns the best sequence found as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for L‑BFGS and additional windows)
# ----------------------------------------------------------------------
try:
    from scipy import optimize, signal
    SCIPY_OPT = True
except Exception:   # pragma: no cover
    SCIPY_OPT = False
    signal = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (length = len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Window families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _make_seed_pool(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large diverse collection of normalised start vectors."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # trivial uniform
    seeds.append(_uniform(n))

    # geometric decays (r from 0.2 to 0.99)
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))

    # symmetric exponential decay from centre
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # combined geometric ∗ power‑law
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    # classic spectral windows
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_gaussian(n, sigma_factor=0.05))
    seeds.append(_gaussian(n, sigma_factor=0.10))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_gaussian(n, sigma_factor=0.30))

    # Dirichlet random draws with various concentrations
    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse blocks (few non‑zeros)
    for k in (10, 20, 30, 50, 100):
        a = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.random(k)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
            seeds.append(a)

    # add a previous construction if it matches the dimension
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ------------------------------------------------------------------
    # Additional SciPy windows (if SciPy available)
    # ------------------------------------------------------------------
    if SCIPY_OPT and signal is not None:
        win = signal.windows
        # Kaiser with a few betas
        for beta in (2, 4, 6, 8, 10, 12, 14, 16):
            try:
                w = win.kaiser(n, beta)
                w = np.asarray(w, dtype=np.float64)
                if w.sum() > 0:
                    seeds.append(w / w.sum())
            except Exception:
                pass
        # triangular / bartlett (same shape)
        for fn in (win.triang, win.bartlett, win.parzen,
                   win.blackmanharris, win.nuttall, win.flattop, win.cosine):
            try:
                w = fn(n)
                w = np.asarray(w, dtype=np.float64)
                if w.sum() > 0:
                    seeds.append(w / w.sum())
            except Exception:
                pass

    # clean‑up – clip tiny negatives (numerical noise) and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective & gradient used by L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    ``beta`` > 0. The gradient w.r.t. the logits ``v`` is supplied.
    """
    a = _softmax(v)                     # probability vector (sum = 1)

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)
    # shift for numerical stability
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # factor 2·n from the definition

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    #   d/da Σₖ wₖ·convₖ(a,a) = 2·(rev(a) * w)   (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                 # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 250) -> tuple[np.ndarray, float]:
    """
    Run a multi‑stage L‑BFGS optimisation on the smooth‑max surrogate.
    Returns the best probability vector found together with its exact objective.
    """
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinements (exponentiated‑gradient – mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Stops when time runs out or ``max_iters`` is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient of max – average over all active indices
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final lightweight polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between
    two random coordinates.  Used only if time is left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

"""
Utility to search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in several stages:

1. Build a large, diverse pool of normalised seed vectors (geometric,
   power‑law, classic spectral windows, random Dirichlet draws and a few
   sparse constructions).  If a previous best (`height_sequence_1`) is
   available it is added to the pool.
2. Evaluate the exact objective on every seed and keep the few best
   raw seeds.
3. Run a multi‑stage smooth‑max L‑BFGS optimisation on each of those
   seeds.  The smooth‑max (log‑sum‑exp) surrogate gradually sharpens
   (`β` grows) so the optimiser eventually targets the true max‑conv.
4. Perform a short high‑β L‑BFGS pass on the current best (time‑permitting).
5. Refine the best vector with a mirror‑descent (exponentiated‑gradient)
   sub‑gradient method that works on the exact non‑smooth objective.
6. Occasionally perturb the best vector and run a brief sub‑gradient
   refinement to escape possible local minima.
7. If time remains, finish with a lightweight Metropolis annealing walk.
All stages respect the supplied time budget (default 1000 s) and the

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

        seeds.append(_geom(n, r))
        seeds.append(_sym_geom_centered(n, r))
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    # classic spectral windows
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    seeds.append(_gaussian(n, sigma_factor=0.05))
    seeds.append(_gaussian(n, sigma_factor=0.10))
    seeds.append(_gaussian(n, sigma_factor=0.18))
    seeds.append(_gaussian(n, sigma_factor=0.30))

    # random Dirichlet draws
    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few sparse constructions (few non‑zeros)
    for k in (10, 20, 30, 50, 100):
        a = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.random(k)
        tot = a.sum()
        if tot > 0:
            a /= tot
            seeds.append(a)

    # previous best, if compatible in size
    if previous is not None and isinstance(previous, np.ndarray) and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # additional SciPy windows (if SciPy is present)
    if SCIPY_OPT and signal is not None:
        win = signal.windows
        for beta in (2, 4, 6, 8, 10, 12, 14, 16):
            try:
                w = win.kaiser(n, beta)
                w = np.asarray(w, dtype=np.float64)
                if w.sum() > 0:
                    seeds.append(w / w.sum())
            except Exception:
                pass
        for fn in (win.triang, win.bartlett, win.parzen,
                   win.blackmanharris, win.nuttall, win.flattop, win.cosine):
            try:
                w = fn(n)
                w = np.asarray(w, dtype=np.float64)
                if w.sum() > 0:
                    seeds.append(w / w.sum())
            except Exception:
                pass

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """Value and gradient of the smooth‑max surrogate."""
    a = _softmax(v)                     # probability vector (sum=1)

    conv = _conv_fft(a, a)              # length 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val

    w = exp_shifted / sum_exp            # weights for the soft‑max

    # gradient w.r.t. a : 2 * (rev(a) * w)  (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through softmax (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 250):
    """Multi‑stage L‑BFGS on the smooth‑max surrogate."""
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent / exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000):
    """Mirror‑descent on the exact non‑smooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient = average of gradients of all active entries
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)            # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000):
    """Small Metropolis walk that swaps a tiny amount of mass between two positions."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump to shake the state
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point – the grader will call this
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4000 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous construction, if present.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a large pool of seed vectors.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])    # lower is better
    top_k = 8
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the selected seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 5000.0, num=12)
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=300)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional high‑β L‑BFGS refinement (if time permits).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        beta_high = np.geomspace(5000.0, 50000.0, num=6)
        seq, val = _l_bfgs_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    beta_schedule=beta_high,
                                    maxiter_per_stage=250)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[high‑β L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Exact sub‑gradient mirror‑descent refinement.
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=250_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 6) Random perturbations + short sub‑gradient runs to escape local minima.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        rng = np.random.default_rng()
        pert = global_best_seq.copy()
        frac = 0.08
        k = max(1, int(frac * n))
        idxs = rng.choice(n, size=k, replace=False)
        mass = pert[idxs].sum()
        pert[idxs] = rng.random(k)
        pert[idxs] /= pert[idxs].sum()
        pert[idxs] *= mass
        pert /= pert.sum()

        seq, val = _subgradient_refine(pert,
                                       n,
                                       deadline,
                                       max_iters=80_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[perturb‑SubGrad] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 7) Final Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=250_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse by a later call.
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as a plain list (the grader will evaluate it).
    return global_best_seq.tolist()
