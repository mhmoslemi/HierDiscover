"""
Optimise a non‑negative step function (vector a) to minimise

    2·n·max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The optimisation proceeds in four
phases that respect a total runtime of ≈1000 s:

1. **Seed generation & quick screening** – a large, diverse collection of
   L¹‑normalised vectors (analytic shapes, Dirichlet draws, previous
   best) is created and the best seed is kept.

2. **Adam on a smooth‑max surrogate** – a soft‑max (temperature β) is
   applied to the convolution values, the gradient is back‑propagated
   through the convolution and the parameters are optimised with Adam.
   The temperature is increased very slowly so that the algorithm ends
   up minimising the true maximum.

3. **Projected sub‑gradient descent on the true max** – the current
   maximal convolution entry `k` is identified; its exact sub‑gradient
   is `g_i = 2·a_{k‑i}` (where defined).  After a simple step we
   project back onto the probability simplex (Euclidean projection).

4. **L₂‑flattening** – minimise ½‖a∗a − μ·1‖² with μ=1/(2n‑1) using the
   same exponentiated‑gradient trick; this drives the auto‑convolution
   towards a constant profile.

5. **Balanced stochastic hill‑climb** – when a maximal entry is
   identified its contributing pairs `(i, k‑i)` are located.  Mass is
   moved from a heavily‑contributing donor to a receiver whose partner
   at the same shift is tiny, thereby directly reducing the maximum
   without inflating other entries.

The best vector found is stored in the global variable
`height_sequence_1` for warm‑starting future calls and is also returned
as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Seed generation (diverse analytic and random families)
# ----------------------------------------------------------------------


def _make_seeds(
    n: int, rng: np.random.Generator, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Half‑sine window
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # Geometric families
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # Power‑law families
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 3.0, 10.0):
        for _ in range(4):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(6):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(6):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal (Gaussian bell) shapes
    for _ in range(6):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # Warm‑start from the previous global best, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

def _adam_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).  The temperature β is grown
    multiplicatively (slowly) so that the surrogate converges to the true max.
    """
    v = np.log(a0 + 1e-30)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 1.5            # start rather smooth
    beta_mul = 1.015      # very slow increase → smoother convergence

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _proj_subgrad_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max.
    After each step we project back onto the probability simplex.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ProjSub] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – diminishing
        lr = 0.6 / np.sqrt(it)
        a_new = a - lr * g

        # project onto simplex (guarantees non‑negativity and sum=1)
        a = _project_simplex(a_new)

    return best_seq, best_val


def _l2_flatten_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a∗a – μ·1‖² with μ = 1/(2n‑1) using exponentiated‑gradient.
    This drives the auto‑convolution towards a constant vector.
    """
    a = a0.copy() / a0.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _balanced_hill_climb_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that deliberately reduces the current maximal
    convolution entry.  Mass is taken from a pair that heavily contributes
    to the max and given to a receiver whose partner at the same shift is
    small, thereby lowering the max without inflating other entries.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # indices that actually contribute to this entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        # donor – weighted by contribution (the larger the product, the likelier)
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        # partner of the donor for this shift (might be out of range)
        partner_of_donor = max_idx - donor
        if not (0 <= partner_of_donor < n):
            partner_of_donor = None

        # choose a receiver that has a *small* partner at the same shift
        # (so the new contribution to the maximal entry stays tiny)
        partner_vals = np.full(n, np.inf)
        partner_idx = max_idx - np.arange(n)
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner_vals[mask] = cur[partner_idx[mask]]

        # avoid picking the donor itself
        candidate_mask = np.arange(n) != donor
        candidate_idxs = np.where(candidate_mask)[0]

        # inverse‑weight by partner value (add small epsilon for stability)
        inv = 1.0 / (partner_vals[candidate_idxs] + 1e-12)
        probs_recv = inv / inv.sum()
        receiver = rng.choice(candidate_idxs, p=probs_recv)

        # amount of mass to shift
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[BalancedHill] it={it:6d} best={val:.6f}")

        # slowly decay step scale
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Length choice (larger n gives a tighter theoretical bound)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    # extra Dirichlet draws with a finer grid of concentration parameters
    for conc in np.logspace(-2, 2, 8):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.45
    seq, val = _adam_optimize(
        global_best_seq, n, phase_deadline, max_iters=500_000
    )
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on true max
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.25
    seq, val = _proj_subgrad_optimize(
        global_best_seq, n, phase_deadline, max_iters=300_000
    )
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑ProjSub] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L2 flattening towards a constant autocorrelation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.15
    seq, val = _l2_flatten_optimize(
        global_best_seq, n, phase_deadline, max_iters=300_000
    )
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Informed stochastic hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _balanced_hill_climb_optimize(
            global_best_seq, n, phase_deadline, max_iters=300_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
