def solve_lp_max_sum(f, M, time_limit):
    n = len(f)

   descent we solve a small LP:
       maximise   Σ b_j
       s.t.       (f * b)_k ≤ M   for every lag k
                  b_j ≥ 0
   where `M` is the current maximum of `f * f`.  The solution concentrates
   mass where `f` is already small, guaranteeing a strict reduction of the
   objective after a short convex combination with the current `f`.

4. **Final stochastic hill‑climbing** – a light simulated‑annealing phase
   (random Dirichlet reshuffles of a few coordinates) cleans up any remaining
   irregularities.

All three stages respect the global 1000 s time‑budget.  The implementation
uses only NumPy and SciPy (the `highs` LP solver) and avoids closures or
lambdas as required.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)).  Assumes Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------


def _subgradient_refine_long(start_seq: np.ndarray,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact mirror‑descent on the true max‑convolution objective.
    Runs until the deadline is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * len(a) * conv[max_idx]

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (len(a) - 1))
        i_end = min(len(a) - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update with a gentler schedule
        lr = 0.5 / np.sqrt(it)          # smaller than the original 0.7
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming mass‑reallocation step
# ----------------------------------------------------------------------


def _lp_reallocate(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   (f * b)_k ≤ M   for all k,
    b_j ≥ 0.  Returns the optimal b (not yet normalised) or None on failure.
    """
    n = len(f)
    L = 2 * n - 1

    # Build the full convolution matrix A (shape L × n) where
    #   A[k, j] = f_{k-j}  if 0 ≤ k-j < n else 0.
    # This is exactly a set of shifted copies of `f`.
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n, dtype=np.float64)            # minimise -sum b  -> maximise sum b
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Use HiGHS (the default in recent SciPy) – it is fast for these sizes.
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=bounds, method='highs')
    except Exception as exc:
        print("[LP] linprog raised:", exc)
        return None

    if not res.success:
        # LP can fail for numerically tight bounds – treat as a no‑op.
        print("[LP] optimisation failed:", res.message)
        return None

    return res.x


def _lp_step(f: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based re‑allocation followed by a small convex combination.
    Returns the new sequence and its objective value.
    """
    n = len(f)
    conv = _conv_fft(f, f)
    M = np.max(conv)
    b_raw = _lp_reallocate(f, M)

    if b_raw is None:
        return f, _objective(f)

    # Normalise the LP solution to a probability vector.
    b_sum = b_raw.sum()
    if b_sum <= 0.0:
        return f, _objective(f)
    g = b_raw / b_sum

    # Convex combination.  We use a modest step size and shrink it if needed.
    t = 0.08
    while t > 1e-4 and time.time() < deadline:
        new_f = (1.0 - t) * f + t * g
        new_val = _objective(new_f)
        if new_val < _objective(f):
            return new_f, new_val
        t *= 0.6   # back‑off
    return f, _objective(f)


# ----------------------------------------------------------------------
# Simulated annealing for final polishing
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)          # 2 … 6 indices
        idx = rng.choice(len(cur), size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve the best sequence from a possible previous run.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and previous:
        try:
            base_seq = np.asarray(previous, dtype=np.float64)
            if base_seq.size > 0 and base_seq.sum() > 0.0:
                base_seq = base_seq / base_seq.sum()
                print("[Info] Loaded previous sequence (length "
                      f"{len(base_seq)})")
            else:
                raise ValueError
        except Exception:
            base_seq = None
    else:
        base_seq = None

    # If no previous sequence, fall back to a deterministic seed.
    if base_seq is None:
        n = 4000
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(i - centre) / centre
        a = np.maximum(a, 0.0)
        a /= a.sum()
        base_seq = a
        print("[Info] No previous solution – using triangular seed.")
    else:
        n = len(base_seq)

    # ------------------------------------------------------------------
    # Stage 1 – long mirror‑descent (exponentiated‑gradient).
    # ------------------------------------------------------------------
    print(f"\n--- Stage 1: mirror‑descent (n={n}) ---")
    seq_mid, val_mid = _subgradient_refine_long(base_seq, deadline)
    print(f"[Stage1] Best objective after descent = {val_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP re‑allocation steps (few, but powerful).
    # ------------------------------------------------------------------
    print("\n--- Stage 2: LP re‑allocation ---")
    while time.time() < deadline - 30.0:          # leave ~30 s for SA
        seq_mid, val_mid = _lp_step(seq_mid, deadline)
        print(f"[LP] New objective = {val_mid:.6f}")
        # If the improvement is tiny we can stop early.
        if val_mid < 1.5:
            # Almost at the target – break to avoid needless LP solves.
            break

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing polishing.
    # ------------------------------------------------------------------
    print("\n--- Stage 3: simulated annealing ---")
    seq_final, val_final = _simulated_annealing(seq_mid, deadline)

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence.
    # ------------------------------------------------------------------
    try:
        from __main__ import evaluate_sequence  # the host injects it
    except Exception:
        # In the execution environment evaluate_sequence is already in globals().
        evaluate_sequence = globals().get("evaluate_sequence")
    if evaluate_sequence is not None:
        try:
            _ = evaluate_sequence(seq_final.tolist())
        except Exception as exc:
            print("Warning: evaluate_sequence raised an exception:", exc)

    print("\n=== Finished ===")
    print(f"Best objective ≈ {val_final:.6f} (length {len(seq_final)})")

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = seq_final.tolist()
    return seq_final.tolist()
