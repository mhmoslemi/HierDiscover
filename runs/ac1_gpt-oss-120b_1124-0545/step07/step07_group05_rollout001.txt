"""
Improved optimiser for the step‑function problem.

The routine builds a diverse pool of normalised seeds (including uniform,
half‑sine, geometric, power‑law, beta‑like, exponential, logistic,
Gaussian, Tukey and Kaiser windows).  The most promising seeds are
optimised with Adam on a smooth log‑sum‑exp approximation of the
max‑convolution, followed by a long exponentiated‑gradient (sub‑gradient)
refinement.  Finally a high‑frequency simulated‑annealing phase with
incremental convolution updates explores the neighbourhood intensively.
All phases respect a global 1000 s budget and the best sequence (normalised
to sum = 1) is returned as a plain ``list[float]``.
"""

import time
from typing import List, Tuple, Optional

import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a simplex‑normalised vector (∑a = 1):
        2·n·max(conv(a,a))
    The factor ∑a² is absorbed because the denominator in
    ``evaluate_sequence`` equals 1 after normalisation.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_centered(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """
    Tukey window with shape parameter alpha in [0,1].
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    width = n / 2.0
    r = np.abs(i - centre) / width
    a = np.where(
        r < alpha / 2,
        0.5 * (1 + np.cos(np.pi * (2 * r / alpha - 1))),
        np.where(r <= 1, 0.5 * (1 + np.cos(np.pi * (r - alpha / 2) / (1 - alpha / 2))), 0.0),
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """
    Kaiser window (beta >= 0). Uses scipy's modified Bessel function i0.
    """
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    a = i0(beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)) / i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a k‑sparse Dirichlet."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _make_seeds(
    n: int,
    previous: Optional[np.ndarray],
    rng: np.random.Generator,
) -> List[np.ndarray]:
    """
    Assemble a diverse list of normalised seeds.
    """
    seeds: List[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families – coarse
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))

    # geometric – gentle decay
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))

    # power‑law
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # asymmetric beta‑like
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2
        seeds.append(_logistic_shape(n, offset, scale))

    # Gaussian windows – several widths
    for sigma in (n / 8, n / 6, n / 4):
        seeds.append(_gaussian_centered(n, sigma))

    # Tukey windows – several tapers
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_tukey_window(n, alpha))

    # Kaiser windows – a few beta values
    for beta in (2.0, 4.0, 6.0):
        seeds.append(_kaiser_window(n, beta))

    # random Dirichlet draws
    for conc in (0.3, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> Tuple[Optional[np.ndarray], float]:
    """Adam optimisation on a log‑sum‑exp approximation of the max‑convolution."""
    eps = 1e-30
    v = np.log(seed + eps)  # parameters of a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 5.0               # initial smoothness
    beta_mul = 1.03          # grow a bit faster than the original code

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:d} it={it:d} best={cur_val:.6f}")

        # ---- smooth max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # stability shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
) -> Tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:d} it={it:d} best={cur_val:.6f}")

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
    max_iters: int = 2_000_000,
    decay: float = 0.9995,
) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing with cheap incremental convolution updates.
    Two move families:
        – Dirichlet re‑allocation on 2‑6 random indices,
        – targeted moves that touch indices contributing to the current maximal entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * decay, min_temp)

        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if rng.random() < 0.5:
            # random Dirichlet re‑allocation on 2‑6 indices
            k = rng.integers(2, 7)                     # 2‑6 indices
            idx = rng.choice(n, size=k, replace=False)
        else:
            # targeted move – focus on indices that appear in the current maximal convolution entry
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # weight each index by its contribution a[i]·a[k_star‑i]
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation on the selected positions (Dirichlet)
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ------------------------------------------------------------------
        # Incremental convolution update (see original implementation)
        # ------------------------------------------------------------------
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a  (commutative)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ term (self‑product)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for i in range(idx.size):
            for j in range(i + 1, idx.size):
                pos = idx[i] + idx[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:d} it={it:d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget and
    returns the best sequence discovered (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin so the outer harness never sees a timeout
    deadline = start_time + budget_s - 3.0

    # Retrieve a possibly‑existing solution to seed the search
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths – we test a few sizes; the best across them is kept.
    lengths = [2000, 4000, 6000, 8000, 10000]

    global_best_seq: Optional[np.ndarray] = None
    global_best_val: float = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # --------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # --------------------------------------------------------------
        for s in seeds:
            if time.time() > deadline:
                break
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Adam on smoothed max (top‑k seeds only)
        # --------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(6, len(seed_vals))
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (pre‑SA)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=700000
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – High‑frequency Simulated Annealing
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            # estimate a generous number of SA moves; we cap it to avoid runaway
            target_iters = int(min(5_000_000, remaining * 17000))
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                n,
                deadline,
                init_temp_factor=0.02,
                min_temp=1e-12,
                max_iters=target_iters,
                decay=0.9998,
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – A short post‑SA sub‑gradient polish
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=250000
            )
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[post‑SubGrad] n={n:5d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Ensure the sequence sums to (approximately) 1 – this makes the
    # ``evaluate_sequence`` call interpretable as the same objective.
    total = global_best_seq.sum()
    if total == 0.0:
        global_best_seq = _uniform(global_best_seq.shape[0])
    else:
        global_best_seq = global_best_seq / total

    # Final verification (optional)
    try:
        ev = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nFinal evaluate_sequence objective ≈ {ev:.6f} "
              f"(length {len(global_best_seq)})")
    except Exception as exc:
        print(f"Evaluation error (ignored): {exc}")

    return global_best_seq.tolist()
