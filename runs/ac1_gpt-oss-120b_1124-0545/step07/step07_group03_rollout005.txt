def _optimize_len(n, deadline, rng):

    # initial u: log of Catalan shape normalized
    a0 = _catalan_shape(n)  # shape normalized
    u = np.log(a0 + 1e-12)

    # Adam states
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a0.copy()
    best_obj = _objective(a0)  # 2*n*max(conv)

    beta = 5.0
    beta_inc = 1.02
    beta_max = 60.0
    iter_num = 0

    while time.time() < deadline:
        iter_num += 1

        a = np.exp(u - np.max(u))  # compute softmax (more stable)
        a /= a.sum()
        # compute conv
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        obj = 2.0 * n * max_conv
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            # optionally print progress
            print(f"[Opt n={n} it={iter_num} obj={obj:.6f} beta={beta:.2f}")

        # gradient approx via smoothed max
        # compute smoothed weights
        scaled = beta * conv
        shift = np.max(scaled)
        w = np.exp(scaled - shift)
        Z = w.sum()
        w /= Z

        # compute gradient of M w.r.t a
        w_rev = w[::-1]
        conv_grad = _conv_fft(a, w_rev)  # full conv length 3n-2
        # slice
        slice_start = n - 1
        slice_end = 2 * n - 1
        g = conv_grad[slice_start:slice_end][::-1]
        grad_M = 2.0 * g  # as derived

        # gradient of O = 2*n* M, ignore constant factor if not needed
        grad_O_a = 2.0 * n * grad_M  # O = 2n M

        # gradient w.r.t u via softmax chain rule
        # dO/du_i = a_i * (grad_O_a_i - dot(a, grad_O_a))
        dot_ag = np.dot(a, grad_O_a)
        grad_u = a * (grad_O_a - dot_ag)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** iter_num)
        v_hat = v / (1 - beta2 ** iter_num)
        u -= 0.1 * m_hat / (np.sqrt(v_hat) + eps)

        # anneal beta
        if iter_num % 2000 == 0:
            beta = min(beta * beta_inc, beta_max)

    return best_a, best_obj

"""
Propose a candidate height sequence that minimizes the given
``evaluate_sequence`` objective.

Algorithm – “smooth‑max mirror‑descent”:

1. **Parameterisation** – a probability vector `a` (non‑negative,
   sum = 1) is written as a soft‑max of unconstrained variables `u`,
   guaranteeing feasibility at every step.

2. **Smooth approximation of the max** – instead of the non‑smooth
   `max(conv(a,a))` we minimise the log‑sum‑exp
   `M_β = (1/β)·log Σ exp(β·conv)`.  As `β` grows the quantity converges
   to the true maximum.  A slowly increasing schedule for `β`
   (starting at 5, ending ≈ 60) lets the optimiser first see a smooth
   landscape and later focus on the true peak.

3. **Exact gradient** – for a fixed `β`

      w_k = exp(β·conv_k) / Σ exp(β·conv)

   and `∂M_β/∂a_i = 2·Σ_k w_k·a_{k‑i}`.
   The sum is a (restricted) convolution and is evaluated with a
   single FFT per iteration.

4. **Mirror descent on the simplex** – we use Adam (adaptive
   moment estimation) on the unconstrained variables `u`.  The chain
   rule for the soft‑max yields

      ∂O/∂u = a ⊙ (∂O/∂a − ⟨a,∂O/∂a⟩),

   where `O = 2·n·M_β` is the objective we are minimising.

5. **Multi‑length & restarts** – a small set of candidate lengths is
   explored (2000 … 32000).  The bulk of the time budget is spent on
   the largest length, but occasional random Dirichlet restarts are
   performed to escape possible local minima.  Deterministic seeds
   (uniform and Catalan‑shape) are also evaluated.

The routine respects the supplied time budget, prints occasional
progress messages and finally returns the best sequence found as a
plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (real‑valued input).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Objective that ``evaluate_sequence`` would return for a
    *already normalised* vector (sum = 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Truncated coefficients of (1‑z)^‑½,
    i.e.  b_i ∝ binom(2i,i) / 4**i .
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0  # i = 0 term
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    tot = coeffs.sum()
    if tot == 0.0:
        return _uniform(n)
    return coeffs / tot


# ----------------------------------------------------------------------
# Random Dirichlet sampling (α = 1)
# ----------------------------------------------------------------------


def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector (normalised exponentials)."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Optimisation core – smooth‑max mirror descent
# ----------------------------------------------------------------------


def _softmax_from_u(u: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    shift = np.max(u)
    exp_shifted = np.exp(u - shift)
    return exp_shifted / exp_shifted.sum()


def _optimize_len(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_a: np.ndarray = None,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max mirror descent on the simplex for a fixed length `n`.
    Returns the best probability vector (sum = 1) seen and its
    objective value 2·n·max(conv).
    """
    # ---- initialise ----------------------------------------------------
    if init_a is None:
        a0 = _catalan_shape(n)
    else:
        a0 = init_a.copy()
        a0 = a0 / a0.sum()
    u = np.log(a0 + 1e-20)                     # parameters for the soft‑max

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.12                                 # raw learning‑rate (tuned empirically)

    # smooth‑max schedule
    beta = 5.0
    beta_inc = 1.02
    beta_max = 60.0

    best_a = a0.copy()
    best_obj = _objective(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current probability vector
        a = _softmax_from_u(u)

        # convolution a*a
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        obj = 2.0 * n * max_conv
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[Refine n={n:5d} it={it:7d} obj={obj:.6f} beta={beta:.2f}")

        # ----- smooth max ------------------------------------------------
        scaled = beta * conv
        shift = np.max(scaled)
        w = np.exp(scaled - shift)
        Z = w.sum()
        w /= Z                     # now w_k = soft‑max weights, Σ w = 1

        # gradient of M_β w.r.t. a  (2·Σ_k w_k·a_{k-i})
        w_rev = w[::-1]
        conv_grad = _conv_fft(a, w_rev)          # length = 3n‑2
        # slice indices n‑1 … 2n‑2 (inclusive) → n elements, then reverse
        g = conv_grad[n - 1 : 2 * n - 1][::-1]
        grad_M = 2.0 * g                         # ∂M/∂a

        # gradient of the true objective O = 2·n·M
        grad_O_a = 2.0 * n * grad_M

        # chain rule through soft‑max
        dot_ag = np.dot(a, grad_O_a)
        grad_u = a * (grad_O_a - dot_ag)

        # Adam update on u
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # gradually tighten the approximation of the max
        if it % 2000 == 0:
            beta = min(beta * beta_inc, beta_max)

    return best_a, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Optional historic solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom, but costs more
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]

    best_seq = None
    best_val = float("inf")

    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    for n in lengths:

                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={seed_func.__name__:8s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=previous   val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – focused optimisation on the longest promising length
    # ------------------------------------------------------------------
    # Allocate the bulk of the remaining time to the largest length.
    # Smaller lengths get a short "quick‑check" run.
    remaining = deadline - time.time()

    # quick runs on the first three lengths (≈ 5 % each)
    quick_frac = 0.05
    for n in lengths[:3]:

        if val_opt < best_val:

        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"[Random‑restart] n={n:5d} new best {best_val:.6f}")

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
