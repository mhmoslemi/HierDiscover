"""
Improved search for a non‑negative step function minimizing

        2·n·max(conv(a,a)) / (∑a)²

The approach builds on the previous implementation but adds a
significant amount of extra exploration:

* a much richer family of seed vectors – especially a dense grid of
  beta‑shaped “U‑curves” (α,β ∈ [0.05, 2.5]), Gaussian blobs,
  geometric decays with fine r‑grid and several random Dirichlet /
  sparse patterns;
* a longer Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate,
  with a slower sharpening schedule and adaptive learning‑rate;
* an exact sub‑gradient mirror‑descent refinement (exponentiated‑gradient);
* a deterministic greedy local optimiser that repeatedly removes a
  small amount of mass from the heaviest contributor to the current
  bottleneck convolution entry and reallocates it to a low‑mass
  position – this never accepts a worsening move, so it only drives
  the objective down.

All phases are time‑aware (the global deadline is respected) and the
best sequence discovered at any length is returned.  The code is fully
self‑contained – the evaluator `evaluate_sequence` is assumed to be
imported in the execution environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# Seed families (all returned already normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i   (0 < r < 1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑like shape:
        a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred at the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_extensive_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
    max_seeds: int = 500,

    alphas = np.linspace(0.05, 2.5, 13)
    betas = np.linspace(0.05, 2.5, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – a handful of σ values (roughly 10‑% to 45‑% of length)
    for sigma_rel in (0.1, 0.15, 0.2, 0.3, 0.4):
        sigma = sigma_rel * n
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 0.8, 1.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns (different sparsities)
    for k in (2, 3, 5, 8, 12):
        for conc in (0.1, 0.4, 0.9):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution (if length matches)
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    # ------------------------------------------------------------------
    # Random shuffling and truncation – keep the pool manageable
    # ------------------------------------------------------------------
    rng.shuffle(seeds)
    if len(seeds) > max_seeds:
        seeds = seeds[:max_seeds]

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    ``seed`` must be already normalised (∑ = 1).
    Returns the best sequence found and its exact objective value.
    """
    eps = 1e-30
    # work in log‑space (ensures positivity)
    v = np.log(seed + eps)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # start with a modest smoothing, then increase slowly
    beta_smooth = 2.0
    beta_mul = 1.005  # slower sharpening than the original

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·conv_k w.r.t. a -----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2, we need the central n entries
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping – stabilises training
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    Returns the best sequence and its exact objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a (only entries that appear in the bottleneck)
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _greedy_local_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy refinement.
    At each step the algorithm identifies the current bottleneck k*,
    removes a small amount of mass from the heaviest contributing index
    and reallocates it to a low‑mass position, accepting only improving
    moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # heuristic schedule for the transferred mass – it shrinks with the iteration count
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # locate the bottleneck entry
        k_star = int(np.argmax(cur_conv))

        # contributors to the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # choosing the index that participates with the largest product
        heavy_idx = idx_range[np.argmax(contrib)]

        # pick a low‑mass target (bottom 10 % quantile, exclude the heavy index)
        low_candidates = np.where(cur <= np.percentile(cur, 10.0))[0]
        low_candidates = low_candidates[low_candidates != heavy_idx]
        if low_candidates.size == 0:
            continue
        target = rng.choice(low_candidates)

        # amount to move – a fraction of the heavy entry, diminishing over time
        epsilon = min(0.5 * cur[heavy_idx], 0.01 / np.sqrt(it))
        if epsilon <= 0.0:
            continue

        # Apply the elementary move (heavy → target) using an incremental convolution update
        changed = np.array([heavy_idx, target], dtype=int)
        deltas = np.array([-epsilon, epsilon], dtype=np.float64)

        # incremental update of the convolution (exact)
        conv_new = cur_conv.copy()
        for p, d in zip(changed, deltas):
            if d == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * d * cur
            conv_new[2 * p] += d * d
        # cross‑terms for i≠j (appears twice)
        conv_new[changed[0] + changed[1]] += 2.0 * deltas[0] * deltas[1]

        new_obj = 2.0 * n * np.max(conv_new)

        # accept only if we improve
        if new_obj < cur_obj:
            cur[changed] += deltas
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[Greedy] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the global time budget and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin – we want to finish BEFORE the process is killed
    deadline = start_time + budget_s - 2.0

    # previous solution, if any, is exposed through the global variable
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to explore – we start from a modest size and grow if time permits
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # 1) Generate a large pool of diverse seeds (time‑aware)
        # ------------------------------------------------------------------
        seeds = _make_extensive_seeds(n, previous, rng, max_seeds=500)

        # quick evaluation of all raw seeds
        seed_vals = []
        for s in seeds:
            if time.time() > deadline:
                break
            seed_vals.append(_objective(s))

        # keep the top‑k seeds for more intensive optimisation
        top_k = 12
        if len(seed_vals) > top_k:
            top_idx = np.argsort(seed_vals)[:top_k]
            top_seeds = [seeds[i] for i in top_idx]
        else:
            top_seeds = seeds

        # update global best with any raw seed
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the most promising seeds (top‑k)
        # ------------------------------------------------------------------
        for idx, s in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} seed#{idx} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline, max_iters=400000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Deterministic greedy local improvement (pure descent)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _greedy_local_optimize(global_best_seq, n, deadline, max_iters=250000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Greedy] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a return value
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # final sanity check (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
