"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)^2)

using a multi‑phase, time‑aware optimization pipeline.
The pipeline:

1. Build a large, diverse set of analytical and random seeds
   (including the classic arcsine shape  α=β=0.5 and a
   singular ‑½ power‑law which analytically yields an almost
   constant self‑convolution).

2. Keep the best seed (lowest objective).

3. Refine the best seed with four optimisation stages,
   all respecting the global time budget:
   – Adam on a smooth‑max surrogate (log‑softmax parametrisation).
   – Exponentiated‑gradient sub‑gradient descent on the true max.
   – L2‑flattening (drive the auto‑convolution towards a constant).
   – Targeted stochastic hill‑climb with an annealed step‑size.

Small random jitters are injected between stages to escape shallow
local minima.  The final vector is stored in the global variable
`height_sequence_1` for warm‑starting future calls.

The implementation is fully vectorised, uses real FFT‐based
convolutions (O(n log n) per iteration) and performs
no I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Return 2·n·max(conv(a,a)); assumes sum(a)=1.
    Uses the fast FFT‑based convolution.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (Σv = 1, v ≥ 0)."""
    s = v.sum()
    if s == 0.0:
        # fallback to uniform – should never happen for a well‑behaved vector
        return np.full_like(v, 1.0 / v.size)
    return np.maximum(v, 0.0) / s

# Seed generation (large, analytically diverse family)
# ----------------------------------------------------------------------


def _make_seeds(
    n: int, rng: np.random.Generator, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Uniform & classic smooth windows
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    i = np.arange(n, dtype=np.float64)

    # half‑sine
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(_normalize(s))

    # Hann window (cosine‑squared)
    hann = 0.5 * (1 - np.cos(2 * np.pi * (i + 0.5) / n))
    seeds.append(_normalize(hann))

    # 2. Power‑law families (including negative exponents)
    for p in np.linspace(-0.5, 4.0, 12):
        a = (i + 1) ** (-p)          # i+1 to avoid div‑by‑zero
        seeds.append(_normalize(a))

    # 3. Symmetric β‑like shapes (Beta(α,β) densities)
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(_normalize(a))

    # 4. Singular “inverse‑sqrt’’ shape (continuous analogue of
    #    f(x) ∝ 1/√x which yields a constant self‑convolution).
    #    Discretised symmetrically:
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    seeds.append(_normalize(a))

    # 5. Dirichlet draws with a wide range of concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(4):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(_normalize(a))

    # 6. Softmax of Gaussian logits (random bell‑shapes)
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(_normalize(a))

    # 7. Random piecewise‑linear shapes (interpolation on a random knot‑grid)
    for _ in range(5):
        K = max(8, int(np.sqrt(n) * rng.uniform(0.4, 1.6)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(_normalize(a))

    # 8. Log‑normal “bell’’ shapes
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(_normalize(a))

    # 9. Warm‑start (if a previous best exists)
    if warm_start is not None and warm_start.shape == (n,):
        seeds.append(_normalize(np.maximum(warm_start, 0.0)))

    # Deduplicate (tiny differences) and return
    uniq = []
    seen = set()
    for a in seeds:
        key = tuple(np.round(a, 12))
        if key not in seen:
            seen.add(key)
            uniq.append(a)
    print(f"Generated {len(uniq)} distinct seeds (n={n})")
    return uniq

def _adam_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000):
    """
    Adam on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).
    The smooth‑max uses an increasing temperature β.
    """
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 0.5            # start with a gentle smoothing
    beta_mul = 1.005      # very slow increase → very stable gradients

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # ---------- soft‑max trick (a on the simplex) ----------
        a = np.exp(v - np.max(v))
        a /= a.sum()

        # ---------- evaluate ----------
        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---------- smooth‑max weights ----------
        beta = min(beta * beta_mul, 2000.0)          # safeguard against overflow
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---------- gradient of Σ w_k·conv_k ----------
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]    # central n entries

        # ---------- convert to gradient wrt logits ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000):
    """
    Exact sub‑gradient descent on the true max,
    using exponentiated‑gradient (mirror descent) on the simplex.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000):
    """
    Minimise ½‖a*a – μ·1‖² where μ = 1/(2n‑1),
    using exponentiated‑gradient.
    Drives the auto‑convolution towards a constant vector.
    """
    a = _normalize(a0.copy())
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[L2Flat] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int, deadline: float, max_iters: int = 500_000):
    """
    Targeted stochastic hill‑climb.
    Each move transfers a tiny amount of mass from a 'donor' index,
    selected according to its contribution to the current maximal
    convolution entry, to a randomly chosen 'receiver'.
    The step‑size decays exponentially.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.03                     # initial relative step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        # pick a receiver different from donor
        receiver = rng.integers(0, n - 1)
        while receiver == donor:
            receiver = rng.integers(0, n - 1)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 10_000 == 0:
                    print(f"[Hill] it={it:7d} best={val:.6f}")

        # geometric decay of step scale every few thousand steps
        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 5e-7)

    return best_seq, best_val

def propose_candidate(
    seed: int = 123, budget_s: int = 1000, **kwargs
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  Returns a plain list of Python floats.
    The function runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    margin = 2.0                     # safety margin
    deadline = start_time + budget_s - margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a tighter theoretical bound, but increases per‑
    # iteration cost only logarithmically.  n = 4000 works comfortably,
    # n = 8000 still fits easily in the time budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds – keep the global best on the simplex
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
    print(f"[Seed] initial global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Helper to inject a tiny random jitter (helps escape shallow minima)
    # ------------------------------------------------------------------
    def _jitter(vec: np.ndarray, magnitude: float = 1e-4) -> np.ndarray:
        pert = vec + magnitude * np.random.standard_normal(vec.shape)
        pert = np.maximum(pert, 0.0)
        return _normalize(pert)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline, max_iters=600_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # jitter before next phase
    global_best_seq = _jitter(global_best_seq, magnitude=2e-4)

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent (mirror‑descent)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline, max_iters=800_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # jitter again
    global_best_seq = _jitter(global_best_seq, magnitude=1e-4)

    # ------------------------------------------------------------------
    # Phase 3 – L2‑flattening (drive the autocorrelation towards uniform)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.20
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline, max_iters=400_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted stochastic hill‑climb (use whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 1.0
        seq, val = _hill_climb_optimize(global_best_seq, n, phase_deadline, max_iters=1_000_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        # The imported evaluate_sequence expects a plain Python list.
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
