   optimisation that escapes shallow local minima.

3. **Linear‑programming refinement** – after the gradient phase we
   optionally perform a few Frank–Wolfe style LP steps.  For a current
   sequence ``f`` we solve

        maximise   ∑_j b_j
        subject to  (f * b)(k) ≤ M   for all k,
                    b_j ≥ 0,

   where ``M = max(f*f)``.  The LP (solved with SciPy HiGHS) yields an
   extreme point ``b`` that can be mixed with ``f`` (convex combination)
   to obtain a strictly better feasible point.  Only a handful of LP
   solves are attempted (the matrix size grows quadratically, so we keep
   n ≤ 2000 for this stage).

The whole routine is time‑aware: it stops as soon as the supplied budget
(1000 s by default) is exhausted and returns the best sequence encountered.
All numbers are forced to be non‑negative; the final list is ready for
``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Quantity that evaluate_sequence would return for a *normalised* sequence
    (sum == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^‑½ (Catalan numbers) truncated and normalised."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(1,…,1) vector."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _random_dirichlet_alpha(n: int, alpha: float,
                           rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(α,…,α) vector for arbitrary α>0."""
    if alpha == 1.0:
        return _random_dirichlet_one(n, rng)
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


def _exponential_decay(n: int, rng: np.random.Generator) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, r∈(0.9,0.99)."""
    r = rng.uniform(0.90, 0.99)
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^p·(n‑i)^q."""
    p = rng.uniform(0.1, 5.0)
    q = rng.uniform(0.1, 5.0)
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1, p) * np.power(n - i, q)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _tent_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric tent (piecewise linear) shape."""
    peak = rng.integers(0, n)
    left = np.linspace(0, 1, peak + 1, dtype=np.float64)
    right = np.linspace(1, 0, n - peak, dtype=np.float64)
    a = np.concatenate([left, right[1:]]) if peak > 0 else right
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max probability vector w_k = exp(α·c_k) / Σ exp(α·c_j).
    Numerically stabilised by subtracting max.
    """
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    return exp_term / exp_term.sum()


def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Full cross‑correlation (w ⋆ a)[k] = Σ_i w[i]·a[i+k] (zero‑padded),
    returned as a 1‑D array of length len(w)+len(a)-1.
    """
    L = len(w) + len(a) - 1
    N = _next_pow2(L)
    W = np.fft.rfft(w, n=N)
    A = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(W * np.conj(A), n=N)[:L]
    return corr


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate with respect to a.
    Result has shape (len(a),) and is given by
        g_i = 2·Σ_k w_k·a_{k-i}.
    Implemented via correlation of w with a.
    """
    corr = _corr_fft(w, a)                      # length = 2n-1 + n-1 = 3n-2
    # the first n entries correspond to i = 0 … n-1
    grad = 2.0 * corr[: len(a)]
    return grad


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Construct the matrix A (shape (2n‑1, n)) such that
        (f * b)[k] = Σ_j A[k, j]·b_j,
    i.e. A[k, j] = f_{k‑j} if 0 ≤ k‑j < n, otherwise 0.
    """
    n = len(f)
    L = 2 * n - 1
    rows, cols, data = [], [], []
    for k in range(L):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # for each j in the feasible range, coefficient = f[k-j]
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(L, n))


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   ∑ b_j
        subject to  (f * b)[k] ≤ M   (k = 0 … 2n‑2)
                    b_j ≥ 0
    Returns the optimal b (not normalised) or None if the LP fails.
    """
    n = len(f)
    A = _build_A_ub(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum b  → minimise -sum
    bounds = [(0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs')
    except Exception:  # pragma: no cover
        return None
    if res.success:
        return res.x
    return None


def _lp_refine(f: np.ndarray, deadline: float,
               max_iter: int = 6, mix: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Apply a few LP steps to improve the current sequence.
    `mix` controls the convex combination: f←(1‑mix)·f + mix·g,
    where g = b / Σb.
    """
    best_f = f.copy()
    best_val = _objective(f)
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(f, f)
        M = conv.max()
        b = _lp_step(f, M)
        if b is None:
            break
        sb = b.sum()
        if sb <= 0.0:
            break
        g = b / sb
        f = (1.0 - mix) * f + mix * g
        f /= f.sum()                     # renormalise (numerical safety)
        val = _objective(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP] it={it} n={len(f)} val={val:.6f}")
            if best_val < 1.503:
                break
        it += 1
    return best_f, best_val


# ----------------------------------------------------------------------
# Annealed soft‑max mirror descent
# ----------------------------------------------------------------------


def _annealed_descent(start_seq: np.ndarray,
                      deadline: float,
                      alphas: list[float] = None,
                      iters_per_alpha: int = 3000) -> tuple[np.ndarray, float]:
    """
    Deterministic‑annealing mirror descent on the simplex.
    Returns the best sequence found and its (exact) objective value.
    """
    if alphas is None:
        alphas = [2, 5, 10, 20, 40, 80, 160, 320]

    a = start_seq.copy()
    n = len(a)
    best_a = a.copy()
    best_val = _objective(a)

    total_it = 0
    eps = 1e-30

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_val

            total_it += 1

            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Descent] it={total_it} n={n} α={alpha} val={cur_val:.6f}")
                if best_val < 1.503:
                    return best_a, best_val

            # soft‑max weights & gradient
            w = _softmax_weights(conv, alpha)
            grad = _gradient_softmax(a, w)

            # step size (diminishing)
            lr = 0.6 / np.sqrt(total_it)

            # exponentiated‑gradient (mirror descent with KL divergence)
            log_a = np.log(a + eps) - lr * grad
            log_a -= np.max(log_a)               # stabilise
            a = np.exp(log_a)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final sanity call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Retrieve a previously stored solution, if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:                # pragma: no cover
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (increase if time permits)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly record a candidate
    # ------------------------------------------------------------------
    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Update] n={len(seq)} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds (fast)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        for s in (_catalan_shape(n), _uniform(n)):
            _maybe_update(s)

        if previous is not None and previous.shape == (n,):
            _maybe_update(previous)

    # ------------------------------------------------------------------
    # Stage 2 – random rich seeds + local optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (bias towards larger n)
        probs = np.array(lengths, dtype=float)
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))

        # choose a random seed generation strategy
        mode = rng.integers(0, 5)
        if mode == 0:
            seq = _random_dirichlet_one(n, rng)
        elif mode == 1:
            alpha_dir = rng.uniform(0.2, 5.0)
            seq = _random_dirichlet_alpha(n, alpha_dir, rng)
        elif mode == 2:
            seq = _exponential_decay(n, rng)
        elif mode == 3:
            seq = _beta_shape(n, rng)
        else:  # mode == 4
            seq = _tent_shape(n, rng)

        # quick descent on the fresh seed
        seq_ref, val_ref = _annealed_descent(seq, deadline,
                                             alphas=[2, 5, 10, 20, 40, 80],
                                             iters_per_alpha=1200)
        _maybe_update(seq_ref)

        # occasional LP refinement (only on modest sizes to keep memory low)
        if n <= 2000 and time.time() < deadline:
            seq_lp, val_lp = _lp_refine(seq_ref, deadline,
                                        max_iter=5, mix=0.2)
            _maybe_update(seq_lp)

        # early exit if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – final long‑run refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq_final, val_final = _annealed_descent(best_seq, deadline,
                                                 alphas=[80, 160, 320, 640],
                                                 iters_per_alpha=2000)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final‑Descent] improved best to {best_val:.6f}")

        # one optional LP pass (if size permits)
        if len(best_seq) <= 2000 and time.time() < deadline:
            seq_lp2, val_lp2 = _lp_refine(best_seq, deadline,
                                          max_iter=4, mix=0.15)
            if val_lp2 < best_val:
                best_seq = seq_lp2
                best_val = val_lp2
                print(f"[Final‑LP] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (grader will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
