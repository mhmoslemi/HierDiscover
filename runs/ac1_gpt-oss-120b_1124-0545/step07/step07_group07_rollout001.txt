def _adam_optimize2(a0, n, deadline, max_iters=200_000):
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta = 1.0
    beta_mul = 1.03
    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute a from logits
        a = np.exp(v - np.max(v))
        a /= a.sum()
        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # smooth max gradient
        beta = beta * beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        w /= w_sum
        # gradient (conv of reversed a and w)
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # softmax jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # adam updates
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val

def _subgradient_optimize2(a0, n, deadline, max_iters=200_000):
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)
    it=0
    while time.time() < deadline and it < max_iters:
        it+=1
        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # exponentiated gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        # optional symmetrization
        a = (_symmetrize(a))
        a /= a.sum()

"""
Optimisation of a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm improves on the previous pipeline by

* enforcing symmetry (the optimum is believed to be palindromic);
* using a larger, adaptive dimension (n≈6000) when time permits;
* a more aggressive Adam schedule for the smooth‑max surrogate;
* a second sub‑gradient phase with explicit symmetrisation;
* an evolutionary “population” phase that mutates the best candidates
  in log‑space (mirrored simplex);
* a final targeted hill‑climb.
All phases respect the global time budget (≈1000 s).  The best
sequence is stored in the global variable ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Return the (unnormalised) part of the evaluation function:
    2·n·max(conv(a,a)), assuming sum(a)=1.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a palindromic shape."""
    return (a + a[::-1]) / 2.0


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    (Duchi et al. 2008).  Returned vector sums to 1 and is non‑negative.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of ℓ₁‑normalised, symmetrised candidate vectors."""
    seeds: list[np.ndarray] = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # half‑sine window (mirrored)
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # geometric families
    for r in np.linspace(0.2, 0.95, 8):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # power‑law families
    for p in np.linspace(0.0, 4.0, 7):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 4):
        for beta in np.linspace(0.5, 2.0, 4):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(3):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(3):
        K = max(6, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal bells
    for _ in range(3):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # Warm‑start (if available)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    # Symmetrise and re‑normalise all seeds
    sym_seeds = []
    for a in seeds:
        a = _symmetrize(a)
        s = a.sum()
        if s > 0:
            sym_seeds.append(a / s)
    return sym_seeds

def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate.
    Parameterisation: a = softmax(v).
    """
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 0.8            # initial smoothness
    beta_mul = 1.04

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 2000 == 0:
                print(f"[Adam] it={t:6d} best={cur_val:.6f}")

        # increase smoothness
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    # final symmetrisation
    best_seq = _symmetrize(best_seq)
    best_seq /= best_seq.sum()
    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max,
    using exponentiated‑gradient (mirror) updates.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # enforce symmetry after each step (helps convergence)
        a = _symmetrize(a)
        a /= a.sum()

    best_seq = _symmetrize(best_seq)
    best_seq /= best_seq.sum()
    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Drive the auto‑convolution towards a constant vector (L2 flattening).
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)    # target uniform conv value

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # enforce symmetry
        a = _symmetrize(a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    best_seq = _symmetrize(best_seq)
    best_seq /= best_seq.sum()
    return best_seq, best_val


def _evolutionary_optimize(parents: list[np.ndarray], n: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          sigma0: float = 0.7,
                          keep: int = 5) -> tuple[np.ndarray, float]:
    """
    Simple evolutionary strategy:
        – keep the `keep` best candidates,
        – mutate them in log‑space with decreasing Gaussian noise,
        – accept improvements.
    """
    rng = np.random.default_rng()
    # initialise population and its objective values
    pop = [p.copy() for p in parents[:keep]]
    vals = [_objective(p, n) for p in pop]

    # keep track of the best overall
    best_idx = int(np.argmin(vals))
    best_seq = pop[best_idx].copy()
    best_val = vals[best_idx]

    sigma = sigma0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick a parent (biased towards better ones)
        probs = np.exp(-np.array(vals) / (np.max(vals) + 1e-12))
        probs /= probs.sum()
        parent = rng.choice(pop, p=probs)

        # mutate in log‑space
        v = np.log(parent + 1e-30)
        v += sigma * rng.normal(size=v.shape)
        child = np.exp(v - np.max(v))
        child /= child.sum()
        child = _symmetrize(child)
        child /= child.sum()

        child_val = _objective(child, n)

        # replace the worst individual if the child improves it
        worst_idx = int(np.argmax(vals))
        if child_val < vals[worst_idx]:
            pop[worst_idx] = child
            vals[worst_idx] = child_val

            if child_val < best_val:
                best_val = child_val
                best_seq = child.copy()
                if it % 3000 == 0:
                    print(f"[Evo] it={it:6d} new best={best_val:.6f}")

        # exponential cooling of mutation strength
        sigma = max(sigma * 0.9997, 1e-4)

    best_seq = _symmetrize(best_seq)
    best_seq /= best_seq.sum()
    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb that moves a tiny amount of mass
    away from the contributors to the current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.03

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each pair (i, j) to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 4000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        # anneal step size slowly
        if it % 2500 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    best_seq = _symmetrize(best_seq)
    best_seq /= best_seq.sum()
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found as a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives a better theoretical bound)
    # ------------------------------------------------------------------
    # If the budget is generous we go for n=6000, otherwise keep n=4000.
    n = 6000 if budget_s >= 900 else 4000
    print(f"Problem dimension set to n={n}")

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    warm_arr = None
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} symmetrised seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            # occasional progress report
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline,
                              max_iters=250_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.25
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline,
                                     max_iters=250_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L2 flattening (drive conv towards constant)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.20
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline,
                                    max_iters=250_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Evolutionary population refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        phase_deadline = time.time() + remaining * 0.15
        # initialise a small population around the current best
        pop = [global_best_seq]
        for _ in range(4):
            pert = np.log(global_best_seq + 1e-30)
            pert += rng.normal(scale=0.2, size=pert.shape)
            cand = np.exp(pert - np.max(pert))
            cand /= cand.sum()
            cand = _symmetrize(cand)
            cand /= cand.sum()
            pop.append(cand)
        seq, val = _evolutionary_optimize(pop, n, phase_deadline,
                                          max_iters=300_000,
                                          sigma0=0.6,
                                          keep=5)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Evo] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Targeted hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _hill_climb_optimize(global_best_seq, n, phase_deadline,
                                        max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final symmetrisation & verification
    # ------------------------------------------------------------------
    global_best_seq = _symmetrize(global_best_seq)
    global_best_seq /= global_best_seq.sum()

    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
