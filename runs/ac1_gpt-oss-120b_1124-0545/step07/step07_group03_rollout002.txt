def _local_search(seq, deadline, rng):

    n = len(a)
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    T = 0.01
    T_min = 1e-6
    decay = 0.9995
    step_scale = 0.1  # proportion of mass to move occasionally
    it = 0
    while time.time() < deadline:
        it += 1
        # choose i with a[i] > 0
        i = rng.integers(n)
        while a[i] <= 0:
            i = rng.integers(n)
        # choose j distinct
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # decide delta
        max_delta = min(a[i], step_scale * a[i] + 1e-6)
        delta = rng.random() * max_delta
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Ensure all nonnegative (by construction)
        # Compute objective
        val = _objective(new_a)
        # acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / (T + 1e-12)):
            a = new_a
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Local] it={it} best={best_val:.6f}")

        T = max(T * decay, T_min)

    return best_a, best_val

"""
Algorithm to find a low‑value step‑function for the provided
evaluate_sequence objective.

The search proceeds in four time‑aware stages:

1. **Deterministic seeds** – several analytically‑motivated families
   (uniform, Catalan, triangular, sine, exponential decay, power‑Catalan)
   are generated for a handful of lengths.  Their objective values are
   recorded and the best few are kept.

2. **Diversified random sampling** – while time permits we draw many
   random vectors from several distributions (Dirichlet with random
   concentration, log‑normal, exponential decay, power‑Catalan, sine).
   The best samples are stored in a small “elite” pool.

3. **Mirror‑descent refinement** – each elite vector is refined on the
   simplex by exponentiated‑gradient (sub‑gradient of the active
   convolution entry).  The refinement runs until its allocated time
   expires.

4. **Local stochastic search** – the current best vector is further
   improved by a temperature‑controlled random walk that transfers a
   small amount of mass between two coordinates (simulated‑annealing
   style).  This stage uses any remaining budget.

All operations work on *probability vectors* (sum = 1); the scaling
invariance of the objective guarantees that the returned list can be
passed directly to ``evaluate_sequence``.
"""

# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a **normalized** sequence (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic seed families (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (series of (1‑z)^‑½).
    Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _triangular_shape(n: int) -> np.ndarray:
    """Ramp up then down linearly."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    coeffs = np.where(i <= half, i + 1.0, n - i)
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _sine_shape(n: int) -> np.ndarray:
    """Half‑sine wave."""
    i = np.arange(1, n + 1, dtype=np.float64)
    coeffs = np.sin(np.pi * i / (n + 1))
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _exp_decay(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (0 < r < 1)."""
    i = np.arange(n, dtype=np.float64)
    coeffs = np.power(r, i)
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _power_catalan(n: int, p: float) -> np.ndarray:
    """Catalan coefficients raised to power p, then normalised."""
    base = _catalan_shape(n)
    coeffs = np.power(base, p)
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total

def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(1,…,1) – exponential‑family sampling."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()

    return x / s

    x = rng.gamma(shape=alpha, scale=1.0, size=n)

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by exponentiated‑gradient steps on the
    max‑convolution functional.  Stops when `time.time()` exceeds
    `deadline`.  Returns the best vector seen and its objective value.
    """
    a = start_seq.copy()
    n = len(a)
    eps = 1e-30
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full convolution and the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()
    return best_seq, best_val

    T_min = 1e-6
    decay = 0.9995
    step_factor = 0.12   # typical fraction of the selected mass to move

    it = 0
    while time.time() < deadline:
        it += 1

        # pick a donor with positive mass
        i = rng.integers(n)
        while a[i] <= 0.0:
            i = rng.integers(n)

        # pick a distinct receiver
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose moving a small amount δ from i to j
        max_delta = min(a[i], step_factor * a[i] + 1e-12)
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        val = _objective(new_a)

        # Metropolis acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / (T + 1e-12)):
            a = new_a
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Local] it={it:7d} best={best_val:.6f}")

        # cool down
        T = max(T * decay, T_min)

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects a time budget
    (default 1000 s) and returns the best sequence as a plain ``list``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final sanity call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore (larger n → potentially smaller bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]

    # ------------------------------------------------------------------
    # Elite pool – always keep the currently best `top_k` candidates
    # ------------------------------------------------------------------
    top_k = 8
    elite = []                     # list of (value, vector, n)

    def _maybe_push(val, vec, n):
        nonlocal elite
        if len(elite) < top_k:
            elite.append((val, vec.copy(), n))
            elite.sort(key=lambda x: x[0])
        elif val < elite[-1][0]:
            elite[-1] = (val, vec.copy(), n)
            elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        deterministic = [
            _uniform,
            _catalan_shape,
            _triangular_shape,
            _sine_shape,
            lambda nn: _exp_decay(nn, 0.90),
            lambda nn: _exp_decay(nn, 0.95),
            lambda nn: _exp_decay(nn, 0.98),
            lambda nn: _power_catalan(nn, 0.5),
            lambda nn: _power_catalan(nn, 2.0),
        ]
        for maker in deterministic:
            try:
                seq = maker(n)
                val = _objective(seq)
                _maybe_push(val, seq, n)
                print(f"[Seed] n={n:5d} type={getattr(maker, '__name__', 'lambda'):<12} val={val:.6f}")
            except Exception as exc:
                print(f"[Seed] n={n} error: {exc}")

    # ------------------------------------------------------------------
    # Optional previous solution (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                prev_norm = prev_arr / prev_arr.sum()
                val = _objective(prev_norm)
                n = prev_norm.size
                _maybe_push(val, prev_norm, n)
                print(f"[Prev] n={n:5d} val={val:.6f}")
        except Exception as exc:
            print("[Prev] failed:", exc)

    # ------------------------------------------------------------------
    # Stage 2 – diversified random sampling
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.45
    sample_deadline = now + time_for_sampling

    random_kinds = [
        "dirichlet1",
        "dirichlet_alpha",
        "lognormal",
        "exp_decay",
        "power_catalan",
        "sine"
    ]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        kind = rng.choice(random_kinds)

        if kind == "dirichlet1":
            seq = _random_dirichlet_one(n, rng)
        elif kind == "dirichlet_alpha":
            alpha = rng.uniform(0.2, 5.0)
            seq = _random_dirichlet_alpha(n, rng, alpha)
        elif kind == "lognormal":
            sigma = rng.uniform(0.2, 2.0)
            seq = _random_lognormal(n, rng, sigma)
        elif kind == "exp_decay":
            r = rng.uniform(0.85, 0.99)
            seq = _exp_decay(n, r)
        elif kind == "power_catalan":
            p = rng.uniform(0.3, 3.0)
            seq = _power_catalan(n, p)
        elif kind == "sine":
            seq = _sine_shape(n)
        else:
            seq = _uniform(n)

        val = _objective(seq)
        _maybe_push(val, seq, n)

        # early exit if we already beat the target by a comfortable margin
        if val < 1.50:
            print(f"[Early] reached impressive bound {val:.6f}")
            break

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient (mirror‑descent) refinement of elite pool
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – fall back to the current best
        final_seq = elite[0][1]
    else:
        per_candidate = remaining / max(1, len(elite))
        best_val = elite[0][0]
        best_seq = elite[0][1].copy()

        for idx, (val, seq, n) in enumerate(elite):
            cand_deadline = time.time() + per_candidate
            refined_seq, refined_val = _refine_subgradient(seq, cand_deadline)
            if refined_val < best_val:
                best_val, best_seq = refined_val, refined_seq.copy()
                print(f"[Refine] elite {idx} improved to {refined_val:.6f}")

        # ------------------------------------------------------------------
        # Stage 4 – local stochastic search using remaining budget
        # ------------------------------------------------------------------
        final_deadline = deadline
        final_seq, best_val = _local_search(best_seq, final_deadline, rng)

    # ------------------------------------------------------------------
    # Final sanity check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(final_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(
        f"\n=== Finished ===\n"
        f"Best objective after all stages ≈ {best_val:.6f} (length {len(final_seq)})"
    )
    return final_seq.tolist()
