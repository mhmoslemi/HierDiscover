def _interpolate(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)

    total = new_seq.sum()
    if total == 0:
        return np.full(new_n, 1.0/new_n)
    return new_seq / total

Implementation:

def _project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    lengths = [2000, 4000, 8000]  # maybe add 12000 if time.

    for idx, n in enumerate(lengths):
        # if no time left, break
        if time.time() > deadline:
            break

        # allocate time portion: remaining_time = deadline - time.time()

        seeds = _make_seeds(n, prev, rng)

        seed_vals = [_objective(s) for s in seeds]

        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()

        # Pick top 3 seeds to refine using Adam
        top_k = min(3, len(seeds))

        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print("[Adam] n={} new global best {:.6f}".format(n, val))

        # subgradient PGD refinement
        if global_best_seq is not None and len(global_best_seq) == n:
            if time.time() > deadline:
                break
            seq, val = _pgd_subgrad_optimize(global_best_seq, n, deadline,

                global_best_val = val
                global_best_seq = seq.copy()

        # simulated annealing
        if global_best_seq is not None and len(global_best_seq) == n:
            if time.time() > deadline:
                break
            seq, val = _simulated_annealing(global_best_seq, n, deadline,

def _project_onto_simplex(v):
    # cast to np
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho: largest i such that u_i > cssv_i / i
    # Actually condition: u_i - (cssv_i)/i > 0.
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive, set to uniform?
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]  # index, 1-indexed
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function minimizing the objective
2·n·max(conv(a,a))/(sum a)^2.

The algorithm:

1. Build a diverse set of seed vectors (uniform, geometric, power‑law,
   beta shapes, random Dirichlet, sparse patterns, and an up‑scaled
   previous best).  All seeds are forced to be symmetric and normalised
   to sum = 1.

2. Run a smooth‑max Adam optimiser on the top‑few seeds.  Optimisation
   is performed in the log‑space of the probabilities and after each
   update the iterate is projected onto the simplex and symmetrised,
   keeping the search inside the symmetric sub‑space.

3. Refine the best vector with a pure sub‑gradient mirror‑descent
   (exponentiated‑gradient) that works on the true max‑convolution
   objective.  The iteration again projects onto the simplex and
   symmetrises.

4. A targeted simulated‑annealing phase that repeatedly attacks the
   current bottleneck entry of the convolution.  Small amounts of mass
   are moved from the heaviest contributors to random locations,
   always re‑symmetrising the vector.  The temperature is annealed
   geometrically.

5. After finishing a length, the vector is up‑sampled (linear
   interpolation) to serve as a seed for the next, larger length.
   The process repeats for several lengths while respecting the global
   time budget (≈ 1000 s).

The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Project a vector onto the set of symmetric vectors on the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s == 0.0:
        return a_sym
    return a_sym / s


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – fall back to uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (monotone decreasing)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑like shape:
        a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return new_seq / s


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse collection of normalised, symmetric seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 10)
    betas = np.linspace(0.30, 1.20, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous.copy())

    # symmetrise all seeds and renormalise
    sym_seeds = []
    for s in seeds:
        s2 = _symmetrize(s)
        sym_seeds.append(s2)
    return sym_seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) approximation.
    Works on the log‑probabilities; each iterate is projected onto the
    symmetric simplex.
    """
    eps = 1e-30
    # initialise log‑probabilities
    v = np.log(seed + eps)               # shape (n,)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing schedule
    beta_smooth = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        a = _symmetrize(a)          # keep search inside symmetric sub‑space

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # numerical stabilisation
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2, keep centre n entries
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- gradient w.r.t. log‑space v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # enforce symmetry on v (so that softmax stays symmetric)
        v = (v + v[::-1]) * 0.5

    return best_seq, best_val


def _pgd_subgrad_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Projected (mirror) sub‑gradient descent on the true max‑convolution.
    The iterate stays on the symmetric simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[PGD] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        # gradient = 2 * a[rev_idx] for each i (including diagonal case)
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step size
        lr = 0.7 / np.sqrt(it)

        # gradient step in Euclidean space
        a_new = a - lr * g

        # project onto simplex
        a = _project_onto_simplex(a_new)

        # enforce symmetry and renormalise
        a = _symmetrize(a)

        # tiny numerical safety
        a = np.maximum(a, 0.0)

        # normalise explicitly (project_onto_simplex already ensures sum=1)
        # but after symmetrisation we may have shifted the sum slightly.
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly attacks the current bottleneck
    convolution entry.  At each step a small amount of mass is relocated
    from the heaviest contributors to a random site, the whole vector
    is then symmetrised and the new objective evaluated.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur = _symmetrize(cur)
    cur = cur / cur.sum()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Initial temperature proportional to current objective
    T = max(0.02 * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, 1e-12)

        # locate bottleneck index
        k_star = int(np.argmax(cur_conv))

        # contributions that form the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a heavy contributor (largest product)
        if len(idx_range) > 0:
            heavy_idx = idx_range[np.argmax(contrib)]
        else:
            heavy_idx = rng.integers(n)

        # pick a random target (different)
        target_idx = rng.integers(n)
        while target_idx == heavy_idx:
            target_idx = rng.integers(n)

        # amount to move: a random fraction of the source mass
        max_move = min(0.5 * cur[heavy_idx], cur[heavy_idx])
        if max_move <= 0.0:
            continue
        delta = rng.uniform(1e-6, max_move)

        new_seq = cur.copy()
        new_seq[heavy_idx] -= delta
        new_seq[target_idx] += delta

        # enforce symmetry and renormalise
        new_seq = _symmetrize(new_seq)
        s = new_seq.sum()
        if s == 0.0:
            continue
        new_seq /= s

        new_conv = _conv_fft(new_seq, new_seq)
        new_obj = 2.0 * n * np.max(new_conv)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur = new_seq
            cur_conv = new_conv
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Load a previously found construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to explore – they grow geometrically.
    lengths = [2000, 4000, 8000, 16000]

    global_best_seq = None
    global_best_val = float("inf")
    prev_seq = previous

    for idx, n in enumerate(lengths):
        if time.time() > deadline:
            break

        # If we have a previous best of a different length, up‑sample it.

"""
Propose a candidate step‑function minimizing the objective

    2·n·max(conv(a,a)) / (sum a)²

The algorithm works inside the symmetric simplex (aᵢ = aₙ₋₁₋ᵢ) and
combines several optimisation phases:

1. Generate a rich family of symmetric seeds (uniform, sine window,
   geometric, power‑law, beta‑shapes, random Dirichlet, sparse patterns,
   and an up‑scaled previous best).  All seeds are normalized to sum = 1.

2. Run a smooth‑max Adam optimiser on the top‑few seeds.  Optimisation
   proceeds in log‑space, projects on the simplex, and after each step
   enforces symmetry.

3. Refine the current best vector with a projected sub‑gradient (PGD)
   descent that works on the true max‑convolution objective, again
   staying on the symmetric simplex.

4. A targeted simulated‑annealing phase that repeatedly attacks the
   current bottleneck convolution entry: a small amount of mass is moved
   from the heaviest contributor to a random site, the vector is
   symmetrised and the new objective is evaluated.  The temperature is
   geometrically cooled.

5. After finishing a length, the best vector is linearly interpolated
   to serve as a seed for the next, larger length.  The process repeats
   for several lengths while respecting the global time budget
   (≈ 1000 s).

The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Project a vector onto the set of symmetric vectors on the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s == 0.0:
        return a_sym
    return a_sym / s


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (monotone decreasing)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑like shape:
        a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return new_seq / s


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse collection of normalised, symmetric seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 10)
    betas = np.linspace(0.30, 1.20, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous.copy())

    # symmetrise all seeds and re‑normalise
    sym_seeds = []
    for s in seeds:
        sym = _symmetrize(s)
        sym_seeds.append(sym)
    return sym_seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) approximation.
    Works in log‑space and keeps iterates symmetric.
    """
    eps = 1e-30
    v = np.log(seed + eps)               # log‑probabilities

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta_smooth = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)
        a = _symmetrize(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # enforce symmetry on log‑space variables
        v = (v + v[::-1]) * 0.5

    return best_seq, best_val


def _pgd_subgrad_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The iterate stays on the symmetric simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[PGD] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step size
        lr = 0.7 / np.sqrt(it)

        # Euclidean gradient step
        a_new = a - lr * g

        # project onto simplex and symmetrise
        a = _project_onto_simplex(a_new)
        a = _symmetrize(a)

        # tiny numeric safety
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly attacks the current bottleneck
    convolution entry.  After each move the vector is symmetrised.
    """
    rng = np.random.default_rng()
    cur = _symmetrize(start_seq)
    cur = cur / cur.sum()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(0.02 * cur_obj, 1e-12)   # initial temperature
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, 1e-12)

        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a heavy contributor (largest product) as source
        heavy_idx = idx_range[np.argmax(contrib)] if idx_range.size else rng.integers(n)

        # pick a random different target
        target_idx = rng.integers(n)
        while target_idx == heavy_idx:
            target_idx = rng.integers(n)

        # move a small random amount from source to target
        max_move = min(0.5 * cur[heavy_idx], cur[heavy_idx])
        if max_move <= 0.0:
            continue
        delta = rng.uniform(1e-6, max_move)

        new_seq = cur.copy()
        new_seq[heavy_idx] -= delta
        new_seq[target_idx] += delta

        # enforce symmetry and renormalise
        new_seq = _symmetrize(new_seq)
        s = new_seq.sum()
        if s == 0.0:
            continue
        new_seq /= s

        new_conv = _conv_fft(new_seq, new_seq)
        new_obj = 2.0 * n * np.max(new_conv)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur = new_seq
            cur_conv = new_conv
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj
