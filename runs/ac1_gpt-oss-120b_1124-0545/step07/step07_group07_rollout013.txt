"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

The new search strategy builds on the previous algorithm but
adds a dedicated L‑BFGS‑B phase that directly optimises a smooth
log‑sum‑exp surrogate of the max‑convolution.  The surrogate is
tightened by a schedule of increasing “temperature’’ β, and the
gradient is computed analytically via FFT‑based convolutions.
After the L‑BFGS refinement we fall back to the proven sub‑gradient
and hill‑climbing phases.  The whole pipeline respects the 1000 s
budget and stores the best vector in the global name
`height_sequence_1` for warm‑starting future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse collection of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # 1. Uniform and half‑sine
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    i = np.arange(n, dtype=np.float64)
    half_sine = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(half_sine / half_sine.sum())

    # 2. Geometric decays (both sides)
    for r in np.linspace(0.2, 0.9, 6):
        left = r ** np.arange(n, dtype=np.float64)
        right = left[::-1]
        a = left + right
        seeds.append(a / a.sum())

    # 3. Power‑law (including reciprocal)
    for p in np.linspace(0.0, 3.0, 7):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p + 1e-12)
        seeds.append(a / a.sum())

    # 4. Beta‑like shapes (symmetric)
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 5. Dirichlet draws (various concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(3):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a)

    # 6. Log‑normal bumps
    for _ in range(3):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.4, 1.5)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 7. Warm‑start (if present)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


class _SmoothMax:
    """
    Callable that returns both smooth‑max objective value and its gradient
    w.r.t. the softmax logits v.

    Parameters
    ----------
    n : int
        Length of the vector a.
    beta : float
        Temperature for the log‑sum‑exp approximation of max.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        # ---- a = softmax(v) ------------------------------------------------
        a = _softmax(v)

        # ---- convolution a*a ------------------------------------------------
        conv = _conv_full(a, a)                     # length 2n‑1

        # ---- smooth‑max (log‑sum‑exp) ---------------------------------------
        # shift for numerical stability
        shift = np.max(self.beta * conv)
        w = np.exp(self.beta * conv - shift)        # un‑normalised weights
        w_sum = w.sum()
        if w_sum == 0.0:                            # should not happen
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum                                   # now a probability vector

        smooth_max = (np.log(w_sum) + shift) / self.beta
        value = 2.0 * self.n * smooth_max

        # ---- gradient w.r.t. a ---------------------------------------------
        # grad_a = 4·n·conv_rev(w, a)  (see analysis)
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)            # length 3n‑2
        grad_a = 4.0 * self.n * grad_full[self.n - 1:self.n - 1 + self.n]

        # ---- gradient w.r.t. logits v (softmax Jacobian) -------------------
        dot = np.dot(grad_a, a)                     # scalar
        grad_v = a * (grad_a - dot)

        return float(value), grad_v.astype(np.float64)

def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away
    from the contributors to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final verification
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length (4000 works nicely within the time budget)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Keep the best seed (exact objective)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – L‑BFGS on smooth‑max (temperature schedule)
    # ------------------------------------------------------------------
    # Log‑space temperature schedule – start smooth, end sharp.
    beta_schedule = [1.0, 3.0, 10.0, 30.0, 80.0, 200.0]
    v = np.log(global_best_seq + 1e-30)  # logits of the current best

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        print(f"[L‑BFGS] starting beta={beta}")

        sm_obj = _SmoothMax(n, beta)

        # we limit the inner optimisation to a modest number of iterations;
        # the outer schedule will tighten the approximation.
        res = minimize(fun=sm_obj,
                       x0=v,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 120, 'disp': False})

        v = res.x
        a_candidate = _softmax(v)
        cur_val = _objective(a_candidate, n)
        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = a_candidate.copy()
            print(f"[L‑BFGS] beta={beta:.1f} improved best={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _subgradient_optimize(global_best_seq, n,
                                     phase_deadline, max_iters=150_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Targeted hill‑climb (use the rest of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _hill_climb_optimize(global_best_seq, n,
                                        phase_deadline, max_iters=200_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
