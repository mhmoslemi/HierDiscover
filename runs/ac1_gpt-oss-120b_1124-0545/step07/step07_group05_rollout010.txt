def _greedy_local_refine(seq, n, deadline, max_iters=100000):
   cur=seq.copy(); conv=_conv_fft(cur,cur); cur_obj = 2*n*np.max(conv)
   while time<deadline and iteration<max_iters:
       # identify indices i in support of max conv
       k_star = int(np.argmax(conv))
       i_start = max(0, k_star - (n - 1))
       i_end = min(n - 1, k_star)
       support = np.arange(i_start, i_end+1)
       # choose i with largest product a_i*a_{k_star-i}
       prods = cur[support] * cur[k_star - support]
       # pick i_max = support[np.argmax(prods)]
       i_max = support[np.argmax(prods)]
       # pick j random not in support
       j = rng.integers(0, n)
       if j == i_max:
           continue
       # move epsilon from i_max to j
       epsilon = cur[i_max] * 0.01  # 1% of its mass
       if epsilon <= 0:
           continue
       new_vals = cur.copy()
       new_vals[i_max] -= epsilon
       new_vals[j] += epsilon
       # ensure non-negative
       if new_vals[i_max] < 0:
           continue
       # compute new conv quickly using deltas
       deltas = np.array([-epsilon, epsilon])
       idx = np.array([i_max, j])

"""
Optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm proceeds in three stages for each candidate length:

1️⃣  **Rich seed generation** – many families (uniform, geometric, power‑law,
    beta‑like, half‑sine, exponential, logistic, Gaussian, random mixtures,
    sparse Dirichlet draws and the previous best solution) are normalised to
    unit L₁‑norm.

2️⃣  **Adam on a soft‑max approximation** – each selected seed is refined by
    Adam on a log‑sum‑exp surrogate of ``max(conv)``.  The smoothing factor
    ``β`` grows gradually, turning the surrogate into the true max.

3️⃣  **High‑frequency simulated‑annealing (SA)** – a fast incremental update
    of the full convolution makes it possible to perform millions of cheap
    moves.  Three move types are used:

        • random Dirichlet re‑allocation on 2‑4 indices,
        • targeted re‑allocation that favours indices participating in the
          current maximal convolution entry,
        • a simple swap of two indices.

    The temperature follows a geometric schedule and the best solution seen
    during the walk is retained.

All stages respect the overall time budget (default 1000 s).  The best
sequence encountered (as a plain ``list[float]``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a point on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float, mu: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_mixture(
    n: int,
    mu1: float,
    sigma1: float,
    mu2: float,
    sigma2: float,
    w: float,
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    a += (1.0 - w) * np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds (∑ = 1)."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families – coarse grid
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    # geometric – gentle decay
    for r in np.linspace(0.90, 0.99, 5):
        seeds.append(_geom(n, r))
    # geometric – very gentle (r up to 0.999)
    for r in np.linspace(0.990, 0.9995, 3):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta‑like shapes – broad grid
    alphas = np.geomspace(0.2, 5.0, 6)
    betas = np.geomspace(0.2, 5.0, 6)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 10, n / 5, n / 2, n):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2.0
        seeds.append(_logistic_shape(n, offset, scale))

    # Gaussian single‑peak family
    for sigma in (n / 15, n / 10, n / 6):
        for mu in (n * 0.3, n * 0.5, n * 0.7):
            seeds.append(_gaussian(n, sigma, mu))

    # Gaussian mixtures (two peaks)
    for _ in range(6):
        mu1 = rng.uniform(0, n)
        mu2 = rng.uniform(0, n)
        sigma1 = rng.uniform(n / 20, n / 5)
        sigma2 = rng.uniform(n / 20, n / 5)
        w = rng.random()
        seeds.append(_gaussian_mixture(n, mu1, sigma1, mu2, sigma2, w))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 8, 12):
        for conc in (0.2, 0.5):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse previous solution if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # parameters of a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # start with a modest smoothness and increase it geometrically
    beta = 5.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # --------------------------------------------------------------
        # Soft‑max weighting of the convolution entries
        # --------------------------------------------------------------
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # Gradient of weighted sum Σ w_k·conv_k w.r.t. a
        # conv = a * a  ⇒  ∂/∂a = 2·(rev(a) * w)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # Convert to gradient w.r.t. v (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Optional clipping to keep updates stable
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on the simplex)
        lr = 0.7 / np.sqrt(it)                # step‑size schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-9,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with incremental convolution updates.

    Three move types:
      1. random Dirichlet re‑allocation on 2‑4 indices,
      2. targeted re‑allocation (focus on indices that compose the current
         maximal convolution entry),
      3. a simple swap of two indices.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # Initial temperature proportional to the current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * decay, min_temp)

        move_type = rng.random()
        if move_type < 0.1:
            # ----- Move type 3: simple swap of two indices -----
            i, j = rng.choice(n, size=2, replace=False)
            idx = np.array([i, j], dtype=int)
            new_vals = cur[[j, i]]                      # swapped values
        elif move_type < 0.6:
            # ----- Move type 2: targeted Dirichlet re‑allocation -----
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # contribution a[i]·a[k_star‑i] for each i in support
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)
            # pick a second index uniformly (different from i)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
            new_vals = new_alloc
        else:
            # ----- Move type 1: random Dirichlet on 2‑4 indices -----
            k = rng.integers(2, 5)                     # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
            new_vals = new_alloc

        # --------------------------------------------------------------
        # Incremental convolution update
        # --------------------------------------------------------------
        deltas = new_vals - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        conv_new = cur_conv.copy()

        # a*Δ + Δ*a  (the linear part)
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ self term
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept move
            cur[idx] = new_vals
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a list of floats representing the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # grab a previously discovered construction, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # lengths to explore – we start from modest sizes and finish with the
    # biggest one that still fits into the remaining budget.
    lengths = [2000, 3000, 4000, 5000, 6000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – raw seed evaluation
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on the soft‑max surrogate
        # ------------------------------------------------------------------
        # keep only a few promising seeds for the costly Adam phase
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(4, len(seed_vals))

        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=30000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement (first pass)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=400000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad1] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing (the most time‑consuming phase)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            remaining = deadline - time.time()
            # estimate moves ≈ 12000 per second for n≈4000, scale with n
            moves_per_sec = max(3000, int(12000 * 2000 / n))
            target_iters = int(min(4_000_000, remaining * moves_per_sec))
            seq, val = _sa_refine(
                global_best_seq,
                n,
                deadline,
                max_iters=target_iters,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – second sub‑gradient pass (fine‑tuning after SA)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=200000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad2] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # final verification (the evaluation routine will clip/normalise)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
