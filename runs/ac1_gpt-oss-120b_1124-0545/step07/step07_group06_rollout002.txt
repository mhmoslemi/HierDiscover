def _lbfgs_optimize(seed, n, deadline, max_iter=150):
    # seed sum = 1
    eps=1e-30
    v0 = np.log(seed + eps)
    beta = 30.0  # starting smoothing parameter
    # We'll run optimization in stages: for beta in [5, 15, 30, 60].
    best_seq = None
    best_val = float('inf')
    for beta_i in [5.0, 15.0, 30.0, 60.0]:
        if time.time()>deadline:
            break
        # define objective functions.
        def func(v):
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            w = np.exp(beta_i * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0/len(conv))
                w_sum = 1.0
            w = w / w_sum
            obj = 2.0 * n * np.dot(w, conv)  # smooth max
            # compute gradient w.r.t a
            rev_a = a[::-1]
            grad_tmp = _conv_fft(rev_a, w)
            grad_a = 2.0 * grad_tmp[n-1:n-1+n]
            # chain rule for logits
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        # use scipy minimize
        from scipy.optimize import minimize
        res = minimize(fun=lambda v: func(v)[0],
                       x0=v0,
                       jac=lambda v: func(v)[1],
                       method='L-BFGS-B',
                       options={'maxiter': max_iter, 'disp': False, 'ftol':1e-12})
        # after optimization, compute a
        a_opt = _softmax(res.x)
        val = _objective(a_opt)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[L-BFGS] n={n} beta={beta_i} best={val:.6f}")
        # update v0 for next beta stage
        v0 = res.x
    return best_seq, best_val
```

"""
propose_candidate – multi‑stage convex‑smooth optimiser for the step‑function problem.

The routine builds a diverse collection of normalised seeds (uniform,
geometric, power‑law, beta‑shapes, sine‑window, mixtures, random Dirichlet
and sparse patterns).  For each chosen length it proceeds through the
following phases, always respecting the global time budget:

1. Quick evaluation of raw seeds → keep the best few.
2. L‑BFGS optimisation of a smoothed “max‑convolution” objective.
   The smoothing is a log‑sum‑exp (soft‑max) with a schedule of β
   values (5 → 15 → 30 → 60 → 120).  The optimisation works on the
   log‑itself (softmax) parametrisation, guaranteeing non‑negativity
   and the simplex constraint.
3. Adam optimisation (as in the previous baseline) for additional
   robustness.
4. Exact sub‑gradient mirror‑descent refinement on the true max.
5. Targeted simulated‑annealing that re‑allocates mass from the
   heaviest contributors to the current bottleneck entry.

The best sequence across all lengths is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# Seed families (all returned already normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (monotone decreasing)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑like shape:
        a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _double_peak(n: int, r1: float, r2: float) -> np.ndarray:
    """
    Two‑sided geometric mixture: left side decays with r1, right side with r2.
    """
    left = r1 ** np.arange(n, dtype=np.float64)
    right = r2 ** np.arange(n - 1, -1, -1, dtype=np.float64)  # reverse
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Create a diverse collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 10)
    betas = np.linspace(0.30, 1.20, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # double‑peak mixtures
    for r1 in (0.2, 0.4, 0.6):
        for r2 in (0.2, 0.4, 0.6):
            if abs(r1 - r2) < 1e-9:
                continue
            seeds.append(_double_peak(n, r1, r2))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds

def _adam_optimize(seed: np.ndarray, n: int, deadline: float, max_iters: int = 20000):
    """
    Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best sequence seen and its true objective value.
    """
    eps = 1e-30
    v = np.log(seed + eps)               # work in log‑space, guarantees positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing starts modest and is amplified each iteration
    beta_smooth = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # keep numbers stable
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·conv_k w.r.t. a ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2, we need the central n entries
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping (helps stability)
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 300000):
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    Returns the best sequence and its true objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray, n: int, deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 1500000):
    """
    Simulated‑annealing that always focuses on the current maximal convolution entry.
    A small random set of the largest contributors to that entry is re‑distributed.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # locate the bottleneck index
        k_star = int(np.argmax(cur_conv))

        # indices that truly contribute to the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few of the heaviest contributors (max 6)
        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to modify (2‑4)
        k_move = rng.integers(2, 5)

        # choose indices: try to use heavy ones first, fill up with random others
        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass among the chosen positions
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a contributions
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur      # a*Δ + Δ*a (old a)
            conv_new[2 * p] += delta * delta            # Δ*Δ self term

        # cross Δ_i * Δ_j for i≠j (appears twice)
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective for L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj(v: np.ndarray, n: int, beta: float) -> float:
    """
    Smoothed objective: 2·n· Σ_k w_k·conv_k  with
    w ∝ exp(beta·(conv - max(conv))).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to uniform weighting – should never happen
        w = np.full_like(conv, 1.0 / conv.size)
        w_sum = 1.0
    w /= w_sum
    obj = 2.0 * n * np.dot(w, conv)
    return obj


def _smooth_grad(v: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the logits v.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
        w_sum = 1.0
    w /= w_sum

    # gradient w.r.t. a of Σ w_k·conv_k  = 2·conv_rev(w, a)
    rev_a = a[::-1]
    grad_tmp = _conv_fft(rev_a, w)
    grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return grad_v

def _lbfgs_optimize(seed: np.ndarray, n: int, deadline: float):
    """
    Run a short L‑BFGS‑B optimisation on the smooth‑max surrogate.
    A schedule of β values is used; the best (true) objective seen is returned.
    """
    # SciPy is optional – if unavailable we simply skip this phase.
    try:
        from scipy.optimize import minimize
    except Exception:
        print("[L‑BFGS] SciPy not available – skipping")
        return None, float("inf")

    eps = 1e-30
    v = np.log(seed + eps)  # logits (sum‑to‑one enforced by softmax)

    best_seq = None
    best_val = float("inf")

    # β schedule – starts smooth and ends very sharp
    for beta in (5.0, 15.0, 30.0, 60.0, 120.0):
        if time.time() > deadline:
            break

        # Number of L‑BFGS iterations – reduced when time is scarce
        remaining = deadline - time.time()
        maxiter = int(min(70, max(10, remaining / 4.0)))  # very rough heuristic

        res = minimize(
            fun=_smooth_obj,
            x0=v,
            args=(n, beta),
            jac=_smooth_grad,
            method="L-BFGS-B",
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False},
        )
        v = res.x
        a_opt = _softmax(v)
        val = _objective(a_opt)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[L‑BFGS] n={n:5d} β={beta:5.1f} best={val:.6f}")

    return best_seq, best_val

        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=350000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 4) Targeted simulated‑annealing
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline,
                                             max_iters=900000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guards against empty result
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity‑check with the provided evaluator
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
