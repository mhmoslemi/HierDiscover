    old_n = seq.shape[0]
    x_old = np.linspace(0,1,old_n)

    new_seq = np.interp(x_new, x_old, seq)
    # ensure nonnegative, clip minor negative due to interpolation (should not happen)
    new_seq = np.maximum(0, new_seq)
    # renorm
    s = new_seq.sum()
    if s == 0:
        new_seq = _uniform(new_n)
    else:
        new_seq = new_seq / s
    return new_seq

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (∑a)²

where ``conv`` is the linear convolution of the vector with itself.
The algorithm works in three stages:

1. **Rich seed generation** – a wide variety of analytic windows
   (uniform, sine, Hann, Hamming, Blackman, Kaiser, Gaussian,
   power‑law, beta‑shapes, geometric, sparse Dirichlet …) are created,
   normalised to sum = 1 and evaluated quickly.

2. **Smooth‑max Adam optimisation** – the best few seeds are refined
   on a log‑sum‑exp approximation of the max‑convolution (smooth‑max),
   staying on the probability simplex via a soft‑max parametrisation.
   The smoothing parameter grows during optimisation, yielding a
   progressively more accurate surrogate.

3. **Exact sub‑gradient mirror descent** – after Adam, the true max‑convolution
   is tackled with an exponentiated‑gradient step (mirror descent) that
   follows the sub‑gradient of the non‑smooth objective.
   A very cheap simulated‑annealing phase then nudges the solution away
   from local plateaux.

The procedure is multiscale: a relatively cheap optimisation is performed
for a modest length (≈2000).  The resulting shape is then up‑sampled by
linear interpolation to larger lengths (4000, 8000 …) and refined with
shorter Adam / sub‑gradient runs.  This strategy exploits the extra
degrees of freedom obtained at higher resolution without exhausting the
time budget.

All steps are time‑aware – the global deadline (default 1000 s) is
checked before each heavy loop, and the best sequence discovered so far
is returned if time runs out.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()

        x = (i - alpha) / alpha
        a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
        return a / a.sum()
except Exception:  # pragma: no cover – SciPy might be unavailable
    _kaiser_window = None


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 8)
    betas = np.linspace(0.30, 1.20, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows with various widths
    for sigma_factor in (0.20, 0.28, 0.35):
        sigma = sigma_factor * n
        seeds.append(_gaussian_window(n, sigma))

    # Kaiser windows (if SciPy is present)
    if _kaiser_window is not None:
        for beta in (2.0, 4.0, 6.0, 8.0):
            seeds.append(_kaiser_window(n, beta))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_sequence, best_value) seen before the deadline.
    """
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑space guarantees positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing schedule
    beta_smooth = 2.0
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # --- smooth‑max weights (log‑sum‑exp) ---
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                 # for numerical stability
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --- gradient of Σ w_k·conv_k w.r.t. a (linear in a) ---
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2, central n entries needed
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # --- convert to gradient w.r.t. v (softmax Jacobian) ---
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping (helps stability)
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # --- Adam update on v ---
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent on the simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 1500000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always focuses on the current maximal convolution entry.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        k_move = rng.integers(2, 5)

        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        # incremental convolution update
        conv_new = cur_conv.copy()
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger size,
    followed by re‑normalisation.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(0.0, new_seq)
    s = new_seq.sum()
    if s == 0.0:
        return _uniform(new_n)
    return new_seq / s

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects a global time budget and returns the
    best sequence found (as a Python list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # keep any previously discovered construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # multiscale lengths – start cheap, then upscale
    base_n = 2000
    lengths = [base_n, 4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")
    prev_seq = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Seed creation
        # ------------------------------------------------------------------
        if n == base_n:
            seeds = _make_seeds(n, previous, rng)
        else:
            # start from an up‑sampled version of the best from the previous size
            seeds = []
            if prev_seq is not None:
                up = _upsample(prev_seq, n)
                seeds.append(up)
                # a few noisy perturbations of the up‑sampled vector
                for _ in range(3):
                    noisy = np.maximum(0.0, up + 0.01 * rng.standard_normal(size=n))
                    noisy /= noisy.sum()
                    seeds.append(noisy)
            # basic analytic baselines for the new size
            seeds.append(_uniform(n))
            seeds.append(_sine_window(n))
            seeds.append(_hann_window(n))
            seeds.append(_hamming_window(n))
            seeds.append(_blackman_window(n))
            # random Dirichlet draws
            for conc in (0.6, 1.0):
                seeds.append(_random_dirichlet(n, conc, rng))

        # ------------------------------------------------------------------
        # Quick evaluation of raw seeds
        # ------------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # Adam optimisation on a few promising seeds
        # ------------------------------------------------------------------
        top_k = 5
        top_idx = np.argsort(seed_vals)[:top_k]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            # pick a reasonable iteration budget depending on n
            max_iters = 20000 if n <= base_n else 8000
            seq, val = _adam_optimize(s, n, deadline, max_iters=max_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Exact sub‑gradient refinement (shorter for large n)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            max_iters_sub = 150000 if n <= base_n else 50000
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=max_iters_sub)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Light targeted simulated‑annealing phase (optional)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            max_iters_sa = 200000 if n <= base_n else 50000
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline,
                                             max_iters=max_iters_sa)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

        # store current best for the next upscale step
        prev_seq = global_best_seq.copy() if global_best_seq is not None else None

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = base_n
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity‑check with the provided evaluator
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    # expose the best sequence globally for possible reuse
    globals()["height_sequence_1"] = global_best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
