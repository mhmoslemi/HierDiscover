"""
Propose a non‑negative step‑function that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The optimiser works in three phases:

1️⃣  **Seed & Adam (soft‑max)** – a wide collection of parameterised
    seeds (uniform, geometric, beta‑like, sine, exponential, random
    Dirichlet and the previous best) is normalised to sum = 1.
    The best few seeds are refined by Adam on a log‑sum‑exp (soft‑max)
    approximation of the “max” operator.  The smoothing factor β is
    increased slowly and the Adam learning‑rate decays ∝ 1/√t.

2️⃣  **Exact sub‑gradient mirror descent** – from the Adam result we run
    exponentiated‑gradient descent on the true sub‑gradient of
    max(conv).  A classic 0.7/√t step‑size schedule is used; the
    iteration count is limited only by the remaining time budget.

3️⃣  **Aggressive simulated‑annealing** – a cheap incremental update of
    the full convolution enables millions of moves per second.
    Three move types are employed:
        • random Dirichlet re‑allocation on 2‑4 indices,
        • targeted re‑allocation that redistributes the mass of the
          indices contributing most to the current maximal convolution
          entry,
        • deterministic “shift” that moves a small ε from the index with
          the largest product a[i]·a[k*‑i] to the one with the smallest.
    Temperature starts at 5 % of the current objective and decays
    geometrically (factor 0.99995).  The SA phase dominates the budget
    and continuously updates the best sequence found.

The routine respects the global 1000 s budget (default) and returns
the best sequence as a plain ``list[float]``.  The final best sequence
is stored in the global variable ``height_sequence_1`` for possible
reuse by later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a probability vector (∑a = 1).
    The factor (∑a)^2 is omitted because a is already normalised.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = (1‑r)·r**i, normalised automatically."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # simple baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.2, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.5, 9):
        seeds.append(_power(n, p))

    # beta‑like shapes (asymmetric)
    for a in (0.2, 0.5, 1.0):
        for b in (0.2, 0.5, 1.0):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for sigma in (n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # logistic shapes – fast drop near beginning
    for scale in (n / 20, n / 10, n / 5):
        offset = scale * 2.0
        seeds.append(_logistic_shape(n, offset, scale))

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse previous solution (if appropriate length)
    if previous is not None and previous.shape == (n,):
        prev = np.maximum(0.0, previous)  # safety clamp
        s = prev.sum()
        if s > 0:
            seeds.append(prev / s)

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                 # parameters for a softmax
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.9

    beta = 5.0            # initial smoothness
    beta_mul = 1.015

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)               # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k w.r.t. a
        # ∂/∂a (a*a) = 2·rev(a) * w   (conv with reversed a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)     # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient steps on the true max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) at k_star
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                     # 2·a_{k*-i}

        lr = 0.7 / np.sqrt(it)                         # classic schedule
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                         # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# High‑frequency simulated‑annealing with incremental updates
# ----------------------------------------------------------------------


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.05,
    min_temp: float = 1e-12,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with three move types:
      1) random Dirichlet re‑allocation on 2‑4 indices,
      2) targeted Dirichlet on indices that contribute most to the
         current maximal convolution entry,
      3) deterministic shift from the index with largest product
         a[i]·a[k*‑i] to the one with smallest (or zero) product.
    The convolution is updated incrementally, allowing millions of
    cheap moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # full length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)   # initial temperature
    decay = 0.99995

    # pre‑allocate small buffers to avoid repeated allocations
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # annealing schedule
        T = max(T * decay, min_temp)

        move_type = rng.random()
        if move_type < 0.45:
            # -------- random Dirichlet move (2‑4 indices) ----------
            k = rng.integers(2, 5)                 # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)
        elif move_type < 0.9:
            # -------- targeted Dirichlet move --------------------
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)
            # second index uniformly from complement
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)
        else:
            # -------- deterministic shift move --------------------
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[support] * cur[k_star - support]
            # index with maximal contribution
            i_max = support[np.argmax(contrib)]
            # index with minimal contribution (prefer zero)
            i_min = support[np.argmin(contrib)]

            if cur[i_max] < 1e-12 or i_max == i_min:
                # fallback to random Dirichlet if nothing sensible
                k = rng.integers(2, 5)
                idx = rng.choice(n, size=k, replace=False)
                new_alloc = rng.dirichlet(np.full(k, 1.0)) * cur[idx].sum()
                deltas = new_alloc - cur[idx]
            else:
                # move a small ε from i_max to i_min
                eps_move = min(0.08 * cur[i_max], 0.002)   # tuned step size
                new_alloc = cur[[i_max, i_min]].copy()
                new_alloc[0] -= eps_move
                new_alloc[1] += eps_move
                idx = np.array([i_max, i_min], dtype=int)
                deltas = new_alloc - cur[idx]

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation for the selected positions (Dirichlet fallback already set)
        if 'new_alloc' not in locals():
            new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
            deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a  (commutative)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ term (self)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j for i≠j (appear twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

        # clean up in case we used the deterministic branch once
        if 'new_alloc' in locals():
            del new_alloc, deltas

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # previous solution, if any
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Choose a single length (the one that performed best before)
    # --------------------------------------------------------------
    # We focus on n = 4000 because the last run gave the best bound.
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Phase 0 – seed generation & quick evaluation
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    global_best_seq = None
    global_best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on smooth max (top‑k seeds)
    # ------------------------------------------------------------------
    # keep only a few promising seeds
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda t: t[1])
    top_k = min(5, len(seed_vals))
    for s, _ in seed_vals[:top_k]:
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑Adam] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient mirror descent
    # ------------------------------------------------------------------
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        seq, val = _subgradient_refine(
            global_best_seq,
            n,
            deadline,
            max_iters=250000,
        )
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall‑SubGrad] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – High‑frequency simulated‑annealing (dominant phase)
    # ------------------------------------------------------------------
    if global_best_seq is not None and global_best_seq.shape[0] == n:
        remaining = deadline - time.time()
        if remaining > 30.0:                     # ensure we have time left
            target_iters = int(min(5_000_000, remaining * 20000))
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                n,
                deadline,
                init_temp_factor=0.05,
                max_iters=target_iters,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with provided evaluate_sequence (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    # expose for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
