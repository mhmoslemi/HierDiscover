"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The search works as follows:

* Choose a modest length (n = 8000 for a full 1000 s budget) – larger
  n enlarges the feasible space and tightens the theoretical lower bound.
* Build a very diverse seed pool (analytic shapes, many Dirichlet draws,
  random piecewise‑linear, log‑normal, sinusoidal, exponential, interpolated
  warm‑start, etc.).
* Keep the best ℓ₁‑normalised seed.
* Run a sequence of time‑aware optimisation phases, each receiving a
  fraction of the remaining budget:
    1. Adam on a smooth‑max surrogate (soft‑max over the convolution entries).
    2. Exact sub‑gradient descent on the true maximum (exponentiated‑gradient).
    3. L₂‑flattening: minimise ½‖conv(a) – μ·1‖² where μ = 1/(2n‑1) (the average
       convolution value for a normalised vector).  This drives the auto‑
       convolution toward a constant.
    4. Deterministic flattening: repeatedly move a tiny amount of mass from
       the index that contributes most to the current maximal convolution
       coefficient to the index with the smallest value.
    5. Stochastic hill‑climb with a simple simulated‑annealing acceptance rule.
* After each phase the current best vector (always kept ℓ₁‑normalised) is
  compared against the global best.
* The final vector is verified with the provided ``evaluate_sequence``.
  If verification fails we fall back to the uniform distribution.

All phases respect the global time budget (≈ 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` for warm‑starting
future calls.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming sum(a) == 1.
    Fast surrogate used throughout the optimisation.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n, dtype=np.float64)
    x_new = np.linspace(0, 1, new_n, dtype=np.float64)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    total = seq_new.sum()
    if total > 0.0:
        seq_new /= total
    return seq_new

def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Very simple analytic families
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))               # uniform
    i = np.arange(n, dtype=np.float64)
    seeds.append(np.sin(np.pi * (i + 1) / (n + 1)) / np.sin(np.pi * (n + 1) / (n + 1)))  # half‑sine
    seeds.append(np.sin(2 * np.pi * (i + 0.5) / n) ** 2)           # sin² window
    seeds.append(np.abs(np.cos(np.pi * (i + 0.5) / n)))            # cosine absolute

    # 2. Geometric / exponential decays
    for r in np.linspace(0.2, 0.97, 12):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # 3. Power‑law families
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # 4. Beta‑like shapes
    for alpha in np.linspace(0.4, 2.2, 5):
        for beta in np.linspace(0.4, 2.2, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 5. Dirichlet draws with many concentration values
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        for _ in range(8):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # 6. Gaussian‑based logistic (softmax) draws
    for _ in range(8):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # 7. Random piecewise‑linear shapes
    for _ in range(8):
        K = max(8, int(np.sqrt(n) * rng.uniform(0.4, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # 8. Log‑normal (bell) shapes
    for _ in range(8):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 9. Warm‑start (interpolated if necessary)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds


def _extra_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a few hundred completely random seeds for extra diversity."""
    seeds = []
    # Random Dirichlet with random concentration per draw
    for _ in range(150):
        conc = rng.uniform(0.01, 8.0)
        a = rng.dirichlet(np.full(n, conc))
        seeds.append(a / a.sum())
    # Random mixtures of two exponential tails
    for _ in range(30):
        a = np.zeros(n, dtype=np.float64)
        left_len = rng.integers(1, n // 2)
        right_len = n - left_len
        r1 = rng.uniform(0.2, 0.9)
        r2 = rng.uniform(0.2, 0.9)
        a[:left_len] = r1 ** np.arange(left_len)
        a[left_len:] = r2 ** np.arange(right_len)
        seeds.append(a / a.sum())
    return seeds


def _select_best_seed(seeds: list[np.ndarray], n: int) -> tuple[np.ndarray, float]:
    """Return the best ℓ₁‑normalised seed according to the surrogate objective."""
    best_seq = None
    best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val

def _adam_optimize(a0: np.ndarray, n: int,

    Adam on a smooth‑max surrogate.  Parameterisation a = softmax(v).
    The temperature β of the soft‑max is increased multiplicatively.
    """
    eps = 1e-12
    v = np.log(a0 + eps)                    # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.6
    beta1, beta2 = 0.9, 0.999

    beta = 5.0          # initial smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        # soft‑max to obtain a (stays on the simplex)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 1000 == 0:
                print(f"[Adam] it={t:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]      # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² with μ = 1/(2n‑1).  This pushes the auto‑convolution
    toward a constant vector (the theoretical optimum if it were achievable).
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _deterministic_flatten_optimize(a0: np.ndarray, n: int,
                                   deadline: float,
                                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted deterministic mass transfer: each iteration moves a tiny amount of
    probability mass away from the index that contributes most to the current
    maximal convolution entry, toward the smallest component.
    """
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of indices to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]

        donor = idx_i[np.argmax(contrib)]          # biggest contributor
        # Choose receiver as the index with the smallest current weight (excluding donor)
        receiver = np.argmin(cur + (np.arange(n) == donor) * 1.0)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        # re‑normalise
        new_sum = new.sum()
        if new_sum > 0.0:
            new /= new_sum

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[DetFlat] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.85, 1e-6)

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a simple simulated‑annealing acceptance rule.
    Mass is transferred between two randomly chosen positions; uphill moves
    are occasionally accepted to escape shallow local minima.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.02
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # donor selection based on contribution to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        new_sum = new.sum()
        if new_sum > 0.0:
            new /= new_sum

        val = _objective(new, n)
        # Acceptance rule
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / (0.001 + 1e-12)):
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  Returns the best sequence found as a
    plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Length choice (larger n gives a tighter theoretical lower bound)
    # ------------------------------------------------------------------
    # We go for 8000 elements – this still fits comfortably into the time budget.
    n = 8000

    # ------------------------------------------------------------------
    # Warm‑start handling (interpolate if size differs)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
            warm_arr = _interpolate(warm_arr, n)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Seed generation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 987654)
    seeds = _make_seeds(n, rng, warm_arr) + _extra_seeds(n, rng)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds
    # ------------------------------------------------------------------
    global_best_seq, global_best_val = _select_best_seed(seeds, n)
    print(f"[Seed] initial global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Time‑aware optimisation phases
    # ------------------------------------------------------------------
    phases = [
        (_adam_optimize, 0.35),
        (_subgradient_optimize, 0.25),
        (_l2_flatten_optimize, 0.20),
        (_deterministic_flatten_optimize, 0.10),
        (_hill_climb_optimize, 0.10),
    ]

    for opt_func, fraction in phases:
        now = time.time()
        remaining = deadline - now
        if remaining <= 2.0:
            break
        phase_deadline = now + remaining * fraction
        print(f"--- Phase {opt_func.__name__} (up to {remaining * fraction:.1f}s) ---")
        seq, val = opt_func(global_best_seq, n, phase_deadline, max_iters=500_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[{opt_func.__name__}] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Store for warm‑starting future calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
