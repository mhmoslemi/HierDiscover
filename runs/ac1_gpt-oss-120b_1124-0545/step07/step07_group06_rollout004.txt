"""
Propose a non‑negative step function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is a “multi‑scale projected‑gradient” search:

1.  Try a few analytically motivated seed families (uniform, U‑shaped
    beta‑shapes, inverse‑product, random Dirichlet / sparse draws).
2.  For each seed evaluate the cheap objective
    ``2·n·max(conv(a,a))`` (the denominator is 1 because the seeds are
    normalised to sum = 1).  Keep the best few seeds.
3.  Run a dedicated projected‑gradient descent on the *squared L₂*
    deviation of the full convolution from the uniform target
    ``1/(2n−1)``.  The gradient can be written as a convolution and is
    computed with a real FFT – this makes many thousands of iterations
    cheap.
4.  After the gradient phase optionally polish the result with the
    sub‑gradient mirror‑descent already used in the reference code.
5.  If time permits repeat the whole procedure on a larger length,
    interpolating the current best solution to the new grid.
6.  Return the best sequence seen (it is automatically clipped to the
    feasible domain by ``evaluate_sequence``).

The search respects the supplied time budget; it prints occasional
progress messages and stops well before the deadline so that the
runtime environment never forces a timeout.
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolutions)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D vectors")

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest ρ satisfying u_ρ - (cssv_ρ / ρ) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the denominator is 1 because all candidates are
# normalised to sum = 1, therefore the objective reduces to 2·n·max(conv)).
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector a that already sums to 1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all are returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_u_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    U‑shaped beta‑like density: a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1} with 0<α,β<1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _inv_product(n: int) -> np.ndarray:
    """
    Heavily edge‑biased shape: a[i] ∝ 1/((i+1)*(n-i)).
    """
    i = np.arange(n, dtype=np.float64) + 1.0          # i+1
    w = 1.0 / (i * (n - i + 1.0))
    s = w.sum()
    return w / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Sparse Dirichlet on a random support of size k (k<n).
    """
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """
    Assemble a diversified list of normalised seeds.
    """
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_inv_product(n))

    # a few U‑shaped beta families (α,β<1)
    for a, b in [(0.4, 0.4), (0.5, 0.5), (0.6, 0.6), (0.7, 0.3), (0.3, 0.7)]:
        seeds.append(_beta_u_shape(n, a, b))

    # geometric decay (for curiosity – usually bad but kept)
    for r in np.linspace(0.15, 0.95, 5):
        geom = r ** np.arange(n, dtype=np.float64)
        s = geom.sum()
        if s > 0:
            seeds.append(geom / s)

    # random Dirichlet draws (different concentrations)
    for conc in (0.3, 0.7, 1.5, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (3, 5, 9, 15):
        for conc in (0.2, 0.5, 0.9):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds


# ----------------------------------------------------------------------
# Projected gradient descent on the squared L₂ deviation from a uniform
# convolution.  This is the “new” optimisation primitive.
# ----------------------------------------------------------------------


def _pgd_sq_error(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise ||conv(a,a) - T||₂²  (T = uniform vector of length 2n‑1)
    with projected gradient descent on the simplex.
    Returns the best sequence seen and its 2·n·max(conv) value.
    """
    a = seed.copy()
    # target uniform convolution (sum of conv is 1, so average = 1/(2n‑1))
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_seq = a.copy()
    best_val = _objective(a)

    # initial step size – tuned empirically
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and residual
        conv = _conv_fft(a, a)                # length 2n‑1
        resid = conv - target                 # length 2n‑1

        # gradient of 4·∑ (conv−T)·a_rev  (central n entries)
        grad_full = _conv_fft(resid, a[::-1])  # length 3n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto simplex
        a_candidate = a - eta * grad
        a = _proj_simplex(a_candidate)

        # evaluate the true (max‑based) objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[PGD] n={n:5d} it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

def _subgradient_refine(

    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent (exponentiated‑gradient)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step (exponentiated gradient)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the global
    time budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the surrounding infrastructure
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Retrieve a previously found construction, if any, and remember it
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to consider – we start modest and enlarge if time permits.
    # ------------------------------------------------------------------
    length_schedule = [2000, 4000, 6000, 8000]  # increasing resolution

    global_best_seq: np.ndarray | None = None
    global_best_val: float = float("inf")

    for idx, n in enumerate(length_schedule):
        # stop early if we run out of time
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # Initialise seeds – reuse the current best (interpolated) when
        # moving to a larger length.
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] != n:
            # linear interpolation onto the new grid
            old_n = global_best_seq.shape[0]
            x_old = np.arange(old_n)
            x_new = np.linspace(0, old_n - 1, n)
            interpolated = np.interp(x_new, x_old, global_best_seq)
            base_for_seeds = interpolated / interpolated.sum()
        else:
            base_for_seeds = None

        seeds = _make_seeds(n, base_for_seeds, rng)

        # --------------------------------------------------------------
        # 1) Quick evaluation of raw seeds (objective = 2·n·max(conv))
        # --------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # --------------------------------------------------------------
        # 2) Focused projected‑gradient exploitation on the most promising
        #    seeds (top‑3 according to the cheap objective).
        # --------------------------------------------------------------
        top_k = min(3, len(seeds))
        top_idx = np.argsort(seed_vals)[:top_k]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _pgd_sq_error(s, n, deadline, max_iters=150_000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑PGD] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 3) Polish the currently best sequence with exact sub‑gradient.
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=300_000)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Early exit if we already beat the target bound.
        # --------------------------------------------------------------
        if global_best_val <= 1.503:
            print(f"Target bound reached with value {global_best_val:.6f}")
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a return value.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest 2·n·max(conv) ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    # The evaluator works with any scaling; we return the probability vector.
    return global_best_seq.tolist()
