"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)^2)

using a hybrid of:

* diverse analytic/random seeds,
* Adam on a smooth‑max surrogate,
* exact sub‑gradient descent,
* a linear‑programming “lifting” step (maximise the total mass of a
  vector that can be added without blowing up the current max‑convolution),
* L2‑flattening (drive the auto‑convolution toward a constant),
* stochastic hill‑climb.

All phases respect a 1000 s wall‑clock budget.
The final sequence is stored in the global variable ``height_sequence_1`` for
warm‑starting future calls.
"""

import time
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Optional imports – fall back gracefully if they are missing
# ------------------------------------------------------------
try:
    import cvxpy as cp

    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False


# ------------------------------------------------------------
# Low‑level helpers
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)).  Assumes sum(a)=1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


# ------------------------------------------------------------
# Seed generation (analytic, random and warm‑start)
# ------------------------------------------------------------
def _make_seeds(
    n: int, rng: np.random.Generator, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Uniform and half‑sine window
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # Geometric families
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())

    # Power‑law families
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # Softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # Random piecewise‑linear shapes
    for _ in range(5):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # Log‑normal (Gaussian bell) shapes
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # Warm‑start (previous best)
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds


# ------------------------------------------------------------
# Optimisation primitives
# ------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate."""
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0
    beta_mul = 1.015

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum of conv entries
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, deadline: float,
                          max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int, deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² (μ = 1/(2n‑1)) using exponentiated‑gradient.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away from the
    contributors to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[Hill] it={it:6d} best={val:.6f}")

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val


def _lp_improve(a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Linear‑programming lifting step.

    Maximise Σ b_j subject to b≥0 and   (a * b)_k ≤ M  (full convolution)
    where M = max(conv(a,a)).  The obtained b is renormalised to unit sum
    and mixed with ``a`` (step size t=0.2).  Returns the mixed vector and its
    objective value.
    """
    if not _HAS_CVXPY:
        # No LP capability – just return the input unchanged.
        return a, _objective(a, n)

    # Current maximal self‑convolution
    conv_aa = _conv_full(a, a)
    M = np.max(conv_aa)

    # Build the (2n‑1)×n convolution matrix A such that A @ b = conv(a,b)
    # A is Toeplitz with constant diagonals equal to a_i.
    offsets = np.arange(n, dtype=int)
    # each diagonal i contains a[i] repeated n times
    diag_data = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")

    # CVXPY model
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M + 1e-12]  # safety epsilon
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, feastol=1e-9, reltol=1e-9)
    except Exception as exc:  # pragma: no cover
        print(f"[LP] Solver exception: {exc}")
        return a, _objective(a, n)

    if b.value is None:
        print("[LP] Solver failed to find a solution.")
        return a, _objective(a, n)

    b_opt = np.asarray(b.value, dtype=np.float64).ravel()
    s = b_opt.sum()
    if s <= 0.0:
        return a, _objective(a, n)

    # Normalise and mix
    g = b_opt / s
    t = 0.2  # mixing coefficient
    new = (1.0 - t) * a + t * g
    new = np.maximum(new, 0.0)
    new_sum = new.sum()
    if new_sum > 0.0:
        new = new / new_sum
    else:
        new = a.copy()

    return new, _objective(new, n)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """

    `evaluate_sequence`.  Returns the best sequence found as a list of
    Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    n = 4000  # fits comfortably and yields a good theoretical bound

    # ------------------------------------------------------------------
    # Build seed pool
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)

    # ------------------------------------------------------------------
    # Evaluate all seeds – keep the best one

    global_best_seq = None
    global_best_val = float("inf")

            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------

    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
