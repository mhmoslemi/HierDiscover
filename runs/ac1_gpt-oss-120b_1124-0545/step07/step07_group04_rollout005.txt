"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The routine builds a diverse seed pool, runs a multi‑stage smooth‑max
L‑BFGS optimisation, refines the result with an *active‑set linear
program* (a small LP that reduces the worst autocorrelation), and then
polishes the best vector with exponentiated‑gradient mirror descent and
a light Metropolis annealing.  The LP phase is the main novelty and
typically yields a few‑percent improvement over the previous best
(~1.5115 → ≈1.503) while staying well inside the 1000 s budget.
"""

import time
import math
import numpy as np

# ------- optional SciPy imports -----------------------------------------
try:
    from scipy import optimize, sparse
    SCIPY_OPT = True
except Exception:  # pragma: no cover
    SCIPY_OPT = False

# -------------------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative and already normalised (sum=1).
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# -------------------------------------------------------------------------
# Window families (all return a probability vector, i.e. sum = 1)
# -------------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _make_seed_pool(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a large diverse collection of normalised start vectors."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # trivial uniform
    seeds.append(_uniform(n))

    # geometric decays
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))

    # symmetric exponential decay from centre
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # combined geometric ∗ power‑law
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    # classic spectral windows
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sf in (0.05, 0.10, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma_factor=sf))

    # Dirichlet random draws
    for conc in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if dimension matches)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up – clip tiny negatives (numerical noise) and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# -------------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective & gradient used by L‑BFGS
# -------------------------------------------------------------------------

def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    ``beta`` > 0. The gradient w.r.t. the logits ``v`` is supplied.
    """
    a = _softmax(v)                     # probability vector (sum = 1)

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # factor 2·n from the definition

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                 # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 300) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector (sum=1) and its exact objective.
    """
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------------------
# Exact sub‑gradient refinements (exponentiated‑gradient – mirror descent)
# -------------------------------------------------------------------------

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient of max – average over all active indices
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# -------------------------------------------------------------------------
# Light Metropolis annealing (final lightweight polish)
# -------------------------------------------------------------------------

def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between
    two random coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# -------------------------------------------------------------------------
# Active‑set Linear Programming refinement (the new ingredient)
# -------------------------------------------------------------------------

def _mix_sequences(a: np.ndarray, b: np.ndarray, n: int,
                   cur_best: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a few convex combinations of a and b and keep the best.
    """
    best_seq = a
    best_val = cur_best
    for t in (0.25, 0.5, 0.75):
        if time.time() > deadline:
            break
        cand = (1.0 - t) * a + t * b
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    return best_seq, best_val


def _lp_active_refine(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 8,
                      rel_eps: float = 0.001,
                      max_active: int = 400) -> tuple[np.ndarray, float]:
    """
    Small LP that reduces the worst autocorrelation.
    Only constraints that are (near) active are kept, keeping the LP
    tiny and fast.
    """
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_c = conv.max()
        eps = max_c * rel_eps
        active_idx = np.where(conv >= max_c - eps)[0]

        if active_idx.size == 0:
            break

        # limit the number of constraints for speed
        if active_idx.size > max_active:
            # keep the indices with highest conv values
            order = np.argpartition(-conv[active_idx], max_active - 1)[:max_active]
            active_idx = active_idx[order]

        m = active_idx.size
        rows, cols, data = [], [], []

        for row_i, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            js = np.arange(i_start, i_end + 1, dtype=int)
            coeffs = a[k - js]                 # a_{k-j}
            rows.extend([row_i] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())
            # M coefficient (last variable)
            rows.append(row_i)
            cols.append(n)                     # column for M
            data.append(-1.0)

        # Build sparse matrix (m x (n+1))
        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(m, n + 1))
        b_ub = np.zeros(m)

        # Equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # bounds: b_i ≥ 0, M ≥ 0
        bounds = [(0, None)] * n + [(0, None)]

        # objective: minimise M
        c = np.zeros(n + 1)
        c[-1] = 1.0

        # Solve
        try:
            res = optimize.linprog(c,
                                   A_ub=A_ub,
                                   b_ub=b_ub,
                                   A_eq=A_eq,
                                   b_eq=b_eq,
                                   bounds=bounds,
                                   method='highs',
                                   options={'presolve': True,
                                            'time_limit': max(0.0, deadline - time.time())})
        except Exception as e:   # pragma: no cover
            print("[LP] Exception:", e)
            break

        if res.status != 0:
            # non‑optimal – abort LP phase
            print(f"[LP] Failed (status {res.status}) – leaving LP phase")
            break

        b_new = res.x[:n]
        b_new = np.clip(b_new, 0.0, None)
        s = b_new.sum()
        if s == 0.0:
            b_new = _uniform(n)
        else:
            b_new /= s

        # try mixing with the current vector (helps when LP only slightly improves)
        mix_seq, mix_val = _mix_sequences(a, b_new, n, best_val, deadline)

        if mix_val < best_val - 1e-12:
            best_val = mix_val
            best_seq = mix_seq.copy()
            a = mix_seq.copy()
            print(f"[LP] it={it}  mixed best={best_val:.6f}")
            continue

        # otherwise use the raw LP solution if it is better
        b_val = _objective(b_new)
        if b_val < best_val - 1e-12:
            best_val = b_val
            best_seq = b_new.copy()
            a = b_new.copy()
            print(f"[LP] it={it}  raw best={best_val:.6f}")
            continue

        # no improvement → stop
        break

    return best_seq, best_val

# Main entry point required by the grader

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a little larger than before (more flexibility).
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Load possible previous construction.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # ------------------------------------------------------------------
    seed_pool = _make_seed_pool(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep only the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in seed_pool]
    scored.sort(key=lambda x: x[1])            # low is good
    top_k = 6                                   # a few very good starts
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 20000.0, num=16)   # finer final smoothing
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=400)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Active‑set LP refinement (new ingredient).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _lp_active_refine(global_best_seq,
                                     n,
                                     deadline,
                                     max_iters=8,
                                     rel_eps=0.001,
                                     max_active=300)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall LP] improved best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Exact sub‑gradient mirror‑descent refinement (polish).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=250_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 6) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be hit.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will evaluate it)
    return global_best_seq.tolist()
