def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)

def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:

    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()

    a = (
        0.42

    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Uniform mass on a random support of size `m` (1 ≤ m ≤ n).
    """
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """
    Deterministic “Sidon‑like” support using triangular numbers.
    The positions are i·(i+1)//2 modulo n, which tends to give many
    distinct pair‑sums for moderate m.
    """
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """
    Positions are i² (mod n).  For a prime n this set is a Sidon set;
    for composite n it still yields a fairly flat sum‑set.
    """
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

# Low‑collision B₂[2] set generator
# ----------------------------------------------------------------------


def _low_collision_set(n: int,
                       target_m: int,
                       max_ordered: int,
                       rng: np.random.Generator,
                       deadline: float) -> np.ndarray | None:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that every ordered
    pair‑sum i+j (i,j∈S) occurs at most `max_ordered` times.
    Tries many random permutations until the time budget runs out.
    Returns the index array of the best set found (the one with most
    elements, up to `target_m`).
    """
    best_mask = None
    best_size = 0
    attempt = 0
    max_attempts = 200  # safety net

    while time.time() < deadline - 0.2 and attempt < max_attempts:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        # ordered‑sum counter; length 2n‑1, values are small (≤max_ordered)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

        order = rng.permutation(n)
        for cand in order:
            # quick check: would the self‑pair overflow?
            if sum_counts[cand * 2] + 1 > max_ordered:
                continue

            # compute sums with already chosen indices
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing               # vectorised
                # each new distinct pair contributes 2 ordered pairs
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue

            # accept the candidate
            mask[cand] = True
            sum_counts[cand * 2] += 1
            if existing.size:
                sum_counts[sums] += 2

            if mask.sum() >= target_m:
                break

        cur_size = mask.sum()
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempt:3d} size {cur_size}")

        # early exit if we already reached the theoretical limit
        # (which is roughly sqrt(2*n*max_ordered/2))
        if best_size >= target_m:
            break

    if best_mask is None or best_size == 0:
        return None
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)  # logits for simplex parametrisation

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = beta * beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k · (a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam moment update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over all active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 5) -> np.ndarray:
    """
    Perform a few cheap linear‑programming refinements.
    For each refinement we:
        * compute the current maximum M of conv(f,f)
        * collect the (few) indices where the max is attained (active set)
        * solve   maximise 1ᵀ g   subject to   f * g ≤ M  and g ≥ 0
          (the constraints are linear because f is fixed)
        * blend the solution g back into f with weight `t_mix`.
    Returns the refined vector (still on the simplex).
    """
    if not CVXPY_AVAILABLE:
        # cvxpy not present – fall back to a simple heuristic
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two random positions and occasionally adds a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        if delta > cur[i]:
            delta = cur[i]

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

# Seed generation (including many sparse candidates)
# ----------------------------------------------------------------------


def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    Adds classic families and a large collection of sparse random vectors.
    """
    seeds: list[np.ndarray] = []

    # classic families (as in the original script)
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # Sparse random seeds – support sizes 30 … 200
    for m in range(30, 210, 20):
        for _ in range(5):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like structures
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # reuse previous construction if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and returns the best sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Build a rich seed pool, adding a low‑collision B₂[2] construction
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # add a low‑collision set (ordered sums ≤ 4) if time permits
    lc_set = _low_collision_set(
        n=n,
        target_m=150,
        max_ordered=4,
        rng=rng,
        deadline=deadline,
    )
    if lc_set is not None and lc_set.size > 0:
        a = np.zeros(n, dtype=np.float64)
        a[lc_set] = 1.0 / lc_set.size
        a /= a.sum()
        seeds.append(a)
        print(f"[seed] added low‑collision set of size {lc_set.size}")

    # ------------------------------------------------------------------
    # Quick evaluation – keep the top few for the expensive phases
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(20, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best obj = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on smoothed max (more iterations)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=25000)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient mirror‑descent (duration adapts to time)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 5.0:
        # allocate a generous share of the remaining budget
        remaining = deadline - time.time()
        # rough cost of one iteration ≈ 0.0015 s, cap iterations accordingly
        max_sub_iters = int(min(500000, remaining / 0.0015))
        seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                       max_iters=max_sub_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Aggressive LP‑refinement (multiple passes)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and global_best_seq is not None and time.time() < deadline - 2.0:
        lp_seq = global_best_seq.copy()
        # run a few LP passes while time permits
        while time.time() + 5.0 < deadline:
            lp_seq = _lp_refine(lp_seq, n, deadline, t_mix=0.3, max_lp_iters=10)
            # short sub‑gradient clean‑up after each LP pass
            sub_seq, sub_val = _subgradient_refine(lp_seq, n, deadline,
                                                   max_iters=50000)
            if sub_seq is not None and sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[LP+Sub] new best {sub_val:.6f}")
                lp_seq = sub_seq.copy()
            else:
                break

    # ------------------------------------------------------------------
    # Phase D – Simulated annealing fine‑tuning (uses all remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # final normalisation (protect against tiny negative drift)
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
