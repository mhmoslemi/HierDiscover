"""
propose_candidate
-----------------
Searches for a non‑negative step function (a probability vector) that
minimises the ``evaluate_sequence`` objective:

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is deliberately time‑aware (budget ≈ 1000 s) and consists
of four stages:

1. **Deterministic seeds** – a few analytically‑motivated families:
   uniform, the classic Catalan shape, and simple power‑law families
   a_i ∝ (i+δ)^‑p with several exponents p and shifts δ.

2. **Massive random Dirichlet sampling** – draws many Dirichlet(α)
   vectors for several lengths (2000 … 8000) with α sampled on a
   logarithmic scale.  The best sample is retained.

3. **Smooth ‑gradient descent** – from the currently best vector a
   performs projected gradient descent on the smooth approximation
   of the max‑norm:
        f(a)=2·n·log(∑_k exp(β·conv_k(a,a)))/β
   The gradient is obtained by a double convolution and the iterate
   is projected onto the probability simplex.  This stage can escape
   shallow local minima.

4. **Local random hill‑climbing** – a cheap simulated‑annealing
   style perturbation that repeatedly moves a small amount of mass
   between two coordinates, accepting improvements or, with a
   temperature that decays to zero, occasional worsening moves.

The best vector found (as a plain ``list``) is returned.  All helper
functions are defined at module level and no closures or file‑IO are
used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution, next power of two)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for moderate n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (for normalised vectors, i.e. sum == 1)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` when ``seq`` already satisfies
    sum(seq) == 1.  Returns   2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution on n points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Truncated coefficients of (1‑z)^‑½
    (i.e. binom(2i,i)/4**i).  Normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    # fall back to uniform if something went wrong
    if total <= 0.0:
        return _uniform(n)
    return coeffs / total


def _power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (i + shift)^‑p   (i = 0 … n‑1)
    Returned vector is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64) + shift
    # avoid division by zero – shift must be > 0 if p > 0
    idx = np.where(idx <= 0.0, 1e-12, idx)
    coeffs = idx ** (-p)
    total = coeffs.sum()
    if total <= 0.0:
        return _uniform(n)
    return coeffs / total


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where u_i > cssv_i / i
    cond = u > cssv / ind
    if not np.any(cond):
        # all components are zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Stage 3 – smooth (log‑sum‑exp) projected gradient descent
# ----------------------------------------------------------------------


def _smooth_optimize(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation of `max`.
    Stops when `time.time()` exceeds `deadline`.  Returns the best
    vector encountered and its exact (non‑smooth) objective.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        # current auto‑convolution
        conv = _conv_fft(a, a)                     # length 2n‑1
        # temperature (β) – start smooth and become sharper
        beta = 5.0 + min(it * 0.005, 30.0)          # β ∈ [5,35]
        # soft‑max weights
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed function:
        #   ∇f = 2·n·Σ_i w_i ∂(a*a)_i/∂a  = 4·n·(w * a_rev) (restricted to length n)
        a_rev = a[::-1]
        conv_rev = _conv_fft(w, a_rev)                 # length 3n‑2
        grad = 4.0 * n * conv_rev[n - 1:n - 1 + n]    # slice of length n

        # diminishing step size
        lr = 0.2 / np.sqrt(it)

        # gradient step followed by projection onto the simplex
        a_new = _project_simplex(a - lr * grad)

        a = a_new
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Smooth] it={it:6d} n={n:5d} val={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stage 4 – cheap random hill‑climbing / simulated annealing
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex: repeatedly move a small amount
    of probability mass between two coordinates.  Accept improvements
    always; occasional worsening moves are accepted with a temperature
    that decays linearly to zero.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    # initial temperature (relative to current objective)
    T0 = max(1e-3, best_val * 0.05)

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # move at most 1 % of the source mass
        delta = min(a[i], rng.random() * 0.01 * a[i])
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            a = cand
            print(f"[Local] it={it:6d} n={n:5d} val={val:.6f}")
        else:
            # linearly decaying temperature
            remaining = max(deadline - time.time(), 0.0)
            fraction = remaining / (deadline - start_seq.__dict__.get('deadline_start', deadline))
            T = T0 * fraction
            if rng.random() < np.exp(-(val - best_val) / max(T, 1e-9)):
                a = cand

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Lengths that will be explored (chosen empirically)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional previous solution (global variable ``height_sequence_1``)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        # uniform and Catalan
        for seq in (_uniform(n), _catalan_shape(n)):
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type={'Uniform' if seq is _uniform(n) else 'Catalan':8s} val={val:.6f}")

        # power‑law families (p around the known optimum 0.5, small shifts)
        for p in (0.45, 0.5, 0.55):
            for shift in (0.0, 0.5, 1.0, 2.0):
                seq = _power_law(n, p, shift)
                val = _objective(seq)
                if val < best_val:
                    best_val, best_seq = val, seq.copy()
                    print(f"[Seed] n={n:5d} p={p:.2f} shift={shift:.1f} val={val:.6f}")

        # previous solution, if it matches the current length
        if previous is not None and previous.shape[0] == n:
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – massive Dirichlet sampling (different concentrations)
    # ------------------------------------------------------------------
    now = time.time()
    # allocate roughly half of the remaining budget for random sampling
    sampling_deadline = now + (deadline - now) * 0.45

    while time.time() < sampling_deadline:
        # pick a length (bias towards larger ones)
        n = int(rng.choice(lengths, p=np.array(lengths) / sum(lengths)))
        # concentration α sampled log‑uniformly in [0.05, 5]
        log_alpha = rng.uniform(np.log(0.05), np.log(5.0))
        alpha = np.exp(log_alpha)
        # Dirichlet draw (the same α for all coordinates)
        x = rng.gamma(alpha, 1.0, size=n)
        seq = x / x.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.3f} val={val:.6f}")
        # early exit if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – smooth projected gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        # give the gradient routine the remaining time (≈30 % of budget)
        gd_deadline = time.time() + (deadline - time.time()) * 0.30
        refined, val_refined = _smooth_optimize(best_seq, gd_deadline, rng)
        if val_refined < best_val:
            best_val, best_seq = val_refined, refined.copy()
            print(f"[Refine‑Smooth] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – local random hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # remaining time (about 20 % of the whole budget)
        local_deadline = deadline
        # store the original deadline inside the array for temperature calc
        best_seq.__dict__ = {'deadline_start': time.time()}
        final_seq, val_final = _local_search(best_seq, local_deadline, rng)
        if val_final < best_val:
            best_val, best_seq = val_final, final_seq.copy()
            print(f"[Refine‑Local] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also call ``evaluate_sequence``)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
