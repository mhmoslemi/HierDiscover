import cvxpy as cp

def lp_step(f, M):
    n = len(f)
    # build sparse matrix T_f
    rows = []
    cols = []
    data = []
    for j in range(n):
        # columns fixed j
        # rows = j:(j+n)
        for d in range(n):
            row = j + d
            col = j
            val = f[d]
            rows.append(row)
            cols.append(col)
            data.append(val)
    # produce scipy.sparse.csr_matrix
    T = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    # Check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback: return f unchanged
        return f, 1.0
    g_val = np.maximum(g.value, 0)
    sum_g = g_val.sum()
    if sum_g <= 0:
        return f, 1.0
    return g_val, sum_g

Now implement simplex projection:

def _project_simplex(v):
    """Projection of vector v onto probability simplex sum=1, v_i >=0."""
    # Based on Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Iter] it={iter_cnt:7d} val={cur_val:.6f}")

        # Update temperature
        tau = min(tau_max, tau * 1.001)   # slowly increase

        # softmax weights
        cmax = np.max(conv)
        # for numerical stability
        exp_weights = np.exp(tau * (conv - cmax))
        w = exp_weights / exp_weights.sum()

        # gradient via FFT convolution w_rev * a
        w_rev = w[::-1]
        grad_full = _conv_fft_full(w_rev, a)
        grad = 2.0 * grad_full[2*n-2 : 2*n-2 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)

        step = step_alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

    # final sanity
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as e:
        print("Evaluation error:", e)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Search algorithm for a non‑negative step‑function that minimises
``evaluate_sequence``.  The method works in three time‑aware phases:

1️⃣  Deterministic seeds – Catalan‑shaped coefficients, uniform vector and a
    few random Dirichlet samples are evaluated and the best one is kept.

2️⃣  Short random Dirichlet sweep – while a small time slice remains,
    additional Dirichlet(1) vectors are drawn to possibly discover a
    better starting point.

3️⃣  Smooth‑max Adam optimisation – the current best vector is refined on
    the probability simplex.  A soft‑max (temperature τ) approximates the
    ``max`` in the objective, allowing a dense sub‑gradient.  Adam’s
    adaptive steps together with projection onto the simplex give a fast
    descent and, in practice, improve the bound beyond the simple
    exponentiated‑gradient approach used previously.

The routine respects a user supplied time budget (default 1000 s) and
returns the best sequence found as a plain ``list`` of ``float`` ready
for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse  # only for the optional previous‑solution check (can be omitted)

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (returns length len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Quantity that ``evaluate_sequence`` would compute for a *normalized*
    vector (i.e. sum(seq) == 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Coefficients proportional to binom(2i,i)/4**i (series of (1‑z)^‑½),
    truncated to length n and normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _random_dirichlet_one(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(1,…,1) vector (normalised exponentials)."""
    x = rng.exponential(1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # a small safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a length – larger n usually yields a smaller bound.
    # Use 8000 when the budget is generous, otherwise fall back to 4000.
    # ------------------------------------------------------------------
    n = 8000 if budget_s >= 300 else 4000

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic seeds
    # ------------------------------------------------------------------
    candidates = [
        _catalan_shape(n),
        _uniform(n),
    ]

    # a handful of random Dirichlet seeds (cheap, give variety)
    for _ in range(5):
        candidates.append(_random_dirichlet_one(n, rng))

    best_seq = None
    best_val = float("inf")
    for seq in candidates:
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Seed] type={'catalan' if np.allclose(seq, _catalan_shape(n)) else 'uniform' if np.allclose(seq, _uniform(n)) else 'rand'}  val={val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short random Dirichlet sweep (≈10 % of the remaining budget)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sweep_deadline = now + min(30.0, 0.15 * remaining)   # at most 30 s or 15 % of remaining time

    while time.time() < sweep_deadline:
        seq = _random_dirichlet_one(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] val={val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam optimisation on the simplex
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate, always sum = 1
    m = np.zeros_like(a)                    # Adam first moment
    v_adam = np.zeros_like(a)               # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    alpha = 0.12                            # base step size for Adam
    tau = 5.0                               # soft‑max temperature (starts low)
    tau_max = 5000.0

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Iter] it={it:7d} val={cur_val:.6f}")

        # anneal temperature slowly (geometric increase)
        tau = min(tau_max, tau * 1.001)

        # soft‑max (smooth approximation of max)
        cmax = np.max(conv)                 # for numerical stability
        exp_w = np.exp(tau * (conv - cmax))
        w = exp_w / exp_w.sum()             # weights sum to 1

        # gradient of the smooth objective:
        #   g = 2 * (w_rev  * a)   (cross‑correlation)
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)     # full length = 2*n-1 + n -1 = 3n-2
        # extract the slice that corresponds to the gradient entries
        start = 2 * n - 2
        grad = 2.0 * grad_full[start:start + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # keep a on the probability simplex
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Final sanity check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
