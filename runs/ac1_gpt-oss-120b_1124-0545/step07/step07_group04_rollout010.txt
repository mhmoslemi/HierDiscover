def _project_simplex(v):
    """Projects onto the probability simplex (non-negative and sum=1)"""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function that (approximately) minimises

    2·n·max(convolve(a,a)) / (sum(a)²)

using a richer seed pool, a fine‑grained smooth‑max L‑BFGS phase and a
longer projected Adam (mirror‑descent) refinement.  The routine
honours a 1000 s wall‑time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    from scipy import optimize
    SCIPY_OPT = True
except Exception:   # pragma: no cover
    SCIPY_OPT = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=1}.
    Returns a new array (the original ``v`` is not modified).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero – fallback
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise for numerical safety
    w_sum = w.sum()
    if w_sum > 0.0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Window families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with a uniform concentration ``alpha``."""
    return rng.dirichlet(np.full(n, alpha))

# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective & gradient used by L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation
        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    ``beta`` > 0.  The gradient w.r.t. the logits ``v`` is supplied.
    """
    a = _softmax(v)                     # probability vector (sum = 1)

    conv = _conv_fft(a, a)              # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # factor 2·n from the definition

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    #   d/da Σₖ wₖ·convₖ(a,a) = 2·(rev(a) * w)   (central n entries)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                 # outer factor 2·n

    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 300) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS optimisation on the smooth‑max surrogate.
    Returns the best probability vector found together with its exact objective.
    """
    if not SCIPY_OPT:          # safety net – should never happen
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent on the exact non‑smooth objective
# ----------------------------------------------------------------------


def _adam_refine(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam (mirror‑descent) on the exact objective.
    Stops when time runs out or ``max_iters`` is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # build the weight vector w (uniform over active set)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        if active.size > 0:
            w[active] = 1.0 / float(active.size)

        # gradient of the (non‑smooth) max – average over active indices
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.2 / math.sqrt(it)          # decreasing learning rate

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)           # keep on the simplex

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it}  best={cur_val:.6f}")

    return best_seq, best_val

# Light Metropolis annealing (final lightweight polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between
    two random coordinates.  Used only if time is left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a rich pool of seeds.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool_extended(n, previous)

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising raw seeds.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])            # low is good
    top_k = 8                                   # keep a few good starters
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 10000.0, num=16)   # finer β schedule
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec,
                                    n,
                                    deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=400)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Adam‑style mirror‑descent refinement (exact non‑smooth objective).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _adam_refine(global_best_seq,
                                n,
                                deadline,
                                max_iters=500_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(global_best_seq,
                                    n,
                                    deadline,
                                    max_iters=400_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be hit.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will call evaluate_sequence on it)
    return global_best_seq.tolist()
