def _subgrad_refine_iter(seq, n, max_iters, lr0=0.9):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters+1):
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        w = np.zeros(2*n - 1)
        if active.size > 0:
            w[active] = 1.0/active.size
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _multiplicative_anneal(seq, n, deadline, init_temp=None):
    cur = seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    it = 0
    start = time.time()
    # set initial temperature as fraction of current objective
    T0 = cur_val * 0.2 if init_temp is None else init_temp
    while time.time() < deadline:
        it += 1
        # exponentially decay temperature
        T = T0 * (0.9999 ** it)
        # propose multiplicative perturbation
        sigma = max(0.01, T / cur_val)  # scale noise with temperature
        perturb = np.exp(np.random.normal(scale=sigma, size=n))
        new_seq = cur * perturb
        new_seq = new_seq / new_seq.sum()
        new_val = _objective(new_seq)
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
                print(f"[AnnealMult] it={it} best={new_val:.6f}")
        # occasionally do pair transfer like original
        if it % 200 == 0:
            i, j = np.random.randint(0, n, size=2)
            delta = np.random.rand() * 0.01 * cur[i]
            delta = min(delta, cur[i])
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_seq = new_seq / new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):

"""
propose_candidate_v3.py

A time‑limited optimiser for the “minimise 2·n·max(conv(a,a))/(sum a)²”
problem.  The optimiser works in four stages:

1️⃣  **Massive seed generation** – many diverse start vectors (uniform,
    geometric, power‑law, classic windows, random Dirichlet and a
    collection of sparse random supports).

2️⃣  **Coarse refinement** – each of the best ~20 seeds receives a short
    exponentiated‑gradient (mirror‑descent) run (≈5 k iterations) to
    leave the basin of the seed.

3️⃣  **Deep refinement** – the top few survivors are refined for a long
    run (≈100 k iterations) with a decreasing step‑size.  This replaces
    the “Adam → sub‑gradient” pipeline of the previous version and
    converges more reliably because it works directly on the true
    non‑smooth objective.

4️⃣  **Global simulated‑annealing** – a multiplicative‑noise annealer
    explores the neighbourhood of the best vector for the remaining
    time budget.  The annealer can make large jumps early on and
    fine‑grained moves later, giving a final push that often drops the
    objective a few × 10⁻³.

Optional cheap LP refinements (via cvxpy) are performed after the deep
refinement if a solver is available.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns the best sequence it found as a plain Python list.
"""

import time
import math
import numpy as np
import sys

# Optional LP support
# ------------------------------------------------------------
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False

# ------------------------------------------------------------
# Low‑level FFT convolution (exact, real‑valued)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)²).
    Input must be non‑negative; the function is scale‑invariant.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size ``m`` (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic “Sidon‑like” support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ------------------------------------------------------------
# Simple utilities
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Optimisation primitives
# ------------------------------------------------------------

def _expon_subgrad(seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 100_000,
                   lr0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / mirror‑descent on the exact non‑smooth
    objective.  Returns the best sequence encountered.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the bound is (almost) active
        active = np.where(conv >= max_val - 1e-12)[0]

        # dual weights – uniform over the active set
        w = np.zeros(2 * n - 1, dtype=np.float64)
        if active.size:
            w[active] = 1.0 / active.size

        # gradient of Σ w_k·(a*a)_k  (see derivation in the original code)
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # step‑size (decreasing like 1/√t)
        lr = lr0 / math.sqrt(it)

        # mirror‑step in the log‑space (KL‑mirror)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)      # prevent overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpSubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_multiplicative(start_seq: np.ndarray,
                           n: int,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with multiplicative log‑normal perturbations.
    Large perturbations are allowed early on; they shrink with the
    temperature schedule.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a fraction of the current value
    T0 = max(1e-6, cur_val * 0.15)
    decay = 0.99995
    it = 0

    while time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ---------- multiplicative perturbation ----------
        sigma = max(0.001, T / cur_val)        # noise scale follows temperature
        perturb = np.exp(np.random.normal(scale=sigma, size=n))
        cand = cur * perturb
        cand /= cand.sum()
        cand_val = _objective(cand)

        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[AnnealMul] it={it} best={cand_val:.6f}")

        # ---------- occasional tiny pair‑wise transfer ----------
        if it % 250 == 0:
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            delta = np.random.rand() * 0.02 * cur[i]
            delta = min(delta, cur[i])
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_val = _objective(cand)
            if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
                cur = cand
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = cand.copy()
                    print(f"[AnnealSwap] it={it} best={cand_val:.6f}")

    return best, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 8) -> np.ndarray:
    """
    Very cheap linear‑programming refinement (identical to the
    implementation used previously).  Only runs if cvxpy is available.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt /= tot

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a

# Seed pool construction
# ------------------------------------------------------------

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a diverse list of start vectors (all normalised to sum=1).
    """
    seeds: list[np.ndarray] = []

    # Classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_gaussian(n, 0.12))
    seeds.append(_gaussian(n, 0.18))
    seeds.append(_gaussian(n, 0.25))

    for r in np.linspace(0.9, 0.1, 9):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.5, 7):
        seeds.append(_power(n, p))
    for r in np.linspace(0.9, 0.1, 5):
        seeds.append(_sym_geom(n, r))

    # Random softmax (Dirichlet‑like)
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # Sparse random supports – a known good source of low max‑conv
    for m in range(30, 210, 30):          # 30,60,90,120,150,180
        for _ in range(3):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like constructions (max size ≈√n)
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))

    # Re‑use the previous best if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Clean‑up – clip negativities (should not occur) and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time
    budget and returns the best sequence discovered as a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin for bookkeeping

    # If a previous construction is stored, reuse it as a seed
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # --------------------------------------------------------
    # Problem dimension – 4000 is a sweet‑spot between quality
    # and speed for the present budget.
    # --------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # --------------------------------------------------------
    # 1) Build a very rich seed pool
    # --------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)
    print(f"[Seeds] generated {len(seeds)} candidates")

    # --------------------------------------------------------
    # 2) Quick assessment – keep the top few for the expensive phases
    # --------------------------------------------------------
    scored = [(s, _objective(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [s for s, _ in scored[:top_k]]
    global_best_seq = None
    global_best_val = float("inf")
    if scored[0][1] < global_best_val:
        global_best_val = scored[0][1]
        global_best_seq = scored[0][0].copy()
        print(f"[Seed] initial global best={global_best_val:.6f}")

    # --------------------------------------------------------
    # 3) Coarse exponentiated‑gradient refinement on the top seeds
    # --------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _expon_subgrad(seed_vec, n, deadline,
                                  max_iters=5_000, lr0=0.9)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Coarse] seed={idx} best={val:.6f}")

    # --------------------------------------------------------
    # 4) Deep refinement on the very best few survivors
    # --------------------------------------------------------
    if global_best_seq is not None:
        # pick the 5 best from the coarse stage for a long run
        # (re‑evaluate the whole top list with a slightly larger budget)
        refined_candidates = [(global_best_seq, global_best_val)]
        for i, seed_vec in enumerate(top_seeds[:5]):
            if time.time() > deadline:
                break
            seq, val = _expon_subgrad(seed_vec, n, deadline,
                                      max_iters=100_000, lr0=0.7)
            refined_candidates.append((seq, val))
        # select the best among these long runs
        refined_candidates.sort(key=lambda x: x[1])
        global_best_seq, global_best_val = refined_candidates[0]
        print(f"[Deep] new global best={global_best_val:.6f}")

    # --------------------------------------------------------
    # 5) Optional cheap LP refinement (adds a small, provably safe step)
    # --------------------------------------------------------
    if CVXPY_AVAILABLE and global_best_seq is not None:
        lp_deadline = time.time() + min(30.0, max(0.0, deadline - time.time()))
        seq_lp = _lp_refine(global_best_seq, n, lp_deadline,
                            t_mix=0.3, max_lp_iters=8)
        val = _objective(seq_lp)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq_lp.copy()
            print(f"[LP] improved best={val:.6f}")

    # --------------------------------------------------------
    # 6) Global simulated‑annealing for the remaining time budget
    # --------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_multiplicative(global_best_seq, n, deadline)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Anneal] final best={val:.6f}")

    # --------------------------------------------------------
    # 7) Fallback – should never be needed
    # --------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[Fallback] using uniform distribution")
        global_best_val = _objective(global_best_seq)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")

    # store the result for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
