def _optimize_seq(start_seq, deadline, max_iters=10000, beta_max=150.0):
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_obj = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv
        conv = _conv_fft(a, a)
        # compute smooth objective
        # schedule beta
        beta = min(beta_max, 0.1 + (beta_max - 0.1) * (it / max_iters))
        # compute softmax weights
        exp_arg = beta * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = np.sum(w)
        w /= w_sum
        # compute gradient via w
        a_rev = a[::-1]
        conv_rev = _conv_fft(w, a_rev)
        g = 2.0 * conv_rev[n-1:n-1+n]
        # step size
        lr = 0.3 / np.sqrt(it + 1)  # tuned
        log_a -= lr * g
        # re-normalize via softmax
        # note: a = softmax(log_a)
        max_log = np.max(log_a)
        a = np.exp(log_a - max_log)
        a /= a.sum()
        # occasionally recompute actual objective
        if (it % 50) == 0:
            obj = _objective(a)
            if obj < best_obj:
                best_obj = obj
                best_seq = a.copy()
                print(f"[Opt] it={it} n={n} obj={obj:.6f}")
                # optional early stop if below target
                if best_obj < 1.503:
                    break
        it += 1
    return best_seq, best_obj

    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1

    fy = np.fft.rfft(y, n=N)

    ``seq`` already satisfies sum(seq)=1 (the function is scale‑invariant).
    Returns 2·n·max(conv(seq,seq)).

    n = len(seq)

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:

def _smooth_obj_and_grad(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation of max(conv(a,a)) and its gradient
    w.r.t. the probability vector ``a``.

    The smooth max is  (1/β)·log(∑_k exp(β·conv_k)).
    Its gradient is Σ_k w_k·∂conv_k/∂a with w_k = softmax(β·conv_k).

    Returns (smooth_obj, grad) where grad has the same shape as ``a``.
    """

def _optimize_seq(start_seq: np.ndarray, deadline: float,
                  beta_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Refine ``start_seq`` (already normalised) by mirror descent.
    The temperature β is increased linearly from 1 to ``beta_max`` over
    roughly the first 5000 iterations; after that it stays at ``beta_max``.
    Returns the best sequence seen and its exact objective (max‑convolution).
    """
    eps = 1e-30
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + eps)

    best_seq = a.copy()
    best_obj = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        beta = 1.0 + (beta_max - 1.0) * min(1.0, it / 5000.0)

        _, grad = _smooth_obj_and_grad(a, beta)

        lr = 0.4 / np.sqrt(it)           # diminishing step size
        log_a -= lr * grad

        # Project back onto simplex (KL‑mirror step)
        max_log = np.max(log_a)
        a = np.exp(log_a - max_log)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        if it % 40 == 0:
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                print(f"[Descent] it={it:6d} n={n:5d} obj={cur_obj:.6f} beta={beta:.1f}")
            if best_obj < 1.503:        # target reached – stop early
                break
    return best_seq, best_obj

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time
    budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored solution, if available.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length schedule – larger n yields a tighter theoretical bound.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]

    global_best_seq = None
    global_best_val = float("inf")

    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline:
            break

        print(f"\n=== Length {n} ===")
        # Allocate a time slice for this length (max 180 s, keep a reserve)
        remaining = deadline - now
        max_alloc = min(180.0, max(30.0, remaining * 0.25))
        length_deadline = now + max_alloc

        # ------------------------------------------------------------------
        # 1️⃣ Deterministic seeds
        # ------------------------------------------------------------------
        seeds = [_uniform(n), _catalan_shape(n)]
        for p in (-0.3, -0.2, -0.1, 0.0, 0.1, 0.2, 0.5, 1.0):
            seeds.append(_beta_shape(p, n))
        if global_best_seq is not None:
            seeds.append(_upsample(global_best_seq, n))
        if previous is not None and previous.shape == (n,):
            seeds.append(previous)

        best_local_seq = None
        best_local_val = float("inf")
        for s in seeds:
            val = _objective(s)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = s.copy()
                print(f"[Seed] n={n:5d} val={val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣ Quick random Dirichlet sampling (≈10 % of the slice)
        # ------------------------------------------------------------------
        sample_deadline = time.time() + max_alloc * 0.10
        while time.time() < sample_deadline:
            seq = _random_dirichlet_one(n, rng)
            val = _objective(seq)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = seq.copy()
                print(f"[Random] n={n:5d} val={val:.6f}")
            if best_local_val < 1.503:
                break

        # ------------------------------------------------------------------
        # 3️⃣ Smoothed mirror‑descent refinement (core optimisation)
        # ------------------------------------------------------------------
        refined_seq, refined_val = _optimize_seq(best_local_seq, length_deadline)
        if refined_val < best_local_val:
            best_local_val = refined_val
            best_local_seq = refined_seq.copy()
            print(f"[Refine] n={n:5d} improved to {refined_val:.6f}")

        # ------------------------------------------------------------------
        # Record the best vector seen so far (any length)
        # ------------------------------------------------------------------
        if best_local_val < global_best_val:
            global_best_val = best_local_val
            global_best_seq = best_local_seq.copy()
            print(f"[Global] new best objective {global_best_val:.6f} (len={n})")

        # ------------------------------------------------------------------
        # 4️⃣ Optional short restarts from fresh random seeds (if time permits)
        # ------------------------------------------------------------------
        restart_deadline = time.time() + max_alloc * 0.05
        while time.time() < restart_deadline:
            seq0 = _random_dirichlet_one(n, rng)
            seq_opt, val_opt = _optimize_seq(seq0, time.time() + 5.0)  # 5 s refinement
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[Restart] n={n:5d} val={val_opt:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check – the grader will also call evaluate_sequence,
    # but we perform one more call to surface any unexpected error.
    # ----------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
