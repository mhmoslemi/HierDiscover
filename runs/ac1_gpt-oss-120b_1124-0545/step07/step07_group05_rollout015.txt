"""
Improved minimiser for 2·n·max(conv(a,a)) / (∑a)².

The algorithm extends the previous four‑phase pipeline with:

1. **Richer seed pool** – adds very low‐rate geometric decays,
   ultra‑sparse anti‑symmetric draws and a few extreme Dirichlet draws.

2. **Longer Adam stage** – more iterations, a slower increase of the
   soft‑max smoothness parameter and a higher learning‑rate ceiling.

3. **Accelerated sub‑gradient stage** – uses a larger iteration budget
   (up to 800 k) and a step‑size adapted to the iteration count.

4. **Enhanced simulated‑annealing** – three move types
   (random Dirichlet, targeted “support” moves and a *mirror‑swap*
   that directly re‑allocates mass between an index and its reversed
   counterpart).  Cooling is gentler (decay = 0.9999) and the
   iteration budget is scaled to the remaining wall‑clock time,
   allowing several millions of cheap updates.

All phases respect the global 1000 s time budget.  The best sequence
found (normalised to sum = 1) is returned as a plain ``list[float]``.
"""

import time
import numpy as np
import math
import concurrent.futures

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a simplex‑normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (0<r<1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, offset: float, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw on k random positions."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _anti_symmetric(n: int, rng: np.random.Generator) -> np.ndarray:
    """For each mirror pair (i, n-1-i) assign all mass to exactly one side."""
    a = np.zeros(n, dtype=np.float64)
    half = n // 2
    for i in range(half):
        # Choose which side receives mass (probability 0.5)
        if rng.random() < 0.5:
            a[i] = rng.exponential(1.0)
        else:
            a[n - 1 - i] = rng.exponential(1.0)
    if n % 2 == 1:
        # middle element can stay zero (its contribution cannot be avoided)
        a[half] = 0.0
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a large, diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # ----- elementary baselines -----
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # ----- geometric families (coarse) -----
    for r in np.linspace(0.02, 0.95, 10):
        seeds.append(_geom(n, r))

    # very high‑rate geometric (almost flat but still asymmetric)
    for r in np.linspace(0.96, 0.995, 5):
        seeds.append(_geom(n, r))

    # ----- power‑law families -----
    for p in np.linspace(0.0, 3.0, 9):
        seeds.append(_power(n, p))

    # ----- beta‑like shapes (asymmetric) -----
    for a in (0.2, 0.5, 1.0, 2.0):
        for b in (0.2, 0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # ----- exponential decays -----
    for sigma in (n / 20, n / 10, n / 5, n / 2, n):
        seeds.append(_exp_decay(n, sigma))

    # ----- logistic shapes -----
    for scale in (n / 20, n / 10, n / 5, n / 2):
        offset = scale * 2.0
        seeds.append(_logistic_shape(n, offset, scale))

    # ----- random Dirichlet draws (various concentrations) -----
    for conc in (0.05, 0.2, 0.5, 1.0, 3.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ----- sparse patterns -----
    for k in (3, 5, 8, 12):
        for conc in (0.05, 0.2):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # ----- anti‑symmetric draws (new) -----
    for _ in range(6):
        seeds.append(_anti_symmetric(n, rng))

    # ----- reuse previous solution if length matches -----
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)                     # softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = 2.0               # start with a gentle smoothing
    beta_mul = 1.015         # increase slowly

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:d} it={it:d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip to avoid exploding steps
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:d} it={it:d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a (true for this k_star)
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with three cheap move types:
        • random Dirichlet on 2‑4 indices,
        • targeted move focusing on the current maximal convolution support,
        • mirror‑swap move that reallocates mass between an index and its
          reversed counterpart.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initialise temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9999

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        move_type = rng.random()

        if move_type < 0.45:
            # ---- random Dirichlet re‑allocation on 2‑4 indices ----
            k = rng.integers(2, 5)                     # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)

        elif move_type < 0.80:
            # ---- targeted move (touch indices that appear in k_star) ----
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)

            # a second index (different) uniformly chosen
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)

        else:
            # ---- mirror‑swap move (pair with its reversal) ----
            i = rng.integers(0, n)
            j = n - 1 - i
            if i == j:
                # middle element – fall back to a random move
                k = rng.integers(2, 4)
                idx = rng.choice(n, size=k, replace=False)
            else:
                idx = np.array([i, j], dtype=int)

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass on the selected positions
        new_alloc = rng.dirichlet(np.full(idx.size, 1.0)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            # a*Δ + Δ*a  (commutative)
            conv_new[p : p + n] += 2.0 * delta * cur
            # Δ*Δ term (self‑product)
            conv_new[2 * p] += delta * delta

        # cross‑terms Δ_i·Δ_j (appears twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            # accept
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n:d} it={it:d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence as a ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final checks

    # retrieve previous best if the notebook provides it
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths – we try a few increasing sizes.
    lengths = [2000, 4000, 6000, 8000, 10000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – evaluate raw seeds (parallelised)
        # ------------------------------------------------------------------
        with concurrent.futures.ThreadPoolExecutor(max_workers=2) as ex:
            future_to_seed = {ex.submit(_objective, s): s for s in seeds}
            for future in concurrent.futures.as_completed(future_to_seed):
                s = future_to_seed[future]
                try:
                    val = future.result()
                except Exception:
                    continue
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = s.copy()
                    print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on the most promising seeds
        # ------------------------------------------------------------------
        # pick top‑k seeds (by seed objective)
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(6, len(seed_vals))          # a bit larger than before
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            seq, val = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=800000
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated‑annealing (dominant phase)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and global_best_seq.shape[0] == n
        ):
            remaining = deadline - time.time()
            # estimate iterations: about 0.00006 s per move for n≈8000
            # use a generous factor to exploit the remaining budget
            target_iters = int(min(5_000_000, max(1_000_000, remaining * 16000)))
            seq, val = _sa_incremental_optimize(
                global_best_seq,
                n,
                deadline,
                init_temp_factor=0.04,
                max_iters=target_iters,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with the supplied evaluate_sequence (the call is cheap)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
