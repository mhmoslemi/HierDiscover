"""
Propose a step function with a lower max‑self‑convolution.

Algorithm
---------
* Build a large pool of candidate vectors (uniform, windows, geometric,
  power‑law, random sparse supports, Sidon‑like constructions,…).  All
  vectors live on the probability simplex (non‑negative, sum = 1).

* Keep the best few candidates (by the exact objective).

* For each kept candidate run a relatively long Adam optimisation on a
  smooth‑max surrogate of the maximum convolution (the “soft‑max”
  temperature is increased slowly).

* Refine the best Adam result with a projected‑gradient descent on the
  true non‑smooth objective.  The gradient is the average of the
  sub‑gradients of the active constraints (those entries of the
  convolution that are within a tiny tolerance of the maximum).  An
  Euclidean projection onto the simplex enforces feasibility.

* Perform several cheap linear‑programming refinements: given the
  current vector `f`, maximise `∑ g_j` subject to the linear constraints
  `f * g ≤ M` (where `M` is the current maximum) and `g ≥ 0`.  The
  solution `g` is mixed back into `f` with a convex combination.
  This step is repeated several times.

* Finally a short simulated‑annealing phase makes tiny random mass
  transfers to escape possible local minima.

All phases respect the overall time budget (default 1000 s).  The best
vector found is stored in the global variable `height_sequence_1` and
returned as a plain list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT.  Returns length len(a)+len(b)-1."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the target function:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    The sequence must be non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size m."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like set of positions (triangular numbers)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions are i² (mod n)."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective value.
    """
    v = np.log(seed + 1e-30)                    # logits (simplex parametrisation)

    # Adam moments
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # temperature for soft‑max of the convolution entries
    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k · (a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _projgrad_descent(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 800000,
                      rel_tol: float = 1e-6,
                      lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected (Euclidean) gradient descent on the exact non‑smooth objective.
    Uses the true max‑convolution and an active‑set sub‑gradient.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        M = conv.max()
        # include points that are very close to the maximum
        eps = max(1e-12, rel_tol * M)
        active = np.where(conv >= M - eps)[0]

        # build uniform weights over the active set
        w = np.zeros(2 * n - 1, dtype=np.float64)
        if active.size:
            w[active] = 1.0 / active.size

        # gradient of the (smooth) surrogate with respect to a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # step size schedule (diminishing)
        lr = lr0 / np.sqrt(it)

        # gradient step + simplex projection
        a = a - lr * grad
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[ProjGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               t_mix: float = 0.3,
               max_lp_iters: int = 12) -> np.ndarray:
    """
    Perform a few cheap linear‑programming refinements.
    For each iteration:
        – compute current max M of conv(f,f)
        – collect active indices (within 1e‑9 of M)
        – maximise total mass of g subject to f * g ≤ M, g ≥ 0
        – mix g back into f with weight t_mix and re‑project onto simplex.
    Returns the refined vector.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        # Build LP
        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            coeff = np.zeros(n, dtype=np.float64)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).ravel()
        tot = g_opt.sum()
        if tot <= 0:
            break
        g_opt = g_opt / tot

        # convex combination and projection
        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing walk that moves a small amount of mass
    between two positions and occasionally injects a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={new_val:.6f}")

        if it % 1000 == 0:
            # global Gaussian jump
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val

def _make_seeds(n: int,
                previous: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse list of probability vectors."""
    seeds: list[np.ndarray] = []

    # Classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.05, 0.45, 5):          # additional edge‑heavy beta windows
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # Sparse random seeds (support sizes spanning a wider range)
    for m in range(20, 301, 30):          # 20, 50, …, 290
        for _ in range(3):
            seeds.append(_sparse_random_seed(n, m, rng))

    # Deterministic Sidon‑like constructions
    for m in (50, 80, 110, 140):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # Re‑use the previous best if it has the right length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Normalise & filter out any degenerate vectors
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned

    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    #  Length selection – 4000 is a good compromise for quality vs. speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    #  1) Build a very rich seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng)

    # ------------------------------------------------------------------
    #  2) Quick evaluation of all seeds – keep the top few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")

    # Record the best raw seed
    best_raw_val = seed_vals[0][1]
    global_best_seq = seed_vals[0][0].copy()
    global_best_val = best_raw_val
    print(f"[seed] best raw seed value = {best_raw_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Projected gradient descent (exact max)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _projgrad_descent(global_best_seq, n, deadline,
                                     max_iters=800000, rel_tol=1e-6, lr0=0.5)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall PGD] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Linear‑programming refinement (multiple iterations)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and global_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 10.0:
            lp_deadline = time.time() + min(30.0, remaining / 2)
            seq_lp = _lp_refine(global_best_seq, n, lp_deadline,
                                t_mix=0.3, max_lp_iters=12)
            # One short projected‑gradient pass after the LP mix
            seq, val = _projgrad_descent(seq_lp, n, deadline,
                                         max_iters=150000, rel_tol=1e-6, lr0=0.5)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[post‑LP PGD] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final annealing fine‑tuning (uses any remaining time)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _anneal_optimize(global_best_seq, n, deadline,
                                    max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.copy()
    return global_best_seq.tolist()
