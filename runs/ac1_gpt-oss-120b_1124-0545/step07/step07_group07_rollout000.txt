    old_n = len(seq)
    if old_n == target_n:

    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n-1, target_n)

    # ensure non-neg and normalized.
    new_seq = np.maximum(new_seq, 0)
    total = new_seq.sum()
    if total > 0:

def _simulated_annealing_optimize(a0, n, deadline, max_iters=500_000):
    rng = np.random.default_rng()
    cur = np.array(a0, dtype=np.float64)
    cur /= cur.sum()
    cur_val = _objective(cur, n)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    step_scale = 0.05 # initial max transfer fraction relative to donor
    T0 = cur_val * 0.1
    T_end = cur_val * 0.001
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # update temperature
        elapsed = time.time() - start
        total = deadline - start
        t = elapsed / total
        T = T0 * ((T_end/T0) ** t)  # exponential cooling
        # pick donor according to contributions
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)
        # propose amount
        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        new_val = _objective(new, n)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
        # adapt step_scale
        if it % 5000 == 0:
            step_scale = max(step_scale * 0.95, 1e-6)
    return best, best_val

"""
Optimise a non‑negative step‑function (vector) to minimise

    2·n·max(conv(a, a)) / (sum(a)²)

using a multi‑phase pipeline:

1. Build a diverse set of analytical & random seeds (including warm‑start
   up‑sampling).
2. Quick evaluation of all seeds → keep global best.
3. Refine the best seed with three deterministic gradient‑type phases:
   * Adam on a smooth‑max surrogate (softmax parametrisation).
   * Exact sub‑gradient descent on the true max (mirror‑descent).
   * L2‑flattening toward a constant auto‑convolution.
4. Exhaust remaining time with Simulated‑Annealing hill‑climbing that moves
   small amounts of mass away from the contributors to the current maximal
   convolution entry, accepting occasional uphill moves.

All phases respect a global time‑budget (≈ 1000 s).  The final vector is
normalised (ℓ₁‑sum = 1), stored as ``height_sequence_1`` for warm‑starting
future runs and returned as a plain list of Python floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution computed via real FFT.
    Length of result = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Return the value of 2·n·max(conv(a, a)) assuming sum(a) == 1.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (ℓ₁ = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _upscale(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    old_n = len(seq)
    if old_n == target_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, target_n, dtype=np.float64)
    up = np.interp(x_new, x_old, seq)
    return _normalize(up)


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator,
                warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a list of ℓ₁‑normalised candidate vectors for length ``n``.
    """
    seeds: list[np.ndarray] = []

    # (1) Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (2) Half‑sine window
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(_normalize(s))

    # (3) Geometric families
    for r in np.linspace(0.2, 0.95, 10):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(_normalize(a))

    # (4) Power‑law families
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(_normalize(a))

    # (5) Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(_normalize(a))

    # (6) Dirichlet draws (various concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(4):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(_normalize(a))

    # (7) Softmax of Gaussian logits
    for _ in range(4):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(_normalize(a))

    # (8) Random piecewise‑linear shapes
    for _ in range(4):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(_normalize(a))

    # (9) Log‑normal (Gaussian bell) shapes
    for _ in range(4):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - mu) / sigma) ** 2)
        seeds.append(_normalize(a))

    # (10) Warm‑start up‑sampling, if available and compatible
    if warm is not None:
        if warm.ndim == 1 and warm.shape[0] > 0:
            ups = _upscale(_normalize(warm), n)
            seeds.append(ups)

    return seeds

def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate.
    Parameters are represented as logits v (softmax(v) = a).
    The temperature β controlling smooth‑max is increased multiplicatively.
    """
    v = np.log(a0 + 1e-30)                 # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0            # initial smoothness
    beta_mul = 1.015      # slower increase → more stable gradients

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # a = softmax(v)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        # current objective (exact max)
        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (using convolution property)
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)                 # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]        # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max, using exponentiated‑gradient
    (mirror descent on the simplex).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry (see derivation in the comment)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Drives a*a towards a constant vector (L2‑flattening) using
    exponentiated‑gradient descent on ½‖a*a‑μ·1‖², where μ = 1/(2n‑1).
    """
    a = _normalize(a0)
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing_optimize(a0: np.ndarray, n: int,
                                 deadline: float,
                                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing.
    At each step a tiny amount of mass is shifted from a “donor” (chosen
    preferably among indices that actively contribute to the current maximal
    convolution entry) to a random “receiver”.  Moves that increase the
    objective are accepted with probability exp(-Δ/T), where the temperature
    T decays exponentially.
    """
    rng = np.random.default_rng()
    cur = _normalize(a0)
    cur_val = _objective(cur, n)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    # Temperature schedule: start at 10 % of current value, end at 0.1 %
    T_start = cur_val * 0.1
    T_end = cur_val * 0.001
    step_scale = 0.04  # initial transfer fraction of donor mass

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exponential cooling
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        t_frac = elapsed / total
        T = T_start * ((T_end / T_start) ** t_frac)

        # identify contributors to the active maximal convolution entry
        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        new_val = _objective(new, n)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it:6d} best={new_val:.6f}")

        # slowly reduce step_scale to fine‑tune
        if it % 4000 == 0:
            step_scale = max(step_scale * 0.92, 1e-7)

    return best, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found as a list of
    Python floats.
    """
    # ------------------------------------------------------------------
    # Timing utilities
    # ------------------------------------------------------------------
    start_time = time.time()
    # 2 s safety margin for final verification & housekeeping
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Randomness & warm‑start handling
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)

    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    # A larger n gives a better theoretical bound, but also higher cost.
    # Empirically n≈6000 works well within the 1000 s budget.
    n = 6000

    # ------------------------------------------------------------------
    # Build diverse seed pool (including up‑scaled warm‑start)
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"[Info] Generated {len(seeds)} seeds for n={n}")

    # ------------------------------------------------------------------
    # Quick seed evaluation – keep the current global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)          # sum(a) is already 1
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase allocation (fractions of the remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # Phase 1 – Adam (≈ 35 % of remaining time)
    phase_deadline = time.time() + remaining * 0.35
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline,
                              max_iters=300_000)
    if val < global_best_val:
        global_best_val, global_best_seq = val, seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # Phase 2 – Exact sub‑gradient descent (≈ 25 %)
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.25
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline,
                                     max_iters=250_000)
    if val < global_best_val:
        global_best_val, global_best_seq = val, seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # Phase 3 – L2‑flattening (≈ 15 %)
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.15
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline,
                                    max_iters=200_000)
    if val < global_best_val:
        global_best_val, global_best_seq = val, seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # Phase 4 – Simulated‑annealing hill‑climb (use whatever time left)
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 1.5  # tiny buffer
        seq, val = _simulated_annealing_optimize(global_best_seq, n,
                                                 phase_deadline,
                                                 max_iters=500_000)
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Phase‑SA] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        # Convert to plain Python list – evaluate_sequence clips automatically.
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as e:
        print("[Verification] sequence rejected (error:", e, "); falling back to uniform.")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Reporting & storing for future warm‑starts
    # ------------------------------------------------------------------
    final_val = _objective(global_best_seq, n)
    print(f"\n=== Finished ===\nBest objective ≈ {final_val:.6f} (length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
