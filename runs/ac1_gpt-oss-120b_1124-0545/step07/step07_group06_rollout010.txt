"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works in three phases:

1. **Rich seed generation** – a large collection of analytically motivated
   families (uniform, sine‑window, geometric, power‑law, beta‑shape,
   Gaussian, linear ramps, random Dirichlet & sparse patterns) together with
   interpolated versions of the seeds from shorter lengths.

2. **Continuous optimisation** – the best few seeds are refined with a
   smooth‑max Adam optimiser (working on the simplex via soft‑max), followed
   by an exact sub‑gradient mirror‑descent that directly targets the true
   max‑convolution entry.

3. **Targeted simulated‑annealing** – a high‑frequency SA that always attacks
   the current bottleneck index by reallocating mass among the heaviest
   contributors. The schedule is time‑aware and stops well before the global
   deadline.

All phases respect a user‑supplied time budget (default 1000 s).  The final
sequence is returned as a plain list of floats and has been sanity‑checked
with the provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the centre of the interval."""
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def _linear_ramp(n: int, bias: float) -> np.ndarray:
    """Monotone linear ramp from bias to 1 (inclusive)."""
    i = np.arange(n, dtype=np.float64)
    a = bias + (1.0 - bias) * i / (n - 1)
    a /= a.sum()
    return a


def _trapezoid(n: int, left_frac: float, right_frac: float) -> np.ndarray:
    """Flat top trapezoid on [left_frac, right_frac] of the normalized index."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.where(
        (i >= left_frac) & (i <= right_frac),
        1.0,
        np.where(i < left_frac, i / left_frac, (1.0 - i) / (1.0 - right_frac)),
    )
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _interp_resample(a: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_len, preserving non‑negativity."""
    old_x = np.linspace(0.0, 1.0, a.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    b = np.interp(new_x, old_x, a)
    b = np.clip(b, 0.0, None)
    b_sum = b.sum()
    if b_sum == 0.0:
        b = _uniform(new_len)
    else:
        b /= b_sum
    return b


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
    small_seeds: list[np.ndarray] | None = None,
) -> list[np.ndarray]:
    """
    Produce a large, diverse collection of normalised seeds for a given length.
    If ``small_seeds`` is supplied each entry is interpolated to length ``n``.
    """
    seeds: list[np.ndarray] = []

    # deterministic baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_linear_ramp(n, bias=0.0))
    seeds.append(_linear_ramp(n, bias=0.3))

    # geometric decays
    for r in np.linspace(0.15, 0.95, 12):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 4.0, 12):
        seeds.append(_power(n, p))

    # beta‑shape grid (broader than previously used)
    alphas = np.linspace(0.10, 2.50, 12)
    betas = np.linspace(0.10, 2.50, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian family
    for sigma_frac in np.linspace(0.07, 0.35, 12):
        seeds.append(_gaussian(n, sigma_frac))

    # Trapezoid family
    for left_frac in np.linspace(0.0, 0.4, 5):
        for right_frac in np.linspace(0.6, 1.0, 5):
            if left_frac < right_frac:
                seeds.append(_trapezoid(n, left_frac, right_frac))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse patterns
    for k in (2, 3, 5, 8, 12):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # Interpolated seeds from smaller lengths (if any)
    if small_seeds is not None:
        for s in small_seeds:
            seeds.append(_interp_resample(s, n))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous.copy()
        prev_sum = prev.sum()
        if prev_sum > 0:
            seeds.append(prev / prev_sum)

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best sequence seen and its objective value.
    """
    eps = 1e-30
    v = np.log(seed + eps)  # log‑space works on the simplex automatically

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.7

    beta_smooth = 2.0        # start smoothing parameter
    beta_mul = 1.03          # increase each iteration (smooth max sharpens)

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                # stays on simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # -------- smooth‑max weighting ----------
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # -------- gradient of Σ w·conv ----------
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]  # central n entries

        # -------- gradient w.r.t. logits v (softmax Jacobian) ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active bottleneck entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirrored on simplex)
        lr = 0.6 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-9,
    max_iters: int = 1500000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always attacks the current maximal convolution entry.
    The move reallocates total mass of a few heavy contributors.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9997

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # current bottleneck
        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a small set of the heaviest contributors
        top_k = min(7, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to move (3‑5)
        k_move = rng.integers(3, 6)

        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass uniformly (Dirichlet with α=1)
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        # incremental convolution update
        conv_new = cur_conv.copy()
        # a * Δ + Δ * a contributions
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross Δ_i * Δ_j for i≠j (appears twice)
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                if it % 50000 == 0:
                    print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # Retrieve any previously found construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a target length – longer lengths tend to give lower bounds.
    # We use 8000 if time permits, otherwise fall back to 4000.
    # ------------------------------------------------------------------
    if (deadline - time.time()) > 300.0:
        target_length = 8000
    else:
        target_length = 4000

    # ------------------------------------------------------------------
    # Generate a pool of seeds from several auxiliary lengths and
    # interpolate them to the target length.
    # ------------------------------------------------------------------
    auxiliary_lengths = [2000, 4000, 6000]
    auxiliary_seeds = []

    for n_aux in auxiliary_lengths:
        if time.time() > deadline:
            break
        aux = _make_seeds(
            n=n_aux,
            previous=previous,
            rng=rng,
            small_seeds=None,
        )
        # keep a few best from each auxiliary length for interpolation
        aux_vals = [_objective(s) for s in aux]
        top_idx = np.argsort(aux_vals)[:5]  # 5 best per auxiliary length
        for idx in top_idx:
            auxiliary_seeds.append(aux[idx])

    # ------------------------------------------------------------------
    # Build the final seed set for the target length.
    # ------------------------------------------------------------------
    seeds = _make_seeds(
        n=target_length,
        previous=previous,
        rng=rng,
        small_seeds=auxiliary_seeds,
    )

    # Quick evaluation of raw seeds
    seed_vals = [_objective(s) for s in seeds]
    for s, v in zip(seeds, seed_vals):
        if v < float("inf"):
            pass  # placeholder – could log, but we only need best later.

    # Determine top‑k seeds to feed into heavy optimisation
    top_k = 8
    top_idx = np.argsort(seed_vals)[:top_k]
    top_seeds = [seeds[i] for i in top_idx]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    for s in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(s, target_length, deadline, max_iters=20000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Adam] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient refinement on the current best
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _subgradient_refine(
            global_best_seq, target_length, deadline, max_iters=400000
        )
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑SubGrad] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted simulated annealing for fine‑tuning
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq, val = _targeted_sa_optimize(
            global_best_seq, target_length, deadline, max_iters=1200000
        )
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑SA] new global best={val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
