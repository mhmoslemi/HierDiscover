"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The search works in four time‑aware phases:

1. **Deterministic seeds** – a collection of analytic families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian) is generated
   for a set of lengths.  The best of these seeds is kept.

2. **Heavy‑weight random sampling** – Dirichlet vectors with several
   concentration parameters (α ∈ {0.5, 1.0, 2.0}) are drawn for the same
   lengths while time permits, keeping the overall best sample.

3. **Smoothed Adam optimisation** – starting from the current best
   vector we minimise a *smooth* version of the max‑convolution
   `max(conv(a,a))`.  The smooth max is approximated by a
   log‑sum‑exp with an annealed temperature β.  The optimisation is
   performed on the soft‑max (log‑its) parametrisation of the
   probability vector using Adam; it yields a finer balance of the
   convolution peaks than the plain sub‑gradient method.

4. **Exact sub‑gradient mirror descent** – a final few thousand
   iterations of the classic exponentiated‑gradient (KL‑mirror descent)
   are performed to polish the solution.

All phases respect the total time budget (default 1000 s).  The routine
always returns a non‑negative list; ``evaluate_sequence`` will clip and
normalise it automatically.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import log, sqrt, pi, exp

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast for large vectors).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a sequence that is already
    normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ---------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    """Raised‑cosine window (Hann)."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """
    Draw a Dirichlet(alpha,…,alpha) vector.
    For alpha=1 we use exponentials; for other alphas we use Gamma.
    """
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Smoothed Adam optimiser (log‑its parametrisation)
# ---------------------------------------------------------------

    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 50.0,
    max_iter: int = 10_000_000,

    * a is represented by logits `logits` --> softmax(a).
    * The smooth max uses a temperature β that is slowly increased.
    * Adam with decaying step size is employed.

    eps = 1e-30

    logits = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.02          # base learning rate (will be scaled down later)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # compute probability vector a
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # full self‑convolution
        conv = _conv_fft(a, a)            # length = 2n‑1

        # anneal β linearly with iteration count
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.5))

        # smooth max via log‑sum‑exp (stabilised)
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()     # soft‑weights, sum=1

        # gradient w.r.t. a :  g_a = 2 * Σ_k p_k a_{k‑i}
        # compute via convolution of p with reversed a
        conv_pa = _conv_fft(p, a[::-1])   # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]   # slice of length n

        # chain rule through soft‑max: ∂F/∂logits = a * (g_a - <g_a,a>)
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays as 1/√it
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:    # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} beta={beta:.1f}")

            # early exit if we already beat the target comfortably
            if best_val < 1.50:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Exact sub‑gradient mirror descent (exponentiated gradient)
# ---------------------------------------------------------------

def _refine_subgradient(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient (KL‑mirror descent) on the simplex,
    using the true sub‑gradient of the active convolution entry.
    Returns the best vector seen and its objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Refine] it={it:7d} n={n:5d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val

# Main entry point
# ---------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small margin for sanity checks

    # ------------------------------------------------------------------
    # Possibility of re‑using a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to investigate – we keep them moderate (LP‑friendly)
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:10s} val={val:.6f}")

        # also try the globally stored previous vector if it matches the length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling (several alphas)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.4   # 40 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.5, 1.0, 2.0]    # give both sparse and smooth samples

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        alpha = rng.choice(alphas)
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.1f} val={val:.6f}")

        # early breakout if we already reach the target comfortably
        if best_val < 1.50:
            break

    # ------------------------------------------------------------------
    # Stage 3 – Smoothed Adam optimisation on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly 30 % of the remaining budget to this phase
        opt_deadline = time.time() + (deadline - time.time()) * 0.3
        seq_adam, val_adam = _smooth_adam_optimize(best_seq, opt_deadline)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"[Adam‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact sub‑gradient mirror descent (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Final‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
