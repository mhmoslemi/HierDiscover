"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (∑a)²

using a three‑phase pipeline:

1. **Diverse seeds** – uniform, geometric, power‑law, beta‑shapes,
   sine‑window, Dirichlet samples and a few manually‑crafted two‑spike
   patterns.  All seeds are normalised to sum = 1.

2. **Smooth‑max Adam optimisation** – optimise a log‑softmax parametrisation
   of the sequence with a gradually sharpening log‑sum‑exp approximation
   of the max.  The learning‑rate schedule and the smoothness parameter are
   fully time‑aware.

3. **Exact sub‑gradient mirror descent** – after Adam the true max‑convolution
   is used.  The sub‑gradient is the vector of contributions to the current
   bottleneck and an exponentiated‑gradient step keeps the iterates on the
   simplex.

4. **Targeted simulated‑annealing** – a high‑frequency SA that always
   attacks the current maximal convolution entry, redistributing a small
   random set of its heaviest contributors.  Incremental convolution updates
   keep each move O(k·n) with k≤4.

The routine adapts the number of iterations to the remaining budget, keeps
the best sequence seen so far and finally returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _two_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two equal spikes placed far apart (roughly opposite ends)."""
    i = rng.integers(0, n // 2)
    j = n - 1 - i
    a = np.zeros(n, dtype=np.float64)
    a[i] = 0.5
    a[j] = 0.5
    return a


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,

    """Create a diverse collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 10)
    betas = np.linspace(0.30, 1.20, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (2, 3, 5, 8):
        for conc in (0.2, 0.5, 0.8):
            for _ in range(2):
                seeds.append(_random_sparse(n, k, conc, rng))

    # a couple of two‑spike patterns (hand‑crafted)
    for _ in range(3):
        seeds.append(_two_spikes(n, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) approximation.
    Returns the best sequence seen and its objective value.
    """
    eps = 1e-30
    v = np.log(seed + eps)           # work in log‑space ⇒ positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.9

    # smoothing (log‑sum‑exp) parameter – starts low, grows exponentially
    beta_smooth = 2.0
    beta_mul = 1.015

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # stabilise exponentials
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a  (convolution with reversed a)
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)      # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping (helps stability)
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated gradient) on the true max.
    Returns the best sequence and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:8d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.6 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always attacks the current maximal convolution
    entry.  A small random set (2‑4) of its heaviest contributors is
    re‑distributed using a Dirichlet draw.  Incremental convolution updates
    keep each move O(k·n) with k≤4.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # initial temperature proportional to current objective
    T = max(0.02 * cur_obj, 1e-12)
    decay = 0.9994

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, 1e-12)

        # locate the bottleneck index
        k_star = int(np.argmax(cur_conv))

        # indices that truly contribute to the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few of the heaviest contributors (max 5)
        top_k = min(5, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to modify (2‑4)
        k_move = rng.integers(2, 5)

        # choose indices: use heavy ones first, fill with random slots if needed
        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass among the chosen positions
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a contributions (ordered pairs)
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            # contribution of a with Δ at position p
            conv_new[p : p + n] += 2.0 * delta * cur
            # self term Δ*Δ (appears once at position 2p)
            conv_new[2 * p] += delta * delta

        # cross Δ_i * Δ_j for i≠j (appears twice)
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:7d} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``. The routine respects the time budget and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the caller that evaluates the result
    deadline = start_time + budget_s - 2.0

    # previous solution (if any) – used as an extra seed
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (increase gradually – larger n costs more time)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000]

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – generate seeds and quick evaluation
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # quick evaluation of raw seeds (no FFT in the inner loop, just one conv)
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Adam optimisation on the few best seeds
        # ------------------------------------------------------------------
        # pick the top‑3 seeds (or fewer if time runs out)
        top_k = 3
        top_idx = np.argsort(seed_vals)[:top_k]
        top_seeds = [seeds[i] for i in top_idx]

        # allocate a fraction of time to Adam (roughly 30 % of remaining budget)
        remaining = deadline - time.time()
        per_seed_budget = remaining * 0.15 / max(1, len(top_seeds))

        for s in top_seeds:
            if time.time() > deadline:
                break

            seed_deadline = time.time() + per_seed_budget
            seq, val = _adam_optimize(s, n, seed_deadline, max_iters=200_000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – exact sub‑gradient refinement (only on the current best)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and len(global_best_seq) == n
            and time.time() < deadline
        ):
            # allocate up to half the remaining time for sub‑gradient work
            remaining = deadline - time.time()
            sub_deadline = time.time() + remaining * 0.45
            iters = 800_000
            seq, val = _subgradient_refine(global_best_seq, n, sub_deadline, max_iters=iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – targeted simulated annealing (fine‑tuning)
        # ------------------------------------------------------------------
        if (
            global_best_seq is not None
            and len(global_best_seq) == n
            and time.time() < deadline
        ):
            remaining = deadline - time.time()
            sa_deadline = time.time() + remaining * 0.35
            seq, val = _targeted_sa_optimize(global_best_seq, n, sa_deadline, max_iters=300_000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Sanity‑check with the official evaluator (optional)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
