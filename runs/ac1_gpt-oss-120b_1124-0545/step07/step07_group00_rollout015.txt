"""
Optimising a non‑negative step‑function (a probability vector) for the
objective

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (∑a)²

The value is scale‑invariant, therefore we normalise every candidate to
∑a = 1 and minimise the surrogate 2·n·max(conv).  The script proceeds in
four stages:

1. **Rich seeding** – dozens of analytic families (uniform, sine‑power,
   raised‑cosine, parabolic, geometric, symmetric geometric, beta, Dirichlet
   and sparse random) are generated for the default length n = 4000.
   Parameters are swept on a fine grid (e.g. γ∈[0.1,2] for the sine‑power
   family).  The best K seeds by the fast surrogate are retained.

2. **Long exponentiated‑gradient descent** – for each retained seed we run a
   mirror‑descent on the true max‑convolution (log‑softmax parametrisation)
   for as many iterations as the remaining budget permits.  A relatively
   aggressive step‑size schedule (lr≈1/√t) together with gradient clipping
   proved more effective than the previous Adam phase.

3. **Perturb‑and‑refine loops** – the current global best is repeatedly
   perturbed on a random subset of coordinates (Dirichlet reshuffle) and
   re‑optimised with the same exponentiated‑gradient routine.  This helps
   escape shallow local minima.

4. **Final sanity check** – the best vector is evaluated with the supplied
   ``evaluate_sequence`` routine and stored in the module globals for possible
   reuse.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – assumes ∑seq == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_power(n: int, gamma: float) -> np.ndarray:
    """sin‑window raised to a power γ (γ>0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, alpha: float) -> np.ndarray:
    """Tukey‑type window with taper α∈[0,1]; α=0 → rectangular,
       α=1 → Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return a / a.sum()


def _parabolic(n: int) -> np.ndarray:
    """Simple parabola a_i = 4·i·(n‑1‑i) (zero at both ends)."""
    i = np.arange(n, dtype=np.float64)
    a = 4.0 * i * (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed collection (diverse families)
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Generate a large, diverse pool of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_parabolic(n))

    # sine‑power family – fine grid
    for gamma in np.linspace(0.1, 2.0, 19):
        seeds.append(_sine_power(n, gamma))

    # raised‑cosine / Tukey family
    for alpha in np.linspace(0.05, 0.95, 19):
        seeds.append(_raised_cosine(n, alpha))

    # geometric families
    for r in np.linspace(0.2, 0.96, 13):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # beta family – extended grid (including values < 1)
    alphas = np.concatenate([np.linspace(0.1, 1.5, 16), np.linspace(1.6, 3.0, 9)])
    betas = np.concatenate([np.linspace(0.1, 2.0, 16), np.linspace(2.1, 5.0, 9)])
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (diverse concentration)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 9, 15, 30):
        for conc in (0.1, 0.3):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser (exact max)
# ----------------------------------------------------------------------


def _expgrad_subgradient(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 600000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    The iterate lives on the probability simplex via a log‑parameterisation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    t = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # compute full convolution and locate the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the max at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]  # because derivative of a_i a_{max_idx-i}

        # step‑size schedule (more aggressive than the former 0.7/√t)
        lr = 1.2 / np.sqrt(t)  # a bit larger learning rate

        # exponentiated‑gradient update (mirror‑descent w.r.t. KL)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # stabilise numerics
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = _uniform(n)
        else:
            a /= s

        # occasional progress report
        if it % 100000 == 0:
            print(f"[ExpGrad] n={n:5d} it={it:7d} cur={cur_val:.6f} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small stochastic perturbation + re‑optimisation
# ----------------------------------------------------------------------


def _perturb_and_refine(
    base_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_trials: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Take the current best sequence, randomly reshuffle a small subset of its
    mass (Dirichlet on 3‑7 coordinates) and run the exp‑gradient optimiser
    again.  Keep the best outcome over a few trials.
    """
    best_seq = base_seq.copy()
    best_val = _objective(best_seq)

    for trial in range(max_trials):
        if time.time() > deadline:
            break

        k = rng.integers(3, 8)                     # size of the reshuffle set
        idx = rng.choice(n, size=k, replace=False)
        total = base_seq[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        cand = base_seq.copy()
        cand[idx] = new_alloc

        # run a short exp‑gradient from this perturbed start
        sub_deadline = min(deadline, time.time() + 5.0)   # at most a few seconds
        seq_opt, val_opt = _expgrad_subgradient(
            cand, n, sub_deadline, max_iters=120000
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Perturb] trial={trial+1}/{max_trials} new best={val_opt:.6f}")

    return best_seq, best_val

    top_k = 7                                     # keep a handful of promising starts
    elite = [vec for vec, val in evaluated[:top_k]]
    print(f"Top‑{top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # initialise global best
    global_best_seq = elite[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best surrogate objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy exponentiated‑gradient optimisation for each elite seed.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(elite):
        if time.time() > deadline:
            break
        print(f"\n--- Exp‑Grad run {idx+1}/{len(elite)} ---")
        seq_opt, val_opt = _expgrad_subgradient(
            seed_vec, n, deadline, max_iters=500000
        )
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Overall‑ExpGrad] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – perturb‑and‑refine cycles to escape possible local minima.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # each cycle consumes only a modest amount of the remaining budget
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        seq_ref, val_ref = _perturb_and_refine(
            global_best_seq, n, deadline, rng, max_trials=8
        )
        if val_ref < global_best_val:
            global_best_val = val_ref
            global_best_seq = seq_ref.copy()
            print(f"[Overall‑Perturb] new global best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n[Final] evaluate_sequence returned {final_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f}"
        f" (length {len(global_best_seq)})"
    )

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
