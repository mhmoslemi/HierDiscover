def adam_optimize(start_seq, n, deadline, beta=80.0, max_iters=200000):
    a = start_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1=0.9; b2=0.999; eps=1e-8
    best = a.copy()
    best_val = _objective(a)  # actual

    # iterate
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        M, grad = _smooth_max_and_grad(a, beta)
        # gradient of objective
        g = 2.0 * n * grad  # since objective = 2*n*M
        # Adam update
        m = b1 * m + (1-b1) * g
        v = b2 * v + (1-b2) * (g**2)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps) * m_hat  # step size 0.1 may adjust.
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
        # compute true value every maybe 1000 steps
        if t % 1000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[Adam] iter={t} best={best_val:.6f}")

    return best, best_val

def _smooth_max_and_grad(a, beta):
    n = a.size
    conv = _conv_fft(a, a)
    # stable softmax
    max_c = conv.max()
    expb = np.exp(beta * (conv - max_c))
    Z = expb.sum()
    w = expb / Z
    # smooth_max = max_c + (1/beta) * np.log(Z)
    M = max_c + (np.log(Z) / beta)
    # gradient w.r.t a via convolution with w
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return M, grad

def _adam_optimize(start_seq, n, deadline, max_iters=200000):
    a = start_seq.clip(1e-12).copy()
    a /= a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr = 0.05
    beta_start = 20.0
    beta_end = 100.0
    half = max_iters // 2

    best = a.copy()
    best_val = _objective(a)

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        beta = beta_end if t > half else beta_start + (beta_end - beta_start) * (t/half)
        M, grad = _smooth_max_and_grad(a, beta)
        g = 2.0 * n * grad
        # Adam updates
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        step = lr / (np.sqrt(v_hat) + eps) * m_hat
        a = a - step
        a = _project_simplex(a)
        if t % 1000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[Adam] it={t} best={best_val:.6f}")

    return best, best_val

"""
Hybrid optimiser for the step‑function bound.

The code searches for a non‑negative vector a (step‑function heights)
that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where conv denotes the linear convolution.  The optimisation proceeds in
the following stages (all respecting the overall time budget):

1.  A rich pool of seed vectors (classical windows, geometric families,
    beta‑pdf windows, random soft‑max vectors, and the previous best
    construction if any) is assembled and the best few are kept.

2.  One‑dimensional golden‑section refinements for the monotone and
    symmetric geometric families are performed.

3.  A smooth‑max surrogate of the max‑convolution is introduced and an
    Adam (exponentiated‑gradient) optimiser is run on the surrogate.
    The smoothness parameter β is annealed from 20 to 100, giving a
    smooth landscape early on and focusing on the true maximum later.

    The Adam updates are projected onto the probability simplex to keep
    the vector non‑negative and normalised.

    The Adam optimiser is invoked on several top seeds (short runs) and
    afterwards a longer run from the currently best vector.

4.  Finally a Metropolis–Hastings simulated‑annealing walk with
    relatively large mass‑transfer moves is used to polish the result.

All phases stop automatically when the remaining time is exhausted.
The best sequence found is stored in the global ``height_sequence_1``
and returned as a plain ``list``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full linear mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classical windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))

    # Gaussian family – fine sweep of σ
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction (must already have the right length)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – clip, renorm
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Golden‑section search (unimodal 1‑D optimisation)
# ----------------------------------------------------------------------


def _golden_section(obj_func, lo: float, hi: float,
                    max_iter: int, deadline: float):
    """Classic golden‑section search on a scalar function."""
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = obj_func(x1)
    f2 = obj_func(x2)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - phi * (hi - lo)
            f1 = obj_func(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = obj_func(x2)
    return (x1, f1) if f1 < f2 else (x2, f2)


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (M, grad) where

        M   = smooth max of conv(a,a) with parameter beta,
        grad = dM/da  (vector of length len(a)).

    M = max_c + (1/beta) * log Σ exp(beta * (c_i - max_c)),
    where c = conv(a,a) and max_c = max(c).
    """
    n = a.size
    conv = _conv_fft(a, a)               # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    expb = np.exp(beta * (conv - max_c))
    Z = expb.sum()
    w = expb / Z                         # still length 2n‑1, sums to 1
    M = max_c + np.log(Z) / beta

    # gradient: dM/da = 2 * conv_rev(w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]   # slice of length n
    return M, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam (exponentiated‑gradient) optimisation on the exact objective
    using a smooth‑max surrogate for the max‑convolution.
    Returns the best vector found and its true objective value.
    """
    a = np.clip(start_seq, 1e-12, None).copy()
    a /= a.sum()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr = 0.05

    beta_start = 20.0
    beta_end = 100.0
    half = max_iters // 2

    best = a.copy()
    best_val = _objective(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # anneal the smooth‑max parameter
        beta = beta_end if t > half else beta_start + (beta_end - beta_start) * (t / half)

        M, grad = _smooth_max_and_grad(a, beta)      # grad = dM/da
        g = 2.0 * n * grad                            # gradient of true objective J = 2·n·M

        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr / (np.sqrt(v_hat) + eps) * m_hat

        a = a - step
        a = _project_simplex(a)

        # occasional true‑objective evaluation
        if t % 1000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[Adam] it={t:6d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (large mass transfers)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with larger mass‑transfer moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # start temperature a little higher than the current value
    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # transfer a random fraction (up to 10 %) of the source mass
        delta = rng.random() * 0.10 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d} best={best_val:.6f}")

        # occasional global perturbation (every 800 steps)
        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] (jump) it={it:6d} best={best_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Helper: up‑sample a short vector to a longer length (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation from len(seq) to new_len.
    The returned vector is normalised to sum = 1.
    """
    old_len = len(seq)
    if old_len == new_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_len, endpoint=False)
    up = np.interp(x_new, x_old, seq)
    up = np.clip(up, 0.0, None)
    total = up.sum()
    if total == 0.0:
        return _uniform(new_len)
    return up / total


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances expressiveness and speed.
    # 8000 works well on the provided hardware.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Retrieve the previous construction (if any) and adapt its length.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1:
                prev_arr = _upsample(prev_arr, n)
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # 0) Build a rich set of seeds and keep the top few.
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, prev_arr)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    best_seq, best_val = seed_vals[0][0].copy(), seed_vals[0][1]
    print(f"[Seeds] best initial val={best_val:.6f}")

    top_seeds = [s for s, _ in seed_vals[:8]]

    # ------------------------------------------------------------------
    # 1) Fine‑tune monotone and symmetric geometric families.
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        lo = max(0.0, 1.0 - 5.0 / n)
        hi = min(1.0, 1.0 - 0.05 / n)

        # monotone geometric
        def obj_mono(r):
            return _objective(_geom(n, r))

        r_opt, v_opt = _golden_section(obj_mono, lo, hi,
                                       max_iter=30, deadline=deadline)
        seq_opt = _geom(n, r_opt)
        if v_opt < best_val:
            best_val, best_seq = v_opt, seq_opt.copy()
            print(f"[Geom] r={r_opt:.12f} val={v_opt:.6f}")

        # symmetric geometric
        def obj_sym(r):
            return _objective(_sym_geom(n, r))

        r_opt2, v_opt2 = _golden_section(obj_sym, lo, hi,
                                         max_iter=30, deadline=deadline)
        seq_opt2 = _sym_geom(n, r_opt2)
        if v_opt2 < best_val:
            best_val, best_seq = v_opt2, seq_opt2.copy()
            print(f"[SymGeom] r={r_opt2:.12f} val={v_opt2:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam runs on a few top seeds (exploration).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120:          # we need at least a couple of minutes left
        per_seed_iters = 60_000
        for idx, seed_vec in enumerate(top_seeds[:4]):
            seq_i, val_i = _adam_optimize(seed_vec, n, deadline,
                                          max_iters=per_seed_iters)
            if val_i < best_val:
                best_val, best_seq = val_i, seq_i.copy()
                print(f"[Adam seed {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Long Adam refinement from the current best vector.
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        seq_long, val_long = _adam_optimize(best_seq, n, deadline,
                                            max_iters=200_000)
        if val_long < best_val:
            best_val, best_seq = val_long, seq_long.copy()
            print(f"[Adam long] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing walk.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_ann, val_ann = _anneal_optimize(best_seq, n, deadline,
                                            max_iters=None)
        if val_ann < best_val:
            best_val, best_seq = val_ann, seq_ann.copy()
            print(f"[Anneal final] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final sanity check with the provided evaluate_sequence (if present).
    # ------------------------------------------------------------------
    try:
        eval_fn = globals().get('evaluate_sequence')
        if callable(eval_fn):
            final_val = eval_fn(best_seq.tolist())
            print(f"[final evaluate_sequence] {final_val}")
        else:
            final_val = best_val
    except Exception as e:
        final_val = best_val
        print("[final evaluate_sequence] error:", e)

    print(f"\n=== Finished ===\nBest objective ≈ {final_val:.6f} (length {n})")

    # store for a possible subsequent run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
