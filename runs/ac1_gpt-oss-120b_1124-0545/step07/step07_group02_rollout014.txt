"""
Hybrid optimiser (v2) for the step‑function bound

The optimiser combines several complementary ideas:

1.  Rich seed pool – classic windows, geometric families,
    beta‑pdf windows and random soft‑max vectors – and keeps the
    best few seeds.

2.  1‑D golden‑section optimisation of the monotone and symmetric
    geometric families (the decay factor r is refined).

3.  Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth
    objective (the same routine as the reference implementation).

4.  **Smooth‑max gradient descent** – a continuation method that
    replaces the hard max by a softmax (log‑sum‑exp) with a
    temperature β that is increased during the run.  The gradient
    of the smooth max can be computed in O(n log n) by a single
    FFT convolution, and the update is performed in the log‑domain
    (mirror‑descent) to keep the iterate on the probability simplex.

5.  A final simulated‑annealing walk with larger mass‑transfer moves.

All stages respect the global time budget (default 1000 s) and the
best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The input is assumed to be a non‑negative vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple families (all return a probability vector on the simplex)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α∈(0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of start vectors, all normalised to sum = 1.
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # basic families ----------------------------------------------------
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_geom(n, 0.0) ** (1.0 / (p + 1)))  # placeholder, not used
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            # a simple geom‑power mixture
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (r ** (idx - 1)) / (idx ** p)
            a /= a.sum()
            seeds.append(a)
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classical windows -------------------------------------------------
    seeds.append(_sine_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_beta_window(n, alpha))

    # random soft‑max vectors -------------------------------------------
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if it has the right length ------------
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up ----------------------------------------------------
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# Euclidean projection onto the simplex (used by the annealer)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # keep numbers in a safe range
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (continuation method)
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth maximum  t = (1/β) log Σ_k exp(β·b_k)  w.r.t. a,
    where b = a * a (full convolution).  Returns (grad, max_b).
    """
    n = a.size
    b = _conv_fft(a, a)                           # length 2n‑1
    max_b = b.max()
    # softmax weights, shifted by max_b for numerical stability
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:                              # degenerate case – fall back to uniform
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum

    # gradient: 2 * (a convolve reversed w)  (slice n‑1 … 2n‑2)
    w_rev = w[::-1]
    conv = _conv_fft(a, w_rev)                    # length 3n‑2
    grad = 2.0 * conv[n - 1:2 * n - 1]             # exactly n components
    return grad, max_b


def _smooth_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300000,
                     beta_start: float = 0.5,
                     beta_factor: float = 1.02,
                     beta_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Continuation method: start with a low β (smooth max) and increase it.
    Uses exponentiated‑gradient updates on the log‑domain to stay on the
    simplex.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    beta = beta_start
    lr0 = 1.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad, _ = _smooth_max_grad(a, beta)

        # learning rate decays like 1/√t (mirroring the sub‑gradient schedule)
        lr = lr0 / np.sqrt(it)

        # exponentiated‑gradient step (move against the gradient)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)          # stabilise numerics
        a = np.exp(log_a)
        a /= a.sum()

        # evaluate from time to time (every 500 steps) – cheap check
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth] it={it:7d} β={beta:.3f} best={cur_val:.6f}")

        # raise temperature slowly
        beta = min(beta * beta_factor, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (larger mass‑transfer moves)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with larger mass‑transfer moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.06)
    decay = 0.9992

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # transfer a random fraction (up to 10 %) of the source mass
        delta = rng.random() * 0.10 * cur[i]
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation (every 800 steps)
        if it % 800 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] (jump) it={it:7d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# 1‑D golden‑section optimisation for a monotone / symmetric geometric family
# ----------------------------------------------------------------------


def _golden_section(obj_func, lo: float, hi: float,
                    max_iter: int, deadline: float):
    """
    Classic golden‑section search on a unimodal scalar function.
    Returns (x_opt, f_opt) within the interval [lo, hi].
    """
    phi = (math.sqrt(5) - 1) / 2  # 0.618…
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = obj_func(x1)
    f2 = obj_func(x2)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - phi * (hi - lo)
            f1 = obj_func(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = obj_func(x2)
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence discovered as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the grader
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length – 5000 is a good compromise (still very fast FFTs,
    #   a little more freedom than 4000).
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Retrieve a possible previous construction (allows chaining runs)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate the previous vector (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) generate a rich set of seeds and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    # keep the best 6 seeds for the expensive phases
    top_seeds = [s for s, _ in seed_vals[:6]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) fine‑tune monotone and symmetric geometric families with a
    #    golden‑section search (tight optimisation on a single parameter)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8:
        lo = max(0.0, 1.0 - 5.0 / n)
        hi = min(1.0, 1.0 - 0.05 / n)

        # monotone geometric ------------------------------------------------
        def obj_mono(r):
            seq = _geom(n, r)
            return _objective(seq)

        r_opt, v_opt = _golden_section(obj_mono, lo, hi,
                                       max_iter=30, deadline=deadline)
        seq_opt = _geom(n, r_opt)
        if v_opt < best_val:
            best_val = v_opt
            best_seq = seq_opt.copy()
            print(f"[geom] r={r_opt:.9f} val={v_opt:.6f}")

        # symmetric geometric -----------------------------------------------
        def obj_sym(r):
            seq = _sym_geom(n, r)
            return _objective(seq)

        r_opt2, v_opt2 = _golden_section(obj_sym, lo, hi,
                                         max_iter=30, deadline=deadline)
        seq_opt2 = _sym_geom(n, r_opt2)
        if v_opt2 < best_val:
            best_val = v_opt2
            best_seq = seq_opt2.copy()
            print(f"[sym‑geom] r={r_opt2:.9f} val={v_opt2:.6f}")

    # ------------------------------------------------------------------
    # 3) Sub‑gradient mirror‑descent on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 12:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=400000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Smooth‑max continuation optimisation (new ingredient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15:
        seq, val = _smooth_optimize(best_seq, n, deadline,
                                    max_iters=250000,
                                    beta_start=0.5,
                                    beta_factor=1.03,
                                    beta_max=300.0)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Smooth] best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final simulated‑annealing walk (uses all remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # 6) Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for a possible subsequent run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
