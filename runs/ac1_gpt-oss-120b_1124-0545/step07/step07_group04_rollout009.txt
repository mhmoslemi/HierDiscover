"""
Optimise a non‑negative step‑function (list of heights) for the
objective

    2·n·max(conv(a,a)) / (sum(a)²)

using a three‑stage approach:

1.  Build a diverse family of “seed” vectors (uniform, geometric,
    power‑law, …) and randomise their order.  The best few permutations
    are kept.
2.  Refine every kept seed with a smooth‑max surrogate and L‑BFGS
    (the usual soft‑max / log‑sum‑exp trick).  This stage can stay
    completely deterministic thanks to a fixed random seed.
3.  Finish with a mirror‑descent / exponentiated‑gradient sub‑gradient
    optimiser and a tiny Metropolis annealer.

All work respects the given time budget (default 1000 s) and the
returned sequence is a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    from scipy import optimize
    SCIPY_OPT = True
except Exception:   # pragma: no cover
    SCIPY_OPT = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative and sum>0.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Families of normalised seeds (all sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate – value + gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation

        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v)))

    ``beta`` > 0.  The gradient w.r.t. the logits ``v`` is supplied.
    """
    a = _softmax(v)                         # probability vector (sum = 1)
    conv = _conv_fft(a, a)                 # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_val               # factor 2·n

    w = exp_shifted / sum_exp                # weights for each shift
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

    # back‑prop through the soft‑max (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)                       # outer factor 2·n
    return obj, grad_v


def _l_bfgs_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: np.ndarray,
                     maxiter_per_stage: int = 300):
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector found and its exact objective.
    """
    if not SCIPY_OPT:
        return start_seq, _objective(start_seq)

    eps = 1e-30
    v = np.log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(fun,
                                v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False,
                                         'gtol': 1e-12,
                                         'ftol': 1e-12})
        v = res.x
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Stops when time runs out or ``max_iters`` is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient of max – average over all active indices
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very small Metropolis walk that swaps a tiny amount of mass between
    two random coordinates.  Used only if time is left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Dimension – 4000 works well and matches earlier runs.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a pool of seed patterns (all are probability vectors).
    # ------------------------------------------------------------------
    base_seeds = [_uniform(n)]

    # geometric decays
    for r in np.linspace(0.2, 0.99, 9):
        base_seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        base_seeds.append(_power(n, p))

    # combined geometric ∗ power‑law
    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            base_seeds.append(_geom_power(n, r, p))

    # previous construction (if any) – normalise to a probability vector
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape[0] != n:
                # resize by truncating / padding with zeros
                if prev_arr.shape[0] > n:
                    prev_arr = prev_arr[:n]
                else:
                    tmp = np.zeros(n, dtype=np.float64)
                    tmp[:prev_arr.shape[0]] = prev_arr
                    prev_arr = tmp
            prev_arr = np.clip(prev_arr, 0.0, None)
            if prev_arr.sum() > 0.0:
                base_seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass  # ignore malformed previous result

    # ------------------------------------------------------------------
    # 1) Random permutation search over the seed pool.
    #    Keep the top‑k raw permutations for downstream refinement.
    # ------------------------------------------------------------------
    top_k = 5                          # how many elite perms we retain
    elite = []                         # list of (value, seq) tuples
    best_seq = None
    best_val = float("inf")

    perm_deadline = start_time + 0.30 * (deadline - start_time)  # ~30 % of budget

    while time.time() < perm_deadline:
        # pick a random base pattern and randomise its order
        base = rng.choice(base_seeds)
        perm = rng.permutation(n)
        seq = base[perm]

        val = _objective(seq)

        # global best
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[perm] new global best {best_val:.6f}")

        # maintain elite list (sorted ascending)
        if len(elite) < top_k:
            elite.append((val, seq.copy()))
            elite.sort(key=lambda x: x[0])
        else:
            if val < elite[-1][0]:
                elite[-1] = (val, seq.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2) L‑BFGS smooth‑max refinement on the elite seeds.
    # ------------------------------------------------------------------
    if SCIPY_OPT:
        beta_sched = np.geomspace(1.0, 8000.0, num=15)
        for val0, start_seq in elite:
            if time.time() > deadline:
                break
            seq_ref, val_ref = _l_bfgs_optimize(start_seq,
                                                n,
                                                deadline,
                                                beta_schedule=beta_sched,
                                                maxiter_per_stage=250)
            if val_ref < best_val:
                best_val = val_ref
                best_seq = seq_ref.copy()
                print(f"[overall L‑BFGS] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient mirror‑descent refinement (if time permits).
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        seq_ref, val_ref = _subgradient_refine(best_seq,
                                               n,
                                               deadline,
                                               max_iters=250_000)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"[overall SubGrad] refined best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        seq_ref, val_ref = _anneal_optimize(best_seq,
                                            n,
                                            deadline,
                                            max_iters=300_000)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"[overall Anneal] final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Store for later runs and return.
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = best_seq.copy()
    print("\n=== Finished ===")
    print(f"Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
