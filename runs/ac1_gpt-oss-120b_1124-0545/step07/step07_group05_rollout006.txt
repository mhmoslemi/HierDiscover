"""
Optimisation of a non‑negative step‑function (vector a) that minimises

    2·n·max(conv(a,a)) / (∑ a)²

The algorithm is deliberately simple and focuses on a long
high‑frequency simulated‑annealing (SA) phase with two complementary
move types:

* **Random Dirichlet moves** – re‑allocate the total mass of 2‑4
  randomly chosen indices.
* **Targeted moves** – pick an index that participates most in the
  current maximal convolution entry (the “peak”) and move part of its
  mass to a random location.  The Dirichlet concentration for these
  moves is set < 1 (0.2) so the transfer is aggressive.

The SA loop runs until the global deadline (≈ 1000 s) and updates the
full convolution incrementally, which makes each move O(n) rather than
O(n log n).  After SA a short mirror‑descent (exact sub‑gradient) phase
polishes the result.

All vectors are normalised to sum = 1, therefore the objective printed
by the code equals ``evaluate_sequence``’s output.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _objective(a: np.ndarray) -> float:
    """
    Objective for a vector that already satisfies ∑a = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Exact sub‑gradient mirror descent (same as in the reference code)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Works on the probability simplex (∑a = 1, a ≥ 0).
    """
    eps = 1e-30
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of the max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror) step on the simplex
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Enhanced high‑frequency simulated annealing
# ----------------------------------------------------------------------


def _sa_enhanced_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
    recompute_interval: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with two move families:
        * random Dirichlet re‑allocation on 2‑4 indices (conc=1.0)
        * targeted move that reduces the current maximal convolution
          entry (i chosen proportionally to a[i]·a[k*‑i]; conc=0.2)

    The full convolution is updated incrementally, which makes each
    move O(n).  The temperature decays geometrically and is occasionally
    reheated if no improvement happens for a long stretch.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)               # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # temperature schedule
    T = max(init_temp_factor * cur_obj, min_temp)
    decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        # cooling
        T = max(T * decay, min_temp)

        # occasional reheating if stuck
        if it % 200_000 == 0 and it - last_improve > 150_000:
            T = max(T, 0.05 * cur_obj)   # give a small temperature bump

        # ---------- select move ----------
        if rng.random() < 0.5:
            # random Dirichlet on 2‑4 indices
            k = rng.integers(2, 5)                     # 2‑4 indices
            idx = rng.choice(n, size=k, replace=False)
            conc = 1.0
        else:
            # targeted move
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)

            # contribution of each i to the peak
            contrib = cur[support] * cur[k_star - support]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                i = rng.choice(support, p=contrib / total_contrib)
            else:
                i = rng.choice(support)

            # pick a second index uniformly (different from i)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)
            conc = 0.2                                 # aggressive shift

        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # new allocation via a Dirichlet draw
        new_alloc = rng.dirichlet(np.full(idx.size, conc)) * total
        deltas = new_alloc - cur[idx]

        if np.allclose(deltas, 0.0):
            continue

        # ---------- incremental convolution update ----------
        conv_new = cur_conv.copy()
        # linear terms a * Δ + Δ * a
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur      # a*Δ and Δ*a
            conv_new[2 * p] += delta * delta            # Δ*Δ self term

        # cross terms Δ_i·Δ_j (appear twice)
        for ii in range(idx.size):
            for jj in range(ii + 1, idx.size):
                pos = idx[ii] + idx[jj]
                conv_new[pos] += 2.0 * deltas[ii] * deltas[jj]

        new_obj = 2.0 * n * np.max(conv_new)

        # ---------- Metropolis acceptance ----------
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept move
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                last_improve = it
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

        # periodic full recomputation to limit numerical drift
        if it % recompute_interval == 0:
            cur_conv = _conv_fft(cur, cur)
            cur_obj = 2.0 * n * np.max(cur_conv)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = cur.copy()
                last_improve = it
                print(f"[SA‑recalc] n={n} it={it} best={best_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for the wrapper

    # try to reuse a previously discovered construction, if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Fixed length – large enough to approach the theoretical lower bound
    n = 8000

    # -------------------- seed generation --------------------
    seeds = []

    # previous solution (if length matches)
    if previous is not None and previous.shape[0] == n:
        seeds.append(previous / previous.sum())

    # uniform baseline

    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val

            print(f"[seed] new global best approx={val:.6f}")

    # -------------------- long SA phase --------------------
    # Allocate most of the remaining budget to SA

    if time_left > 10.0:   # give at least a few seconds to SA
        sa_deadline = time.time() + 0.85 * time_left
        seq, val = _sa_enhanced_optimize(best_seq, n, sa_deadline)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall‑SA] new global best approx={val:.6f}")

    # -------------------- final sub‑gradient polish --------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_seq is not None:
        sub_deadline = time.time() + remaining
        seq, val = _subgradient_refine(best_seq, n, sub_deadline,
                                       max_iters=200_000)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[final‑SubGrad] new global best approx={val:.6f}")

    # Fallback – should never be needed
    # ------------------------------------------------------------------

        print("[fallback] using uniform distribution")

    total = best_seq.sum()

        pass

    print(
        f"\n=== Finished ===\nBest approx objective ≈ {best_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
