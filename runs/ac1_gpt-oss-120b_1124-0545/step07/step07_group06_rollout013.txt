def _lp_step(f, M, n, deadline):
    # Build sparse matrix A where A[k,i] = f_{k-i} if 0<=k-i<n.
    rows = []
    cols = []
    data = []
    for i in range(n):
        # column i corresponds to b_i coefficient f_{k-i} in each constraint.
        # For each k where k-i in [0, n-1] -> k from i to i+n-1
        start = i
        end = i + n - 1
        for k in range(start, end+1):
            rows.append(k)
            cols.append(i)
            data.append(f[k - i]) # because f_{k-i}
    # Now A shape (2n-1, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs', options={'time_limit': time_remaining})
```

def _gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    a = np.exp(-((i - center) ** 2) / (2 * sigma**2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _cauchy(n, tau):
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    a = 1.0 / (1.0 + ((i - center) / tau) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _raised_cosine(n, a):
    # a in (0,1] controls broadness: weight = (1 + cos(pi * (i - center)/ (a * (n/2)))) /2 for |i-center| <= a*(n/2)
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    width = a * (n/2)
    diff = np.abs(i - center)
    a_vals = np.where(diff <= width, 0.5 * (1 + np.cos(np.pi * diff / width)), 0.0)
    s = a_vals.sum()
    return a_vals / s if s > 0 else _uniform(n)

    M = np.max(_conv_fft(f, f))
    # Build sparse matrix for convolution f * b
    rows = []
    cols = []
    data = []
    for i in range(n):
        # positions where b_i contributes: k = i + j where j in [0, n-1], so k range i..i+n-1
        k_start = i
        k_end = i + n - 1

        rows.append(ks)
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(f[ks - i])  # f_{k-i}
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum(b) -> minimize -sum(b)
    # time remaining:
    time_left = deadline - time.time()

        return f, 2.0 * n * M
    # Not all LP solvers accept time limit; use options:
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                    method='highs',
                                    options={'presolve': True, 'time_limit': time_left})
    except Exception as e:
        print("[LP] linprog failed:", e)
        return f, 2.0 * n * M
    if not res.success:
        print("[LP] not successful:", res.message)
        return f, 2.0 * n * M
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0: 
        return f, 2.0 * n * M
    # scale:
    alpha = 1.0 / sum_b   # since we have sum(f)=1, we want same sum after scaling? Actually target sum = 1
    b_scaled = b * alpha
    # compute new max after mixing

"""
propose_candidate – a multi‑phase search for the optimal non‑negative step‑function.

The routine builds a diverse portfolio of analytical seed families (uniform,
beta‑shapes, Gaussian, Cauchy, raised‑cosine, geometric, power‑law, random
Dirichlet / sparse patterns).  Each seed is normalised to sum = 1.

For the most promising seeds we run three optimisation phases:

1.  Adam on a smooth‑max (log‑sum‑exp) surrogate of the max‐convolution.
    The surrogate becomes sharper over time, driving the iterates toward a
    true minimiser of 2·n·max(conv).

2.  Exact sub‑gradient mirror descent (exponentiated gradient) on the genuine
    max­‑convolution objective.

3.  A two‑stage simulated‑annealing:
    – a “targeted” SA that always attacks the current bottleneck entry of the
      convolution by redistributing mass among its heaviest contributors;
    – a “global” SA that performs random reallocations of a few indices,
      allowing escape from local minima.

Optionally, for the smallest length (n = 2000) a linear‑programming step is
performed: maximising the total mass of a vector *b* under the linear constraints
 f ∗ b ≤ M (where M = max(f∗f)).  The resulting *b* is rescaled to sum = 1 and
mixed with the current iterate, guaranteeing a reduction of the peak
convolution value.

All phases respect the global deadline (default 1000 s) and constantly keep
the best sequence seen so far.  The final sequence (list of floats) is returned.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, no circular wrap)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a normalised probability vector (∑a = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)

# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i (monotone decreasing)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a[i] = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑like shape:
        a[i] ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}.
    α,β>0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cauchy(n: int, tau: float) -> np.ndarray:
    """Cauchy (Lorentzian) shape."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + ((i - centre) / tau) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, rel_width: float) -> np.ndarray:
    """
    Raised cosine supported on a relative width of the interval.
    rel_width ∈ (0,1] controls the half‑width w = rel_width·(n/2).
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = rel_width * (n / 2.0)
    diff = np.abs(i - centre)
    a = np.where(diff <= w,
                 0.5 * (1.0 + np.cos(np.pi * diff / w)),
                 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a full Dirichlet vector (conc>0)."""
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse vector with exactly k non‑zero entries (Dirichlet on the support)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _make_seeds_enhanced(n: int,
                         previous: np.ndarray | None,
                         rng: np.random.Generator) -> list[np.ndarray]:
    """Create a rich collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.12, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families

    # dense beta‑shape grid
    alphas = np.linspace(0.30, 1.50, 7)
    betas = np.linspace(0.30, 1.50, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # new analytic shapes
    for sigma in np.linspace(n / 10.0, n / 2.0, 5):

    for tau in np.linspace(n / 10.0, n / 2.0, 5):

    for rel in np.linspace(0.25, 1.0, 5):

    for conc in (0.1, 0.3, 0.7, 1.5):

    if previous is not None and previous.shape == (n,):

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 50000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Operates on log‑probabilities to stay on the simplex automatically.
    """
    eps = 1e-30
    v = np.log(seed + eps)               # working in log‑space guarantees positivity

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta_smooth = 2.0
    beta_mul = 1.015   # slower sharpening than the original

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # stays on the simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w·conv w.r.t. a ----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]

        # ---- convert to gradient w.r.t. v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping (helps stability)
        max_norm = 15.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ---- Adam update on v ----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)   # decreasing step size
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 3000000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that focuses on the current maximal convolution entry.
    The update always redistributes mass from the heaviest contributors to that
    entry among a few selected indices.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9997

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        # locate the bottleneck index
        k_star = int(np.argmax(cur_conv))

        # indices that truly contribute to the bottleneck
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few of the heaviest contributors (max 6)
        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # decide how many indices to modify (2‑4)
        k_move = rng.integers(2, 5)

        # choose indices: try to use heavy ones first, fill up with random others
        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the total mass among the chosen positions
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross Δ_i * Δ_j for i≠j (appears twice)
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


def _global_sa_optimize(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        init_temp_factor: float = 0.02,
                        min_temp: float = 1e-9,
                        max_iters: int = 2000000) -> tuple[np.ndarray, float]:
    """
    Global simulated annealing with random reallocations.
    Allows the search to jump out of narrow basins that targeted SA may miss.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9998

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * decay, min_temp)

        # decide how many indices to modify (2‑5)
        k_move = rng.integers(2, 6)
        chosen = rng.choice(n, size=k_move, replace=False)

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]

        conv_new = cur_conv.copy()
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[GlobalSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               mix_factor: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement for small n (e.g. n=2000).
    Maximise the total mass of b subject to f*b ≤ M where M = max(f*f).
    The resulting b is rescaled to sum = 1 and mixed with f.
    """
    # Compute current peak
    conv_f = _conv_fft(f, f)
    M = float(np.max(conv_f))

    # Build sparse convolution matrix A where (A @ b)[k] = sum_i f_{k-i} b_i
    rows = []
    cols = []
    data = []
    # Vectorised construction: for each column i, the rows are i..i+n-1
    for i in range(n):
        ks = np.arange(i, i + n, dtype=np.int32)          # length n
        rows.append(ks)
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(f[ks - i])                           # f_{k-i}
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    c = -np.ones(n)  # maximise sum(b) → minimise -sum(b)

    time_left = deadline - time.time()
    if time_left < 1.0:
        return f, 2.0 * n * M

    try:
        res = scipy.optimize.linprog(c,
                                    A_ub=A,
                                    b_ub=b_ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'time_limit': time_left})
    except Exception as e:
        print("[LP] linprog exception:", e)
        return f, 2.0 * n * M

    if not res.success:
        # LP not solved – skip this refinement
        print("[LP] not successful:", res.message)
        return f, 2.0 * n * M

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return f, 2.0 * n * M

    # Scale b back to have sum = 1 (same as f)
    b_scaled = b / sum_b

    # Mix
    f_new = (1.0 - mix_factor) * f + mix_factor * b_scaled
    conv_new = _conv_fft(f_new, f_new)
    new_obj = 2.0 * n * np.max(conv_new)
    return f_new, new_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the global time budget and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Load a previously found construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to explore – we start small (cheap) and finish with a larger
    # length if time permits.
    lengths = [2000, 4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds_enhanced(n, previous, rng)

        # ------------------------------------------------------------------
        # 1) Quick evaluation of raw seeds
        # ------------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the most promising seeds (top‑3)
        # ------------------------------------------------------------------
        top_idx = np.argsort(seed_vals)[:3]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=50000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=800000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Targeted simulated annealing (high‑frequency)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline,
                                             max_iters=3000000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Global random annealing (exploratory)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _global_sa_optimize(global_best_seq, n, deadline,
                                           max_iters=2000000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑GlobalSA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 6) Optional LP refinement for the smallest size (n=2000)
        # ------------------------------------------------------------------
        if n == 2000 and global_best_seq is not None:
            seq, val = _lp_refine(global_best_seq, n, deadline, mix_factor=0.2)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[LP‑refine] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Verify with the provided evaluator (optional sanity‑check)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
