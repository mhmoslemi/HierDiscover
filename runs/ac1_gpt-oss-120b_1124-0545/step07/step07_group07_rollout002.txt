"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear convolution.  The algorithm

* builds a very diverse set of analytic and random seeds (including wide
  Gaussian, triangular, cosine‑bell and exponential shapes that were
  missing from the previous version),
* keeps the best seed,
* then performs four time‑aware optimisation phases:
    1. Adam on a smooth‑max (log‑sum‑exp) surrogate,
    2. Exponentiated‑gradient sub‑gradient descent on the true max,
    3. L2‑flattening (driving the auto‑convolution toward a constant),
    4. Simulated‑annealing hill‑climb that moves tiny amounts of mass
       from indices that dominate the current maximal convolution entry
       to random receivers.
All phases respect a global 1000 s budget and the final vector is stored
in the global variable ``height_sequence_1`` for warm‑starting future
calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Return 2·n·max(conv(a,a)) under the implicit constraint sum(a)=1.
    The caller must ensure a is non‑negative and normalised.
    """
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)

# Seed generation (much richer families)
# ----------------------------------------------------------------------


def _make_seeds(
    n: int, rng: np.random.Generator, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # 1. Simple baselines
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))                      # uniform
    i = np.arange(n, dtype=np.float64)
    seeds.append(np.sin(np.pi * (i + 1) / (n + 1)) / np.sum(np.sin(np.pi * (i + 1) / (n + 1))))  # half‑sine

    # 2. Geometric / power‑law families
    for r in np.linspace(0.2, 0.95, 9):
        a = r ** np.arange(n, dtype=np.float64)
        seeds.append(a / a.sum())
    for p in np.linspace(0.0, 4.0, 9):
        a = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
        seeds.append(a / a.sum())

    # 3. Beta‑like shapes
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            idx = np.arange(1, n + 1, dtype=np.float64)
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 4. Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(4):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # 5. Softmax of random logits
    for _ in range(4):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # 6. Random piecewise‑linear shapes
    for _ in range(4):
        K = max(10, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(np.arange(n, dtype=np.float64), knots, p)
        seeds.append(a / a.sum())

    # 7. Wide Gaussian bells (σ up to a sizable fraction of n)
    centre = (n - 1) / 2.0
    for sigma in (n / 20.0, n / 12.0, n / 8.0, n / 6.0, n / 4.0):
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 8. Symmetric exponential (Laplace) shapes
    for tau in (n / 15.0, n / 10.0, n / 6.0):
        a = np.exp(-np.abs(np.arange(n) - centre) / tau)
        a += 1e-12
        seeds.append(a / a.sum())

    # 9. Triangular shape (linear up then down)
    mid = n // 2
    tri = np.concatenate([np.arange(1, mid + 1), np.arange(mid, 0, -1)])
    seeds.append(tri / tri.sum())

    # 10. Raised cosine (half‑cosine bell)
    half_cos = np.maximum(
        0.0,
        0.5 * (1.0 + np.cos(np.pi * (np.arange(n) - centre) / centre)),
    )
    seeds.append(half_cos / half_cos.sum())

    # 11. Log‑normal bells
    for _ in range(3):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        a += 1e-12
        seeds.append(a / a.sum())

    # 12. Warm‑start from previous global best, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

# Optimisation primitives

def _adam_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp).  Parameterisation:
    a = softmax(v).  Temperature β is increased multiplicatively.
    """
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 1.0            # start temperature
    beta_mul = 1.016

    best_seq = a0.copy()
    best_val = _objective(a0, n)
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1

        # softmax → simplex
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max using exponentiated‑gradient."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² (μ = 1/(2n‑1)) using exponentiated‑gradient.
    This drives the auto‑convolution towards a constant.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)   # length = 3n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[L2Flat] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


def _annealing_hill_climb_optimize(
    a0: np.ndarray, n: int, deadline: float, max_iters: int = 300_000
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb: move a small amount of mass from
    indices that dominate the current maximal convolution entry to a
    random receiver.  Accept worsening moves with a temperature that
    decays exponentially.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val

    step_scale = 0.08               # start with up‑to‑8 % of donor mass
    init_temp = 0.012               # initial temperature for uphill moves
    decay_rate = 1e-5               # exponential decay factor

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        # objective for the candidate
        new_val = _objective(new, n)

        # temperature for this iteration
        temp = init_temp * np.exp(-decay_rate * it)

        # acceptance rule
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temp > 0.0:
                prob = np.exp(-(new_val - cur_val) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[AnnealHill] it={it:6d} best={new_val:.6f}")

        # slowly shrink the step size
        if it % 3000 == 0:
            step_scale = max(step_scale * 0.90, 1e-5)

        if it % 5000 == 0:
            print(f"[AnnealHill] it={it:6d} cur={cur_val:.6f} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None
    else:
        warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length – larger n gives more degrees of freedom
    # ------------------------------------------------------------------
    n = 6000  # still comfortably within the time budget

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Allocate time for deep refinement of the current global best
    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase_deadline = time.time() + remaining * 0.40
        seq, val = _adam_optimize(global_best_seq, n, phase_deadline, max_iters=500_000)
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Phase‑Adam] best={val:.6f}")

    # Phase 2 – Exact sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase_deadline = time.time() + remaining * 0.30
        seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline, max_iters=500_000)
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Phase‑SubGrad] best={val:.6f}")

    # Phase 3 – L2‑flattening descent
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.15
        seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline, max_iters=500_000)
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Phase‑L2Flat] best={val:.6f}")

    # Phase 4 – Simulated‑annealing hill‑climb (use whatever time is left)
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 2.0
        seq, val = _annealing_hill_climb_optimize(global_best_seq, n, phase_deadline, max_iters=500_000)
        if val < global_best_val:
            global_best_val, global_best_seq = val, seq.copy()
            print(f"[Phase‑HillAnneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
