"""
Propose a non‑negative step function (list of heights) that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm builds a large and diverse set of analytic/random seeds, keeps the
best, and then spends the remaining time in a sequence of sophisticated
optimisation phases:

1. Adam on a smooth‑max surrogate (softmax parametrisation).
2. Exact sub‑gradient descent (exponentiated‑gradient).
3. L₂‑flattening (drive the auto‑convolution towards a constant).
4. **Additive gradient descent** on the true max‑norm (new phase).  Here we
   explicitly move mass away from the indices that contribute to the maximal
   convolution entry and redistribute it uniformly – a simple projected gradient
   step that often pushes the objective further down.
5. Targeted stochastic hill‑climb for final polishing.

All phases respect the global 1000 s wall‑clock budget.  The best sequence found
is stored in the global variable `height_sequence_1` for warm‑starting future
calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)) assuming Σa = 1."""
    conv = _conv_full(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation (diverse analytic & random families)
# ----------------------------------------------------------------------


def _make_seeds(n: int, rng: np.random.Generator,
                warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # 1. trivial uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. half‑sine window (used previously)
    s = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(s / s.sum())

    # 3. Hann, Hamming, Blackman windows
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    seeds.append(hann / hann.sum())

    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    seeds.append(hamming / hamming.sum())

    blackman = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    seeds.append(blackman / blackman.sum())

    # 4. geometric families
    for r in np.linspace(0.2, 0.95, 12):
        a = r ** i
        seeds.append(a / a.sum())

    # 5. power‑law families
    for p in np.linspace(0.0, 4.0, 11):
        a = 1.0 / ((i + 1) ** p)
        seeds.append(a / a.sum())

    # 6. beta‑like shapes
    idx = i + 1
    for alpha in np.linspace(0.5, 2.0, 5):
        for beta in np.linspace(0.5, 2.0, 5):
            a = (idx ** (alpha - 1.0)) * ((n - idx + 1.0) ** (beta - 1.0))
            seeds.append(a / a.sum())

    # 7. Dirichlet draws (different concentrations)
    for conc in (0.1, 0.3, 1.0, 3.0, 10.0):
        for _ in range(5):
            a = rng.dirichlet(np.full(n, conc))
            seeds.append(a / a.sum())

    # 8. Softmax of Gaussian logits
    for _ in range(5):
        logits = rng.standard_normal(n)
        a = np.exp(logits - np.max(logits))
        seeds.append(a / a.sum())

    # 9. Random piecewise‑linear shapes
    for _ in range(5):
        K = max(8, int(np.sqrt(n) * rng.uniform(0.5, 2.0)))
        knots = np.linspace(0, n - 1, K)
        p = rng.dirichlet(np.ones(K))
        a = np.interp(i, knots, p)
        seeds.append(a / a.sum())

    # 10. Log‑normal (bell) shapes
    for _ in range(5):
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.5, 2.0)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2) + 1e-12
        seeds.append(a / a.sum())

    # 11. Kaiser window (beta=5)
    beta = 5.0
    t = (i - (n - 1) / 2) / ((n - 1) / 2)  # range [-1,1]
    a = np.i0(beta * np.sqrt(1 - t ** 2))
    seeds.append(a / a.sum())

    # Warm‑start from the previous global best, if compatible
    if warm_start is not None and warm_start.shape == (n,):
        a = np.maximum(warm_start, 0.0)
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)

    return seeds

def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate (softmax parametrisation)."""
    v = np.log(a0 + 1e-30)                     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta = 5.0            # initial smoothness
    beta_mul = 1.02

    best_seq = a0.copy()
    best_val = _objective(a0, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # a = softmax(v)
        a = np.exp(v - np.max(v))
        a /= a.sum()

        conv = _conv_full(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # smooth‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (using correlation)
        rev_a = a[::-1]
        grad_full = _conv_full(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max (exponentiated‑gradient)."""
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent)
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


def _l2_flatten_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖a*a – μ·1‖² (μ = 1/(2n‑1)) using exponentiated‑gradient.
    Drives the auto‑convolution towards a uniform vector.
    """
    a = a0.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective(a, n)

    mu = 1.0 / (2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        residual = conv - mu
        rev_a = a[::-1]
        grad_full = _conv_full(residual, rev_a)
        grad = 4.0 * grad_full[n - 1:n - 1 + n]

        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _additive_gradient_optimize(a0: np.ndarray, n: int,
                               deadline: float,
                               max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple projected (additive) gradient descent on the true max‑norm.
    At each iteration we move mass away from indices that currently
    contribute to the maximal convolution entry and redistribute it uniformly.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    base_step = 0.5          # initial step scale
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # compute sub‑gradient for the active max entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]          # positive only on the active band

        pos = g > 0
        if not np.any(pos):
            break  # nothing to move

        # safety: avoid division by zero
        ratios = a[pos] / (g[pos] + 1e-30)
        alpha_max = np.min(ratios)          # largest step that keeps a ≥ 0
        # step schedule – decreasing with iteration count
        step = min(alpha_max * 0.5, base_step / np.sqrt(it))

        if step <= 0.0:
            continue

        total_removed = step * np.sum(g)   # mass taken away from active indices
        # additive update
        a = a - step * g
        a += total_removed / n              # uniform redistribution

        # numerical clean‑up
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # optional early break if stuck for a long time
        if no_improve >= 5000 and it > 20000:
            break

    return best_seq, best_val


def _hill_climb_optimize(a0: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb: move a tiny amount of mass away
    from the contributors to the current maximal convolution coefficient.
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_val = _objective(cur, n)
    best_seq = cur.copy()
    best_val = cur_val
    step_scale = 0.02

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(cur, cur)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        contrib = cur[idx_i] * cur[rev_idx]
        total = contrib.sum()

        if total > 0.0:
            probs = contrib / total
            donor = rng.choice(idx_i, p=probs)
        else:
            donor = rng.integers(0, n)

        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta

        val = _objective(new, n)
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()

        if it % 2000 == 0:
            step_scale = max(step_scale * 0.8, 1e-6)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimizes
    `evaluate_sequence`.  Returns the best sequence found as a plain list
    of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    warm_arr = None
    if warm is not None:
        try:
            warm_arr = np.asarray(warm, dtype=np.float64)
        except Exception:
            warm_arr = None

    # ------------------------------------------------------------------
    # Choose a length (power of two for cheap FFTs)
    # ------------------------------------------------------------------
    n = 4096   # works comfortably within the time budget

    # ------------------------------------------------------------------
    # Build a diverse set of seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    seeds = _make_seeds(n, rng, warm_arr)
    print(f"Generated {len(seeds)} seeds (n={n})")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds, keep the global best
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")
    for a in seeds:
        val = _objective(a, n)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[Seed] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.30
    seq, val = _adam_optimize(global_best_seq, n, phase_deadline,
                              max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑Adam] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.20
    seq, val = _subgradient_optimize(global_best_seq, n, phase_deadline,
                                     max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L₂‑flattening descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.15
    seq, val = _l2_flatten_optimize(global_best_seq, n, phase_deadline,
                                    max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑L2Flat] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Additive gradient descent on the true max‑norm
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_deadline = time.time() + remaining * 0.20
    seq, val = _additive_gradient_optimize(global_best_seq, n,
                                            phase_deadline,
                                            max_iters=300_000)
    if val < global_best_val:
        global_best_val = val
        global_best_seq = seq.copy()
        print(f"[Phase‑AddGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Targeted hill‑climb (use whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining - 1.0
        seq, val = _hill_climb_optimize(global_best_seq, n,
                                        phase_deadline,
                                        max_iters=300_000)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Phase‑Hill] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        print("[Verification] sequence rejected – falling back to uniform")
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
