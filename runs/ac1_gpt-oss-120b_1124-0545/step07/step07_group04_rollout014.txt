def _double_gaussian(n, sigma_factor, offset_ratio, w1):
    a = np.exp(-0.5 * ((np.arange(n) - (n-1)*offset_ratio) / (sigma_factor*n))**2)
    b = np.exp(-0.5 * ((np.arange(n) - (n-1)*(1-offset_ratio)) / (sigma_factor*n))**2)
    a = w1 * a
    b = (1-w1) * b
    s = a + b
    return s / s.sum()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, math, random
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4000
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape[0] != n:
                previous = None
        except Exception:
            previous = None

    seeds = _make_seed_pool(n, previous)
    # add extra seeds
    rng = np.random.default_rng(seed+123)
    # Dirichlet draws
    for alpha in np.geomspace(0.05, 5.0, num=6):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, alpha)))
    # random permutations of uniform
    for _ in range(5):
        p = rng.permutation(n)
        u = np.ones(n) / n
        seeds.append(u[p])
    # double-gaussian
    for sigma in [0.04, 0.08, 0.12]:
        for offset in [0.2, 0.4, 0.6, 0.8]:
            for w in [0.5, 0.7]:
                g = _double_gaussian(n, sigma, offset, w)
                seeds.append(g)

    # Clean seeds: clip negative, normalise.
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    seeds = cleaned

    # Quick screen: evaluate each seed, keep top_k (e.g., 12)
    scored = [(s, _objective(s)) for s in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(12, len(scored))
    top_seeds = [s for s,_ in scored[:top_k]]
    best_seq = scored[0][0].copy()
    best_val = scored[0][1]
    print(f"[seed] best raw seed obj={best_val:.6f}")

    # L-BFGS runs
    beta_sched = np.geomspace(1.0, 20000.0, num=15)
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(seed_vec, n, deadline,
                                    beta_schedule=beta_sched,
                                    maxiter_per_stage=400)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best={val:.6f}")

    # subgradient refinement
    if time.time() < deadline:
        seq, val = _subgradient_refine(best_seq, n, deadline,
                                       max_iters=600_000)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[overall SubGrad] refined best={val:.6f}")

    # annealing
    if time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,

    print("\n=== Finished ===")
    print(f"Best obj ≈ {best_val:.6f} (len={len(best_seq)})")

    globals()["height_sequence_1"] = best_seq.copy()

"""

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation   2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Window families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom_centered(n: int, r: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = r ** np.abs(i - centre)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window (0 ≤ α ≤ 1, α=0 → rectangle, α=1 → Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _double_gaussian(
    n: int,
    sigma_factor: float,
    offset_ratio: float,
    w1: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Two Gaussian bumps, placed at positions offset_ratio·(n‑1) and
    (1‑offset_ratio)·(n‑1).  ``w1`` controls the relative weight of the
    first bump.
    """
    i = np.arange(n, dtype=np.float64)
    centre1 = offset_ratio * (n - 1)
    centre2 = (1.0 - offset_ratio) * (n - 1)
    sigma = sigma_factor * n
    g1 = np.exp(-0.5 * ((i - centre1) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((i - centre2) / sigma) ** 2)
    a = w1 * g1 + (1.0 - w1) * g2
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _make_seed_pool(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Create a large, diverse collection of normalised start vectors."""
    seeds: list[np.ndarray] = []

    # Classical families
    seeds.append(_uniform(n))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))

    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_sym_geom_centered(n, r))

    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    for r in (0.3, 0.5, 0.7, 0.9):
        for p in (0.5, 1.0, 1.5, 2.0):
            seeds.append(_geom_power(n, r, p))

    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    for sigma in (0.05, 0.10, 0.18, 0.30):
        seeds.append(_gaussian(n, sigma_factor=sigma))

    # Dirichlet draws – wide range of concentration parameters
    for alpha in np.geomspace(0.05, 5.0, 6):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # Random permutations of the uniform vector (breaks symmetry)
    for _ in range(5):
        perm = rng.permutation(n)
        seeds.append(_uniform(n)[perm])

    # Two‑Gaussian mixtures with varied parameters
    for sigma in (0.04, 0.08, 0.12):
        for offset in (0.2, 0.3, 0.4, 0.6, 0.7, 0.8):
            for w1 in (0.5, 0.7):
                seeds.append(_double_gaussian(n, sigma, offset, w1, rng))

    # include a previous construction if present
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # Clean‑up – clip tiny negatives, normalise.
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0.0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate – objective & gradient for L‑BFGS
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (value, gradient) of the smooth approximation

        Fβ(v) = 2·n·(1/β)·log Σₖ exp(β·convₖ(softmax(v),softmax(v))).

    The gradient is with respect to the logits v.
    """
    a = _softmax(v)                     # probability vector, sum = 1

    conv = _conv_fft(a, a)               # length = 2n‑1
    shift = np.max(conv)
    exp_shifted = np.exp(beta * (conv - shift))
    sum_exp = exp_shifted.sum()
    smooth_val = shift + np.log(sum_exp) / beta

    obj = 2.0 * n * smooth_val           # factor 2·n from definition

    # weights for the weighted sum of convolutions
    w = exp_shifted / sum_exp

    # gradient w.r.t. a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax (Jacobian = diag(a) – a aᵀ)
    dot = np.dot(a, grad_a)
    grad_v = a * (grad_a - dot)
    grad_v *= (2.0 * n)

    return obj, grad_v


def _l_bfgs_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: np.ndarray,
    maxiter_per_stage: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS on the smooth‑max surrogate.
    Returns the best probability vector found together with its exact objective.
    """
    if not SCIPY_OPT:                     # fallback – plain gradient descent
        # simple projected gradient descent using the same surrogate
        v = np.log(start_seq + 1e-30)
        best_seq = start_seq.copy()
        best_val = _objective(best_seq)
        for beta in beta_schedule:
            if time.time() > deadline:
                break
            for _ in range(maxiter_per_stage):
                obj, grad = _smooth_obj_and_grad(v, n, beta)
                # step size heuristics (decrease with iteration count)
                step = 0.5 / (1.0 + _
                              )  # we will not reach here when SciPy available
        return best_seq, best_val

    eps = 1e-30
    v = np.log(start_seq + eps)          # initialise logits

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta)

        res = optimize.minimize(
            fun,
            v,
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "disp": False,
                "gtol": 1e-12,
                "ftol": 1e-12,
            },
        )
        v = res.x

        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[L‑BFGS] β={beta:.1f}   best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact non‑smooth objective.
    Stops when time runs out or ``max_iters`` is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient of max – average over all active indices
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        if len(active) > 0:
            g /= float(len(active))

        # exponentiated‑gradient update
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d}  best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light Metropolis annealing (final polish)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis walk that moves a small percentage of mass between
    two random coordinates.  Used only if time is left.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]   # ≤ 2 % of source mass
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:6d}  best={new_val:.6f}")

        # occasional Gaussian jump to shake out of plateaus
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= tot
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:6d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – keep it large to gain more degrees of freedom.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load possible previous construction (if any).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.shape != (n,):
                previous = None
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1) Build a very rich pool of seeds.
    # ------------------------------------------------------------------
    all_seeds = _make_seed_pool(n, previous, rng)

    # ------------------------------------------------------------------
    # 2) Quick screen – keep the most promising candidates.
    # ------------------------------------------------------------------
    scored = [(s, _objective(s)) for s in all_seeds]
    scored.sort(key=lambda x: x[1])            # low is good
    top_k = min(12, len(scored))
    top_seeds = [s for s, _ in scored[:top_k]]

    global_best_seq = None
    global_best_val = float("inf")
    if scored:
        global_best_seq = scored[0][0].copy()
        global_best_val = scored[0][1]
        print(f"[seed] best raw seed = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Multi‑stage smooth‑max L‑BFGS on the best seeds.
    # ------------------------------------------------------------------
    beta_sched = np.geomspace(1.0, 20000.0, num=15)   # finer schedule, larger final β
    for seed_vec in top_seeds:
        if time.time() > deadline:
            break
        seq, val = _l_bfgs_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            maxiter_per_stage=400,
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall L‑BFGS] new best = {val:.6f}")

        # early exit if we already beat the target
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient mirror‑descent refinement.
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=600_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall SubGrad] refined best = {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final tiny Metropolis annealing (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(
            global_best_seq, n, deadline, max_iters=500_000
        )
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[overall Anneal] final best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be hit.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    print("\n=== Finished ===")
    print(f"Best objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # Store for possible reuse in a later invocation
    globals()["height_sequence_1"] = global_best_seq.copy()

    # Return as plain Python list (the grader will call evaluate_sequence on it)
    return global_best_seq.tolist()
