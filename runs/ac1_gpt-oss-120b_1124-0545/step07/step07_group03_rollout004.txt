   The loop runs until the remaining time budget is exhausted.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, stable)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Evaluate the quantity that `evaluate_sequence` would return for a
    *normalised* sequence (sum = 1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """
    Truncated coefficients of (1‑z)^‑½ (Catalan series) normalised to sum = 1.
    """
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _semicircle_shape(n: int) -> np.ndarray:

    total = vals.sum()
    if total == 0.0:
        return _uniform(n)
    return vals / total

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:

        # gradient w.r.t. the log‑weights (softmax chain rule)
        avg = np.dot(a, grad_a)               # ⟨a,∇_a O⟩
        grad_w = a * (grad_a - avg)           # ∂O/∂w

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # 0.5 s safety margin for the final sanity check
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Optional previous solution (if one is stored globally)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – longer sequences give a tighter asymptotic bound.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    for n in lengths:
        seeds = [
            _uniform(n),
            _catalan_shape(n),
            _semicircle_shape(n),
            _beta_shape(n, 0.5, 0.5),   # arcsine law
            _beta_shape(n, 2.0, 2.0),   # tent‑like
        ]
        if previous is not None and previous.shape == (n,):
            seeds.append(previous)

        for seq in seeds:
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={seq.shape} val={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – massive random Dirichlet sampling (various α)
    # ------------------------------------------------------------------
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]          # exploration of sparsity / flatness
    # allocate roughly half the remaining budget to sampling
    while time.time() < deadline * 0.5:
        # pick a length with a bias towards larger n
        probs = np.array(lengths, dtype=float)
        probs /= probs.sum()
        n = int(rng.choice(lengths, p=probs))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # occasional Gaussian perturbation of the current best
        if best_seq is not None and rng.random() < 0.05:
            log_best = np.log(best_seq + 1e-30)
            pert = log_best + rng.normal(scale=0.2, size=best_seq.shape)
            pert_seq = np.exp(pert - np.max(pert))
            pert_seq /= pert_seq.sum()
            val_pert = _objective(pert_seq)
            if val_pert < best_val:
                best_val = val_pert
                best_seq = pert_seq.copy()
                print(f"[Perturb] n={n:5d} val={val_pert:.6f}")

        # early exit if we already beat the target comfortably
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 3 – Adam‑based mirror‑descent refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        # give the optimiser the remaining time (minus a tiny safety margin)
        refine_deadline = deadline - 0.1
        if time.time() < refine_deadline:
            seq_refined, val_refined = _refine_adam(best_seq, refine_deadline)
            if val_refined < best_val:
                best_val = val_refined
                best_seq = seq_refined
                print(f"[Overall‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
