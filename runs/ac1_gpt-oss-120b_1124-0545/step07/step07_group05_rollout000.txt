"""
Robust optimizer for the step‑function bound problem.

The objective (after normalising Σa = 1) is
        J(a) = 2·n·max(conv(a,a))

We minimise J by a multi‑stage global search:

1. **Rich seed pool** – many analytically‑motivated families (geometric,
   power‑law, Beta‑shapes with a broad range of parameters, classic windows,
   random Dirichlet / sparse spikes, mixtures of two Beta shapes, and the
   previous best solution if any).  All seeds are normalised to unit sum.

2. **Soft‑max Adam refinement** – each of the top‑k seeds is optimised with
   Adam on a smooth approximation of the max (log‑sum‑exp) while keeping the
   vector on the simplex via a soft‑max parametrisation.

3. **Exact sub‑gradient mirror descent** – the best Adam result is further
   refined with exponentiated‑gradient steps using the true sub‑gradient of
   `max(conv)`.

4. **High‑frequency simulated‑annealing** – a cheap incremental convolution
   update makes it possible to perform millions of moves.  Four move types
   are mixed: (a) Dirichlet re‑allocation on 2‑5 random indices,
   (b) targeted moves that touch indices contributing to the current maximal
       convolution entry, (c) swaps of two entries, and (d) small ε‑transfer
       of mass between two indices.  Temperature decays slowly to allow
   careful final polishing.

The routine respects a 1000 s wall‑clock budget, automatically falls back
to a shorter schedule if time runs low and finally returns the best
candidate as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full length) via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax (numerically stable)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """
    Objective for a vector a already normalised to sum=1:
        2 * n * max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families – all return a normalised (sum=1) ndarray
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_mixture(n: int, w: float,
                  alpha1: float, beta1: float,
                  alpha2: float, beta2: float) -> np.ndarray:
    a1 = _beta_shape(n, alpha1, beta1)
    a2 = _beta_shape(n, alpha2, beta2)
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hanning(n: int) -> np.ndarray:
    a = np.hanning(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming(n: int) -> np.ndarray:
    a = np.hamming(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman(n: int) -> np.ndarray:
    a = np.blackman(n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, sigma: float) -> np.ndarray:
    a = np.exp(-np.arange(n, dtype=np.float64) / sigma)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _spike_seed(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k equally‑weighted spikes on random positions."""
    if k >= n:
        a = np.full(n, 1.0 / n)
        return a
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


def _make_seeds(
    n: int,
    previous: np.ndarray | None,
    rng: np.random.Generator,
) -> list[np.ndarray]:
    """Collect a very diverse set of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_hanning(n))
    seeds.append(_hamming(n))
    seeds.append(_blackman(n))

    # geometric families (coarse and fine)
    for r in np.linspace(0.2, 0.9, 7):
        seeds.append(_geom(n, r))
    for r in np.linspace(0.90, 0.99, 7):
        seeds.append(_geom(n, r))

    # power‑law family
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # beta shapes – broad range of exponents
    alphas = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    betas = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # mixture of two beta shapes (randomised)
    for _ in range(12):
        w = rng.uniform(0.2, 0.8)
        a1 = rng.choice(alphas)
        b1 = rng.choice(betas)
        a2 = rng.choice(alphas)
        b2 = rng.choice(betas)
        seeds.append(_beta_mixture(n, w, a1, b1, a2, b2))

    # exponential decays (different sigmas)
    for sigma in (n / 30, n / 20, n / 10, n / 5, n / 2):
        seeds.append(_exp_decay(n, sigma))

    # random Dirichlet draws
    for conc in (0.1, 0.3, 0.7, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse / spike patterns
    for k in (max(3, n // 2000), max(5, n // 1000), max(8, n // 800)):
        for conc in (0.1, 0.3, 0.7):
            seeds.append(_random_sparse(n, k, conc, rng))
    # pure spikes (very sparse)
    for k in (max(2, n // 3000), max(4, n // 1500)):
        seeds.append(_spike_seed(n, k, rng))

    # reuse previous solution if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

def _adam_optimize(
    seed: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
    top_beta: float = 5.0,
    beta_mul: float = 1.015,
) -> tuple[np.ndarray | None, float]:
    """Adam on a log‑sum‑exp (soft‑max) approximation of max(conv)."""
    eps = 1e-30
    v = np.log(seed + eps)  # params for soft‑max (stay on simplex)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    beta = top_beta
    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)               # simplex‑projected vector
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it} best={cur_val:.6f}")

        # ------- smooth‑max weights (log‑sum‑exp) -------
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)
        grad_a = 2.0 * grad_tmp[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. v (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional gradient clipping
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 350000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror descent on max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _sa_incremental_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with incremental convolution updates.
    Four move types:
        1) Dirichlet re‑allocation on 2‑5 positions,
        2) Targeted move (indices contributing to current max entry),
        3) Swap of two entries,
        4) Small ε‑transfer between two entries.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # full convolution, length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    # temperature proportional to current objective
    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9997

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # ----- choose move type -----
        move_type = rng.random()
        if move_type < 0.45:
            # ---- Dirichlet on 2‑5 random indices ----
            k = rng.integers(2, 6)          # 2‑5 indices
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        elif move_type < 0.70:
            # ---- Targeted move ----
            k_star = int(np.argmax(cur_conv))
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            support = np.arange(i_start, i_end + 1, dtype=int)
            contrib = cur[support] * cur[k_star - support]
            w_sum = contrib.sum()
            if w_sum > 0:
                i = rng.choice(support, p=contrib / w_sum)
            else:
                i = rng.choice(support)
            # pick a second index uniformly (could be the same, we reject)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            idx = np.array([i, j], dtype=int)
            total = cur[idx].sum()
            new_alloc = rng.dirichlet(np.full(2, 1.0)) * total
        elif move_type < 0.85:
            # ---- Swap two entries ----
            i, j = rng.choice(n, size=2, replace=False)
            idx = np.array([i, j], dtype=int)
            new_alloc = cur[idx][::-1]      # swapped values
        else:
            # ---- ε‑transfer (tiny shift) ----
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] <= 0.0:
                continue
            eps_transfer = rng.random() * min(0.1, cur[i])  # move at most 10 %
            new_val_i = cur[i] - eps_transfer
            new_val_j = cur[j] + eps_transfer
            if new_val_i < 0.0:
                continue
            idx = np.array([i, j], dtype=int)
            new_alloc = np.array([new_val_i, new_val_j], dtype=np.float64)

        # ---- compute deltas ----
        deltas = new_alloc - cur[idx]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental convolution update -----
        conv_new = cur_conv.copy()
        # linear term 2 * cur * delta (shifted)
        for p, delta in zip(idx, deltas):
            if delta == 0.0:
                continue
            conv_new[p : p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta
        # cross‑terms delta_i * delta_j (i != j)
        for i_ in range(idx.size):
            for j_ in range(i_ + 1, idx.size):
                pos = idx[i_] + idx[j_]
                conv_new[pos] += 2.0 * deltas[i_] * deltas[j_]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < np.exp(-delta_obj / T):
            # accept move
            cur[idx] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[SA] n={n} it={it} best={new_obj:.6f}")

    return best_seq, best_obj

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # retrieve previous solution if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Choose a collection of lengths – we start from medium sizes
    # and optionally try a larger one if time permits.
    lengths = [2000, 4000, 6000, 8000]
    # add a longer length only when we have extra budget
    if deadline - time.time() > 200:
        lengths.append(10000)

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # Phase 0 – quick evaluation of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation on smooth max (top‑k seeds)
        # ------------------------------------------------------------------
        seed_vals = [(s, _objective(s)) for s in seeds]
        seed_vals.sort(key=lambda t: t[1])
        top_k = min(6, len(seed_vals))          # more seeds than previous version
        for s, _ in seed_vals[:top_k]:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline,
                                       max_iters=30000,
                                       top_beta=5.0,
                                       beta_mul=1.018)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=500000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – High‑frequency simulated‑annealing (dominant phase)
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            # allocate remaining time proportionally
            remaining = deadline - time.time()
            # Very rough estimate: ~0.00007 s per SA iteration for n≈4000
            # target up to 2 M moves if time allows
            target_iters = int(min(2_500_000, remaining * 14_000))
            seq, val = _sa_incremental_optimize(global_best_seq, n,
                                                deadline,
                                                init_temp_factor=0.03,
                                                max_iters=target_iters)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SA] n={n:5d} new global best={val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # -----------------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence (optional)
    # -----------------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
