beta‑shapes, edge‑blocks and symmetric sparse patterns), runs a multi‑stage
optimisation pipeline and respects the global 1000 s budget.

Pipeline (per length):
  1. Evaluate many seeds, keep the best few.
  2. Adam on a smooth‑max (log‑sum‑exp) objective (few thousand steps).
  3. Exact sub‑gradient mirror‑descent (hundreds of thousands steps).
  4. Targeted simulated‑annealing (millions of cheap incremental updates).
  5. A short final Adam + sub‑gradient polishing.
The best sequence over all explored lengths is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    c = np.fft.irfft(fa * fb, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _objective(a: np.ndarray) -> float:
    """Objective for a normalised probability vector (∑a = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_block(n: int, frac: float) -> np.ndarray:
    """Two equal blocks at both ends, each receiving half the total mass."""
    L = max(1, int(n * frac))
    if 2 * L > n:                     # blocks would overlap → fall back to uniform
        return _uniform(n)
    a = np.zeros(n, dtype=np.float64)
    a[:L] = 0.5 / L
    a[-L:] = 0.5 / L
    return a


def _mirror_geom(n: int, r: float) -> np.ndarray:
    """Geometric decay on the left half, mirrored on the right."""
    half = (n + 1) // 2
    left = r ** np.arange(half, dtype=np.float64)
    full = np.concatenate([left, left[: n - half][::-1]])
    s = full.sum()
    return full / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc))


def _random_sparse_sym(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric sparse vector with exactly k non‑zero pairs (or a centre element)."""
    a = np.zeros(n, dtype=np.float64)
    half = n // 2
    # choose distinct indices in the left half
    idx_left = rng.choice(half, size=min(k, half), replace=False)
    vals = rng.dirichlet(np.full(len(idx_left), conc))
    a[idx_left] = vals / 2.0
    a[n - 1 - idx_left] = vals / 2.0
    # if n odd, possibly add a centre mass
    if n % 2 == 1 and len(idx_left) < k:
        centre = half
        a[centre] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diversified collection of normalised seeds."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for r in np.linspace(0.15, 0.95, 9):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # moderate beta‑shape grid
    alphas = np.linspace(0.30, 1.20, 8)
    betas = np.linspace(0.30, 1.20, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # *extreme* beta‑shapes (more mass at the ends)
    alphas_ext = np.linspace(0.05, 0.40, 5)
    betas_ext = np.linspace(0.05, 0.40, 5)
    for a in alphas_ext:
        for b in betas_ext:
            seeds.append(_beta_shape(n, a, b))

    # edge‑block patterns
    for frac in (0.01, 0.02, 0.04, 0.08, 0.12, 0.16):
        seeds.append(_edge_block(n, frac))

    # mirrored geometric
    for r in np.linspace(0.20, 0.90, 7):
        seeds.append(_mirror_geom(n, r))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.15, 0.30, 0.60, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # symmetric sparse patterns
    for k in (2, 3, 5):
        for conc in (0.05, 0.15, 0.40):
            for _ in range(2):
                seeds.append(_random_sparse_sym(n, k, conc, rng))

    # reuse a previous solution if its length matches
    if previous is not None and previous.shape == (n,):
        prev = previous / previous.sum()
        seeds.append(prev)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitives (Adam, sub‑gradient, SA, final polish)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate.
    Works in log‑space to keep positivity and uses softmax to stay on simplex.
    """
    eps = 1e-30
    v = np.log(seed + eps)                     # log‑space parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.8

    # smoothing factor will grow geometrically
    beta_smooth = 2.0
    beta_mul = 1.01

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                     # a stays on simplex automatically
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ----- smooth‑max weighting -----
        beta_smooth *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w·conv -----
        rev_a = a[::-1]
        grad_tmp = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * grad_tmp[n - 1:n - 1 + n]   # keep centre n entries

        # ----- map gradient to log‑space (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # optional clipping
        max_norm = 10.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_star = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_star]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of max‑conv at the bottleneck index
        g = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k_star - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _targeted_sa_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          init_temp_factor: float = 0.02,
                          min_temp: float = 1e-9,
                          max_iters: int = 1500000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that always focuses on the current maximal convolution
    entry (the “bottleneck”). A small set of the heaviest contributors to that
    entry is re‑distributed each step.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_conv = _conv_fft(cur, cur)          # length 2n‑1
    cur_obj = 2.0 * n * np.max(cur_conv)

    best_seq = cur.copy()
    best_obj = cur_obj

    T = max(init_temp_factor * cur_obj, 1e-12)
    decay = 0.9995

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # cooling
        T = max(T * decay, min_temp)

        # bottleneck location
        k_star = int(np.argmax(cur_conv))

        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_star - idx_range
        contrib = cur[idx_range] * cur[opp]

        # pick a few heavy contributors (max 6)
        top_k = min(6, len(idx_range))
        heavy_pos = idx_range[np.argpartition(-contrib, top_k - 1)[:top_k]]

        # number of indices to modify (2‑4)
        k_move = rng.integers(2, 5)

        if len(heavy_pos) >= k_move:
            chosen = rng.choice(heavy_pos, size=k_move, replace=False)
        else:
            needed = k_move - len(heavy_pos)
            extra = rng.choice(n, size=needed, replace=False)
            chosen = np.concatenate([heavy_pos, extra])

        total = cur[chosen].sum()
        if total <= 0.0:
            continue

        # re‑allocate the mass among the chosen positions
        new_alloc = rng.dirichlet(np.full(k_move, 1.0)) * total
        deltas = new_alloc - cur[chosen]
        if np.allclose(deltas, 0.0):
            continue

        # ----- incremental update of the full convolution -----
        conv_new = cur_conv.copy()
        # a*Δ + Δ*a terms (old a)
        for p, delta in zip(chosen, deltas):
            if delta == 0.0:
                continue
            conv_new[p:p + n] += 2.0 * delta * cur
            conv_new[2 * p] += delta * delta

        # cross Δ_i * Δ_j (i≠j) appears twice
        for i in range(k_move):
            for j in range(i + 1, k_move):
                pos = chosen[i] + chosen[j]
                conv_new[pos] += 2.0 * deltas[i] * deltas[j]

        new_obj = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        delta_obj = new_obj - cur_obj
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            cur[chosen] = new_alloc
            cur_conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = cur.copy()
                print(f"[TargetSA] n={n:5d} it={it:8d} best={new_obj:.6f}")

    return best_seq, best_obj


def _final_polish(seq: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    One short Adam run with a very sharp smooth‑max, followed by a brief
    sub‑gradient refinement. This cleans up any remaining tiny irregularities.
    """
    # Adam with higher starting β and fewer iterations (fast)
    seq_adam, val_adam = _adam_optimize(seq, n, deadline, max_iters=8000)

    # sub‑gradient refinement (short)
    seq_sub, val_sub = _subgradient_refine(seq_adam if seq_adam is not None else seq,
                                            n, deadline, max_iters=80000)

    # Return whichever is better
    if val_adam < val_sub:
        return seq_adam, val_adam
    else:
        return seq_sub, val_sub

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``. The routine respects the time budget (≈1000 s) and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # Load a previously discovered construction if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # Lengths to explore – larger n offers more freedom but costs more time.
    # 4000 proved good; we also try 6000 and a very long 8000 for a final push.
    lengths = [4000, 6000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # ------------------------------------------------------------------
        # 1) Quick evaluation of raw seeds
        # ------------------------------------------------------------------
        seed_vals = [_objective(s) for s in seeds]
        for s, v in zip(seeds, seed_vals):
            if v < global_best_val:
                global_best_val = v
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={v:.6f}")

        # ------------------------------------------------------------------
        # 2) Adam optimisation on the most promising seeds (top‑7)
        # ------------------------------------------------------------------
        top_idx = np.argsort(seed_vals)[:7]
        top_seeds = [seeds[i] for i in top_idx]

        for s in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(s, n, deadline, max_iters=25000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑Adam] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 3) Exact sub‑gradient refinement (long run)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _subgradient_refine(global_best_seq, n, deadline,
                                           max_iters=600000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑SubGrad] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 4) Targeted simulated‑annealing (high‑frequency)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _targeted_sa_optimize(global_best_seq, n, deadline,
                                             max_iters=3000000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall‑TargetSA] n={n:5d} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # 5) Final polishing (Adam + sub‑gradient)
        # ------------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq, val = _final_polish(global_best_seq, n, deadline)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[final‑polish] n={n:5d} new global best={val:.6f}")

        # store the best of this length as possible future seed
        previous = global_best_seq

    # ----------------------------------------------------------------------
    # Fallback (should never be required)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq)
        print("[fallback] using uniform distribution")

    # Optional sanity‑check with the provided evaluator (ignores its return)
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
