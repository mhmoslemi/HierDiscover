```
def simplex_projection(v, target_sum):
    # v is array
    if target_sum <=0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, len(v)+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```
