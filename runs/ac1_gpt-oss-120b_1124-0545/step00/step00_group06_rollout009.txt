def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    # try multiple lengths in a loop
    while time.time() < deadline:
        n = np.random.randint(800, 3000)  # choose length
        # random init: Dirichlet distribution with many entries
        a = np.random.rand(n)
        # Scale to sum = sqrt(2n)
        target_sum = np.sqrt(2 * n)
        a = a / np.sum(a) * target_sum
        # Clip to 1000
        a = np.clip(a, 0, 1000)  # though after scaling likely <1000.
        # gradient descent loop
        t_end = time.time() + min(2.0, deadline - time.time())
        # we allocate e.g. each restart 1 sec for iterative improvement
        while time.time() < t_end:
            conv = np.convolve(a, a)
            max_val = conv.max()
            # Local objective = max_val (since sum fixed)
            if max_val < best_score:
                best_score = max_val
                best_seq = a.copy()
            # find index of maximum
            idx_max = np.flatnonzero(conv == max_val)
            k = np.random.choice(idx_max)  # random among ties
            # compute gradient w.r.t a
            grad = np.zeros_like(a)
            # For each i from max(0, k-(n-1)) to min(n-1, k)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # vectorized: j = k - i
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            grad[i_range] += a[j_range]
            grad[j_range] += a[i_range]  # double count when i!=j
            # note double counting; actually gradient = 2 * a[j] (since both terms)
            # for i == j (i == k/2) we added twice a_i, which matches derivative 2a_i.
            # we want to move opposite direction:
            direction = -grad
            # step size adapt
            norm = np.linalg.norm(direction)
            if norm == 0:
                # random jitter
                a += np.random.randn(n) * 0.01
                a = np.maximum(a, 0)
                a = a / np.sum(a) * target_sum
                continue
            step = 0.01 * target_sum / norm  # scaled step
            a_new = a + step * direction
            a_new = np.maximum(a_new, 0)
            # renormalize sum
            sum_a = a_new.sum()
            if sum_a == 0:
                a_new = np.random.rand(n)
                sum_a = a_new.sum()
            a_new = a_new / sum_a * target_sum
            a = a_new
        # after inner loop, adjust outer time.
    # After search, return best_seq or fallback uniform if none.
    if best_seq is None:
        # fallback uniform sequence
        n = 1000
        best_seq = [np.sqrt(2 * n) / n] * n
    # ensure not too large values: clip to range
    best_seq = [float(min(max(x,0.0),1000.0)) for x in best_seq]
    return best_seq

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    
    best_seq = None
    best_score = float('inf')
    
    # define list of lengths to try
    lengths = [1024, 1500, 2000, 2500, 3000]
    
    while time.time() < deadline:
        # pick n randomly from lengths
        n = np.random.choice(lengths)
        # init random seq
        a = np.random.rand(n).astype(np.float64)
        a = a / a.sum() * np.sqrt(2 * n)  # normalize sum
        # local optimization loop
        iter_deadline = time.time() + min(2.0, deadline - time.time())
        while time.time() < iter_deadline:
            # compute conv via FFT for speed
            # using real FFT
            size = 1 << ((2*n-1).bit_length())
            fa = np.fft.rfft(a, size)
            conv = np.fft.irfft(fa * fa, size)[:2*n-1]
            # evaluate
            cur_score = float(conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                # print intermediate progress
                print(f"New best score {best_score:.6f} at n={n}")
            # pick index of max conv
            max_inds = np.flatnonzero(conv == conv.max())
            k = np.random.choice(max_inds)
            # compute gradient for max index
            grad = np.zeros_like(a)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            # accumulate gradient
            grad[i_range] += a[j_range]
            grad[j_range] += a[i_range]
            # direction opposite gradient
            direction = -grad
            norm = np.linalg.norm(direction)
            if norm == 0:
                # jitter
                a += np.random.randn(n) * 1e-3
                a = np.clip(a, 0, None)
                a = a / a.sum() * np.sqrt(2 * n)
                continue
            # step size schedule
            step_size = 0.1 * np.sqrt(2 * n) / norm  # adjust factor
            a_new = a + step_size * direction
            # enforce non-negativity
            a_new = np.maximum(a_new, 0.0)
            # renormalize sum
            sum_a = a_new.sum()
            if sum_a == 0:

"""
Simple stochastic optimizer for the step‑function problem.

The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the ordinary discrete convolution.  The objective is
scale‑invariant, therefore the algorithm works with a fixed L1‑norm
(e.g. sum(a)=1).  The optimizer performs many short simulated‑annealing
runs on randomly chosen lengths (from a few dozens up to a few
thousands).  Each run starts from a random non‑negative vector, then
repeatedly transfers a small amount of mass between two coordinates.
A Metropolis accept/reject step is used with a linear cooling schedule.
The convolution maximum is evaluated with an FFT (O(n log n)) which is
fast enough for thousands of evaluations.

The best sequence found during the whole budget (default 1000 s) is
returned.  If the search does not improve over the trivial uniform
distribution, a known three‑element construction `[1, 0.5, 0.5]` (which
gives a score 1.875 < 2) is returned as a safe fallback.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def max_convolution(seq: np.ndarray) -> float:
    """
    Return the maximum entry of the discrete convolution of `seq` with itself.
    Uses an FFT for speed (O(n log n)).
    """
    n = seq.shape[0]
    # length of the linear convolution is 2*n-1; pad to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def objective(seq: np.ndarray) -> float:
    """
    Compute the objective that `evaluate_sequence` returns, but
    without the extra security checks.  The implementation follows
    the definition in the problem statement.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:                # protect against the same pathological case
        return float("inf")
    maxb = max_convolution(seq)
    return 2.0 * n * maxb / (s * s)


def random_transfer(x: np.ndarray, eps: float) -> tuple:
    """
    Propose a new vector by moving a small amount of mass from one
    coordinate to another while keeping the total L1‑norm unchanged.
    `eps` controls the typical step size (relative to the source value).
    Returns the new vector and the absolute change `delta`.
    """
    n = x.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        i, j = np.random.randint(0, n, size=2)

    # propose to move a fraction of x[i] to x[j]
    delta = eps * np.random.randn() * x[i]
    # clip to keep both entries non‑negative
    if delta > x[i]:
        delta = x[i]
    if delta < -x[j]:
        delta = -x[j]

    if delta == 0.0:
        return None, 0.0

    y = x.copy()
    y[i] -= delta
    y[j] += delta
    return y, delta


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and returns the best sequence found.

    The algorithm:
      * repeatedly pick a random length n (50 … 2000)
      * initialise a random probability vector (sum = 1)
      * run a short simulated‑annealing loop that repeatedly
        transfers a small amount of mass between two coordinates,
        accepting downhill moves always and uphill moves with a
        temperature‑dependent probability.
      * keep the best sequence over the whole budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # keep the globally best candidate
    best_seq = None
    best_val = float("inf")

    # a few lengths that work well in practice (try many restarts)
    possible_lengths = [50, 80, 120, 180, 280, 420, 640, 960, 1440, 2000]

    # safety fallback (known improvement over the uniform case)
    fallback = np.array([1.0, 0.5, 0.5], dtype=float)

    # Main search loop – many short independent runs
    while time.time() < deadline:
        n = int(np.random.choice(possible_lengths))

        # start from a random point on the simplex (sum = 1)
        a = np.random.rand(n).astype(float)
        a /= a.sum()                     # enforce sum = 1  (scale‑invariant)

        cur_val = objective(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[{time.time():.1f}] New global best: {best_val:.6f} (n={n})")

        # annealing schedule parameters
        start_temp = 1.0
        end_temp   = 1e-4
        max_iters  = 3000                # cheap runs, many restarts

        for it in range(max_iters):
            if time.time() >= deadline:
                break
            # linear cooling
            temp = start_temp + (end_temp - start_temp) * (it / max_iters)

            # propose a random transfer; eps controls typical step size
            eps = 0.2
            proposal, _ = random_transfer(a, eps)
            if proposal is None:
                continue

            new_val = objective(proposal)

            # Metropolis acceptance
            accept = False
            if new_val < cur_val:
                accept = True
            else:
                prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = proposal
                cur_val = new_val
                if cur_val < best_val:
                    best_val, best_seq = cur_val, a.copy()
                    print(f"[{time.time():.1f}] New global best: {best_val:.6f} (n={n})")

        # short random perturbation phase to diversify starts
        # (helps escaping possible plateaus)
        if time.time() < deadline:
            # perturb a few coordinates randomly and re‑normalise
            for _ in range(5):
                idx = np.random.randint(n)
                a[idx] = max(0.0, a[idx] + np.random.randn() * 0.02)
            a /= a.sum()
            cur_val = objective(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[{time.time():.1f}] New global best after perturb: {best_val:.6f} (n={n})")

    # -----------------------------------------------------------------
    # Return result
    # -----------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = fallback

    # The evaluation routine clamps values to [0,1000]; our numbers are tiny,
    # but we still enforce the bounds explicitly.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Convert to plain Python list of floats for the caller
    return best_seq.tolist()
