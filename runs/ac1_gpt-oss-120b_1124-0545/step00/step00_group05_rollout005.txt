so only the *shape* of the vector matters, not its absolute scale.

The algorithm searches the shape space with many short random restarts.
Each restart runs a few hundred iterations of a *soft‑max* gradient descent:

*   For a current sequence a the convolution c = a*a is computed.
*   A smooth approximation of   max(c)   is taken:
        w_k = exp(c_k / T) / Σ exp(c / T)          (soft‑max)
    with an annealing temperature T that is reduced every iteration.
*   The gradient of   f(a)=2·n·softmax(c)/ (Σa)²   is obtained analytically:
        ∂softmax/∂a_i = 2· Σ_j a_j·w_{i+j}
        ∂f/∂a_i   = 2·n·( (∂softmax/∂a_i)·S – 2·softmax ) / S³,
        where S = Σ a .
*   A small step along the negative gradient is taken, the result is
    projected onto the non‑negative orthant and renormalised to keep Σa
    unchanged (scale does not affect the score).

Occasionally a random perturbation is applied to escape shallow minima.
Throughout the search the best sequence (lowest score) is stored and
returned at the end of the allotted time.
"""

import time
import numpy as np

def _smooth_max(conv, T):
    """Soft‑max (log‑sum‑exp) used as a smooth approximation of max(conv)."""
    # shift to avoid overflow
    shift = conv.max()
    eps = 1e-12
    weights = np.exp((conv - shift) / max(T, eps))
    weights /= weights.sum()
    smooth = np.dot(weights, conv)            # ≈ max(conv) when T→0
    return smooth, weights


def _gradient(a, conv, smooth, weights):
    """
    Compute the gradient of the evaluation objective with respect to a,
    using the smooth max approximation.
    """
    n = len(a)
    S = a.sum()
    # grad_soft = 2 * Σ_j a_j * w_{i+j}
    # This is a cross‑correlation of 'weights' (length 2n‑1) with a.
    # np.correlate with mode='valid' yields exactly Σ_j a_j·w_{i+j}.
    grad_soft = 2.0 * np.correlate(weights, a, mode='valid')
    # derivative of the whole objective
    df = (2.0 * n) * (grad_soft * S - 2.0 * smooth) / (S ** 3)
    return df


def _gradient_step(a, learning_rate, temperature):
    """One gradient‑descent step with annealed temperature."""
    conv = np.convolve(a, a)
    smooth, weights = _smooth_max(conv, temperature)
    grad = _gradient(a, conv, smooth, weights)

    # step opposite to the gradient
    step = -learning_rate * grad
    a_new = a + step

    # projection onto non‑negative orthant
    a_new = np.maximum(a_new, 0.0)

    # renormalise to keep the L1‑norm fixed (scale does not matter)
    s = a_new.sum()
    if s > 0:
        a_new *= a.sum() / s
    else:
        # if everything collapsed to zero, fall back to the old vector
        a_new = a.copy()
    return a_new, smooth


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The function runs for at most *budget_s* seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # start with a few obvious baselines
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # (a) the previous best construction, if present
    if 'height_sequence_1' in globals():
        cand = list(height_sequence_1)
        # trim to a manageable size (very long vectors are expensive)
        if len(cand) > 2000:
            cand = cand[:2000]
        sc = evaluate_sequence(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f'Init from height_sequence_1 : score={best_score:.6f}')

    # (b) the optimal 2‑element pattern (2,1) gives 16/9 ≈ 1.7778
    cand2 = [2.0, 1.0]
    sc2 = evaluate_sequence(cand2)
    if sc2 < best_score:
        best_seq, best_score = cand2, sc2
        print(f'Init from 2‑elem pattern   : score={best_score:.6f}')

    # ------------------------------------------------------------------
    # main stochastic search
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) random restart: pick a length and an initial shape
        # --------------------------------------------------------------
        n = np.random.randint(12, 120)          # modest size → cheap convolutions
        # Dirichlet with concentration <1 creates sparse, heavy‑tailed vectors
        alpha = np.random.uniform(0.05, 0.8)
        a = np.random.dirichlet(np.full(n, alpha)).astype(float)

        # keep a reasonable L1 norm (scale does not matter, but avoids tiny numbers)
        a *= np.sqrt(2.0 * n) / a.sum() * np.random.uniform(0.5, 2.0)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_seq, best_score = a.tolist(), cur_score
            print(f'Random init better         : score={best_score:.6f}  n={n}')

        # --------------------------------------------------------------
        # 2) inner optimisation loop (soft‑max gradient descent)
        # --------------------------------------------------------------
        temperature = max(a.max(), 1.0)       # initial temperature
        learning_rate = 0.15
        iters = 0
        stagnation = 0

        while time.time() < deadline and iters < 300:
            iters += 1

            # gradient step
            a, smooth_val = _gradient_step(a, learning_rate, temperature)

            # anneal temperature (geometric cooling)
            temperature *= 0.97
            temperature = max(temperature, 1e-4)

            # occasionally re‑scale learning rate
            if iters % 50 == 0:
                learning_rate *= 0.9
                learning_rate = max(learning_rate, 0.01)

            # evaluate the true (non‑smooth) objective
            seq_candidate = a.tolist()
            try:
                new_score = evaluate_sequence(seq_candidate)
            except Exception:
                new_score = float('inf')

            if new_score < cur_score:
                cur_score = new_score
                stagnation = 0
                if new_score < best_score:
                    best_seq, best_score = seq_candidate, new_score
                    print(f'New global best           : score={best_score:.6f}  n={n}')
            else:
                stagnation += 1
                # if we are stuck, inject a small random jitter
                if stagnation > 30:
                    idx = np.random.randint(n)
                    a[idx] = max(0.0, a[idx] * (1.0 + np.random.randn() * 0.07))
                    stagnation = 0

        # --------------------------------------------------------------
        # 3) occasional large random perturbation to escape deep local minima
        # --------------------------------------------------------------
        if np.random.rand() < 0.15 and best_seq is not None:
            # take the best known vector, randomise a few entries and continue
            a = np.array(best_seq, copy=True)
            for _ in range(np.random.randint(1, 4)):
                i = np.random.randint(len(a))
                a[i] = max(0.0, a[i] * (1.0 + np.random.randn() * 0.2))
            # possibly change length (trim or pad with tiny values)
            if np.random.rand() < 0.5 and len(a) > 20:
                a = a[:len(a) - np.random.randint(1, 5)]
            else:
                # pad with a few tiny entries
                pad = np.random.rand(np.random.randint(1, 5)) * 0.01
                a = np.concatenate([a, pad])
            cur_score = evaluate_sequence(a.tolist())
            # let the inner loop improve this perturbed candidate
            # (the next outer iteration will handle it)

    # ------------------------------------------------------------------
    # final sanity check – make sure all entries are within allowed bounds
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f'Finished search – best score = {best_score:.6f}, length = {len(best_seq)}')
    return best_seq
