"""
Simple gradient‑descent optimizer for the step‑function inequality.

The objective after normalising a sequence a to 
    sum(a) = sqrt(2·n) 
is exactly  max_k (a * a)[k] .  

We minimise the variance of the convolution towards its average value  

    mu = sum(a)^2 / (2·n‑1)

by gradient descent on the squared‑error
    L(a) = Σ_k ( (a*a)[k] – mu )^2 .

For a given k the derivative of L is
    ∂L/∂a_i = 2 Σ_j a_j ( (a*a)[i+j] – mu )
which can be computed efficiently with two convolutions.
After each gradient step we project back onto the simplex 
{ a ≥ 0 , Σ a = sqrt(2·n) }.
The algorithm starts from the uniform vector, adds a tiny random
perturbation to break symmetry and then iterates until the time
budget expires.  The best sequence encountered (as measured by
evaluate_sequence) is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex with a given
# total mass `s`.  The implementation follows the classic O(n log n)
# algorithm (Wang & Carreira‑Perpiñán, 2013).
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex total must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the largest rho where u_i > (cssv_i / i)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Core optimisation loop (gradient descent on the L2‑error of the
# convolution).  It works for any length `n`.
# ------------------------------------------------------------
def _optimise_length(
    n: int,
    time_deadline: float,
    seed: int = 0,
    target_score: float = 1.51,
) -> np.ndarray:
    """Return a good sequence of length n found before time_deadline."""
    rng = np.random.default_rng(seed)

    # required total mass after normalisation
    target_sum = np.sqrt(2.0 * n)

    # start from the uniform vector and break symmetry slightly
    a = np.full(n, target_sum / n, dtype=np.float64)
    a += rng.normal(scale=target_sum * 1e-4, size=n)  # tiny Gaussian noise
    a = np.maximum(a, 0.0)
    a = _project_onto_simplex(a, target_sum)

    # constants used in the optimisation
    mu = target_sum ** 2 / (2 * n - 1)          # desired average of the convolution
    best_a = a.copy()
    # direct evaluation (no extra clipping – evaluate_sequence will handle it)
    best_score = 2.0 * n * np.max(np.convolve(best_a, best_a)) / (target_sum ** 2)

    # hyper‑parameters for the descent
    eta = 0.5            # initial step size
    decay = 0.9995       # geometric decay per iteration
    min_eta = 1e-9
    max_iters = 10_000   # safety upper bound

    it = 0
    while time.time() < time_deadline and it < max_iters:
        # 1) convolution of the current vector with itself
        conv = np.convolve(a, a)                     # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (target_sum ** 2)

        # keep the best observed vector
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[n={n}] iter {it:5d}  score={best_score:.6f}  eta={eta:.2e}")

            # early exit when we have already passed the target
            if best_score < target_score:
                break

        # 2) residual of the convolution w.r.t. the desired average
        resid = conv - mu                                 # length 2n‑1

        # 3) gradient of L = Σ (conv‑mu)^2 w.r.t a
        #    grad_i = 2 * Σ_j a_j * resid[i+j]  (windowed inner product)
        a_rev = a[::-1]
        grad_full = np.convolve(resid, a_rev)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]        # extract the n components

        # 4) gradient step with projection back onto the simplex
        a = a - eta * grad
        # projection enforces non‑negativity and the total sum constraint
        a = _project_onto_simplex(a, target_sum)

        # 5) schedule step size
        eta = max(eta * decay, min_eta)

        it += 1

    return best_a


# ------------------------------------------------------------
# Main entry point required by the competition harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights (list of floats) that
    attempts to minimise `evaluate_sequence`.  The routine runs a
    simplex‑projected gradient descent on the squared error of the
    convolution, starting from the uniform distribution and using a
    small random perturbation to escape the trivial stationary point.
    It may restart with different lengths if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a small safety margin

    # Try to reuse the length of the previously found best sequence
    # if the variable exists; otherwise pick a reasonable size.
    base_n = None
    if "height_sequence_1" in globals():
        try:
            base_n = int(len(globals()["height_sequence_1"]))
        except Exception:
            base_n = None

    # If no length is known fall back to a moderate size
    if base_n is None or base_n < 500:
        base_n = 2000

    # Store the overall best sequence seen across all attempts
    overall_best_seq = None
    overall_best_score = float("inf")

    # We will perform a few independent runs (different seeds / lengths)
    # until the time budget runs out.
    run_id = 0
    while time.time() < deadline:
        # occasionally vary the length a little to explore the effect of n
        n = base_n
        if run_id % 3 == 1:
            n = int(base_n * 0.9)
        elif run_id % 3 == 2:
            n = int(base_n * 1.1)

        # guarantee a minimum of 300 items (too small makes the bound weak)
        n = max(300, n)

        # allocate a slice of the remaining budget for this run
        run_deadline = min(deadline, time.time() + 30.0)   # at most 30 s per run

        seq = _optimise_length(
            n=n,
            time_deadline=run_deadline,
            seed=seed + run_id * 100,
            target_score=1.51,
        )
        # evaluate the candidate with the provided black‑box function
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            score = float("inf")

        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(
                f"*** New global best (run {run_id}) – n={n}, score={overall_best_score:.6f}"
            )

        # stop early if we have already reached the goal
        if overall_best_score < 1.51:
            break

        run_id += 1

    # Fallback – uniform sequence (should never be needed)
    if overall_best_seq is None:
        n = base_n
        uniform = [np.sqrt(2.0 * n) / n] * n
        return uniform

    # Return as plain Python list (the harness expects a list)
    return overall_best_seq.tolist()
