def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1  # leave 1 sec margin

    n = 600  # could be larger
    sum_target = np.sqrt(2 * n)

    # Initialize sequence random dirichlet
    seq = np.random.rand(n)
    seq = seq / np.sum(seq) * sum_target

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # annealing schedule
    T0 = 1.0
    while time.time() < deadline:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 1)
        T = T0 * (1 - frac)  # linear cooling

        # propose move
        i, j = np.random.randint(n, size=2)
        while i == j:
            j = np.random.randint(n)
        # amount to move
        delta = np.random.exponential(scale=0.1) * seq[i]  # maybe small
        # limit delta not exceed seq[i]
        delta = min(delta, seq[i])
        # apply move
        seq[i] -= delta
        seq[j] += delta

        # evaluate
        score = evaluate_sequence(seq.tolist())
        # accept if improvement or with probability exp(-(score-best_score)/T)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # optionally keep seq as new current
        else:
            # accept with probability
            if np.random.rand() < np.exp(-(score - best_score) / max(T, 1e-12)):
                # accept worse move
                # keep seq as is (score becomes new baseline)
                best_score = score  # we can treat current baseline not best
            else:
                # revert move
                seq[i] += delta
                seq[j] -= delta

        # occasionally apply smoothing to reduce peaks
        if np.random.rand() < 0.01:
            # smoothing: replace each element by weighted average with neighbors
            seq = 0.5*seq + 0.25*np.roll(seq,1) + 0.25*np.roll(seq,-1)
            # re-normalize sum exactly to target (as rounding may cause drift)
            seq = seq / np.sum(seq) * sum_target

    return best_seq.tolist()

    best_seq = None

    for init_type in ['random', 'uniform']:
        # initialize
        if init_type == 'uniform':
            n = 2000
            seq = np.full(n, np.sqrt(2*n)/n)  # uniform such that sum sqrt(2n); but each value = sum/n, indeed sum = sqrt(2n)
        else:
            n = np.random.randint(800, 2500)

            seq = raw / raw.sum() * np.sqrt(2*n)
        # local search
        seq = seq.copy()
        # ensure nonnegative
        seq = np.maximum(seq, 0)
        curr_score = evaluate_sequence(seq.tolist())
        # if improvement:
        if curr_score < best_score:
            best_score = curr_score
            best_seq = seq.copy()
        # hill climb
        while time.time() < deadline:
            # compute convolution and max
            conv = np.convolve(seq, seq)
            kmax = int(np.argmax(conv))
            # compute contributions
            # compute array of pairwise contributions:

The optimizer builds a random (or uniform) starting vector whose L¹‑norm is
sqrt(2·n) – the same normalisation used by the reference algorithm, because
after the normalisation the evaluation value equals the maximal entry of the
self‑convolution.  A simple stochastic hill‑climbing / simulated‑annealing loop
is then applied:

* The current convolution is examined and the index of its maximal coefficient
  (kmax) is found.
* For this shift we compute the individual pair contributions a[i]·a[kmax‑i].
  The pair that contributes most to the current maximum is weakened by moving

* The move is accepted if it improves the score; otherwise it may be accepted
  with a temperature‑controlled probability (simulated annealing) to escape
  local minima.
* Occasionally the whole vector is smoothed (averaging neighbours) and
  re‑normalised, and random perturbations are injected to keep the search

        target_norm = np.sqrt(2.0 * len(seq1))
        if np.sum(seq1) > 0:
            seq1 = seq1 / np.sum(seq1) * target_norm
        init_strategies.append(("previous", lambda _: seq1))

    # ------------------------------------------------------------------
    # Main optimisation loop (one pass per initialisation)
    # ------------------------------------------------------------------
    for name, maker in init_strategies:
        # --------------------------------------------------------------
        # initialise a candidate
        # --------------------------------------------------------------
        seq = maker(None)
        n = len(seq)
        target_sum = np.sqrt(2.0 * n)          # L1‑norm we keep invariant
        curr_score = _evaluate(seq)

        if curr_score < best_score:
            best_score, best_seq = curr_score, seq.copy()
            print(f"[{name}] initial score improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # Simulated‑annealing / hill‑climbing iterations
        # ------------------------------------------------------------------
        iter_cnt = 0
        while time.time() < deadline:
            iter_cnt += 1

            # ----------------------------------------------------------
            # 1) compute convolution and locate the current worst shift
            # ----------------------------------------------------------
            conv = np.convolve(seq, seq)
            kmax = int(np.argmax(conv))

            # ----------------------------------------------------------
            # 2) find the pair (i, j) that contributes most to that max
            # ----------------------------------------------------------
            i_arr = np.arange(n)
            j_arr = kmax - i_arr
            valid = (j_arr >= 0) & (j_arr < n)
            contrib = np.zeros_like(seq)
            contrib[valid] = seq[valid] * seq[j_arr[valid]]

            i_star = int(np.argmax(contrib))          # worst contributor
            # pick a destination index that currently holds little mass
            j_star = int(np.argmin(seq))

            # ----------------------------------------------------------
            # 3) propose a small transfer from i_star to j_star
            # ----------------------------------------------------------
            # step size adapts to the magnitude of the source entry
            max_transfer = min(seq[i_star] * 0.25, 0.02 * target_sum)
            if max_transfer <= 0:
                # nothing to move – give up on this iteration
                continue
            delta = np.random.rand() * max_transfer

            seq[i_star] -= delta
            seq[j_star] += delta

            # ----------------------------------------------------------
            # 4) evaluate the new vector
            # ----------------------------------------------------------
            new_score = _evaluate(seq)

            # ----------------------------------------------------------
            # 5) keep the move if it improves or with a small annealing prob.
            # ----------------------------------------------------------
            accept = False
            if new_score < curr_score:
                accept = True
            else:
                # temperature gently decays towards zero
                elapsed = time.time() - start_time
                frac = elapsed / (budget_s - 2.0)
                temperature = max(1e-4, (1.0 - frac) * 0.1)
                prob = np.exp(-(new_score - curr_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

            if accept:
                curr_score = new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, seq.copy()
                    print(
                        f"[{name}] iter {iter_cnt:6d} -> new best {best_score:.6f}"
                    )
            else:
                # revert the transfer
                seq[i_star] += delta
                seq[j_star] -= delta

            # ----------------------------------------------------------
            # 6) occasional smoothing to reduce sharp spikes
            # ----------------------------------------------------------
            if np.random.rand() < 0.005:
                seq = (
                    0.5 * seq
                    + 0.25 * np.roll(seq, 1)
                    + 0.25 * np.roll(seq, -1)
                )
                # re‑normalise exactly to the required L1 norm
                seq = seq / np.sum(seq) * target_sum
                curr_score = _evaluate(seq)

            # ----------------------------------------------------------
            # 7) occasional random jitter to escape deep plateaus
            # ----------------------------------------------------------
            if np.random.rand() < 0.001:
                jitter = np.random.randn(n) * 0.001 * target_sum
                seq = np.maximum(seq + jitter, 0.0)
                seq = seq / np.sum(seq) * target_sum
                curr_score = _evaluate(seq)

        # end of while‑time loop for this initial strategy
        print(f"[{name}] finished after {iter_cnt} iterations, best so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1000
        n_fallback = 1000
        best_seq = _initial_uniform(n_fallback)

    # final sanity: ensure non‑negative, capped at 1000 and summed correctly
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    # re‑normalise to the required sum (protect against tiny drift)
    target_sum = np.sqrt(2.0 * len(best_seq))
    best_seq = best_seq / np.sum(best_seq) * target_sum

    return best_seq.tolist()
