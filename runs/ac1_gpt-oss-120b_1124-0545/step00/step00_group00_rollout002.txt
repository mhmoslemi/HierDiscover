"""
Propose a step‑function (non‑negative heights) that minimizes
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two phases:

1.  **Flattening phase** – A multiplicative‑gradient descent that
    reduces the variance of the self‑convolution.
    The loss is  L = mean( (conv(a,a) - avg)^2 )  where
    avg = sum(a)^2 / (2*n-1).  The gradient of L w.r.t. a can be
    expressed as a convolution with the residuals and the current
    vector.  An exponentiated update
        a_i ← a_i * exp( -η * grad_i )
    keeps all entries non‑negative; the vector is re‑normalised to
    sum(a)=1 after each step.  The learning rate η is annealed.

2.  **Refinement phase** – A cheap simulated‑annealing / hill‑climbing
    that repeatedly transfers a small amount of mass between two
    positions.  Moves that improve the objective are always accepted,
    otherwise they are accepted with probability exp(-(Δ)/T) where T
    slowly decays.

Multiple random restarts with different lengths are executed while
the time budget permits.  The best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: loss and gradient (variance of the convolution)
# ----------------------------------------------------------------------
def _loss_and_grad(a: np.ndarray):
    """Return (loss, grad) for the variance loss described above."""
    n = a.shape[0]
    # Convolution a*a (full length 2n‑1)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    avg = sum_a * sum_a / (2 * n - 1)          # mean value of the convolution
    resid = conv - avg                         # same length 2n‑1

    # sliding window view: each row i contains resid[i : i+n]
    # shape (n, n)
    windows = np.lib.stride_tricks.sliding_window_view(resid, n)
    # gradient of  sum(resid**2)  w.r.t. a  (the factor 4 comes from the
    # derivative of the squared residuals and the symmetry of the convolution)
    grad = 4.0 * windows.dot(a)                # shape (n,)
    loss = np.mean(resid * resid)
    return loss, grad


# ----------------------------------------------------------------------
# Phase 1 – Multiplicative gradient descent
# ----------------------------------------------------------------------
def _flatten_until_deadline(a: np.ndarray, deadline: float,
                            max_iter: int = 3000,
                            eta_start: float = 0.5) -> np.ndarray:
    """
    Perform exponentiated gradient descent until the time deadline or
    max_iter is reached.  The vector a is kept on the probability simplex.
    """
    n = a.shape[0]
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a[:] = 1.0 / n
    a = a / a.sum()
    eta = eta_start

    for it in range(max_iter):
        if time.time() > deadline:
            break

        loss, grad = _loss_and_grad(a)

        # Exponentiated update (multiplicative weights)
        tmp = -eta * grad
        # shift to avoid overflow/underflow; adding a constant does not affect
        # the result after normalisation.
        m = np.max(tmp)
        factors = np.exp(tmp - m)
        a = a * factors
        a = a / a.sum()          # project back to the simplex

        # Simple schedule: decay η every 200 steps
        if (it + 1) % 200 == 0:
            eta *= 0.9

    return a

# ----------------------------------------------------------------------

    Randomly move small mass between two coordinates while accepting
    occasional uphill moves.  Stops when the time budget is exhausted.
    """

    cur = seq.copy()
    best = seq.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best_score = cur_score

        if i == j:

        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta

        # Numerical safety – clip tiny negatives (evaluate_sequence also does it)
        nxt = np.maximum(nxt, 0.0)

        try:
            nxt_score = evaluate_sequence(nxt.tolist())
        except Exception:
            continue

        accept = False
        if nxt_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(nxt_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur = nxt
            cur_score = nxt_score
            if nxt_score < best_score:
                best = nxt.copy()
                best_score = nxt_score

        step += 1
        # Temperature schedule
        if step % 500 == 0:
            T *= 0.95
            T = max(T, 1e-6)

        # occasional progress report
        if step % 2000 == 0:
            print(f"  hill‑climb step {step}, best score {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step function.

    The function runs several random restarts with different lengths,
    applies the flattening gradient descent and finishes with a short
    hill‑climbing phase.  The best sequence found before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # Possible lengths – a compromise between quality and speed
    candidate_lengths = [500, 800, 1100, 1500, 2000]

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Random restarts
    # ------------------------------------------------------------------
    for length in candidate_lengths:
        if time.time() > deadline:
            break

        # initialise – combine a few strategies
        init_vec = np.random.rand(length) + 0.1   # avoid exact zeros
        # also try the previous best if it exists
        try:
            prev = globals().get('height_sequence_1')
            if prev is not None and len(prev) == length:
                init_vec = np.array(prev, dtype=float) + 1e-6
        except Exception:
            pass

        init_vec = np.maximum(init_vec, 0.0)
        init_vec = init_vec / init_vec.sum()

        # Phase 1 – flattening
        a = _flatten_until_deadline(init_vec, deadline,
                                    max_iter=2500, eta_start=0.5)

        # quick evaluation
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            score = float('inf')

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{length}] New best after flattening: {best_score:.6f}")

        # Phase 2 – local refinement (only a few seconds per length)
        sub_deadline = min(deadline, time.time() + 3.0)   # limit per restart
        refined = _hill_climb(a, sub_deadline)

        try:
            refined_score = evaluate_sequence(refined.tolist())
        except Exception:
            refined_score = float('inf')

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[{length}] New best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final long hill‑climb on the current best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("Final refinement on the global best...")
        final_seq = _hill_climb(best_seq, deadline)
        try:
            final_score = evaluate_sequence(final_seq.tolist())
        except Exception:
            final_score = float('inf')
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"Final best score after extra climb: {best_score:.6f}")

    # safety fall‑back
    if best_seq is None:
        best_seq = np.ones(1, dtype=float)

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq.tolist()
