def project_simplex(v, S=1.0):
    # v is 1D np array
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n+1)
    # find rho = max i with u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all negative?? set all zeros
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function sequence that minimizes
evaluate_sequence() = 2·n·max(convolution(a,a)) / (sum(a))**2.

The algorithm:
1. Choose a length n (use the existing height_sequence_1 if present,
   otherwise a moderate size ~5000) and initialise a non‑uniform,
   non‑negative vector a with sum 1 (random Dirichlet).
2. Iteratively minimise the smooth approximation of the max‑convolution:
   * compute b = a * a (self‑convolution)
   * soft‑max the entries of b with a temperature α to obtain
     probabilities p_k ≈ 1_{k = argmax b_k}.
   * the gradient of the smooth max w.r.t. a is
       grad_i = 2·∑_k p_k·a_{k‑i},
     which can be obtained by a single convolution of a with the reversed
     probability vector.
   * take a gradient step, then project the new vector onto the simplex
     {x ≥ 0, ∑x = 1}.
3. Use a simple back‑tracking line search to guarantee a strict
   improvement of the true objective f = 2·n·max(b).  If no improvement
   is found for many iterations, inject a small random perturbation to
   escape possible plateaux.
4. Increase the soft‑max temperature α gradually so the optimisation
   focuses more on the actual maximum.
5. Keep track of the best sequence encountered.  The loop stops a few
   seconds before the global deadline, after which the best sequence is
   returned as a plain list of floats.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return the zero vector
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation score.  The function returns a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the problem size and the starting vector
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the surrounding environment
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.size
        a = init_seq.copy()
    except Exception:
        # fallback length – large enough to give a good theoretical bound
        n = 5000
        a = np.random.rand(n)

    # Ensure a is a probability vector (sum = 1, all non‑negative)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    # Occasionally randomise the start to break symmetry
    if np.random.rand() < 0.5:
        a = np.random.rand(n)
        a = a / a.sum()

    best_a = a.copy()
    conv = np.convolve(a, a)
    best_f = 2.0 * n * np.max(conv)    # because sum(a) == 1
    print(f"[init] n={n}, f={best_f:.6f}")

    # ------------------------------------------------------------------
    # 2. Optimisation parameters
    # ------------------------------------------------------------------
    lr = 0.5               # initial learning rate
    min_lr = 1e-7
    alpha = 1.0            # temperature for the soft‑max
    alpha_inc = 0.5
    max_alpha = 200.0

    iter_cnt = 0
    stagnation = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # ---- compute current convolution and soft‑max weights ----
        conv = np.convolve(a, a)
        max_b = conv.max()
        # soft‑max (numerically stable)
        exp_vals = np.exp(alpha * (conv - max_b))
        probs = exp_vals / exp_vals.sum()          # p_k ≥ 0, ∑p = 1

        # ---- gradient of the smooth max (2·∑_k p_k a_{k-i}) ----
        probs_rev = probs[::-1]
        grad_conv = np.convolve(a, probs_rev)      # length = n + (2n-1) - 1 = 3n-2
        L = probs.size                               # = 2n-1
        grad = 2.0 * grad_conv[(L - 1) - np.arange(n)]

        # ---- back‑tracking line search (guarantee improvement) ----
        step_factor = lr
        improved = False
        while step_factor > min_lr:
            a_cand = a - step_factor * grad
            a_cand = _project_simplex(a_cand, s=1.0)

            conv_cand = np.convolve(a_cand, a_cand)
            f_cand = 2.0 * n * np.max(conv_cand)

            if f_cand < best_f - 1e-12:
                # Accept the move
                best_f = f_cand
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            else:
                step_factor *= 0.5

        # ---- adapt learning rate and handle stagnation ----
        if improved:
            lr = min(lr * 1.02, 1.0)      # slowly raise the step size
            stagnation = 0
        else:
            lr = max(lr * 0.9, min_lr)
            stagnation += 1
            if stagnation % 200 == 0:
                # random jitter to escape a possible plateau
                jitter = np.random.randn(n) * 0.01
                a = _project_simplex(a + jitter, s=1.0)
                stagnation = 0

        # ---- raise the soft‑max temperature gradually ----
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha + alpha_inc)

        # ---- occasional progress report ----
        if iter_cnt % 100 == 0:
            print(f"[iter {iter_cnt:05d}] best_f={best_f:.6f}, lr={lr:.4e}, alpha={alpha:.2f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations={iter_cnt}, best_f={best_f:.6f}")
    return best_a.tolist()
