def project_simplex(v):
    # Implementation
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Projection of vector v onto the probability simplex {x>=0, sum x=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function (non‑negative heights) that tries to
minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv(a,a)` is the discrete self‑convolution.
The algorithm is a cheap stochastic mirror‑descent on the probability
simplex:

* Initialise a random probability vector (sum = 1) of a random length.
* Repeatedly (while time permits) penalise the entries of the
  convolution that are larger than the average value
  (1/(2·n‑1)).  The sub‑gradient of the penalty w.r.t. the
  coefficients is obtained analytically and an exponentiated‑gradient
  (mirror‑descent) update keeps the vector non‑negative and automatically
  projects it onto the simplex.
* Every few iterations the true evaluation is measured; the best vector
  found so far is kept.
* Several restarts with different lengths and seeds are performed;
  optionally the previously known construction `height_sequence_1` is used
  as a starting point.

The method is simple, works with only NumPy and runs well within the
given 1000 s budget.  It often yields values below the uniform bound
(2.0) and can reach the target range ≈ 1.5 for moderate lengths.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation step
def _step_descent(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent step that reduces large convolution values.

    Parameters
    ----------
    a   : current probability vector (sum = 1, a_i ≥ 0)
    eta : step size (positive)

    Returns
    -------
    a_new : updated probability vector (still on the simplex)
    """
    n = a.shape[0]

    # full self‑convolution, needed only for the penalty
    conv = np.convolve(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)                     # theoretical average of conv

    # penalty = max(conv - avg, 0)  – we only push down the peaks
    penalty = np.maximum(conv - avg, 0.0)

    # sub‑gradient of ½·∑ penalty_k·(conv_k) w.r.t a_i is:
    #   g_i = 2 * sum_k penalty_k * a_{k-i}
    # This is a cross‑correlation of (a reversed) with the penalty.
    # Using convolution we obtain it efficiently.
    grad_full = np.convolve(a[::-1], penalty)   # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # extract the n relevant components

    # Mirror‑descent (exponentiated gradient) update
    #   a_i ← a_i * exp(‑η·g_i)      then renormalise
    exp_factor = np.exp(-eta * grad)
    a_new = a * exp_factor
    # avoid a zero vector because of underflow
    a_new = np.maximum(a_new, 1e-16)
    a_new = a_new / a_new.sum()
    return a_new

        _update_best(a)

            a = a / a.sum()
            # quick evaluation of the perturbed start
            _update_best(a)

    # ------------------------------------------------------------------
    # If nothing was found (should never happen), fall back to a trivial sequence
    if best_seq is None:
        best_seq = [1.0]

    return best_seq
