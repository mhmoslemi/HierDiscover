def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = np.random.randint(200, 800)   # moderate length
    # initialize w as random positive numbers and normalize
    w = np.random.rand(n)
    w = w / w.sum()
    best_seq = w.copy()
    best_score = evaluate_sequence(w.tolist())
    # parameters
    beta = 0.5
    eta = 0.1
    iteration = 0
    while time.time() < deadline:
        conv = np.convolve(w, w)
        M = conv.max()
        score = 2 * n * M    # sum is 1
        if score < best_score:
            best_score = score
            best_seq = w.copy()
            # optional early stop if below target
            if best_score < 1.503:
                break
        # compute softmax weights p
        diff = conv - M
        # cap diff for stability
        # compute exp scaled
        p = np.exp(beta * diff)
        p_sum = p.sum()
        if p_sum == 0:
            p = np.ones_like(p) / len(p)
        else:
            p = p / p_sum
        p_rev = p[::-1]
        # compute gradient using convolution
        grad_full = np.convolve(w, p_rev)
        idx = 2 * n - 2 - np.arange(n)
        grad = 2 * grad_full[idx]
        # multiplicative update
        w = w * np.exp(- eta * grad)
        # projection onto simplex (normalize)
        w = np.maximum(w, 0)
        sum_w = w.sum()
        if sum_w == 0:
            w = np.random.rand(n)
            w = w / w.sum()
        else:
            w = w / sum_w
        # update beta, eta
        beta = min(beta * 1.001, 5.0)
        eta = max(eta * 0.9995, 1e-4)
        iteration += 1
        # occasional random perturb
        if iteration % 500 == 0:
            # add small noise
            w += np.random.randn(n) * 0.001
            w = np.clip(w, 0, None)
            w = w / w.sum()
    return best_seq.tolist()

"""
Propose candidate sequence for the convolution‐ratio minimisation.

The algorithm is a simple stochastic optimisation that works directly on the
probability simplex (the entries are forced to be non‑negative and to sum to 1,
so the scale does not affect the score).  It repeatedly :

* draws a random problem size n (small enough to keep evaluations cheap);
* initialises a random probability vector a;
* performs a short simulated‑annealing / hill‑climbing phase where a tiny
  amount of probability mass is moved from one coordinate to another;
* keeps the best vector seen (according to ``evaluate_sequence``).

The method uses only ``numpy`` and the supplied ``evaluate_sequence`` function,
runs for the allotted time budget and returns the best sequence it has found.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1).

    Standard O(n log n) algorithm (see Duchi et al., 2008).
    """
    if v.sum() == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    # Make a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical guard – fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Normalise explicitly to avoid tiny drift
    w /= w.sum()
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs until ``budget_s`` seconds have
    elapsed and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds.
    **kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best height sequence discovered.
    """
    np.random.seed(seed)
    # Small margin to guarantee clean exit
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float("inf")
    # --------------------------------------------------------------------
    # Main outer loop – many short random restarts.
    # --------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a moderate problem size (the score does not depend on n
        # apart from the factor 2·n, so any n gives comparable quality).
        n = np.random.randint(200, 800)

        # Random start on the simplex.
        cur = np.random.rand(n)
        cur = _project_to_simplex(cur)

        # Evaluate the starting point.
        cur_score = evaluate_sequence(cur.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.tolist()
            print(f"[Init] n={n}  score={best_score:.6f}")

        # Parameters for the inner simulated‑annealing phase.
        temperature = 1.0          # controls acceptance probability
        step_fraction = 0.1        # fraction of the source mass to move
        inner_deadline = time.time() + 0.12   # a few hundred ms per restart

        # ----------------------------------------------------------------
        # Short annealing run – move a small amount of probability mass.
        # ----------------------------------------------------------------
        while time.time() < inner_deadline:
            # Pick two distinct indices.
            i, j = np.random.choice(n, 2, replace=False)

            # Amount to transfer from i to j.
            delta = step_fraction * cur[i]
            if delta <= 0.0:
                continue

            # Propose the new vector.
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

            # Keep the vector on the simplex (numerical safety).
            cand = _project_to_simplex(cand)

            # Evaluate the candidate.
            cand_score = evaluate_sequence(cand.tolist())

            # Accept if it is better, otherwise accept with a Metropolis prob.
            improve = cand_score < cur_score
            if improve or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temperature, 1e-12)):
                cur, cur_score = cand, cand_score
                # Update global best.
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.tolist()
                    print(f"[Update] n={n}  score={best_score:.6f}")

                    # Early exit if we beat the target.
                    if best_score < 1.503:
                        return best_seq

            # Annealing schedule – slowly cool and shrink steps.
            temperature *= 0.9995
            step_fraction *= 0.9997

        # Occasionally shake the current vector to escape plateaus.
        if np.random.rand() < 0.15:
            cur += np.random.randn(n) * 0.001
            cur = np.maximum(cur, 0.0)
            cur = _project_to_simplex(cur)

    # If nothing was found (should not happen) return a trivial safe sequence.
    if best_seq is None:
        best_seq = [1.0]
    return best_seq
