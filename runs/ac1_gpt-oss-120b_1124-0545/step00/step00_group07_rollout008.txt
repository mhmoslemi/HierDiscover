def simplex_projection(v):
    """Project onto probability simplex."""
    v = np.array(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Projects a vector onto the probability simplex {x >=0, sum(x)=1}."""
    # shift to avoid negative large values
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zeros case
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time
import math

def simplex_projection(v):
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(x, y):
    """
    Linear convolution using FFT (real-valued input).
    Returns the full convolution (len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    # Find next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    z = np.fft.irfft(X * Y, N)[:n]
    return z

def _optimize_once(init_vec, deadline):
    """
    Perform a projected (soft‑max) gradient descent on a single start vector.
    Returns the best sequence found and its score.
    """
    a = np.asarray(init_vec, dtype=float)
    # force feasibility (probability simplex)
    a = simplex_projection(a)

    n = a.shape[0]

    # keep best local result
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # smoothing parameter for the soft‑max (starts relatively large)
    tau = 0.5
    # exponential decay schedule for tau
    tau_decay = 0.98
    min_tau = 0.02

    it = 0
    while time.time() < deadline:
        # -------------------  convolution a*a  -------------------
        conv_aa = convolve_fft(a, a)          # length 2n-1
        max_conv = conv_aa.max()

        # -------------------  soft‑max weights  ----------------
        # shift by the maximum to avoid overflow
        shifted = conv_aa - max_conv
        # exponentiate safely
        exp_shift = np.exp(shifted / tau)
        w = exp_shift / exp_shift.sum()       # length 2n-1, sums to 1

        # -------------------  gradient of the smoothed max  -----------------
        # gradient of each entry of conv_aa w.r.t a is 2 * a reversed shifted.
        # The gradient of the weighted sum is the weighted sum of those.
        a_rev = a[::-1]
        grad_full = convolve_fft(w, a_rev)    # length 3n-2
        grad = 2.0 * grad_full[n-1 : 2*n-1]    # keep the n components we need

        # -------------------  step (projected gradient)  -----------------
        # step size decays like 1/sqrt(it)
        step = 0.4 / math.sqrt(it + 1.0)
        a = simplex_projection(a - step * grad)

        # -------------------  bookkeeping  -----------------
        it += 1

        # gradually tighten the smoothing
        if it % 200 == 0 and tau > min_tau:
            tau = max(tau * tau_decay, min_tau)

        # evaluate the true objective every few iterations
        if it % 5 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] local best score {best_score:.6f} (iter {it})")

        # occasional random jitter to help escape plateaus
        if it % 250 == 0:
            # add a tiny Gaussian noise and re‑project
            noise = np.random.randn(n) * 0.005
            a = simplex_projection(a + noise)

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm uses
    projected gradient descent on the smooth approximation of the
    max‑convolution objective.  Several random restarts are performed
    within the allocated time budget; the best sequence seen is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – we must return before the global timeout
    deadline = start_time + budget_s - 1.0

    # Length of the candidate.  If the competition already supplies a
    # reference construction we inherit its length; otherwise we fall back
    # to a moderate size that still fits comfortably into memory.
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 2000
    n = base_len

    best_seq = None
    best_score = float('inf')
    restart = 0

    # We keep launching short optimisation runs until time runs out.
    while time.time() < deadline:
        restart += 1
        # Choose an initial vector.
        if restart == 1 and 'height_sequence_1' in globals():
            init = np.array(height_sequence_1, dtype=float)
        elif restart == 2:
            init = np.random.rand(n)
        else:
            # Dirichlet distribution guarantees positivity and a unit sum.
            init = np.random.dirichlet(np.ones(n))

        # Reserve a slice of the remaining budget for this run.
        # Each inner optimisation gets at most 30 seconds; if less time
        # is left we use whatever is available.
        inner_deadline = min(deadline, time.time() + 30.0)

        seq, score = _optimize_once(init, inner_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL best updated: {best_score:.6f} after restart {restart}")

    # Fallback in the unlikely case that nothing was produced.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ``evaluate_sequence`` is scale‑invariant, but it rejects sequences
    # whose total sum is below 0.01.  Our vectors live on the simplex,
    # therefore they are safe.
    return best_seq.tolist()

"""
Projected gradient descent on a smooth approximation of the max‑convolution
objective.

The evaluation function for a step‑function f = (a_0,…,a_{n‑1}) is

    score(f) = 2·n·max_k (f*f)[k] / (∑ a_i)² ,

where (f*f) is the ordinary linear convolution.
The ratio is invariant under scaling, therefore we restrict the search to the
probability simplex Σ a_i = 1, a_i ≥ 0.

For a fixed vector a, the quantity max_k (a*a)[k] is not smooth.  We replace
the max by a soft‑max:

    φ_τ(a) = τ·log Σ_k exp( (a*a)[k] / τ )    (τ > 0)

∇φ_τ(a) = 2·Σ_k w_k·a_{k−·}   with   w_k = exp((a*a)[k]/τ) / Σ_j exp((a*a)[j]/τ).

Thus a gradient step is

    a ← a – η·∇φ_τ(a) ,

followed by a Euclidean projection onto the simplex.  The smoothing
parameter τ is decreased gradually so that the algorithm first moves
according to a smooth landscape and later focuses on the true maximum.

The routine performs several short restarts (uniform, random Dirichlet,
linearly increasing/decreasing profiles and the supplied `height_sequence_1`
if any).  Each restart runs the projected gradient descent until the
allocated time slice expires; the best sequence seen overall is returned.

All calculations use FFT‑based convolutions (O(n log n)) to keep the per‑
iteration cost modest even for n ≈ 2000.
"""

import numpy as np
import time
import math

# ------------------------------------------------------------
# Helper: projection onto the probability simplex
# ------------------------------------------------------------
def simplex_projection(v):
    """Project a vector onto {x : x_i ≥ 0, Σ x_i = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: linear convolution via FFT (real inputs)
# ------------------------------------------------------------
def convolve_fft(x, y):
    """Linear convolution of real vectors x and y using the FFT."""
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    z = np.fft.irfft(X * Y, N)[:n]
    return z
