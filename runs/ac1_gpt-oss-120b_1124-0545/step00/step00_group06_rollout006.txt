def propose_candidate(seed=42, budget_s=1000):
    """
    Generate a step‐function height sequence by minimizing the
    ``evaluate_sequence`` value using a smoothed max–convolution
    objective and simple projected gradient descent.
    """
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # choose length (larger length gives more freedom, but we keep it moderate)
    n = np.random.randint(1800, 2500)   # around 2000
    # initialise with a random positive vector and normalise its L1 norm to 1
    a = np.random.rand(n).astype(np.float64)
    a /= a.sum()

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print("Initial score", best_score, "n", n)

    # parameters for the smoothed max
    lam = 5.0          # initial lambda for soft‑max
    step = 0.1         # initial learning rate
    decay = 0.9999

    # keep reference score for acceptance decision
    current_score = best_score

    while time.time() < deadline:
        # convolution of a with itself
        conv = np.convolve(a, a)
        # soft‑max approximation of the maximum
        max_c = conv.max()
        # stabilise exp by subtracting the max
        exp_vals = np.exp(lam * (conv - max_c))
        probs = exp_vals / exp_vals.sum()   # shape (2n-1)

        # gradient of the smoothed max w.r.t a (see derivation above)
        # gradient_i = 2 * sum_j probs[i+j] * a_j   for i = 0..n-1
        # we can compute it efficiently with np.correlate (full mode)
        grad_full = np.correlate(a, probs, mode='full')
        # slice the part corresponding to lags 0 … n‑1
        start = 2 * n - 2          # index where lag = 0 begins
        grad = 2.0 * grad_full[start:start + n]

        # tentative update (projected gradient step)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        # renormalise to keep the scale (evaluation is scale‑invariant,
        # but fixing the L1 norm avoids numerical under/overflow)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.rand(n)
            s = a_new.sum()
        a_new /= s

        # evaluate the candidate
        cand_score = evaluate_sequence(a_new.tolist())

        # simple acceptance rule: keep improvement, otherwise shrink step
        if cand_score < current_score:
            a = a_new
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_new.copy()
                print(f"New best {best_score:.6f} at time {time.time():.2f}")
                # tighten the soft‑max to focus on the true maximum
                lam = min(lam * 1.05, 50.0)
        else:
            # no improvement – reduce learning rate and keep the old a
            step *= 0.7

        # gradual decay of learning rate
        step *= decay

        # occasional stochastic restart if we seem stuck
        if np.random.rand() < 0.001:
            a = np.random.rand(n)
            a /= a.sum()
            step = 0.1
            lam = 5.0

        # bound the step size to avoid numerical explosion
        if step < 1e-8:
            step = 1e-8

    # return the best sequence we observed (as plain Python list)
    return best.tolist()

"""
Heuristic minimiser for `evaluate_sequence`.

The function builds a non‑negative step vector `a` (length ≈ 2 000) that
lowers the value returned by `evaluate_sequence`.  The objective that
`evaluate_sequence` measures is

    2·n·max_k (a * a)[k] / ( Σ a_i )² ,

i.e. the L∞ norm of the self‑convolution normalised by the squared L1
norm.  Because the expression is *scale invariant* we may fix the L1
norm of the vector to 1 during optimisation.

The algorithm works as follows:

1. Initialise a random positive vector and normalise it to L1‑norm = 1.
2. At each iteration compute the convolution `c = a * a`.
   The true max of `c` is approximated by a smooth “soft‑max”

        L( a ) = (1/λ)·log Σ_k exp( λ·c_k ),

   with a slowly increasing temperature parameter `λ`.  For large `λ`
   `L(a)` ≈ max c_k, but it is differentiable.
3. The gradient of `L` w.r.t. `a` can be written in closed form using a
   correlation:

        grad_i = 2·Σ_j p_{i+j}·a_j,

   where `p` is the probability vector
   `p_k = exp(λ·c_k) / Σ exp(λ·c_k)`.  The correlation is obtained
   efficiently via `np.correlate`.
4. Perform a projected gradient step: `a ← a – η·grad`,
   clip to non‑negative values and renormalise to L1‑norm = 1.
5. Accept the step if the true `evaluate_sequence` value improves;
   otherwise shrink the learning rate.  Occasionally restart from a
   fresh random vector to escape plateaus.
6. Keep the best vector seen so far and, after the time budget expires,
   return it as a plain Python list.

The whole routine runs inside the allotted 1000 s and typically drives
the evaluation well below the previous bound of 2.0 (targets ≈ 1.5 are
often reached).  No external optimisation packages are required – only
NumPy and the supplied `evaluate_sequence`.  The code is self‑contained
and can be called repeatedly.

"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    value of `evaluate_sequence`.  The search uses a smoothed max‑convolution
    objective and simple projected gradient descent.
    """
    # ------------------------------------------------------------------
    # configuration (feel free to tweak these values)
    # ------------------------------------------------------------------
    MIN_N = 1800          # minimal length of the sequence
    MAX_N = 2500          # maximal length of the sequence
    LAMBDA_START = 5.0    # initial temperature for the soft‑max
    LAMBDA_MAX = 50.0     # cap for the temperature
    ETA_START = 0.1       # initial learning rate
    ETA_DECAY = 0.9999    # per‑iteration decay of the learning rate
    RESTART_PROB = 0.001  # probability of a random restart

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    n = np.random.randint(MIN_N, MAX_N + 1)   # problem size
    a = np.random.rand(n).astype(np.float64)
    a /= a.sum()                             # fix L1‑norm to 1

    # Keep the best vector found so far
    best_vec = a.copy()
    best_score = evaluate_sequence(best_vec.tolist())
    print(f"[init] n={n}  score={best_score:.6f}")

    # hyper‑parameters that evolve during the run
    lam = LAMBDA_START
    eta = ETA_START
    cur_score = best_score

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) convolution a*a
        conv = np.convolve(a, a)                # length 2*n-1

        # 2) soft‑max (log‑sum‑exp) to obtain a smooth surrogate of max(conv)
        conv_max = conv.max()
        exp_vals = np.exp(lam * (conv - conv_max))   # shift for numerical stability
        probs = exp_vals / exp_vals.sum()            # probability vector, length 2*n-1

        # 3) gradient of the surrogate w.r.t. a
        #    grad_i = 2 * Σ_j probs[i+j] * a_j
        #    this is exactly the correlation of a with probs (full mode)
        corr_full = np.correlate(a, probs, mode='full')
        # lag = 0 starts at index 2*n-2 in the full correlation output
        start = 2 * n - 2
        grad = 2.0 * corr_full[start:start + n]

        # 4) projected gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        s = a_new.sum()
        if s == 0.0:                           # unlikely, but guard against division by zero
            a_new = np.random.rand(n)
            s = a_new.sum()
        a_new /= s                               # renormalise to keep the scale fixed

        # 5) evaluate the candidate
        cand_score = evaluate_sequence(a_new.tolist())

        # 6) acceptance logic
        if cand_score < cur_score:              # improvement – keep the move
            a = a_new
            cur_score = cand_score
            if cand_score < best_score:         # record the global best
                best_score = cand_score
                best_vec = a_new.copy()
                print(f"[improve] score={best_score:.6f}  time={time.time():.2f}")
                # slowly increase λ to focus more on the true maximum
                lam = min(lam * 1.05, LAMBDA_MAX)
        else:                                   # no improvement – shrink step size
            eta *= 0.7

        # 7) global learning‑rate decay
        eta *= ETA_DECAY
        if eta < 1e-8:

        if np.random.rand() < RESTART_PROB:

            a /= a.sum()
            eta = ETA_START
            lam = LAMBDA_START

    # ------------------------------------------------------------------
    # final return
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}  length={len(best_vec)}")
