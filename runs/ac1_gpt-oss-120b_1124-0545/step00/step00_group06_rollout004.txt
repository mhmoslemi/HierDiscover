"""
Propose a step‑function (non‑negative sequence) that reduces the evaluation
score 2 * n * max(conv(a,a)) / (sum(a)^2).  The score is scale‑invariant, so
the algorithm works with a probability distribution (sum = 1).

The method is a soft‑max gradient descent on the full convolution:
    loss(a) = (1/α) * log Σ_k exp(α * (a*a)[k])
which smoothly approximates the maximum of the convolution.  Its gradient can
be written as a convolution of the current distribution with the soft‑max
weights, allowing an O(n log n) update using FFTs.

A small random jitter is added each iteration to avoid the uniform fixed‑point,
and the vector is projected onto the probability simplex and symmetrised.
The algorithm runs until the time budget expires, keeping the best sequence
found (as measured by the provided evaluate_sequence function) and then
performs a short hill‑climbing refinement.

The routine is deterministic given the seed and works for any length n
(up to a few thousand) within the 1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: linear convolution via FFT (real inputs, full mode)
# ------------------------------------------------------------
def linear_convolution(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear convolution of two real vectors x and y using FFT.
    The result has length len(x) + len(y) - 1.
    fft_len must be a power of two ≥ len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)
    return conv[: len(x) + len(y) - 1]


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The algorithm uses soft‑max gradient descent on the
    self‑convolution, with periodic random jitter and a final local refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2  # a tiny safety margin

    # ---- problem size -------------------------------------------------
    # Length can be tuned; 1500‑2000 gives a good trade‑off between quality
    # and speed (FFT size stays modest).
    n = 1800

    # ---- initialise a random probability vector -----------------------
    a = np.random.rand(n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    # enforce mirror symmetry (helps keep the convolution centred)
    a = (a + a[::-1]) * 0.5

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())

    # ---- FFT length (covers both a*a and a_rev * weights) -------------
    max_len = (n + (2 * n - 1) - 1)          # length of a_rev * weights = 3n-2
    fft_len = 1 << (max_len - 1).bit_length()   # next power of two

    # ---- hyper‑parameters ---------------------------------------------
    alpha = 1.0               # start with a smooth max
    alpha_max = 30.0
    alpha_inc = (alpha_max - alpha) / 500.0   # linear schedule
    lr = 0.12                 # learning rate for the gradient step
    jitter_scale = 1e-3       # relative jitter amplitude
    iter_cnt = 0
    no_improve = 0

    # ------------------------------------------------------------
    # Main optimisation loop (soft‑max gradient descent)
    # ------------------------------------------------------------
    while time.time() < deadline:
        # 1) full self‑convolution a*a
        conv = linear_convolution(a, a, fft_len)          # length 2n‑1

        # 2) soft‑max weights (scale‑invariant, approximates arg‑max)
        max_c = conv.max()
        # numerical stability: shift by the max before exponentiation
        shifted = conv - max_c
        exp_term = np.exp(alpha * shifted)
        w = exp_term / exp_term.sum()                     # Σ w = 1

        # 3) gradient of the soft‑max loss:
        #    ∂loss/∂a_i = 2 * Σ_k w_k * a_{k-i}
        #    This is a convolution of a_rev with w.
        grad_full = linear_convolution(a[::-1], w, fft_len)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]             # keep n entries

        # 4) gradient step + tiny random jitter
        a -= lr * grad
        jitter = jitter_scale * (np.random.rand(n) - 0.5) * a
        a += jitter

        # 5) projection onto the non‑negative simplex
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:                     # safety fallback
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # 6) re‑impose symmetry (optional but stabilises the centre)
        a = (a + a[::-1]) * 0.5

        iter_cnt += 1

        # ----- schedule updates ---------------------------------------
        if iter_cnt % 50 == 0 and alpha < alpha_max:
            alpha = min(alpha + alpha_inc, alpha_max)
        if iter_cnt % 120 == 0:
            lr *= 0.94                     # slowly decay learning rate

        # ----- occasional evaluation ----------------------------------
        if iter_cnt % 20 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"Iter {iter_cnt:5d} | α={alpha:5.2f} | lr={lr:6.4f} | score={cur_score:.6f}")
            else:
                no_improve += 1

        # ----- re‑initialise if we are stuck for a long time ----------
        if no_improve > 350:
            a = np.random.rand(n)
            a = a / a.sum()
            a = (a + a[::-1]) * 0.5
            no_improve = 0
            print(f"Re‑initialised at iteration {iter_cnt}")

    # ------------------------------------------------------------
    # Local hill‑climbing refinement (small mass swaps)
    # ------------------------------------------------------------
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # move a tiny fraction of mass from i to j
        delta = min(best_seq[i], 0.005 * best_seq[i])
        if delta <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()
        new_seq = (new_seq + new_seq[::-1]) * 0.5

        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            print(f"Hill‑climb improvement: score={new_score:.6f}")

    # ------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------
    return best_seq.tolist()
