def gradient_step(a, b, lr):
    k = np.argmax(b)
    n = len(a)
    g = np.zeros_like(a)
    # indices where i in [max(0, k-(n-1)), min(n-1, k)] correspond to a_{k-i}
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    # For i in this range:
    # a_{k-i} = a[ k - i]
    # g[i] = 2 * a[k - i]
    # else g[i] = 0
    idx_i = np.arange(i_start, i_end + 1)
    g[idx_i] = 2.0 * a[k - idx_i]
    # update
    a_new = a - lr * g
    a_new = np.maximum(0, a_new)
    # renormalize sum
    a_new = normalize_sum(a_new)
    b_new = np.convolve(a_new, a_new)
    return a_new, b_new

"""
Propose a step‑function sequence that minimises the evaluation function.

The algorithm is a simple simulated‑annealing / gradient descent hybrid:

* Start from either the provided `height_sequence_1` (if available) or a
  uniform sequence, normalised so that  sum(a)=√(2 n) .
* The objective after normalisation is simply  max(conv(a,a)) .
  (The original evaluation returns 2 n·max/ sum² = max  when
   the sum constraint holds.)

* Two kinds of moves are tried repeatedly until the time budget expires
  – a cheap “mass‑transfer” move that swaps a small amount of weight
  between two entries, and a “peak‑gradient” move that takes a step
  opposite the gradient of the current maximal convolution coefficient.
* Moves are accepted if they improve the maximal convolution value,
  otherwise with a small Metropolis probability that slowly cools.
* Every few thousand iterations a light smoothing (3‑point moving‑average)
  is attempted; if it lowers the objective it is kept.
* The best sequence seen so far (lowest max‑convolution) is stored and
  finally returned as a plain Python list.

The whole routine runs inside the given time‑budget (default 1000 s) and
conforms to the required API (`propose_candidate`).  No file or network
access is performed.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _normalize_sum(a: np.ndarray) -> np.ndarray:
    """
    Scale `a` so that sum(a) = sqrt(2*len(a)).
    The scaling does not change the evaluation value (except for the
    clipping inside evaluate_sequence, which is never hit for the
    magnitudes used here).
    """
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, target / n)
    else:
        a = a * (target / s)
    return a


def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """Return a normalised start sequence (uniform, or from the global
    `height_sequence_1` if it exists and has the right length)."""
    np.random.seed(seed)
    # try to use the existing construction if it matches the length
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        seq = np.asarray(global_seq, dtype=float)
        if seq.shape[0] == n:
            return _normalize_sum(seq)
    # otherwise a uniform start
    a = np.full(n, 1.0, dtype=float)
    return _normalize_sum(a)


def _apply_delta(a: np.ndarray, b: np.ndarray, idx: int, delta: float) -> None:
    """
    Apply an increment `delta` to a[idx] (i.e. a[idx] += delta) and update
    the convolution vector `b` in‑place using the exact O(n) formula:

        b[k] += 2*delta*a[k-idx]   for idx ≤ k < idx+n
        b[2*idx] += delta**2
    The array `a` must be the *old* version (before the change) when the
    slice update is performed.
    """
    n = a.shape[0]
    start = idx
    end = idx + n
    # vectorised update of the overlapping part
    b[start:end] += 2.0 * delta * a
    # central term gets the extra quadratic contribution
    b[2 * idx] += delta * delta
    # now really update a
    a[idx] += delta


def _mass_transfer_move(a: np.ndarray, b: np.ndarray, max_step_frac: float = 0.2):
    """
    Perform a single mass‑transfer move:
        a[i] <- a[i] - d
        a[j] <- a[j] + d
    where the feasible interval for d is
        -a[j] ≤ d ≤ a[i] .
    The move respects the sum constraint exactly.
    Returns the new maximum convolution after the tentative move and the
    indices / delta used (or None if no feasible move could be generated).
    """
    n = a.shape[0]
    # bias the choice of i (centre) and j (edges) – this usually helps
    centre_radius = int(0.2 * n)
    edge_width = int(0.1 * n)

    # choose i near centre with probability 0.6
    if np.random.rand() < 0.6:
        i = np.random.randint(n // 2 - centre_radius, n // 2 + centre_radius + 1)
    else:
        i = np.random.randint(n)

    # choose j near an edge with probability 0.6
    if np.random.rand() < 0.6:
        if np.random.rand() < 0.5:
            j = np.random.randint(0, edge_width + 1)
        else:
            j = np.random.randint(n - edge_width, n)
    else:
        j = np.random.randint(n)

    if i == j:
        return None

    # feasible interval for the transferred amount d
    d_low = -a[j]               # cannot make a[j] negative
    d_high = a[i]               # cannot make a[i] negative
    # restrict the step size to a modest fraction of the smaller bucket
    max_step = max_step_frac * min(a[i], a[j])
    d_low = max(d_low, -max_step)
    d_high = min(d_high, max_step)

    if d_low >= d_high:
        return None

    d = np.random.uniform(d_low, d_high)

    # make a temporary copy of b for the trial
    b_trial = b.copy()
    # apply the two opposite deltas (order matters because the second uses the updated a)
    _apply_delta(a, b_trial, i, -d)   # a[i] -= d
    _apply_delta(a, b_trial, j, d)    # a[j] += d

    new_max = b_trial.max()
    # revert the in‑place changes to a (they will be reapplied only if we accept)
    a[i] += d          # undo -d
    a[j] -= d          # undo +d
    return (new_max, i, j, d, b_trial)


def _gradient_peak_step(a: np.ndarray, lr: float = 0.05):
    """
    Take a gradient descent step targeting the current maximal convolution
    coefficient.  The gradient of the peak b[k*] w.r.t. a is
        g[i] = 2 * a[k* - i]   if 0 ≤ k* - i < n, else 0 .
    The step is: a <- a - lr * g, projected onto the non‑negative orthant
    and finally re‑normalised to keep the sum fixed.
    Returns the new sequence and its maximal convolution.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_star = int(b.argmax())

    # build the gradient for the peak index
    g = np.zeros_like(a)
    i_start = max(0, k_star - (n - 1))
    i_end = min(n - 1, k_star)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k_star - idx]

    a_new = a - lr * g
    a_new = np.maximum(0.0, a_new)
    a_new = _normalize_sum(a_new)
    b_new = np.convolve(a_new, a_new)
    return a_new, b_new.max()


def _smooth_sequence(a: np.ndarray) -> (np.ndarray, float):
    """Apply a short 3‑point smoothing filter and return the smoothed
    sequence together with its maximal convolution."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    a_s = np.convolve(a, kernel, mode="same")
    a_s = np.maximum(0.0, a_s)
    a_s = _normalize_sum(a_s)
    b_s = np.convolve(a_s, a_s)
    return a_s, b_s.max()


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a problem size.  Use the length of the previously known
    # construction if it exists, otherwise fall back to ~2500.
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        n = len(global_seq)
    else:
        n = 2500

    # ------------------------------------------------------------------
    # Initialise the state
    # ------------------------------------------------------------------
    a = _initial_sequence(seed, n)                     # current sequence
    b = np.convolve(a, a)                              # current conv
    cur_max = b.max()                                  # current objective (max conv)
    best_a = a.copy()
    best_max = cur_max

    # parameters for the annealing schedule
    start_temp = 0.5          # initial temperature (controls uphill jumps)
    end_temp = 1e-4          # final temperature
    iteration = 0
    last_print = start_time

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 0.2)
        # linear cooling schedule
        temp = (1.0 - frac) * start_temp + frac * end_temp

        # Occasionally take a gradient step toward the peak
        if np.random.rand() < 0.08:          # ~8 % of the iterations
            lr = 0.02 * (np.sqrt(2.0 * n) / n)   # small learning rate
            a_candidate, new_max = _gradient_peak_step(a, lr=lr)
            # accept if it improves, otherwise keep old
            if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / (temp + 1e-12)):
                a = a_candidate
                b = np.convolve(a, a)
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
        else:
            # standard mass‑transfer move
            move = _mass_transfer_move(a, b)
            if move is None:
                continue
            new_max, i, j, d, b_trial = move

            # Metropolis acceptance
            if new_max < cur_max or np.random.rand() < np.exp(-(new_max - cur_max) / (temp + 1e-12)):
                # accept the move – apply it for real
                _apply_delta(a, b, i, -d)          # a[i] -= d
                _apply_delta(a, b, j, d)           # a[j] += d
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                # occasionally re‑normalise to fight floating point drift
                if iteration % 5000 == 0:
                    a = _normalize_sum(a)
            # else: reject – a and b stay unchanged

        # every few thousand iterations try a smoothing operation
        if iteration % 3000 == 0 and iteration > 0:
            a_s, max_s = _smooth_sequence(a)
            if max_s < cur_max:
                a = a_s
                b = np.convolve(a, a)
                cur_max = max_s
                if max_s < best_max:
                    best_max = max_s
                    best_a = a.copy()

        # occasional progress print (once per second)
        if time.time() - last_print > 1.0:
            print(
                f"[{time.time() - start_time:6.1f}s] "
                f"cur_max={cur_max: .6f}  best_max={best_max: .6f}  n={n}"
            )
            last_print = time.time()

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence seen
    # ------------------------------------------------------------------
    best_list = best_a.tolist()
    # sanity check (optional, can be removed if speed is critical)
    try:
        import numpy as _np
        from math import isfinite as _isfinite

        # evaluate_sequence is assumed to be already imported in the environment
        score = evaluate_sequence(best_list)
        print(f"Final evaluated score = {score:.6f}")
    except Exception:
        pass

    return best_list
