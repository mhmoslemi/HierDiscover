def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    # Build convolution matrix A of shape (2*n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For each i, j = k-i
        # Set A[k, j] = f_seq[i]
        # We'll iterate i
        for i in range(i_start, i_end+1):
            j = k - i
            A[k, j] = f_seq[i]
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # Use high-performance method 'highs')
    try:
        result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 1.0})
    except Exception:
        return None
    if result.success:
        return result.x
    return None

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a combination of:
* a deterministic linear‑programming direction (maximise the L1‑norm of a
  vector g under the linear constraints f * g ≤ M, where M is the current
  maximum of f * f);
* a simple line‑search blending f and the normalised g;
* occasional random pairwise mass transfers (hill‑climbing) to escape
  stagnation.

The sequence length stays moderate (~2‑3k) so that the full convolution
(and the LP with ~5 000 constraints) fits easily into the 1000 s time
budget.  The LP is solved with SciPy’s HiGHS backend, which is fast and
has a built‑in time limit.

The function returns the best sequence found before the deadline.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """Return max of the full discrete convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ g) = conv(f, g) (full).

    For each k = 0 … 2n‑2 and each j = 0 … n‑1:
        A[k, j] = f[k‑j] if 0 ≤ k‑j < n else 0
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=f.dtype)
    # column j contributes f[i] to row i + j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_direction(f: np.ndarray, M: float, time_limit: float = 1.0):
    """
    Solve the LP:
        maximise   sum(g_i)
        subject to conv(f, g) ≤ M   (element‑wise)
                  g_i ≥ 0

    Returns the optimal g (as a 1‑D NumPy array) or None on failure.
    """
    n = f.size
    A = _build_conv_matrix(f)                 # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=f.dtype)  # RHS for every convolution entry
    c = -np.ones(n, dtype=f.dtype)            # maximise sum  → minimise -sum
    bounds = [(0, None)] * n

    # HiGHS is the default solver for recent SciPy versions.
    # We give it a modest per‑LP time budget so the outer loop stays fast.
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def _normalise_to_target_sum(seq: np.ndarray) -> np.ndarray:
    """
    Scale the sequence so that its L1‑norm equals sqrt(2·len(seq)).
    The scaling does not change the objective value (it depends only on
    the ratio max(conv) / sum²), but keeping the sum fixed simplifies the
    line‑search step.
    """
    target = np.sqrt(2.0 * seq.size)
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target / s)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (expects list)."""
    # The function evaluate_sequence is assumed to be already imported in the
    # execution environment.
    return evaluate_sequence(seq.tolist())  # type: ignore[name-defined]

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (floats) that gives a low value of
    the evaluation function.  The search runs for at most ``budget_s`` seconds.

    The algorithm mixes a linear‑programming direction (maximise L1 under a
    convolution bound) with simple stochastic hill‑climbing steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Initialise: try to reuse an existing construction if available.
    # ------------------------------------------------------------------
    try:

        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Ensure the sequence respects the problem limits
        init = np.clip(init, 0.0, 1000.0)
        if init.size < 2:
            raise ValueError
        print(f"Initialising from existing height_sequence_1 (len={init.size})")
    except Exception:
        # Uniform start – length similar to the reference construction.
        n_start = 2500
        init = np.full(n_start, np.sqrt(2.0 * n_start) / n_start, dtype=float)
        print(f"Initialising from uniform sequence (len={n_start})")

    # Normalise to the standard target sum sqrt(2·n) – this makes the
    # evaluation equal to the maximal convolution value.
    cur = _normalise_to_target_sum(init.copy())
    best_seq = cur.copy()
    best_score = _evaluate(best_seq)
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = cur.size

        else:

            #    Move a small amount of mass from one index to another.
            # ----------------------------------------------------------------
            i, j = np.random.choice(n, size=2, replace=False)
            if cur[i] > 0:
                # Move a fraction (up to 10 %) of the source entry.

                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                # Re‑normalise to keep the L1‑norm fixed.
                new_seq = _normalise_to_target_sum(new_seq)
                score_new = _evaluate(new_seq)
                if score_new < best_score:
                    best_score = score_new
                    best_seq = new_seq.copy()
                    cur = new_seq.copy()
                    print(f"[{iteration}] Random transfer i→j improved score to {best_score:.6f}")

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        if iteration % 250 == 0 and np.random.rand() < 0.3:

            cur = np.full(n_new, np.sqrt(2.0 * n_new) / n_new, dtype=float)
            cur = _normalise_to_target_sum(cur)
            print(f"[{iteration}] Random restart with length {n_new}")

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
