def project_onto_simplex(v, s=1):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_gradient_smooth(a, alpha):
    n = len(a)
    # compute convolution
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    # compute p = softmax(alpha*conv)
    # To avoid overflow, subtract max
    maxc = np.max(conv)
    exp_vals = np.exp(alpha * (conv - maxc))
    Z = np.sum(exp_vals)
    p = exp_vals / Z
    # gradient: 2 * convolution of a_rev and p
    a_rev = a[::-1]
    # compute convolution of a_rev and p
    # Need length L2 for convolution
    L2 = 1 << ((len(a_rev) + len(p) - 1).bit_length())

import numpy as np
import time

def project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x | x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def convolve_fft(x, y):
    """Linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[:n]
    return conv

def compute_score(a):
    """
    Returns the value of evaluate_sequence for an array a that already
    sums to one (so the denominator is 1).  The function also returns
    the full convolution and its maximum – useful for the optimiser.
    """
    n = a.shape[0]
    conv = convolve_fft(a, a)  # length 2n‑1
    max_b = np.max(conv)
    # the denominator sum(a)**2 equals 1 because we keep a on the simplex
    score = 2.0 * n * max_b
    return float(score), conv, max_b

def gradient_l2_error(a, target):
    """
    Gradient of ½||conv(a) - target||₂² with respect to a.
    The scaling factor 4 comes from the chain rule:
        g = ∂/∂a Σ (c_k - t)² = 2 Σ (c_k - t) ∂c_k/∂a
        and ∂c_k/∂a_i = 2 a_{k-i}.
    """
    n = a.shape[0]
    conv = convolve_fft(a, a)           # (2n‑1,)
    diff = conv - target                # same shape
    a_rev = a[::-1]
    grad_full = convolve_fft(diff, a_rev)   # length 3n‑2
    start = n - 1
    grad = 4.0 * grad_full[start:start + n]
    return grad

"""
Projected Gradient Descent for Flattening the Self‑Convolution of a Step Function.

The evaluation function is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

and is scale‑invariant, i.e. we can force sum(a)=1 without loss of generality.
With this normalisation the objective reduces to 2·n·max(conv).  Minimising the
maximum of the convolution is equivalent to making the convolution as flat as
possible (its total mass is 1).  The algorithm therefore minimises the L2
distance between the current convolution and the uniform distribution:

    L(a) = ½ ‖conv(a, a) – u‖₂² ,   u_k = 1/(2n‑1).

The gradient of L is
    ∇L = 4 * conv(diff, a_rev)_{n‑1 : n‑1+n},
where diff = conv(a, a) – u and a_rev is a reversed copy of a.
A projected gradient step (followed by a Euclidean projection onto the simplex
{a ≥ 0, sum(a)=1}) reduces L and, in practice, also lowers the original
evaluation score.  Several random restarts and a simple back‑tracking line
search are performed within the allotted time budget; the best sequence found
is returned.

The code uses only NumPy and the standard library and respects the required
time limit.
"""

import time
import numpy as np


def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using the FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[:n]
    return conv


def compute_score(a: np.ndarray) -> tuple[float, np.ndarray, float]:
    """
    Returns the value of the original evaluation function for a vector a that
    already satisfies sum(a)=1 (hence the denominator is 1).  Also returns the
    full convolution and its maximum – useful for the optimiser.
    """
    n = a.shape[0]
    conv = convolve_fft(a, a)           # length 2n‑1
    max_b = np.max(conv)
    score = 2.0 * n * max_b              # because (sum a)² = 1
    return float(score), conv, max_b


def gradient_l2_error(a: np.ndarray, target: np.ndarray) -> np.ndarray:
    """
    Gradient of ½‖conv(a,a) – target‖₂² with respect to a.
    The factor 4 comes from the chain rule (∂c_k/∂a_i = 2·a_{k‑i}).

    conv = convolve_fft(a, a)                 # current convolution
    diff = conv - target                       # (2n‑1,) array
    a_rev = a[::-1]
    # convolution of diff with reversed a gives the gradient (pre‑slice)
    grad_full = convolve_fft(diff, a_rev)     # length 3n‑2
    start = n - 1
    grad = 4.0 * grad_full[start:start + n]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative coefficients) that minimises
    evaluate_sequence().  The method performs several random restarts of a
    projected gradient descent that flattens the self‑convolution.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin so the harness does not kill us.
    deadline = start_time + budget_s - 5.0

    # Length of the sequence – a few thousand gives a good balance.
    n = 3000

    # Uniform target for the convolution (sum(a)=1 ⇒ total mass = 1)
    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val, dtype=np.float64)

    best_score = np.inf
    best_a = None

    # ------------------------------------------------------------
    # Main loop: keep launching random restarts until time runs out
    # ------------------------------------------------------------
    while time.time() < deadline:
        # ---- 1️⃣ Initialise -------------------------------------------------
        if np.random.rand() < 0.2:
            a = np.full(n, 1.0 / n)                 # uniform start
        else:
            a = np.random.dirichlet(np.ones(n))    # random point on the simplex

        # ---- 2️⃣ Evaluate the start point ----------------------------------
        curr_score, _, _ = compute_score(a)
        if curr_score < best_score:
            best_score = curr_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] Init best score {best_score:.6f}")

        # ---- 3️⃣ Gradient‑descent on the L2 error -------------------------
        eta = 0.5          # initial step length
        no_improve = 0     # counts consecutive unrewarded steps

        while time.time() < deadline:
            grad = gradient_l2_error(a, target)
            a_new = project_onto_simplex(a - eta * grad, 1.0)

            new_score, _, _ = compute_score(a_new)

            # Simple back‑tracking line search
            if new_score < curr_score:
                a = a_new
                curr_score = new_score
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.1f}s] New global best {best_score:.6f}")
            else:
                # step rejected – shrink step size
                eta *= 0.5
                no_improve += 1
                if eta < 1e-9:          # step too tiny → abort this restart
                    break

            # Occasionally revive the step size to avoid stagnation
            if no_improve > 200:
                eta = 0.5
                no_improve = 0

            # Heuristic termination of this restart
            if no_improve > 2000:
                break

    # Fallback – should never happen, but just in case
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    print(f"Search finished after {time.time() - start_time:.1f}s, best score {best_score:.6f}")
    return best_a.tolist()
