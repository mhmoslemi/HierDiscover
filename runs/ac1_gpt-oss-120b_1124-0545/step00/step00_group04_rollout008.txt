import time
import numpy as np

def _build_cosine_sequence(n: int, amp: float = 1.0) -> np.ndarray:
    """
    Return a numpy array of length n with the cosine shape.
    The result is already normalised to sum sqrt(2n).
    """
    # Normalisation constant (ensures the sum is sqrt(2n) when sum(cos)=0)
    C = np.sqrt(2.0 * n) / n
    i = np.arange(n, dtype=float)

    # Cosine term runs from +1 (i=0) to –1 (i=n‑1)
    a = C * (1.0 + amp * np.cos(np.pi * i / (n - 1)))

    # Numerical safety – negative values can only appear because of
    # floating‑point round‑off when amp is very close to 1.
    a = np.maximum(a, 0.0)

    # The analytical construction already gives the correct sum,
    # but we renormalise for safety (especially when amp ≠ 1).
    total = a.sum()
    if total > 0:
        a *= (np.sqrt(2.0 * n) / total)

    return a
