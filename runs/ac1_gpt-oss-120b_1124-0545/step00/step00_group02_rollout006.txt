def project_simplex(v):
    # Input: array v (not necessarily sum 1).
    # Output: projection onto simplex {x>=0, sum=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""Projected gradient search for a low‑value step‑function sequence.

The algorithm minimizes a smooth approximation of the max convolution
value (the objective of ``evaluate_sequence``) using a softmax (log‑sum‑exp)
surrogate.  At each iteration we

* compute the self‑convolution of the current probability vector,
* form softmax weights from the convolution,
* compute the gradient of the surrogate w.r.t. the vector (via a single
  convolution, O(n log n)),
* take a small step in the negative gradient direction and finally
  project back onto the probability simplex (non‑negative entries that sum
  to one).

Several lengths and random restarts are tried within the time budget,
and the best sequence found (according to the true ``evaluate_sequence``
function) is returned.
"""

import time
import numpy as np


def _project_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x >= 0, sum(x) = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen for non‑zero vectors, but guard anyway
        return np.maximum(v, 0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    ``evaluate_sequence`` value.  The function respects the given time
    budget (seconds) and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a small safety margin

    best_seq = None
    best_score = float("inf")

    # lengths to explore – larger n gives more degrees of freedom.
    lengths = [80, 120, 200, 300, 400, 500, 600, 800]

    # hyper‑parameters (tuned empirically)
    max_iters_per_run = 2000        # optimization steps per restart
    restarts_per_length = 2        # different initialisations
    beta_start = 0.5                # start of soft‑max temperature
    beta_end = 30.0                 # end of temperature schedule
    lr_start = 0.2                  # initial step size (norm of step)
    eval_every = 100                # how often to call evaluate_sequence

    for n in lengths:
        if time.time() > deadline:
            break

        for restart in range(restarts_per_length):
            # ----- initialise a probability vector -----
            if restart == 0 and "height_sequence_1" in globals():
                init = np.array(height_sequence_1[:n], dtype=float)
                if init.size < n:
                    init = np.pad(init, (0, n - init.size), constant_values=0.0)
                if init.sum() == 0.0:
                    init = np.ones(n) / n
                else:
                    init = init / init.sum()
                seq = init.copy()
            else:
                seq = np.random.rand(n)
                seq = seq / seq.sum()

            # ----- optimisation loop -----
            for it in range(max_iters_per_run):
                if time.time() > deadline:
                    break

                # temperature schedule for smooth max approximation
                beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_run)

                # self‑convolution
                conv = np.convolve(seq, seq)               # length 2n‑1
                # softmax weights (log‑sum‑exp surrogate of max)
                exp_vals = np.exp(beta * conv)
                w = exp_vals / exp_vals.sum()               # length 2n‑1, sums to 1

                # gradient of the surrogate:
                # g_i = 2 * sum_{t} seq[t] * w[i + t]
                w_rev = w[::-1]
                grad_full = np.convolve(seq, w_rev)        # length 3n‑2
                idx = (2 * n - 2) - np.arange(n)           # positions giving g_i
                grad = 2.0 * grad_full[idx]

                # normalise the direction to keep steps comparable across n
                gn = np.linalg.norm(grad) + 1e-12
                step_dir = grad / gn

                # step‑size schedule (decreases linearly)
                lr = lr_start * (1.0 - it / max_iters_per_run)

                # gradient step + projection onto simplex
                seq = seq - lr * step_dir
                seq = _project_simplex(seq)

                # occasional evaluation with the true objective
                if (it % eval_every) == 0:
                    cur_score = evaluate_sequence(seq.tolist())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = seq.copy()
                        print(f"[{time.time():.2f}] New best: {best_score:.6f} "
                              f"(n={n}, it={it}, restart={restart})")

                # small random jitter to avoid stagnation
                if gn < 1e-8 and np.random.rand() < 0.3:
                    seq += 0.01 * (np.random.rand(n) - 0.5)
                    seq = _project_simplex(seq)

            # final evaluation for this restart
            cur_score = evaluate_sequence(seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Improved after restart: {best_score:.6f} "
                      f"(n={n}, restart={restart})")

    # fallback – extremely unlikely to be needed
    if best_seq is None:
        n = 200
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform sequence, score={best_score:.6f}")

    print(f"Best final score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
