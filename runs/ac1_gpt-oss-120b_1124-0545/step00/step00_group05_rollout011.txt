"""
Gradient descent on a smooth approximation of the max‐convolution.
-----------------------------------------------------------------

We minimise the evaluation function

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

where ``a`` is a non‑negative step‑function (a list of heights).
The function is scale invariant: multiplying all entries by a constant
does not change the value, therefore we keep the ℓ¹‑norm of the vector
fixed (set to 1) during the optimisation.

The max‑operator is non‑smooth, but a smooth approximation

    M_α(a) = (1/α)·log( Σ_k exp(α·c_k) ),    c = a ⊛ a

has the same minimisers as the original problem when α → ∞.
Its gradient is

    ∂M_α/∂a_i = 2· Σ_k w_k·a_{k‑i} / Σ_k w_k,
    w_k = exp(α·(c_k – max(c)))

so the gradient is a weighted correlation of the current convolution
with the vector a.  Using this gradient we perform projected gradient
descent (project onto the non‑negative orthant and renormalise the sum).

The algorithm:
    * start from a few different initial vectors (uniform random and a
      deterministic “half‑triangle” shape);
    * repeatedly compute the convolution (FFT based), the smooth weights,
      the gradient and take a small step;
    * keep the best vector seen (according to the exact ``evaluate_sequence``).

The whole search respects the time budget (default 1000 s).  The returned
sequence is a list of non‑negative floats ready for evaluation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are defined at module level (no closures)
# ----------------------------------------------------------------------

def convolve_self_fft(x: np.ndarray) -> np.ndarray:
    """
    Autoconvolution of a real vector using FFT.
    Returns the full convolution of length 2*len(x)-1.
    """
    n = x.size
    # power‑of‑two size for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    return conv[: 2 * n - 1]


def objective_value(x: np.ndarray) -> float:
    """
    Exact objective used by the competition (matches evaluate_sequence).
    """
    b = np.convolve(x, x)               # direct convolution – small enough for scoring
    max_b = np.max(b)
    s = np.sum(x)
    if s < 0.01:                         # protection in evaluate_sequence
        return np.inf
    return 2.0 * x.size * max_b / (s * s)


def smooth_max_gradient(x: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of the smooth max approximation of max(conv(x)).
    Returns a vector of the same shape as x.
    """
    conv = convolve_self_fft(x)                     # full convolution, length 2n-1
    max_val = np.max(conv)

    # weights – shifted by max to avoid overflow, all values in (0,1]
    w = np.exp(alpha * (conv - max_val))
    w_sum = np.sum(w) + 1e-30                         # avoid division by zero

    # correlation (valid) gives Σ_k w_k·x_{k‑i}
    grad = 2.0 * np.correlate(w, x, mode='valid') / w_sum
    return grad


def project_nonnegative_and_normalise(x: np.ndarray) -> np.ndarray:
    """
    Enforce x ≥ 0 and renormalise the ℓ¹‑norm to 1.
    """
    x = np.maximum(x, 0.0)
    total = np.sum(x)
    if total == 0.0:
        # fall back to a tiny uniform vector to keep the search alive
        x = np.full_like(x, 1.0 / x.size)
    else:
        x = x / total
    return x


def initialise_vectors(n: int, rng: np.random.Generator):
    """
    Produce a small list of starting vectors.
    """
    # 1) deterministic half‑triangle (high on the left, low on the right)
    idx = np.arange(n)
    half = np.concatenate([np.linspace(1.0, 0.0, n // 2),
                           np.linspace(0.0, 1.0, n - n // 2)])
    v1 = half / np.sum(half)

    # 2) random uniform Dirichlet (positive entries, sum = 1)
    v2 = rng.random(n)
    v2 = v2 / np.sum(v2)

    # 3) another random start (different seed)
    v3 = rng.random(n)
    v3 = v3 / np.sum(v3)

    return [v1, v2, v3]

    """
    Search for a step‑function with a low evaluation score.

    budget_s : float

    """

    lr_initial = 0.12          # step size

            # gradient step on the smooth max
            grad = smooth_max_gradient(a, alpha)
            a = a - lr * grad
            a = project_nonnegative_and_normalise(a)

            # Occasionally compute the exact objective (cheap enough)
            if (it & 31) == 0:          # every 32 iterations
                score = objective_value(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"Iter {it:5d} – new best {best_score:.6f}")

            it += 1

        # simple early exit if we already reached a very good bound
        if best_score < 1.51:          # close to the target 1.503
            break

    # Fallback: if nothing was found (should never happen) use a uniform vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = objective_value(best_seq)
        print("Fallback to uniform sequence, score:", best_score)

    # Return as plain Python list (as required by the evaluator)
    return best_seq.tolist()
```
