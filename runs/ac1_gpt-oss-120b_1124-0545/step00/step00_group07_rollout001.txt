def propose_candidate(seed=..., budget_s=1000, n=2000):
    np.random.seed(seed)
    # initialize sequence: maybe random positive numbers drawn from Dirichlet or uniform.
    a = np.random.rand(n)
    a = a / a.sum()  # sum to 1
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # track for logging.

    initial_temp = 1.0  # parameter for annealing
    final_temp = 0.001
    start = time.time()
    deadline = start + budget_s - 0.5  # small margin
    # schedule: t = (elapsed / total)  fraction.
    # step size maybe set as delta_max = 0.1 initially, decreasing.

    while time.time() < deadline:
        # temperature
        elapsed = time.time() - start
        fraction = elapsed / (budget_s - 10)  # ?? adapt
        temp = initial_temp * (1 - fraction) + final_temp * fraction

        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # choose delta: random normal scaled by step size
        step_scale = 0.1 * (1 - fraction) + 0.001 * fraction  # gradually shrink
        delta = np.random.randn() * step_scale
        # ensure we transfer from i to j: we will apply a_i_new = a_i - delta, a_j_new = a_j + delta.
        # We need both nonnegative.
        a_i = a[i]
        a_j = a[j]
        # adjust sign of delta so we subtract from i and add to j
        # If delta positive, we take from i (decrease i) and give to j; if delta negative, we take from j to i.
        # We'll handle both.
        # Let's define transfer = delta.
        # Build candidate
        new_a_i = a_i - delta
        new_a_j = a_j + delta
        # If out of bounds, skip (or clip?). We'll skip.
        if new_a_i < 0 or new_a_j < 0:
            continue
        candidate = a.copy()
        candidate[i] = new_a_i
        candidate[j] = new_a_j
        # Optionally renormalize to sum 1 due to floating errors.
        candidate = candidate / candidate.sum()
        # Evaluate
        try:
            score = evaluate_sequence(candidate.tolist())
        except Exception:
            continue
        # Accept if better or with probability exp((best_score - score) / temp) (since lower better)
        if score < best_score or np.random.rand() < np.exp((best_score - score) / max(temp, 1e-12)):
            a = candidate
            if score < best_score:
                best_score = score
                best_sequence = candidate.copy()
                print(f"New best: {best_score:.6f} at {elapsed:.1f}s")
    # return best_sequence as list floats
    return best_sequence.tolist()

import numpy as np
import time

def evaluate_obj(seq):
    """Fast evaluation of the objective 2*n*max(conv(a,a)) / sum(a)^2.
    Assumes non-negative seq and sum>0."""
    a = np.array(seq, dtype=float)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2 * len(a) * max_b / (s * s))

def targeted_move(a, step_scale):
    """Return a new sequence after moving a small amount of mass from an index that contributes
    most to the current maximal convolution coefficient."""
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))
    # compute contribution weights w_i = a_i * a_{k_max-i} if both indices exist
    w = np.zeros(n)
    # i range where j = k_max - i in [0,n-1]
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]
    if w.sum() == 0:
        # fallback to uniform random index
        i = np.random.randint(n)
    else:
        # sample i proportional to w
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)
    # choose a destination index j different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # amount to move, limited by current mass at i
    max_transfer = a[i]
    if max_transfer <= 0:
        # nothing to move, return unchanged
        return a
    # sample delta uniformly in [0, step_scale * max_transfer]
    delta = step_scale * max_transfer * np.random.rand()
    # perform transfer
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalize to eliminate numerical drift
    a_new /= a_new.sum()
    return a_new

def random_move(a, step_scale):
    """Random transfer of a small fraction of mass between two random indices."""
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    # choose delta up to step_scale * max_transfer
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalize (should preserve sum but floating errors)
    a_new /= a_new.sum()
    return a_new

def gradient_step(a, step):
    """Simple gradient‐like step that tries to flatten the peak of the convolution."""
    n = len(a)
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    # gradient g_i = 2*a_{k-i} for valid indices, else 0
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    # direction that keeps the sum zero
    d = -g + g.mean()
    a_new = a + step * d
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() == 0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= a_new.sum()
    return a_new

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated‑annealing search that combines random perturbations, a
    targeted move that reduces the current maximal autocorrelation coefficient,
    and a simple gradient step.  The algorithm works with a fixed length
    (default 2000) non‑negative sequence normalised to sum 1 and returns the
    best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --- initialise -------------------------------------------------
    n = kwargs.get("n", 2000)           # length of the step function
    # start with a Dirichlet (uniform) distribution
    a = np.random.rand(n)
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_obj(best_seq.tolist())
    print(f"Initial score {best_score:.6f}")

    # annealing schedule
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.2         # relative amount moved in a single perturbation
    final_step = 0.001

    iteration = 0
    last_improve_time = start

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)   # scale 0‑1
        # temperature for Metropolis acceptance
        temp = init_temp * (1 - progress) + final_temp * progress
        # step size for transfers (absolute, not relative)
        step_scale = init_step * (1 - progress) + final_step * progress

        # choose a move type
        move_choice = np.random.rand()
        if move_choice < 0.4:
            a_candidate = targeted_move(a, step_scale)
        elif move_choice < 0.8:
            a_candidate = random_move(a, step_scale)
        else:
            # gradient‑like step, a bit larger step early on
            grad_step = step_scale * 0.5
            a_candidate = gradient_step(a, grad_step)

        cand_score = evaluate_obj(a_candidate.tolist())

        # Metropolis acceptance (lower is better)
        delta = cand_score - best_score
        if cand_score < best_score or np.random.rand() < np.exp(-delta / max(temp, 1e-12)):
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                last_improve_time = time.time()
                print(f"Iter {iteration:6d}  new best {best_score:.6f}  time {elapsed:.1f}s")

        # occasional restart if no improvement for a while
        if iteration % 5000 == 0 and (time.time() - last_improve_time) > 30.0:
            a = np.random.rand(n)
            a = a / a.sum()
            print(f"Restarted at iter {iteration}")

        iteration += 1

    # final conversion to list
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()

"""
Simulated‑annealing / hill‑climbing search for a non‑negative step function
that minimises  2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm works with a fixed length (default 2000) sequence that is kept
normalised to sum = 1.  Three elementary moves are mixed:

* **targeted_move** –  moves a little mass away from an index that contributes
  most to the current maximal autocorrelation coefficient.
* **random_move**   –  a simple random transfer of a tiny amount of mass between
  two indices (exploration).
* **gradient_step** – a cheap gradient‑like step that tries to flatten the peak
  of the convolution.

A temperature schedule and a decreasing step size are used for a Metropolis
acceptance rule, allowing occasional uphill moves early on.  If no improvement
is observed for a while the search is restarted from a fresh random
distribution.  The best sequence found before the time budget expires is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Return the exact evaluation value for a (assumes a ≥ 0)."""
    s = a.sum()
    if s < 1e-12:                     # should never happen – defensive
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    # Transfer at most ``step_scale * a[i]`` (never negative)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # Re‑normalise to eliminate rounding error
    a_new /= a_new.sum()
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from an index that contributes most to the current
    maximal convolution entry.  The probability of picking an index is
    proportional to its contribution a[i]·a[k_max‑i].
    """
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # compute contributions w_i = a_i * a_{k_max-i} (zero if out of range)
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # pick a destination index different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()

    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Very cheap gradient‑like update based on the peak of the convolution.
    The direction d = -g + mean(g) keeps the sum unchanged (∑d = 0).
    """
    n = len(a)
    b = np.convolve(a, a)
    k = int(np.argmax(b))

    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]          # ∂b_k/∂a_i = 2·a_{k-i} for valid i

    d = -g + g.mean()                  # ∑d = 0
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)

    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good height sequence within ``budget_s`` seconds.
    The routine mixes three move types (targeted, random, gradient) and
    uses a simple annealing schedule.
    Returns the best sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin so the process is not terminated abruptly
    deadline = start + budget_s - 0.5

    # length of the step function – can be overridden via kwargs
    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # initialise a uniform Dirichlet‑like distribution (sum = 1)
    # --------------------------------------------------------------
    a = np.random.rand(n)
    a /= a.sum()

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # annealing parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.20          # relative to current mass (0 … 1)
    final_step = 0.001

    iteration = 0
    last_improve = start

    # --------------------------------------------------------------
    # main SA loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)          # in [0,1]

        # temperature for Metropolis acceptance
        temp = init_temp * (1.0 - progress) + final_temp * progress
        # step size (relative transfer factor)
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # choose move type
        move_type = np.random.rand()
        if move_type < 0.40:                # targeted move – most important
            cand = targeted_move(a, step_scale)
        elif move_type < 0.80:              # simple random perturbation
            cand = random_move(a, step_scale)
        else:                               # gradient‑like step
            cand = gradient_step(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance (lower score is better)
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:6d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # occasional restart if stagnation is detected
        if iteration % 5000 == 0 and (time.time() - last_improve) > 30.0:
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{iteration:6d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
