"""
Propose a high‑quality step function.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a monotone decreasing polynomial shape

    a[i] = ((n - i) / n) ** p          (0 ≤ i < n)

the convolution peak occurs at the centre and can be expressed
as an integral.  In the continuous limit the score simplifies to

    score(p) = 2 * (p + 1) ** 2 * Beta(p + 1, p + 1),

which quickly drops below the target 1.503 as soon as p ≥ 2.
We therefore generate a few candidates with different exponents,
measure the true score with `evaluate_sequence`, and return the
best one.  The construction needs only O(n) work, the evaluation
is done by the provided routine, and the whole search finishes
well within the allotted time.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Generate a decreasing polynomial sequence and return the one
    with the smallest evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed (unused, kept for API compatibility).
    budget_s : float, optional
        Time budget in seconds – the routine uses only a tiny
        fraction of it.
    **kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    # The function runs well within a few seconds; we keep a safety margin.
    deadline = time.time() + min(budget_s, 30.0)

    # Length of the step function – a few thousand values give fast conv.
    n = 5000

    # Try several exponents; higher exponents concentrate mass at the
    # left end and dramatically lower the score.
    exponents = [1, 2, 3, 4, 5, 6]

    best_seq = None
    best_score = float("inf")

    for p in exponents:
        # Build the candidate: ((n - i) / n) ** p  for i = 0 .. n-1
        seq = [((n - i) / n) ** p for i in range(n)]

        # Evaluate – the function handles clipping and all safety checks.
        try:
            score = evaluate_sequence(seq)
        except Exception as e:
            # Any unexpected error – skip this candidate.
            print(f"Evaluation failed for p={p}: {e}")
            continue

        print(f"p={p:<2d} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # Stop early if we already beat the target comfortably.
        if best_score < 1.0:
            break

        # Respect the overall runtime budget.
        if time.time() > deadline:
            break

    # As a fallback, if something went wrong, return a simple uniform list.
    if best_seq is None:
        print("Fallback to uniform sequence.")
        best_seq = [1.0] * n

    print(f"Best score achieved: {best_score:.6f} with exponent {exponents[exponents.index(p)]}")
    return best_seq
