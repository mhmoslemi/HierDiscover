Pseudo:

def project_simplex(v):
    # Projects v onto probability simplex (sum to 1, all >=0)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(a, alpha, eta, n_fft):
    fft_a = np.fft.rfft(a, n=n_fft)
    b = np.fft.irfft(fft_a * fft_a, n=n_fft)[:2*len(a)-1]

def project_simplex(v):
    """Projects vector v onto the probability simplex (nonnegative, sum=1)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a new height sequence using a smoothed‐max gradient descent.

The objective of evaluate_sequence is
    score = 2·n·max(conv(a,a)) / (sum(a)^2) .
Because the score is scale invariant we may fix sum(a)=1.
Let b = a∗a (linear convolution).  The term max(b) is the
worst probability of the sum of two independent draws from the
distribution a.  To minimise the score we want this peak to be as
flat as possible.

We minimise a smooth approximation of max(b) :
    f(a)= (1/α)·log( Σ_k exp(α·b_k) )
whose gradient is
    ∇f_i = 2· Σ_k w_k·a_{k‑i},   w_k = softmax(α·b_k) .
Both the convolution and the gradient are computed with FFTs
( O(L log L) with L ≈ 3·n ).  After a gradient step we project the
vector onto the probability simplex (non‑negative, sum=1).
Random restarts and occasional noise injections avoid stagnation.
The best sequence found within the time budget is returned.
"""

import time
import numpy as np

# ----- helper utilities -------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_step(a: np.ndarray, alpha: float, eta: float, L: int):
    """
    One smoothed‑max gradient descent step.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector (sum≈1, non‑negative).
    alpha : float
        Smoothing parameter for the soft‑max (larger → closer to true max).
    eta : float
        Step size (learning rate).
    L : int
        FFT length (≥ 3·n, power‑of‑two).

    Returns
    -------
    a_new : ndarray, shape (n,)
        Updated probability vector (projected onto the simplex).
    max_b : float
        Maximum entry of the linear convolution a∗a (computed in this step).
    """
    n = a.shape[0]

    # ---------- convolution a * a ----------
    fft_a = np.fft.rfft(a, n=L)
    b_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]  # linear convolution
    max_b = float(b_full.max())
    # ---------- soft‑max weights ----------
    # shift by max to avoid overflow, keep values ≤ 1
    exp_arg = alpha * (b_full - max_b)
    # clip to avoid underflow in exp (exp(-700) is already ~0)
    np.maximum(exp_arg, -700.0, out=exp_arg)
    w = np.exp(exp_arg)  # w_max = 1, 0 ≤ w ≤ 1

    # ---------- gradient = 2·(w * a_rev) ----------
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_rev = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(fft_w * fft_rev, n=L)
    # gradient entries are the slice centred at index n‑1
    grad = 2.0 * conv[n - 1 : n - 1 + n]

    # ---------- gradient descent + projection ----------
    a_tmp = a - eta * grad
    a_new = simplex_projection(a_tmp)

    return a_new, max_b

    print(f"[done] iterations={it}, best score={best_score:.6f}")
    # ensure the returned list satisfies the evaluator's safety checks
    return best_a.tolist()
