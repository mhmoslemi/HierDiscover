def propose_candidate(budget_s=1000, seed=0):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0 # margin

    # Choose length n: maybe 2000 or adaptively increase if time.
    # Could try starting with n=2000 (like height_sequence_1 length). Might be good.
    n = 2000 # could choose 3000.

    # Initialize sequence: uniform random positive, e.g., a = np.random.rand(n)
    a = np.random.rand(n)
    # Ensure sum>0; project later
    # Normalize to sum 1
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score {best_score}')

    # Hyperparameters:
    beta = 1.0
    beta_inc = 1.5
    max_beta = 50.0
    step_size = 0.1 # adjust
    # Could adapt step size.

    iteration = 0
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(a, a) # length 2n-1
        max_conv = conv.max()
        # compute smooth approximation weight w
        # To avoid overflow, subtract max before exp
        scaled = beta * (conv - conv.max())  # shift to avoid overflow
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()  # shape length 2n-1

        # compute correlation term: corr_j = sum_i a_i * w_{i+j}
        # Use convolution of reversed a and w
        rev_a = a[::-1]
        corr_full = np.convolve(rev_a, w)  # length 3n-2
        # correlation at lag j corresponds to index (n-1 + j)
        corr = corr_full[n-1 : n-1 + n]  # length n

        # compute gradient
        # diagonal correction term: w[2*j] * a_j
        diag = np.zeros_like(a)
        # Determine indices where 2*j < len(w)
        idxs = 2 * np.arange(n)
        mask = idxs < len(w)
        diag[mask] = w[idxs[mask]] * a[mask]
        grad = 2 * corr - diag

        # Gradient descent step: a = a - step_size * grad
        a_new = a - step_size * grad

        # Project onto non-negative
        a_new = np.maximum(0.0, a_new)

        # Renormalize to avoid sum too small: ensure sum >= 0.01. Scale to sum=1
        s = a_new.sum()
        if s < 1e-8:
            # reinitialize random small maybe
            a_new = np.random.rand(n)
            s = a_new.sum()
        a_new = a_new / s  # maybe scaling does not affect ratio, but keep stable.

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f'Iter {iteration} new best {best_score:.6f} beta {beta:.2f} step {step_size}')
        # Update step size maybe adapt
        # If not improving for many steps, reduce step
        # Also gradually increase beta to focus on max
        iteration += 1
        # adjust step size adaptively:
        if iteration % 50 == 0:
            step_size *= 0.9  # decay
            beta = min(max_beta, beta * beta_inc)

    print(f'Finished after {iteration} iterations, best score {best_score}')
    return best_seq.tolist()

"""
Propose a candidate step‑function by minimizing the max‑convolution objective.

The evaluation score is 2*n*max(a∗a)/(sum(a)^2) and is invariant under
uniform scaling of the whole vector.  Hence we can keep the vector normalised
to unit sum without loss of generality.

The algorithm performs a projected gradient descent on a smooth
approximation of the max‑convolution:

    w_k(β) = exp(β·( (a∗a)_k – max(a∗a) )) / Σ_j exp(β·( (a∗a)_j – max(a∗a) ))

For β→∞ the weighted sum Σ_k w_k (a∗a)_k converges to
max_k (a∗a)_k.  The gradient of this smooth objective can be written in
closed form using a correlation, which we obtain via a second convolution.
After each step we project onto the non‑negative orthant and renormalise the
vector to keep the sum equal to 1 (any other scaling gives the same score).

A simple schedule raises β periodically, the learning rate decays, and a
tiny Gaussian noise is added to help escape shallow local minima.  The
current best sequence is kept and the process restarts from it when no
improvement occurs for a while.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence found.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    **kwargs
        Ignored; kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered before the deadline.
    """
    # --------------------------------------------------------------------
    # Basic setup
    # --------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --- initialise the vector -----------------------------------------
    # Try to reuse the existing construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        # Limit the size to keep each iteration cheap.
        max_n = 3000
        if n > max_n:
            base = base[:max_n]
            n = max_n
    except NameError:
        # No global `height_sequence_1`; fallback to a random start.
        n = 2000
        base = np.random.rand(n)

    # Ensure feasibility.
    base = np.maximum(0.0, base)
    if base.sum() == 0.0:
        base = np.random.rand(n)
    base /= base.sum()                     # unit sum – scale‑invariant

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[INIT] n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------------
    # Hyper‑parameters (tuned by hand)
    # --------------------------------------------------------------------
    beta = 1.0               # temperature for the soft‑max
    beta_mult = 1.20         # multiplicative increase every 100 iters
    beta_max = 60.0

    lr_initial = 0.5         # initial learning rate
    lr_decay = 200           # denominator for 1/(1+it/decay)

    noise_scale = 1e-3       # std dev of isotropic noise added each step

    max_iters_without_improve = 1000  # restart after this many stagnant iters

    # --------------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------------
    current = base.copy()
    it = 0
    stagnant = 0

    while time.time() < deadline:
        # ----- compute a∗a ------------------------------------------------
        conv = np.convolve(current, current)          # length 2n‑1

        # ----- soft‑max weights -----------------------------------------
        max_c = conv.max()
        # shift before exponentiation to avoid overflow
        shifted = beta * (conv - max_c)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / exp_shifted.sum()      # w_k(β)

        # ----- correlation term (Σ_i a_i w_{i+j}) -----------------------
        # correlation = (rev_a ⋆ w); rev_a = a reversed.
        rev_a = current[::-1]
        corr_full = np.convolve(rev_a, weights)       # length 3n‑2
        # The relevant slice corresponds to lags j = 0…n‑1.
        corr = corr_full[n - 1 : n - 1 + n]

        # ----- diagonal correction (the case i = k‑i) -------------------
        idx2 = 2 * np.arange(n)                       # positions 2j in w
        diag = np.zeros_like(current)
        valid = idx2 < len(weights)
        diag[valid] = weights[idx2[valid]] * current[valid]

        # ----- gradient of the smooth objective -------------------------
        grad = 2.0 * corr - diag

        # ----- step size schedule ---------------------------------------
        lr = lr_initial / (1.0 + it / lr_decay)

        # Normalise the gradient to keep updates comparable across iterations.
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            grad_norm = grad / max_abs
        else:
            grad_norm = grad

        # ----- projected gradient step with noise -----------------------
        noise = noise_scale * np.random.randn(n)
        new_vec = current - lr * grad_norm + noise
        new_vec = np.maximum(0.0, new_vec)            # enforce a_i ≥ 0

        # Renormalise to unit sum (any positive scaling is equivalent).
        s = new_vec.sum()
        if s < 1e-12:                                 # protect against collapse
            new_vec = np.random.rand(n)
            s = new_vec.sum()
        new_vec /= s

        # ----- evaluation -----------------------------------------------
        try:
            cur_score = evaluate_sequence(new_vec.tolist())
        except Exception:  # safety net
            cur_score = float("inf")

        # ----- keep the best found ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_vec.copy()
            stagnant = 0
            elapsed = time.time() - start_time
            print(f"[ITER {it:5d}] new best {best_score:.6f}  β={beta:.2f}  lr={lr:.4f}  time={elapsed:.1f}s")
        else:
            stagnant += 1

        # ----- temperature schedule --------------------------------------
        if (it + 1) % 100 == 0:
            beta = min(beta_max, beta * beta_mult)

        # ----- restart if stuck -----------------------------------------
        if stagnant > max_iters_without_improve:
            # Perturb the current best slightly and restart from there.
            pert = best_seq * (1.0 + 0.1 * np.random.randn(n))
            current = np.maximum(0.0, pert)
            if current.sum() == 0.0:
                current = np.random.rand(n)
            current /= current.sum()
            beta = 1.0
            stagnant = 0
            print(f"[RESTART] after {it} iters, restarting with perturbed best.")
        else:
            current = new_vec

        # ----- early exit if we already beat the target -----------------
        if best_score < 1.503:
            print(f"[GOAL] reached target {best_score:.6f} at iteration {it}.")
            break

        it += 1

    print(f"[DONE] Best score {best_score:.6f} after {it} iterations.")
    # Convert to plain Python list as required by the evaluation harness.
    return best_seq.tolist()
