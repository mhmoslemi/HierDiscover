def construct_sidon(n):
    positions=[]
    sums=set()
    for i in range(n):
        ok=True
        for p in positions:
            if (i+p) in sums:
                ok=False
                break
        if ok:
            # also check i+i sum
            if (i+i) in sums:
                ok=False
            else:
                # add sums
                for p in positions:
                    sums.add(i+p)
                sums.add(i+i)
                positions.append(i)
    return positions

        try_seq(dir_seq)

    # Local search starting from best_seq

def propose_candidate(seed=0, budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    n = 1200  # moderate length to keep runtime manageable
    # initial sequence: exponential
    r = 0.998

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print('initial score', best_score)

    while time.time() < deadline:
        # compute current max bound M
        conv = np.convolve(seq, seq)
        M = conv.max()
        # solve LP to get direction that increases sum under bound M
        b = solve_convolution_lp(seq, M)
        if b is None:
            break
        sum_seq = seq.sum()
        sum_b = b.sum()
        if sum_b <= sum_seq * (1 + 1e-12): # no improvement
            # maybe adjust step size?
            break
        # scale b to have same sum as seq
        g = b * (sum_seq / sum_b)

        # try mixing with various t values
        improved = False

"""
Propose a candidate sequence for the convolution evaluation problem.

Approach
--------
1. **Parameter sweep** – generate many simple families of non‑negative
   sequences (exponential, mixtures of exponentials, power‑law,
   shifted reciprocals, random Dirichlet vectors).  Each candidate is
   evaluated with `evaluate_sequence` and the best one is stored.

2. **Local hill‑climbing** – starting from the best candidate, repeatedly
   move a small amount of mass from one index to another (maintaining
   non‑negativity).  Accept changes that improve the score.

3. **LP‑based refinement** – use the linear programme from the reference
   algorithm to find a direction that raises the total sum while keeping
   the convolution bound unchanged.  After scaling the direction to the
   same ℓ₁‑norm as the current sequence we mix the two sequences with
   several mixing factors and keep any improvement.  This step is
   iterated as long as time permits.

All steps respect the limits of the `evaluate_sequence` function
(non‑negative entries, clipping at 0 … 1000).  The search runs for up to
`budget_s` seconds and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme: maximise sum(b) subject to  conv(f,b) <= M  and b >= 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, M: float):
    """Return the vector b solving the LP or None on failure."""
    n = len(f_seq)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints: for each k, sum_i f_i * b_{k-i} <= M
    A_ub = []
    b_ub = []

    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs from lo..hi, and column index j = k-i
        for i in range(lo, hi + 1):
            j = k - i
            row[j] = f_seq[i]
        A_ub.append(row)
        b_ub.append(M)

    # non‑negativity constraints  -b_i <= 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # Use the robust HiGHS solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None
