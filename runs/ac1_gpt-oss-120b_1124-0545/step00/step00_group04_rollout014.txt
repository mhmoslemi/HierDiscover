def simplex_projection(v, s):
    # ensure v >= 0 after projection but algorithm yields nonnegative.
    # Implementation per Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import random, time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = height_sequence_1 (if exists) else random.

    best_val = evaluate_sequence(best_seq)

    # define function to evaluate quickly: compute from evaluate_sequence or direct?

    def get_eval(seq):
        # Convert to list of floats maybe.
        return evaluate_sequence(seq)

    # define optimization routine for one given n and starting seq:
    def optimize(seq, time_limit):
        nonlocal best_seq, best_val
        n = len(seq)
        target_sum = np.sqrt(2 * n)
        # normalize seq to target sum
        sum_seq = np.sum(seq)
        if sum_seq == 0:
            seq = np.full(n, target_sum / n)
        else:
            seq = np.array(seq) * (target_sum / sum_seq)
        seq = np.maximum(seq, 0.0)  # maintain nonneg
        # Compute initial evaluation.
        cur_val = get_eval(seq.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.tolist()

        iteration = 0
        step = 0.5
        while time.time() < time_limit:
            # compute convolution
            conv = np.convolve(seq, seq)
            max_b = np.max(conv)
            # indices where max occurs (within tolerance)
            max_idx = np.where(conv >= max_b - 1e-12)[0]
            # compute gradient
            grad = np.zeros(n)
            for k in max_idx:
                i0 = max(0, k - (n - 1))
                i1 = min(n - 1, k)
                # loop i:
                for i in range(i0, i1 + 1):
                    j = k - i
                    grad[i] += seq[j]
            # direction to reduce max_b under sum constraint
            mean_grad = np.mean(grad)
            direction = mean_grad - grad  # -grad + mean*1
            # step size decreasing
            eta = step * (0.99 ** iteration)
            # propose new sequence
            cand = seq + eta * direction
            # project onto simplex
            cand = simplex_projection(cand, target_sum)
            # evaluate
            val = get_eval(cand.tolist())
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                # Accept new sequence
                seq = cand
                cur_val = val
            else:
                # maybe accept with small probability to escape
                # Use probability exp(-(val-cur_val)/temp)
                # temperature decays
                temp = max(1e-4, step * (0.99 ** iteration))
                if np.random.rand() < np.exp(-(val - cur_val) / temp):
                    seq = cand
                    cur_val = val
            iteration += 1
            # occasional random perturbation every 200 iterations
            if iteration % 200 == 0:
                # random perturb small noise
                noise = np.random.randn(n) * 0.01 * target_sum / n
                seq = simplex_projection(seq + noise, target_sum)
        # end while

    # Determine list of lengths and allocate time accordingly
    # If height_sequence_1 exists, its length will be used as one of the lengths.
    lengths = []
    if 'height_sequence_1' in globals():
        lengths.append(len(height_sequence_1))
    # add some other lengths
    lengths += [200, 400, 800, 1600]
    # deduplicate and sort
    lengths = sorted(set(lengths))
    # allocate time equally among lengths
    per_len_time = (deadline - start_time) / len(lengths)
    # start with each length
    for n in lengths:
        # create initial sequence
        if n == len(height_sequence_1):
            init_seq = list(height_sequence_1)
        else:
            # random positive sequence of length n
            init_seq = np.random.rand(n).tolist()
        # run optimization for per_len_time seconds
        opt_deadline = time.time() + per_len_time
        optimize(init_seq, opt_deadline)

    print(f"Best value found: {best_val:.6f} with length {len(best_seq)}")
    return best_seq

    """Generate a step-height sequence with low convolution bound.

    The algorithm builds an antisymmetric perturbation of the uniform
    sequence (which yields evaluation 2). By adding opposite signed
    deviations at mirrored positions the central convolution peak is
    reduced while preserving the total sum. Random draws are performed
    until a sequence with evaluation ≤ target is found or time runs out.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose a length – we try a few typical values.
    lengths = []
    if 'height_sequence_1' in globals():

    lengths += [512, 1024, 2048, 4096]
    # keep only reasonable lengths (<= 5000)
    lengths = [L for L in lengths if L <= 5000 and L > 0]
    lengths = sorted(set(lengths))

    best_seq = None
    best_val = float('inf')
    target_val = 1.5  # stop early if we beat the target

    # helper to compute evaluation using fft (fast)
    def eval_fast(arr):
        n = arr.shape[0]
        size = 1 << ((2*n - 1).bit_length())
        A = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(A * A, n=size)
        max_b = np.max(conv[:2*n - 1])

"""
Propose a step‑height sequence with a low autocorrelation bound.

The idea is simple but powerful:
* For a length n the uniform sequence (all entries equal) gives
  evaluation = 2.0 regardless of n.
* By pairing symmetric positions i and n‑1‑i and adding opposite
  signed perturbations δi (so a[i]=c+δi, a[n‑1‑i]=c‑δi)
  the total ℓ1‑norm stays exactly √(2n) while the central
  convolution value drops by 2·∑δi².
* Choosing the δi uniformly in [‑c, c] (c = √(2n)/n) makes the
  expected drop about 2/3, i.e. the expected evaluation ≈ 4/3 < 1.5.
* Random draws are cheap; we generate many candidates (different
  lengths, different seeds) and keep the best one.
* A fast evaluation using FFT‑based convolution is employed; the
  official `evaluate_sequence` is called only for promising candidates
  to guarantee exactness.

The search stops early as soon as a sequence with evaluation ≤ 1.5
is found or the time budget is exhausted.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: fast max‑convolution using FFT (no external imports)
# -------------------------------------------------------------
def max_convolution_fft(arr: np.ndarray) -> float:
    """
    Returns max( (arr * arr) ) where * denotes linear convolution.
    Computed via FFT for speed.
    """
    n = arr.shape[0]
    # next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT is enough because the vector is real
    A = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(A * A, n=size)
    # keep only the relevant 2*n-1 entries
    return float(np.max(conv[: 2 * n - 1]))


def fast_evaluation(arr: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence.
    The sequence is assumed to be non‑negative and its sum is >0,
    so the expression below matches the official evaluator.
    """
    n = arr.shape[0]
    max_b = max_convolution_fft(arr)
    sum_a = float(np.sum(arr))
    return 2.0 * n * max_b / (sum_a * sum_a)


# -------------------------------------------------------------
# Main entry point required by the grader
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function height sequence with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # lengths we will try – include the known construction length if present
    lengths = []
    if "height_sequence_1" in globals():
        lengths.append(len(height_sequence_1))
    # a few even lengths up to a few thousand (larger n gives more freedom)
    lengths.extend([512, 1024, 2048, 2584, 4096])
    lengths = [L for L in lengths if L > 0 and L % 2 == 0 and L <= 5000]
    lengths = sorted(set(lengths))

    best_seq = None
    best_val = float("inf")
    target_val = 1.5          # early‑stop threshold

    for n in lengths:
        if time.time() >= deadline:
            break

        half = n // 2
        c = math.sqrt(2.0 * n) / n               # uniform entry that satisfies the sum constraint

        # ---- baseline (uniform) -------------------------------------------------
        uniform = np.full(n, c, dtype=np.float64)
        val = fast_evaluation(uniform)
        # verify with the official evaluator for safety (only a few times)
        if val < best_val:
            best_val = val
            best_seq = uniform.tolist()
            print(f"[{n}] uniform baseline value = {val:.6f}")

        # ---- random antisymmetric search ---------------------------------------
        # each iteration creates a pairwise‑antisymmetric perturbation
        while time.time() < deadline:
            # draw δ_i ∈ [‑c, c] for the first half of the indices
            delta = np.random.uniform(-c, c, size=half)

            # build the full vector
            a = np.full(n, c, dtype=np.float64)
            idx = np.arange(half)
            a[idx] = c + delta
            a[n - 1 - idx] = c - delta          # mirrored positions

            # fast estimate
            val_fast = fast_evaluation(a)

            # only call the official (slower) evaluator when we see improvement
            if val_fast < best_val:
                # correctness check – the official function also clips numbers,
                # but our values are far below the limits.
                val_real = evaluate_sequence(a.tolist())
                if val_real < best_val:
                    best_val = val_real
                    best_seq = a.tolist()
                    print(f"[{n}] new best value = {best_val:.6f}")

                # stop early if we beat the target
                if best_val <= target_val:
                    deadline = time.time() + 0.1   # break loops shortly
                    break

        if best_val <= target_val:
            break   # target reached – no need to try larger lengths

    # fallback: if for some reason we never updated best_seq
    if best_seq is None:
        # return a simple uniform sequence (always a valid candidate)
        n = lengths[0] if lengths else 256
        c = math.sqrt(2.0 * n) / n
        best_seq = [float(c)] * n
        best_val = evaluate_sequence(best_seq)

    print(f"Finished search – best value = {best_val:.6f} (length {len(best_seq)})")
    return best_seq
