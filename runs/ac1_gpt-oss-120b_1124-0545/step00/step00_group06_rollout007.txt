def simplex_proj(v):
    # v: array, may be not summing to 1.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    # Duchi
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def update_expgd(p, k_max, step):

    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(p)
    g[mask] = 2 * p[idx[mask]]
    # exponentiated gradient
    p = p * np.exp(-step * g)
    p = p / p.sum()
    return p

    n = len(p)

def propose_candidate(seed=0, budget_s=900):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    best_seq = None
    best_score = np.inf

    # function to compute score efficiently (internal)
    def compute_score(seq):
        n = len(seq)
        conv = np.convolve(seq, seq)
        maxb = conv.max()
        s = np.sum(seq)
        if s < 1e-9:
            return np.inf
        return 2 * n * maxb / (s**2)

    # Try a few deterministic candidates: Gaussian, exponential, uniform
    for n in [800, 1200, 2000, 2800, 3400]:
        if time.time() > deadline:
            break
        # gaussian candidate
        for alpha in [0.3, 0.35, 0.4, 0.45, 0.5]:
            sigma = alpha * n
            i = np.arange(n)
            center = (n-1)/2
            seq = np.exp(- ((i-center)**2) / (2 * sigma**2))
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
        # exponential candidate
        for c in [0.7, 1.0, 1.25, 1.5, 2.0]:
            eta = c / n
            q = 1 - eta
            seq = q ** np.arange(n)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
        # uniform baseline
        seq = np.ones(n)
        score = compute_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()

    # Gradient descent refinement
    while time.time() < deadline:
        # pick starting candidate: either best_seq (if exists) with random perturbation, or random new
        if best_seq is not None and np.random.rand() < 0.7:
            n = len(best_seq)
            p = np.array(best_seq)
            # add small random noise then renorm
            p = p * (1 + 0.01*np.random.randn(n))
            p = np.clip(p, 0, None)
            if p.sum() == 0:
                continue
            p = p / p.sum()
        else:
            n = np.random.randint(1000, 3000)
            p = np.random.rand(n)
            p = p / p.sum()
        # run few inner iterations
        iters = np.random.randint(100, 500)
        step0 = 0.5
        for t in range(iters):
            # compute convolution max index
            conv = np.convolve(p, p)
            k_max = np.argmax(conv)
            # compute gradient
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            g = np.zeros_like(p)
            g[mask] = 2 * p[idx[mask]]
            # step size
            step = step0 / np.sqrt(t+1)
            # exponentiated gradient
            p = p * np.exp(-step * g)
            # renormalize
            p_sum = p.sum()
            if p_sum == 0:
                p = np.random.rand(n)
                p = p / p.sum()
            else:
                p = p / p_sum
            # occasional projection onto simplex (already normalized)
            # evaluate occasionally
            if t % 20 == 0:
                score = compute_score(p)
                if score < best_score:
                    best_score = score
                    best_seq = p.tolist()
        # after inner loop evaluate
        if time.time() > deadline:
            break
    # Return best_seq (ensure list of floats)
    if best_seq is None:
        # fallback to simple uniform
        best_seq = [1.0] * 1000
    print(f"Best score found: {best_score}")
    return best_seq

"""
Propose a candidate step‑function heights sequence.

The algorithm consists of three stages:

1. **Fast parametric sweep** – test a handful of simple families
   (uniform, exponential, Gaussian‑like) with several lengths and
   shape parameters.  The best among them provides a good starting
   point.

2. **Projected (exponentiated) gradient descent** – the objective
   `E(a)=2·n·max_k (a*a)_k / (∑a)^2` is scale‑invariant, so we work
   with a probability vector `p = a/∑a`.  The max‑convolution value is
   convex in `p`; a sub‑gradient at the current worst index `k*` is

        g_i = 2·p_{k*‑i}          (0≤i<n, 0 elsewhere)

   We perform a mirror‑descent step using the exponentiated gradient
   update `p_i ← p_i·exp(−η·g_i)` and renormalise to the simplex.
   The step size η decays like 1/√t.  Periodically we evaluate the
   true objective and keep the best vector seen.

3. **Random restarts / refinements** – while time permits we launch
   many short optimisation runs from random initialisations or from
   the current best sequence (or from the provided `height_sequence_1`
   if available).  The best sequence found before the deadline is
   returned.

All numbers stay non‑negative and far below the 1000‑cap, so the
security checks inside ``evaluate_sequence`` never reject the
candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_max(p: np.ndarray) -> float:
    """Return max value of the aperiodic convolution p*p."""
    # Using FFT for speed (real inputs)
    n = p.size
    m = 1 << ((2 * n - 1).bit_length())       # next power of two
    f = np.fft.rfft(p, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    return float(conv.max())


def _score(p: np.ndarray) -> float:
    """Internal (fast) evaluation of the objective for a probability vector."""
    n = p.size
    maxb = _conv_max(p)
    # p already sums to 1 -> denominator = 1
    return 2.0 * n * maxb


def _exp_gradient_step(p: np.ndarray, k_max: int, eta: float) -> np.ndarray:
    """One exponentiated‑gradient update that tries to lower the worst
    convolution entry (`k_max`)."""
    n = p.size
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    # sub‑gradient of the max‑convolution (positive values only)
    g = np.zeros_like(p)
    g[mask] = 2.0 * p[idx[mask]]
    # exponentiated‑gradient update, stays non‑negative
    p = p * np.exp(-eta * g)
    # renormalise to the simplex (sum = 1)
    s = p.sum()
    if s == 0.0:                     # numerical safeguard – re‑randomise
        p = np.random.rand(n)
        p /= p.sum()
        return p
    return p / s


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (Duchi et al.)."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for finalisation

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Parametric sweep (uniform / exponential / Gaussian families)
    # ------------------------------------------------------------------
    def _maybe_update(p_vec):
        nonlocal best_seq, best_score
        sc = _score(p_vec)
        if sc < best_score:
            best_score = sc
            best_seq = p_vec.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best (parametric) score: {sc:.6f}")

    lengths = [800, 1200, 1800, 2400, 2960, 3400]
    for n in lengths:
        if time.time() > deadline:
            break

        # uniform (baseline)
        _maybe_update(np.ones(n) / n)

        # exponential families a_i = q^i  (q in (0,1))
        for c in [0.6, 0.9, 1.2, 1.5, 2.0]:
            eta = c / n
            q = 1.0 - eta
            p = q ** np.arange(n, dtype=float)
            p /= p.sum()
            _maybe_update(p)

        # Gaussian‑like families centered in the middle
        centre = (n - 1) / 2.0
        indices = np.arange(n, dtype=float)
        for alpha in [0.30, 0.35, 0.40, 0.45, 0.48]:
            sigma = alpha * n
            p = np.exp(-((indices - centre) ** 2) / (2.0 * sigma ** 2))
            if p.sum() == 0.0:
                continue
            p /= p.sum()
            _maybe_update(p)

    # ------------------------------------------------------------------
    # 2. Gradient‑descent refinement with random restarts
    # ------------------------------------------------------------------
    # if ``height_sequence_1`` is defined we can reuse it as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0.0, None)
            prev = prev / prev.sum()
    except Exception:
        prev = None

    while time.time() < deadline:
        # ---------- choose an initial point ----------
        if prev is not None and np.random.rand() < 0.3:
            p = prev.copy()
            # add a tiny jitter to avoid staying at a stationary point
            p = p * (1.0 + 0.005 * np.random.randn(p.size))
            p = np.clip(p, 0.0, None)
            if p.sum() == 0.0:
                p = np.random.rand(p.size)
        else:
            # random length in a sensible interval
            n = np.random.randint(1500, 3400)
            p = np.random.rand(n)
        p /= p.sum()

        # ---------- inner optimisation loop ----------
        max_iters = np.random.randint(150, 400)   # cheap inner runs
        step0 = 0.8                               # initial step size
        for it in range(max_iters):
            # compute current worst convolution index
            conv = np.convolve(p, p)
            k_max = int(np.argmax(conv))

            # exponentiated‑gradient step
            eta = step0 / np.sqrt(it + 1.0)
            p = _exp_gradient_step(p, k_max, eta)

            # occasional evaluation
            if (it & 7) == 0:                      # every 8th iteration
                sc = _score(p)
                if sc < best_score:
                    best_score = sc
                    best_seq = p.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"New best (grad) score: {sc:.6f}")

            # early stop if time runs out
            if time.time() >= deadline:
                break

        # after each run, keep the result as a possible new seed
        if best_seq is not None:
            prev = best_seq.copy()

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list of floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a simple uniform vector
        best_seq = np.ones(1000) / 1000.0
        print("Fallback to uniform sequence.")

    final_seq = best_seq.tolist()
    print(f"Finished. Best score: {best_score:.6f} with length {len(final_seq)}")
    return final_seq
