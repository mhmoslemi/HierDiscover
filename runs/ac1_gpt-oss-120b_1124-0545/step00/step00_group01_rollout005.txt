def simplex_projection(v, z):
    # returns w s.t. sum w = z, w >= 0, and w closest to v (euclidean)
    # Implementation from Duchi.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) ==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Subgradient descent for the max‑self‑convolution problem.

The evaluation function is
    D = 2 * n * max(conv(a, a)) / (sum(a)^2)

and is invariant under scaling of the sequence.
Therefore we keep the L¹‑norm fixed to sqrt(2*n) (the same
normalisation used in the reference implementation) and minimise
the peak of the convolution.

Algorithm
---------

1. Initialise a sequence `a` (either the provided `height_sequence_1`
   or a random exponential vector) and scale it to the target sum.

2. Repeatedly
   * compute the full convolution `b = np.convolve(a, a)`;
   * locate the index set `K` where the maximum `M = max(b)` is attained;
   * build a sub‑gradient `g` of the function `M`:
       g_i = Σ_{k∈K} a_{k-i}   (ignoring out‑of‑range indices)
     The sub‑gradient points in the direction that would increase `M`;
     we therefore move opposite to it.  To keep the sum unchanged we
     subtract the mean of `g`, obtaining a zero‑sum direction `d`.
   * take a step `a' = a + η * d` with a slowly decaying step size `η`;
   * project `a'` onto the probability simplex of radius `target_sum`
     (non‑negative entries with the prescribed L¹‑norm);
   * evaluate the new sequence.  If the score improves we keep the move,
     otherwise we fall back to a small random mass‑transfer perturbation.

3. Occasionally (every few hundred iterations) a random restart is tried
   to escape possible stagnation.

The loop runs until the time budget is exhausted and the best sequence
found is returned.

The implementation is pure NumPy, avoids any file or network I/O,
and respects the 1000 s wall‑time limit.
"""

# Helper: projection onto the simplex {x >= 0, sum(x) = z}
def simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1   # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    evaluate_sequence().  The function respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Determine problem size and target L1 norm
    try:
        n = len(height_sequence_1)          # use the known construction if present
        init_from_known = True
    except Exception:
        n = 4595                           # fallback length (as mentioned in the prompt)
        init_from_known = False

    target_sum = math.sqrt(2.0 * n)        # sqrt(2n)

    # ------------------------------------------------------------------
    # Initialise the sequence
    if init_from_known and np.random.rand() < 0.5:
        a = np.array([float(x) for x in height_sequence_1], dtype=float)
        # re‑normalise to the target sum (height_sequence_1 may not be perfectly normalised)
        a = simplex_projection(a, target_sum)
    else:
        # Exponential distribution gives a nice spread of values
        a = np.random.exponential(scale=1.0, size=n)
        a = simplex_projection(a, target_sum)

    # Evaluate the starting point
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # Parameters for the sub‑gradient descent
    step_base = target_sum / n * 2.0           # roughly twice the average entry
    step_decay = 0.9997                        # slow geometric decay
    iteration = 0
    max_no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # Full convolution and current maximum
        conv = np.convolve(a, a)
        M = conv.max()

        # --------------------------------------------------------------
        # Compute a sub‑gradient of the max‑convolution
        # (indices where the peak is attained)
        K = np.where(conv >= M - 1e-12)[0]

        grad = np.zeros_like(a)
        n_len = a.shape[0]
        for k in K:
            i_start = max(0, k - (n_len - 1))
            i_end = min(n_len - 1, k)
            i_vals = np.arange(i_start, i_end + 1)
            j_vals = k - i_vals
            grad[i_vals] += a[j_vals]

        # Zero‑sum direction: move away from large gradient components
        grad_mean = grad.mean()
        direction = -(grad - grad_mean)

        # --------------------------------------------------------------
        # Step size decreasing with iteration number
        eta = step_base * (step_decay ** iteration)

        # Candidate move and projection back to the simplex
        cand = a + eta * direction
        cand = simplex_projection(cand, target_sum)

        # Evaluate candidate
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            # improvement – accept
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            max_no_improve = 0
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")
            continue
        else:
            max_no_improve += 1

        # --------------------------------------------------------------
        # If no improvement for a while, try a tiny random transfer
        if np.random.rand() < 0.2 or max_no_improve > 200:
            i, j = np.random.choice(n_len, size=2, replace=False)
            # Transfer a small proportion of mass from i to j
            delta = 0.01 * a[i] * (np.random.rand() - 0.5)   # signed, ~±0.5 % of a[i]
            # Keep non‑negativity
            delta = np.clip(delta, -a[i], a[i])
            cand2 = a.copy()
            cand2[i] -= delta
            cand2[j] += delta
            cand2 = simplex_projection(cand2, target_sum)
            cand2_score = evaluate_sequence(cand2.tolist())
            if cand2_score < best_score:
                best_score = cand2_score
                best_seq = cand2.copy()
                a = cand2
                max_no_improve = 0
                print(f"Iter {iteration:6d}: random tweak improved score = {best_score:.6f}")
                continue

        # --------------------------------------------------------------
        # Occasionally restart from a fresh random point (helps escape local minima)
        if iteration % 1500 == 0:
            a = np.random.exponential(scale=1.0, size=n_len)
            a = simplex_projection(a, target_sum)
            # Not resetting best_seq – we keep the best ever found
            print(f"Iter {iteration:6d}: random restart")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
