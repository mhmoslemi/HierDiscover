def solve_linearized(a_curr):
    n = len(a_curr)
    b_curr = np.convolve(a_curr, a_curr)
    rows = []
    cols = []
    data = []
    rhs = []
    # for each k
    for k in range(2*n-1):
        # for each i where i in [0,n-1] and j=k-i in [0,n-1]
        # coefficient for a_i is 2 * a_curr[j] where j = k-i
        # So we can iterate over i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            j = k - i
            coeff = 2 * a_curr[j]  # because a_{k-i} = a_j
            rows.append(k)
            cols.append(i)
            data.append(coeff)
        # add column for t with coefficient -1
        rows.append(k)
        cols.append(n)  # t variable index
        data.append(-1.0)
        rhs.append(b_curr[k])
    # Build sparse matrix
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    # equality sum a_i = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # bounds: a_i >=0, t >=0
    bounds = [(0, None)] * n + [(0, None)]
    result = linprog(c=np.concatenate([np.zeros(n), [1.0]]), A_ub=A_ub, b_ub=rhs,
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                    method='highs', options={'time_limit': 5.0})
    if result.success:
        a_new = result.x[:n]
        t_new = result.x[-1]
        return a_new, t_new
    else:
        return None

def max_convolution(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)[:2*n-1]
    return conv.max()
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random, math
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 500
    # start uniform
    best_a = np.full(n, 1.0/n)
    best_score = fast_evaluate(best_a)
    # maybe also include height_sequence_1 if available
    if 'height_sequence_1' in globals():
        cand = np.array(height_sequence_1, dtype=float)
        cand = np.maximum(cand, 0)
        if cand.sum() == 0:
            cand = best_a.copy()
        else:
            cand = cand / cand.sum()
        score = fast_evaluate(cand)
        if score < best_score:
            best_a = cand.copy()
            best_score = score
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        method = np.random.choice(['exp','power','two_block','dirichlet','perturb'], p=[0.2,0.2,0.2,0.3,0.1])
        if method == 'exp':
            # sample lambda
            lam = np.random.uniform(0.0, 0.02)  # up to 0.02 should not dominate
            r = np.exp(-lam)
            a = r ** np.arange(n, dtype=float)
            a_sum = a.sum()
            a = a / a_sum
        elif method == 'power':
            p = np.random.uniform(-5, 5)
            indices = np.arange(1, n+1, dtype=float)
            a = np.power(indices, p)
            a_sum = a.sum()
            a = a / a_sum
        elif method == 'two_block':
            m = np.random.randint(int(0.1*n), int(0.9*n)+1)
            f = np.random.uniform(0.0, 1.0)  # fraction of total mass in high block
            # compute high and low values
            high = f / m
            low = (1.0 - f) / (n - m) if n != m else 0.0
            a = np.empty(n, dtype=float)
            a[:m] = high
            a[m:] = low
        elif method == 'dirichlet':
            # concentration parameter alpha controlling dispersion
            alpha = np.random.uniform(0.1, 5.0)  # lower alpha => sparser
            a = np.random.dirichlet([alpha]*n)
        elif method == 'perturb':
            # small perturbation of current best
            a = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.uniform(-0.5, 0.5)
            # ensure not exceed bounds
            delta = max(-a[i]*0.999, delta)  # keep non-negative
            # also ensure not make a[j] exceed 1000 (never)
            a[i] -= delta
            a[j] += delta
        else:
            continue
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a < 1e-8:
            continue
        a = a / sum_a
        # Evaluate fast
        score = fast_evaluate(a)
        # occasional more exact evaluate
        if score < best_score:
            # verify with official evaluate_sequence
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    print(f"Iter {iter_count}: improved to {best_score:.6f}")
            except Exception:
                pass
        # also allow occasional random acceptance to escape; not needed.
        if iter_count % 1000 == 0:
            print(f"Iter {iter_count}, current best {best_score:.6f}, time left {deadline - time.time():.1f}s")
    return best_a.tolist()

"""
Fast heuristic search for a step‐function that reduces the evaluation
2·n·max(conv(a,a))/sum(a)^2.  The idea is to use simple parametrised families
(exponential, mixtures of exponentials, two‑block, power‑law, random Dirichlet)
and keep the best candidate found within the time budget.  A small local
hill‑climbing phase refines the best parameters.  The code works with any
length (the default is the length of the provided `height_sequence_1`).
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def _conv_max(a: np.ndarray) -> float:
    """
    Return max(a * a) using an FFT based convolution (real numbers only).
    """
    n = a.shape[0]
    # length of linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence`:  sum(a) is forced to 1,
    therefore the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (lam ≥ 0)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p  (any real p)."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    """
    First `m` entries have the same value, the remaining `n-m` entries
    share the rest of the mass.  `high_frac` is the fraction of total mass
    given to the first block.
    """
    if m <= 0 or m >= n:
        raise ValueError("invalid block size")
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    """
    Mixture of two exponentials:
        a_i ∝ w·exp(-lam1·i) + (1-w)·exp(-lam2·i)
    """
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a

# Main candidate generator
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation score.
    The function runs for at most `budget_s` seconds and returns the best
    sequence (as a list of floats) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a safety margin

    # ----------------------------------------------------------
    # baseline length and starting point
    # ----------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                              # fallback size

    # start with the uniform sequence (its score is exactly 2.0)
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_score = _fast_score(best_a)

    # also try the provided construction if it exists
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=float)
        cand = np.clip(cand, 0.0, None)       # enforce non‑negativity
        if cand.sum() > 0.0:
            cand = cand / cand.sum()
            sc = _fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

    # ----------------------------------------------------------
    # random search over parametrised families
    # ----------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1
        # pick a generation method
        method = np.random.choice(
            ["exp", "power", "two_block", "dirichlet", "mix_exp", "perturb"],
            p=[0.15, 0.15, 0.20, 0.25, 0.15, 0.10],
        )

        if method == "exp":
            # exponential decreasing, lam ∈ [0, 5/n]
            lam = np.random.uniform(0.0, 5.0 / n)
            a = _make_exponential(n, lam)

        elif method == "power":
            # power‑law, p ∈ [-5, 5]
            p = np.random.uniform(-5.0, 5.0)
            a = _make_power_law(n, p)

        elif method == "two_block":
            # two‑block shape
            m = np.random.randint(int(0.1 * n), int(0.9 * n) + 1)
            high_frac = np.random.uniform(0.0, 1.0)
            a = _make_two_block(n, m, high_frac)

        elif method == "dirichlet":
            # random Dirichlet (concentration controls sparsity)
            alpha = np.random.uniform(0.1, 5.0)
            a = np.random.dirichlet([alpha] * n)

        elif method == "mix_exp":
            # mixture of two exponentials
            w = np.random.uniform(0.0, 1.0)
            lam1 = np.random.uniform(0.0, 5.0 / n)
            lam2 = np.random.uniform(0.0, 5.0 / n)
            a = _make_mix_exponential(n, w, lam1, lam2)

        else:  # "perturb"
            # small random move from the current best
            a = best_a.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.uniform(-0.4, 0.4)
            delta = np.clip(delta, -a[i] + 1e-12, 1e12)  # keep a[i] ≥ 0
            a[i] -= delta
            a[j] += delta
            # renormalise to keep sum = 1
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum

        # ensure non‑negativity and normalise (just in case)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # Fast surrogate score
        score = _fast_score(a)

        # Keep the best – verify with the official evaluator only when an
        # improvement is seen (this saves a few milliseconds per iteration).
        if score < best_score:
            # exact check (the official function also clamps to the same limits)
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    # occasional progress report
                    elapsed = time.time() - start
                    print(
                        f"[{elapsed:6.1f}s] New best: {best_score:.6f} "
                        f"(iter {it}, method={method})"
                    )
            except Exception:
                # Should never happen, but just in case
                pass

        # occasional housekeeping output
        if it % 2000 == 0:
            elapsed = time.time() - start
            remaining = max(0.0, deadline - time.time())
            print(
                f"[{elapsed:6.1f}s] iter={it} best={best_score:.6f} "
                f"time left≈{remaining:.0f}s"
            )

    # ----------------------------------------------------------
    # final local refinement (tiny random tweaks)
    # ----------------------------------------------------------
    a = best_a.copy()
    for _ in range(2000):
        i, j = np.random.choice(n, 2, replace=False)
        # move a small fraction of mass from i to j
        delta = a[i] * np.random.uniform(-0.05, 0.05)
        delta = np.clip(delta, -a[i] + 1e-12, 1e12)
        a[i] -= delta
        a[j] += delta
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        score = _fast_score(a)
        if score < best_score:
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_a = a.copy()
            except Exception:
                pass

    # Return the best sequence found (as a plain Python list)
    return best_a.tolist()
