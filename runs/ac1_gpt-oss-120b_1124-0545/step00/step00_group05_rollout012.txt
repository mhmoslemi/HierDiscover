def project_simplex(v):
    # v is 1D array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

But careful: If rho is None, fallback.

import numpy as np
import time

def linear_convolution_fft(x, y):
    L = len(x) + len(y) - 1
    N = 1 << ((L - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def compute_eval(a):
    conv = linear_convolution_fft(a, a)
    return float(2 * len(a) * np.max(conv))

def project_simplex(v):
    # Euclidean projection onto simplex {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def generate_sequence_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    if a.sum() == 0:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a

def generate_sequence_power(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    if a.sum() == 0:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a

def generate_sequence_geometric_power(n, r, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = (r ** (idx-1)) / (idx ** p)
    if a.sum() == 0:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    return a

def generate_sequence_symmetric_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n-1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a non‑negative step‑function heights sequence that minimises the evaluation function.
    
    The algorithm explores several parametric families (geometric, power‑law, combined) and then
    refines the best candidate with a smoothed‑max gradient descent using FFT‑based convolutions.
    It respects the time budget and returns the best sequence found.
    """
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a buffer

    best_seq = None
    best_val = float('inf')

    # --- 1. Quick parametric sweep over a few lengths -----------------
    lengths = [500, 1000, 2000]  # modest sizes keep iterations cheap
    for n in lengths:
        # geometric series (r close to 1)
        # use parameter c = (1-r)*n where c in [0.5, 5]
        cs = np.linspace(0.5, 5.0, 30)
        for c in cs:
            r = 1.0 - c / n
            if r <= 0:
                continue
            a = generate_sequence_geometric(n, r)
            val = compute_eval(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"new best (geom) n={n} c={c:.3f} val={val:.6f}")
        # power‑law families
        ps = np.linspace(0.0, 2.0, 25)
        for p in ps:
            a = generate_sequence_power(n, p)
            val = compute_eval(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"new best (pow) n={n} p={p:.3f} val={val:.6f}")
        # combined geometric * power
        for c in cs:
            r = 1.0 - c / n
            if r <= 0:
                continue
            for p in ps:
                a = generate_sequence_geometric_power(n, r, p)
                val = compute_eval(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"new best (geo‑pow) n={n} c={c:.3f} p={p:.3f} val={val:.6f}")
        # symmetric geometric
        for c in cs:
            r = 1.0 - c / n
            if r <= 0:
                continue
            a = generate_sequence_symmetric_geometric(n, r)
            val = compute_eval(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"new best (sym‑geo) n={n} c={c:.3f} val={val:.6f}")

    # -----------------------------------------------------------------
    # 2. Refine the best candidate with smoothed‑max gradient descent.
    if best_seq is None:
        # fallback to a uniform start
        n = 1000
        best_seq = np.ones(n) / n
        best_val = compute_eval(best_seq)

    a = best_seq.copy()
    n = len(a)

    # parameters for the descent
    max_iters = 5000
    beta0 = 5.0      # start smoothing factor
    beta_factor = 1.02   # geometric increase each iteration
    eta0 = 0.5        # initial step size

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # convolution a*a
        conv = linear_convolution_fft(a, a)
        max_conv = np.max(conv)
        # check progress
        val = 2.0 * n * max_conv
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"refine iter {it} new best val={val:.6f}")

        # smoothed max weights
        beta = beta0 * (beta_factor ** it)
        # shift to avoid overflow
        shift = max_conv
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum

        # gradient = 2 * (a_rev * w) [slice]
        grad_full = linear_convolution_fft(a[::-1], w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # multiplicative update
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        # re‑normalise to sum to 1 (simple simplex projection)
        s = a.sum()
        if s <= 0:

"""
Propose a sequence of non‑negative heights that minimizes the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution.  The algorithm works in two
phases:

1.  A cheap parametric sweep over several families (geometric,
    power‑law, combined geometric‑power and a symmetric geometric) for a
    few different lengths.  Every candidate is normalised to sum to 1 and
    its score is computed with an FFT‑based convolution.

2.  The best candidate from the sweep is refined by a smoothed‑max
    gradient descent.  The loss ``log‑sum‑exp(beta * conv)`` approximates
    the maximum of the convolution; its gradient is computed efficiently
    by another FFT convolution.  A multiplicative update together with
    renormalisation (simple simplex projection) drives the sequence toward
    a flatter autocorrelation, lowering the evaluation value.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT.  Returns the full (len(x)+len(y)-1) vector.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _evaluate(a: np.ndarray) -> float:
    """Return the value of the objective for a sequence that already sums to 1."""
    conv = _conv_fft(a, a)
    return float(2.0 * len(a) * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x >=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        theta = cssv[-1] / n
    else:
        rho = np.where(mask)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------- families of parametric sequences ----------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64) + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a /= a.sum()
    return a


# ---------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function runs for roughly ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    best_seq = None
    best_val = float("inf")

    # 1.  Parametric sweep -------------------------------------------------
    lengths = [500, 1000, 2000, 4000]          # reasonable sizes for rapid search
    c_vals = np.linspace(0.5, 5.0, 30)        # controls r = 1 - c/n  (r close to 1)
    p_vals = np.linspace(0.0, 2.0, 25)        # power‑law exponent

    for n in lengths:
        # geometric series
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            a = _geom(n, r)
            val = _evaluate(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] geom  n={n:4d} c={c:5.2f}  val={val:.6f}")

        # pure power‑law
        for p in p_vals:
            a = _power(n, p)
            val = _evaluate(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] power n={n:4d} p={p:5.2f}  val={val:.6f}")

        # combined geometric * power
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in p_vals:
                a = _geom_power(n, r, p)
                val = _evaluate(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()
                    print(f"[sweep] g×p   n={n:4d} c={c:5.2f} p={p:5.2f}  val={val:.6f}")

        # symmetric geometric
        for c in c_vals:
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            a = _sym_geom(n, r)
            val = _evaluate(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[sweep] sym‑g n={n:4d} c={c:5.2f}  val={val:.6f}")

        if time.time() >= deadline:
            break

    # If nothing was found (should not happen), fall back to uniform
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = _evaluate(best_seq)

    # 2.  Smoothed‑max gradient descent refinement -------------------------
    a = best_seq.copy()
    n = len(a)

    max_iters = 5000
    beta0 = 5.0          # initial smoothing strength
    beta_mul = 1.02      # geometric increase each iteration
    eta0 = 0.5           # initial step size

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[refine] iter {it:4d} new best val={cur_val:.6f}")

        # smoothed max (log‑sum‑exp) weights
        beta = beta0 * (beta_mul ** it)
        shift = cur_max                     # avoid overflow
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient = 2 * (a_rev * w)  (slice of length n)
        grad_full = _conv_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # multiplicative update, then renormalise to the simplex
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.ones_like(a) / n
        else:
            a /= a_sum

    # Return the best sequence found (as a plain list)
    return best_seq.tolist()
