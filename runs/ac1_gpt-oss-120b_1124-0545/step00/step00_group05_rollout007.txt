def simplex_projection(v):
    v = np.asarray(v)
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max i such that u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected gradient descent on a smooth approximation of the maximal
convolution value.

The algorithm keeps the coefficients on the probability simplex
(non‑negative, sum = 1).  At each iteration we compute the self‑convolution,
use a softmax with a temperature β to obtain a smooth weight vector
that approximates the location of the maximum, compute the sub‑gradient
of the weighted sum via an FFT accelerated cross‑correlation,
take a small step opposite to the gradient and project back onto the
simplex.  The temperature β is slowly increased so that the optimisation
focuses first on a global smoothing of the distribution and later on the
actual maximum.  Occasional random perturbations and optional LP moves
are used to escape local minima.

The best sequence encountered during the run is returned.
"""
import numpy as np
import time
from scipy import optimize

def simplex_projection(v):
    """Project a vector onto the probability simplex {x >=0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def project_gradient_step(a, beta, step):
    """Perform one projected gradient step on the soft‑max approximated objective."""
    n = a.shape[0]
    # convolution a * a
    conv = np.convolve(a, a)          # length 2n-1
    # soft‑max weights (more emphasis on large entries as β grows)
    # shift for numerical stability
    shift = np.max(conv)
    w = np.exp(beta * (conv - shift))
    w /= w.sum()
    # cross‑correlation g_i = 2 * sum_j a_j * w_{i+j}
    L = 1
    while L < n + len(w) - 1:
        L <<= 1
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    cross = np.fft.irfft(np.conj(A) * W, n=L)
    g = 2.0 * cross[:n]               # real‑valued gradient
    # gradient descent (we minimise the objective, thus move opposite to g)
    a_new = a - step * g
    # projection onto simplex
    a_new = simplex_projection(a_new)
    return a_new, conv

def evaluate_fast(a):
    """Exact evaluation used inside the search (no clipping needed)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

Projected gradient descent on a smooth approximation of the maximal

The algorithm keeps the coefficients on the probability simplex
(non‑negative, sum = 1).  In each iteration:

2. Form a soft‑max weight vector w that concentrates on the largest

4. Take a small step opposite to g and project back onto the simplex.
