def project_onto_simplex(v, z=1):
    """Projects vector v onto the simplex {x >= 0, sum x = z}."""
    # based on algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time

def simplex_projection(v, z):
    """Project v onto the simplex {x >= 0, sum x = z}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(a):
    """Return the evaluation function value for a (assumes evaluate_sequence exists)."""
    # Using the analytic form for speed; evaluate_sequence applies clippings that
    # do not affect our valid ranges, so the two are identical.
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    maxc = conv.max()
    return 2.0 * n * maxc / (S * S)

def improve_sequence(a, n, target_sum, deadline):
    """Run a few thousand gradient‑descent steps to flatten the convolution."""
    best_a = a.copy()
    best_val = compute_score(a)
    # initial step size – roughly a fraction of the average entry
    eta = target_sum / n * 0.5
    # pre‑compute constant used later
    two_n_minus_one = 2 * n - 1

    while time.time() < deadline:
        # convolution and current objective
        conv = np.convolve(a, a)
        maxc = conv.max()
        val = 2.0 * n * maxc / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"  improvement: score={best_val:.6f}")

        # target mean of the convolution (average over all 2n‑1 entries)
        mu = (a.sum() ** 2) / two_n_minus_one
        diff = conv - mu

        # gradient of sum(diff**2) w.r.t a (steepest descent)
        corr = np.correlate(a, diff, mode='full')
        grad = 4.0 * corr[2 * n - 2: 2 * n - 2 + n]

        # adapt step length – we aim for a modest change of the vector
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        step = eta * (maxc - mu) / (grad_norm + 1e-12)

        # tentative update
        a_candidate = a - step * grad
        # project onto the simplex of fixed ℓ1 norm
        a_candidate = simplex_projection(a_candidate, target_sum)

        # evaluate the candidate
        cand_val = compute_score(a_candidate)

        if cand_val <= val:

            a = a_candidate
            eta *= 1.05
        else:
            # reject – shrink step size and try again
            eta *= 0.5
            # if step becomes tiny we abort the inner loop
            if eta < 1e-12:
                break
    return best_a, best_val

def _simplex_projection(v: np.ndarray, z: float) -> np.ndarray:
    """
    Project a vector onto the simplex  {x ≥ 0, Σ x = z}.
    Implementation of the O(n) algorithm by Duchi et al.
    """
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying u_i – cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """
    Fast computation of the objective 2*n*max(conv)/ (sum(a)^2).
    The values are always non‑negative and far below the clipping limits
    imposed inside ``evaluate_sequence``.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def _flatten_step(a: np.ndarray, target_sum: float,
                  eta: float, deadline: float) -> tuple:
    """
    One iteration of projected gradient descent that tries to make the
    convolution as flat as possible (minimise Σ (conv‑µ)^2).
    Returns the new vector, the new step size and the achieved score.
    """
    n = a.shape[0]
    # convolution
    conv = np.convolve(a, a)
    maxc = conv.max()
    cur_score = 2.0 * n * maxc / (a.sum() ** 2)

    # average value of the convolution (µ = Σconv / (2n‑1))
    mu = (a.sum() ** 2) / (2 * n - 1)
    diff = conv - mu

    # gradient of Σ diff^2 w.r.t a  (see analysis for derivation)
    # np.correlate(a, diff) gives Σ a_i * diff_{i+lag}
    corr = np.correlate(a, diff, mode='full')
    grad = 4.0 * corr[2 * n - 2: 2 * n - 2 + n]      # length n

    # adaptive step length – we scale it by the current excess over the mean
    grad_norm = np.linalg.norm(grad)
    if grad_norm == 0.0:
        return a, eta, cur_score    # nothing to do

    step = eta * (maxc - mu) / (grad_norm + 1e-12)

    # tentative update and projection onto the simplex (ℓ₁‑norm fixed)
    a_new = a - step * grad
    a_new = _simplex_projection(a_new, target_sum)

    # evaluate candidate
    new_score = _score(a_new)

    # accept only if we do not lose the current score
    if new_score <= cur_score:
        # a successful step – slightly increase the step size
        eta = max(eta, step) * 1.05
        return a_new, eta, new_score
    else:
        # reject – shrink step size, keep the old point
        eta *= 0.5
        return a, eta, cur_score


def _optimise(a0: np.ndarray, target_sum: float,
              deadline: float) -> tuple:
    """
    Run a few thousand projected‑gradient steps, keeping the best vector
    found so far.  The routine stops as soon as the time budget is exhausted.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(best_a)
    # initial learning‑rate (roughly a fraction of the average entry)
    eta = target_sum / a.shape[0] * 0.5

    while time.time() < deadline:
        a, eta, cur_val = _flatten_step(a, target_sum, eta, deadline)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"    inner improvement → score {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Gradient‑descent search for a non‑negative step function that minimises
    ``evaluate_sequence``.
    * fixes the ℓ₁‑norm to √(2·n) (the scaling does not affect the score);
    * starts from a few different random initialisations (uniform, Dirichlet,
      power‑law);
    * each start is refined by projected gradient descent aiming at a flat
      self‑convolution;
    * after each tentative move the vector is projected onto the simplex
      (non‑negative entries, fixed sum) so the constraints stay satisfied;
    * the best vector found before the time limit is returned.
    """
    # ------------------------------------------------------------------
    # problem size – use the length of the previously found construction if it exists
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        # fallback to a modest default length
        n = 2000
    target_sum = np.sqrt(2.0 * n)

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    # start from the SOTA construction if it is available
    try:
        best_vec = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        best_vec = np.clip(best_vec, 0.0, 1000.0)
        best_vec = best_vec * (target_sum / best_vec.sum())
        best_score = _score(best_vec)
        print(f"initial SOTA score {best_score:.6f}")
    except Exception:
        best_vec = None
        best_score = np.inf

    restart = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # choose an initial vector for this run
        # --------------------------------------------------------------
        if restart == 0:
            a0 = np.full(n, target_sum / n)                     # perfectly uniform
        elif restart == 1:
            a0 = np.random.rand(n)
            a0 = a0 * (target_sum / a0.sum())                   # Dirichlet‑like
        else:
            # power‑law (more mass at the beginning) – often a good start
            expo = np.random.rand(n) ** 2
            a0 = expo * (target_sum / expo.sum())

        # --------------------------------------------------------------
        # run the inner optimiser (will stop automatically on time out)
        # --------------------------------------------------------------
        a_opt, val_opt = _optimise(a0, target_sum, deadline)

        # --------------------------------------------------------------
        # keep the best vector seen across all restarts
        # --------------------------------------------------------------
        if val_opt < best_score:
            best_score = val_opt
            best_vec = a_opt.copy()
            print(f"restart {restart}: new global best {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Return a plain Python list – the external ``evaluate_sequence`` will
    # apply the same clipping rules that we used internally.
    # ------------------------------------------------------------------
    final_sequence = best_vec.tolist()
    final_score = evaluate_sequence(final_sequence)   # type: ignore[name-defined]
    print(f"final score reported by evaluate_sequence: {final_score:.6f}")
    print(f"total restarts performed: {restart}")
    return final_sequence
