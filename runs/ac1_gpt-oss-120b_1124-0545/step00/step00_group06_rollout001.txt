def propose_move(seq):
    n = len(seq)
    i = random.randint(0, n-1)
    # ensure source non-zero
    if seq[i] < 1e-12:
        # find another index with >0
        nonzero = [k for k in range(n) if seq[k] > 1e-12]
        if not nonzero:
            return None
        i = random.choice(nonzero)
    j = random.randint(0, n-1)
    while j == i:
        j = random.randint(0, n-1)
    max_transfer = seq[i]
    delta = random.random() * max_transfer * move_factor
    # create new seq copy
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq
```

def generate_initial_seq(n, mode):
    if mode == "random":
        seq = [random.random() for _ in range(n)]
    elif mode == "sin":
        p = random.uniform(0.5, 3.0)
        seq = [math.sin(math.pi * (i + 0.5) / n) ** p for i in range(n)]
    elif mode == "triangular":
        seq = [min(i+1, n-i) for i in range(n)]
    elif mode == "exp":
        sigma = random.uniform(0.1*n, 0.4*n)
        center = n/2
        seq = [math.exp(-((i - center) ** 2) / (2 * sigma ** 2)) for i in range(n)]
    else:
        seq = [random.random() for _ in range(n)]
    # ensure non-zero
    # shift small epsilon
    seq = [max(x, 1e-12) for x in seq]
    # normalize to sum sqrt(2n)
    scale = math.sqrt(2 * n) / sum(seq)
    seq = [x * scale for x in seq]
    return seq

                if new_val < best_val:
                    best_val = new_val
                    best_seq = seq.tolist()
                    print(
                        f"[{time.time():.1f}] total_iter {total_iters + inner_iters} – "
                        f"new best value {best_val:.6f}"
                    )

            inner_iters += 1
            if time.time() >= deadline:
                break

        total_iters += inner_iters

    # -------------------------------------------------------------------------
    # 4) Post‑processing – guarantee a valid, properly normalised output
    # -------------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence (value = 2)
        best_seq = [sqrt2n / n] * n
        best_val = max_convolution(np.array(best_seq, dtype=float))

    # Re‑normalise to counter any tiny drift
    total = sum(best_seq)
    if total <= 0:
        best_seq = [sqrt2n / n] * n
        total = sum(best_seq)
    if abs(total - sqrt2n) > 1e-9:
        scale = sqrt2n / total
        best_seq = [x * scale for x in best_seq]

    # Return a plain Python list (as required by evaluate_sequence)
    return best_seq
