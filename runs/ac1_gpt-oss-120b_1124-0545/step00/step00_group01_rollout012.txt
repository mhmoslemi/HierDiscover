"""
propose_candidate
-----------------

Search for a non‑negative step‑function that yields a low value of the
objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

The strategy mixes three ideas:

1. **Parametric families** –  Inverse‑square‑root shape
   a_i = (i+offset+1)^(-p) is known to give the classical bound
   2·n·max/∑² ≈ π/2 ≈ 1.57 (p≈0.5).  By sampling the exponent *p* around
   0.5 and a small integer *offset* we obtain many diverse candidates.

2. **Simple random exploration** –  Randomly vary the length, the
   exponent, the offset and also try a few other families
   (exponential decay, pure random vectors, an arcsine‑like shape).  
   Every candidate is evaluated with the supplied ``evaluate_sequence``,
   and the best one seen so far is kept.

3. **Local refinement using the LP “direction” step** –  For any promising
   sequence we invoke the existing LP routine
   ``get_good_direction_to_move_into`` (which implements the iteration
   described in the problem statement).  A handful of such LP steps can
   push the score further down without a large computational cost.

The whole search runs for at most ``budget_s`` seconds (default 990 s,
leaving a small safety margin).  Progress is printed so that the
interaction log shows how the bound improves.

The returned sequence is the best one discovered within the time limit.
"""

import numpy as np
import time
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear‑program used in the LP “direction” step
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """
    Solve  max  Σ b_j   s.t.  conv(f,b) ≤ rhs,  b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_sequence)
    # objective: maximize Σ b  →  minimize -Σ b
    c = -np.ones(n)

    # Build constraints conv(f,b) ≤ rhs  for each possible shift k
    a_ub = []
    b_ub = []

    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f[i] where i + j = k   ⇔   i = k - j
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  -b ≤ 0
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)

    # Use a modest time limit on each LP call (10 s max)
    res = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        options={"time_limit": 10.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Return a new sequence that moves a short step towards a direction
    found by the LP described in the statement.
    The step size t is kept small (0.01) to ensure stability.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None

    # Normalise to the traditional scale (∑a = √(2n))
    norm_seq = np.array(sequence) * np.sqrt(2.0 * n) / sum_seq
    rhs = np.max(np.convolve(norm_seq, norm_seq))

    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None

    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    norm_g = g * np.sqrt(2.0 * n) / sum_g
    t = 0.01
    new_seq = (1.0 - t) * np.array(sequence) + t * norm_g
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def generate_power_sequence(n, exponent=0.5, offset=0):
    """a_i = (i + offset + 1)^{-exponent}  for i = 0..n‑1."""
    idx = np.arange(n, dtype=float)
    seq = (idx + offset + 1.0) ** (-exponent)
    return seq.tolist()


def generate_exponential_sequence(n, base):
    """a_i = base^{i}  with 0 < base < 1."""
    idx = np.arange(n, dtype=float)
    seq = base ** idx
    return seq.tolist()


def generate_arcsine_like_sequence(n):
    """
    a_i ∝ 1 / sqrt( (i+0.5) * (n-i+0.5) ), the discrete analogue of the
    Beta(½,½) density.  It is symmetric and heavily weighted at the ends.
    """
    i = np.arange(n, dtype=float) + 0.5
    seq = 1.0 / np.sqrt(i * (n - i + 0.5))
    return seq.tolist()

    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1

        # ----- pick a family and random parameters --------------------

        elif family == "power_shift":

        except Exception:

            best_score = score

            if refined is not None:

                            f"[{iter_cnt:05d}] Perturb improved to "
                            f"{best_score:.6f}"
                        )
                except Exception:
                    pass

    print(f"[done] Best score = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq
