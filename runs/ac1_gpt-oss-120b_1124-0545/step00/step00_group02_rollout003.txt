def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a non-negative sequence minimizing evaluate_sequence.
    Uses an iterative LP-based improvement together with random local perturbations.
    """
    import numpy as np, time
    from scipy.optimize import linprog
    # use given seed
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # margin

    # Choose length
    n = 500  # moderate size that balances LP speed and bound potential
    # Initialize with constant sequence
    init_val = np.sqrt(2 * n) / n
    seq = np.full(n, init_val, dtype=float)  # sum = sqrt(2n)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # Helper: compute max conv and bound after normalization
    def normalize(v):
        s = v.sum()
        if s < 1e-12:
            return v
        return v * (np.sqrt(2*n) / s)

    # main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # ensure sequence is normalized to meet the sum sqrt(2n)
        f = normalize(seq)
        # compute current bound M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # If M already equals lower theoretical bound 2*n/(2*n-1) within tolerance, stop
        lower_theoretical = 2 * n / (2 * n - 1)
        # early break if already very good
        if best_score <= lower_theoretical * 1.02:
            print(f"Reached near theoretical bound after {iteration} iterations.")
            break

        # ----- LP step -----
        # build A matrix (2n-1) x n
        rows = []
        for k in range(2 * n - 1):
            row = np.zeros(n)
            # a_i contributes to b_{k-i}
            # we need indices j where 0<=j<n, i = k-j within [0,n-1]
            # i = k - j
            # for each i in 0..n-1, j = k-i
            # we fill row[j] = f[i] if j in range
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # loop through i
            for i in range(i_start, i_end + 1):
                j = k - i
                row[j] = f[i]
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M, dtype=float)

        # objective: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n, dtype=float)

        # solve LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 5})
        except Exception as e:
            print(f"LP failed with exception: {e}")
            res = None

        if res is not None and res.success:
            b = res.x
            sum_b = b.sum()
            if sum_b > 0:
                g = b * (np.sqrt(2 * n) / sum_b)
                # Consider several mixing coefficients
                for t in (0.2, 0.4, 0.6, 0.8):
                    cand = (1 - t) * f + t * g
                    cand = normalize(cand)   # keep total sum stable
                    score = evaluate_sequence(cand.tolist())
                    if score < best_score:
                        best_score = score
                        best_seq = cand.copy()
                        seq = cand.copy()
                        print(f"Iter {iteration}: improved score to {best_score:.6f} with t={t}")
                        break
        else:
            # LP didn't produce useful direction, perform small random tweak
            idx = np.random.randint(n)
            delta = np.random.randn() * 0.05 * seq[idx]
            seq[idx] = max(0.0, seq[idx] + delta)

        # Occasionally (10% of iterations) apply a random Gaussian smoothing to whole vector
        if iteration % 10 == 0:
            sigma = np.random.uniform(0.01, 0.05)
            pert = np.random.randn(n) * sigma * seq.mean()
            seq = np.maximum(0.0, seq + pert)
            seq = normalize(seq)

        # Check time
        if time.time() >= deadline:
            break

    print(f"Finished after {iteration} iterations, best score: {best_score:.6f}")
    # Return best sequence as plain list
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.
The algorithm mixes three ingredients:

1.  **Normalization** – the sequence is always scaled so that its
    L1‑norm equals √(2·n).  With this scaling the evaluation value equals the
    maximal self‑convolution ``M = max_k (f*f)[k]``.
2.  **LP‑move** – for the current (normalized) sequence *f* we solve a
    linear program that maximizes the total mass of a new vector *g*
    under the linear constraints ``(f*g)[k] ≤ M``.
    The optimiser pushes the mass of *g* towards the ends of the interval,
    where the convolution constraints are weakest.
3.  **Mixing & local perturbations** – we replace the current sequence by a
    convex combination ``(1‑t)·f + t·g`` for several mixing parameters
    ``t`` and keep the best improvement.  Occasionally a small random
    perturbation is applied to avoid stagnation.

The loop runs until the time budget expires (default 1000 s) and returns
the best sequence found.  The method is a direct implementation of the
state‑of‑the‑art iterative LP scheme, but with adaptive mixing,
periodic random noise and early‑stopping when the theoretical lower bound
``2·n/(2·n‑1)`` is approached.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _normalize_to_sqrt2n(vec: np.ndarray, n: int) -> np.ndarray:
    """Scale *vec* so that its L1‑norm becomes √(2·n)."""
    total = vec.sum()
    if total <= 0:
        return vec
    factor = np.sqrt(2.0 * n) / total
    return vec * factor


def _build_conv_ub_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
    ``(f * b)[k] = A[k] @ b``   for every shift k.
    """
    n = f.size
    rows = []
    # for each shift k we need the coefficients of b_j that appear
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over indices of f that participate, j = k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    return np.vstack(rows)


def _solve_lp_for_g(f_norm: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve    max sum(b)   s.t.   (f_norm * b)[k] ≤ M   for all k,
    b ≥ 0.
    Returns the optimal b (numpy array) or None if the LP fails.
    """
    n = f_norm.size
    A = _build_conv_ub_matrix(f_norm)
    b_ub = np.full(A.shape[0], M, dtype=float)

    # maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # high‑level LP solver (HiGHS) – fast and robust
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": 5.0, "presolve": True},
        )
    except Exception:      # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation value.
    Returns a plain ``list[float]`` of heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    #    Larger n gives a tighter theoretical lower bound (≈1),
    #    but also larger linear programmes.
    #    n = 500 offers a good trade‑off for the allotted time.
    # ------------------------------------------------------------------
    n = 500

    # ------------------------------------------------------------------
    # 2. Initialise the sequence.
    #    Prefer the provided SOTA sequence if it exists and is at least n long,
    #    otherwise start from a constant (which already yields score 2).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, list):
        hs = np.array(height_sequence_1[:n], dtype=float)
        if hs.size == n and np.all(np.isfinite(hs)):
            seq = _normalize_to_sqrt2n(hs, n)
        else:
            seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
    else:
        seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Normalise current candidate (keeps the L1‑norm fixed)
        # --------------------------------------------------------------
        f = _normalize_to_sqrt2n(seq, n)

        # current maximal convolution value M
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # theoretical minimal value for this length (uniform bound)
        theoretical_min = 2.0 * n / (2 * n - 1)

        # early exit if we are already very close to the bound
        if best_score <= theoretical_min * 1.02:
            print(f"[stop] achieved near‑theoretical bound after {iteration} iters.")
            break

        # --------------------------------------------------------------
        # 1) LP direction
        # --------------------------------------------------------------
        g_vec = _solve_lp_for_g(f, M)

        moved = False
        if g_vec is not None:
            # scale g to have the same L1‑norm as f
            g = _normalize_to_sqrt2n(g_vec, n)

            # try several mixing coefficients, keep the best improvement
            for t in (0.8, 0.6, 0.4, 0.2):
                cand = (1.0 - t) * f + t * g
                cand = _normalize_to_sqrt2n(cand, n)
                score = evaluate_sequence(cand.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    seq = cand.copy()
                    moved = True
                    print(f"[iter {iteration}] LP‑mix t={t:.2f} → score={score:.6f}")
                    break

        # --------------------------------------------------------------
        # 2) Small random perturbations (if LP gave no improvement)
        # --------------------------------------------------------------
        if not moved:
            # pick a random index and perturb it slightly
            idx = np.random.randint(n)
            # relative perturbation magnitude adapts to the current scale
            scale = max(seq[idx], 1e-8)
            delta = np.random.randn() * 0.05 * scale
            seq[idx] = max(0.0, seq[idx] + delta)
            seq = _normalize_to_sqrt2n(seq, n)

        # --------------------------------------------------------------
        # 3) Periodic global noise to escape plateaus
        # --------------------------------------------------------------
        if iteration % 10 == 0:
            sigma = np.random.uniform(0.01, 0.04) * seq.mean()
            noise = np.random.randn(n) * sigma
            seq = np.maximum(0.0, seq + noise)
            seq = _normalize_to_sqrt2n(seq, n)

        # occasionally report progress
        if iteration % 25 == 0:
            elapsed = time.time() - (deadline - budget_s + 5.0)
            print(f"[iter {iteration}] best score={best_score:.6f}, elapsed={elapsed:.1f}s")

    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    # ``evaluate_sequence`` is scale‑invariant, so returning the raw
    # (non‑clamped) values is fine.
    return best_seq.tolist()
