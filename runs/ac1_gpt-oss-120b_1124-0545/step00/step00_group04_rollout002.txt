    # choose n = random between 200 and 3000 maybe
    n = len(height_sequence_1) if start_from_sota else random within range

    # Let's create f as uniform: each entry = sqrt(2*n)/n
    f = np.full(n, np.sqrt(2*n)/n)
    # optionally apply random perturbations
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    iteration = 0
    while time.time() < deadline:
        # compute M: max conv of f with itself
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # solve LP for g
        g = solve_lp_for_g(f.tolist(), M)
        if g is None:
            # random perturbation fallback
            # Add small noise to random index
            idx = np.random.randint(n)
            f[idx] = max(0, f[idx] + np.random.randn() * 0.01)
            continue
        # compute optimal mixing coefficient
        alpha = find_best_alpha(f, g, conv_f)
        # update f
        f = (1-alpha) * f + alpha * g
        # renorm sum
        sum_f = np.sum(f)
        if sum_f == 0:
            f = np.full(n, np.sqrt(2*n)/n)
            continue
        scale = np.sqrt(2*n) / sum_f
        f = f * scale
        # evaluate
        curr_score = evaluate_sequence(f.tolist())
        if curr_score < best_score:
            best_score = curr_score
            best_seq = f.copy()
            print(f"Iter {iteration}: improved best score {best_score}")
        iteration += 1
    return best_seq.tolist()

def find_best_alpha(f,g,conv_f=None):
    # compute conv_f if not given
    if conv_f is None: conv_f = np.convolve(f, f)
    conv_fg = np.convolve(f,g)
    conv_gg = np.convolve(g,g)
    # compute quadratic expression for each k as function of alp:
    # h(alp) = (1-alp)^2 * conv_f[k] + 2*alp*(1-alp)*conv_fg[k] + alp^2 * conv_gg[k]
    # we need max over k of h(alp).
    # grid search over alp in [0,1] maybe 101 points.
    alphas = np.linspace(0, 1, 101)
    best_alpha = 0
    best_val = np.max(conv_f) # initial M
    for a in alphas:
        term = (1-a)**2 * conv_f + 2*a*(1-a)*conv_fg + a*a * conv_gg
        val = np.max(term)
        if val < best_val:
            best_val = val
            best_alpha = a
    return best_alpha

                seq = np.full(n, np.sqrt(2*n)/n)
                no_improve = 0
            continue

        seq = (1 - alpha) * seq + alpha * g
        # renormalize sum
        sum_seq = np.sum(seq)

            seq = np.full(n, np.sqrt(2*n)/n)

"""
LP‑driven direction + line‑search step‑function optimizer.

The algorithm iteratively
1. normalises the current sequence  (sum = √(2·n)),
2. computes its self‑convolution peak M,
3. solves a linear programme that finds a non‑negative direction `g`
   with maximal ℓ¹‑norm while keeping the linear constraints
   (f * g)(k) ≤ M   for every convolution index k,
4. performs a one‑dimensional line search on the mixing weight α
   to minimise the new peak of the self‑convolution of (1‑α)f+αg,
5. mixes, renormalises and keeps the best sequence seen.

Random perturbations and occasional restarts avoid stagnation.
The routine stops after the allotted wall‑clock time (default 1000 s)
or as soon as a score ≤ 1.503 is reached.
"""

import time
import numpy as np
from scipy import optimize, sparse

# Helper functions
# ----------------------------------------------------------------------


def build_convolution_matrix(f):
    """
    Build a sparse (2·n‑1) × n matrix A such that A @ b = np.convolve(f, b).
    Each row k corresponds to the linear constraint
        sum_i f[i] * b[k‑i] ≤ M   (with out‑of‑range indices ignored).
    """
    n = len(f)
    f = np.asarray(f, dtype=np.float64)

    # row indices: i + j  (i for f, j for b)
    i = np.arange(n, dtype=np.int32).reshape(-1, 1)      # (n,1)
    j = np.arange(n, dtype=np.int32).reshape(1, -1)      # (1,n)
    rows = (i + j).ravel()                               # length n*n
    cols = np.tile(np.arange(n, dtype=np.int32), n)      # length n*n
    data = np.repeat(f, n)                               # length n*n

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_g(f, rhs):
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f, b) ≤ rhs   (for every index)
                   b ≥ 0
    Returns the optimal vector b (as a NumPy array) or None on failure.
    """
    n = len(f)
    A = build_convolution_matrix(f)

    c = -np.ones(n, dtype=np.float64)            # maximise sum(b)  →  minimise -sum(b)
    b_ub = np.full(2 * n - 1, rhs, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is the fastest solver in SciPy; give it a modest time limit.
    result = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"time_limit": 4.0, "presolve": True},
    )
    if result.success:
        return result.x
    return None


def best_alpha_for_mix(f, g, conv_f=None):
    """
    Given two non‑negative vectors f and g, find α∈[0,1] that minimises
        max_k   ((1‑α)²·(f*f)_k + 2α(1‑α)·(f*g)_k + α²·(g*g)_k).

    A simple grid search (101 points) is sufficient for the time budget.
    """
    if conv_f is None:
        conv_f = np.convolve(f, f)
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)

    alphas = np.linspace(0.0, 1.0, 101)
    best_a = 0.0
    best_val = np.max(conv_f)   # current peak

    for a in alphas:
        # vectorised evaluation of the quadratic form for every convolution index
        cur = (1 - a) ** 2 * conv_f + 2 * a * (1 - a) * conv_fg + a ** 2 * conv_gg
        cur_max = np.max(cur)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a
    return best_a


def normalise_to_target(seq):
    """
    Scale `seq` so that its ℓ¹‑norm equals √(2·n) (the standard normalisation).
    """
    n = len(seq)
    target = np.sqrt(2.0 * n)
    s = np.sum(seq)
    if s == 0.0:
        return np.full(n, target / n)
    return seq * (target / s)

def propose_candidate(seed=42, budget_s=1000, **_kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine runs for at most `budget_s` seconds and
    returns the best sequence it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Initialise – try to start from the provided SOTA sequence if available.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is assumed to be a global list of floats.
        starter = np.array(height_sequence_1, dtype=np.float64)
        n = len(starter)
        use_sota = np.random.rand() < 0.55
    except Exception:
        # No global variable – fall back to a modest size.
        n = 2000
        starter = None
        use_sota = False

    if starter is not None and use_sota:
        f = normalise_to_target(starter)
        print(f"Starting from provided sequence (n={n})")
    else:
        # Uniform start – the simple constant sequence already yields score 2.
        f = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        print(f"Starting from uniform sequence (n={n})")

    # Keep the best result seen so far.
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    iteration = 0
    stagnation_counter = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. Compute current peak of the self‑convolution (M)
        # --------------------------------------------------------------
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # --------------------------------------------------------------
        # 2. LP to obtain a direction g with a larger ℓ¹‑norm
        # --------------------------------------------------------------
        g = solve_lp_for_g(f.tolist(), M)

        if g is None:
            # LP failure – apply a tiny random nudge to escape possible degenerate region.
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            continue

        g = np.asarray(g, dtype=np.float64)

        # --------------------------------------------------------------
        # 3. Find the optimal mixing coefficient α
        # --------------------------------------------------------------
        alpha = best_alpha_for_mix(f, g, conv_f)

        # If α is effectively zero, the LP direction does not improve the peak.
        # Apply a small random perturbation and keep track of stagnation.
        if alpha <= 1e-6:
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            stagnation_counter += 1
            if stagnation_counter >= 50:
                # Restart from a fresh uniform vector to avoid long dead ends.
                f = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
                stagnation_counter = 0
                print("Restarted from uniform due to stagnation.")
            continue

        # --------------------------------------------------------------
        # 4. Mix, renormalise and evaluate
        # --------------------------------------------------------------
        f = (1.0 - alpha) * f + alpha * g
        f = normalise_to_target(f)

        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            stagnation_counter = 0
            print(
                f"Iter {iteration:04d}: improved score = {best_score:.6f} "
                f"(α={alpha:.4f}, peak={np.max(np.convolve(f,f)):.6f})"
            )
            # Early exit if we have already reached the target bound.
            if best_score <= 1.503:
                print("Target bound achieved – terminating early.")
                break
        else:
            stagnation_counter += 1

        iteration += 1

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    return best_seq.tolist()
