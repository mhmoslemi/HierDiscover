"""
Local‐search optimizer for the convolution‑based evaluation function.

The algorithm maintains a non‑negative sequence with fixed ℓ₁‑norm.
At each step it:
  * computes the self‑convolution and finds the index k with maximal value,
  * identifies the pair (i, j) contributing most to that peak,
  * moves a small amount of mass from i and j to two currently smallest entries,
    thereby lowering the dominant convolution term,
  * accepts the move if it improves the score (or with a tiny
    simulated‑annealing probability to escape local minima).

Multiple random restarts with different lengths and initialisations are
performed until the time budget expires.  The best sequence found is
returned.  The score is exactly the same as `evaluate_sequence`
(the function is reproduced locally for speed).
"""

import time
import numpy as np


def _score(seq: np.ndarray) -> float:
    """
    Compute the objective 2 * n * max(conv(seq,seq)) / (sum(seq)**2).
    Values are guaranteed to be non‑negative and sum >= 0.01 in the
    search, so no special handling is required.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

                if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    seq = new_seq
                    cur_score = new_score
                    no_improve = 0
                else:
                    no_improve += 1

            # decay the step size slowly to allow fine‑grained tuning
            step_frac *= 0.9999

            if no_improve >= max_no_improve:
                break  # restart with a fresh random initialisation

    # ------------------------------------------------------------------
    # Return the best sequence found (fallback to a uniform sequence if
    # nothing was ever accepted – this should never happen).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()
