   set_positions = []
   sums = set()
   cur = 0
   while len(set_positions) < m_target and cur < N:

       conflict = False
       for x in set_positions:
           if (cur + x) in sums: # would duplicate sum with x+cur
               conflict = True
               break
       if not conflict:
           # also check duplicate sum with self (2*cur) not already in sums
           if (2*cur) in sums:
               conflict = True
       if not conflict:
           # add cur
           for x in set_positions:
               sums.add(cur + x)
           sums.add(2*cur)
           set_positions.append(cur)
       cur += 1
