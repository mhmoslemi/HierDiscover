def propose_candidate(seed=0, budget_s=1000):
    import numpy as np
    import time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # choose length
    if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
        base_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = np.random.randint(500, 3000)
        base_seq = np.random.rand(n) * 0.1  # small values

    # Ensure non-zero sum
    if np.sum(base_seq) < 0.1:
        base_seq += 0.1

    # Keep best
    best_seq = base_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    curr_seq = best_seq.copy()
    curr_score = best_score

    # set temperature schedule
    init_temp = best_score * 0.1
    temp = init_temp
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature cooling: exponential based on iteration count
        temp = init_temp * (0.99 ** (it / 1000.0))

        # propose move
        seq_new = curr_seq.copy()
        n = len(seq_new)

        # decide move type
        move_type = np.random.rand()
        if move_type < 0.7:
            # mass transfer between two indices
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            # transfer amount epsilon (relative)
            epsilon = np.random.exponential(scale=0.01) * np.max(seq_new)
            epsilon = min(epsilon, seq_new[i])
            seq_new[i] -= epsilon
            seq_new[j] += epsilon
        elif move_type < 0.9:
            # scale a random index up or down
            i = np.random.randint(n)
            factor = 1.0 + np.random.randn() * 0.1
            if factor < 0:
                factor = 0
            seq_new[i] *= factor
        else:
            # rearrange: swap two values
            i, j = np.random.choice(n, 2, replace=False)
            seq_new[i], seq_new[j] = seq_new[j], seq_new[i]

        # enforce non-negative
        if np.any(seq_new < 0):
            seq_new = np.maximum(seq_new, 0)

        # maybe random small smoothing
        if np.random.rand() < 0.02:
            # apply moving average filter of radius 1-2
            kernel = np.ones(np.random.randint(2,5))
            seq_new = np.convolve(seq_new, kernel, mode='same') / kernel.sum()

        # compute score
        try:
            new_score = evaluate_sequence(seq_new.tolist())
        except Exception:
            continue

        # accept if better or by SA criteria
        delta = new_score - curr_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (temp + 1e-12)):
            curr_seq = seq_new
            curr_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_seq = seq_new.copy()
                print(f"Iter {it}, new best score: {best_score:.6f}")

        # occasional restart if stuck
        if it % 5000 == 0 and curr_score > best_score * 1.1:
            # random perturb to escape
            seq_new = best_seq + np.random.randn(n) * 0.01
            seq_new = np.maximum(seq_new, 0)
            try:
                new_score = evaluate_sequence(seq_new.tolist())
                curr_seq = seq_new
                curr_score = new_score
            except:
                pass

    print(f"Finished: best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Simulated‑annealing search with O(n) convolution updates.

The evaluation function is
    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv` denotes the aperiodic convolution.
A naive call to ``evaluate_sequence`` would recompute the full
convolution (O(n²)) each time.  This implementation keeps the
convolution array up‑to‑date with cheap rank‑one updates, allowing
tens of thousands of iterations in the allotted time.

The algorithm starts from the provided ``height_sequence_1`` (if
available) or from a random non‑negative vector, and then repeatedly
performs one of several inexpensive moves:

* random mass transfer between two positions,
* a *targeted* transfer that tries to reduce the current peak of the
  convolution,
* scaling a single entry,
* a tiny global perturbation (full recompute of the convolution).

A simple simulated‑annealing acceptance rule permits occasional
up‑hill moves early on, while the temperature decays exponentially.
The best sequence encountered is returned at the end.

All numbers are kept in the interval [0,1000] (the same clipping that
``evaluate_sequence`` applies), and the sum is never allowed to drop
below 0.01, so the returned sequence is always valid.
"""

import numpy as np
import time
import math


def _fast_eval(seq: np.ndarray, conv: np.ndarray) -> float:
    """Return the same value as evaluate_sequence but without the safety checks."""
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * seq.size * max_b / (sum_a * sum_a)


def _apply_delta(seq: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Apply an additive change ``delta`` to ``seq[idx]`` and update the
    convolution array in‑place using the exact rank‑one formula:

        conv += 2 * delta * shifted(seq) + delta**2 * e_{2*idx}
    """
    n = seq.shape[0]
    # update the convolution slice before changing seq[idx]
    conv[idx: idx + n] += 2.0 * delta * seq
    conv[2 * idx] += delta * delta
    # finally update the sequence entry
    seq[idx] += delta


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """Return a sequence of non‑negative heights with low evaluation score.

    The function runs for at most ``budget_s`` seconds (leaving a small
    margin for the caller) and prints occasional progress messages.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise the starting sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and np.random.rand() < 0.6:
        seq = np.array(height_sequence_1, dtype=np.float64)
        # optionally trim a few elements to avoid pathological border effects
        if np.random.rand() < 0.3 and len(seq) > 200:
            cut = np.random.randint(1, min(50, len(seq) // 5))
            seq = seq[cut:-cut] if cut * 2 < len(seq) else seq
    else:
        # random start – length between 800 and 3000
        n0 = np.random.randint(800, 3000)
        seq = np.random.rand(n0) * 0.5   # modest initial magnitudes

    # enforce the safety limits used by ``evaluate_sequence``
    np.clip(seq, 0.0, 1000.0, out=seq)
    if seq.sum() < 0.1:
        seq += 0.1 / seq.size

    n = seq.size
    conv = np.convolve(seq, seq)          # full aperiodic convolution
    best_seq = seq.copy()
    best_conv = conv.copy()
    best_score = _fast_eval(best_seq, best_conv)

    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    it = 0
    init_temp = max(best_score * 0.1, 1e-6)   # a modest starting temperature

    while time.time() < deadline:
        it += 1
        # exponential cooling – 0.995 per 200 iterations
        temp = init_temp * (0.995 ** (it / 200.0))

        # --------------------------------------------------------------
        # Propose a move (copy‑on‑write to keep the current state unchanged)
        # --------------------------------------------------------------
        seq_new = seq.copy()
        conv_new = conv.copy()
        move_type = np.random.rand()

        # ------------------------------------------------------------------
        # 1) Random mass transfer between two distinct indices
        # ------------------------------------------------------------------
        if move_type < 0.45:
            i = np.random.randint(n)
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)

            max_transfer = seq_new[i]
            if max_transfer <= 0.0:
                continue
            eps = np.random.rand() * max_transfer * 0.5   # up to 50 % of a_i

            # i -> j
            _apply_delta(seq_new, conv_new, i, -eps)
            _apply_delta(seq_new, conv_new, j,  eps)

        # ------------------------------------------------------------------
        # 2) Targeted transfer aimed at the current convolution peak
        # ------------------------------------------------------------------
        elif move_type < 0.75:
            # locate the index of the current maximum of the convolution
            k_max = int(np.argmax(conv_new))

            # indices that actually contribute to conv[k_max]
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k_max - i_vals

            # product a_i * a_j for each contributing pair
            products = seq_new[i_vals] * seq_new[j_vals]
            if not products.size:
                continue
            best_local = int(np.argmax(products))
            i_src = i_vals[best_local]

            # pick a destination index different from the source
            j_dst = np.random.randint(n)
            while j_dst == i_src:
                j_dst = np.random.randint(n)

            max_transfer = seq_new[i_src]
            if max_transfer <= 0.0:
                continue
            eps = np.random.rand() * max_transfer * 0.5

            _apply_delta(seq_new, conv_new, i_src, -eps)
            _apply_delta(seq_new, conv_new, j_dst,  eps)

        # ------------------------------------------------------------------
        # 3) Scale a single entry (multiplicative perturbation)
        # ------------------------------------------------------------------
        elif move_type < 0.90:
            i = np.random.randint(n)
            factor = 1.0 + np.random.randn() * 0.08   # roughly ±8 %
            if factor < 0.0:
                factor = 0.0
            delta = (factor - 1.0) * seq_new[i]
            if delta == 0.0:
                continue
            _apply_delta(seq_new, conv_new, i, delta)

        # ------------------------------------------------------------------
        # 4) Tiny global Gaussian perturbation (full recompute)
        # ------------------------------------------------------------------
        else:
            perturb = np.random.randn(n) * 0.0008
            seq_new += perturb
            np.clip(seq_new, 0.0, 1000.0, out=seq_new)
            conv_new = np.convolve(seq_new, seq_new)

        # --------------------------------------------------------------
        # Clip values that may have left the allowed range.
        # (If clipping changed any entry we fall back to a full conv.)
        # --------------------------------------------------------------
        if np.any(seq_new > 1000.0) or np.any(seq_new < 0.0):
            np.clip(seq_new, 0.0, 1000.0, out=seq_new)
            conv_new = np.convolve(seq_new, seq_new)

        # --------------------------------------------------------------
        # Evaluation and acceptance test
        # --------------------------------------------------------------
        new_score = _fast_eval(seq_new, conv_new)
        delta_score = new_score - best_score   # we compare to the *best* for acceptance

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion (temperature may become very small)
            if temp > 0.0:
                prob = math.exp(- (new_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            seq = seq_new
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_seq = seq_new.copy()
                best_conv = conv_new.copy()
                print(f"[iter {it:07d}] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Every few thousand steps recompute the convolution from scratch
        # to avoid accumulation of floating‑point drift.
        # ------------------------------------------------------------------
        if it % 7000 == 0:
            conv = np.convolve(seq, seq)

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, best_score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
