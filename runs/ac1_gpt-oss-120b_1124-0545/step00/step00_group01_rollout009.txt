def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random, math

    np.random.seed(seed)

    deadline = time.time() + budget_s - 5 # some margin

    best_seq = None
    best_score = float('inf')
    
    # We'll try multiple lengths attempts.
    lengths = [50, 100, 200, 500, 800, 1200] # restrict max length to not too heavy.
    # random shuffle lengths

    while time.time() < deadline:
        n = random.choice(lengths)
        # Start either from uniform or random
        if random.random() < 0.3:
            p = np.full(n, 1.0/n)
        else:
            # Dirichlet random
            alpha = np.full(n, 0.5) # Dirichlet with low concentration to get sparse distribution
            p = np.random.dirichlet(alpha)
        
        # local search: hill climbing with simulated annealing
        # schedule: start temperature high and decay
        start_temp = 1.0
        end_temp = 0.001
        steps = 0
        max_steps = 5000
        while time.time() < deadline and steps < max_steps:
            # compute current score
            seq = p.tolist()
            cur_score = evaluate_sequence(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f'New best {best_score:.6f} at n={n} step {steps}')
            
            # propose new candidate
            # mix of random perturb and gradient descent
            if random.random() < 0.7:
                # random perturb
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                delta = (np.random.rand() - 0.5) * 0.2 * p[i] # fraction of p[i]
                new_pi = p[i] - delta
                new_pj = p[j] + delta
                if new_pi < 0:
                    delta = p[i]  # move all
                    new_pi = 0.0
                    new_pj = p[j] + delta
                # copy and assign
                p_new = p.copy()
                p_new[i] = new_pi
                p_new[j] = new_pj
                # renorm to sum=1 (should be unchanged)
            else:
                # gradient step
                # compute convolution
                conv = np.convolve(p, p)
                max_c = conv.max()
                # active indices within tolerance
                tol = 1e-12
                active = np.where(np.abs(conv - max_c) < 1e-12)[0]
                # compute gradient
                g = np.zeros_like(p)
                for k in active:
                    # i ranges over indices that make k-i in range
                    i_min = max(0, k-(n-1))
                    i_max = min(n-1, k)
                    # vectorized: indices = np.arange(i_min,i_max+1)
                    idx = np.arange(i_min, i_max+1)
                    j = k - idx
                    g[idx] += 2 * p[j]
                # step size decreasing
                lr = 0.1 * (1 - steps / max_steps)
                p_new = p - lr * g
                # enforce nonnegative
                p_new = np.maximum(p_new, 0.0)
                # renorm
                sum_p = p_new.sum()
                if sum_p > 0:
                    p_new = p_new / sum_p
                else:
                    # if all zero (unlikely), reinitialize
                    p_new = np.full(n, 1.0/n)

            # evaluate new candidate
            new_score = evaluate_sequence(p_new.tolist())
            # acceptance: SA
            elapsed = time.time()
            temp = start_temp * ( (end_temp/start_temp) ** (steps / max_steps) )
            # If better, accept, else accept with prob exp(-(new - cur)/temp)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # degrade acceptance probability
                prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
                if random.random() < prob:
                    accept = True
            if accept:
                p = p_new
                cur_score = new_score
            steps += 1

        # proceed to next random length
    return best_seq

def project_onto_simplex(v):
    # ensure v is numpy array
    # from https://arxiv.org/pdf/1101.6081.pdf (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Projected sub‑gradient descent with random restarts.

The evaluation metric for a non‑negative sequence `a` of length *n* is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the score is invariant under a global scaling of `a`,
we may restrict ourselves to probability vectors (`sum(a) = 1`).
The problem then reduces to minimising the peak of the
self‑convolution of a probability distribution on {0,…,n‑1}.
The algorithm repeatedly:

1. draws a random start (Dirichlet vector, i.e. a random probability
   distribution),
2. iteratively reduces the current maximal convolution value using a
   projected sub‑gradient step,
3. projects the perturbed vector back onto the probability simplex,
4. occasionally injects a tiny random jitter to escape shallow local
   minima,
5. keeps the best sequence seen over the whole time budget.

The sub‑gradient of  `M(p) = max_k (p * p)_k` is obtained from the set
of indices `k` attaining the maximum:
    ∂M/∂p_i = Σ_{k∈K} p_{k-i}   (only indices inside the support are used).

A simple O(n log n) projection onto the simplex is used after each
step.  Several different lengths `n` are tried (100, 200, 400, 800,
1600) and the best result overall is returned.
"""

import time
import random
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the simplex {x ≥ 0, sum(x)=1}."""
    # Duchi et al. (2008) – O(n log n) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # the last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# -----------------------------------------------------------------
# Core optimisation routine for a fixed length `n`
# -----------------------------------------------------------------
def _optimize_length(
    n: int,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.8,
) -> tuple[list[float], float]:
    """
    Run a projected sub‑gradient descent for a given length `n`.

    Returns the best sequence (as a list of floats) and its score.
    """
    # start from a random Dirichlet vector – sparsity factor 0.5 works well
    p = np.random.dirichlet(np.full(n, 0.5))

    # evaluate the starting point (score = 2 * n * max(conv))
    conv = np.convolve(p, p)
    best_score = 2.0 * n * conv.max()
    best_seq = p.copy()

    for t in range(1, max_iter + 1):
        # stop if we run out of time
        if time.time() >= time_deadline:
            break

        # ----------------------------------------------------
        # 1) compute the convolution and locate the active set
        # ----------------------------------------------------
        conv = np.convolve(p, p)
        M = conv.max()
        # treat values within a tiny relative tolerance as active
        tol = 1e-12 * (1.0 + M)
        active = np.where(conv >= M - tol)[0]

        # ----------------------------------------------------
        # 2) sub‑gradient of the max‑convolution
        # ----------------------------------------------------
        # indicator vector of the active indices
        indicator = np.zeros(2 * n - 1, dtype=float)
        indicator[active] = 1.0

        # Using convolution with the reversed probability vector gives
        # g_i = Σ_{k∈active} p_{k-i} (out‑of‑range terms are ignored automatically)
        p_rev = p[::-1]
        tmp = np.convolve(indicator, p_rev)          # length = 3n‑2
        g = tmp[(n - 1) : (n - 1 + n)]                # slice the needed part

        # ----------------------------------------------------
        # 3) sub‑gradient step + projection onto the simplex
        # ----------------------------------------------------
        step = step0 / math.sqrt(t)                   # diminishing step size
        v = p - step * g                               # descent direction
        p = _project_simplex(v)                        # keep p a probability vector

        # ----------------------------------------------------
        # 4) occasional random jitter to avoid stagnation
        # ----------------------------------------------------
        if t % 250 == 0 and random.random() < 0.15:
            jitter = np.random.normal(scale=0.01, size=n)
            p = np.maximum(p + jitter, 0.0)
            p = p / p.sum()

        # ----------------------------------------------------
        # 5) evaluate current candidate (exactly as the judge does)
        # ----------------------------------------------------
        cur_score = 2.0 * n * np.max(np.convolve(p, p))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
    return best_seq.tolist(), best_score

    """
    Return a step‑function (list of non‑negative heights) that
    minimises the provided `evaluate_sequence` function.

    The implementation uses a series of projected sub‑gradient
    descents on random initialisations for several problem sizes.
    """
    np.random.seed(seed)
    random.seed(seed)

    # allow a small margin for the wrapper to finish cleanly
    deadline = time.time() + budget_s - 2.0

    # lengths that have shown a good trade‑off between speed and quality
    lengths = [100, 200, 400, 800, 1600]
    random.shuffle(lengths)

    global_best_seq = None
    global_best_score = float("inf")

    # keep trying new starts until the time budget expires
    while time.time() < deadline:
        n = random.choice(lengths)

        # allocate a portion of the remaining time for this run
        per_run_deadline = min(deadline, time.time() + 0.15 * (deadline - time.time()))
        # the number of internal iterations is scaled with the problem size
        max_iter = max(300, int(120000 / n))

        seq, score = _optimize_length(n, per_run_deadline, max_iter=max_iter)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq
            print(
                f"[{time.strftime('%H:%M:%S')}] New global best: "
                f"{global_best_score:.6f} (n={n})"
            )

    # fallback – a uniform sequence of modest length – should never be used
    if global_best_seq is None:
        n = 50
        global_best_seq = [1.0 / n] * n

    # the judge expects a plain Python list of floats
    return global_best_seq
