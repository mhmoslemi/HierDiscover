function gradient_step(a, eta):
    // compute convolution a*a
    b = np.convolve(a, a)
    max_val = b.max()
    # identify indices where b is maximum
    max_idx = np.where(b == max_val)[0]
    # compute gradient: grad_i = 2 * sum_{k in max_idx} a_{k-i}
    n = len(a)
    grad = np.zeros(n)
    for k in max_idx:
        # compute contributions
        i_start = max(0, k - (n-1))  # i such that j = k-i within range
        # Actually for each i from max(0,k-(n-1)) to min(n-1,k)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # Then j = k - i
        # gradient contribution: 2 * a_j
        # So we can loop over i
        # Could vectorize using slice assignment
        idx_range = np.arange(i_low, i_high+1)
        j = k - idx_range
        grad[idx_range] += 2 * a[j]
    # we have grad (subgradient).
    # Move opposite direction:
    a_new = a - eta * grad
    # clip to non-negative (or after projection onto simplex).
    # But moving opposite direction will reduce a_i with large gradient (i.e., those that contribute heavily to max peaks) and increase those with small gradient (since we have to keep sum). The projection will handle.
    a_proj = project_onto_simplex(a_new)  # ensures sum =1, nonneg.
    return a_proj

def project_simplex(v):
    # v: array, project onto simplex {x >=0, sum x = 1}
    # Implementation from Duchi et al. 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v))+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Now the gradient step:

def gradient_step(a, eta):
    n = len(a)
    b = np.convolve(a, a)
    max_val = b.max()
    max_idx = np.flatnonzero(b == max_val)
    # compute grad
    grad = np.zeros(n)
    for k in max_idx:
        # indices i from max(0,k-(n-1)) to min(n-1,k)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high+1)
        j = k - i_range
        grad[i_range] += 2 * a[j]
    # gradient direction for decreasing max
    a_new = a - eta * grad
    # project onto simplex
    a_proj = project_simplex(a_new)
    return a_proj

"""
Simple stochastic optimizer for the step‑function evaluation problem.

The algorithm mixes three complementary moves:

* ``reduce_max_lp`` – a linear program that directly minimizes the
  current convolution peak while keeping the total sum unchanged.
* ``increase_sum_lp`` – the LP from the reference implementation that
  tries to increase the ℓ₁‑norm without increasing the peak.
* a tiny random mass‑transfer perturbation for exploration.

The current best sequence (according to ``evaluate_sequence``) is kept
throughout the search and returned when the time budget expires.
The routine works for any sequence length; we use a moderate size
(≈300‑500) to keep the LP cheap, but the final candidate may be of any
length.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog

# Helper: project onto simplex (non‑negative, sum = 1)
def _project_onto_simplex(v):
    """Projects the vector v onto the probability simplex."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

# LP that tries to increase the ℓ₁‑sum without raising the convolution bound.
def _solve_increase_sum_lp(f_seq, rhs):
    """Maximise ∑b subject to b≥0 and conv(f,b) ≤ rhs."""
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum(b)  ⇔ minimise -sum(b)
    # Build the convolution constraints Conv(f, b) ≤ rhs
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)
    # Non‑negativity is handled via variable bounds
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=np.vstack(rows), b_ub=np.array(rhs_vec), bounds=bounds,
                     method='highs', options={'disp': False, 'time_limit': 10})
    if result.success:
        return result.x
    return None

def _get_good_direction_to_move_into(sequence):
    """LP step used by the reference implementation (increase‑sum direction)."""
    n = len(sequence)
    s = np.sum(sequence)
    if s < 1e-12:
        return None
    normalized = [x * np.sqrt(2 * n) / s for x in sequence]
    rhs = np.max(np.convolve(normalized, normalized))
    g = _solve_increase_sum_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    normalized_g = [x * np.sqrt(2 * n) / sum_g for x in g]
    # small mixing factor
    t = 0.05
    return [(1 - t) * x + t * y for x, y in zip(sequence, normalized_g)]

# New LP that directly decreases the current maximum of the convolution.
def _solve_reduce_max_lp(f_seq):
    """
    Solve the LP:
        minimise t
        subject to conv(f, d) * 2 + conv(f, f) ≤ t  (for all k)
                 sum(d) = 0
                 d ≥ -f   (so f+d stays non‑negative)
    The returned sequence is f+d (renormalised to sum 1).
    """
    n = len(f_seq)
    conv_f = np.convolve(f_seq, f_seq)
    # Variables: d_0 … d_{n‑1}, t   → total n+1 variables
    var_cnt = n + 1
    t_idx = n

    # Build inequality matrix A_ub * x ≤ b_ub
    rows = []
    rhs = []
    # 1)   2·conv(f, d) - t ≤ -conv_f
    for k in range(2 * n - 1):
        row = np.zeros(var_cnt)
        # coefficients for d_j: 2·f_{k‑j}
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] += 2.0 * f_seq[i]
        row[t_idx] = -1.0
        rows.append(row)
        rhs.append(-conv_f[k])

    # 2)   -d_i ≤ f_i   (i.e. d_i ≥ -f_i)
    for i in range(n):
        row = np.zeros(var_cnt)
        row[i] = -1.0
        rows.append(row)
        rhs.append(f_seq[i])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

                     method='highs',

    if not result.success:

    if t_val >= current_max - 1e-12:
        return None
    new_f = f_seq + d
    new_f = np.maximum(new_f, 0.0)

    # Renormalise to sum‑1 (the equality already enforces the sum, numerical errors may appear)
    total = new_f.sum()
    if total <= 0:
        return None
    new_f = new_f / total
    return new_f.tolist()

"""
Simple stochastic optimizer for the step‑function evaluation problem.

The algorithm mixes three complementary moves:

* ``_solve_reduce_max_lp`` – a linear program that directly lowers the
  current convolution peak while keeping the ℓ₁‑sum unchanged.
* ``_get_good_direction_to_move_into`` – the LP from the reference
  implementation that tries to increase the sum without increasing the
  peak.
* a tiny random mass‑transfer perturbation for exploration.

The best sequence encountered (according to ``evaluate_sequence``) is
kept and returned when the allocated time budget expires.  The routine
works for any sequence length; we use a moderate size (≈200‑400) to keep
the LP cheap, but the final candidate may be of any length.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the largest rho satisfying the KKT condition
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# LP that tries to increase the ℓ₁‑sum without raising the convolution bound
# (the “good direction” used by the reference implementation).
# ----------------------------------------------------------------------
def _solve_increase_sum_lp(f_seq: np.ndarray, rhs: float):
    """Maximise Σb subject to b ≥ 0 and conv(f, b) ≤ rhs."""
    n = f_seq.size
    c = -np.ones(n)                     # minimise –Σb  ⇔ maximise Σb
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs_vec)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method="highs")
    if res.success:
        return res.x
    return None


def _get_good_direction_to_move_into(sequence):
    """Return a new sequence after a small LP‑based mixing step."""
    n = len(sequence)
    s = np.sum(sequence)
    if s < 1e-12:
        return None
    # normalise to the convention used in the paper (∑a = √(2n))
    normalized = np.array(sequence) * np.sqrt(2 * n) / s
    rhs = np.max(np.convolve(normalized, normalized))
    g = _solve_increase_sum_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    norm_g = g * np.sqrt(2 * n) / sum_g
    t = 0.05                      # small mixing coefficient
    return list((1 - t) * np.array(sequence) + t * norm_g)


# ----------------------------------------------------------------------
# New LP that directly reduces the current maximum of the convolution
# ----------------------------------------------------------------------
def _solve_reduce_max_lp(f_seq):
    """
    Solve the LP:
        minimise t
        subject to 2·conv(f, d) + conv(f, f) ≤ t   (for all k)
                 Σ d_i = 0
                 d_i ≥ –f_i   (so f+d stays non‑negative)

    Returns f+d (renormalised to sum‑1) if a strict improvement is found,
    otherwise None.
    """
    n = f_seq.size
    conv_f = np.convolve(f_seq, f_seq)
    var_cnt = n + 1                     # d_0 … d_{n-1}, t
    t_idx = n

    rows = []
    rhs = []

    # 1)  2·conv(f, d) – t ≤ –conv_f   (one row per convolution index)
    for k in range(2 * n - 1):
        row = np.zeros(var_cnt)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] += 2.0 * f_seq[i]            # coefficient for d_j
        row[t_idx] = -1.0                       # coefficient for t
        rows.append(row)
        rhs.append(-conv_f[k])

    # 2)  –d_i ≤ f_i   ⇔   d_i ≥ –f_i   (ensure non‑negativity of f+d)
    for i in range(n):
        row = np.zeros(var_cnt)
        row[i] = -1.0
        rows.append(row)
        rhs.append(f_seq[i])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # Equality: Σ d_i = 0   (preserve total mass)
    A_eq = np.zeros((1, var_cnt))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Variable bounds: d_i free, t ≥ 0
    bounds = [(None, None)] * n + [(0, None)]

    # Objective: minimise t
    c = np.zeros(var_cnt)
    c[t_idx] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method="highs")
    if not res.success:
        return None

    d = res.x[:n]
    t_val = res.x[t_idx]
    if t_val >= np.max(conv_f) - 1e-12:          # no improvement
        return None

    new_f = f_seq + d
    new_f = np.maximum(new_f, 0.0)

    # Renormalise to sum‑1 (numerical drift may appear)
    total = new_f.sum()
    if total <= 0.0:
        return None
    new_f /= total
    return new_f.tolist()
