"""
Propose a step‑height sequence that minimizes the evaluation function.

The algorithm runs a fast, directed hill‑climbing / simulated‑annealing
search on the following observation:

For a fixed L1‑norm S = sqrt(2·n) the evaluation equals the peak of the
auto‑convolution M = max_k (a * a)[k].  The search therefore tries to
flatten this convolution.  At each iteration the index k where the
convolution is maximal is identified.  From the contributing pairs
(i, k‑i) we pick a “donor’’ index that contributes the most to the peak
and a “receiver’’ index that contributes the least (or is very small).  A
small amount of mass δ is moved from donor to receiver.  The convolution
is updated analytically in O(n) time, which makes it possible to evaluate
millions of moves within the 1000 s budget.

The move is accepted if it reduces the peak, otherwise it may still be
accepted with a temperature‑dependent probability (simulated annealing)
to escape local minima.  The current best sequence is stored and finally
returned.
"""

import time
import numpy as np

def _project_to_simplex(v, s):
    """
    Projects the vector v onto the simplex {x >= 0, sum(x) = s}.
    This is used only occasionally to correct numerical drift.
    """
    # Standard O(n log n) algorithm
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        donor = idx_range[np.argmax(contrib)]

        # 4. pick a receiver – a position with relatively little mass
        #    (choose among the 10 % smallest entries to keep moves focused)
        low_thr = np.percentile(a, 10)
        low_candidates = np.where(a <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = np.argmin(a)
        else:
            receiver = np.random.choice(low_candidates)
        if receiver == donor:
            # fall back to the absolute minimum if we accidentally chose the donor
            receiver = np.argmin(a + (np.arange(n) == donor) * 1e6)

        # 5. amount of mass to transfer
        max_delta = a[donor] * move_frac
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()   # random fraction of the allowed move

        # ------------------------------------------------------------------
        # 6. Efficiently update the convolution for this move
        # ------------------------------------------------------------------
        conv_new = conv.copy()

        # linear part: 2·δ·(a_{k‑receiver} – a_{k‑donor})
        # donor contribution
        start = donor
        end = min(donor + n, L)
        idx = np.arange(start, end, dtype=int)
        conv_new[idx] -= 2.0 * delta * a[idx - donor]

        # receiver contribution
        start = receiver
        end = min(receiver + n, L)
        idx2 = np.arange(start, end, dtype=int)
        conv_new[idx2] += 2.0 * delta * a[idx2 - receiver]

        # quadratic correction (exact)
        conv_new[2 * donor] += delta * delta
        conv_new[2 * receiver] += delta * delta
        conv_new[donor + receiver] -= 2.0 * delta * delta

        new_max = conv_new.max()

        # ------------------------------------------------------------------
        # 7. Simulated‑annealing acceptance test
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 0.5)
        temperature = max(1e-4, initial_temp * (1.0 - progress))

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # uphill move accepted with probability exp(-(Δ)/T)
            prob = np.exp(-(new_max - cur_max) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[donor] -= delta
            a[receiver] += delta
            conv = conv_new
            cur_max = new_max
            iter_cnt += 1
            stagnation = 0

            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] iter {iter_cnt:,d} – new best max = {best_max:.6f}")

        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # 8. Periodic re‑normalisation (keeps sum exactly target_sum)
        # ------------------------------------------------------------------
        if stagnation % 5000 == 0:
            # renormalise to the exact target L1‑norm; this does not
            # change the evaluation value because it scales both sum and
            # convolution by the same factor.
            a = _project_to_simplex(a, target_sum)
            conv = np.convolve(a, a)
            cur_max = conv.max()

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"Finished: best max = {best_max:.6f} after {iter_cnt:,d} accepted moves")
    return best_a.tolist()
