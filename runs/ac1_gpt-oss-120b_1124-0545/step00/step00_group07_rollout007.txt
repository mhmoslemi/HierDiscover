def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    import time
    start = time.time()
    deadline = start + budget_s - 5

    # Choose length n (maybe 2000)
    n = 2000
    # initialize a (uniform)
    a = np.full(n, 1.0/n)  # sum=1

    # initial best
    b = np.convolve(a, a)
    best_score = 2 * n * np.max(b) / (np.sum(a)**2)  # can compute directly
    best_seq = a.copy()

    iter = 0
    step0 = 0.1
    beta = 2.0

    while time.time() < deadline:
        # compute conv
        b = np.convolve(a, a)
        # compute score
        score = 2 * n * np.max(b) / (np.sum(a)**2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iter}: new best {best_score:.6f}")

        # softmax weights
        maxb = np.max(b)
        exp_terms = np.exp(beta * (b - maxb))
        w = exp_terms / np.sum(exp_terms)

        # compute gradient using correlation
        # compute convolution of a with reversed w
        conv_full = np.convolve(a, w[::-1])
        # slice
        grad = 2 * conv_full[n-1:2*n-1][::-1]  # length n

        # adaptive step size
        step = step0 / np.sqrt(iter + 1)
        # update
        a_new = a - step * grad
        # enforce non-negative
        a_new = np.maximum(a_new, 0.0)
        # renormalize to sum=1
        sum_a_new = np.sum(a_new)
        if sum_a_new <= 1e-12:
            # reinitialize random
            a_new = np.random.rand(n)
            a_new = a_new / np.sum(a_new)
            sum_a_new = 1.0
        a = a_new / sum_a_new  # keep sum=1

        # occasionally increase beta to sharpen focus
        if iter % 100 == 0 and beta < 50:
            beta *= 1.05

        iter += 1

    # after loop, evaluate best_seq with evaluate_sequence before returning
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final best score (evaluate_sequence): {final_score}")
    except Exception as e:
        print("Error evaluating final sequence:", e)

    return best_seq.tolist()

"""
Propose a candidate step‑function using a smooth‐approximation of the
max‑convolution objective and projected gradient descent.

The evaluation metric is
    E(a) = 2·n·max(a∗a) / (sum(a))²
where “∗” is the full discrete convolution.
Because the expression is scale–invariant we fix sum(a)=1 and minimise
the maximum entry of the convolution.  The exact max is non‑smooth, so we
replace it by a softmax (log‑sum‑exp) with a temperature β that is raised
during the run.  The gradient of this smooth surrogate is

    ∂/∂a_i  Mβ(a) = 2 * Σ_k  w_k  a_{k-i},
    w_k = exp(β·(b_k - max(b))) / Σ_j exp(β·(b_j - max(b))) ,

where b = a∗a.

Each iteration:
  1. Compute b = a∗a (via FFT for speed).
  2. Form the soft‑max weights w.
  3. Obtain the gradient by correlating a with w (again via FFT).
  4. Take a small step opposite the gradient, project onto the simplex
     (non‑negative and sum = 1), and enforce symmetry a_i = a_{n‑1‑i}.

The step size decays like 1/√t and the temperature β grows slowly,
allowing the algorithm to first explore the landscape and then focus on
flattening the convolution.  Random restarts are performed when the
algorithm stalls for many iterations.  The best sequence seen within the
time budget is returned.

The method is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence` for the final verification, and runs well
within the 1000 s allocation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[: len(x) + len(y) - 1]
    return conv


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries with temperature beta."""
    # shift by max to avoid overflow
    m = np.max(b)
    exp_term = np.exp(beta * (b - m))
    w = exp_term / np.sum(exp_term)
    return w


def _gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient d/d a of the soft‑max surrogate.
    It is 2 * correlation(a, w) restricted to the first n entries.
    """
    n = a.shape[0]
    # correlation = convolution with reversed w
    conv = _full_convolution_fft(a, w[::-1])          # length = 3n-2
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse to obtain g_i
    grad = 2.0 * conv[n - 1 : 2 * n - 1][::-1]
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the probability simplex."""
    # standard algorithm (Michelot / Duchi et al.)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate_score(a: np.ndarray) -> float:
    """Exact evaluation used for bookkeeping (identical to evaluate_sequence up to scaling)."""
    n = a.shape[0]
    b = np.convolve(a, a)           # full convolution, cheap for our sizes
    return float(2.0 * n * np.max(b) / (np.sum(a) ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation metric.  The algorithm uses a temperature‑controlled
    soft‑max surrogate of the max‑convolution and performs projected gradient
    descent with occasional random restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Parameters (tuned empirically)
    # ------------------------------------------------------------------
    n = 2000                     # length of the sequence; fits comfortably in time budget
    step0 = 0.15                 # initial step size for additive updates
    beta_start = 0.6             # start temperature for soft‑max
    beta_max = 30.0              # upper bound for temperature
    beta_inc = 0.02              # incremental increase per 100 iterations

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    beta = beta_start
    # start from a symmetric uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # random‑restart bookkeeping
    last_improve_iter = 0
    max_no_improve = 4000        # after this many iterations, re‑initialise

    while time.time() < deadline:
        # ----- evaluate current point -------------------------------------------------
        score = _evaluate_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"[{iteration:07d}] New best: {best_score:.6f}")

        # stop early if we have already reached the target (optional)
        if best_score < 1.503:      # target from the statement; can stop sooner
            break

        # ----- compute soft‑max weights ------------------------------------------------
        b = _full_convolution_fft(a, a)               # full convolution (length 2n‑1)
        w = _softmax_weights(b, beta)

        # ----- gradient of the surrogate ------------------------------------------------
        grad = _gradient(a, w)

        # ----- step size schedule (1/√t) -----------------------------------------------
        step = step0 / np.sqrt(iteration + 1.0)

        # ----- simple additive projected gradient step ----------------------------------
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)                # non‑negative
        a_new = _project_to_simplex(a_new)            # enforce sum = 1
        # enforce symmetry (helps keep convolution close to constant)
        a_new = (a_new + a_new[::-1]) * 0.5

        a = a_new

        # ----- temperature schedule -----------------------------------------------------
        if iteration % 100 == 0 and beta < beta_max:
            beta = min(beta_max, beta * (1.0 + beta_inc))

        # ----- occasional random restart if stuck --------------------------------------
        if iteration - last_improve_iter > max_no_improve:
            # re‑initialise with a small random perturbation around the current best
            if best_seq is not None:
                a = best_seq + 0.02 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a = _project_to_simplex(a)
                a = (a + a[::-1]) * 0.5
            else:
                a = np.random.rand(n)
                a = _project_to_simplex(a)
                a = (a + a[::-1]) * 0.5
            last_improve_iter = iteration
            print(f"[{iteration:07d}] Restarted (no improvement)")

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification with the supplied evaluate_sequence function
    # ------------------------------------------------------------------
    if best_seq is not None:
        try:
            final_score = evaluate_sequence(best_seq.tolist())
            print(f"Final evaluated score (evaluate_sequence): {final_score:.6f}")
        except Exception as exc:
            print("Error during final evaluation:", exc)

    # Return the best sequence found as a plain Python list
    return best_seq.tolist() if best_seq is not None else []
