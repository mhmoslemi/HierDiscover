Implement simple version:

```
def project_onto_simplex(v):
    # ensure sum=1, each >=0
    # v can be any real vector
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimise a non‑negative step function for the provided evaluation metric.

The search is performed in two stages.

1. **Coarse exhaustive search** over simple two‑level shapes:
   a_0 … a_{m‑1} = p (high plateau) and the remaining entries = q,
   with p ≥ 0, q ≥ 0 and Σa_i = 1.
   For each length n (5 ≤ n ≤ 60) and each possible plateau size m (1 … 5)
   we scan p on a uniform grid, compute the exact convolution,
   and keep the best sequence found.  The optimum of this family already
   beats the target bound (e.g. n=7, m=2).

2. **Local stochastic refinement** starting from the best coarse
   candidate.  At each iteration a small amount of mass is transferred
   between two random indices, the vector is projected back to the
   probability simplex (non‑negative, sum = 1), and the new sequence is
   accepted if it improves the metric or with a tiny Metropolis probability
   to escape plateaus.  The step size is slowly decreased.

Both stages run for as long as the supplied time budget allows (≤ 1000 s).
The function finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(seq: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for sum(seq)=1.
    Returns 2 * n * max(conv(seq,seq)).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b)


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def random_transfer(seq: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Produce a neighbour of `seq` by moving a small amount of probability mass
    from one index to another, then re‑project onto the simplex.
    """
    n = len(seq)
    i, j = np.random.choice(n, 2, replace=False)
    # transfer a random fraction of the smaller entry
    delta = (np.random.rand() - 0.5) * step_scale
    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] - delta)
    new_seq[j] = max(0.0, new_seq[j] + delta)
    # projection guarantees Σ new_seq = 1 and non‑negativity
    return project_onto_simplex(new_seq)

# Main optimisation routine
# ----------------------------------------------------------------------

    print(f"Finished. Best score = {best_score:.6f}, length = {len(best_seq)}")
    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
