"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence(sequence).

The algorithm is a simple continuous optimisation:
* The sum of the coefficients is kept fixed (normalised to 1), because
  evaluate_sequence is homogeneous of degree 0.
* For a given sequence a we compute the linear convolution c = a∗a.
* The objective is to make max(c) as close as possible to its
  theoretical lower bound (the average of c, which equals 1/(2n‑1)).
* We repeatedly identify a few shifts k where c[k] is largest
  (the “hot spots”) and move mass away from the entries that
  participate most in those hot spots.
  The move is a projected gradient step that preserves the total sum
  and respects non‑negativity.
* Several random restarts with different lengths (≈500–4000) are tried
  within the allocated time budget.  The best sequence encountered
  (lowest score) is returned.

The implementation uses FFT‑based convolution (O(n log n)) so that
many iterations fit comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence)
def _score(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (sum(a) must be >0)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.size
    # linear convolution via FFT (real → real)
    size = 2 * n - 1
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, top_m: int = 5, lr: float = 0.5) -> np.ndarray:
    """
    Perform one projected gradient step that tries to lower the largest
    entries of the convolution a∗a.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative vector (its sum is 1).
    top_m : int
        Number of largest convolution positions that are used to build the
        gradient (default 5).  Using a few hot spots is enough to push mass
        away from the peaks while keeping the cost per iteration O(n).
    lr : float
        Learning rate (step size).  The step is multiplied by `lr`.

    Returns
    -------
    np.ndarray
        Updated vector (still non‑negative, not yet re‑normalised).
    """
    n = a.size
    size = 2 * n - 1
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)

    # indices of the `top_m` largest entries of the convolution
    if top_m >= conv.size:
        hot = np.arange(conv.size)
    else:
        hot = np.argpartition(-conv, top_m)[:top_m]

    # gradient accumulator
    g = np.zeros_like(a)

    # accumulate contributions from every selected shift k
    for k in hot:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)                # inclusive
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] += a[j]                         # a[j] is the partner in the product a[i]*a[j]

    # projected step: keep total sum unchanged (zero‑mean update)
    mean_g = g.mean()
    step = -g + mean_g                         # sum(step) == 0
    return a + lr * step


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluate_sequence value.

    The function is called by the evaluation harness.  It must return a
    Python list of floats.  The total running time is limited by
    `budget_s` seconds (default 1000 s).  The implementation performs
    many short optimisation runs with random restarts and keeps the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny margin

    best_score = float("inf")
    best_seq = None

    # Parameters that control the optimisation; they are deliberately
    # generous because the inner loop is cheap (FFT based).
    min_len = 800      # smallest length we try
    max_len = 4000     # largest length we try
    max_outer_iters = 200   # max number of random restarts (bounded by time)
    max_inner_iters = 3000  # iterations per restart (bounded by time)

    outer = 0
    while time.time() < deadline and outer < max_outer_iters:
        outer += 1

        # --------------------------------------------------------------
        # Choose a problem size and an initial vector.
        n = np.random.randint(min_len, max_len + 1)

        # Three possible initialisations: uniform, random, and a “arcsine‑like”
        # distribution that already has a slightly flatter autocorrelation.
        init_type = np.random.choice(["random", "arcsine", "uniform"])
        if init_type == "uniform":
            a = np.full(n, 1.0 / n)
        elif init_type == "arcsine":
            # a_i ∝ 1 / sqrt( (i+0.5)*(n-i-0.5) )
            i = np.arange(n, dtype=float)
            a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + 1e-12)
            a = a / a.sum()
        else:   # random
            a = np.random.rand(n)
            a = a / a.sum()

        # Starting learning rate – we decay it slowly.
        lr = 0.8
        lr_min = 1e-4
        decay = 0.9995

        # --------------------------------------------------------------
        # Inner optimisation loop.
        for it in range(max_inner_iters):
            if time.time() >= deadline:
                break

            # One gradient step with a few hot spots.
            a = _gradient_step(a, top_m=5, lr=lr)

            # Project onto the feasible set: non‑negative & sum = 1.
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:            # numerical safety – re‑initialise if all zero
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / s

            # Decay learning rate.
            lr = max(lr * decay, lr_min)

            # Periodically evaluate the true objective (cheaper than calling
            # the black‑box every iteration).
            if (it & 15) == 0:      # every 16th iteration
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{time.time():.2f}] New best = {best_score:.6f} "
                          f"(n={n}, outer={outer}, it={it}, type={init_type})")
                # early break if we already beat the target comfortably
                if best_score < 1.30:   # well below the required 1.503
                    deadline = time.time()    # stop as soon as possible
                    break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded (should never happen) return a safe uniform vector.
    if best_seq is None:
        n_fallback = 1000
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = _score(np.array(best_seq))
        print("Fallback used, score =", best_score)

    # The evaluator caps values at 1000, but our sequences are already ≤1.
    return best_seq
