    # Stage 1: random structured searches
    while time.time() < deadline - 5:
        # choose family
        family = np.random.choice(['linear','exp','two'])
        n = int(np.random.uniform(200, 3000))
        if family == 'linear':
            # slope param s in (0, 1], offset to keep positive

            idx = np.arange(n)

            q = np.random.uniform(1.0, 5.0)  # ratio high/low

            a = np.full(n, low)

        # Normalize to avoid all zeros
        if np.sum(a) == 0:
            continue
        # Scale to mean 1 (optional) to keep numbers moderate
        a = a / np.mean(a)  # ensures average ~1
        seq = a.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_sequence = seq.copy()
            print(f"New best {best_score:.6f} at n={n} family={family}")

    # Stage 2: local hill climbing on current best
    cur_seq = np.array(best_sequence, dtype=float)
    n = len(cur_seq)
    # Precompute some
    while time.time() < deadline:
        # compute conv and max index
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()

        k = np.random.choice(max_idxs)  # pick one max index

        # for each i where 0<=i<n and 0<=k-i<n, we can reduce a_i and a_{k-i}

            # Occasionally accept worse move with small probability
            if np.random.rand() < 0.001:
                cur_seq = new_seq

    print(f"Final best score: {best_score}")

def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence().  The method mixes cheap parametric random sampling,
    a simple hill‑climbing optimisation on the best candidate found and an
    LP‑based direction update (the same idea as the SOTA code).  The whole
    search stops after ``budget_s`` seconds and returns the best sequence.
    """
    import time, numpy as np, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a safety margin

    # ----- initialise from known construction ---------------------------------
    if 'height_sequence_1' in globals():
        best_seq = list(height_sequence_1)
    else:
        # fall back to a modest random start
        best_seq = np.random.rand(500).tolist()
    best_score = evaluate_sequence(best_seq)

    print(f"[init] score={best_score:.6f} length={len(best_seq)}")

    # -------------------------------------------------------------------------
    # 1. Random parametric sampling
    # -------------------------------------------------------------------------
    while time.time() < deadline - 5:
        n = int(np.random.uniform(200, 3000))
        family = np.random.choice(['linear', 'exp', 'twolev'])
        if family == 'linear':
            # a_i = max(0, 1 - slope * i/(n-1))
            slope = np.random.rand() * 0.95 + 0.05     # between 0.05 and 1.0
            idx = np.arange(n, dtype=float)
            a = 1.0 - slope * idx / (n - 1)
            a = np.clip(a, 0, None)
        elif family == 'exp':
            # a_i = r**i   (decreasing geometric)
            r = np.random.uniform(0.5, 0.99)
            idx = np.arange(n, dtype=float)
            a = r ** idx
        else:  # two‑level block
            cut = np.random.randint(1, n)
            high_low_ratio = np.random.uniform(1.0, 10.0)
            low = 1.0
            high = low * high_low_ratio
            a = np.full(n, low, dtype=float)
            a[:cut] = high

        # Normalise to keep the numbers in a comfortable range (scale does not matter)
        if a.sum() == 0:
            continue
        a = a / np.mean(a)  # average ≈1 => typical values around 1
        seq = a.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[sample] new best {best_score:.6f} n={n} fam={family}")

    # -------------------------------------------------------------------------
    # 2. Local refinement – hill climbing + LP direction updates
    # -------------------------------------------------------------------------
    cur_seq = np.array(best_seq, dtype=float)
    # keep a small history of best
    while time.time() < deadline:
        # ----- hill‑climbing step ------------------------------------------------
        # pick a random index that currently contributes to the maximal convolution term
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()
        max_pos = np.flatnonzero(conv == max_val)
        k = int(np.random.choice(max_pos))   # one of the binding indices

        # all pairs (i, j) with i+j == k that are inside the array
        possible_pairs = [(i, k - i) for i in range(len(cur_seq))
                          if 0 <= k - i < len(cur_seq)]
        if possible_pairs:
            i, j = possible_pairs[np.random.randint(len(possible_pairs))]
            # move a small amount from the larger of the two to a random other index
            source = i if cur_seq[i] > cur_seq[j] else j
            target = np.random.randint(len(cur_seq) - 1)
            if target >= source:
                target += 1
            eps = cur_seq[source] * (0.02 + 0.03 * np.random.rand())   # 2‑4 % of the source
            eps = min(eps, cur_seq[source])  # stay non‑negative
            new_seq = cur_seq.copy()
            new_seq[source] -= eps
            new_seq[target] += eps
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                cur_seq = new_seq
                print(f"[hill] improved to {best_score:.6f}")
            else:
                # with a tiny probability we still accept (simulated‑annealing style)
                if np.random.rand() < 5e-4:
                    cur_seq = new_seq

        # ----- occasional LP based direction (same as the baseline algorithm) ----
        if np.random.rand() < 0.02:   # about once every 50 iterations
            # reuse the helper defined earlier (make sure it exists)
            try:
                nxt = get_good_direction_to_move_into(cur_seq.tolist())

                        cur_seq = np.array(nxt, dtype=float)
                        print(f"[LP] improved to {best_score:.6f}")
            except Exception:
                pass

    print(f"[final] best score {best_score:.6f} length {len(best_seq)}")
    return best_seq

"""
search_step_function.py

A fast, self‑contained search for a non‑negative step function (list of heights)
that minimises the provided ``evaluate_sequence`` objective.

Algorithm
---------

1. **Initialisation** – start from the global construction ``height_sequence_1`` if
   it exists, otherwise fall back to a modest random vector.

2. **Random parametric sampling** – generate many candidate sequences from three
   cheap families:
        * linear decay   a_i = max(0, 1 - s·i/(n‑1))                     (s∈(0,1])
        * geometric decay a_i = r**i                                    (r∈(0.5,0.99))
        * two‑level block   high value on a prefix, low value everywhere.
   The candidates are normalised (scale does not affect the objective) and
   immediately evaluated.  The best candidate so far is kept.

3. **Local refinement** – on the current best vector a simple hill‑climbing move
   is repeatedly applied:
        * locate a convolution index that attains the maximum,
        * pick a contributing pair (i, j) with i+j = k,
        * move a small fraction of the larger element to a random other index,
          thereby reducing the maximal convolution entry while roughly keeping the
          total sum unchanged.
   Improvements are accepted; a very small probability of accepting a worse
   move provides a tiny amount of simulated‑annealing.

4. **Occasional LP direction** – once in a while the LP helper used by the
   reference implementation is called (``get_good_direction_to_move_into``).  It
   produces a direction that increases the ℓ₁‑norm while keeping the convolution
   bound, and the new point is kept when it yields an improvement.

5. **Time budget** – the whole routine stops after ``budget_s`` seconds (a small
   safety margin is kept).  The best sequence found is returned.

The implementation uses only NumPy and SciPy (for the optional linear program)
and contains no closures, no filesystem access and no global side‑effects.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper – linear program used by the reference method (optional).
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f, b) <= rhs   (for all convolution indices)
                   b >= 0
    Returns the optimal b as a NumPy array, or None if the LP failed.
    """
    try:
        # SciPy may not be present in all environments – import lazily.
        from scipy.optimize import linprog
    except Exception:
        return None

    n = len(f_sequence)
    # Objective: maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # Build the convolution constraints: for each k = 0 .. 2n-2
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        # conv_k = Σ_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints b >= 0  →  -b <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # Use the high‑performance default method; give a short time limit.
    opts = {"disp": False, "time_limit": 8.0}
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs", options=opts)

    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Implements the LP‑based direction step described in the reference code.
    Returns a new sequence (list of floats) that is a convex combination of the
    current sequence and the LP solution, or None if the LP fails.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq == 0:
        return None
    # Normalise to the same ℓ₁‑norm used by the reference algorithm.
    normalized = [x * np.sqrt(2 * n) / sum_seq for x in sequence]
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    normalized_g = [x * np.sqrt(2 * n) / sum_g for x in g]

    t = 0.01  # small mixing weight, identical to the reference implementation
    new_seq = [(1 - t) * x + t * y for x, y in zip(sequence, normalized_g)]
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the judging harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence (list of non‑negative floats) discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    # ------------------------------------------------------------------
    # 0. Initialise with the known construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best_seq = list(height_sequence_1)
    else:
        best_seq = np.random.rand(500).tolist()  # fallback
    best_score = evaluate_sequence(best_seq)
    print(f"[init] score={best_score:.6f}  len={len(best_seq)}")

    # ------------------------------------------------------------------
    # 1. Random parametric sampling (cheap families).
    # ------------------------------------------------------------------
    families = ["linear", "exp", "twolev"]
    while time.time() < deadline - 5:
        n = int(np.random.uniform(200, 3000))
        fam = np.random.choice(families)

        if fam == "linear":
            # a_i = max(0, 1 - s * i/(n-1)),  s ∈ (0.05, 1.0)
            s = np.random.rand() * 0.95 + 0.05
            idx = np.arange(n, dtype=float)
            a = 1.0 - s * idx / (n - 1)
            a = np.clip(a, 0, None)

        elif fam == "exp":
            # a_i = r**i,  r ∈ (0.5, 0.99)
            r = np.random.uniform(0.5, 0.99)
            idx = np.arange(n, dtype=float)
            a = r ** idx

        else:  # two‑level block
            cut = np.random.randint(1, n)                 # size of the high block
            ratio = np.random.uniform(1.0, 10.0)          # high / low ratio
            low = 1.0
            high = low * ratio
            a = np.full(n, low, dtype=float)
            a[:cut] = high

        if a.sum() == 0:
            continue
        # Normalise – the objective is scale invariant
        a = a / np.mean(a)  # average ≈ 1 → numbers stay modest
        seq = a.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[sample] new best {best_score:.6f}  n={n}  fam={fam}")

    # ------------------------------------------------------------------
    # 2. Local refinement – hill climbing + occasional LP step.
    # ------------------------------------------------------------------
    cur_seq = np.array(best_seq, dtype=float)

    while time.time() < deadline:
        # ---- a cheap hill‑climbing move ---------------------------------
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()
        max_positions = np.flatnonzero(conv == max_val)
        k = int(np.random.choice(max_positions))

        # all index pairs (i, j) with i + j = k inside the array
        pairs = [(i, k - i) for i in range(len(cur_seq))
                 if 0 <= k - i < len(cur_seq)]
        if not pairs:
            continue

        i, j = pairs[np.random.randint(len(pairs))]
        # pick the larger one as the source of the mass transfer
        source = i if cur_seq[i] > cur_seq[j] else j
        # pick a random different target index
        target = np.random.randint(len(cur_seq) - 1)
        if target >= source:
            target += 1

        # transfer a small fraction (2‑5 %) of the source value
        eps = cur_seq[source] * (0.02 + 0.03 * np.random.rand())
        eps = min(eps, cur_seq[source])          # stay non‑negative

        new_seq = cur_seq.copy()
        new_seq[source] -= eps
        new_seq[target] += eps

        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.tolist()
            cur_seq = new_seq
            print(f"[hill] improved to {best_score:.6f}")
        else:
            # tiny chance to accept a worse move (helps escaping shallow traps)
            if np.random.rand() < 5e-4:
                cur_seq = new_seq

        # ---- occasional LP‑based direction (same idea as the baseline) ---
        if np.random.rand() < 0.02:   # about once every 50 iterations
            try:
                nxt = get_good_direction_to_move_into(cur_seq.tolist())
                if nxt is not None:
                    nxt_score = evaluate_sequence(nxt)
                    if nxt_score < best_score:
                        best_score = nxt_score
                        best_seq = nxt
                        cur_seq = np.array(nxt, dtype=float)
                        print(f"[LP] improved to {best_score:.6f}")
            except Exception:
                # any problem (e.g. missing scipy) is ignored – the loop continues
                pass

    print(f"[final] best score {best_score:.6f}  length {len(best_seq)}")
    return best_seq
