```
def project_simplex(v):
    """Projects vector v onto the probability simplex {x: x>=0, sum x = 1}."""
    n = len(v)
    # ensure v is array
    v = np.asarray(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum, so minimize negative sum
    # Build constraint matrix: for each k, sum_i f_i * b_{k-i} <= M
    # Equivalent to A_ub @ b <= b_ub
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i in [0,n-1]; j = k - i must be between 0 and n-1 => i between max(0,k-(n-1)) and min(n-1,k)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # For each i, b_j appears where j = k - i; coefficient is f_i.
        # So A[k, j] += f_i for each i.
        # This can be vectorized:
        # set rows: A[k, (k - i_range)] = f_seq[i_range].
        i_range = np.arange(i_min, i_max+1)
        j_range = k - i_range
        A[k, j_range] = f_seq[i_range]
    # Non-negativity constraints handled by bounds param.
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # Build A_ub matrix
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1)
        j_range = k - i_range
        A[k, j_range] = f_seq[i_range]
    b = np.full(rows, M)
    bounds = [(0, None)] * n
    # Solve using high efficiency method
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        return None

Iterative improvement of a non‑negative step function.

The algorithm keeps the sequence normalized to have unit ℓ¹‑norm (the
objective is scale‑invariant).  In each iteration we

1. compute the current auto‑convolution and its maximum M,
2. solve a linear program that finds a feasible direction `g`
   with maximal ℓ¹‑norm under the same convolution bound M,
   3. mix the current sequence `f` with the direction `g` using an
   optimal convex coefficient `t` (found by a tiny ternary search)
   that minimises the new maximum auto‑convolution,
4. remember the best sequence seen so far and occasionally
   inject a random perturbation to escape stagnation.

The LP is built from the Toeplitz matrix that represents the
convolution `f * b`.  Using a sparse matrix makes the LP fast enough
to run many iterations within the 1000 s budget.
"""

def _build_convolution_matrix(f):
    """Return a sparse (2n‑1)×n matrix A such that A @ b = conv(f, b)."""
    n = len(f)
    rows = 2 * n - 1
    data = []
    row_idx = []
    col_idx = []
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k - i_range               # column indices
        data.extend(f[i_range])
        row_idx.extend([k] * len(i_range))
        col_idx.extend(j_range)
    return sparse.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))


def _solve_convolution_lp(f, M):
    """
    Solve   max sum(b)    s.t.  conv(f, b) ≤ M ,  b ≥ 0 .
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f)
    A = _build_convolution_matrix(f)
    c = -np.ones(n)                     # maximize sum(b)  ⇔  minimise -sum(b)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    # ‘highs’ is the modern, fast LP solver in SciPy
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else None


def _optimal_mix_t(ff, fg, gg):
    """

    that minimises max_k [(1‑t)²·ff + 2t(1‑t)·fg + t²·gg]_k
    via a short ternary search (convex in t).
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        c1 = (1 - t1) ** 2 * ff + 2 * t1 * (1 - t1) * fg + t1 ** 2 * gg
        c2 = (1 - t2) ** 2 * ff + 2 * t2 * (1 - t2) * fg + t2 ** 2 * gg
        if c1.max() < c2.max():
            hi = t2
        else:
            lo = t1
