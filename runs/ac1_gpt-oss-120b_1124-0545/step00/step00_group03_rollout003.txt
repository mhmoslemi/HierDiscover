import time, numpy as np, random

def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes `evaluate_sequence`.  The routine explores a family of
    parametric shapes (linear, quadratic, exponential, power‑law,
    sinusoidal, triangular etc.) for several lengths, keeps the best
    candidate and then refines it by a simple simulated‑annealing /
    hill‑climbing loop.  The total running time is bounded by the
    supplied `budget_s` seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1
    np.random.seed(seed)
    random.seed(seed)

    # Helper to evaluate and keep best
    best_seq = None
    best_score = float('inf')

    # Helper to try a candidate (list or np.ndarray)
    def try_candidate(seq):
        nonlocal best_seq, best_score
        # Convert to list for evaluate_sequence
        seq_list = seq.tolist() if isinstance(seq, np.ndarray) else list(seq)
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list.copy()
            print(f"[{time.time()-start_time:.1f}s] New best: {best_score:.6f} length {len(seq_list)}")
        return score

    # ------------------------------------------------------------------
    # Phase 1 – deterministic exploration of parameterised families
    # ------------------------------------------------------------------
    lengths = [64, 128, 256, 512, 1024, 2048, 4096, 8192]

    # If a known good construction exists, start from it
    if 'height_sequence_1' in globals():
        try_candidate(np.array(height_sequence_1, dtype=float))

    for n in lengths:
        if time.time() >= deadline:
            break
        # constant
        try_candidate(np.ones(n, dtype=float))

        # linear decreasing / increasing
        try_candidate(np.arange(n, 0, -1, dtype=float))
        try_candidate(np.arange(1, n + 1, dtype=float))

        # quadratic decreasing / increasing
        try_candidate(np.square(np.arange(n, 0, -1, dtype=float)))
        try_candidate(np.square(np.arange(1, n + 1, dtype=float)))

        # triangular (mountain)
        t = np.arange(1, n + 1, dtype=float)
        tri = np.minimum(t, n - t + 1)
        try_candidate(tri)

        # valley shape (large at the ends, small in centre)
        centre = (n - 1) / 2.0
        valley = 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))
        try_candidate(valley)

        # sin^2 shape
        sin_sq = np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2
        try_candidate(sin_sq)

        # exponential decay for a few bases
        for r in (0.9, 0.8, 0.7, 0.6, 0.5):
            exp_seq = np.power(r, np.arange(n, dtype=float))
            try_candidate(exp_seq)

        # power‑law decay
        for p in (0.5, 1.0, 1.5, 2.0, 2.5):
            i = np.arange(1, n + 1, dtype=float)
            pow_seq = 1.0 / (i ** p)
            try_candidate(pow_seq)

    # ------------------------------------------------------------------
    # Phase 2 – local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    # Turn the best sequence into a float array for convenient mutation
    if best_seq is None:
        # fallback: a short random array
        best_seq = [np.random.random() for _ in range(128)]
        best_score = evaluate_sequence(best_seq)

    cur = np.array(best_seq, dtype=float)

    # Pre‑compute a small temperature schedule
    start_temp = 1.0
    end_temp = 0.001

    while time.time() < deadline:
        # Current temperature (linear cooling)
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)
        # Choose mutation type
        if np.random.rand() < 0.75:
            # Gaussian perturbation on a random subset
            m = max(1, int(0.01 * len(cur)))          # ~1 % of entries
            idx = np.random.choice(len(cur), size=m, replace=False)
            delta = np.random.randn(m) * cur.mean() * 0.1
            new = cur.copy()
            new[idx] += delta
            # enforce non‑negativity
            new = np.maximum(new, 0.0)
        else:
            # Small reversal of a random block – changes the ordering
            start = np.random.randint(len(cur))
            block_len = np.random.randint(1, min(200, len(cur) - start) + 1)
            new = cur.copy()
            new[start:start + block_len] = new[start:start + block_len][::-1]

        # If the sum became too tiny, inject a tiny constant to stay valid
        if new.sum() < 0.01:
            new += (0.01 - new.sum()) / len(new)

        # Evaluate the candidate
        cand_score = evaluate_sequence(new.tolist())

        # Accept if better, otherwise maybe accept probabilistically
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time()-start_time:.1f}s] Improved to {best_score:.6f}")
        else:
            # Simulated annealing acceptance
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
                cur = new

        # Early exit if we already beat the target
        if best_score <= 1.503:
            print(f"Target reached (score {best_score:.6f}) – stop searching.")
            break

    return best_seq
```

"""
Proposal: Stochastic shape‑search for the step‑function coefficients.

The evaluation function is homogeneous, i.e. scaling a sequence does not
change its score.  Therefore the algorithm works with *shapes* only:
every candidate is first normalised so that its largest entry is < 1 (far
below the 1000‑clip limit) while keeping a non‑negligible total sum.

Phase 1 enumerates a collection of analytically‑defined families
(constant, linear, quadratic, triangular, valley, sinusoidal,
exponential‑decay and power‑law).  For each length in a geometric series
(64 … 8192) every shape is normalised and evaluated; the best sequence is
kept.

Phase 2 refines the current best candidate with a simple simulated‑annealing
loop: random Gaussian perturbations of a small subset of entries and random
reversals of short blocks are tried.  Improvements are always accepted,
worse candidates are accepted with a temperature‑controlled probability.
The loop respects the given time budget and stops early if the target
score ≤ 1.503 is reached.

All helper functions are defined at module level (no closures) and the
public entry point is `propose_candidate`.
"""

import time
import numpy as np


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale the array so that its maximum is 0.9 (well below the 1000 cap)
    and guarantee a total sum of at least 0.01 (otherwise evaluate_sequence
    returns ∞).  The scaling does not affect the final score because the
    evaluation is homogeneous.
    """
    a = np.array(a, dtype=float)
    max_v = a.max()
    if max_v > 0:
        a = a * (0.9 / max_v)          # keep values < 1
    s = a.sum()
    if s < 0.01:                      # avoid the “sum too close to zero” guard
        a = a * (0.01 / s) if s > 0 else a + (0.01 / a.size)
    return a


def _candidate_shapes(n: int):
    """
    Generate a list of candidate shape vectors of length `n`.  All returned
    arrays are plain float ndarrays; they are normalised later before
    evaluation.
    """
    cand = []

    # 1. constant
    cand.append(np.ones(n, dtype=float))

    # 2. linear (decreasing / increasing)
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # 3. quadratic (decreasing / increasing)
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # 5. valley (large at the ends, small in the centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # 6. sine‑squared shape
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7. exponential decays
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 8. power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimizes
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # a small safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Start from height_sequence_1 "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of parametric families
    # ------------------------------------------------------------------
    lengths = [64, 128, 256, 512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _candidate_shapes(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(cand)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] "

            if best_score <= 1.503:   # early exit if target already reached
                break

    if best_seq is None:

        best_score = evaluate_sequence(best_seq)

    # Phase 2 – stochastic hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)

    # Linear temperature schedule: start at 1.0, finish at 0.001
    start_temp, end_temp = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # ----- mutation -------------------------------------------------
        if rng.random() < 0.70:

            m = max(1, int(0.01 * len(cur)))
            idx = rng.choice(len(cur), size=m, replace=False)

            new = cur.copy()
            new[idx] += delta
            new = np.clip(new, 0.0, 999.0)          # stay below the 1000 clamp

            start_idx = rng.integers(0, len(cur))

            new = cur.copy()

        if new.sum() < 0.01:
            new += (0.01 - new.sum()) / len(new)

        # ----- evaluation -----------------------------------------------
        cand_score = evaluate_sequence(new.tolist())

        # ----- acceptance ------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] Improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – stopping search.")
                break
        else:
            # Simulated‑annealing acceptance of worse moves
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

    return best_seq
