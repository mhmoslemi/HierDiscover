    n = len(a)
    size = int(2**np.ceil(np.log2(2*n-1)))  # next power of 2
    fa = np.fft.rfft(np.pad(a, (0, size - n)))

    deadline = time.time() + budget_s

    # Initialize sequence: random expo distribution, normalized sum to sqrt(2*n)

    print(f'Initial score {best_score:.6f}')

    T0 = best_score * 0.1  # estimate temperature (maybe)
    T = T0

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # propose move: transfer delta
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.2 * a[i]   # up to +-10% of a[i]
        new_i = a[i] - delta
        new_j = a[j] + delta

            delta = a[i]
            new_i = 0
            new_j = a[j] + delta

        a_new = a.copy()
        a_new[i] = new_i
        a_new[j] = new_j

        # ensure sum > 0.01
        if a_new.sum() < 0.01:
            continue
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.tolist()
            a = a_new
            # print improvement
            print(f'Iter {iteration}: improved score {best_score:.6f}')

            if np.random.rand() < np.exp(-(score - best_score) / (T + 1e-12)):

"""
Propose a better step‑function (non‑negative sequence) for the given
evaluation function.

The algorithm is a simple stochastic local search that repeatedly tries to
lower the peak of the self‑convolution while keeping the total L¹ norm
(= sum of the sequence) fixed.

* Initialise the sequence either from the supplied SOTA sequence
  ``height_sequence_1`` or from a random exponential vector
  (scaled to the standard normalisation √(2·n)).
* In each iteration we compute the convolution a*a, locate the index
  ``k_max`` where the convolution attains its maximum and pick the pair
  (i, j) with i+j = k_max that contributes the largest product
  a[i]·a[j].
* A small amount of mass (δ) is taken from both i and j and is added
  to a few random target positions; the best target (the one that yields
  the smallest evaluation score) is kept.
* The move is accepted if it improves the best score seen so far;
  otherwise it may be accepted with a tiny probability (simulated
  annealing) to escape plateaus.
* If the search stagnates for a while we re‑initialise the whole vector
  to a fresh random draw.

The procedure stops a few seconds before the overall time budget
(`budget_s`) expires and returns the best sequence discovered.
"""

import time
import math
import numpy as np

def _max_contributing_pair(seq, conv_max_idx):
    """
    Return indices (i, j) with i+j == conv_max_idx that have the largest
    product a[i] * a[j] (the pair that most contributes to the current peak).
    """
    n = len(seq)
    i_start = max(0, conv_max_idx - (n - 1))
    i_end = min(n - 1, conv_max_idx)
    i_vals = np.arange(i_start, i_end + 1)
    j_vals = conv_max_idx - i_vals
    products = seq[i_vals] * seq[j_vals]
    best = int(np.argmax(products))
    return int(i_vals[best]), int(j_vals[best])


def propose_candidate(seed=0, budget_s=950, **kwargs):
    """
    Stochastic local optimisation of the step‑function coefficients.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will terminate a few seconds
        before this limit to be safe).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------ #
    # 1. initialise ---------------------------------------------------- #
    # ------------------------------------------------------------------ #
    if 'height_sequence_1' in globals() and np.random.rand() < 0.5:
        a = np.array(height_sequence_1, dtype=float)
        # ensure the sequence is not empty and has reasonable total mass
        if a.size == 0 or a.sum() < 0.01:
            a = np.random.exponential(scale=1.0, size=800)
    else:
        # a random length between 500 and 1500 gives a good trade‑off
        n = np.random.randint(800, 1200)
        a = np.random.exponential(scale=1.0, size=n)

    # Normalise the sum to √(2·n) – the scale does not affect the score,
    # but keeping it in a reasonable range avoids the “sum too small”
    # guard in evaluate_sequence.
    n = a.size
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------ #
    # 2. main stochastic search ---------------------------------------- #
    # ------------------------------------------------------------------ #
    no_improve_cnt = 0
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Compute the full convolution (numpy’s implementation is fast enough)
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # Identify the pair (i, j) that contributes most to this peak
        i, j = _max_contributing_pair(a, k_max)

        move_frac = 0.05

        trial = a.copy()

            no_improve_cnt = 0

    print(f"[finished] iterations={iteration}, best score={best_score:.6f}")
    return best_seq
