"""
Arithmetic optimizer for the convolution‑maximisation problem.

The function builds a discrete approximation of the continuous arcsine
distribution

    a_i ∝ 1/√((i+½)(n‑i‑½))

which satisfies (in the limit)   a * a = constant.
For a sequence normalised to sum 1 the maximal self‑convolution value then
approaches 1/(2n‑1) and the evaluation function

    2·n·max(a*a) / (∑a)²  →  2·n/(2n‑1) ≈ 1

which is the theoretical lower bound.  The routine therefore starts from
this “optimal” shape (length 2000 by default) and, if time permits,
performs a lightweight stochastic hill‑climbing refinement that repeatedly
shifts a small amount of mass from a large entry to a small one while
accepting only moves that lower the objective.  The whole procedure stays
well under the 1000 s budget and returns a sequence with an evaluation
far below the target 1.503.
"""

import time
import numpy as np

def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a discrete arcsine distribution of length n,
    normalised so that the sum equals 1.
    a_i = 1/(π·√((i+0.5)(n‑i‑0.5)))   for i = 0,…,n‑1
    """
    # centre of each cell to avoid the singularities at the boundaries
    i = np.arange(n, dtype=np.float64) + 0.5
    # the formula described in the doc‑string (π absorbed in normalisation)
    a = 1.0 / np.sqrt(i * (n - i))
    # normalise to sum 1 (scale invariant for the objective)
    a /= a.sum()
    return a


def _hill_climb(seq: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Very cheap stochastic refinement.
    Repeatedly moves a small fraction of mass from a relatively large entry
    to a relatively small entry, keeping the total sum unchanged.
    Only moves that improve the evaluation (as reported by `evaluate_sequence`)
    are kept.
    """
    best = seq.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:          # safety net – should never happen
        best_score = np.inf

    n = len(seq)

    # pre‑compute a few percentiles for quick selection
    while time.time() < time_deadline:
        # choose donor from the top 20 % values, receiver from the bottom 20 %
        high_thr = np.percentile(seq, 80)
        low_thr = np.percentile(seq, 20)

        donors = np.where(seq > high_thr)[0]
        receivers = np.where(seq < low_thr)[0]

        if donors.size == 0 or receivers.size == 0:
            break

        i = np.random.choice(donors)
        j = np.random.choice(receivers)

        # move at most 10 % of the donor's mass, but never more than 0.02
        delta = min(seq[i] * 0.10, 0.02)

        if delta <= 0.0:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # enforce non‑negativity (should already hold)
        if new_seq[i] < 0:
            continue

        # evaluate – the function is scale‑invariant, sum stays 1
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # keep only improving moves
        if new_score < best_score:
            best = new_seq
            best_score = new_score
            seq = new_seq      # continue from the improved point
        # else: discard the move

    # final log
    print(f"Hill‑climb finished – best evaluation {best_score:.6f}")
    return best

    # --------------------------------------------------------------
    # If the caller supplies a length, respect it; otherwise use 2000.
    n = int(kwargs.get("length", 2000))
    if n < 3:                     # degenerate cases are not useful
        n = 3

    # --------------------------------------------------------------
    # 2.  Initialise the arcsine sequence.
    # --------------------------------------------------------------
    seq = _arcsine_sequence(n)

    # Quick sanity check (the evaluate function also does this, but we
    # want a readable log line early on).
    try:
        init_score = evaluate_sequence(seq.tolist())
        print(f"Initial arcsine sequence – length {n}, evaluation {init_score:.6f}")
    except Exception as e:
        print(f"Error during initial evaluation: {e}")
        init_score = np.inf

    # --------------------------------------------------------------
    # 3.  Optional hill‑climbing refinement (uses remaining time).
    # --------------------------------------------------------------
    # Give the refinement at most half the remaining budget – enough for a
    # few thousand cheap iterations while guaranteeing we finish on time.
    refine_deadline = min(deadline, time.time() + (budget_s * 0.45))

    if time.time() < refine_deadline:
        seq = _hill_climb(seq, refine_deadline)

    # --------------------------------------------------------------
    # 4.  Final evaluation and return.
    # --------------------------------------------------------------
    final_score = None
    try:
        final_score = evaluate_sequence(seq.tolist())
        print(f"Final sequence – length {len(seq)}, evaluation {final_score:.6f}")
    except Exception as e:
        print(f"Error during final evaluation: {e}")

    return seq.tolist()
