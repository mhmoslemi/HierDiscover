def random_transfer(seq, max_frac=0.05):
    n = len(seq)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i==j: return seq
    amount = seq[i] * max_frac * np.random.rand()
    if amount > seq[i]:
        amount = seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    return new_seq

The evaluation function is homogeneous: scaling a sequence does not change the score.  
If we normalise the sum of a sequence to  

```
S = √(2·n)      (n = length of the sequence)
```

the score is simply the maximal entry of the *auto‑convolution*  

```
c = a * a .
```

For a uniform sequence `a_i = S/n` the maximum of the convolution equals `2` and the
score is `2`.  

A non‑uniform distribution can flatten the convolution.  In particular an
exponential (geometric) shape

```
a_i ∝ r^i      with 0 < r < 1
```

has a convolution whose peak is about  

```
max(c) ≈ (1‑r) / e .
```

Choosing `r = 1‑ε` with `ε ≈ e·target/(2·n)` makes  

```
score ≈ 2·n·max(c) ≈ target .
```

Hence a simple parametric family (geometric) already reaches the target
≈ 1.5 for the usual lengths (≈ 2500).  
The search algorithm therefore

1. tries many lengths,
2. samples a wide range of `ε` (i.e. of `r`),
3. evaluates the corresponding geometric sequences (and the reversed ones),
4. also samples a small set of power‑law sequences for robustness,
5. keeps the best sequence found,
6. performs a cheap local optimisation (`random_transfer`) to fine‑tune the
   best solution until the time limit expires.

All steps use only `evaluate_sequence`, no external I/O, and respect the
time budget.

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (all at top level – no closures)
# ----------------------------------------------------------------------


def geometric_sequence(n: int, epsilon: float) -> list[float]:
    """
    Return a non‑negative sequence of length n of the form r**i,
    where r = 1 - epsilon, scaled so that sum(seq) == sqrt(2*n).
    """
    if epsilon <= 0.0:
        epsilon = 1e-12
    r = max(0.0, 1.0 - epsilon)
    idx = np.arange(n, dtype=np.float64)
    seq = r ** idx
    # normalise to S = sqrt(2*n)
    target_sum = math.sqrt(2.0 * n)
    total = seq.sum()
    if total == 0.0:                     # extremely unlikely, but guard against div‑by‑0
        seq = np.ones(n, dtype=np.float64) / n
        total = 1.0
    seq *= (target_sum / total)
    return seq.tolist()


def power_sequence(n: int, beta: float) -> list[float]:
    """
    Return a sequence a_i = (i+1)^(-beta) (beta>=0) scaled to the same sum as above.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    # avoid division by zero for beta==0 (uniform)
    if beta == 0.0:
        seq = np.ones(n, dtype=np.float64)
    else:
        seq = idx ** (-beta)
    target_sum = math.sqrt(2.0 * n)
    total = seq.sum()
    if total == 0.0:
        seq = np.ones(n, dtype=np.float64)
        total = n
    seq *= (target_sum / total)
    return seq.tolist()


def random_transfer(seq: list[float], max_frac: float = 0.02) -> list[float]:
    """
    Produce a neighbour of ``seq`` by moving a random fraction (up to ``max_frac``)
    of mass from one index to another.  The total sum stays unchanged.
    """
    n = len(seq)
    if n < 2:
        return seq.copy()
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)

    src = seq[i]
    # don't move more than is available
    delta = src * max_frac * np.random.rand()
    if delta > src:
        delta = src

    new_seq = seq.copy()
    new_seq[i] = src - delta
    new_seq[j] = seq[j] + delta
    # tiny negative values can appear because of floating point noise – clip
    new_seq[i] = max(0.0, new_seq[i])
    return new_seq


def evaluate(seq: list[float]) -> float:
    """
    Thin wrapper around the provided evaluate_sequence for brevity.
    """
    return evaluate_sequence(seq)  # evaluate_sequence is assumed to be in the global namespace


# ----------------------------------------------------------------------
# Main candidate‑generation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative height sequence) with a low
    evaluation score.

    Strategy:
    1. Try geometric (exponential) sequences a_i ∝ (1‑ε)^i for many lengths
       and many ε values (log‑uniform sampling).  Both forward and reversed
       orders are examined.
    2. Try a few power‑law families a_i ∝ (i+1)^(-β) as a secondary family.
    3. Keep the best candidate seen so far.
    4. Refine the best candidate with cheap random mass‑transfer moves until
       the time budget (≈ 1000 s) is exhausted.
    5. Return the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin for the outer loop
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. systematic sampling of geometric and power‑law families
    # ------------------------------------------------------------------
    length_choices = [256, 512, 1024, 2048, 2584]  # include the known length

    # sampling parameters
    n_geo_samples = 200          # per length
    n_pow_samples = 200

    # log‑uniform epsilon in [1e-5, 0.4] – the upper bound is far above the useful range
    log_eps_low, log_eps_high = -5.0, -0.3

    for n in length_choices:
        # ---- geometric candidates -------------------------------------------------
        for _ in range(n_geo_samples):
            eps = 10 ** np.random.uniform(log_eps_low, log_eps_high)
            # Guard against eps too large (would make r negative)
            if eps >= 1.0:
                continue
            seq = geometric_sequence(n, eps)
            sc = evaluate(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[{time.time() - start_time:.1f}s] New best {best_score:.6f} "
                      f"(geometric n={n}, eps={eps:.2e})")
            # also test the reversed order
            rev = seq[::-1]
            sc_rev = evaluate(rev)
            if sc_rev < best_score:
                best_score = sc_rev
                best_seq = rev
                print(f"[{time.time() - start_time:.1f}s] New best {best_score:.6f} "
                      f"(geometric rev n={n}, eps={eps:.2e})")

        # ---- power‑law candidates -----------------------------------------------
        for _ in range(n_pow_samples):
            beta = np.random.uniform(0.0, 3.0)   # beta=0 gives uniform, larger → more front‑heavy
            seq = power_sequence(n, beta)
            sc = evaluate(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[{time.time() - start_time:.1f}s] New best {best_score:.6f} "
                      f"(power n={n}, beta={beta:.3f})")

    # ------------------------------------------------------------------
    # 2. include the previously known construction if it exists
    # ------------------------------------------------------------------
    try:
        if height_sequence_1:                     # noqa: F821 – global may or may not exist
            sc = evaluate(height_sequence_1)
            if sc < best_score:
                best_score = sc
                best_seq = list(height_sequence_1)
                print(f"[{time.time() - start_time:.1f}s] Adopted height_sequence_1 "
                      f"score={best_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. local refinement – cheap random mass transfers
    # ------------------------------------------------------------------
    # we keep a small “temperature” that slowly decays; this helps to escape tiny plateaus
    init_temp = 0.02
    temp_decay = 0.9995

    while time.time() < deadline:
        if best_seq is None:
            # fallback: random start (should never happen)
            n = np.random.randint(100, 2000)
            seq = np.random.rand(n).tolist()
            # normalise to the required sum
            target_sum = math.sqrt(2.0 * n)
            seq = (np.array(seq) * (target_sum / np.sum(seq))).tolist()
            best_seq = seq
            best_score = evaluate(best_seq)

        # generate a neighbour
        step = max(1e-4, init_temp)      # gradually shrink
        cand = random_transfer(best_seq, max_frac=step)
        try:
            sc = evaluate(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[{time.time() - start_time:.1f}s] Improved to {best_score:.6f} "
                      f"via random transfer")
        except Exception:
            # evaluation can return np.inf for malformed sequences – ignore
            pass

        # slowly cool the step size
        init_temp *= temp_decay

    # final report
    print(f"Finished. Best length = {len(best_seq)}   best score = {best_score:.6f}")
    return best_seq
