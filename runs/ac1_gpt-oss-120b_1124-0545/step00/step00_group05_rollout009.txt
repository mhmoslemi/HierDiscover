"""
Propose a better step‑function by iteratively improving the
self‑convolution maximum (the quantity returned by evaluate_sequence).

The algorithm works with a fixed length `n` (chosen as a few hundred to
keep every iteration cheap).  The sequence is always re‑scaled so that
its sum equals `sqrt(2*n)`.  For this scaling the evaluation function
simplifies to `max(conv(seq,seq))`, i.e. the maximal entry of the
autocorrelation.

Given a current sequence `f` we solve a linear program:

    maximise   sum(g)
    subject to  (f * g)[k] ≤ M   for all k
                g ≥ 0

where `M = max(f*f)`.  Because the convolution is linear in the second
argument the constraints are linear, so a standard LP finds the best
direction `g` that increases the total mass without increasing the
current bound `M`.  Afterwards we blend `f` and `g` using a line search
to keep the maximum autocorrelation non‑increasing.  The blend is
always a convex combination, therefore the sum stays fixed and the
objective value (the maximum convolution) can only go down.

A few random perturbations are applied when the LP fails or the progress
stalls.  The routine runs until the allocated time budget (default
≈ 1000 s) expires and returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that  A @ g  gives the convolution (f * g)
    for all possible output indices (full convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # indices i where 0 ≤ i < n and 0 ≤ k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals               # = index of g
        A[k, j_vals] = f[i_vals]
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP described in the module docstring.
    Returns the optimal g (as a NumPy array) or None if infeasible.
    """
    n = f.shape[0]
    A_conv = _build_conv_matrix(f)               # (2n-1) × n
    # inequality: A_conv @ g ≤ M
    A_ub = np.vstack([A_conv, -np.eye(n)])       # add g ≥ 0 as -g ≤ 0
    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    c = -np.ones(n)                               # maximise sum(g) -> minimise -sum(g)

    # use the default HiGHS method – fast and robust
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None


def _line_search(f: np.ndarray, g: np.ndarray, M_cur: float) -> tuple[np.ndarray, float]:
    """
    Find the largest step size t∈(0,1] such that the maximum of the
    convolution does not exceed M_cur.  Returns the new sequence and its
    new max‑convolution value.
    """
    # Full step first – if it works we keep it
    t_low, t_high = 0.0, 1.0
    new = (1 - t_high) * f + t_high * g
    M_new = np.max(np.convolve(new, new))
    if M_new <= M_cur:
        return new, M_new

    # binary search for the best feasible t
    for _ in range(12):          # enough for double precision
        t_mid = (t_low + t_high) / 2.0
        cand = (1 - t_mid) * f + t_mid * g
        M_cand = np.max(np.convolve(cand, cand))
        if M_cand <= M_cur:
            t_low = t_mid
            new = cand
            M_new = M_cand
        else:
            t_high = t_mid
    return new, M_new


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a tiny safety margin

    # ----- choose a working length -------------------------------------------------
    # A few hundred gives a good trade‑off between expressiveness and speed.
    n = 600
    target_sum = np.sqrt(2.0 * n)                    # keep sum fixed to simplify the objective

    # ----- start from a uniform sequence ------------------------------------------
    seq = np.full(n, target_sum / n)                 # uniform, sum = target_sum
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # current maximum of the autocorrelation
        M = np.max(np.convolve(seq, seq))

        # ----- try the LP direction -------------------------------------------------
        g = _solve_lp(seq, M)
        if g is not None:
            # rescale g to the same total mass
            sum_g = np.sum(g)
            if sum_g == 0:
                # degenerate solution – skip
                continue
            g = g * (target_sum / sum_g)

            # blend using line search, keep max‑convolution non‑increasing
            new_seq, new_M = _line_search(seq, g, M)

            # evaluate the candidate (evaluate_sequence works with any scaling)
            cur_score = evaluate_sequence(new_seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = new_seq.copy()
                # reporting
                print(f"Iter {iteration:4d}: improved score = {best_score:.6f}  (M={new_M:.6e})")
            seq = new_seq
            continue

        # ----- LP failed – apply a small random perturbation -----------------------
        # Move a tiny amount of mass from a random index to another.
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.randn() * 0.02 * target_sum   # typical step size ~2%
        # keep both entries non‑negative
        if seq[i] + delta < 0:
            delta = -seq[i] + 1e-12
        if seq[j] - delta < 0:
            delta = seq[j] - 1e-12
        seq[i] += delta
        seq[j] -= delta
        # re‑normalise to keep the exact target sum (helps numerical stability)
        seq = seq * (target_sum / np.sum(seq))

        # occasional cheap evaluation
        if iteration % 20 == 0:
            cur_score = evaluate_sequence(seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"Iter {iteration:4d}: random perturb improved score = {best_score:.6f}")

        # stop early if we already beat the target bound (≈1.5)
        if best_score <= 1.55:
            print(f"Target reached (score {best_score:.6f}) – stopping early.")
            break

    # -------------------------------------------------------------------------------
    # Return the best sequence found, making sure all entries are non‑negative
    # and clipped to the allowed range (the evaluator will also do this).
    final = [float(max(0.0, x)) for x in best_seq.tolist()]
    return final
