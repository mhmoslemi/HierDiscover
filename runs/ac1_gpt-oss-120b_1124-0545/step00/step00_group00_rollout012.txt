def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Determine length
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1000
    # Catalan base
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

        a[i] = a[i-1] * (2*i - 1) / (2*i)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # set parameters for hill climbing
    eps = 0.05  # relative amount moved each iteration
    iteration = 0
    while time.time() < deadline:

        if i == j:

        delta = eps * best_seq[i]  # transfer up to eps proportion of value at i

        # propose new sequence

        cand[i] -= delta

            best_seq = cand

            eps = max(eps * 0.99, 1e-4)

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑type coefficients
        c_i = (2i choose i) / 4**i   (i = 0 … n‑1)
    and normalise them so that they sum to one.
    These coefficients are the power‑series of (1‑x)^‑½ and make the
    (infinite) self‑convolution perfectly flat.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence  c_{i} = c_{i-1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute the full (linear) convolution a*a using the FFT,
    return the maximal value and its index.
    """
    # FFT of a padded to fft_len (next power of two ≥ 2·len(a) )
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * len(a) - 1]          # keep only the linear part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def gradient_step(a: np.ndarray, step: float, fft_len: int) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective
        max_k (a*a)[k] .
    The sub‑gradient of the maximal term is 2·a_{k‑i} at each
    position i that participates in the maximal shift k.
    The vector is then moved opposite to the gradient,
    clipped to non‑negative values and re‑normalised.
    """
    _, k = max_convolution_fft(a, fft_len)

    n = len(a)
    # indices i that contribute to the maximal shift
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=np.int64)

    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max term

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)        # projection onto the non‑negative orthant
    s = a_new.sum()
    if s == 0.0:                          # safeguard against degenerate vectors
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Small random move: transfer a fraction *eps* of the mass of a random
    index i to a different random index j (i ≠ j). The vector stays
    normalised.
    """
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]                     # never removes more than what we have
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

    1. Initialise with the Catalan‑type sequence (coefficients of
       (1‑x)^‑½).  This gives an almost flat self‑convolution and already
       improves the bound to ≈π/2 ≈ 1.57.

    2. Run a projected sub‑gradient descent on the objective
       max(conv(a,a)) while keeping the vector non‑negative and normalised.
       The step size decays slowly; periodic calls to ``evaluate_sequence``
       keep the current best solution.

    3. Finish with a short stochastic local‑search phase that repeatedly
       moves a small amount of mass from one coordinate to another.
       Occasionally uphill moves are accepted with a tiny temperature to
       escape shallow plateaus.

    The best sequence seen before the time budget expires is returned.
    """

    deadline = time.time() + budget_s - 0.5   # leave a tiny margin for final return

    # ------------------------------------------------------------------

    except NameError:

    # FFT size – next power of two ≥ 2·n  (fast convolution)
    fft_len = 1 << ((2 * n - 1).bit_length())

        temperature *= 0.9997
        it += 1

    # ------------------------------------------------------------------

    return final_seq.tolist()
