def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    end_time = time.time() + budget_s - 5.0
    best_seq = None
    best_score = float('inf')
    
    # Try a handful of lengths and window shapes
    lengths = [200, 500, 1000]
    window_funcs = [make_hann, make_hamming, make_blackman, make_gaussian]
    # Also random start to explore
    while time.time() < end_time:
        # choose length randomly among lengths (or random in range)
        n = np.random.choice(lengths)
        # pick a generator
        if np.random.rand() < 0.7:
            # use a deterministic window shape
            gen = np.random.choice(window_funcs)
            if gen is make_gaussian:
                sigma = np.random.uniform(0.1, 0.3) * n
                seq = gen(n, sigma=sigma)
            else:
                seq = gen(n)
        else:
            # random positive vector, then smooth
            seq = np.random.rand(n)
            seq = np.convolve(seq, np.ones(3)/3, mode='same')
        # normalize sum to 1
        seq = seq / (np.sum(seq) + 1e-12)
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:

            best_seq = seq.tolist()
        # local refinement

            score2 = evaluate_sequence(seq_refine.tolist())

                best_seq = seq_refine.tolist()
    return best_seq

def projected_subgradient(a, max_iters=2000, time_limit=None):
    n = len(a)
    a = a.copy()
    for t in range(1, max_iters+1):
        if time_limit is not None and time.time()>time_limit:
            break
        conv = np.convolve(a, a)
        k = np.argmax(conv)
        # gradient g_i = 2*a_{k-i} if in range else 0
        g = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = 2 * a[j]
        # step size
        eta = 1.0 / np.sqrt(t)  # simple diminishing step
        a = a - eta * g
        # project onto simplex (sum=1, nonnegative)
        a = project_onto_simplex(a)
        # If any numerical issues lead to NaN, break and return None
        if np.isnan(a).any():
            return None
    return a

"""
propose_candidate
=================

A lightweight search that looks for a short (odd‑length) probability vector
whose self‑convolution has a strictly smaller peak than the uniform vector.
The uniform vector gives a score of exactly 2.0, because

    score = 2·n·max(conv(a,a)) / (sum(a))**2 = 2·n·(1/n) = 2.

For n = 3 the optimum can be obtained analytically by equalising the three
largest convolution entries.
The optimal vector is

    a = [8/15, 4/15, 1/5] ≈ [0.53333, 0.26667, 0.20000]

which yields   max(conv) = 64/225 ≈ 0.28444
and a score of   2·3·64/225 = 384/225 ≈ 1.706666…   (< 2).

The routine first evaluates this exact solution, then spends the remaining
time performing a cheap random search over a few other odd lengths
(5,7,9, …) in case a still better configuration is found.
All sequences are normalised to unit sum (the score is scale‑invariant)
and are kept non‑negative, satisfying the security checks of
``evaluate_sequence``.
"""

import time
import numpy as np

# If a global variable ``height_sequence_1`` exists we ignore it – the
# algorithm does not depend on any pre‑computed sequence.


def _analytic_n3() -> list[float]:
    """Exact optimum for length‑3 (proved by equalising the active constraints)."""
    a0 = 8.0 / 15.0
    a1 = 4.0 / 15.0
    a2 = 1.0 / 5.0   # 3/15
    return [a0, a1, a2]


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex of dimension n."""
    # Dirichlet(1,…,1) is uniform over the simplex.
    return np.random.dirichlet(np.ones(n))


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    # ``evaluate_sequence`` expects a plain Python list of floats.
    return evaluate_sequence(seq.tolist())


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a coefficient sequence with a score lower than 2.0.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.  The function will return before the budget
        expires (leaving a short safety margin).

    Returns
    -------
    list[float]
        The best sequence found.  It is guaranteed to consist of non‑negative
        numbers and to have a non‑zero sum, so ``evaluate_sequence`` will not
        return ``np.inf``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a safety margin

    # ------------------------------------------------------------------
    # 1) start with the known analytic optimum for n = 3
    # ------------------------------------------------------------------
    best_seq = _analytic_n3()
    best_score = _evaluate(np.array(best_seq))
    print(f"Initial analytic n=3 candidate score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) cheap random search on a few odd lengths (5,7,9,11,13,15)
    # ------------------------------------------------------------------
    odd_lengths = [5, 7, 9, 11, 13, 15]
    # Number of random samples grows with length because the search space
    # gets larger; we keep the total work modest so we stay well within 1 000 s.
    samples_per_n = {
        5: 12000,
        7: 10000,
        9:  8000,
        11: 6000,
        13: 4000,
        15: 3000,
    }

    for n in odd_lengths:
        nsamples = samples_per_n.get(n, 2000)
        print(f"--- random search for n={n} ({nsamples} samples) ---")
        for _ in range(nsamples):
            if time.time() > deadline:
                break
            a = _random_dirichlet(n)
            score = _evaluate(a)
            if score < best_score:
                best_score = float(score)
                best_seq = a.tolist()
                print(f"  New best (n={n}) score: {best_score:.6f}")

        # simple local refinement: tweak the best vector of this length
        # (if it happened to be found) by a few coordinate‑wise moves.
        if best_seq and len(best_seq) == n:
            a = np.array(best_seq, copy=True)
            # Perform a handful of tiny random redistributions that keep the sum = 1.
            for step in range(30):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, size=2, replace=False)
                # move a small amount delta from i to j
                delta = (np.random.rand() - 0.5) * 0.02   # |delta| ≤ 0.01
                delta = np.clip(delta, -a[i], a[j])      # keep non‑negative
                a[i] -= delta
                a[j] += delta
                score = _evaluate(a)
                if score < best_score:
                    best_score = float(score)
                    best_seq = a.tolist()
                    print(f"  Improvement after tweak (n={n}) score: {best_score:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) final sanity check – ensure the returned sequence is safe
    # ------------------------------------------------------------------
    # The evaluator will clamp negatives and NaNs, but we keep everything tidy.
    best_seq = [max(0.0, float(x)) for x in best_seq]
    total = sum(best_seq)
    if total < 1e-12:
        # fallback to a trivial uniform vector (should never happen)
        n = max(1, len(best_seq))
        best_seq = [1.0 / n] * n
        total = 1.0
    # Normalise – not required for the score, but makes the sequence nicer.
    best_seq = [x / total for x in best_seq]

    print(f"Best score after search: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
