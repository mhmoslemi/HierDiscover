Pseudo:

def simplex_projection(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Works for S>0.

def project_simplex(v, S):
    """Project vector v onto the simplex {x>=0, sum x = S}."""
    # algorithm from Duchi et al. 2008
    v = np.asarray(v)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simple stochastic search and gradient descent to minimize evaluate_sequence.
    Starts from several analytic shapes (uniform, triangular, sinusoidal, arcsine,
    exponential), evaluates them, picks the best, then performs projected
    sub‑gradient descent on the max‑convolution entry.
    Random Dirichlet restarts are also used when time permits.
    Returns the best sequence found before the deadline.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # use length from existing best if available
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000
    
    S = math.sqrt(2 * n)
    
    best_seq = None
    best_score = float('inf')
    
    def evaluate(seq):
        try:
            return evaluate_sequence(seq)
        except Exception:
            return float('inf')
    
    # analytic shape generators
    def uniform():
        a = np.full(n, S / n)
        return a
    
    def triangular():
        # symmetric triangular with peak at middle
        x = np.arange(n)
        peak = (n - 1) / 2
        a = peak - np.abs(x - peak) + 1  # values 1..peak+1
        a = a.astype(float)
        a = a / a.sum() * S
        return a
    
    def sine():
        x = np.arange(1, n + 1)
        a = np.sin(np.pi * x / (n + 1))
        a = a / a.sum() * S
        return a
    
    def cosine():
        x = np.arange(n)
        a = np.cos(np.pi * (x - (n-1)/2) / n)
        a = np.maximum(a, 0)  # ensure nonnegative
        a = a / a.sum() * S
        return a
    
    def arcsine():
        # avoid zero at ends by small epsilon shift
        eps = 1e-6
        i = np.arange(n) + 0.5
        a = 1.0 / np.sqrt(i * (n + 0.5 - i))
        a = a / a.sum() * S
        return a
    
    def exponential(lam=0.001):
        i = np.arange(n)
        a = np.exp(-lam * i)
        a = a / a.sum() * S
        return a
    
    # List of generators
    generators = [uniform, triangular, sine, arcsine, lambda: exponential(0.001), lambda: exponential(0.0005), lambda: exponential(0.005)]
    
    # Evaluate each analytic candidate
    for gen in generators:
        if time.time() > deadline:
            break
        a = gen()
        # ensure nonnegatives (should be)
        a = np.maximum(a, 0)
        # adjust small negatives to zero
        a_sum = a.sum()
        if a_sum == 0:
            continue
        # Normalize sum to S
        a = a * (S / a_sum)
        score = evaluate(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"New best from analytic {gen.__name__ if hasattr(gen, '__name__') else 'lambda'}: {score:.6f}")
    
    # Define projection onto simplex
    def project_simplex(v, S):
        """Project onto the simplex {x >= 0, sum x = S}."""
        v = np.asarray(v, dtype=np.float64)
        n = v.shape[0]
        # sort descending
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - S
        # find rho
        ind = np.arange(1, n+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all zero case
            theta = 0.0
        else:
            rho = ind[cond][-1] - 1  # zero-index
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w
    
    # Subgradient descent from current best
    def descent(start_vec, max_iters=300, init_alpha=1.0):
        nonlocal best_score, best_seq
        a = start_vec.copy()
        alpha = init_alpha
        # compute base evaluation
        base_score = evaluate(a.tolist())
        if base_score < best_score:
            best_score = base_score
            best_seq = a.copy()
        for it in range(max_iters):
            if time.time() > deadline:
                break
            # compute convolution
            b = np.convolve(a, a)
            max_b = b.max()
            # tolerance for active set
            eps = 1e-12 * max_b
            active_idxs = np.where(b >= max_b - eps)[0]
            # gradient of max_b
            grad = np.zeros_like(a)
            n_local = len(a)
            for k in active_idxs:
                i_start = max(0, k - (n_local - 1))
                i_end = min(n_local - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    if j < 0 or j >= n_local:
                        continue
                    if i == j:
                        grad[i] += a[i]
                    else:
                        grad[i] += a[j]
                        grad[j] += a[i]
            # take a step opposite to gradient
            direction = -grad
            if np.linalg.norm(direction) < 1e-12:
                # gradient zero -> local optimum
                break
            # try step with line search
            success = False
            for _ in range(5):
                candidate = project_simplex(a + alpha * direction, S)
                cand_score = evaluate(candidate.tolist())
                if cand_score < base_score:
                    a = candidate
                    base_score = cand_score
                    success = True
                    # possibly increase step size a bit
                    alpha *= 1.2
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = a.copy()
                    break
                else:
                    alpha *= 0.5
            if not success:
                # if cannot improve, shrink step drastically and continue
                alpha *= 0.5
                # optionally add small random perturbation to escape flat region
                if np.random.rand() < 0.1:
                    perturb = np.random.randn(n_local) * 0.01 * S
                    a = project_simplex(a + perturb, S)
                    base_score = evaluate(a.tolist())
            # optional occasional termination if below target
            if best_score < 1.51:
                break
    
    # Run descent from best analytic
    if best_seq is not None:
        descent(best_seq, max_iters=500, init_alpha=0.5)
    
    # Random restarts
    while time.time() < deadline - 0.1:
        # generate random Dirichlet with concentration parameter maybe 0.5 for sparsity
        alpha_param = np.random.uniform(0.1, 2.0)
        # sample gamma variables
        g = np.random.gamma(alpha_param, 1.0, size=n)
        a = g / g.sum() * S
        # a already nonnegative
        # maybe few descent steps
        descent(a, max_iters=150, init_alpha=0.5)
    
    # Return best as list
    if best_seq is None:
        # fallback: uniform
        best_seq = uniform()
    return best_seq.tolist()

def descent(start_vec, n, total_sum, deadline, best_seq, best_score, max_iters=500, init_alpha=0.5):
    a = start_vec.copy()
    alpha = init_alpha
    # compute initial evaluation
    curr_score = evaluate_sequence(a.tolist())
    if curr_score < best_score:
        best_score = curr_score
        best_seq = a.copy()
    # loop
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute convolution and max
        b = np.convolve(a, a)
        max_b = b.max()
        # active indices
        eps = 1e-12 * max_b
        active = np.where(b >= max_b - eps)[0]
        # compute gradient
        grad = np.zeros_like(a)
        n_local = n
        for k in active:
            i_start = max(0, k - (n_local - 1))
            i_end = min(n_local - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if j < 0 or j >= n_local:
                    continue
                if i == j:
                    grad[i] += a[i]
                else:
                    grad[i] += a[j]
                    grad[j] += a[i]
        # direction = -grad
        direction = -grad
        if np.linalg.norm(direction) < 1e-14:
            break
        # try step with line search up to 5 attempts
        improved = False
        for _ in range(5):
            candidate = project_simplex(a + alpha * direction, total_sum)
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < curr_score - 1e-12:  # improvement
                a = candidate
                curr_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                alpha *= 1.2
                break
            else:
                alpha *= 0.5
        if not improved:
            # small random perturbation occasionally
            if np.random.rand() < 0.05:
                perturb = (np.random.rand(n) - 0.5) * 0.02 * total_sum
                a = project_simplex(a + perturb, total_sum)
                curr_score = evaluate_sequence(a.tolist())
                if curr_score < best_score:
                    best_score = curr_score
                    best_seq = a.copy()
            # reduce step size
            alpha *= 0.7
        # early exit if we already beat target
        if best_score < 1.51:
            break
    return best_seq, best_score

"""
Simple stochastic optimizer for the step‑function bound.

The algorithm works in three phases:

1. **Analytic seeds** – generate a handful of deterministic shapes
   (uniform, triangular, sine, arcsine, exponential) whose sum is
   normalised to √(2 n).  The best of these is kept.

2. **Projected sub‑gradient descent** – starting from the best seed,
   repeatedly reduce the maximal convolution entry (which equals the
   evaluation value when the sum is fixed).  The gradient of the
   maximal entry is computed analytically, a step is taken against it,
   and the new point is projected back onto the simplex
   {x ≥ 0, Σx = √(2 n)}.  A simple back‑tracking line search adjusts the
   step length.

3. **Random restarts** – while time permits, draw random Dirichlet
   vectors (different concentration parameters) and run a short
   descent from each.  Any improvement updates the global best.

All helper functions are defined at module level (no closures), and the
search stops before the allocated time limit, returning the best
sequence found.
"""

import time
import math
import numpy as np

# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray, total_sum: float) -> np.ndarray:
    """Project vector `v` onto the simplex {x ≥ 0, Σx = total_sum}."""
    if total_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - total_sum
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def evaluate_internal(a: np.ndarray) -> float:
    """Fast internal evaluation: 2·n·max(conv(a,a)) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    max_b = np.convolve(a, a).max()
    return 2.0 * a.shape[0] * max_b / (s * s)


def compute_gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                grad[i] += a[i]
            else:
                grad[i] += a[j]
                grad[j] += a[i]
    return grad

    max_iters: int = 300,
    init_alpha: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns possibly updated (best_seq, best_score).
    """
    a = start_vec.copy()
    curr_score = evaluate_internal(a)
    if curr_score < best_score:
        best_score = curr_score
        best_seq = a.copy()

    alpha = init_alpha
    for _ in range(max_iters):
        if time.time() > deadline:
            break

        # convolution and active set
        b = np.convolve(a, a)
        max_b = b.max()
        eps = 1e-12 * max_b
        active = np.where(b >= max_b - eps)[0]

        # gradient
        grad = compute_gradient(a, active)
        if np.linalg.norm(grad) < 1e-14:
            break

        direction = -grad
        improved = False

        # simple back‑tracking line search (max 5 tries)
        for _ in range(5):
            cand = project_simplex(a + alpha * direction, total_sum)
            cand_score = evaluate_internal(cand)
            if cand_score < curr_score - 1e-12:
                a = cand
                curr_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                alpha *= 1.2
                break
            else:
                alpha *= 0.5

        if not improved:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.05:
                jitter = (np.random.rand(n) - 0.5) * 0.02 * total_sum
                a = project_simplex(a + jitter, total_sum)
                curr_score = evaluate_internal(a)
                if curr_score < best_score:
                    best_score = curr_score
                    best_seq = a.copy()
            alpha *= 0.7

        # early exit if we already beat a reasonable target
        if best_score < 1.51:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns a list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a safety margin

    # --------------------------------------------------------------
    # Determine problem size (use the known construction if available)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2000

    total_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Analytic seed shapes
    # --------------------------------------------------------------
    seed_funcs = [
        shape_uniform,
        shape_triangular,
        shape_sine,
        shape_arcsine,
        shape_exponential_0_001,
        shape_exponential_0_0005,
        shape_exponential_0_005,
    ]

    for func in seed_funcs:
        if time.time() > deadline:
            break
        a = func(n, total_sum)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a = a * (total_sum / a.sum())
        score = evaluate_internal(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"New best from {func.__name__}: {score:.6f}")

    # --------------------------------------------------------------
    # Descent from the best analytic seed
    # --------------------------------------------------------------
    if best_seq is not None:
        best_seq, best_score = descent(
            best_seq,
            n,
            total_sum,
            deadline,
            best_seq,
            best_score,
            max_iters=500,
            init_alpha=0.5,
        )

    # --------------------------------------------------------------
    # Random Dirichlet restarts (quick limited descents)
    # --------------------------------------------------------------
    while time.time() < deadline - 0.2:
        conc = np.random.uniform(0.1, 1.5)
        a = random_dirichlet(n, total_sum, conc)
        best_seq, best_score = descent(
            a,
            n,
            total_sum,
            deadline,
            best_seq,
            best_score,
            max_iters=150,
            init_alpha=0.5,
        )

    # --------------------------------------------------------------
    # Fallback to uniform if everything else failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = shape_uniform(n, total_sum)

    # Return plain Python list as required by the framework
    return best_seq.tolist()
