def solve_lp_active(f, M, max_active=10):
    n = len(f)
    # Determine active shift indices: top max_active conv values
    conv_ff = np.convolve(f, f)
    # find indices sorted descending by conv value
    top_indices = np.argpartition(-conv_ff, max_active-1)[:max_active]
    # Build constraints rows
    A = []
    b = []
    for k in top_indices:
        row = np.zeros(n)
        # i runs from max(0, k-(n-1)) to min(k, n-1)
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        # fill row entries
        row[j_vals] = f[i_vals]
        A.append(row)
        b.append(M)
    A = np.array(A)
    b = np.array(b)
    # Objective: maximize sum(b) -> c = -1
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # Solve LP using HiGHS
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if result.success:
        return result.x
    return None

def compute_max_feasible_t(conv_ff, conv_fg, conv_gg, M):
    n = len(conv_ff) # length 2n-1
    # compute for each shift the quadratic inequality
    # we want (1-t)^2 * A + 2t(1-t) * B + t^2 * C <= M
    # Convert to a2*t^2 + a1*t + a0 <= 0 where a0 = A - M
    a0 = conv_ff - M   # vector length
    a1 = 2 * (conv_fg - conv_ff)    # 2(B - A)
    a2 = conv_ff - 2*conv_fg + conv_gg  # A - 2B + C
    # For each entry, determine positive t range where D(t) <=0.
    t_candidates = []
    # We only care about positive t
    eps = 1e-12
    for i in range(len(conv_ff)):
        # Solve a2 t^2 + a1 t + a0 <= 0
        # with a0 <= 0 at t=0
        # Find the smallest positive root where D becomes 0 (if any)
        A = a2[i]
        B = a1[i]
        C = a0[i]
        if abs(A) < 1e-14:
            # Linear: B t + C <= 0 -> t <= -C/B if B>0
            if abs(B) < 1e-14:
                # Both A and B ~0: constant <=0 (since C <=0). No restriction.
                continue
            else:
                t_boundary = -C / B
                if B > 0:
                    if t_boundary > eps:
                        t_candidates.append(t_boundary)
                else:
                    # B<0: inequality holds for t >= -C/B, but we want t small positive; no restriction
                    continue
        else:
            discriminant = B*B - 4*A*C
            if discriminant < 0:
                continue
            sqrt_disc = np.sqrt(discriminant)
            t1 = (-B - sqrt_disc) / (2*A)
            t2 = (-B + sqrt_disc) / (2*A)
            # For a2>0 (convex parabola upward): D(t) <=0 between the roots
            # For a2<0: D(t)<=0 outside the roots.
            # Since D(0)=C <=0, we start within feasible region.
            # We need the smallest positive root where D(t) becomes 0.
            # Choose the positive root larger than eps.
            # For a2>0: the region is [t_left, t_right]; since at t=0 we are at left boundary (since C <= 0 maybe equality). So the right bound is the larger root >0.
            # For a2<0: region includes t>=0 up to the smaller positive root (if the region is t<=t_lower or t>=t_upper). Need analysis.
            # Simpler: compute positive roots > eps, take the smallest.
            positives = [root for root in (t1, t2) if root > eps]
            if not positives:
                continue
            t_boundary = min(positives)
            t_candidates.append(t_boundary)
    if not t_candidates:
        return 0.0
    t_max = min(t_candidates)
    # Clip to 1.0
    t_max = min(t_max, 1.0)
    return t_max

def compute_max_feasible_t(A, B, C, M):
    # D(t) = (1-t)^2 * A + 2*t*(1-t) * B + t**2 * C - M
    # Expand:
    a2 = A - 2*B + C
    a1 = 2*(B - A)
    a0 = A - M
    # For each element, find smallest positive root
    # We'll compute discriminant, handle cases.
    T = np.full_like(a0, np.inf, dtype=float)
    # Where a2 ~ 0:
    mask_linear = np.abs(a2) < 1e-12
    # linear case: a1 * t + a0 <= 0
    # if a1 >0 then t <= -a0/a1 else if a1 <=0 no restriction (since D(t) never >0 for t>=0)
    # a0 <=0 always at t=0 (since M >= A)
    a0_linear = a0[mask_linear]
    a1_linear = a1[mask_linear]
    # For a1 > 0:
    mask_pos_a1 = (a1_linear > 1e-12)
    T[mask_linear][mask_pos_a1] = (-a0_linear[mask_pos_a1] / a1_linear[mask_pos_a1])
    # For cases a2 !=0:
    mask_quad = ~mask_linear
    a2_q = a2[mask_quad]
    a1_q = a1[mask_quad]
    a0_q = a0[mask_quad]
    disc = a1_q**2 - 4*a2_q*a0_q
    # disc >=0 needed
    valid = disc >= 0
    sqrt_disc = np.sqrt(np.clip(disc, 0, None))
    # Compute both roots
    t1 = (-a1_q - sqrt_disc) / (2*a2_q)
    t2 = (-a1_q + sqrt_disc) / (2*a2_q)
    # For each case, we pick the smallest positive root greater than zero
    # Only consider roots > 1e-12
    roots = np.stack([t1, t2], axis=0)
    # select small positive root per element
    # Set any negative/zero roots to inf
    roots[roots <= 1e-12] = np.inf
    # Choose min across roots
    min_root = np.min(roots, axis=0)
    # feasible bound is min_root where valid (discriminant) else Inf
    T[mask_quad][valid] = min_root[valid]
    # Where not valid (disc<0) keep Inf
    # Finally, t_max = min(T) across all elements
    t_max = np.min(T)
    if t_max == np.inf or t_max < 0:
        return 0.0
    # bound by 1
    return min(t_max, 1.0)

"""
Propose a better step‑function (height) sequence.

The algorithm mixes two complementary ideas:

1. **Active‑set linear programming** –  
   Given the current sequence `f` (already normalized so that
   sum(f) = sqrt(2·n)), we solve a tiny LP that maximizes the total
   mass of a new sequence `g` while respecting the most critical
   convolution constraints `conv(f,g) ≤ M` where `M = max(conv(f,f))`.
   Only the shifts whose convolution with `f` is close to the current
   maximum are kept (active set), which makes the LP tiny (≈10 rows).

2. **Optimal mixing via line‑search** –  
   After a feasible `g` is obtained we evaluate the whole family
   `(1‑t)·f + t·g` (0 ≤ t ≤ 1).  For a few candidate `t` values we
   compute the resulting convolution analytically:
   `conv_t = (1‑t)²·conv(f,f) + 2·t·(1‑t)·conv(f,g) + t²·conv(g,g)`.
   The `t` that yields the smallest maximal convolution (hence the
   smallest evaluation score) is selected.  The new sequence becomes
   the current iterate.

Occasionally a random perturbation is applied to escape possible
local minima.  The process repeats until the time budget is exhausted,
always keeping the best sequence seen.

All sequences are kept non‑negative and have the required fixed sum,
so `evaluate_sequence` can be called directly.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative sequence so that sum = sqrt(2·len(seq))."""
    n = len(seq)
    target = np.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0:
        # fallback to uniform distribution
        return np.full(n, target / n, dtype=float)
    return seq * (target / s)


def _active_constraint_rows(f: np.ndarray, M: float, max_active: int = 5) -> np.ndarray:
    """
    Build a small set of rows for the LP:
    each row corresponds to a shift k where conv(f,f)[k] is among the
    largest values (active set).  The constraint is sum_i f[i] * b[k-i] ≤ M.
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    # pick the indices with largest values (ties are fine)
    if max_active >= len(conv_ff):
        active = np.arange(len(conv_ff))
    else:
        # argpartition gives unsorted top‑k; we keep them all
        active = np.argpartition(-conv_ff, max_active)[:max_active]
    rows = []
    for k in active:
        row = np.zeros(n, dtype=float)
        # j runs over positions where 0 ≤ k‑j < n  →  j ∈ [max(0,k‑(n‑1)), min(k,n‑1)]
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        j_vals = np.arange(j_start, j_end + 1)
        i_vals = k - j_vals
        row[j_vals] = f[i_vals]
        rows.append(row)
    return np.array(rows)


def _solve_lp_active(f: np.ndarray, M: float, max_active: int = 5) -> np.ndarray | None:
    """
    Solve the LP:
        maximize   sum(b)
        subject to conv(f,b) ≤ M for a few active shifts,
                    b ≥ 0.
    Returns the optimal vector `b` or None on failure.
    """
    n = len(f)
    A = _active_constraint_rows(f, M, max_active)
    b = np.full(A.shape[0], M, dtype=float)

    # objective   –sum(b)   (linprog minimizes)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _mixing_line_search(f: np.ndarray, g: np.ndarray,
                        conv_ff: np.ndarray, conv_fg: np.ndarray,
                        conv_gg: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Search a few `t` values for the convex combination
        h = (1‑t)·f + t·g
    and return the combination with the smallest maximum convolution.
    Also returns the corresponding maximal convolution value.
    """
    # candidate t's – dense near 0 and also full range
    candidates = np.concatenate([
        np.linspace(0.0, 1.0, 21),               # uniform grid
        np.random.rand(10)                       # some random points
    ])
    best_t = 0.0
    best_max = np.max(conv_ff)  # current max

    for t in candidates:
        # conv of the mixture using the quadratic expansion
        conv_mix = ((1 - t) ** 2) * conv_ff + \
                   2 * t * (1 - t) * conv_fg + \
                   (t ** 2) * conv_gg
        cur_max = conv_mix.max()
        if cur_max < best_max:
            best_max = cur_max
            best_t = t

    # construct the new sequence
    new_seq = (1 - best_t) * f + best_t * g
    return new_seq, best_max


def _random_perturb(seq: np.ndarray, magnitude: float = 0.02) -> np.ndarray:
    """
    Small random Gaussian perturbation respecting non‑negativity and
    the required sum (sqrt(2·n)).
    """
    n = len(seq)
    noise = np.random.randn(n) * magnitude
    cand = seq + noise
    cand = np.clip(cand, 0.0, None)
    # renormalize to the fixed sum
    target = np.sqrt(2.0 * n)
    s = cand.sum()
    if s == 0.0:
        return seq.copy()
    cand = cand * (target / s)
    return cand

    cur_seq = _normalize(init)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = len(cur_seq)

        conv_ff = np.convolve(cur_seq, cur_seq)
        M = conv_ff.max()
