Optimizes the step‑function heights for the given evaluation measure.
The algorithm explores several parametric families (Gaussian, sinusoidal,
triangular) for different lengths, normalises them, and then applies a
simple “mass‑flattening” local search that repeatedly moves a fraction of
the weight from the index contributing most to the current maximal
convolution entry to an index contributing least.  The objective
(2·n·max(conv)) is evaluated using a fast FFT based convolution, so the
search can run many iterations within the 1000 s budget.  The best
candidate found is returned.
"""
import time
import numpy as np

def _max_conv_fft(a):
    """Return the maximum value of the linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[: 2 * n - 1]
    return conv.max(), conv

def _score_from_seq(a):
    """Score = 2*n*max(conv) for a normalised sequence (sum=1)."""
    n = len(a)
    maxc, _ = _max_conv_fft(a)
    return 2.0 * n * maxc

def _flatten_max(a, max_iters=200, step_factor=0.2, min_step=1e-6):
    """Iteratively move mass from the index that contributes most to the current
    maximal convolution entry to the index that contributes least.
    The sum of the vector stays exactly 1, non‑negativity is enforced."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.shape[0]
    for it in range(max_iters):
        # compute full convolution and locate the maximal entry
        maxc, conv = _max_conv_fft(a)
        k = int(np.argmax(conv))
        # compute contributions w_i = a_{k-i} for indices that appear
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(a)
        w[mask] = a[j[mask]]
        # pick source (largest w) and sink (smallest positive w)
        # ignore indices with zero contribution in the sink selection
        i_src = int(np.argmax(w))
        pos = np.where(w > 0)[0]
        if pos.size == 0:
            break
        i_sink = int(pos[np.argmin(w[pos])])
        if i_src == i_sink:
            # no direction to move – stop
            break
        eps = step_factor * a[i_src]
        if eps < min_step:
            break
        # perform transfer
        a[i_src] -= eps
        a[i_sink] += eps
        # keep numerical stability
        a[i_src] = max(0.0, a[i_src])
        # sum is preserved because we transferred epsilon
        # optional early stop if the max does not improve
        new_maxc, _ = _max_conv_fft(a)
        if new_maxc >= maxc:
            # reduce step size and try again
            step_factor *= 0.5
            # revert the last transfer
            a[i_src] += eps
            a[i_sink] -= eps
            if step_factor < 1e-4:
                break
        # else keep the new a and continue
    return a.tolist()

    """Gaussian bell centred in the middle, sigma = sigma_factor * n."""
    x = np.arange(n) - (n - 1) / 2.0

    s = a.sum()
    if s == 0:

    else:

    return a.tolist()

    """sin^power window on [0,1]."""
    x = np.arange(n) / (n - 1)
    a = np.sin(np.pi * x) ** power
    s = a.sum()
    if s == 0:

    else:
        a = a / s
    return a.tolist()

def _make_triangular(n):
    """Symmetric triangular (tent) shape."""
    a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
    s = a.sum()
    a = a / s
    return a.tolist()

"""
Optimizes the step‑function heights for the supplied evaluation function.

The algorithm proceeds in four stages:

1. **Parametric families** – Gaussian bells, powers of a sine window and a
   symmetric triangular shape are generated for several lengths.  Each shape
   is normalised (∑aᵢ=1) and then refined by a simple “mass‑flattening’’ local
   search that repeatedly moves a fraction of weight from the index that
   contributes most to the current maximal convolution entry to an index that
   contributes least.  The objective 2·n·max(a∗a) (which equals the value
   returned by `evaluate_sequence` for a normalised sequence) is evaluated
   with a fast FFT‑based convolution, allowing many iterations.

2. **Historic solution** – if the global `height_sequence_1` is available it
   is normalised and refined in the same way; any improvement is kept.

3. **Random Dirichlet starts** – a random probability vector (Dirichlet
   distribution) of a random length is produced, refined and kept if it
   yields a better score.

4. **Short random tweaks** – occasional small transfers of mass between two
   positions of the current best sequence are tried, followed by a brief
   refinement.

All candidates are kept as plain Python lists of non‑negative floats.
The best sequence found before the time limit expires is returned.
"""

def _max_conv_fft(a):
    """
    Compute the maximum entry of the linear convolution a*a using FFT.
    Returns (max_value, full_convolution_array).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    # Use a power‑of‑two length for efficiency
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[: 2 * n - 1]
    return conv.max(), conv


def _score_from_seq(a):
    """
    For a sequence normalised to sum(a)=1, the evaluation function equals
    2 * n * max(a∗a).  This fast surrogate is used during the search.
    """
    n = len(a)
    maxc, _ = _max_conv_fft(a)
    return 2.0 * n * maxc


def _flatten_max(a, max_iters=200, step_factor=0.2, min_step=1e-8):
    """
    Simple local optimiser:
    repeatedly move a fraction of the mass from the index that contributes
    most to the current maximal convolution entry to the index that
    contributes the least (ignoring zero‑contributing sinks).  The sum stays
    exactly 1 and all entries remain non‑negative.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.shape[0]

    for _ in range(max_iters):
        maxc, conv = _max_conv_fft(a)
        k = int(np.argmax(conv))

        # contributions w_i = a_{k-i} for indices that appear in the max entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(a)
        w[mask] = a[j[mask]]

        # source = index with largest contribution, sink = smallest positive
        i_src = int(np.argmax(w))
        positive = np.where(w > 0)[0]
        if positive.size == 0:
            break
        i_sink = int(positive[np.argmin(w[positive])])

        if i_src == i_sink:
            break

        eps = step_factor * a[i_src]
        if eps < min_step:
            break

        # transfer eps from source to sink
        a[i_src] -= eps
        a[i_sink] += eps

        # numerical safety
        a[i_src] = max(0.0, a[i_src])

        # check if the move helped; if not, backtrack and shrink step
        new_maxc, _ = _max_conv_fft(a)
        if new_maxc >= maxc:
            # revert
            a[i_src] += eps
            a[i_sink] -= eps
            step_factor *= 0.5
            if step_factor < 1e-4:
                break
        # otherwise keep the new a and continue

    return a.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The method explores several parametric families,
    refines them with a lightweight mass‑flattening optimisation and finally
    performs a short random‑perturbation phase.  The best sequence found
    before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # reserve a few seconds for safety

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1. Explore a few handcrafted families for several lengths
    # --------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 6000]
    sigma_factors = np.linspace(0.12, 0.45, 5)   # Gaussian widths
    sin_powers = np.linspace(0.5, 5.0, 5)       # sin^p exponents

    for n in lengths:
        # ---- Gaussian family -------------------------------------------------
        for sf in sigma_factors:
            seq = _make_gaussian(n, sf)
            seq = _flatten_max(seq, max_iters=300, step_factor=0.2)
            score = _score_from_seq(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                print(f'New best (Gaussian)  n={n:4d} sigma={sf:.3f}  score={score:.6f}')
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

        # ---- Sin^p family ----------------------------------------------------
        for p in sin_powers:
            seq = _make_sin_power(n, p)
            seq = _flatten_max(seq, max_iters=300, step_factor=0.2)
            score = _score_from_seq(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                print(f'New best (Sin^p)    n={n:4d} p={p:.2f}   score={score:.6f}')
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

        # ---- Triangular shape -------------------------------------------------
        seq = _make_triangular(n)
        seq = _flatten_max(seq, max_iters=300, step_factor=0.2)
        score = _score_from_seq(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f'New best (Triangular) n={n:4d}      score={score:.6f}')
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2. Try to improve the historic construction (if present)
    # --------------------------------------------------------------
    try:
        hist = list(height_sequence_1)          # global provided by the framework
        s = sum(hist)
        if s > 0:
            hist = [x / s for x in hist]       # normalise to sum=1
            seq = _flatten_max(hist, max_iters=500, step_factor=0.15)
            score = _score_from_seq(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                print(f'Improved historic sequence, score={score:.6f}')
    except Exception:
        # variable missing or malformed – simply ignore
        pass

    # --------------------------------------------------------------
    # 3. Random Dirichlet starts + brief refinement
    # --------------------------------------------------------------
    while time.time() < deadline:
        n = np.random.randint(800, 4000)   # moderate size for quick updates
        r = np.random.exponential(scale=1.0, size=n)
        seq = (r / r.sum()).tolist()
        seq = _flatten_max(seq, max_iters=150, step_factor=0.2)
        score = _score_from_seq(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f'Random boost, n={n:4d}, score={score:.6f}')

        # ----------------------------------------------------------
        # 4. Occasionally perturb the current best solution
        # ----------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.2:
            a = np.array(best_seq, dtype=np.float64)
            i, j = np.random.choice(len(a), 2, replace=False)
            # transfer a small random fraction (≈±2.5%) from i to j
            delta = 0.05 * a[i] * (np.random.rand() - 0.5)
            # keep both entries non‑negative
            if delta > 0:
                delta = min(delta, a[i])
            else:
                delta = -min(-delta, a[j])
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            a = np.array(_flatten_max(a.tolist(), max_iters=80, step_factor=0.1))
            new_score = _score_from_seq(a)
            if new_score < best_score:
                best_score, best_seq = new_score, a.tolist()
                print(f'Local tweak improved best to {new_score:.6f}')

        # keep loop cheap
        if time.time() + 0.05 >= deadline:
            break

    # --------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------
    print(f'Final internal best score: {best_score:.6f}')
    # One final call to the official evaluator (optional, not required)
    try:
        official = evaluate_sequence(best_seq)
        print(f'Official evaluate_sequence returns {official:.6f}')
    except Exception:
        pass
    return best_seq
