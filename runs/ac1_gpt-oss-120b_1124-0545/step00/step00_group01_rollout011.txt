def project_onto_simplex(v, s=1):
    # returns w s.t. w >= 0 and sum(w) = s and w is closest to v in Euclidean norm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]  # last true
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    max_val = conv.max()

    low_i = max(0, k-(n-1))

def descent(a, max_iter=10000, time_deadline=None):
    best_a = a.copy()
    best_val = np.max(np.convolve(a,a))
    step = 0.5

def subgradient_update(a, step):
    n = len(a)
    conv = np.convolve(a, a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    k = ks[np.random.randint(len(ks))]
    # compute gradient:
    g = np.zeros_like(a)
    low_i = max(0, k - (n-1))
    high_i = min(n-1, k)
    idx_i = np.arange(low_i, high_i+1)
    idx_j = k - idx_i
    # valid pairs have idx_j in range [0,n-1]; already assured
    g[idx_i] = 2 * a[idx_j]
    # compute lambda to keep sum zero:
    lam = g.mean()  # since sum(g)/n
    d = -g + lam
    a_new = a + step * d
    # enforce sum constraint via projection onto simplex
    # target sum s:
    s = np.sqrt(2*len(a))
    a_new = project_simplex(a_new, s)
    # ensure positivity
    # projection already ensures nonnegativity
    return a_new, max_val

"""
Optimisation by sparse Sidon‑type constructions.

The evaluation of a step function is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

When the L1‑norm of the sequence is fixed to sqrt(2*n) the denominator
cancels and the score is simply the maximal convolution value.
For a sequence consisting of a constant height `w` on a set of
indices `S` and zero elsewhere we have

    w = sqrt(2*n) / |S|
    max(conv) = w**2 * (2 * Cmax - self)

where `Cmax` is the largest number of *unordered* pairs
(i, j) with i ≤ j, i, j ∈ S and i + j = t for any sum t,
and `self` is 0 or 1 (a self‑pair contributes only once).
Thus the score is roughly

    score ≈ 4 * n * Cmax / |S|**2

If we can find a large set `S` with `Cmax = 2` (each sum occurs at most
twice) the score drops below the current bound of 2.  A greedy random
construction of such a set is cheap (O(n·|S|)) and works very well for
`n` in the few‑thousand range.

The algorithm below repeatedly builds random Sidon‑type sets with a
collision limit `Cmax = 2`, keeps the best one found within the time
budget and returns the corresponding step heights.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: project a vector onto the simplex {x >= 0, sum(x)=s}
def _project_simplex(v, s):
    """Euclidean projection of v onto the simplex of total mass s."""
    if s < 0:
        raise ValueError("Simplex target must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    # Find the ridge where condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries forced to zero (should not happen for our use case)
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Greedy construction of a sparse index set with limited pair‑sum collisions
def _greedy_sparse_set(n, Cmax, rng):
    """
    Build a set S ⊆ {0,…,n‑1} such that each integer sum i+j (i≤j) appears
    at most Cmax times.  The order of candidates is randomised by rng.
    Returns (list_of_indices, max_collision).
    """
    order = np.arange(n, dtype=int)
    rng.shuffle(order)

    selected = []                         # list of chosen indices
    sum_counts = {}                       # dict: sum -> number of unordered pairs

    for i in order:
        # --- self‐pair check -------------------------------------------------
        s_self = i + i
        if sum_counts.get(s_self, 0) >= Cmax:
            continue

        # --- check all existing pairs ---------------------------------------
        feasible = True
        for j in selected:
            s = i + j
            if sum_counts.get(s, 0) >= Cmax:
                feasible = False
                break
        if not feasible:
            continue

        # --- add the index ----------------------------------------------------
        # update counts for self pair
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        # update counts for pairs with previous members
        for j in selected:
            s = i + j
            sum_counts[s] = sum_counts.get(s, 0) + 1
        selected.append(i)

    max_coll = max(sum_counts.values()) if sum_counts else 0
    return selected, max_coll


# --------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a low‑score step function.
    The algorithm repeatedly creates random Sidon‑type sets with at most
    two collisions per sum (Cmax = 2) for several lengths n.
    For each set we build the corresponding height vector
        a[i] = w   if i ∈ S,   else 0,
    where w = sqrt(2*n) / |S| guarantees the required normalisation.
    The best sequence seen before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a little safety margin

    # lengths that give a good trade‑off between speed and quality
    length_pool = [2000, 3000, 4000, 4595]

    best_seq = None
    best_score = float("inf")

    # Helper to evaluate a candidate set quickly (theoretical score)
    def _score_from_set(k, Cmax, n):
        # score ≈ 4 * n * Cmax / k**2   (exact: w**2 * (2*Cmax - self))
        return (4.0 * n * Cmax) / (k * k)

    # Main search loop
    while time.time() < deadline:
        # randomly pick a length
        n = int(rng.choice(length_pool))
        # collision limit (2 works very well; occasionally try 3 for robustness)
        Cmax = 2
        if rng.random() < 0.05:      # 5 % of attempts try a looser bound
            Cmax = 3

        # Build a random sparse set
        S, max_coll = _greedy_sparse_set(n, Cmax, rng)

        if len(S) < 3:
            continue        # trivially bad, try again

        # theoretical score (fast to compute)
        theor = _score_from_set(len(S), max_coll, n)

        # only keep candidates that could beat the current best
        if theor >= best_score:
            continue

        # build the concrete height vector
        weight = np.sqrt(2.0 * n) / float(len(S))
        heights = np.zeros(n, dtype=float)
        heights[S] = weight

        # final, exact evaluation (handles the tiny rounding differences)
        try:
            exact = evaluate_sequence(heights.tolist())
        except Exception:         # safety net – should never happen
            continue

        if exact < best_score:
            best_score = exact
            best_seq = heights.tolist()
            print(
                f"[{time.time() - start:6.1f}s] New best: {best_score:.6f} "
                f" (n={n}, |S|={len(S)}, Cmax={max_coll})"
            )
            # if we already beat the target we can stop early
            if best_score < 1.503:
                break

    # Fallback: uniform sequence (never used unless something goes wrong)

        uniform = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = uniform.tolist()
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence, score =", best_score)

    # final sanity check (the platform will also call evaluate_sequence)
    print("Finished search – best score found:", best_score)
    return best_seq
