def project_onto_simplex(v, s):
    # standard algorithm
    n=len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=123, budget_s=1000):

    # choose n: maybe len(height_sequence_1) or adjust

    seq0 = np.array(height_sequence_1, dtype=float)
    sum0 = seq0.sum()

    # ensure non-negative (should be)
    seq = seq0.copy()
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # functions for compute max conv
    def compute_max_conv(a):
        # compute convolution via fft:
        # zero pad length 2n-1
        fa = np.fft.rfft(a, n*2-1)
        conv = np.fft.irfft(fa * fa, n*2-1)
        # due to floating errors may get tiny negative values
        return conv.real
    # gradient for max conv index
    def gradient_max_conv(a):
        conv = compute_max_conv(a)
        max_val = conv.max()
        # handle ties via first index
        k = np.argmax(conv)
        # compute g = 2 * a shifted
        g = np.zeros_like(a)
        # For each j where index in range
        # g_j = 2 * a_{k - j} if 0 <= k-j < n
        # Using vectorized:
        idx = np.arange(n)
        partner_idx = k - idx
        # mask for valid
        mask = (partner_idx >= 0) & (partner_idx < n)
        g[mask] = 2 * a[partner_idx[mask]]
        return max_val, g
    # we may also compute if we need conv for final evaluation

def simplex_projection(v, s):
    # v is numpy array.
    # if sum(v) == s and all(v >=0): return v
    # else apply algorithm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def conv_fft(a):
    n = a.shape[0]
    L = 2*n - 1
    N = 1 << ((L-1).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv
```

def compute_max_conv(a):
    # compute convolution via FFT
    n = len(a)
    L = 2*n - 1
    N = 1 << ((L-1).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    maxb = conv.max()
    return maxb
def compute_score(a):
    n = len(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    maxb = compute_max_conv(a)
    return 2 * n * maxb / (sum_a ** 2)

"""
Proposed candidate generator.

The evaluation function is
    J(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)
where conv(a,a) is the discrete self‑convolution.
For non‑negative coefficients with a fixed L1‑norm (the function normalises
to sum(a)=sqrt(2n)), minimising J is equivalent to minimising the maximum
convolution value.

A simple construction that dramatically reduces the maximal convolution is
to enforce a perfect anti‑correlation between each coefficient a_i and its
mirror a_{n‑1‑i}.  Let the common average be

    c = sqrt(2*n) / n                                 (uniform value)

and choose a deviation d = α·c with 0 ≤ α ≤ 1.
For each pair (i, n‑1‑i) we set

    a_i       = c + α·c   (large)   if i is even,
    a_{n-1-i} = c - α·c   (small),

and the opposite assignment when i is odd.
All pairs sum to 2c, therefore the total L1‑norm stays √(2n).

For this alternating pattern the centre convolution value is

    conv_center = 2·(1‑α²)

while the convolution near the middle (k≈n/2) is

    conv_middle ≈ 1 + α²   (the small O(1/n) term is negligible).

The worst of the two controls the score, so the optimal α solves

    2·(1‑α²) = 1 + α²   ⇒   α = 1/√3 ≈ 0.57735,

giving a bound J ≈ 4/3 ≈ 1.333 which beats the target 1.503.

The code below builds the alternating sequence, scans a small grid of α
values around the theoretical optimum and returns the best sequence found.
All operations respect the non‑negativity and sum constraints required by
`evaluate_sequence`.
"""

import math
import time
import numpy as np

def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) using an FFT based convolution."""
    n = a.shape[0]
    L = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return float(conv.max())

def _score_from_array(a: np.ndarray) -> float:
    """Compute the same objective that evaluate_sequence returns."""
    n = a.shape[0]
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float('inf')
    max_b = _max_convolution(a)
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """
    Generate a step‑function sequence with a very low evaluation score.

    The construction uses an alternating anti‑correlated pattern with an
    optimised parameter α ≈ 1/√3.  A small grid search over α is performed,
    the best sequence (according to the provided evaluate_sequence) is
    returned.
    """
    # Fixed seed for reproducibility (not strictly needed here)
    np.random.seed(seed)

    # Use the length of the provided construction; fall back to 2000 if unavailable.
    n = globals().get('height_sequence_1')
    if isinstance(n, list) or isinstance(n, np.ndarray):
        n = len(n)
    else:
        n = 2000

    # Target sum required by the evaluation (sum = sqrt(2*n))
    target_sum = math.sqrt(2.0 * n)
    c = target_sum / n                     # uniform value

    # Search a modest range of α; the optimum is 1/√3 ≈ 0.57735
    alphas = np.linspace(0.45, 0.70, 51)   # dense enough for a quick scan
    best_score = float('inf')
    best_seq = None

    start_time = time.time()
    deadline = start_time + budget_s - 2  # a couple of seconds safety margin

    for alpha in alphas:
        if time.time() > deadline:
            break
        d = alpha * c
        a = np.empty(n, dtype=np.float64)

        half = n // 2
        # alternating signs: +1 for even i, -1 for odd i
