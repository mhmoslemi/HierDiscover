    # parameters

    decay = 0.999 # per iteration
    no_improve_iters = 0
    max_no_improve = 2000

    while time.time() < deadline:
        # compute conv
        conv = np.convolve(a, a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # compute gradient direction: g_i = a[partner] for partner within range
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = a[partner[mask]]
        # step
        step = eta * g
        new_a = a - step
        new_a[new_a < 0] = 0.0
        # Keep sum constant scaling to 1
        total = new_a.sum()
        if total == 0:
            # reinitialize randomly
            new_a = np.random.rand(n)
            total = new_a.sum()
        new_a = new_a / total

        # random perturbation occasionally
        if np.random.rand() < 0.001:
            # add small Gaussian noise
            noise = np.random.randn(n) * 0.01
            new_a = new_a + noise
            new_a = np.clip(new_a, 0, None)
            new_a = new_a / new_a.sum()

        # evaluate
        score = evaluate_sequence(new_a.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_a.copy()
            print(f"New best {best_score:.6f} at time {time.time() - start:.2f}s")
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # accept if improvement else maybe accept sometimes
        if score <= best_score + 1e-9: # changed
            a = new_a
        else:
            # maybe accept with prob decreasing
            prob = np.exp(-(score - best_score) / (best_score * 0.01))
            if np.random.rand() < prob:
                a = new_a

                # keep old a
                a = a

        # decrease step size

            no_improve_iters = 0
