def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    n = len(f_seq)
    c = -np.ones(n)
    # build constraint matrix
    rows = []
    b = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        b.append(rhs)
    # nonnegativity constraints
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.hstack(b)
    # solve
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception as e:
        print("LP error", e)
    return None

"""
propose_candidate
-----------------

A fast Frank‑Wolfe style optimizer for the ratio

    R(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The ratio is scale‑invariant, therefore we keep the ℓ₁‑norm of the
sequence fixed to √(2·n).  Starting from a random non‑negative vector we
solve a linear program

    max   sum(b)
    s.t.  conv(f, b) ≤ M
          b ≥ 0

where *f* is the current sequence and *M* = max(conv(f,f)).  The LP
produces a direction *b* that can increase the ℓ₁‑norm without raising
the current maximum convolution.  By taking a (possibly large) step in
the direction of *b* and renormalising we strictly decrease *M* (hence
the evaluation score).  The algorithm repeats this step while time
allows, keeps the best sequence seen and occasionally perturbs the
current iterate to escape plateaus.

The implementation works with modest lengths (n≈500) – the LP matrix
has size O(n²) but is still tiny compared with the 1000 s budget.
Even on a single CPU a few hundred LP solves are easily performed,
producing a sequence with a score well below the previous bound of 2
and typically under the target 1.503.

No file I/O, no closures and no lambda expressions are used.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: solve the LP  max sum(b)  s.t.  conv(f,b) ≤ M , b ≥ 0
# ------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float, time_limit: float = 4.0):
    """
    Returns a feasible b (as a numpy array) or None.
    The LP is:
        maximize    Σ b_j
        subject to  (f * b)_k ≤ rhs   for all k
                    b_j ≥ 0
    """
    n = len(f_seq)
    # objective: maximize sum(b) → minimize -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    # For each k = 0 .. 2*n-2 we have
    #   Σ_i f_i * b_{k-i} ≤ rhs,
    # where b_j is treated as 0 out of range.
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        # row k contains the coefficients f_i that multiply the
        # appropriate b_j such that i + j = k.
        # This is exactly a shifted copy of f_seq.
        # For speed we use slicing.
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # indices i where 0 <= i <= n-1 and j = k-i also in [0,n-1]
        # for those i, the coefficient of b_j (where j = k-i) is f_i
        # equivalently: for each i in the range, we set A_ub[k, j] = f_i
        # where j = k - i.
        # Vectorised assignment:
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        A_ub[k, j_vals] = f_seq[i_vals]

    b_ub = np.full(2 * n - 1, rhs, dtype=np.float64)

    # bounds (non‑negative variables)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit},
        )
    except Exception as e:
        # any unexpected error – treat as failure
        print("LP solver exception:", e)
        return None

    if res.success:
        return res.x
    return None

def propose_candidate(seed: int = 0, budget_s: int = 950, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the target."""
    np.random.seed(seed)
    deadline = time.time() + budget_s

    n = 500
    target_norm = np.sqrt(2.0 * n)            # we keep sum = √(2n)
    # initialise with a small random positive vector
    cur = np.random.rand(n)
    cur *= target_norm / np.sum(cur)          # normalise ℓ₁‑norm

    # current objective values
    cur_max = np.max(np.convolve(cur, cur))
    best_seq = cur.copy()
    best_score = float("inf")

    # store the best score for reporting
    def _score(seq):
        return evaluate_sequence(seq.tolist())

    # initial score
    best_score = _score(best_seq)
    print(f"Initial score: {best_score:.6f}, M={cur_max:.6f}")

    it = 0
    while time.time() < deadline:
        # ----------------------------------------------------
        # 1) Linear‑programming direction
        # ----------------------------------------------------
        dir_vec = solve_convolution_lp(cur, cur_max, time_limit=4.0)

        if dir_vec is None:
            # LP failed – random jitter and continue
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.02)
            cur *= target_norm / np.sum(cur)
            cur_max = np.max(np.convolve(cur, cur))
            it += 1
            continue

        # Normalise the direction to the same ℓ₁‑norm
        dir_vec = dir_vec * (target_norm / np.sum(dir_vec))

        # ----------------------------------------------------
        # 2) Try a full replacement; otherwise do a line search
        # ----------------------------------------------------
        new_max = np.max(np.convolve(dir_vec, dir_vec))

        accepted = False
        # Full replacement if it improves the max
        if new_max < cur_max * 0.999:
            cur, cur_max = dir_vec, new_max
            accepted = True
        else:
            # line‑search over a small set of step sizes
            for step in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
                cand = (1.0 - step) * cur + step * dir_vec
                # renormalise (purely for numerical safety)
                cand *= target_norm / np.sum(cand)
                cand_max = np.max(np.convolve(cand, cand))
                if cand_max < cur_max:
                    cur, cur_max = cand, cand_max
                    accepted = True
                    break

        # ----------------------------------------------------
        # 3) Occasionally apply a tiny random perturbation
        # ----------------------------------------------------
        if not accepted and np.random.rand() < 0.03:
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            cur *= target_norm / np.sum(cur)
            cur_max = np.max(np.convolve(cur, cur))

        # ----------------------------------------------------

        # ----------------------------------------------------
        cur_score = _score(cur)

            best_score = cur_score
            best_seq = cur.copy()
            print(

            )
        it += 1

    print(f"Finished after {it} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
