        k_max = np.argmax(b)
        max_b = b[k_max]

        grad = np.zeros_like(a)

        low = max(0, k_max - (n-1))
        high = min(n-1, k_max)

        i_range = np.arange(low, high+1)

        grad[i_range] = 2 * a[j_range]
        # Determine step size:
        # Using subgradient step: α = step_factor / (iteration+1)**0.5
        # Choose step_factor adaptively: maybe start with 0.5
        step_factor = 0.3  # approximate
        alpha = step_factor / np.sqrt(iteration + 1)
        # Proposed update
        a_new = a - alpha * grad
        # Enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)
        # Normalize sum to 1
        sum_new = np.sum(a_new)
        if sum_new == 0:
            # if all zero, reinitialize random
            a_new = np.random.rand(n)
            a_new = a_new / np.sum(a_new)
        else:
            a_new = a_new / sum_new
        a = a_new
        # evaluate
        # Could compute evaluate using evaluate_sequence directly (calls conv again). Instead compute evaluation from max_b and sum (1). compute score = 2*n*max_b/(1**2) = 2*n*max_b
        # Because sum = 1 after normalization, eval = 2*n*max_b.

        # But to avoid repeated conv, compute eval manually.
        # compute current_score = 2 * n * max_b
        curr_score = 2.0 * n * max_b
        if curr_score < best_score:
            best_score = curr_score
            best_a = a.copy()
            # print progress

def linear_convolve(a):
    n=len(a)
    Nfft=1<<((2*n-1).bit_length())
    A=np.fft.rfft(a,Nfft)
    B=A*A
    conv=np.fft.irfft(B,Nfft)[:2*n-1]
    return conv

import time

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Generate a monotone increasing integer sequence a_i = i+1 ( i = 0 … n‑1 )
    with n ≤ 1000.  The self‑convolution of this distribution has a maximal
    value

        max_{k} (a * a)_k = n (n+1)(n+2) / 6 ,

    the total sum is S = n (n+1) / 2 and therefore

        evaluate_sequence(a) = 2·n·max / S²
                             = (4/3)·(n+2)/(n+1) → 4/3 ≈ 1.333…

    for large n.  With n = 1000 the score is ≈ 1.336, well below the target
    1.503.  All entries lie in the allowed range [0, 1000] and the sequence is
    non‑negative, so it passes every security check in ``evaluate_sequence``.
    """
    # The per‑element ceiling of evaluate_sequence is 1000.0,
    # therefore we choose n = 1000 so that the last element equals the limit.
    n = 1000
    # Build the increasing sequence 1, 2, …, n  (as floats).
    sequence = [float(i + 1) for i in range(n)]

    # Optional: print the score for debugging/verification.
    try:
        score = evaluate_sequence(sequence)
        print(f"Generated linear increasing sequence of length {n}, score = {score:.6f}")
    except Exception as exc:          # pylint: disable=broad-except
        print("Error while evaluating the candidate sequence:", exc)
