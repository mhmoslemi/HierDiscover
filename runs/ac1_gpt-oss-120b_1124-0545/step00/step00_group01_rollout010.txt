def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        # compute score
        score = evaluate_sequence(p.tolist())
        if score < best_score:
            best_score = score
            best_seq = p.copy()
            # log
            print(f"Iter {iteration}: improved score {best_score:.6f}")
        # gradient descent step
        # softmax temperature lambda
        lam = 5.0 + 20.0 * iteration / 5000.0  # adapt, limited
        # To avoid overflow, subtract max
        conv_max = np.max(conv)
        weights_exp = np.exp(lam * (conv - conv_max))

"""
Propose a sequence of non‑negative heights that minimizes the evaluation
function.  The algorithm works on the probability vector p (the sequence
normalised to sum 1) and iteratively reduces the maximal value of the
self‑convolution p∗p.

Key ideas
---------

*   The objective 2 n max(c) / (∑a)² is scale‑invariant, therefore we can
    optimise a probability vector p (∑p=1) and later rescale it to stay below the
    1000‑value clipping required by ``evaluate_sequence``.
*   For a given p the convolution c = p∗p is a discrete probability distribution
    of the sum of two independent draws.  Its average value is 1/(2 n‑1); we
    minimise the peak ``max(c)``.
*   A smooth approximation of the max is the soft‑max (log‑sum‑exp).  For a
    temperature λ the loss  L(p)=log∑ₖexp(λ·cₖ) is differentiable and its
    gradient can be written as a correlation of the soft‑max weights with p.
*   Gradient descent on the simplex (projection onto the probability simplex
    after each step) drives the mass away from indices that contribute most to
    the current peak.  The temperature λ is increased slowly so that the
    optimisation first smooths the whole distribution and later focuses on the
    true maximum.
*   Random restarts (Dirichlet initialisation) and occasional Gaussian
    perturbations avoid getting stuck in poor local minima.

The routine respects the 1000 s budget, uses only NumPy (no external solvers)
and returns a list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i >= 0, sum x_i = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the largest rho that satisfies the condition
    rho = ind[u - cssv / ind > 0][-1]
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before the budget ends).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, each entry ≤ 1000).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0            # leave a little margin

    # Length of the problem – use the existing construction if present.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000                             # fallback length

    # ------------------------------------------------------------------
    # Evaluate a probability vector (it will be scaled later)
    # ------------------------------------------------------------------
    def _score(p_vec: np.ndarray) -> float:
        # convert to a plain list; evaluate_sequence does the clipping etc.
        return evaluate_sequence(p_vec.tolist())

    best_p = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop – several random restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- initial point -------------------------------------------------
        if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
            # start from the known construction, normalised to sum 1
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            init = np.maximum(init, 0.0)
            if init.sum() == 0:
                init = np.random.dirichlet(np.ones(n))
            else:
                init = init / init.sum()
        else:
            # fresh Dirichlet start – gives a dense random probability vector
            init = np.random.dirichlet(np.ones(n))

        p = init.copy()
        local_iters = 0
        # ----------------------------------------------------------------
        # Local optimisation (projected gradient descent on the simplex)
        # ----------------------------------------------------------------
        while time.time() < deadline and local_iters < 2500:
            # current convolution and score
            conv = np.convolve(p, p)                     # length 2n‑1
            cur_score = _score(p)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(
                    f"[{time.time() - start_time:6.1f}s] "
                    f"new best = {best_score:.6f}"
                )

            # ------- soft‑max on the convolution (temperature λ) -------------
            # λ grows slowly so we start with a smooth objective
            lam = 5.0 + 15.0 * (local_iters / 2500.0)      # range ≈[5,20]
            c_max = np.max(conv)
            # stabilise the exponentials
            exp_weights = np.exp(lam * (conv - c_max))
            w = exp_weights / exp_weights.sum()           # shape (2n‑1,)

            # ------- gradient of the soft‑max loss -------------------------
            # grad_i = 2·λ·∑_k w_k·p_{k‑i}  (indices outside 0..n‑1 omitted)
            # this is a cross‑correlation; NumPy's correlate does exactly that.
            p_full = np.concatenate([p, np.zeros(2 * n - 1 - n)])
            corr = np.correlate(w, p_full, mode="full")
            start = len(p_full) - 1                     # index ≡ shift 0
            grad = 2.0 * lam * corr[start:start + n]

            # ------- step size (scale‑invariant) ---------------------------
            # move only a fraction of the largest component to keep p non‑negative
            step = 0.05 / (np.max(np.abs(grad)) + 1e-12)

            p_next = p - step * grad
            p_next = _project_simplex(p_next)            # stay on the simplex

            # ------- occasional random jitter to escape plateaus ----------
            if local_iters % 400 == 0:
                noise = np.random.randn(n) * 0.001
                p_next = p_next + noise
                p_next = np.maximum(p_next, 0.0)
                p_next = _project_simplex(p_next)

            p = p_next
            local_iters += 1

        # after a restart, if we still have time we begin a new one
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Post‑process the best probability vector into a concrete height list
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.random.dirichlet(np.ones(n))

    # Scale to keep every entry comfortably below the 1000‑clip limit.
    max_allowed = 500.0
    scale = 1.0
    if best_p.max() > max_allowed:
        scale = max_allowed / best_p.max()
    heights = (best_p * scale).tolist()

    return heights
