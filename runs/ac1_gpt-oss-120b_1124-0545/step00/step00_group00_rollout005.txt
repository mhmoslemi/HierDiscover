def fast_score(seq):
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2 * len(seq) * np.max(conv) / (s * s))

def improve(seq, deadline, init_step=0.1, min_step=1e-6):
    n = len(seq)
    best_seq = seq.copy()
    best_score = fast_score(seq)
    step = init_step
    while time.time() < deadline:
        # compute convolution and max index
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        # compute contributions
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        i_range = np.arange(start, end+1)
        j_range = max_idx - i_range
        contrib = seq[i_range] * seq[j_range]
        # pick i_max with max contribution 
        idx_max_contrib = int(i_range[np.argmax(contrib)])
        # choose j_target as index with smallest value not equal idx_max_contrib
        idx_min_val = int(np.argmin(seq))
        if idx_min_val == idx_max_contrib:
            # choose second smallest
            sorted_idx = np.argsort(seq)
            idx_min_val = int(sorted_idx[1])
        # compute delta
        delta = min(step * seq[idx_max_contrib], seq[idx_max_contrib] * 0.5)
        if delta <= 0:
            # can't move
            step *= 0.5
            continue
        new_seq = seq.copy()
        new_seq[idx_max_contrib] -= delta
        new_seq[idx_min_val] += delta
        
        # optionally ensure sum unchanged (it is)
        # compute score
        new_score = fast_score(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            # possibly increase step
            step = min(step * 1.2, 0.5)
        else:
            # maybe accept with probability decreasing with time
            # or just discard
            # reduce step
            step *= 0.9

        # occasionally random perturb for escaping local minima
        if np.random.rand() < 0.01:
            # random small perturb at random index
            idx = np.random.randint(n)
            eps = (np.random.rand() - 0.5) * step * seq[idx]
            new_seq = seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + eps)
            # adjust another index to keep sum constant
            idx2 = np.random.randint(n)
            if idx2 != idx:

            new_score = fast_score(new_seq)
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                seq = new_seq

    deadline = time.time() + budget_s - 5

            s = seq_arr.sum()

    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
