    sums_used = set()
    positions = []
    for i in range(n):
        # test if adding i does not create duplicate sums
        conflict = False
        for j in positions:
            if (i + j) in sums_used:
                conflict = True
                break
        # also i+i sum
        if (i + i) in sums_used:
            conflict = True
        if not conflict:
            # add i

                sums_used.add(i + j)
            sums_used.add(i + i)
            positions.append(i)

def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected‑gradient descent for flattening the self‑convolution of a non‑negative
step‑function.

The evaluation metric is 2·n·max(f∗f)/(sum(f)²).  Since the metric is invariant
under scaling, we work with a probability vector p (p_i ≥ 0, Σp_i = 1) and
minimise 2·n·max(p∗p).  The maximum of the convolution is a non‑smooth
function, therefore we minimise a smooth upper–bound:
    J(p) = log( Σ_k exp(β·(p∗p)_k) ),
which converges to the max as β grows.  The gradient of J is
∇J = 2·conv(p_rev, w)[n‑1:2n‑1] with w the normalised soft‑max weights.
We take a step opposite to the gradient, project the result onto the simplex
(Σp = 1, p_i ≥ 0) and repeat.  Random restarts and a simple annealing schedule
provide exploration.  The best sequence encountered (converted back to a list)
is returned.

The algorithm runs for the allotted time (default 1000 s) and is
fully self‑contained – no external files or network access are required.
"""

import time
import numpy as np

def _project_simplex(v):
    """Projection of a vector onto the probability simplex (Σv = 1, v ≥ 0)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:      # all entries are zero – return uniform vector
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # numerical safety – renormalise exactly
    w /= w.sum()
    return w


def _smooth_max_gradient(p, beta):
    """
    Returns the gradient of log‑sum‑exp(β·(p∗p)) w.r.t. p.
    The gradient is 2·conv(p_rev, w)[n‑1:2n‑1] where
    w_k ∝ exp(β·(p∗p)_k) and Σ w_k = 1.
    """
    n = p.shape[0]

    # self‑convolution (full length 2n‑1)
    conv_full = np.convolve(p, p)

    # soft‑max weights (stable formulation)
    max_val = conv_full.max()
    exp_weights = np.exp(beta * (conv_full - max_val))
    w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

    # gradient via a single convolution (p reversed)
    p_rev = p[::-1]
    grad_full = np.convolve(p_rev, w)            # length 3n‑2
    # extract the part that corresponds to ∂/∂p_i
    grad = 2.0 * grad_full[n - 1:2 * n - 1]
    return grad

            try:
                score = evaluate_sequence(a.tolist())
            except Exception:   # should not happen – keep safety net
                score = float("inf")

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best: {best_score:.6f} "
                      f"(n={n}, iter={it})")

            # --- gradient step ----------------------------------------------
            # compute smooth‑max gradient
            grad = _smooth_max_gradient(a, beta)

            # descent direction (negative gradient) with zero‑sum constraint
            direction = -grad
            direction -= direction.mean()        # projection onto the sum‑zero subspace

            a_new = a + eta * direction
            a = _project_simplex(a_new)

            # adapt hyper‑parameters
            beta *= beta_increment
            eta *= eta_decay

        # end of one restart – try another with a fresh random start

    # Fallback in the unlikely case no sequence was ever accepted
    if best_seq is None:
        best_seq = np.ones(100) / 100.0

    # final sanity check – ensure we really improve over the initial upper bound
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final best score returned: {final_score:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
