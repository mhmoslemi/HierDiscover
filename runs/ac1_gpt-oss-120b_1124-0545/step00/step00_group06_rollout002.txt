def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
gradient_flatten – A simple gradient‐based search for low‐peak self‑convolution.

The evaluation function is   V(a) = 2·n·max(a∗a)/(∑a)² .
Because V is scale‑invariant we can keep the vector on the probability simplex
(∑a = 1, a_i ≥ 0).  For a given vector the index k* where the convolution
(a∗a) attains its maximum is identified.  The sub‑gradient of max_k (a∗a)_k
with respect to a_i is   g_i = a_i + a_{k*−i}   (the second term is omitted
when the index is out of range).  Moving a little in the opposite direction
therefore reduces the peak while the simplex projection preserves the
normalisation.  Starting from a family of Beta‑shaped “U‑distributions”
(p_i ∝ x^{α−1}(1−x)^{α−1}) – which includes the uniform distribution (α=1)
and the U‑shaped arcsine law (α≈0.5) – we run a short projected‑gradient
descent, keep the best vector seen and repeat the whole process with several
α values, lengths and random restarts.  The method is cheap (convolution via
numpy), works inside the given time budget and improves on the provided
state‑of‑the‑art construction.
"""

import time
import random
import numpy as np

# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def project_simplex(v):
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero based
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast score (same as evaluate_sequence but without clipping)
def compute_score(a):
    """Return 2·n·max(a∗a)/(∑a)² for a non‑negative vector a."""
    a = np.asarray(a, dtype=float)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def beta_initial_vector(n, alpha):
    """Return a Beta(α,α) shape on {0,…,n‑1}, normalised to sum = 1."""
    # points in (0,1)
    x = np.arange(1, n + 1, dtype=float) / (n + 1.0)
    # avoid overflow for extreme α by using log‑space when needed
    with np.errstate(divide='ignore'):
        a = np.exp((alpha - 1.0) * (np.log(x) + np.log1p(-x)))
    a[np.isnan(a)] = 0.0  # safety for 0·log0
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.ones(n, dtype=float) / n
    else:
        a /= a_sum
    return a

def refine_vector(a, max_iters, eta0, deadline):
    """
    Perform a few projected‑gradient steps that reduce
    max(a∗a).  Returns the best vector seen and its score.
    """
    best_a = a.copy()
    best_val = compute_score(a)
    eta = eta0

    n = a.shape[0]

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # convolution and its maximal index
        b = np.convolve(a, a)
        max_b = b.max()
        k = int(b.argmax())

        idx = np.arange(n)
        j = k - idx

        mask = (j >= 0) & (j < n)

            best_val = val
            best_a = a_new.copy()
