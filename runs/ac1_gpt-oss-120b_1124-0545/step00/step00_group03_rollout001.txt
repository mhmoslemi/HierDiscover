        seq = best_seq.copy()
        i, j = random.sample(range(n), 2)

        # epsilon is chosen so that the change is far below the precision
        # that influences the evaluation (≈1e‑12 relative).
        eps = (random.random() - 0.5) * 1e-6

        # Respect the global caps [0,1000] after the move.
        if not (0.0 <= seq[i] + eps <= 1000.0 and 0.0 <= seq[j] - eps <= 1000.0):
            continue

        seq[i] += eps
        seq[j] -= eps

        score = evaluate_sequence(seq)
        # We are minimizing the evaluation.
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[improve] score={best_score:.12f}")

    # -------------------------------------------------------------
    # 3.  Return the best candidate discovered.
    # -------------------------------------------------------------
    print(f"[final] length={len(best_seq)}  best_score={best_score:.12f}")
    return best_seq
