"""
Propose a better step‐function coefficients sequence.

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a))²
and is invariant to a global scaling of the vector.
Hence we can keep the ℓ₁‑norm fixed ( √(2n) ) and only try to
flatten the self‑convolution (i.e. the antidiagonal sums of the
rank‑1 matrix a·aᵀ).

The algorithm works as follows:

1. Fix the target L1‑norm  S = √(2·n)  (the same normalisation used by
   the state‑of‑the‑art construction).  All candidates are projected
   onto the simplex {x ≥ 0,  ∑x = S} after every update.

2. Initialise several random starts (uniform on the simplex).  For each
   start we run a simple projected sub‑gradient descent that tries to
   reduce the currently largest convolution entry.

   * For the current vector a we compute the full convolution `c = a∗a`
     using an FFT ( O(n log n) ).
   * Let `kmax` be the index where `c` attains its maximum (or all
     indices whose value is within 0.5 % of the maximum – this gives a
     smoother descent direction).
   * The gradient of `c[k]` w.r.t. a is known analytically:
       ∂c[k]/∂a_i = 2·a_{k-i}  (zero if the index is out of range).
     We build the sub‑gradient by summing the above expression over the
     selected `k` values.
   * Perform a projected step  a ← a – η·grad , clip to non‑negative,
     and rescale to the target L1‑norm.
   * The step size η decays geometrically; if no improvement is seen
     for a while it is reduced further.

3. Every few hundred iterations a tiny Gaussian jitter is added to
   avoid stagnation.

4. The best vector seen (according to the exact `evaluate_sequence`
   function) across **all** restarts is returned.

The method is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and respects the 1000 s budget.
"""

import time

def _proj_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x≥0, Σx = target_sum}."""
    total = v.sum()
    if total == 0.0:
        # fall back to the uniform distribution
        return np.full_like(v, target_sum / v.size, dtype=float)
    return v * (target_sum / total)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a with itself using an FFT."""
    n = a.size
    size = 2 * n - 1
    # rfft works with real input and is a bit faster than the full FFT
    f = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv


def _gradient_of_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Compute ∇_a (a∗a)[k] = 2·a_{k-i} for all i (zero where the index is out of range).
    Returned gradient already contains the factor 2.
    """
    n = a.size
    grad = np.zeros_like(a)

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=int)
    j = k - i

    # each pair (i, j) contributes a_j to grad_i and a_i to grad_j
    grad[i] += a[j]
    grad[j] += a[i]

    # for i == j (i.e. k even and i = k/2) the above adds a_i twice,
    # which is exactly the required 2·a_i.
    return grad

            # (1) compute full convolution and locate the current maximum
            conv = _conv_fft(a)
            max_b = conv.max()
            # current objective (exactly the one used in evaluate_sequence,
            # except that we skip the clipping performed there – all our values stay inside the safe range)
            cur_score = 2.0 * n * max_b / target_sum_sq

            # (2) keep the best ever seen
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration}] new best  score={best_score:.6f}  it={it}")

            # (3) simple improvement‑stagnation logic
            if cur_score < best_score - 1e-8:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 800:
                    # shrink the learning rate if we are stuck
                    eta *= 0.5
                    no_improve = 0

            # (4) build a (sub)gradient from the largest convolution entries
            #    we use every entry whose value is within 0.5 % of the maximum.
            thr = max_b * 0.995
            ks = np.where(conv >= thr)[0]
            if ks.size == 0:                     # fallback – should never happen
                ks = np.array([conv.argmax()], dtype=int)

            grad = np.zeros_like(a)
            for k in ks:
                grad += _gradient_of_shift(a, int(k))

            # (5) projected gradient step
            a -= eta * grad
            a = np.maximum(a, 0.0)               # keep non‑negative
            a = _proj_to_simplex(a, target_sum)  # re‑normalise

            # (6) occasional jitter to escape shallow plateaus
            if (it + 1) % 250 == 0:
                jitter = np.random.randn(n) * eta * 0.05
                a += jitter
                a = np.maximum(a, 0.0)
                a = _proj_to_simplex(a, target_sum)

            iteration += 1

        # after one restart we go for another one (if time permits)
        # – the outer while‑loop handles the time budget

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score returned = {final_score:.6f} (iterations: {iteration})")
    return best_seq.tolist()
