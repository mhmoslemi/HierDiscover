"""
Propose a candidate sequence using random sparse binary patterns.

The evaluation function is:
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a binary (0/1) vector of length n with m ones:
    average(conv) = sum(a)^2 / (2n-1) = m^2 / (2n-1)
    the objective equals  max(conv) / average(conv).

Thus minimizing the ratio is equivalent to making the number of ordered
pairs i+j=k (with a_i = a_j = 1) as uniform as possible.
Random subsets of size m ≈ 0.25‑0.45·n typically give max ≤ 1.5·average,
yielding scores below the target 1.503.

The algorithm repeatedly draws random subsets of indices, evaluates the
score with `evaluate_sequence`, and keeps the best one.  After a good
candidate is found it performs a cheap hill‑climbing phase consisting of
random swaps of a 1 ↔ 0 entry; any improving swap is kept.
All work stops before the allotted budget (default 990 s, leaving a
small safety margin for the caller).
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """Return a list of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total time budget in seconds. The function stops a few seconds
        before this limit to avoid being killed.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)

    # leave a small safety margin (10 s) for the caller / cleanup
    deadline = time.time() + max(0.0, budget_s - 10.0)

    # problem size – 1000 works well with the 1000‑value clamp and is fast.
    n = 1000

    best_seq: list[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – random sampling of sparse binary vectors.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a density between 20 % and 45 % (empirically good)
        density = np.random.uniform(0.20, 0.45)
        m = max(1, int(density * n))

        # create a binary vector with exactly m ones
        seq = np.zeros(n, dtype=float)
        idx = np.random.choice(n, size=m, replace=False)
        seq[idx] = 1.0

        # evaluate – the function itself performs all the security checks
        score = evaluate_sequence(seq.tolist())

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time():.2f}] New best score: {best_score:.6f} "
                  f"(density={density:.3f}, ones={m})")

        # early exit if we already beat the target comfortably
        if best_score < 1.45:      # a comfortable margin below 1.503
            break

    # ------------------------------------------------------------------
    # Phase 2 – simple hill‑climbing by swapping a 1 with a 0.
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback to uniform (score = 2)
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    # work on a mutable numpy array for speed
    cur_seq = np.array(best_seq, dtype=float)
    cur_score = best_score

    # pre‑compute positions of ones and zeros for fast updates
    while time.time() < deadline:
        ones = np.flatnonzero(cur_seq > 0.0)
        zeros = np.flatnonzero(cur_seq == 0.0)

        if len(ones) == 0 or len(zeros) == 0:
            break  # nothing to swap

        # pick a random 1 → 0 and a random 0 → 1 swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_seq = cur_seq.copy()
        new_seq[i] = 0.0
        new_seq[j] = 1.0

        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"[{time.time():.2f}] Improved by swap: {best_score:.6f}")

    # final sanity check – evaluate once more to be sure the returned
    # sequence is valid and its score is the best we have recorded.
    final_score = evaluate_sequence(best_seq)
    if final_score != best_score:
        # If something changed (unlikely), keep the evaluated value.
        best_score = final_score

    print(f"Finished search. Best score = {best_score:.6f} (length={n})")
    return best_seq
