"""
Propose a candidate step‑function (list of non‑negative heights) that minimizes
the evaluation metric

    2 * n * max(conv(a,a)) / (sum(a)**2)

where n = len(a).  The search works in two stages:

1.  Randomised parameterised families (single exponential, mixture of two
    exponentials and a piecewise exponential) are sampled.  Each family can
    generate a full‑length sequence with a very small “peak” in the convolution
    and therefore a low objective.  The best sample found during the allotted
    time is kept.

2.  A quick hill‑climbing / simulated‑annealing phase refines the best
    sequence by moving a tiny amount of mass from indices that contribute most
    to the current maximum convolution value to indices that contribute the
    least.  The refinement continues until time runs out.

No external files are accessed and the function respects the required
signature `propose_candidate`.  All intermediate progress is printed.
"""
import time
import numpy as np
import random

# The evaluation routine is already available in the execution environment:
# from <environment> import evaluate_sequence


def _exp_sequence(n, epsilon):
    """Single exponential a_i = exp(-epsilon * i)   (unnormalised)."""
    a = np.exp(-epsilon * np.arange(n, dtype=float))
    return a / a.sum()


def _double_exp_sequence(n, epsilon1, epsilon2, weight):
    """Mixture of two exponentials."""
    i = np.arange(n, dtype=float)
    a = weight * np.exp(-epsilon1 * i) + (1.0 - weight) * np.exp(-epsilon2 * i)
    return a / a.sum()


def _piecewise_exp_sequence(n, epsilon, gamma, frac):
    """
    Piecewise exponential:
        a_i = exp(-epsilon * i)                for i < m
        a_i = exp(-epsilon * m) * exp(-gamma * (i-m))  for i >= m
    with m = int(frac * n).
    """
    m = int(frac * n)
    i = np.arange(n, dtype=float)
    a = np.where(i < m,
                 np.exp(-epsilon * i),
                 np.exp(-epsilon * m) * np.exp(-gamma * (i - m)))
    return a / a.sum()


def _hill_climb(seq, best_score, deadline):
    """
    Simple refinement: move a small amount of mass from the index that
    contributes most to the peak convolution to a random other index.
    Uses an ever‑decreasing step size.
    """
    n = len(seq)
    step = 0.05                # relative step size
    while time.time() < deadline and step > 1e-6:
        # locate the shift where the convolution is maximal
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))

        # contribution of each index to that peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = seq[i] * seq[j]

        i_high = int(np.argmax(contrib))
        j_high = k_max - i_high
        # pick a low‑contribution index (try a few random draws)
        i_low = None
        for _ in range(5):
            cand = random.randint(0, n - 1)
            j = k_max - cand
            if 0 <= j < n and (seq[cand] * seq[j]) < 1e-12:
                i_low = cand
                break
        if i_low is None:                     # fallback – just choose a random one
            i_low = random.randint(0, n - 1)
        j_low = k_max - i_low

        # move a tiny amount of mass from the high‑contribution pair to the low one
        delta = min(seq[i_high], seq[j_high]) * step
        if delta <= 0:
            step *= 0.9
            continue

        new_seq = seq.copy()
        new_seq[i_high] = max(0.0, new_seq[i_high] - delta)
        new_seq[j_high] = max(0.0, new_seq[j_high] - delta)

        # distribute the same total mass to the low‑contribution pair
        new_seq[i_low] += delta
        new_seq[j_low] += delta

        # numerical safety – renormalise to keep the sum exactly 1
        new_seq /= new_seq.sum()

        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            seq = new_seq
            print(f"Hill‑climb: improved score {best_score:.6f}")
        else:
            # slowly reduce the step if no improvement
            step *= 0.99
    return seq.tolist(), best_score

            gamma = tg / n
            frac = random.uniform(0.1, 0.7)   # where the switch happens
            seq = _piecewise_exp_sequence(n, eps, gamma, frac)

        # normalise (the generators already normalise, but guard against numerical drift)
        seq /= seq.sum()

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Iter {it:06d}: new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – lightweight hill‑climbing refinement
    # ------------------------------------------------------------------
    best_seq.tolist()            # ensure we have a plain list for the final call
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    print(f"Finished. Best score found: {best_score:.6f}")
    return best_seq
