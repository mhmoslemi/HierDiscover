def conv_self(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    fb = fa * fa
    conv = np.fft.irfft(fb, m)[:2*n-1]
    return conv
```

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 2000  # choose length maybe same as height_sequence_1 if exists
    # Option: use length of height_sequence_1 if it's longer.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    # initialize a
    a = np.random.rand(n)  # uniform random [0,1)
    a = a / a.sum()  # sum to 1
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # compute parameters
    lr = 0.5
    it = 0
    while time.time() < deadline and it < 10000:
        conv = conv_self(a)  # convolution length 2n-1
        max_b = conv.max()
        # compute objective
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {it}, improved score: {best_score:.6f}")
        # compute avg
        sum_a = a.sum()
        avg = sum_a**2 / (2*n - 1)
        diff = conv - avg
        # gradient of variance (optional) - but we want to move to reduce max.
        # Instead more direct: compute subgradient for max_b: for max index(s)
        max_idx = np.flatnonzero(conv == max_b)
        # compute gradient tick for max index
        # compute gradient = 2 * a reversed aligned at max_idx
        g = np.zeros_like(a)
        for idx in max_idx:
            # g_i = 2*a[max_idx - i] if within range
            # Equivalent to vector shift
            # We can compute using indexing
            i = np.arange(n)
            j = idx - i
            mask = (j >=0) & (j < n)
            g[mask] += 2 * a[j[mask]]
        # Use gradient descent step
        a = a - lr * g
        # enforce non-negative, small epsilon
        a = np.maximum(a, 0.0)
        # renormalize sum to 1
        total = a.sum()
        if total < 1e-12:
            # reinitialize random
            a = np.random.rand(n)
            total = a.sum()
        a = a / total
        # adapt learning rate maybe lower gradually
        lr *= 0.9995
        # occasional random perturbation
        if it % 200 == 0 and it > 0:
            # random shuffle small portion
            rand_idx = np.random.choice(n, size=int(0.01*n), replace=False)
            a[rand_idx] = np.random.rand(len(rand_idx))
            a = np.maximum(a, 0)
            a = a / a.sum()
        it += 1

    return best_seq.tolist()

"""
Propose a candidate step‑function using a fast gradient‑descent scheme that
flattens the self‑convolution of the height vector.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is equivalent (for a fixed sum) to minimising the maximal entry of
the convolution.  By reducing the variance of the convolution vector we drive
all its entries towards the average value, which makes the maximum close to
the theoretical lower bound 2 n/(2 n‑1) ≈ 1.0.  The program performs many
iterations of a projected Adam update on the variance loss, using FFTs for
O(n log n) per step, and keeps the best sequence seen during the allotted
time.

The implementation is completely self‑contained, respects the security
checks of ``evaluate_sequence`` and works for any reasonable length (the
default is taken from the supplied ``height_sequence_1`` if present).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Convolution a * a using FFT of given padded size."""
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[: 2 * len(a) - 1]
    return conv


def _correlate_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear correlation of x with y (i.e. sum_i x[i] * y[i + lag]).
    Returned length = len(x) + len(y) - 1.
    """
    Fx = np.fft.rfft(x, size)
    Fy = np.fft.rfft(y, size)
    corr = np.fft.irfft(Fx * np.conj(Fy), size)[: len(x) + len(y) - 1]
    return corr


def _flatten_step(a: np.ndarray, iters: int, lr: float, size_fft: int) -> np.ndarray:
    """
    Perform a few Adam steps that minimise the variance of the self‑convolution.
    """
    n = a.shape[0]
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    for t in range(iters):
        # self‑convolution
        conv = _conv_self_fft(a, size_fft)                     # length 2n-1
        s = a.sum()
        avg = s * s / (2 * n - 1)                              # average of conv
        diff = conv - avg

        # gradient of variance = 4 * correlation(diff, a)
        corr_full = _correlate_fft(diff, a, size_fft)        # length 3n-2
        grad = 4.0 * corr_full[n - 1 : n - 1 + n]            # slice to length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # non‑negativity and renormalisation
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total < 1e-12:                 # avoid division by zero – re‑initialise
            a = np.random.rand(n)
            total = a.sum()
        a = a / total

        # slowly decay the learning rate
        lr *= 0.9995
    return a
