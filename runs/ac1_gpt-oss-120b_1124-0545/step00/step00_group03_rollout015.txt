def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # initial length
    n = 300

    # initial sequence: exponential shape maybe with lambda = 0.01
    lam = 0.002  # try moderate
    a = np.exp(-lam * np.arange(n))
    a = a / np.sum(a)  # normalize sum to 1 (scale later)
    seq = a.copy()

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print('Initial score', best_score)

    # gradient descent loop
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]

        # compute gradient g = 2 * seq shifted by max_idx
        g = np.zeros_like(seq)
        # For i where max_idx - i in [0, n-1]:
        low = max(0, max_idx - (n - 1))
        high = min(n-1, max_idx)
        # indices i in [low, high]
        # for these, g[i] = 2 * seq[max_idx - i]
        idx_range = np.arange(low, high+1)
        g[idx_range] = 2 * seq[max_idx - idx_range]

        # step size adapt: maybe eta = 0.05 / (iteration+1)**0.5
        # Use simple constant small step.
        eta = 0.01 * (best_score - 1.5) / 0.5  # not negative
        if eta <= 0:
            eta = 0.001
        # gradient descent step
        seq_new = seq - eta * g
        # ensure nonnegative
        seq_new = np.maximum(seq_new, 0.0)
        # renormalize to sum 1
        s = np.sum(seq_new)
        if s < 1e-12:
            seq_new = seq  # skip if too small
        else:
            seq_new /= s

        # evaluate
        score = evaluate_sequence(seq_new.tolist())
        # if improvement accept
        if score < best_score:
            best_score = score
            best_seq = seq_new.copy()
            print('Improved score', best_score, 'iteration')
        # optionally, accept this new seq as current regardless of improvement
        seq = seq_new

        # occasional random perturbation to escape local minima
        if np.random.rand() < 0.01:
            # add small random noise to seq
            noise = np.random.normal(scale=0.01, size=n)
            seq = np.maximum(seq + noise, 0.0)
            seq /= np.sum(seq)
    # after loop, scale best_seq to have sum >0.01 (already sum=1)
    result = best_seq.tolist()
    return result

"""
propose_candidate
-----------------

A simple stochastic optimizer that tries to flatten the self‑convolution of a
non‑negative step function.  The evaluation function is

    E(a) = 2·n·max(a*a) / (sum(a)**2)

which is scale‑invariant.  The optimizer works with a vector whose entries sum
to one (the scale does not matter) and repeatedly moves a small amount of mass
from positions that contribute most to the current maximal convolution value to
positions that contribute the least.  Occasionally a random perturbation is
added and occasional restarts are performed.  The best sequence found before the
deadline is returned.

The method is fast (O(n) per iteration) and, on modest lengths (n≈200‑300),
typically drives the evaluation from the uniform value 2.0 down below the
target 1.503.
"""

import time
import numpy as np
from typing import List


def _max_convolution(seq: np.ndarray) -> (float, int, np.ndarray):
    """
    Compute the full convolution of seq with itself, return the maximal value,
    the index where it occurs and the full convolution array.
    """
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx, conv


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Generate a step‑function (list of non‑negative floats) that minimises the
    provided evaluate_sequence() metric.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds. The function will stop a few seconds
        before this limit and return the best sequence found so far.

    Returns
    ----------
    List[float]
        The best coefficient sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # leave a tiny safety margin

    # --------------------------------------------------------------------
    # 1. Choose a problem size.
    #    Larger n gives more freedom but also more work per iteration.
    #    n≈250 works well within the time budget.
    # --------------------------------------------------------------------
    n = 250

    # --------------------------------------------------------------------
    # 2. Initial sequence – start from a smooth decreasing profile.
    #    The exact scale does not matter because the score is scale invariant,
    #    we normalise so that the total mass equals one.
    # --------------------------------------------------------------------
    lam = 0.02                     # exponential decay rate
    seq = np.exp(-lam * np.arange(n))
    seq /= seq.sum()               # normalise to unit sum

    best_seq = seq.copy()
    best_score = float("inf")

    # --------------------------------------------------------------------
    # 3. Main optimisation loop.
    # --------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # current max convolution and its location
        cur_max, k_max, _ = _max_convolution(seq)

        # contribution of each index to that maximal convolution entry
        # (only indices that actually participate have a non‑zero contribution)
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[valid] = seq[idx[valid]] * seq[j[valid]]

        # indices that contribute most / least to the current peak
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        # move a small amount of mass from a “heavy” index to a “light” one
        delta = min(seq[i_max] * 0.1, 0.005)   # keep moves modest
        if delta > 0 and i_max != i_min:
            seq[i_max] -= delta
            seq[i_min] += delta

        # occasional renormalisation to avoid drift caused by rounding
        if it % 500 == 0:
            seq = np.maximum(seq, 0.0)
            total = seq.sum()
            if total > 0:
                seq /= total

        # random shake‑up to help escape shallow local minima
        if np.random.rand() < 0.005:
            noise = np.random.normal(scale=0.01, size=n)
            seq = np.maximum(seq + noise, 0.0)
            seq /= seq.sum()

        # evaluate the current candidate (the external function handles all checks)
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            # any unexpected problem – just skip this iteration
            continue

        # keep the best one seen so far
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # give a quick progress report
            print(f"Iter {it:6d} – new best score: {best_score:.6f}")

        # simple restart logic: if we have stagnated for a while, start anew
        if it % 4000 == 0 and best_score > 1.7:
            # generate a fresh random Dirichlet vector
            seq = np.random.gamma(shape=0.5, scale=1.0, size=n)
            seq /= seq.sum()

    # --------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python floats).
    # --------------------------------------------------------------------
    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
