def solve_convolution_lp(f, rhs):
    n = len(f)
    c = -np.ones(n)  # maximize sum => minimize -sum
    rows = []
    rhs_vec = []
    # inequality: conv(f, b) <= rhs for each k
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv contribution: sum_i f[i] * b[k-i]
        # for each i where 0<=k-i<n, b[k-i] coefficient is f[i]
        # So we fill row[j] = f[i] where j = k - i
        # Equivalent to row[j] = f[k - j] if 0 <= k - j < n
        for j in range(max(0, k - (n - 1)), min(n, k + 1)):
            i = k - j
            row[j] = f[i]
        rows.append(row)
        rhs_vec.append(rhs)
    # nonnegativity:
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.concatenate(rhs_vec)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return result.x
    else:
        return None
```

"""
Hybrid optimizer for the step‑function inequality.

The search combines two complementary ideas:

1. **Projected gradient descent on a smooth approximation of the
   max‑convolution**.  The objective
   `max_k (a*a)[k]` is approximated by a soft‑max with a temperature
   parameter `beta`.  The gradient of this surrogate is
   ∂/∂a_i  Σ_k w_k (a*a)[k] = 2 Σ_j a_j w_{i+j},
   which can be evaluated with a few vectorised slices.
   After each step we project onto the non‑negative orthant and
   renormalise the vector (the objective is homogeneous, so fixing the
   ℓ₁‑norm is harmless).

2. **Linear‑programming refinement**.  Given a current sequence `f`,
   we solve the LP  
       maximise   Σ_j g_j
       subject to  (f * g)_k ≤ M_f   for all k,
                    g_j ≥ 0,
   where `M_f = max_k (f*f)[k]`.  The optimal `g` adds as much mass as
   possible without increasing the convolution bound.  We then
   mix `f` and `g` using a line‑search over the mixing coefficient `t`
   and keep the best improvement.  This step reduces `M_f`
   monotonically and is cheap for moderate lengths (n≈300).

The algorithm runs several random restarts with different lengths,
alternating the two phases while respecting the overall time budget.
The best sequence encountered is returned.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of the convolution values; used as a smooth max."""
    m = np.max(conv)
    exp_vals = np.exp(beta * (conv - m))
    s = np.sum(exp_vals)
    if s == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient_step(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    One projected gradient step for the smooth max‑convolution objective.
    The vector is kept non‑negative and renormalised to unit ℓ₁‑norm.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                # length 2n‑1
    w = _softmax_weights(conv, beta)        # same length
    grad = np.zeros_like(a)

    # grad_i = 2 * Σ_j a_j * w[i+j]
    for j in range(n):
        grad += 2.0 * a[j] * w[j:j + n]

    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    total = np.sum(a_new)
    if total > 0.0:
        a_new /= total
    else:
        # occasional re‑initialisation if everything collapsed
        a_new = np.random.rand(n)
        a_new /= np.sum(a_new)
    return a_new


def _make_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A where A[k, j] = f_{k‑j} (0 if out of range).
    Shape: (2n‑1) × n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # fill column‑wise using the pattern described in the analysis
    for j in range(n):
        A[j:j + n, j] = f[: rows - j]
    return A


def _solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise Σ g_j
        s.t. (f * g)_k ≤ rhs  (k = 0 … 2n‑2)
             g_j ≥ 0.
    Returns the optimal `g` or None if the solver fails.
    """
    n = f.shape[0]
    # objective (maximise sum → minimise -sum)
    c = -np.ones(n, dtype=float)

    # convolution constraints
    A_conv = _make_conv_matrix(f)           # (2n‑1) × n
    b_conv = np.full(2 * n - 1, rhs, dtype=float)

    # non‑negativity constraints: -I * g ≤ 0
    A_ub = np.vstack([A_conv, -np.eye(n, dtype=float)])
    b_ub = np.concatenate([b_conv, np.zeros(n, dtype=float)])

    # solve with HiGHS (fast for moderate sizes)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None


def _lp_direction(f: np.ndarray) -> np.ndarray | None:
    """
    Compute a feasible direction `g` that respects the current
    convolution bound of `f`.  The returned vector has the same ℓ₁‑norm
    as `f` (∑g = √(2n)), which guarantees mixing does not change the
    denominator of the evaluation function.
    """
    n = f.shape[0]
    s = np.sum(f)
    if s == 0.0:
        return None
    # normalise to the usual scaling √(2n)
    f_norm = f * np.sqrt(2.0 * n) / s
    rhs = np.max(np.convolve(f_norm, f_norm))

    g_raw = _solve_convolution_lp(f_norm, rhs)
    if g_raw is None:
        return None

    sg = np.sum(g_raw)
    if sg == 0.0:
        return None
    # renormalise to the same ℓ₁‑norm as `f`
    g_norm = g_raw * np.sqrt(2.0 * n) / sg
    return g_norm


def _best_mixing(f: np.ndarray, g: np.ndarray, cur_M: float) -> tuple[float, float]:
    """
    Simple line‑search for the mixing coefficient t ∈ (0,1] that yields
    the smallest max‑convolution of h = (1‑t) f + t g.
    Returns (best_t, best_M).  If no improvement is possible, best_t = 0.
    """
    best_t = 0.0
    best_M = cur_M

    # coarse grid
    for t in np.linspace(0.0, 1.0, 21)[1:]:
        h = (1.0 - t) * f + t * g
        M = np.max(np.convolve(h, h))
        if M < best_M:

    if best_t > 0.0:

        hi = min(1.0, best_t + 0.1)
        for t in np.linspace(lo, hi, 21):
            h = (1.0 - t) * f + t * g
            M = np.max(np.convolve(h, h))
            if M < best_M:
                best_M = M
                best_t = t
    return best_t, best_M

        best_seq = [float(np.sqrt(2.0 * n) / n)] * n

        print(f"[Fallback] using uniform n={n}  score={best_score:.6f}")

    return best_seq
