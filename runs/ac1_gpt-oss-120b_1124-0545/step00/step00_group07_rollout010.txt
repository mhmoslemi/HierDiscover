    # v can be any real vector

    return w

def proj_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find rho
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def conv_max_fft(a):
    n = len(a)
    m = 1 << ((2*n - 1).bit_length())
    # compute rfft of a
    A = np.fft.rfft(a, m)
    conv = np.fft.irfft(A * A, m)[:2*n-1]
    # due to numerical error, ensure nonnegative
    conv = np.real(conv)
    # conv may have tiny negative rounding, clip.
    return np.max(conv)
```

    seq = start_seq.copy()

    seq = proj_simplex(seq, S)

    best_score = compute_objective(seq)

    step = S * 0.2

    while time.time() - start < time_limit:
        # compute current convolution max and index
        conv = conv_full_fft(seq)
        k_max = int(np.argmax(conv))
        # gradient descent step

        improved = False

            new_score = compute_objective(new_seq)
            if new_score < best_score:

            if new_score < compute_objective(seq):

                break

            seq = proj_simplex(seq, S)
        step = max(step * decay, min_step)
    return best_seq, best_score

"""
Propose a low‑maximum convolution step sequence.

The algorithm works in three stages:

1. **Parameterized global search** –  a handful of simple families
   (exponential, power‑law, triangular and sinusoidal shapes) are
   generated for several sequence lengths.  Each candidate is scaled
   to have the required ℓ1‑norm `sqrt(2·n)`.  Its objective
   `2·n·max(conv)/sum^2` (which equals the maximal convolution value
   because of the normalisation) is evaluated quickly with an FFT‑based
   convolution.

2. **Random Dirichlet restarts** –  a number of random Dirichlet
   vectors are drawn, sorted in decreasing order (which minimises the
   central product) and evaluated.

3. **Projected sub‑gradient refinement** –  starting from the best
   candidate found above a few seconds of projected gradient descent
   are performed.  The gradient of the currently maximal convolution
   entry is `g_i = 2·a_{k_max‑i}`; after a step the vector is projected
   onto the simplex `{x≥0, sum(x)=sqrt(2·n)}`.  Occasionally a random
   mass transfer is applied to escape plateaus.

All work is bounded by the provided time budget (default 1000 s).  The
resulting sequence is returned as a plain list of floats and is
compatible with the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def proj_simplex(v, s):
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, sum(x) = s }.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; fall back to uniform
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a):
    """
    Full convolution a*a using an FFT of the next power of two.
    Returns a real ndarray of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    conv.real.clip(min=0, out=conv.real)   # eliminate tiny negatives
    return conv.real


def max_conv_value(a):
    """Fast evaluation of max(conv(a,a))."""
    return np.max(conv_full_fft(a))


def objective_value(a):
    """
    Exact objective used by `evaluate_sequence`.
    Works for any non‑negative vector a.
    """
    n = a.shape[0]
    sum_a = np.sum(a)
    if sum_a < 1e-12:        # safeguard – evaluate_sequence would return inf
        return float("inf")
    max_b = max_conv_value(a)
    return float(2.0 * n * max_b / (sum_a ** 2))


def gradient_step(a, k_max, step, target_sum):
    """
    One projected sub‑gradient step aimed at reducing the entry
    `conv[a,a][k_max]`.  The gradient of that entry is
    g_i = 2 * a_{k_max-i} whenever the index is valid.
    The step is `a - step * g` followed by projection onto the simplex.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] = 2.0 * a[k_max - idx]
    new = a - step * g
    new = proj_simplex(new, target_sum)
    return new


def generate_exponential(n, r):
    """a_i ∝ r**i  (0<r<1)  and scaled to the correct ℓ1‑norm."""
    v = r ** np.arange(n, dtype=float)
    v = v / v.sum() * np.sqrt(2.0 * n)
    return v


def generate_powerlaw(n, alpha):
    """a_i ∝ (i+1)^{-alpha}  (alpha>0)  scaled to the ℓ1‑norm."""
    i = np.arange(1, n + 1, dtype=float)
    v = i ** (-alpha)
    v = v / v.sum() * np.sqrt(2.0 * n)
    return v


def generate_power_exp(n, alpha, r):
    """a_i ∝ i^{-alpha}·r**i  with two tunable parameters."""
    i = np.arange(1, n + 1, dtype=float)
    v = (i ** (-alpha)) * (r ** (i - 1))
    v = v / v.sum() * np.sqrt(2.0 * n)
    return v


def generate_triangular(n, m):
    """
    Linear decreasing sequence of length `m` (values m, m‑1, …, 1)
    padded with zeros to length n, then scaled.
    """
    if m > n:
        m = n
    vals = np.arange(m, 0, -1, dtype=float)
    if m < n:
        vals = np.pad(vals, (0, n - m), mode="constant")
    vals = vals / vals.sum() * np.sqrt(2.0 * n)
    return vals


def generate_sine(n):
    """a_i = sin(π·(i+0.5)/n), all positive, then scaled."""
    i = np.arange(n, dtype=float) + 0.5
    v = np.sin(np.pi * i / n)
    v = v / v.sum() * np.sqrt(2.0 * n)
    return v


def generate_dirichlet_sorted(n):
    """Random Dirichlet (exponential) draw, sorted decreasing."""
    expo = np.random.exponential(size=n)
    v = expo / expo.sum()
    v = np.sort(v)[::-1]                 # opposite order minimises central product
    v = v * np.sqrt(2.0 * n)
    return v


def local_refinement(start_seq, time_limit):
    """
    Projected sub‑gradient descent starting from `start_seq`
    for at most `time_limit` seconds.
    Returns the best sequence found and its objective value.
    """
    n = start_seq.shape[0]
    S = np.sqrt(2.0 * n)
    a = proj_simplex(start_seq.copy(), S)   # ensure feasibility
    best_seq = a.copy()
    best_val = objective_value(a)

    # parameters for the descent loop
    step = S * 0.2          # initial step magnitude
    decay = 0.9995
    min_step = S * 1e-6

    start = time.time()
    while time.time() - start < time_limit:
        conv = conv_full_fft(a)
        k_max = int(np.argmax(conv))

        # try a few scaled step sizes; accept the first improvement
        improved = False
        for scale in (1.0, 0.5, 0.2):
            candidate = gradient_step(a, k_max, step * scale, S)
            cand_val = objective_value(candidate)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = candidate.copy()
            if cand_val < objective_value(a):
                a = candidate
                improved = True
                break

        if not improved:
            # occasional random mass transfer to escape flat regions
            i, j = np.random.randint(n, size=2)
            delta = np.random.rand() * step
            delta = min(delta, a[i])
            a[i] -= delta
            a[j] += delta
            a = proj_simplex(a, S)

        step = max(step * decay, min_step)

    return best_seq, best_val
