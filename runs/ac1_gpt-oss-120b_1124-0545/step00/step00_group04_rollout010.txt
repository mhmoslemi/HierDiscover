    # parameters

    move_prob = 0.7

    while time.time() < deadline:

        k_max = int(np.argmax(conv))
        # compute products for indices contributing to k_max
        # products = a * a_rev where a_rev[i] = a[k_max - i] if valid
        # compute vector p
        p = np.zeros(n)

            j = k_max - i

            # find max product index
            idx_max = int(np.argmax(p))
            # choose src as larger of idx_max and k_max-idx_max
            tail = k_max - idx_max
            src = idx_max if a[idx_max] >= a[tail] else tail
            # propose move
            step = 0.05 * a[src] * np.random.rand()
            if step == 0:
                continue
            dest = np.random.randint(n-1)
            if dest >= src:
                dest += 1
            a_new = a.copy()
            a_new[src] -= step
            a_new[dest] += step
            if a_new[src] < 0:
                continue
            # evaluate
            score = evaluate_sequence(a_new.tolist())
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                a = a_new
            else:
                # occasional acceptance
                if np.random.rand() < 0.001:
                    a = a_new
        # optionally apply random perturb tiny
        if best_score < 1.51:
            break
    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that lowers the evaluation
function  eval = 2·n·max(conv(a,a)) / (sum(a))².

The algorithm is a cheap stochastic local search:
* start from several random initial vectors (sum normalised to 1);
* repeatedly identify the convolution index where the current maximum
  occurs, pick the pair of entries that contributes most to that
  maximum and move a small amount of mass from the larger end of the
  pair to a random other position;
* keep the best vector seen.
The search is performed entirely with NumPy (no external LP solver) and
stays well within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for non‑negative vectors.
    Returns 2·n·max(conv(a,a)) / (sum(a)²).
    """
    s = a.sum()
    if s < 0.01:                     # protect against near zero sum
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * a.size * b.max() / (s * s))


def _targeted_move(a: np.ndarray, max_idx: int, step_ratio: float) -> np.ndarray:
    """
    Produce a new vector after moving a small amount of mass away from the
    entry that participates most in the current maximum convolution entry.
    """
    n = a.size
    # product a[i] * a[max_idx-i] for all feasible i
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    products = np.zeros_like(a)
    products[mask] = a[mask] * a[j[mask]]

    i_max = int(products.argmax())
    j_max = max_idx - i_max

    # choose the larger of the two participants as the source of the move
    src = i_max if a[i_max] >= a[j_max] else j_max
    if a[src] <= 0.0:
        return a  # nothing to move

    # amount to transfer – a random fraction of the source entry
    delta = step_ratio * a[src] * np.random.rand()
    if delta <= 0.0 or delta > a[src]:
        return a

    # pick a destination index different from src
    dest = np.random.randint(n - 1)
    if dest >= src:
        dest += 1

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dest] += delta
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """Search for a good height sequence within the allotted time."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # length of the sequence – a moderate size keeps each iteration cheap
    n = 800

    # keep the overall best vector found
    best_vec = np.full(n, 1.0 / n)   # start from the uniform vector
    best_val = _score(best_vec)

    # parameters for the stochastic descent
    step_ratio = 0.07               # how much mass is moved each step
    max_iters_per_start = 4000      # limit per random restart
    restart_counter = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) create a fresh start (uniform or random Dirichlet)
        if restart_counter % 2 == 0:
            cur = np.full(n, 1.0 / n)               # uniform start
        else:
            cur = np.random.rand(n)
            cur = np.maximum(0.0, cur)
            cur_sum = cur.sum()
            if cur_sum == 0:
                cur = np.full(n, 1.0 / n)
            else:
                cur /= cur_sum                        # normalise to sum 1

        cur_val = _score(cur)
        # ------------------------------------------------------------------
        # 2) local stochastic improvement loop
        it = 0
        while it < max_iters_per_start and time.time() < deadline:
            # compute the convolution and locate its maximum
            conv = np.convolve(cur, cur)
            k_max = int(conv.argmax())

            # propose a targeted move that reduces the dominant term
            cand = _targeted_move(cur, k_max, step_ratio)

            # small random perturbation every few iterations to escape plateaus
            if it % 1000 == 0 and np.random.rand() < 0.15:
                cand = np.maximum(0.0, cand + np.random.randn(n) * 0.002)
                cand_sum = cand.sum()
                if cand_sum > 0:
                    cand /= cand_sum

            cand_val = _score(cand)

            # accept if we improve, otherwise keep the old vector
            if cand_val < best_val:
                best_val = cand_val
                best_vec = cand.copy()
                cur = cand
                cur_val = cand_val
            elif cand_val <= cur_val:
                # non‑worsening move – still adopt to keep the walk moving
                cur = cand
                cur_val = cand_val
            else:
                # occasional uphill step (simulated‑annealing style)
                prob = np.exp(-(cand_val - cur_val) / max(0.001, 0.5 - it / max_iters_per_start))
                if np.random.rand() < prob:
                    cur = cand
                    cur_val = cand_val

            # gradually shrink the step size to allow fine tuning
            step_ratio = max(0.005, step_ratio * 0.9999)

            it += 1
        # ------------------------------------------------------------------
        restart_counter += 1

        # early exit if we already beat the target bound
        if best_val < 1.51:          # target is 1.503, give a tiny margin
            break

    # final safety: if nothing better than uniform was found, return uniform
    if best_vec is None or best_val == np.inf:
        best_vec = np.full(n, 1.0 / n)

    # return a plain Python list as required
    return best_vec.tolist()
