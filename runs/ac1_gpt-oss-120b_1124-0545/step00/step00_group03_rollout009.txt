def project_onto_simplex(v):
    # ensure nonnegative and sum to 1
    v = np.maximum(v, 0)
    if np.sum(v) == 0:
        v = np.ones_like(v) / len(v)

def _run_gradient_descent(a, deadline, rng):
    n = a.shape[0]
    best_local_seq = a.copy()
    best_local_score = 2.0 * n * np.max(np.convolve(a, a))

    max_iter = 3000
    init_lr = 0.5
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute score
        score = 2.0 * n * max_val
        if score < best_local_score:
            best_local_score = score
            best_local_seq = a.copy()
        # compute subgradient
        mask = conv == max_val
        w = np.zeros_like(conv)
        w[mask] = 1.0 / mask.sum()

        # use sliding window view for gradient
        # windows shape (n, n)
        windows = np.lib.stride_tricks.sliding_window_view(w, n)
        grad = 2.0 * windows @ a  # shape (n,)

        lr = init_lr * (1 - it / max_iter)
        a = a - lr * grad

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            # random restart
            a = rng.random(n)
            a /= a.sum()
        else:
            a = a / s

        # occasional random jitter to escape flat region
        if it % 200 == 0 and rng.random() < 0.3:
            # perturb a little
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
    return best_local_seq, best_local_score

"""
propose_candidate
-----------------

A fast projected sub‑gradient descent that directly minimizes the
max‑convolution value of a non‑negative step‑function under the simplex
constraint (non‑negative entries, sum = 1).  

The evaluation function (after normalising the sum to 1) simplifies to

    value = 2 * n * max(conv(a, a))

so we minimise the peak of `conv(a, a)`.  For a convex combination of a few
random starts we repeatedly

1. compute the full self‑convolution,
2. identify the active shift(s) attaining the maximum,
3. build a sub‑gradient of the maximum (the active shifts receive equal
   weight), and
4. take a projected sub‑gradient step onto the probability simplex.

The algorithm is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and respects the 1000 s budget.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _run_subgradient_descent(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on a single start vector ``a``.
    Returns the best vector found and its (exact) score ``2 * n * max(conv)``.
    """
    n = a.shape[0]
    best_vec = a.copy()
    # exact score (no need to call evaluate_sequence while iterating)
    best_score = 2.0 * n * np.max(np.convolve(a, a))

    max_iter = 3000          # enough for a decent descent, still cheap
    init_lr = 0.5            # initial learning rate

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = np.convolve(a, a)               # full self‑convolution
        peak = conv.max()
        score = 2.0 * n * peak
        if score < best_score:
            best_score = score
            best_vec = a.copy()

        # -------------------- sub‑gradient of the max --------------------
        active = conv == peak                   # mask of active indices
        w = np.zeros_like(conv, dtype=float)
        w[active] = 1.0 / np.count_nonzero(active)   # uniform weight

        # gradient = 2 * sum_k w_k * a_{k-i}
        # using sliding windows: each row i contains w[i:i+n]
        windows = np.lib.stride_tricks.sliding_window_view(w, n)  # (n, n)
        grad = 2.0 * windows @ a          # shape (n,)

        # -------------------- update & projection --------------------
        lr = init_lr * (1.0 - it / max_iter)   # linear decay
        a = a - lr * grad

        # project back onto the simplex
        a = _project_to_simplex(a)

        # occasional jitter to escape flat plateaux
        if (it + 1) % 200 == 0 and rng.random() < 0.3:
            a = _project_to_simplex(a + rng.normal(scale=0.01, size=n))

    return best_vec, best_score

    """

    deadline = time.time() + budget_s - 2.0   # safety margin

    best_sequence = None
    best_score = float("inf")
