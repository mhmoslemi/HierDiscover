"""
Propose a candidate sequence of non‑negative heights that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a very fast projected‑subgradient / simulated‑annealing
search:

* start from a random Dirichlet vector (sum = 1)
* repeatedly
    - compute the full self‑convolution using an FFT (O(n log n))
    - locate the index k_max where the convolution is maximal
    - locate the index k_min where it is minimal
    - the contribution of a_i to the maximal entry is a_i * a_{k_max-i}
      (zero if the second index is out of range).  The same holds for
      k_min.
    - move a small amount of mass from the index that contributes most
      to k_max towards an index that contributes the least to k_min.
    - accept the move if the objective improves; otherwise reject it.
    - occasionally perform a random “kick” to escape plateaus.
* keep the best sequence seen so far and return it when the time budget
  (default 1000 s) expires.

All heavy work is done with NumPy/FFT, so thousands of iterations fit
comfortably into the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a * a using a real FFT.
    The result length is 2*len(a)-1.
    """
    n = a.shape[0]
    m = _next_pow_two(2 * n - 1)
    # rfft works with real input and returns the positive frequencies only
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    # Numerical noise can produce tiny negatives; clip to zero
    np.maximum(conv, 0.0, out=conv)
    return conv


def evaluation_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a non‑negative vector a.
    This mimics evaluate_sequence but avoids the security‐clamping.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = convolution_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of heights) with a low evaluation score.
    The function runs for ``budget_s`` seconds (minus a safety margin) and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Parameters (feel free to tune)
    # ------------------------------------------------------------------
    n = 3000                     # length of the sequence – larger gives a lower
                                # theoretical bound (≈1) while still fast.
    max_iters = 10_000_000       # hard limit – the loop also stops on timeout
    delta_factor = 0.12          # fraction of the source entry moved each step
    min_delta = 1e-5             # absolute lower bound for a move
    kick_interval = 200          # perform a random kick every so many iterations
    restart_interval = 250.0     # seconds after which we restart from a fresh random start

    # ------------------------------------------------------------------
    # Initialise a random feasible sequence (Dirichlet -> sum = 1)
    # ------------------------------------------------------------------
    a = np.random.rand(n).astype(np.float64)
    a /= a.sum()                 # enforce sum = 1 (scale does not matter)
    cur_score = evaluation_score(a)

    best_seq = a.copy()
    best_score = cur_score

    it = 0
    last_restart = start_time
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        # 1) compute the self‑convolution and locate extrema
        conv = convolution_fft(a)
        k_max = int(np.argmax(conv))
        k_min = int(np.argmin(conv))

        # 2) compute contributions to the maximal and minimal entries
        idx = np.arange(n)
        j_max = k_max - idx
        mask_max = (j_max >= 0) & (j_max < n)
        contrib_max = np.zeros_like(a)
        contrib_max[mask_max] = a[idx[mask_max]] * a[j_max[mask_max]]

        j_min = k_min - idx
        mask_min = (j_min >= 0) & (j_min < n)
        contrib_min = np.zeros_like(a)
        contrib_min[mask_min] = a[idx[mask_min]] * a[j_min[mask_min]]

        # 3) choose donor (large contribution to max) and receiver (tiny contribution to min)
        donor = int(np.argmax(contrib_max))                     # i with highest impact on max
        # For receiver we prefer an index with *very* small impact on the minimal entry.
        # If many indices qualify we pick one at random to add stochasticity.
        min_val = contrib_min.min()
        candidates = np.where(contrib_min == min_val)[0]
        receiver = int(np.random.choice(candidates))

        # 4) propose a move of a small chunk from donor to receiver
        delta = min(a[donor] * delta_factor, min_delta if a[donor] * delta_factor < min_delta else a[donor] * delta_factor)
        if delta <= 0.0:
            # fallback – pick a random donor with positive mass
            donor = int(np.random.choice(np.where(a > 0)[0]))
            delta = min(a[donor] * delta_factor, min_delta)

        a[donor] -= delta
        a[receiver] += delta

        # 5) evaluate the new candidate
        new_score = evaluation_score(a)

        # 6) accept if improvement (otherwise revert)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta

        # ------------------------------------------------------------------
        # occasional random “kick” to escape shallow plateaus
        # ------------------------------------------------------------------
        if it % kick_interval == 0 and it > 0:
            i, j = np.random.choice(n, size=2, replace=False)
            # move a tiny random amount (up to 5 % of the source mass)
            d = min(a[i] * 0.05, min_delta)
            if d > 0:
                a[i] -= d
                a[j] += d
                kick_score = evaluation_score(a)
                if kick_score < cur_score:
                    cur_score = kick_score
                    if kick_score < best_score:
                        best_score = kick_score
                        best_seq = a.copy()
                else:
                    a[i] += d
                    a[j] -= d

        # ------------------------------------------------------------------
        # occasional full restart if we have been stuck for a while
        # ------------------------------------------------------------------
        now = time.time()
        if now - last_restart > restart_interval:
            # keep the best we have found, but re‑initialise a fresh start
            a = np.random.rand(n).astype(np.float64)
            a /= a.sum()
            cur_score = evaluation_score(a)
            last_restart = now
            # (the best sequence is retained across restarts)

        it += 1

        # optional progress print (once per few thousand iterations)
        if it % 5000 == 0:
            elapsed = now - start_time
            print(f"Iter {it:,} | time {elapsed:.1f}s | cur {cur_score:.6f} | best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"Finished after {it:,} iterations. Best score = {best_score:.6f}")
    return best_seq.tolist()
