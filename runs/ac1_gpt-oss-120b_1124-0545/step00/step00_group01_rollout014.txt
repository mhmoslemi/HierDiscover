def step(seq, t):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    grad = np.zeros_like(seq)
    lo = max(0, k - (n -1))

    for i in range(lo, hi+1):
        j = k - i
        grad[i] = seq[j]
    # projection to sum-zero
    grad = grad - np.mean(grad)
    # step
    new = seq - t * grad
    # clip
    new = np.maximum(new, 0)
    # renormalize to sum=1
    new = new / np.sum(new)
    return new

"""
Propose a sequence that minimizes the evaluation function.

The idea is based on the continuous model a(x) = 1/√x, whose
self‑convolution is (almost) constant.  Discretising this gives a
sequence a_i ∝ 1/√(i+1) which is known to achieve an evaluation close
to the theoretical lower bound (≈π/2 ≈ 1.57).  Starting from this
sequence we perform a simple projected gradient descent that tries to
flatten the actual discrete convolution further.  The algorithm runs
until the allotted time expires and returns the best sequence found.
"""
import time
import numpy as np

# evaluate_sequence is assumed to be already imported in the execution environment


def _initial_root_sequence(n: int) -> np.ndarray:
    """Return a normalised sequence a_i ∝ 1/√(i+1) of length n."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _projected_gradient_step(seq: np.ndarray, step_size: float) -> np.ndarray:
    """
    One gradient step that tries to reduce the maximum entry of the
    linear convolution `c = seq * seq`.

    The gradient of the max entry `c[k_max]` w.r.t. `seq[i]` is
    `seq[k_max-i]` (if the index is inside the support).  We move in the
    direction opposite to this gradient, project onto the affine subspace
    `sum(seq)=1` and clip to non‑negative values.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # gradient of the active convolution entry
    grad = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    # i runs over indices that actually contribute to c[k_max]
    indices = np.arange(lo, hi + 1, dtype=int)
    grad[indices] = seq[k_max - indices]

    # make the step tangent to the hyperplane sum(seq)=1
    grad -= grad.mean()

    # take a step and keep non‑negativity
    new_seq = seq - step_size * grad
    new_seq = np.maximum(new_seq, 0.0)

    # renormalise to keep sum exactly 1 (scale invariance of the score)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:  # fallback – should never happen
        new_seq = _initial_root_sequence(n)

    return new_seq


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a high‑quality sequence within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds.
    kwargs : dict
        Ignored – kept for compatibility with the testing harness.

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, sum>0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ---- 1. Initialise -------------------------------------------------
    # a moderately large length gives a good approximation of the
    # continuous optimum while still being cheap to evaluate.
    n = 5000
    seq = _initial_root_sequence(n)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # ---- 2. Simple projected gradient descent -------------------------
    # Step size schedule – start relatively aggressive and decay.
    step = 0.2
    decay = 0.99995

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # gradient step
        seq = _projected_gradient_step(seq, step)

        # occasional random swap to escape flat regions
        if iteration % 50 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = step * seq[i]  # a small amount of mass
            eps = min(eps, seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            seq = cand

        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # update step size
        step = max(step * decay, 1e-6)

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()
