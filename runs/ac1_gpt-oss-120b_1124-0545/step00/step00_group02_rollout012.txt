"""
Propose a candidate sequence for the convolution bound problem.

The algorithm searches for a family of power‑law sequences
    a_i = (i+1)^p   (i = 0 … n‑1)

which are known analytically to give a much smaller
max‑convolution than the uniform sequence (the current bound 2.0).
For a fixed length `n` the evaluation value depends only on the
exponent `p`; by scanning `p` we can quickly locate a region where
the score drops below the target 1.503.  After a coarse random search
the algorithm performs a finer local refinement around the best
parameters.  All values are scaled (if necessary) so that the maximum
element is ≤ 1000, which avoids the clipping step inside `evaluate_sequence`.
The evaluation is scale‑invariant, so this scaling does not affect the
objective.

The routine respects the 1000 s time budget, returns the best sequence
found and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def generate_powerlaw_sequence(length: int, exponent: float) -> list[float]:
    """
    Returns a non‑negative sequence a_i = (i+1)^exponent of the given length.
    The whole sequence is scaled so that the maximum entry does not exceed 1000.
    (Scaling does not change the value of evaluate_sequence because it is
    homogeneous of degree zero.)
    """
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = np.power(idx, exponent)

    # scale to respect the 0‑1000 clipping performed inside evaluate_sequence
    max_val = seq.max()
    if max_val > 1000.0:
        seq = seq * (1000.0 / max_val)

    # convert to plain Python list for evaluate_sequence
    return seq.tolist()


def evaluate(seq: list[float]) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq)

# Main search routine
# ----------------------------------------------------------------------

    """
    Search for a step‑function (non‑negative height list) that minimizes
    evaluate_sequence.  The search consists of three stages:

    1. Initialise with a uniform baseline (taken from the global
       `height_sequence_1` if it exists) and keep the best score.
    2. Randomly explore power‑law families a_i = (i+1)^p for several
       lengths n and exponents p.  The best candidate is kept.
    3. Refine the best exponent/length by a small local grid around it and

    # ------------------------------------------------------------------

        # jitter relative to the current scale (10 % of the max value)
        scale = max(cur) * 0.1
        cur[idxs] += np.random.randn(k) * scale
        cur = np.clip(cur, 0.0, None)                # enforce non‑negativity

        # keep values ≤1000 (scale uniformly if needed)
        if cur.max() > 1000.0:
            cur = cur * (1000.0 / cur.max())

        seq = cur.tolist()
        sc = evaluate(seq)

        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"[jitter] improved to {sc:.6f}")

        # occasionally re‑scale to keep the sum comfortably away from zero
        if np.sum(best_seq) < 0.1:
            best_seq = (np.array(best_seq) + 1.0).tolist()
            best_score = evaluate(best_seq)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq
