import time

from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Heuristic optimiser for the step‑function bound.
#
#  The algorithm keeps a non‑negative sequence f and its current
#  maximal self‑convolution M = max(f*f).  A linear programme finds a
#  partner g with a larger ℓ₁‑sum while guaranteeing conv(f,g) ≤ M.
#  The largest feasible convex combination (1‑λ)·f + λ·g that does not
#  increase M is then taken (binary search on λ).  Repeating this never
#  raises the bound M but raises the total sum, thereby decreasing the
#  evaluation 2·n·M / sum(f)².  Occasionally we fall back to small random
#  perturbations and perform a short hill‑climb at the end.
# ----------------------------------------------------------------------


def solve_convolution_lp(f, M):
    """
    Linear programme:
        maximise    Σ b_j
        subject to  Σ_i f_i * b_{k-i} ≤ M   for all k = 0 … 2n‑2
                     b_j ≥ 0
    The constraints are linear in b because f is fixed.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f)
    # Build A_ub for the convolution constraints.
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        rhs.append(M)
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    # Objective: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use the HiGHS solver; a short time limit keeps the overall run fast.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 0.5})
        if res.success:
            return res.x
    except Exception:      # pragma: no cover
        pass
    return None

    Find the largest λ∈[0,1] such that
        h = (1‑λ)·f + λ·g
        max(conv(h,h)) ≤ M

    lo, hi = 0.0, 1.0
    for _ in range(steps):
        lam = (lo + hi) * 0.5
        h = (1.0 - lam) * f + lam * g
        if np.max(np.convolve(h, h)) <= M + 1e-12:
            lo = lam
        else:
            hi = lam
    return lo

    else:
        a /= total
