def simplex_projection(v, s):
    # v is 1D np array, s > 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0

def project_onto_simplex(v, s):
    # v: np.array
    # s: target sum > 0
    # returns w: np.array, sum(w)=s, w>=0
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all entries zero? then return uniform
        return np.full_like(v, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimization of the step‑function heights.

The objective is to minimise

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

with a_i ≥ 0.  The function is scale‑invariant, therefore we keep the
ℓ₁‑norm fixed to sqrt(2*n) (the usual normalisation) and minimise the
maximum entry of the self‑convolution.  The algorithm combines:

* a simple simulated‑annealing / random walk on the simplex,
* a sub‑gradient step that directly reduces the currently largest
  convolution entry,
* a projection onto the probability simplex to keep the sum fixed.

Multiple random restarts are performed until the time budget is exhausted
and the best sequence found is returned.
"""

import time
import numpy as np
import random
import math

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x ≥ 0, sum(x)=s}
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w >= 0 , sum(w) = s }."""
    # based on the algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find rho = max { j : u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero (should not happen for s>0)
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the given
    evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    **kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best coefficient sequence found within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the problem size.  If a previous construction is present
    # we reuse its length, otherwise we fall back to a modest size.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1500                     # a safe default when no prior data

    target_sum = math.sqrt(2.0 * n)  # the normalising ℓ₁‑norm

    # ------------------------------------------------------------------
    # Initial candidates
    uniform = np.full(n, target_sum / n)
    best_seq = uniform.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Occasionally keep a random start as well
    random_seq = np.random.exponential(scale=1.0, size=n)
    random_seq = random_seq / random_seq.sum() * target_sum
    rand_score = evaluate_sequence(random_seq.tolist())
    if rand_score < best_score:
        best_seq = random_seq.copy()
        best_score = rand_score

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a little safety margin
    iteration = 0

    # Temperature schedule for simulated‑annealing style acceptance
    T_start = 1.0
    T_end = 1e-4

    # Main optimisation loop
    while time.time() < deadline:
        # Linear cooling for the temperature
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)

        # --------------------------------------------------------------
        # 1) Propose a small random move on the simplex.
        #    Transfer a random amount from i -> j.
        proposal = best_seq.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            # Amount to move – scale with temperature to shrink later
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.02 * T)
            delta = max(-proposal[i], min(delta, proposal[i]))  # keep non‑negative
            proposal[i] -= delta
            proposal[j] += delta

        # Re‑project onto the simplex (keeps sum exactly target_sum)
        proposal = _simplex_projection(proposal, target_sum)

        # --------------------------------------------------------------
        # 2) Sub‑gradient step aimed at the currently largest convolution entry.
        conv = np.convolve(proposal, proposal)
        k_max = int(np.argmax(conv))

        # Gradient of the entry conv[k_max] w.r.t. proposal is
        #   g_i = proposal[k_max - i]   (zero when the index is out of range)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        gradient = np.zeros_like(proposal)
        gradient[mask] = proposal[partner[mask]]

        # Step size – also decays with the temperature
        eta = target_sum * 0.1 * T
        proposal -= eta * gradient
        proposal = _simplex_projection(proposal, target_sum)

        # --------------------------------------------------------------
        # 3) Evaluate and possibly accept.
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            # Any unexpected numerical issue – simply skip this candidate
            continue

        # Accept if improvement; otherwise accept with a small probability
        # (simulated‑annealing).
        if score < best_score or np.random.rand() < math.exp(-(score - best_score) / (T + 1e-12)):
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                print(f"[{iteration:6d}] New best score: {best_score:.6f} (time {elapsed:.1f}s)")
            else:
                # occasional uphill move to escape plateaus
                best_seq = proposal.copy()
        iteration += 1

    # Final sanity check – make sure we return a plain Python list.
    return best_seq.tolist()
