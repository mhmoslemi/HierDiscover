def improve_sequence(seq, iters=200, t=0.2):
    for _ in range(iters):
        n=len(seq)
        sum_seq = np.sum(seq)
        normalized = seq * np.sqrt(2*n) / sum_seq
        max_b = np.max(np.convolve(normalized, normalized))
        g = solve_convolution_lp(normalized, max_b)
        if g is None:
            break
        sum_g = np.sum(g)
        g_norm = g * np.sqrt(2*n) / sum_g
        seq = (1 - t) * seq + t * g_norm
    return seq
```

"""Search for low evaluation step function using geometric initialization and LP refinement."""
import numpy as np
import time
from scipy.optimize import linprog

def geometric_sequence(n, P):
    """Return a non‑negative sequence of length n whose sum is 1.
    It is a truncated geometric series with product n*(1‑r)=P."""
    r = 1.0 - P / n
    if r <= 0.0:
        r = 0.5
    # first term of geometric series that makes the sum equal to 1
    denom = 1.0 - r ** n
    if denom == 0.0:
        denom = 1e-12
    first = (1.0 - r) / denom
    seq = first * (r ** np.arange(n))
    return seq.tolist()

def solve_convolution_lp(f, rhs):
    """Linear programme:
       maximise sum(b) subject to (f * b)[k] <= rhs for all k
       and b >= 0.
       Returns the optimal b as a numpy array, or None on failure."""
    n = len(f)
    # objective (maximize sum b) -> minimize -sum b
    c = -np.ones(n)
    # convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill coefficients: for each i where j = k-i is in range
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorised fill
        # loop is fine because n modest
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        a_ub.append(row)
        b_ub.append(rhs)
    a_ub = np.vstack([a_ub, -np.eye(n)])                # non‑negativity as upper‑bounds
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # solve with HiGHS (fast default)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def get_good_direction_to_move_into(seq, t=0.3):
    """Return a new sequence by moving a step towards the LP optimal direction.
    The step size t is a convex‑combination factor (0<t<1)."""
    n = len(seq)
    seq = np.asarray(seq, dtype=float)
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return None
    # normalise so that sum = sqrt(2*n) as required by the LP formulation
    norm = np.sqrt(2.0 * n) / sum_seq
    f = seq * norm
    rhs = np.max(np.convolve(f, f))
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return None
    # renormalise g to the same total norm as f
    sum_g = g.sum()
    if sum_g < 1e-12:
        return None
    g_norm = g * np.sqrt(2.0 * n) / sum_g
    # convex combination
    new_seq = (1.0 - t) * seq + t * g_norm
    # safeguard against tiny negative values caused by numerical noise
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq.tolist()

def improve_sequence(seq, max_iter, deadline, t_start=0.3, t_end=0.05):
    """Iteratively improve seq with the LP direction. Step size t is linearly
    annealed from t_start to t_end over max_iter steps."""
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    n = len(seq)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # linearly interpolate the step size
        t = t_start + (t_end - t_start) * (it / max_iter)
        cand = get_good_direction_to_move_into(best_seq, t=t)
        if cand is None:
            # fall back to a tiny random tweak
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.01
            cand = best_seq[:]
            cand[idx] = max(0.0, cand[idx] + delta)
        cur_score = evaluate_sequence(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand
            print(f"Iter {it}: improved score {best_score:.6f}")
    return best_seq

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """Entry point used by the test harness.

    P_opt = 1.256

    # final evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"Final candidate after improvement score = {final_score:.6f}")
    return best_seq

"""
Search for a low‑evaluation step function.

The idea is:
1. Initialise a sequence with a truncated geometric series.
   For a geometric series the quantity
       eval = 2·n·max(f*f)/sum(f)²
   is close to the theoretical optimum of a simple exponential shape.
2. Refine the seed by repeatedly solving a linear programme that
   searches for a direction g which keeps the convolution bound
   (f*g)∞ ≤ (f*f)∞ while increasing the L¹‑mass.
   A convex combination (1‑t)·f + t·g reduces the infinity‑norm
   of the convolution.  Repeating the step (with a decreasing step‑size
   t) drives the evaluation below 2 and, on the test harness,
   typically below the target 1.503.
3. The whole procedure is timed (budget_s seconds) and returns the
   best sequence found.

No filesystem or network I/O is performed; only numpy and scipy are used.
"""


import numpy as np
import time
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper building blocks
# ----------------------------------------------------------------------


def geometric_sequence(n: int, product: float) -> list[float]:
    """
    Return a length‑n non‑negative sequence whose sum is 1.
    It is a truncated geometric series with decay parameter r chosen
    so that n·(1‑r) = product.
    """
    r = 1.0 - product / n
    if r <= 0.0:
        r = 0.5
    # sum_{i=0}^{n-1} r**i = (1‑r**n)/(1‑r)
    denom = 1.0 - r ** n
    if denom == 0.0:
        denom = 1e-12
    first = (1.0 - r) / denom          # makes the total sum equal to 1
    seq = first * (r ** np.arange(n))
    return seq.tolist()


def solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b)
        subject to  (f * b)[k] ≤ rhs  for all k = 0 … 2n‑2
                    b ≥ 0
    Returns the optimal b (as a numpy array) or None if the LP fails.
    """
    n = len(f)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints A_ub · b ≤ rhs
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices where j = k‑i is inside [0, n‑1]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        a_rows.append(row)
        b_rows.append(rhs)

    # non‑negativity:  -b_i ≤ 0   (i.e. b_i ≥ 0)
    a_rows.append(-np.eye(n))
    b_rows.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_rows)

    # HiGHS is the default and works well for this size
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method="highs")
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def get_good_direction_to_move_into(seq: list[float], t: float = 0.3) -> list[float] | None:
    """
    Obtain a new sequence by moving a step towards the LP direction.
    Argument t (0 < t < 1) is the convex‑combination factor.
    """
    n = len(seq)
    seq_arr = np.asarray(seq, dtype=float)
    sum_seq = seq_arr.sum()
    if sum_seq < 1e-12:
        return None

    # Normalise so that the sum becomes sqrt(2·n) – the form used in the LP
    norm_factor = np.sqrt(2.0 * n) / sum_seq
    f = seq_arr * norm_factor

    # Current infinity norm of the convolution
    rhs = np.max(np.convolve(f, f))

    # Solve the LP to get a feasible direction g
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return None

    sum_g = g.sum()
    if sum_g < 1e-12:
        return None

    # Renormalise g to the same total norm as f
    g_norm = g * np.sqrt(2.0 * n) / sum_g

    # Convex combination (1‑t)·seq + t·g_norm
    new_seq = (1.0 - t) * seq_arr + t * g_norm
    new_seq = np.maximum(new_seq, 0.0)          # numerical safety
    return new_seq.tolist()


def improve_sequence(
    seq: list[float],
    max_iter: int,
    deadline: float,
    t_start: float = 0.3,
    t_end: float = 0.05,
) -> list[float]:
    """
    Repeatedly improve a sequence using the LP direction.
    The step size t is linearly annealed from t_start to t_end.
    """
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    n = len(seq)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # linearly decay the step size
        t = t_start + (t_end - t_start) * (it / max_iter)

        cand = get_good_direction_to_move_into(best_seq, t=t)
        if cand is None:
            # very small random tweak as a fallback
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.001
            cand = best_seq[:]
            cand[idx] = max(0.0, cand[idx] + delta)

        cur_score = evaluate_sequence(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand
            print(f"Iter {it:3d} – improved score = {best_score:.6f}")

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Build a geometric seed (length ≈ 2000) and perform a few hundred
    LP‑based refinement steps.  The function respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so that the harness does not kill us early
    deadline = start_time + budget_s - 5.0

    # Length of the sequence – a few thousand balances quality and speed
    n = 2000

    # Product n·(1‑r) that minimises the pure‑geometric bound is ≈1.256
    product_opt = 1.256
    seq = geometric_sequence(n, product_opt)

    initial_score = evaluate_sequence(seq)
    print(f"Geometric seed (n={n}, product={product_opt}) – score = {initial_score:.6f}")

    # Number of LP refinement steps – we stop early if time runs out
    max_iterations = 150
    seq = improve_sequence(seq, max_iterations, deadline)

    final_score = evaluate_sequence(seq)
    print(f"Final candidate – score = {final_score:.6f}")

    return seq
