"""
Propose a sequence of non‑negative heights that minimises the evaluation function.

Algorithm
---------
1. Choose a length `n`.  If the global variable ``height_sequence_1`` exists,
   its length is used, otherwise the default length (2584) is taken.
2. Initialise the search with a few simple candidates:
   * uniform vector (all entries equal)
   * a geometric (exponential) decreasing vector  a_i ∝ r**i .
     The exponent `r` is swept in a fine grid and the best one is kept.
   * a random Dirichlet sample.
3. The best of the above candidates becomes the current solution.
4. A fast local‑search (simulated‑annealing style) is then performed until
   the allotted time expires:
   * The convolution `c = a * a` and the index `k_max` where `c` attains its
     maximum are computed.
   * For this index a contribution for each position `i` is approximated by
     `contrib[i] = a[i] * a[k_max - i]` (zero when the partner index is out of
     range).  The positions with largest contributions are the main sources of
     the peak.
   * One source position is chosen from the top‑10 % of contributions and a
     target position from the bottom‑10 % (both randomly).  A small amount
     `ε` of mass is moved from source to target, the vector is re‑normalised
     (the evaluation function is scale‑invariant) and the new score is
     evaluated.
   * The move is accepted if it improves the score; otherwise it may be
     accepted with a tiny probability to escape shallow local minima.
   * Every 5 000 unsuccessful moves a random restart (new Dirichlet vector)
     is performed.
5. The best sequence seen during the whole run is returned.

The routine is completely self‑contained, uses only NumPy, SciPy and the
provided ``evaluate_sequence`` for the final verification, and respects the
time budget supplied via ``budget_s``.
"""

import numpy as np
import time
from scipy import optimize

linprog = optimize.linprog


def _fast_score(seq: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` but without the safety checks."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _make_geometric(n: int, r: float) -> np.ndarray:
    """Geometric decreasing vector a_i ∝ r**i, normalised to sum 1."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _initial_candidates(n: int) -> list[np.ndarray]:
    """Return a small list of diverse starting vectors."""
    cand = []

    # 1) uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand.append(uniform)

    # 2) geometric sweep
    # sweep r in a region that is known to give scores below 2
    r_vals = np.linspace(0.90, 0.9999, 200)
    best_geo = None
    best_score = float("inf")
    for r in r_vals:
        a = _make_geometric(n, r)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_geo = a
    if best_geo is not None:
        cand.append(best_geo)

    # 3) random Dirichlet (alpha = 1)
    rng = np.random.default_rng(12345)
    a = rng.dirichlet(np.ones(n))
    cand.append(a)

    # 4) random Dirichlet (alpha = 0.5) – sparser vectors
    a = rng.dirichlet(0.5 * np.ones(n))
    cand.append(a)

    # 5) if a previous solution exists, include it
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        old = np.asarray(global_vars["height_sequence_1"], dtype=np.float64)
        if old.shape[0] != n:
            # trim or pad with zeros to fit the required length
            if old.shape[0] > n:
                old = old[:n]
            else:
                old = np.pad(old, (0, n - old.shape[0]), constant_values=0.0)
        # normalise to unit sum
        if old.sum() > 0:
            old = old / old.sum()
        cand.append(old)

    return cand


def _local_search(start_seq: np.ndarray, deadline: float) -> np.ndarray:
    """Hill‑climbing / simulated‑annealing style local optimisation."""
    n = start_seq.shape[0]
    best_seq = start_seq.copy()
    best_score = _fast_score(best_seq)

    curr_seq = best_seq.copy()
    curr_score = best_score

    # counters for restart logic
    no_improve_iters = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution and position of the current maximum
        conv = np.convolve(curr_seq, curr_seq)
        k_max = int(conv.argmax())

        # contributions of each index to that peak (i + j = k_max)
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(curr_seq)
        contrib[mask] = curr_seq[mask] * curr_seq[partner[mask]]

        # choose a source (high contribution) and a target (low contribution)
        # we work with a small stochastic sample to keep the step cheap
        # source from the top 10 % of contributions
        idx_sorted = np.argsort(contrib)
        top_start = int(0.9 * n)
        bot_end = int(0.1 * n) + 1

        src = idx_sorted[rng.integers(top_start, n)]
        tgt = idx_sorted[rng.integers(0, bot_end)]

        if src == tgt:
            continue

        # move a modest amount of mass from src to tgt
        eps = min(curr_seq[src] * 0.2, 0.01) * rng.random()
        if eps <= 0.0:
            continue
        new_seq = curr_seq.copy()
        new_seq[src] -= eps
        new_seq[tgt] += eps
        # re‑normalise (the objective is homogeneous)
        new_seq /= new_seq.sum()

        new_score = _fast_score(new_seq)

        # accept if improving, otherwise with a tiny probability
        if new_score < curr_score:
            curr_seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                no_improve_iters = 0
                print(f"[{time.time():.2f}] New best score: {best_score:.6f}")
        else:
            # occasional uphill move (simulated annealing)
            if rng.random() < 0.001:
                curr_seq = new_seq
                curr_score = new_score

        # periodic random restart if search stalls
        no_improve_iters += 1
        if no_improve_iters > 5000:
            # fresh random Dirichlet with alpha=0.7 (moderately sparse)
            curr_seq = rng.dirichlet(0.7 * np.ones(n))
            curr_score = _fast_score(curr_seq)
            no_improve_iters = 0

        # very occasional global shuffling (rotate the whole vector)
        if rng.random() < 0.005:
            shift = rng.integers(1, n)
            curr_seq = np.roll(curr_seq, shift)
            curr_score = _fast_score(curr_seq)

    return best_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    **kwargs :
        Optional ``n`` (length of the sequence).  If not supplied,
        the length of ``height_sequence_1`` is used when available,
        otherwise 2584.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size
    # ------------------------------------------------------------------
    n = kwargs.get("n")
    if n is None:
        if "height_sequence_1" in globals():
            n = len(globals()["height_sequence_1"])
        else:
            n = 2584  # default length used in the reference construction

    deadline = time.time() + float(budget_s) - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Generate a few diverse starting points and pick the best one.
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n)
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local optimisation phase (the majority of the time budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        improved_seq = _local_search(best_seq, deadline)
        improved_score = _fast_score(improved_seq)
        if improved_score < best_score:
            best_seq = improved_seq
            best_score = improved_score

    print(f"[{time.time():.2f}] Final best score: {best_score:.6f}")

    # The evaluation function clips values to [0,1000]; our numbers are far
    # from those limits, but we still return a plain Python list.
    return best_seq.tolist()
