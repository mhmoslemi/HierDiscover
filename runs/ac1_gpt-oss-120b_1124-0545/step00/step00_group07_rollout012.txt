def simplex_projection(v):
    # Projects v onto probability simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def simplex_projection(v):
    # Projects onto simplex sum=1, v>=0
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative, project onto uniform?
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list) that minimizes
evaluate_sequence.  The search uses a simple projected gradient descent
on the homogeneous objective

    E(a) = 2 * n * max_k (a * a)[k]   ,   sum(a)=1 , a_i >= 0

because scaling does not change the score.  The gradient of the
“max‑convolution” term is easy to compute: if k* is the index where the
convolution attains its maximum, then

    ∂max/∂a_i = 2 * a_{k*‑i}    (when 0 ≤ k*‑i < n, else 0).

A step of gradient descent is projected onto the probability simplex
(sum = 1, a_i ≥ 0) using the standard O(n log n) projection.
Several random restarts (uniform, geometric, Dirichlet) are performed
within the time budget; the best sequence found is returned.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (non‑negative, unit sum)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become negative – return uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # ------------------------------------------------------------------

    max_inner_iters = 400          # per restart
    step_init = 0.1                # initial step size (on the simplex)
    step_inc = 1.05                # modest increase after a successful step
    step_dec = 0.5                 # reduction after a failure
    min_step = 1e-6                # stop decreasing beyond this

    # ------------------------------------------------------------------
    # Pre‑computed constant for the (near‑optimal) geometric start.
    # The optimal α solves e^α = 1 + 2α, giving α ≈ 1.256.
    alpha_opt = 1.256
    eps_opt = alpha_opt / n
    r_opt = max(0.0, 1.0 - eps_opt)

    # Main search loop (multiple random restarts)
    while time.time() < deadline:
        # ----- choose an initial distribution --------------------------------
        choice = np.random.rand()
        if choice < 0.25:
            # uniform (already known, but serves as a cheap restart)
            a = np.full(n, 1.0 / n)
        elif choice < 0.5:
            # geometric tail (close to the analytic optimum)
            a = np.power(r_opt, np.arange(n, dtype=float))
            a = a / a.sum()
        elif choice < 0.75:
            # Dirichlet with concentration 0.5 (spiky, many tiny values)
            a = np.random.gamma(shape=0.5, scale=1.0, size=n)
            a = a / a.sum()
        else:
            # pure exponential (shape parameter 1)
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum()

        step = step_init
        cur_score = 2.0 * n * np.max(np.convolve(a, a))

        # ----- projected gradient descent ------------------------------------
        for _ in range(max_inner_iters):
            if time.time() >= deadline:
                break

            # convolution and the index of the maximum term
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))
            max_b = conv[k_max]

            # gradient of max_b w.r.t. a (2 * a_{k_max-i} where defined)
            grad = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(k_max, n - 1)
            if i_min <= i_max:
                idx = np.arange(i_min, i_max + 1, dtype=int)
                grad[idx] = 2.0 * a[k_max - idx]

            # ---- backtracking line search on the simplex --------------------
            improved = False
            for _ in range(6):               # at most a few reductions
                a_candidate = _simplex_projection(a - step * grad)
                candidate_score = 2.0 * n * np.max(np.convolve(a_candidate, a_candidate))
                if candidate_score < cur_score:
                    a = a_candidate
                    cur_score = candidate_score
                    step *= step_inc
                    improved = True
                    break
                else:
                    step *= step_dec
                    if step < min_step:
                        break
            if not improved:

                break

        # ----- keep the best candidate ---------------------------------------
        if cur_score < best_score:

            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best score: {best_score:.6f}")

    # Return as a plain Python list (the evaluator expects a list of floats)
    return best_seq.tolist()
