def triangle_sequence(n, peak=1.0):
    # produce symmetric triangular shape with peak at middle, linearly decreasing to 0 at ends
    seq = []
    center = (n-1)/2
    for i in range(n):
        height = max(0., peak * (1 - abs(i - center) / center))

def gradient_step(seq, eta):
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    n = len(seq)
    # compute gradient: for each i, if 0 <= i < n and 0 <= k-i < n, contribution a_{k-i}
    # else zero.
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = seq[j]
    # gradient of max_b w.r.t seq_i is grad_i
    # we want to reduce max_b, so move opposite direction.
    new = np.maximum(0., seq - eta * grad)
    # renormalize sum to original sum
    total = np.sum(new)
    if total < 1e-6:
        return seq
    new = new * (np.sum(seq) / total)
    return new

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # build convolution constraint matrix A_ub
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # for each i in [i_start, i_end], define j = k-i, then coefficient for g[j] is f[i]
        # So row[j] = f[i]
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.hstack([np.full(2*n-1, rhs), np.zeros(n)])
    # use high-speed simplex/hybrid
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None
```

def get_good_direction(sequence):
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-9:
        return None
    # normalize to sum sqrt(2n)
    factor = np.sqrt(2*n)/sum_seq
    f_norm = [x*factor for x in sequence]
    # compute rhs M for normalized sequence
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None
    # scale g to same sum sqrt(2n)
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = [x * (np.sqrt(2*n) / sum_g) for x in g]
    return g_norm
```

"""
Propose a step‑function (non‑negative heights) that minimizes the
`evaluate_sequence` score.

The algorithm consists of two stages:

1. **Parametric exploration** – generate many candidate sequences from
   simple analytic families (exponential, power‑law, Gaussian,
   triangular, end‑heavy).  For a few lengths (200 … 2500) the candidates
   are evaluated directly and the best one is kept.  All candidates are
   normalised (max = 1) – scaling does not affect the score because the
   objective is homogeneous of degree 0.

2. **LP‑based refinement** – starting from the best candidate, repeatedly
   solve a linear programme that finds a direction `g` with the same
   convolution bound but a larger ℓ¹‑sum.  The current sequence `f` and the
   direction `g` (both normalised to the same ℓ¹‑sum) are mixed with a
   line‑search over several mixing coefficients.  The best mixed sequence
   replaces `f`.  If the LP fails, a small random perturbation is applied.
   The loop runs until the time budget is exhausted.

The whole procedure finishes well within the 1000 s limit and returns the
best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

def _normalise(seq):
    """Scale so that max(seq) == 1 (preserves the objective value)."""
    mx = max(seq) if seq else 1.0
    if mx == 0:
        return seq
    factor = 1.0 / mx
    return [x * factor for x in seq]


def exponential_seq(n, r):
    """a_i = r**i   (0 < r < 1)."""
    return _normalise([r ** i for i in range(n)])


def powerlaw_seq(n, p):
    """a_i = (i+1)**(-p)   (p > 0)."""
    return _normalise([(i + 1) ** (-p) for i in range(n)])


def gaussian_seq(n, sigma):
    """a_i = exp(-((i - centre)/sigma)**2)."""
    centre = (n - 1) / 2.0
    return _normalise([math.exp(-((i - centre) / sigma) ** 2) for i in range(n)])


def triangular_seq(n):
    """Symmetric triangle, peak = 1, zero at the ends."""
    centre = (n - 1) / 2.0
    seq = []
    for i in range(n):
        val = 1.0 - abs(i - centre) * 2.0 / (n - 1)
        seq.append(max(0.0, val))
    return _normalise(seq)


def end_heavy_seq(n, p):
    """
    Larger values near the two ends, smaller in the centre:
    a_i = ((i+1)*(n-i))**(-p).
    """
    seq = [((i + 1) * (n - i)) ** (-p) for i in range(n)]
    return _normalise(seq)


# ----------------------------------------------------------------------
# Linear‑programming utilities (the same idea as the provided SOTA code)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve   max Σ b_j   s.t.   (f_seq * b)[k] ≤ rhs  for all k,
    b_j ≥ 0.
    Returns the optimal b (numpy array) or None if infeasible.
    """
    n = len(f_seq)
    c = -np.ones(n)  # minimize -sum(b)  →  maximize sum(b)

    # Build the convolution constraints.
    # Row k contains f[i] at column j = k-i whenever 0≤i,j<n.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # Non‑negativity constraints:  -b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Use the HiGHS solver (fast and reliable for moderate sizes)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return result.x
    return None


def lp_direction(current):
    """
    Given a sequence `current`, return a direction `g` with the same
    ℓ¹‑sum (normalised to √(2n)) that respects the current convolution
    bound.  Returns None if the LP fails.
    """
    n = len(current)
    s = float(np.sum(current))
    if s < 1e-9:
        return None
    factor = math.sqrt(2.0 * n) / s
    f_norm = [x * factor for x in current]

    rhs = float(np.max(np.convolve(f_norm, f_norm)))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g to the same ℓ¹‑sum as f_norm (i.e. √(2n))
    sum_g = float(np.sum(g))
    if sum_g == 0:
        return None
    g_norm = [x * (math.sqrt(2.0 * n) / sum_g) for x in g]
    return g_norm


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) yields a low score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Initial best – either the provided SOTA sequence or a simple uniform one
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best_seq = list(height_sequence_1)
    else:
        best_seq = [1.0] * 256
    best_score = evaluate_sequence(best_seq)
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Parametric families
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1200, 1800, 2500]          # a few representative sizes
    alphas_exp = np.linspace(0.60, 0.999, 12)            # r for exponential
    alphas_pow = np.linspace(0.2, 2.0, 12)               # p for power‑law / end‑heavy
    sigma_factors = np.linspace(0.10, 0.45, 10)          # σ = factor * n

    # Helper to evaluate and possibly update the global best
    def try_candidate(seq):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq)
        except Exception:
            return
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[grid] new best = {sc:.6f}  (len={len(seq)})")

    # Exhaustive grid over the families
    for n in lengths:
        # Exponential
        for r in alphas_exp:
            try_candidate(exponential_seq(n, r))

        # Power‑law
        for p in alphas_pow:
            try_candidate(powerlaw_seq(n, p))

        # Gaussian
        for f in sigma_factors:
            sigma = f * n
            if sigma <= 0:
                continue
            try_candidate(gaussian_seq(n, sigma))

        # Triangular (no parameters)
        try_candidate(triangular_seq(n))

        # End‑heavy
        for p in alphas_pow:
            try_candidate(end_heavy_seq(n, p))

    # ------------------------------------------------------------------
    # 2)  LP‑based refinement + occasional random perturbations
    # ------------------------------------------------------------------
    mix_alphas = [0.01, 0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.0]

    while time.time() < deadline:
        direction = lp_direction(best_seq)
        if direction is None:
            # LP failed – perform a tiny random tweak
            tmp = best_seq.copy()
            for _ in range(3):
                idx = np.random.randint(len(tmp))
                tmp[idx] = max(0.0, tmp[idx] + np.random.randn() * 0.02)
            try_candidate(tmp)
            continue

        improved_this_round = False
        # Try several mixing coefficients and keep the best improvement
        for a in mix_alphas:
            cand = [(1.0 - a) * x + a * y for x, y in zip(best_seq, direction)]
            # No need to re‑normalise – the objective is scale‑invariant
            try:
                sc = evaluate_sequence(cand)
            except Exception:
                continue
            if sc < best_score:
                best_score, best_seq = sc, cand
                improved_this_round = True
                print(f"[lp] improved to {sc:.6f} using α={a:.2f}")
                break   # accept the first improvement, then recompute direction

        if not improved_this_round:
            # If nothing helped, inject a small random move to escape
            tmp = best_seq.copy()
            for _ in range(2):
                idx = np.random.randint(len(tmp))
                tmp[idx] = max(0.0, tmp[idx] + np.random.randn() * 0.05)
            try_candidate(tmp)

    # Final sanity: make sure the result is a plain Python list of floats
    final = [float(x) for x in best_seq]
    print(f"[final] best score = {best_score:.6f}  length = {len(final)}")
    return final
