"""
Optimization of the step‑function bound
======================================

The goal is to minimise

    E(a) = 2·n·max_k (a*a)_k / (Σ a_i)²

where `a` is a non‑negative vector (the heights of the step function) and
`(a*a)` denotes the discrete self‑convolution.  The value of the objective
is invariant under scaling, therefore we fix Σ a_i = 1 and strive to make
the maximum convolution entry as small as possible.

The algorithm combines three simple ideas:

1. **Warm start** – either the historic construction (`height_sequence_1`)
   or a uniform distribution.
2. **Local mass transfer** – choose two indices `i` and `j` and move a
   small amount `δ` from `i` to `j`.  The convolution can be updated in
   O(n) time using the linearity of the convolution, so we avoid the
   expensive `np.convolve` call at every step.
3. **Simulated annealing** – accept a move that worsens the objective
   with probability `exp(-(ΔE)/T)`.  The temperature `T` is slowly
   cooled, which helps the search escape shallow local minima.
   Occasionally we recompute the full convolution to correct accumulated
   floating‑point drift.

The routine runs for the allocated time budget (default 1000 s) and
returns the best sequence found.  All values stay in the allowed range
`[0, 1000]`, and the final sequence is compatible with the supplied
`evaluate_sequence` function.  Progress is printed periodically.
"""

import math
import random
import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution x*x (length 2·len(x)-1)."""
    return np.convolve(x, x)


def _objective(conv: np.ndarray, total: float, n: int) -> float:
    """Compute 2·n·max(conv)/(total**2).  `total` should be Σ a_i."""
    return 2.0 * n * conv.max() / (total * total)


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop a few seconds
        before the limit to guarantee a return value).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise the sequence (uniform or the historic construction)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the competition environment
        base_seq = np.array(list(height_sequence_1), dtype=np.float64)
        n = base_seq.size
        if np.random.rand() < 0.5 and base_seq.sum() > 0.0:
            a = base_seq.copy()
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    except NameError:
        # fallback: a modest size uniform vector
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure a positive total (the evaluate function rejects totals < 0.01)
    if a.sum() < 0.01:
        a = np.full_like(a, 1.0 / a.size)

    # --------------------------------------------------------------
    # 2.  Pre‑compute the convolution and the current objective value
    # --------------------------------------------------------------
    conv = _full_convolution(a)
    total = a.sum()
    best_score = _objective(conv, total, n)
    best_seq = a.copy()

    # Timing
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a safety margin

    # Annealing schedule
    temperature = 0.1
    it = 0
    RECOMPUTE_EVERY = 5000   # full recompute to avoid drift

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 3.  Choose a move
        # ----------------------------------------------------------
        # 3a)  With probability 0.7 aim at the currently maximal k
        if np.random.rand() < 0.7:
            k_max = int(conv.argmax())
            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[valid] = a[valid] * a[partner[valid]]

            i = int(np.argmax(contrib))           # index that contributes most
            # pick a low‑contributing partner (different from i)
            low_candidates = np.where(contrib == contrib.min())[0]
            low_candidates = low_candidates[low_candidates != i]
            if low_candidates.size == 0:
                j = np.random.randint(n)
            else:
                j = int(np.random.choice(low_candidates))
        else:
            # 3b)  Random move
            i, j = np.random.choice(n, size=2, replace=False)

        if a[i] <= 0.0:
            continue

        # Choose the amount to transfer (up to 50 % of a[i])
        max_delta = a[i]
        delta = max_delta * np.random.rand() * 0.5 + 1e-12
        di = -delta
        dj =  delta

        # ----------------------------------------------------------
        # 4.  Incrementally update the convolution
        # ----------------------------------------------------------
        # Copy the old convolution (O(n) – cheap compared to a full FFT)
        new_conv = conv.copy()

        # (+) contributions from the linear part 2·δ·a
        #    (the slice a appears at positions i … i+n‑1)
        upd_i = 2.0 * di * a
        upd_j = 2.0 * dj * a
        new_conv[i:i + n] += upd_i
        new_conv[j:j + n] += upd_j

        # (+) quadratic terms (only a few single positions)
        new_conv[2 * i] += di * di
        new_conv[2 * j] += dj * dj
        new_conv[i + j] += 2.0 * di * dj

        # ----------------------------------------------------------
        # 5.  Update the sequence and evaluate the new objective
        # ----------------------------------------------------------
        new_a = a.copy()
        new_a[i] += di
        new_a[j] += dj

        # Safeguard: values must stay non‑negative and not explode
        if new_a[i] < 0.0 or new_a[j] < 0.0:
            continue
        if new_a.max() > 1000.0:      # the evaluate function clips at 1000
            continue

        new_score = _objective(new_conv, total, n)
        curr_score = _objective(conv, total, n)

        # ----------------------------------------------------------
        # 6.  Acceptance (simulated annealing)
        # ----------------------------------------------------------
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # probabilistic acceptance of uphill moves
            prob = math.exp(-(new_score - curr_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = new_a
            conv = new_conv
            # Remember the best solution seen so far
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()
                if it % 1000 == 0:
                    print(f"[{time.time() - start_time:6.1f}s] iter {it:7d}  best_score {best_score:.6f}")

        # ----------------------------------------------------------
        # 7.  Temperature cooling and occasional full recompute
        # ----------------------------------------------------------
        if it % 1000 == 0:
            temperature *= 0.99   # exponential cooling

        if it % RECOMPUTE_EVERY == 0:
            # Full recompute eliminates accumulated floating‑point drift
            conv = _full_convolution(a)

    # ------------------------------------------------------------------
    # 8.  Return the best sequence found
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations – best_score = {best_score:.6f}")
    # Convert to a plain Python list (the evaluation harness expects this)
    return best_seq.tolist()
