def compute_max_conv(seq):
    a = np.array(seq, dtype=np.float64)
    n = a.size
    m = 1 << ((2*n-1).bit_length())  # next power of 2 size for FFT

    conv = np.fft.irfft(Fa*Fa, n=m)[:2*n-1]
    return np.max(conv)

        fa = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]

    curr_seq = seq.copy()

        fa = np.fft.rfft(curr_seq, n=m)
        conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
        k_max = np.argmax(conv)

        # We'll compute using slicing logic
        c = np.zeros(n, dtype=np.float64)
        # i from max(0, k_max-(n-1)) to min(n-1, k_max)
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        # This range inclusive: indices i in [i_start, i_end]
        # correspond j = k_max - i
        idx = np.arange(i_start, i_end+1)
        jdx = k_max - idx
        c[idx] = curr_seq[idx] * curr_seq[jdx]

        # choose donor i with large contribution
        # weighting by contribution to pick high value
        # If all contributions zero, fallback random
        if np.max(c) > 0:
            i_donor = np.argmax(c)
        else:
            i_donor = np.random.randint(n)

# -*- coding: utf-8 -*-
"""
Simulated‑annealing flattening of the auto‑convolution.

The evaluation metric for a sequence `a` of length `n` is
    2 * n * max(conv(a, a)) / (sum(a)**2)

Because the metric is homogeneous, we may enforce the
normalisation  sum(a) = sqrt(2*n).  Under this normalisation the
metric reduces to `max(conv(a,a))`.  The goal is therefore to
make the convolution as flat as possible – i.e. to lower the
largest entry of `a*a`.

The algorithm works as follows

* start from a uniform non‑negative step function (the current
  SOTA construction is also available as ``height_sequence_1``);
* repeatedly move a small amount of mass from an index that
  contributes heavily to the current maximal convolution entry to
  an index that contributes little;
* accept improving moves always and worsening moves with a
  temperature‑dependent probability (simulated annealing);
* slowly cool the temperature; occasionally restart from a random
  Dirichlet‑distributed sequence if progress stalls;
* keep the best sequence seen and return it when the time budget
  (≈ 1000 s) is exhausted.

All heavy work (the convolution) is performed with an FFT
so that each iteration costs only O(n log n).  The routine
``propose_candidate`` follows the required signature.
"""

import time
import numpy as np

# ------------------------------------------------------------
# helper: fast computation of the maximum value of the full convolution
def _max_convolution(seq: np.ndarray, fft_len: int) -> float:
    """Return max(conv(seq, seq)) using a real FFT of length ``fft_len``."""
    # rfft/irfft keep the result real and are faster than a full complex FFT
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * len(seq) - 1]
    return float(conv.max())


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds). The function must return before the
        budget expires; a few seconds are kept as a safety margin.
    kwargs
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we work with.
    # Prefer the length of the previously found construction if it exists.
    if "height_sequence_1" in globals() and isinstance(
        globals()["height_sequence_1"], (list, np.ndarray)
    ):
        n = len(globals()["height_sequence_1"])
    else:
        n = np.random.randint(500, 1500)  # fall‑back length

    target_sum = np.sqrt(2.0 * n)          # we keep sum(seq) = sqrt(2*n)

    # --------------------------------------------------------------
    # FFT length – next power of two that contains the full convolution.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Starting point – uniform distribution (the obvious SOTA baseline).
    seq = np.full(n, target_sum / n, dtype=np.float64)

    # Compute initial score directly via the provided evaluator (for logging).
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())   # this equals max(conv) after normalisation
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Variables that drive the annealing loop
    curr_seq = seq.copy()
    curr_max = _max_convolution(curr_seq, fft_len)

    temperature = best_score                     # start temperature
    min_temp = 1e-6
    iteration = 0
    no_improve = 0                               # counter for stagnation
    deadline = time.time() + budget_s - 5.0      # keep a few seconds free

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) compute the full convolution of the current sequence
        f = np.fft.rfft(curr_seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())
        max_val = float(conv[k_max])

        # --------------------------------------------------------------
        # 2) Determine which indices contribute most/least to this
        #    maximal entry.
        #    Contribution of index i is a_i * a_{k_max-i} when the partner
        #    index lies in [0, n-1]; otherwise the contribution is zero.
        contrib = np.zeros(n, dtype=np.float64)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx
        contrib[idx] = curr_seq[idx] * curr_seq[jdx]

        # donor – a position with a large contribution
        if contrib.max() > 0:
            i_donor = int(contrib.argmax())
        else:
            i_donor = np.random.randint(n)

        # receiver – a position with a small contribution (bottom 30 %)
        low_thr = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size > 0:
            j_receiver = int(np.random.choice(low_candidates))
            if j_receiver == i_donor:
                # ensure the two indices differ
                j_receiver = (i_donor + np.random.randint(1, n)) % n
        else:
            j_receiver = np.random.randint(n)
            while j_receiver == i_donor:
                j_receiver = np.random.randint(n)

        # --------------------------------------------------------------
        # 3) Propose a move: transfer a fraction of the donor's mass.
        donor_val = curr_seq[i_donor]
        if donor_val <= 0:
            # fallback – random small move
            i_donor = np.random.randint(n)
            donor_val = curr_seq[i_donor]

        max_delta = donor_val * 0.5               # we never move more than half
        # temperature controls the size of the step
        temp_factor = max(min(temperature, 1.0), 0.01)
        delta = max_delta * np.random.rand() * temp_factor
        if delta <= 0:
            delta = max_delta * 0.01

        # --------------------------------------------------------------
        # 4) Apply the move and re‑normalise the sum.
        new_seq = curr_seq.copy()
        new_seq[i_donor] -= delta
        new_seq[j_receiver] += delta

        # Clip possible numerical negatives (should be very rare)
        if new_seq[i_donor] < 0:
            # move whatever we could take from the donor
            delta += new_seq[i_donor]          # new_seq[i_donor] is negative
            new_seq[i_donor] = 0.0
            new_seq[j_receiver] = max(new_seq[j_receiver], 0.0)

        # Enforce the exact sum = sqrt(2*n) (scaling is cheap and keeps the metric unchanged)
        s = new_seq.sum()
        if s == 0:
            # extremely unlikely – re‑initialise uniformly
            new_seq = np.full(n, target_sum / n, dtype=np.float64)
        else:
            new_seq *= (target_sum / s)

        # --------------------------------------------------------------
        # 5) Evaluate the new maximum of the auto‑convolution.
        new_max = _max_convolution(new_seq, fft_len)

        # --------------------------------------------------------------
        # 6) Acceptance test (simulated annealing).
        accept = False
        if new_max < curr_max:
            accept = True
        else:
            if temperature > min_temp:
                prob = np.exp((curr_max - new_max) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            curr_seq = new_seq
            curr_max = new_max

            # Track the best sequence ever seen.
            if new_max + 1e-12 < best_score:
                # Use the official evaluator for the true score (handles the tiny
                # safety clamping performed inside the grader).
                best_score = evaluate_sequence(new_seq.tolist())
                best_seq = new_seq.copy()
                print(
                    f"[{iteration:06d}] new best: score={best_score:.6f}  "
                    f"max_conv={new_max:.6f}  temp={temperature:.4f}"
                )
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 7) Temperature schedule (geometric cooling)
        temperature = max(min_temp, temperature * 0.9999)

        # --------------------------------------------------------------
        # 8) Restart if we have been stuck for a while.
        if no_improve > 5000:
            # Random Dirichlet start – keeps the sum fixed after scaling.
            rand_vec = np.random.rand(n)
            rand_vec *= target_sum / rand_vec.sum()
            curr_seq = rand_vec
            curr_max = _max_convolution(curr_seq, fft_len)
            temperature = best_score          # re‑heat a little
            no_improve = 0
            # occasional log
            print(f"[{iteration:06d}] restart due to stagnation")

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification and return.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score returned = {final_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
