def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Constructs a non‑negative step function whose self‑convolution is
    approximately flat.  The shape is a discretised arcsine (Beta(0.5,0.5))
    density, which is known to convolve with itself to a (nearly) uniform
    distribution.  After normalising the sum of the heights to √(2·n) the
    evaluation function equals the maximal convolution entry; a uniform
    convolution yields a value close to 1, well below the target 1.503.
    The routine also tries a few neighbouring Beta‑α shapes (α∈[0.3,0.8])
    and keeps the best sequence found before the time budget expires.
    """
    import numpy as np, time, math

    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)   # keep a little margin

    # use the same length as the current best construction – it gives a
    # reasonable trade‑off between resolution and computation time.
    n = 4595

    best_seq = None
    best_score = float('inf')

    # try a few symmetric Beta(α,α) densities (α=0.5 gives the arcsine law)
    alphas = [0.5, 0.45, 0.55, 0.4, 0.6, 0.35, 0.65]
    for alpha in alphas:
        if time.time() > deadline:
            break
        # points in (0,1) – use midpoint rule to avoid division by zero
        i = np.arange(n, dtype=np.float64)
        x = (i + 0.5) / n                     # centre of each bin
        # β(α,α) density up to a normalising constant: x^{α‑1}(1‑x)^{α‑1}
        w = np.power(x * (1.0 - x), alpha - 1.0)
        # normalise the sum to √(2·n) as required by the evaluation function
        Z = np.sum(w)
        target_sum = math.sqrt(2.0 * n)
        a = target_sum * w / Z                # a_i ≥ 0
        # clip aggressively – the construction never exceeds the 1000‑limit
        a = np.clip(a, 0.0, 1000.0)

        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            score = float('inf')

        print(f"alpha={alpha:.3f}, score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.tolist()

    # If time permits, perform a tiny random local search starting from the
    # best sequence obtained above.  The move keeps the total sum exactly
    # constant by transferring a small amount of mass from one index to
    # another.
    if best_seq is not None and time.time() < deadline:

        target_sum = np.sum(seq)          # should already be √(2·n)
