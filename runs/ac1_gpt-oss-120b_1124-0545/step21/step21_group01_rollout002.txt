"""
Propose candidate step function.

Algorithm:
1. Load previous best sequence if available.
2. Build a diverse set of seed sequences (uniform, arcsine, triangular,
   beta distributions, Dirichlet draws, sinusoidal, random‑phase FFT).
   Evaluate each and keep the best.
3. Perform a high‑budget stochastic hill‑climbing / simulated‑annealing
   search that repeatedly transfers a small amount of mass between two
   coordinates while staying on the probability simplex.
   The step size decays over time and the temperature of the annealing
   schedule also falls, allowing fine‑grained adjustments.
4. Return the best sequence found (the global variable ``height_sequence_1`` is updated).
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (complex round‑off errors removed)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    # Numerical noise may produce tiny imaginary parts; keep real part.
    return conv.real

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)). Assumes sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a

def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a = a / a.sum()
    return a

def _phase_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative sequence via a random‑phase FFT."""
    # Use length M = power of two ≥ 2n‑1 for a convenient real IFFT.
    M = _next_pow2(2 * n - 1)
    # Random phases for the positive frequencies (excluding 0 and Nyquist).
    half = M // 2
    phases = rng.uniform(0.0, 2.0 * np.pi, size=half - 1)
    X = np.empty(M, dtype=np.complex128)
    X[0] = 1.0  # DC component (real)
    # Nyquist component for even M must be real.
    if M % 2 == 0:
        X[half] = 1.0
        pos = np.arange(1, half)
    else:
        pos = np.arange(1, half + 1)
    X[pos] = np.exp(1j * phases)
    X[-pos] = np.conj(X[pos])
    seq = np.fft.ifft(X).real
    # Shift to non‑negative, discard tiny negatives.
    seq = np.maximum(seq, 0.0)
    # Keep only first n entries (the rest are wrap‑around artefacts).
    seq = seq[:n]
    total = seq.sum()
    if total <= 0.0:
        return _uniform(n)
    seq /= total
    return seq

def _evaluate_surrogate(seq: np.ndarray) -> float:
    """Wrapper that guarantees sum(seq)=1 before calling the surrogate."""
    s = seq.sum()
    if s == 0.0:
        return np.inf
    return _objective_surrogate(seq / s)

def _initial_seed_pool(n: int, rng: np.random.Generator, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = [_uniform(n)]
    if prev_seq is not None:
        seeds.append(prev_seq)
    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_distribution(n))
    # a few beta shapes
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, alpha, beta))
    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Random‑phase FFT seeds
    for _ in range(5):
        seeds.append(_phase_random(n, rng))
    return seeds

def _stochastic_mass_transfer_search(start_seq: np.ndarray,
                                     n: int,
                                     deadline: float,
                                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climbing that repeatedly moves a tiny amount of mass.
    Returns the best sequence (and its surrogate value) encountered.
    """
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    cur_seq = best_seq.copy()
    cur_val = best_val

    start_time = time.time()
    total_budget = deadline - start_time
    # Parameters for the schedule
    eps_start = 0.02   # maximal relative move at the beginning
    eps_end = 0.0005   # minimal relative move near the end
    temp_start = 0.01
    temp_end = 1e-6

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start_time
        frac = elapsed / total_budget  # 0 → 1
        eps = eps_start * (1.0 - frac) + eps_end * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose a delta that is a fraction of the source mass.
        source = cur_seq[i]
        if source <= 0.0:
            continue
        delta = eps * source * rng.random()
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # Numerical safeguard
        if new_seq[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_seq)

        # Accept if improvement, otherwise with Metropolis probability.
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

    # final normalisation (tiny drift may have occurred)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = _uniform(n)
    return best_seq, best_val

    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sine_distribution(n: int) -> np.ndarray:
    """Sine‑shaped density."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a /= a.sum()
    return a


def _phase_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a non‑negative vector via a random‑phase FFT.
    The construction creates a real sequence, discards tiny negatives,
    truncates to length n and normalises.
    """
    M = _next_pow2(2 * n - 1)          # length for a clean linear convolution
    half = M // 2
    phases = rng.uniform(0.0, 2.0 * np.pi, size=half - 1)
    X = np.empty(M, dtype=np.complex128)
    X[0] = 1.0
    if M % 2 == 0:
        X[half] = 1.0
        pos = np.arange(1, half)
    else:
        pos = np.arange(1, half + 1)
    X[pos] = np.exp(1j * phases)
    X[-pos] = np.conj(X[pos])
    seq = np.fft.ifft(X).real
    seq = np.maximum(seq, 0.0)          # enforce non‑negativity
    seq = seq[:n]
    total = seq.sum()
    if total <= 0.0:
        return _uniform(n)
    seq /= total
    return seq


def _initial_seed_pool(
    n: int,
    rng: np.random.Generator,
    prev_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse list of candidate vectors (all sum to 1)."""
    seeds: list[np.ndarray] = [_uniform(n)]
    if prev_seq is not None:
        seeds.append(prev_seq)
    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_distribution(n))

    # a few beta shapes
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, alpha, beta))

    # Dirichlet draws with several concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random‑phase FFT seeds
    for _ in range(5):
        seeds.append(_phase_random(n, rng))

    return seeds


def _stochastic_mass_transfer_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climbing that repeatedly moves a tiny amount
    of probability mass between two coordinates.
    Returns the best sequence seen and its surrogate value.
    """
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    cur_seq = best_seq.copy()
    cur_val = best_val

    start_time = time.time()
    total_budget = deadline - start_time

    # schedules (linear decay)
    eps_start, eps_end = 0.02, 0.0005      # relative step size
    temp_start, temp_end = 0.01, 1e-6     # Metropolis temperature

    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start_time
        frac = elapsed / total_budget  # 0 → 1

        eps = eps_start * (1.0 - frac) + eps_end * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src = cur_seq[i]
        if src <= 0.0:
            continue
        delta = eps * src * rng.random()
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:               # safeguard against round‑off
            continue

        new_val = _objective_surrogate(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

    # final normalisation (tiny drift may have occurred)
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = _uniform(n)

    return best_seq, best_val


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence is stored in the global variable ``height_sequence_1``
    and also returned as a plain ``list`` of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small margin so we can still call ``evaluate_sequence`` at the end
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Retrieve previously found solution (if it matches the current size)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣ Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = _initial_seed_pool(n, rng, prev_arr)

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate all seeds and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating initial seed pool ---")
    for idx, seq in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(seq)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Additional cheap random search (more Dirichlet draws)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Extra Dirichlet random search ---")
        conc_choices = (0.01, 0.02, 0.05, 0.1)
        trials = 1500
        for t in range(trials):
            if time.time() > deadline:
                break
            conc = rng.choice(conc_choices)
            cand = _random_dirichlet(n, conc, rng)
            val = _objective_surrogate(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"  [Dir] new best surrogate = {best_val:.6f} (iter {t})")

    # ------------------------------------------------------------------
    # 3️⃣ Fine‑grained stochastic refinement (mass‑transfer SA)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Stochastic mass‑transfer refinement ---")
        refined_seq, refined_val = _stochastic_mass_transfer_search(
            best_seq, n, deadline, rng
        )
        print(f"[Refine] surrogate = {refined_val:.6f}")
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq.copy()
            print("   >>> Updated global best (refine)")

    # ------------------------------------------------------------------
    # 4️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
