    p = _softmax(y)
    conv = _conv_fft(p, p)
    k = np.argmax(conv)

    g = np.zeros_like(p)

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – a diverse pool of probability vectors of length
   ``n`` (uniform, Dirichlet, sparse Dirichlet, Gaussian‑mixture, Beta‑shaped
   PDFs and, if available, the previous best).  All seeds are ℓ₁‑normalised
   (they sum to one) so that the surrogate objective
   ``2·n·max(conv(p,p))`` coincides with the true ``evaluate_sequence`` value.

2. **Initial selection** – evaluate the surrogate for every seed and keep the
   best one.

3. **Mirror‑descent polishing** – run an *Adam* version of exponentiated‑gradient
   (mirror descent on the simplex) on the *exact* max‑convolution objective.
   The update works on log‑probabilities, uses a decaying step‑size
   ``ηₜ = η₀/√t`` and stops when the time budget is exhausted.

4. **Stochastic refinements** – as long as time permits, perturb the current
   best sequence with a tiny Gaussian noise, run a short Adam polishing and
   keep any improvement.

5. **Final evaluation** – call ``evaluate_sequence`` on the best sequence
   found, store it in the global variable ``height_sequence_1`` and return it.

All heavy work respects the global 1000 s budget; the optimizer is
implemented with only NumPy (no external optimisation packages) and
uses a fast FFT‑based convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact objective for a probability vector ``p`` (sum(p)=1):
        2 * n * max(conv(p, p))
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick ``k`` positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of ``K`` Gaussian windows.
    ``params`` layout: [w_0..w_{K‑1}, μ_0..μ_{K‑1}, σ_0..σ_{K‑1}]
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Normalised discrete Beta(α,β) density on ``n`` points.
    The grid avoids the exact 0 and 1 to keep the PDF finite.
    """
    # centre of each bin, never exactly 0 or 1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _perturb(p: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = p + sigma * rng.normal(size=p.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return p.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _expadam_opt(start_p: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 2_500_000,
                 eta0: float = 2.0,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8):
    """
    Adam‑style exponentiated‑gradient on the true max‑convolution.
    Operates on log‑probabilities (mirror descent w.r.t. the negative
    entropy).  Returns the best probability vector encountered and its
    surrogate objective.
    """
    # initialise log‑probabilities
    y = np.log(start_p + 1e-30)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_p = start_p.copy()
    best_val = _objective_surrogate(start_p)

    t = 0
    while True:
        if time.time() > deadline:
            break
        if max_iters is not None and t >= max_iters:
            break
        t += 1

        # softmax of y (stable)
        y_shift = y - np.max(y)
        p = np.exp(y_shift)
        p_sum = p.sum()
        if p_sum == 0.0:
            p = _uniform(n)
        else:
            p = p / p_sum

        # convolution and active shift
        conv = _conv_fft(p, p)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(p)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * p[rev_idx]

        # gradient w.r.t. log‑probabilities (mirror‑descent rule)
        dot = np.dot(g, p)
        grad_y = p * (g - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size (inverse‑sqrt schedule)
        eta = eta0 / math.sqrt(t)
        y = y - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional progress report and best‑track
        if t % 5000 == 0:
            cur_val = _objective_surrogate(p)
            if cur_val < best_val:
                best_val = cur_val
                best_p = p.copy()
                print(f"[Adam] iter {t:7d} – best surrogate = {best_val:.8f}")

    return best_p, best_val

            eta0=1.5
        )
        print(f"[Pert‑Adam] surrogate = {cand_val:.8f}")
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"  >>> New best after perturb‑Adam = {best_val:.8f}")

    # --------------------------------------------------------------
    # 3️⃣  Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.8f}, evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
