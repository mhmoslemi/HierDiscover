def _exp_gradient_opt(start_seq, n, deadline, max_iters=500_000, eta0=5.0):
    a = start_seq.copy()
    # ensure sum=1
    sum_a = a.sum()
    if sum_a <= 0: a = _uniform(n)
    a = a / sum_a

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log a representation
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # produce gradient wrt log_a (self)
        # derivative of log domain: ∂/∂ log a_i = a_i * ∂obj/∂a_i - a_i * sum a * ∂obj/∂a.
        # but easier: we can compute subgradient w.r.t a: g_i = 2 * a_{k-i} for i in overlapping region
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        # multiply update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = best_seq.copy()
            continue
        a = a / s

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val

"""
Propose a step‑function (non‑negative list of heights) that minimizes the
provided ``evaluate_sequence`` objective.

The algorithm is deliberately simple yet aggressive:
  1. Build a large, diverse pool of seed vectors (uniform, arcsine,
     triangular, several beta‑shapes, geometric decay, linear decay,
     Dirichlet draws with many concentration parameters, and small
     perturbations of the uniform vector).  All seeds are normalised to
     sum = 1.
  2. Evaluate the exact surrogate 2·n·max(conv(seq,seq)) for every seed
     and keep the best one.
  3. While time permits, draw many additional Dirichlet samples and keep
     any improvement.
  4. Perform a long exponentiated‑gradient (mirror‑descent) optimisation
     on the true max‑convolution.  This uses the exact sub‑gradient at the
     active lag and a decreasing step‑size η₀/√t.
  5. Finish with a cheap random local walk that moves a tiny amount of mass
     from one coordinate to another and accepts improvements.
  6. The best sequence found is stored in the global ``height_sequence_1``
     variable and returned as a plain list of floats.

The whole procedure respects the 1000 s budget; the heavy optimisation
stage (exponentiated‑gradient) automatically stops when the deadline is
reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq, seq)), assuming sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc`` (vector of length n)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric‑decay distribution (rate close to 1)."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    # normalise to sum = 1
    a = (1.0 - rate) * np.power(rate, i)
    eps = 1e-12
    a_sum = a.sum()
    if a_sum <= eps:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    """Deterministic linear decay from left to right."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (exact sub‑gradient of max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 8.0):
    """
    Mirror‑descent on the true max‑convolution.
    ``start_seq`` must already sum to 1.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                # work in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact convolution
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active lag (see analysis for factor 2)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step‑size
        eta = eta0 / math.sqrt(it)

        # additive update on log‑variables, then softmax to stay on simplex
        log_a = log_a - eta * g
        a = _softmax(log_a)                # projects back onto the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random amount from one
    coordinate to another and keep the move if it improves the exact surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a relative amount from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
