"""
optimised_step_function_v2.py

Search for a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The new strategy builds on the previous pipeline but
adds two important upgrades that have proved useful for breaking the 1.5085
barrier:

1.  **Richer seed pool** – extra analytic families (edge‑heavy power law) and
    a much finer grid for the U‑shaped beta family.  More diverse initial
    points give the optimizer better chances to escape local minima.

2.  **Adam‑based smoothed max‑convolution polishing** – after the classical
    exponentiated‑gradient and permutation hill‑climbing stages we run a
    custom Adam optimizer on a log‑sum‑exp smooth approximation of the
    objective.  The schedule of the smoothing parameter β is increased
    geometrically, while Adam’s adaptive step‑sizes let us make larger,
    more directed updates than the simple EG scheme.

The overall budget is 1000 s.  Time is split between seed generation,
local refinement of the best seeds, a final Adam polishing phase, and a
short permutation fine‑tune.  The best sequence is stored in the global
variable ``height_sequence_1`` for possible continuation runs.
"""

# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    """Gaussian with adjustable mean (mu_frac∈[0,1] gives position in the array)."""
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_heavy(n: int, power: float) -> np.ndarray:
    """
    Edge‑heavy shape: a_i ∝ ((i+1)*(n-i))^{-power}.
    Gives large values near both ends, small in the middle.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-power)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Normalisation & surrogate helper

def _norm_and_surrogate(vec: np.ndarray):
    """Clip, normalise to sum = 1 and return (normalised_vec, surrogate)."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return None, np.inf
    vec = vec / s
    return vec, _objective_surrogate(vec)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the active index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val

# Optional L‑BFGS smoothing optimiser (kept from the baseline)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution (log‑sum‑exp) objective.
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          iters_per_beta: int = 20_000,
                          lr: float = 0.25):
    """
    Adam optimisation on the smooth approximation
        M_β(a) = 2·n·(max_c + log∑exp(β·(c‑max_c))/β)
    where c = a∗a.
    The optimizer works in the log‑softmax (logits) space; the sequence
    is always kept on the probability simplex.
    """
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800,
                         50_000, 200_000, 800_000, 2_000_000]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    # Adam hyper‑parameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            # ---- forward pass ------------------------------------------------
            # softmax to obtain a
            a = _softmax(v)

            # convolution a*a
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # smooth weights (log‑sum‑exp)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth objective (not needed for the update but handy for logging)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp

            # ---- gradient w.r.t. a (smooth max) -------------------------------
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. logits (chain rule for softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ---- Adam update -------------------------------------------------
            t += 1
            m = b1 * m + (1 - b1) * grad_v
            v2 = b2 * v2 + (1 - b2) * (grad_v ** 2)
            m_hat = m / (1 - b1 ** t)
            v_hat = v2 / (1 - b2 ** t)

            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- keep best ----------------------------------------------------
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The pipeline:
      1. Generate a very large pool of analytic seeds (including a dedicated
         edge‑heavy family and a fine β‑grid for the U‑shaped Beta family).
      2. Keep the top K seeds according to the fast surrogate.
      3. Refine each seed with deterministic ordering, permutation hill‑climbing,
         exponentiated‑gradient polishing and (optional) L‑BFGS smoothing.
      4. Run a dedicated Adam‑based smoothed‑max polishing on the best seed.
      5. One last short permutation tweak.
      6. Return the best sequence (also stored as ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0          # safety margin

    # --------------------------------------------------------------
    # Problem dimension – a balance between expressiveness and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution (if present) for continuation.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Massive seed generation (dense deterministic sweeps + random draws)
    # --------------------------------------------------------------
    seeds = []

    # Analytic families – dense grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine windows – dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 70):
        seeds.append(_sine_window(n, gamma))

    # Gaussian – sweep sigma fraction
    for sigma_frac in np.linspace(0.04, 0.34, 35):
        seeds.append(_gaussian(n, sigma_frac))

    # Shifted Gaussians – sweep centre and width
    for mu_frac in np.linspace(0.15, 0.85, 16):
        for sigma_frac in np.linspace(0.04, 0.30, 14):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # Exponential decay – fine log‑space sweep
    for rate in np.logspace(-5, 0.5, 80):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Power‑law – wide sweep
    for p in np.linspace(0.0, 6.0, 35):
        seeds.append(_power(n, p))

    # Edge‑heavy family (U‑shape) – finer grid of power exponents
    for power in np.linspace(0.2, 3.0, 30):
        seeds.append(_edge_heavy(n, power))

    # Beta‑shape – systematic fine grid, concentrating on U‑shapes (α,β < 1)
    alphas = np.geomspace(0.05, 0.9, 12)
    betas  = np.geomspace(0.05, 0.9, 12)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # Random Dirichlet draws (various concentrations)
    conc_vals = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec, val = _norm_and_surrogate(vec)
        if vec is None:
            continue
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refinement of each top seed
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a dynamic slice of the remaining budget (~10 % each)
        remaining = deadline - time.time()
        share = remaining * 0.10 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) deterministic monotone ordering (descending) – often already good
        seq_ordered = np.sort(seed_vec)[::-1]

        # b) permutation hill‑climb – give it a generous iteration budget
        max_hc_iters = int(180_000 * (share / 30.0))   # scale with allocated time
        seq_perm, val_perm = _perm_hill_climb(seq_ordered, n,
                                              sub_deadline,
                                              rng,
                                              max_iters=max_hc_iters)

        # c) exponentiated‑gradient polishing
        if time.time() < sub_deadline:
            eg_iters = int(250_000 * (share / 30.0))
            seq_eg, val_eg = _exp_gradient_opt(seq_perm, n,
                                               sub_deadline,
                                               max_iters=eg_iters,
                                               eta0=7.5)
        else:
            seq_eg, val_eg = seq_perm, val_perm

        # d) optional L‑BFGS smoothing (if time permits)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n,
                                               sub_deadline,
                                               max_iter_per_beta=1800)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # pick the best of the refinements
        candidates = [
            (seq_ordered, _objective_surrogate(seq_ordered)),
            (seq_perm, val_perm),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long Adam polishing (smooth max approximation)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam polishing (smooth max) ---")
        # Use a relatively aggressive schedule and many iterations
        final_seq, final_val = _adam_smooth_optimize(global_best_seq,
                                                     n,
                                                     deadline,
                                                     iters_per_beta=25_000,
                                                     lr=0.30)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Adam] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. One last lightweight permutation tweak (tiny budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 22.0   # about half a minute
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=90_000)
        if val_last < global_best_val:
            global_best_seq, global_best_val = seq_last, val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
