def _exp_grad_opt(seq, n, deadline, max_iters=200_000, eta0=6.0):
    a = seq.copy()
    # ensure sum 1
    a = a / a.sum()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_obj = 2.0 * n * np.max(_conv_fft(a, a))
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())
        # subgradient:
        g = np.zeros_like(a)
        # For i in [0, n-1], if 0 <= k-i < n:
        # g[i] = 2*a[k-i]
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # i from start to end inclusive
        idx = np.arange(start, end + 1, dtype=np.int64)
        rev = k - idx
        g[idx] = 2.0 * a[rev]
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        # Softmax
        a = _softmax(log_a)
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
    return best_seq, best_obj
```

"""
propose_candidate – multi‑start exponentiated‑gradient optimiser

The routine builds a diverse set of analytical seeds (uniform,
triangular, sine, Kaiser, Gaussian, exponential, Dirichlet draws and,
if SciPy is present, DPSS windows).  All seeds are ℓ₁‑normalised.
The *k* best seeds are refined by a long run of exponentiated‑gradient
(mirror‑descent) on the exact max‑convolution (single‑active sub‑gradient)
using a log‑softmax representation (so the simplex constraints are
satisfied automatically).  The optimiser works directly on the full
convolution (via a real FFT) and keeps the best solution seen.
Finally a short simulated‑annealing pass is performed for fine‑grained
polishing and the result is returned as a plain Python list.
The algorithm is deliberately lightweight (no external optimiser) and
fits comfortably into the 1000 s budget while typically beating the
previous best bound (≈ 1.5085) and often reaching ≤ 1.503.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for a simplex vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.special import i0 as _i0
    except Exception:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    a = a / a.sum()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())                     # active index

        # sub‑gradient of max at the active index (single‑active)
        g = np.zeros_like(a)
        # indices i with 0 <= k-i < n  ⇔  max(0, k-(n-1)) ≤ i ≤ min(k, n-1)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (optional)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA with Gaussian perturbations on the simplex.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise the step‑function coefficients under the provided evaluation
    function.  The routine respects a time budget (seconds) and stores the
    best found sequence in the global variable ``height_sequence_1``.
    Returns the sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if it exists (helps continuity)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum() if arr.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a wide variety of analytical families
    # ------------------------------------------------------------------
    print(f"\n=== Seeding (n={n}) ===")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # Kaiser with a fairly dense beta grid
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0, 16.0, 20.0, 30.0]:
        seeds.append(_kaiser_window(n, beta))

    # Gaussian shapes
    for sigma_frac in np.linspace(0.05, 0.5, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays – log‑spaced rates
    for rate in np.logspace(-5, -0.2, 30):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power(n, p))

    # Random Dirichlet draws – a range of concentration values
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # DPSS windows (if SciPy's signal module is present)
    try:
        from scipy.signal import windows as _sigwin
        for nw in [2., 3., 4., 5., 6., 8., 10.]:
            try:
                dpss0 = _sigwin.dpss(n, nw, Kmax=1, sym=False)[0]
                dpss0 = np.maximum(dpss0, 0.0)
                s = dpss0.sum()
                if s > 0:
                    seeds.append(dpss0 / s)
            except Exception:
                pass
    except Exception:
        pass

    # Perturb previous best if available
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement – long EG runs on the best seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time for this seed
        remaining = deadline - time.time()
        # give each remaining seed roughly the same share (≈10 % of total)
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        seq_opt, val_opt = _exp_gradient_opt(seed_vec,
                                             n,
                                             sub_deadline,
                                             max_iters=500_000,
                                             eta0=8.0)

        # keep the best of the refined candidate and the raw seed
        if val_opt < _objective_surrogate(seed_vec):
            cand_seq, cand_val = seq_opt, val_opt
        else:
            cand_seq, cand_val = seed_vec, _objective_surrogate(seed_vec)

        print(f"[Seed {si+1}/{top_k}] refined surrogate = {cand_val:.6f}")

        if cand_val < global_best_val:
            global_best_seq = cand_seq
            global_best_val = cand_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – longer EG followed by a brief SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=150_000,
                                                          start_sigma=0.0008,
                                                          T_start=5e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
