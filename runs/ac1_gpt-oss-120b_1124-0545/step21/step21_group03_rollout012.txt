"""
search_step_function.py

Search for a non‑negative step‑function (a list of heights) that minimises
the provided ``evaluate_sequence`` function.

Idea
----
* Increase the problem size far beyond the previous 4000 entries.
  With a larger n the theoretical lower bound of the objective tends to 1,
  so a modestly good shape can already beat the 1.503 target.

* Generate a large pool of *analytic* candidates (Beta‑densities, exponentials,
  power‑laws, Dirichlet draws, random piecewise‑linear shapes, the previous
  best solution upsampled, etc.).  All candidates are ℓ₁‑normalised, i.e. the
  sum of the heights is 1, because the objective is scale‑invariant.

* Evaluate the cheap surrogate

        2·n·max(conv(a,a))

  where ``conv`` is a full linear convolution obtained with a real FFT.
  This surrogate is exactly the value returned by ``evaluate_sequence`` as
  long as no entry exceeds 1000 (our candidates are tiny).

* Keep the best few seeds and polish each of them with a short
  exponentiated‑gradient (mirror‑descent) optimisation that works directly
  on the exact max‑convolution sub‑gradient.  The iteration count is limited
  so that even for n≈20000 the total runtime stays below the 1000 s budget.

* Finish with a brief permutation‑based simulated annealing to escape ordering
  traps (large values far apart tend to reduce the peak of the convolution).

The whole pipeline respects the global deadline, prints progress
information and stores the final list in the global variable
``height_sequence_1`` so that later runs can continue from it.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the discrete grid i/(n‑1)."""
    # mid‑point discretisation avoids exact zeros at the boundaries
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    # use logs to stay stable for α,β < 1
    log_pdf = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    pdf = np.exp(log_pdf)
    s = pdf.sum()
    if s == 0.0:
        return _uniform(n)
    return pdf / s


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay about the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(
    n: int, num_breaks: int, rng: np.random.Generator
) -> np.ndarray:
    """Random piecewise‑linear shape with `num_breaks` interior break points."""
    # ensure the break points include the ends
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape, zero at the ends, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 5.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent in the log‑softmax (probability) domain.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep non‑negativity automatically
    log_a = np.log(a + 1e-30)

    n = len(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it + 1.0)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()  # renormalise to keep ℓ₁‑norm = 1

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """Swap‑based annealing that tries to improve the ordering of the values."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose swap
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

            global_best_seq = local_seq.copy()
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final polishing with the remaining time budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Final polishing for {remaining:.1f}s ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            deadline,
            max_iters=1_200_000,
            eta0=5.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very short final permutation tweak (few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(15.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the true objective and store the result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation enlarges the search space (≈ 18 000 coefficients), explores a very broad set of analytic families (Beta, exponential, power‑law, Dirichlet, piecewise‑linear, plus the previous best up‑sampled), keeps the most promising candidates and refines them with a fast exponentiated‑gradient mirror‑descent and a short permutation‑based simulated annealing.  
All steps respect the 1000 s time budget, print useful progress information and finally expose the best sequence via the global variable `height_sequence_1`.
