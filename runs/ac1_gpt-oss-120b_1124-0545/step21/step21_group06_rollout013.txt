def exp_gradient_adam(start_seq, n, deadline, max_iters=200000, eta0=1.0, single_active=True):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _softmax(log_a)
        # compute subgrad
        g, max_val = subgrad_max(a, single_active)
        # update momentum
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # learning rate schedule can be eta0 / sqrt(it)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Broad seeding** – generate thousands of candidates from analytic families
   (uniform, triangular, beta‑shapes, Gaussians, power‑laws, exponentials,
   mixtures) and a handful of Dirichlet draws.  All vectors are ℓ₁‑normalised
   and evaluated with a fast FFT surrogate.  The best few seeds are kept.

2. **Quick local refinement** – each top seed is improved by a *mirror‑descent*
   (exponentiated‑gradient) optimiser that works in log‑softmax space.
   The optimiser uses Adam‑style momentum and an adaptive learning‑rate.
   Only a modest number of iterations (≈80 k) are spent on each seed,
   which is enough to move the seed towards a basin of attraction while
   preserving the global time budget.

3. **Heavy polishing** – the currently best sequence receives a long EG run
   (hundreds of thousands of iterations, limited only by the remaining time)
   followed by a high‑budget simulated‑annealing pass.  The final vector is
   returned and saved in the global variable ``height_sequence_1`` for reuse.

The whole routine respects the 1000 s outer deadline (a tiny safety margin
is kept).  All operations stay inside NumPy; SciPy is used only if available
for an optional L‑BFGS smoothing step (kept low‑budget to stay within the
time limit).  The returned sequence consists of non‑negative floats and
already satisfies the security checks performed by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for a short L‑BFGS smoothing phase)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – assumes sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (left‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i))  (right‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (single active index)
# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, single_active: bool = True):
    """Return (subgradient, max_value) of max(conv(a,a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    if single_active:
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        g = np.zeros_like(a)
        # accumulate a[rev] onto the appropriate positions in g
        np.add.at(g, idx_i, a[rev])
        g = 2.0 * g
    else:
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            np.add.at(g_tot, idx_i, a[rev])
        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)
    return g, max_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (Adam style) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_adam(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 200_000,
                       eta0: float = 5.0,
                       single_active: bool = True):
    """
    Mirror‑descent (exponentiated‑gradient) with Adam momentum.
    Operates in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm = 1.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # stay in log domain
    m = np.zeros_like(a)              # first‑order momentum
    v = np.zeros_like(a)              # second‑order momentum
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # map back to probability simplex
        a = _softmax(log_a)

        # sub‑gradient of max(conv)
        g, _ = _subgrad_max(a, single_active=single_active)

        # Adam‑style updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule
        eta = eta0 / math.sqrt(it)

        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 500):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 500_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.005,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1``.
    """

    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Determine problem size – reuse previous solution if present
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = arr.shape[0]
            if n > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 4000
        prev_arr = None

    print(f"\n=== Optimisation started (n={n}) ===")

    # --------------------------------------------------------------
    # 1. Broad seeding – analytic families + random draws
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_power(n, p=1.0))

    # beta‑shapes (coarse grid)
    alphas = np.linspace(0.1, 8.0, 15)
    betas = np.linspace(0.1, 8.0, 15)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # monotone exponentials
    rates = np.logspace(-4, -0.2, 30)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # power‑law family
    ps = np.linspace(0.0, 5.0, 15)
    for p in ps:
        seeds.append(_power(n, p))

    # sine‑window with varied gamma
    gammas = np.linspace(0.2, 5.0, 20)
    for g in gammas:
        seeds.append(_sine_window(n, gamma=g))

    # Gaussian with varied width
    sigs = np.linspace(0.05, 0.5, 25)
    for s in sigs:
        seeds.append(_gaussian(n, sigma_frac=s))

    # Random Dirichlet draws – explore a range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturb previous best (if any) – encourage local exploration
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr.copy())
        for _ in range(8):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")

                                                  single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high‑budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=600_000,
                                                          start_sigma=0.001,
                                                          T_start=4e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
