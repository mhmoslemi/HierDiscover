def _exp_gradient_multi(start_seq, n, deadline, max_iters, eta0=5.0, epsilon=1e-12):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - epsilon)[0]
        # compute gradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
step‑function optimiser – fresh implementation

The routine builds a very wide collection of seeds (analytically‑generated
families, fine grids of exponential rates, β‑shapes, plateau‑exponential
mixtures and numerous Dirichlet draws).  All seeds are ℓ₁‑normalised and
ranked by the exact surrogate 2·n·max(conv(a,a)).  The best *top_k*
seeds are refined by a mirror‑descent (exponentiated‑gradient) optimiser
that uses **all** active constraints (all shifts whose convolution value
is within a tiny tolerance of the maximum) and an Adam‑style momentum
update.  After the EG stage a brief simulated‑annealing polish is run.
Finally a long EG polish is performed on the current global best.
The best sequence is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_plateau(n: int, offset: int, rate: float) -> np.ndarray:
    """
    Plateau‑exponential shape.
    First `offset` entries are equal, the rest decay geometrically.
    """
    if offset >= n:
        return _uniform(n)
    a = np.empty(n, dtype=np.float64)
    a[:offset] = 1.0
    rest = np.arange(n - offset, dtype=np.float64)
    a[offset:] = np.exp(-rate * rest)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, then re‑normalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_multi(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300_000,
                        eta0: float = 6.0,
                        epsilon: float = 1e-12):
    """
    EG optimiser that averages the sub‑gradient over **all** active
    constraints (all shifts whose convolution value is within `epsilon`
    of the maximum).  Adam‑style momentum is used for a smoother
    trajectory.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose value is essentially the maximum
        active = np.where(conv >= max_val - epsilon)[0]

        # sub‑gradient of the max (averaged over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam update in the log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.004,
                         T_start: float = 0.02,
                         T_end: float = 1e-8):
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence encountered before `deadline`.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence` function.  The routine respects a time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 works well with the FFT based convolution
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previously found solution (if any) for continuity
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a large, diverse set of analytic seeds
    # --------------------------------------------------------------
    seeds = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # exponential decays (fine grid around the previously found optimum)
    for rate in np.linspace(5e-5, 6e-4, 30):
        seeds.append(_exp_decay(n, rate))

    # plateau‑exponential mixtures (offset + decay)
    for offset_frac in [0.0, 0.05, 0.10, 0.20, 0.30]:
        offset = int(offset_frac * n)
        for rate in np.linspace(5e-5, 6e-4, 20):
            seeds.append(_exp_plateau(n, offset, rate))

    # beta‑shapes – dense log‑grid for α,β (very small values are useful)
    alphas = np.logspace(-3, 0.5, 20)
    betas = np.logspace(-3, 0.5, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws with a variety of concentrations
    for conc in [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of uniform and exponential (different mix ratios)
    for mix in [0.02, 0.08, 0.15, 0.30]:
        exp_seq = _exp_decay(n, 2e-4)
        seeds.append(_mix_shapes(_uniform(n), exp_seq, mix))

    # perturbation of previous best (if available)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        # no valid seed – fall back to uniform
        best_seq = _uniform(n)
        globals()["height_sequence_1"] = best_seq.tolist()
        return best_seq.tolist()

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement (EG + short SA)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget for this seed
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        # ---- (a) exponentiated‑gradient with multi‑active set ----------
        seq_eg, val_eg = _exp_gradient_multi(seed_vec.copy(),
                                             n,
                                             sub_deadline,
                                             max_iters=300_000,
                                             eta0=8.0,
                                             epsilon=1e-12)

        # ---- (b) brief simulated annealing ---------------------------
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_eg,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=80_000,
                                                  start_sigma=0.0015,
                                                  T_start=0.015,
                                                  T_end=5e-8)
        else:
            seq_sa, val_sa = seq_eg, val_eg

        # pick the best among the three candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"[Seed {si+1}/{top_k}] New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long‑run polishing (EG then SA)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (multi‑active) ---")
        final_seq, final_val = _exp_gradient_multi(global_best_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=4_000_000,
                                                   eta0=6.0,
                                                   epsilon=1e-12)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=120_000,
                                                          start_sigma=0.0008,
                                                          T_start=0.006,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
