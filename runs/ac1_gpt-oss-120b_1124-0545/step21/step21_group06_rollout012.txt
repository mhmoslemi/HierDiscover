def propose_candidate(seed=123,budget_s=1000):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2

    # Determine n:
    prev = globals().get('height_sequence_1')
    if prev is not None and isinstance(prev,list) and len(prev)>0:
        n = len(prev)
        prev_arr = np.array(prev,dtype=float)
        prev_arr /= prev_arr.sum()
    else:
        n = 2000
        prev_arr = None

    # Define seed vectors:
    seeds = []
    # uniform
    seeds.append(np.full(n, 1.0/n))
    # exponential left with moderate rate (based on n)
    seed_rate = 5.0 / n   # produce r = exp(-rate * i)
    seeds.append(_exp_decay_left(n, seed_rate))
    # exponential right similarly
    seeds.append(_exp_decay_right(n, seed_rate))
    # random Dirichlet low concentration
    for conc in (0.01, 0.02, 0.05):
        seeds.append(_random_dirichlet(n, conc, rng))
    # previous solution if any
    if prev_arr is not None:
        seeds.append(prev_arr)

    # Now for each seed run optimizer
    best_seq = None
    best_val = np.inf

    # We'll limit number of seeds to perhaps up to 5 or 6.
    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline: break
        # ensure normalization and positivity
        a0 = np.maximum(seed_vec, 0.0)
        s = a0.sum()
        if s == 0:
            continue
        a0 = a0 / s

        # run Adam-based smoothing optimizer
        seq_opt, val_opt = _adam_smooth_optimize(a0, n, deadline, rng)
        # compute true evaluation
        try:
            true_val = evaluate_sequence(seq_opt.tolist())
        except Exception as e:
            true_val = None

        if true_val is not None and true_val < best_val:
            best_val = true_val
            best_seq = seq_opt.copy()
            print(f"[Seed {idx}] New best value {best_val:.6f}")

        # also consider the seed directly (maybe already better)
        try:
            val_seed = evaluate_sequence(a0.tolist())
        except:
            val_seed = None
        if val_seed is not None and val_seed < best_val:
            best_val = val_seed
            best_seq = a0.copy()
            print(f"[Seed {idx}] Seed beats best {best_val:.6f}")

    # final polishing:
    if best_seq is not None and time.time() < deadline:
        # final SA with small sigma
        final_seq, final_val = _simulated_annealing(best_seq, n, deadline, rng,
                                                   max_attempts=200_000,
                                                   start_sigma=0.0005,
                                                   T_start=0.005,
                                                   T_end=1e-9)
        # compute true
        try:
            val_sa = evaluate_sequence(final_seq.tolist())
        except:
            val_sa = None
        if val_sa is not None and val_sa < best_val:

"""
Optimiser for the step‑function bound (target ≈ 1.503)

The algorithm follows a two‑phase strategy:

1. **Global seeding** – a few analytically‑motivated distributions
   (uniform, left / right exponential, sparse Dirichlet draws) and the
   previous best solution are normalised to sum = 1 and stored as
   starting points.

2. **Smooth‑max Adam descent** – each seed is refined by minimising a
   *smooth maximum* of the convolution using Adam on log‑softmax
   variables.  The smooth‑max is the log‑sum‑exp approximation
   ``max(x) ≈ shift + log Σ exp(β·(x‑shift))/β`` with a continuation
   schedule that raises β from 1 to 10⁵.  Because the variables are kept
   in log‑softmax space the constraints “non‑negative & ℓ₁‑normalised”
   are satisfied automatically.

3. **Final polishing** – a short simulated‑annealing pass explores
   nearby points that may improve the nonsmooth objective.

All work respects the 1000 s budget; the best sequence found is stored
in the global ``height_sequence_1`` variable for reuse across runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities – FFT‑based full convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a unit‑ℓ₁ vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families used as seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n entries (ℓ₁‑normalised)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) (left‑hand exponential, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i)) (right‑hand exponential, normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth‑max optimiser – Adam on log‑softmax variables
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          beta_schedule: list[float] | None = None,
                          lr: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate
        L(v) = 2·n·(shift + log Σ exp(β·(conv‑shift))/β)
    where v are log‑softmax variables (a = softmax(v)).
    Returns the best sequence found and its surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    # log‑softmax variables
    v = np.log(start_seq + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # iterator over beta values – we increase β every few hundred steps
    beta_idx = 0
    beta = beta_schedule[beta_idx]
    steps_per_beta = max_iters // len(beta_schedule) + 1

    while time.time() < deadline and t < max_iters:
        # optional β‑schedule update
        if t // steps_per_beta > beta_idx and beta_idx + 1 < len(beta_schedule):
            beta_idx += 1
            beta = beta_schedule[beta_idx]

        # current probability vector
        a = _softmax(v)

        # full convolution a * a
        conv = _conv_fft(a, a)
        shift = conv.max()                     # numeric stabiliser

        # smooth‑max weights (log‑sum‑exp gradient)
        w = np.exp(beta * (conv - shift))
        sum_w = w.sum()
        if sum_w == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / sum_w

        # gradient w.r.t. a (see derivation in the description)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule to obtain ∂L/∂v (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        # learning‑rate adaptation (simple 1/√t decay)
        lr_t = lr / math.sqrt(t)

        v -= lr_t * m_hat / (np.sqrt(v2_hat) + eps)

        # track best surrogate value
        cur_val = float(2.0 * n * shift)      # exact surrogate = 2·n·max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny perturbations, keeps ℓ₁‑norm)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA that perturbs the current solution with Gaussian noise,
    accepts worsening moves with a temperature schedule and returns the
    best point seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
