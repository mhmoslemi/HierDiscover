"""
Propose a high‑quality step‑function for the leaderboard.

Algorithm
---------
1.  Initialise from the previous best sequence (global ``height_sequence_1``)
    or from a simple exponential decay.  The vector is normalised to a
    probability distribution (sum = 1).

2.  Smooth optimisation: minimise a log‑sum‑exp surrogate of the
    ``max(conv(a,a))`` objective using SciPy's L‑BFGS‑B.  The surrogate
    becomes tighter as the temperature parameter ``beta`` grows.
    A schedule of increasing ``beta`` values is used; each stage performs
    many L‑BFGS iterations (up to a few thousand) while time permits.

3.  Intensive exponentiated‑gradient (mirror‑descent) polishing.  The
    exact sub‑gradient of the active convolution entry is used.  The
    step size decays as ``eta0 / sqrt(t)`` with a relatively large
    ``eta0`` to explore the landscape.  This run consumes the bulk of
    the time budget.

4.  Optional short random‑restart refinements: a few noisy copies of the
    current best are polished for a brief period; any improvement is kept.

5.  The final vector is evaluated with the provided ``evaluate_sequence``
    function, printed and stored in the global variable
    ``height_sequence_1`` for possible continuation runs.

All helper functions are defined at module level (no closures or lambdas)
and only standard scientific libraries are used.
"""

import time
import math
import numpy as np

# optional SciPy import – used only for the L‑BFGS smoothing phase
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (used for a cheap initial guess if no previous solution)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to stay on the simplex
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) of the smoothed objective:
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c                     # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine focuses on a long L‑BFGS smoothing phase followed by an
    extensive exponentiated‑gradient polishing, using the previous best
    solution (if any) as a starting point.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from a previously discovered sequence if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_seq = arr
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        # fall back to a simple exponential decay and sort descending
        init_seq = _exp_decay(n, rate=0.001)
        init_seq = np.sort(init_seq)[::-1]

    # Normalise to a probability vector (sum = 1) – invariant for the objective
    init_seq = np.maximum(init_seq, 0.0)
    init_seq = init_seq / init_seq.sum()

    # ------------------------------------------------------------------
    # Phase 1 – smoothing via L‑BFGS on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    v = np.log(init_seq + 1e-30)   # log‑softmax variables
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]

    # allocate at most 20 % of the total budget to the smoothing stage,
    # but stop early if the remaining time is needed for polishing.
    smoothing_deadline = deadline - 300.0   # keep ~5 min for the final EG + restarts
    for beta in beta_schedule:
        if time.time() > smoothing_deadline:

        # maxiter is chosen heuristically; 3000–4000 L‑BFGS steps are cheap

                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': max_iter_per_beta,
                                             'disp': False,
                                             'ftol': 1e-12})
                v = res.x
            except Exception:   # pragma: no cover
                # fall back silently – a later EG will still improve things
                pass
        else:
            # If SciPy is unavailable, skip this stage.
            break

    # Convert back to probability vector
    cur_seq = _softmax(v)
    cur_val = _objective_surrogate(cur_seq)
    print(f"[After smoothing] surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive exponentiated‑gradient polishing (the bulk of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Main EG polishing ---")
        cur_seq, cur_val = _exp_gradient_opt(cur_seq, n, deadline,
                                             max_iters=5_000_000,
                                             eta0=10.0)
        print(f"[After EG] surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short random‑restart refinements (if any time left)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:

        if s == 0.0:

        noisy = noisy / s

        sub_deadline = time.time() + 25.0   # give each restart ~25 s

                                           max_iters=2_000_000,
                                           eta0=10.0)
        if val_rr < cur_val:
            cur_seq, cur_val = seq_rr, val_rr
            print(f"[Random restart] improved surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny permutation tweak (very fast)
    # ------------------------------------------------------------------

                                              n,

                                              rng,
                                              max_iters=80_000)
        if val_last < cur_val:
            cur_seq, cur_val = seq_last, val_last
            print(f"[Final permutation] improved surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate the *true* objective via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:

        print(f"\n=== Finished ===\nSurrogate ≈ {cur_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = cur_seq.tolist()
    return cur_seq.tolist()
