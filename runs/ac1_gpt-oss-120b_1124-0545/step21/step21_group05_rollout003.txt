"""
Optimise a non‑negative step‑function (vector a) to minimise

        2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation function `evaluate_sequence` normalises the sum automatically,
so we search among vectors that sum to 1 and minimise the surrogate

        S(a) = 2·n·max(conv(a,a))

which is exactly the value returned by `evaluate_sequence` for a normalised
vector.

The optimiser works in four stages:

1. **Massive seed generation** – analytic families (uniform, triangular,
   sine‑window, Gaussian, shifted Gaussian, exponential, power‑law,
   beta‑shapes) and random Dirichlet draws are created for a relatively
   large dimension (n = 8000).  The best few seeds are kept.

2. **Short Adam polishing** – each kept seed is refined with a mirror‑descent
   (Adam) optimiser that minimises a smooth approximation of the max‑convolution
   (log‑sum‑exp with an increasing temperature).  The optimisation stays on the
   probability simplex by working in log‑space and applying the soft‑max.

3. **Global long‑run polishing** – the current best vector is further polished
   with a long Adam run (large β → true max) using the remaining time budget.

4. **Final tiny permutation hill‑climb** – a brief simulated‑annealing style
   permutation/perturbation search is performed to squeeze out any remaining
   improvement.

All heavy work uses FFT‑based convolutions (O(n log n)).  The total run time
is kept well below the 1000 s limit; the final sequence is stored in the
global variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT‑based convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Numerics
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector with sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _norm_and_surrogate(vec: np.ndarray):
    """Clip to ≥0, normalise to sum = 1 and return (vector, surrogate)."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return None, np.inf
    vec = vec / s
    return vec, _objective_surrogate(vec)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   beta_start: float = 1.0,
                   beta_end: float = 1e6,
                   lr: float = 0.12,
                   eps: float = 1e-12):
    """
    Adam optimisation in log‑probability space.
    Objective: 2·n·log‑sum‑exp_{β}(conv)   (β→∞ → true max).
    Returns the best vector encountered (ℓ₁‑normalised) and its surrogate.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    beta0 = float(beta_start)
    beta1 = float(beta_end)
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current vector
        a = _softmax(log_a)

        # convolution
        conv = _conv_fft(a, a)                 # length = 2n‑1
        shift = np.max(conv)

        # schedule for β (exponential from beta_start to beta_end)
        beta = beta0 * (beta1 / beta0) ** (it / max_iters)

        # soft‑max weights over the convolution entries
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp                  # shape = 2n‑1

        # gradient w.r.t. a  (see comment in the description)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule for log‑softmax parameterisation
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_log
        v = 0.999 * v + 0.001 * (grad_log ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # current surrogate (exact max, cheap)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Light permutation‑based hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly swap two entries or apply a small
    Gaussian perturbation, accepting worsening moves with a linear temperature.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule – from a small fraction of cur_val downwards
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:          # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                            # tiny Gaussian noise
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Interpolation helper for a previous solution of different length
# ----------------------------------------------------------------------
def _interpolate_to_n(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length n."""
    if old.shape[0] == n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.shape[0])
    x_new = np.linspace(0.0, 1.0, n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        return _uniform(n)
    return new / s


# ----------------------------------------------------------------------
# Main entry point used by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine builds a large set of analytic seeds,
    keeps the best few, refines each with a short Adam + hill‑climb phase,
    and finally runs a long Adam polishing on the current global best.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Problem dimension – 8000 provides a good balance of expressivity
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load previous solution (if any) and interpolate it to the current n
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] > 0:
                prev_arr = _interpolate_to_n(arr, n)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 45):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.04, 0.30, 25):
        seeds.append(_gaussian(n, sigma_frac))

    for mu_frac in np.linspace(0.15, 0.85, 12):
        for sigma_frac in np.linspace(0.04, 0.25, 10):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    for rate in np.logspace(-5, 0.4, 60):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    for p in np.linspace(0.0, 6.0, 28):
        seeds.append(_power(n, p))

    alphas = np.geomspace(0.1, 10.0, 12)
    betas = np.geomspace(0.1, 10.0, 12)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in conc_vals:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include the previous best (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec, val = _norm_and_surrogate(vec)
        if vec is None:
            continue
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of each top seed (short Adam + hill‑climb)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # we reserve a small chunk for the final polishing later
        reserve = 150.0
        share = max(2.0, (remaining - reserve) / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] allocated time ≈ {share:.1f}s")

        # a) deterministic ordering – descending (often improves surrogate)
        seq_ordered = np.sort(seed_vec)[::-1]

        # b) short permutation hill‑climb
        stage_deadline = time.time() + share * 0.30
        seq_perm, val_perm = _perm_hill_climb(seq_ordered, n,
                                              stage_deadline,
                                              rng,
                                              max_iters=120_000)

        # c) short Adam polishing (moderate β schedule)
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_optimize(seq_perm, n,
                                                sub_deadline,
                                                max_iters=250_000,
                                                beta_start=1.0,
                                                beta_end=3e4,
                                                lr=0.12)
        else:
            seq_adam, val_adam = seq_perm, val_perm

        # pick the best local variant
        candidates = [
            (seq_ordered, _objective_surrogate(seq_ordered)),
            (seq_perm, val_perm),
            (seq_adam, val_adam),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])
        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best ← {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long Adam polishing (uses almost all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_optimize(global_best_seq, n,
                                              deadline,
                                              max_iters=2_500_000,
                                              beta_start=5.0,
                                              beta_end=1e6,
                                              lr=0.10)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short permutation tweak (few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 22.0
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq, global_best_val = seq_last, val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function (for logging only)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
