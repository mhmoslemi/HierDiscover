"""
Hybrid search for a step‑function that minimises
evaluate_sequence().  The method builds many random / structured
initialisations, keeps the most promising few and then refines each of
them by a long exponentiated‑gradient (mirror‑descent) run on the true
max‑convolution objective.  Finally a short L‑BFGS smoothing on the
log‑sum‑exp surrogate is applied.  All phases respect the total 1000 s
budget and the best sequence is stored in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# optional SciPy – used for L‑BFGS and differential evolution
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# --------------------------------------------------------------
# Low‑level numeric helpers
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Smooth surrogate: 2·n·max(conv(seq, seq))  (seq already normalised).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective_exact(seq: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence() for a normalised seq."""
    # evaluate_sequence expects a Python list, but we can compute
    # the same formula directly (avoiding the overhead of a Python call).
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    return float(2.0 * n * max_b / (sum_a ** 2))

# Seed factories (all ℓ₁‑normalised)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k distinct indices and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 16) -> np.ndarray:
    """
    Gaussian mixture with K components on the discrete grid {0,…,n‑1}.
    params = [w_0…w_{K‑1}, μ_0…μ_{K‑1}, σ_0…σ_{K‑1}]
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """Fourier‑softmax: a_i ∝ exp(θ·φ(i))."""
    logits = basis @ theta
    return _softmax(logits)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 4.0) -> (np.ndarray, float):

    Exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# L‑BFGS on the smooth log‑sum‑exp surrogate
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 25000):
    """L‑BFGS‑B on the smooth surrogate (variables = log‑softmax)."""
    if _opt is None:
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1_000_000.0, 2_000_000.0, 5_000_000.0,
                         10_000_000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 199, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The best sequence found is stored in the global
    variable ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it the same as the reference run (4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds (all already ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    # Dirichlet draws with various concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse Dirichlet – supports of size 200–600
    for k in (200, 300, 500):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # Gaussian‑mixture seeds (more components than the reference)
    K_mix = 16
    for _ in range(6):
        params = np.concatenate([
            rng.random(K_mix),                     # raw weights
            rng.random(K_mix),                     # means
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n, K=K_mix))

    # Fourier‑softmax seeds (40 basis functions)
    D_fourier = 40
    i_grid = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis = []
    for k in range(1, D_fourier // 2 + 1):
        basis.append(np.sin(2.0 * np.pi * k * i_grid))
        basis.append(np.cos(2.0 * np.pi * k * i_grid))
    basis = np.column_stack(basis)   # (n, D_fourier)

    for _ in range(6):
        theta = rng.normal(scale=1.0, size=D_fourier)
        seeds.append(_fourier_softmax(theta, basis))

    # Small random perturbations of the uniform seed – helpful for escaping
    for _ in range(3):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate the pool (fast surrogate) and keep the top few
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    print("\n--- Evaluating initial seeds (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # Keep the 5 best distinct seeds for intensive polishing
    # (simple ranking – we redo the surrogate evaluation to sort)
    surrogate_vals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    surrogate_vals.sort(key=lambda x: x[1])
    polish_seeds = [vec for vec, _ in surrogate_vals[:5]]

    # ------------------------------------------------------------------
    # 3️⃣  Intensive mirror‑descent polishing of the top seeds
    # ------------------------------------------------------------------
    print("\n--- Mirror‑descent polishing of top seeds ---")
    per_seed_deadline = deadline - 1.0   # leave a little slack for final steps
    for s_idx, seed_vec in enumerate(polish_seeds):
        # allocate a fair share of the remaining time to each seed
        time_left = per_seed_deadline - time.time()
        if time_left <= 0:
            break
        # we allow up to 600 k iterations per seed (adjusted to time)
        max_iters = int(min(600_000,
                            max(100_000,
                                time_left / (deadline - start_time) * 2_000_000)))
        seq_eg, val_eg = _exp_gradient_opt(
            seed_vec, n, deadline, max_iters=max_iters, eta0=5.0)
        print(f"[EG seed {s_idx+1}] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"   >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  L‑BFGS smoothing on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- L‑BFGS smoothing of the current best ---")
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45
        seq_lbfgs = _lbfgs_smooth_optimize(
            best_seq, n, sub_deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                           200.0, 500.0, 1000.0,
                           2000.0, 5000.0,
                           20000.0, 50000.0,
                           100000.0, 200000.0, 500000.0,
                           1_000_000.0, 2_000_000.0, 5_000_000.0,
                           10_000_000.0],
            max_iter_per_beta=30000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"   >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final polishing – short EG runs from stochastic perturbations
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        candidate = _perturb(best_seq, rng, sigma=0.0012)
        short_deadline = time.time() + 0.45
        cand_seq, cand_val = _exp_gradient_opt(
            candidate, n,
            short_deadline,
            max_iters=150_000,
            eta0=3.0,
        )
        print(f"[Pert‑EG] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"   >>> New best after perturb‑EG = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
