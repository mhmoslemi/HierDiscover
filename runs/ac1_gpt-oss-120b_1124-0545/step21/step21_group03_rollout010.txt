def _adam_smooth_opt(start_seq, deadline):
    n = len(start_seq)
    a = start_seq.copy()
    u = np.log(a + 1e-30)   # logits
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    beta_schedule = [0.5, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_schedule:
        # choose iterations per beta based on remaining time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # estimate iterations we can afford: e.g., each iter ~estim_time = 0.0015 sec for n=4000 (two FFT).
        # compute max_iters = min(5000, int(remaining / 0.0015))
        est_iter_time = 0.0015   # maybe adjust?
        max_iters = int(min(5000, remaining / est_iter_time))
        for _ in range(max_iters):
            if time.time() >= deadline:
                break
            t += 1
            # compute a from logits
            a = np.exp(u)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(u, 1.0/len(u))
                a_sum = 1.0
            a = a / a_sum
            # conv
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            # grad w.r.t a:
            full = _conv_fft(a[::-1], w)
            grad_a = 2.0 * full[n-1:n-1+n]   # length n
            # gradient w.r.t logits:
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            # step size schedule
            lr = 0.05 / (np.sqrt(t) + 1e-8)  # could constant maybe
            u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
            # evaluate surrogate
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    # final normalize best_seq sum to 1
    best_seq = best_seq / best_seq.sum()
    return best_seq, best_val

def _local_swap_search(start_seq, deadline, max_iters=500000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    # ensure normalized to sum 1
    cur_seq = cur_seq / cur_seq.sum()
    cur_max = np.max(_conv_fft(cur_seq, cur_seq))
    cur_val = 2.0 * n * cur_max
    best_seq = cur_seq.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline and it < max_iters:
        # choose random i with positive weight
        i = rng.integers(0, n)
        if cur_seq[i] < 1e-12:
            continue
        # choose random j (different)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # propose moving fraction epsilon = cur_seq[i] * alpha where alpha random in [0,0.2]
        alpha = rng.random() * 0.2
        delta = cur_seq[i] * alpha
        ns = cur_seq.copy()
        ns[i] -= delta
        ns[j] += delta
        # renormalize (should still sum 1)
        # compute conv max efficiently: but must recompute
        conv = _conv_fft(ns, ns)
        new_max = np.max(conv)
        new_val = 2.0 * n * new_max
        # accept if better, or with probability if not (simple Metropolis)
        if new_val < cur_val:
            cur_seq = ns
            cur_val = new_val
            if new_val < best_val:
                best_seq = ns.copy()
                best_val = new_val
        else:
            # occasional uphill moves using temperature schedule
            # simple linear cooling
            T = (deadline - time.time()) / (deadline - start_time)  # temperature descending to 0
            # Actually T goes from ~1 to 0 as time passes.
            if rng.random() < math.exp(-(new_val - cur_val) / (0.01 * T + 1e-12)):
                cur_seq = ns
                cur_val = new_val
        it += 1
    return best_seq, best_val

Optimise a non‑negative step‑function to minimise the evaluate_sequence objective.

The optimiser proceeds in three phases:

1. **Broad seed generation** – analytic families (uniform, exponential,
   symmetric exponential, power‑law, Gaussian, sinusoidal, beta, Dirichlet,
   piecewise‑linear) are instantiated with many parameter choices, and the
   best few are kept.

2. **Smooth Adam descent** – each retained seed is refined by minimizing a
   log‑sum‑exp surrogate of ``2·n·max(conv)``.  The surrogate is differentiable
   and its gradient is obtained analytically; an Adam optimiser in the
   log‑softmax (probability) space is used.  The smoothing parameter β is
   increased gradually so that the optimisation first sees a smooth landscape
   and finally focuses on the true max‑convolution.

3. **Fine‑grained permutation / mass‑transfer search** – a short simulated‑annealing
   stage performs random transfers of a small amount of mass between two
   positions, accepting downhill moves and occasional uphill moves to escape
   local minima.

All phases respect the global time budget; the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) under ℓ₁‑normalisation."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma: float) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _half_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / (2.0 * n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:

    return a / s

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None) -> tuple[np.ndarray, float]:
    """
    Minimises the log‑sum‑exp surrogate of 2·n·max(conv) via Adam.
    The iterate lives in the log‑softmax (probability) space.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0
        ]

    n = len(start_seq)
    # logits (log‑softmax before normalisation)
    u = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    iter_count = 0
    beta_idx = 0
    beta = beta_schedule[beta_idx]

    while time.time() < deadline:
        # Every few thousand iterations we move to a tighter beta
        if iter_count > 0 and iter_count % 5000 == 0:
            beta_idx = min(beta_idx + 1, len(beta_schedule) - 1)
            beta = beta_schedule[beta_idx]

        iter_count += 1

        # soft‑max to obtain a probability vector
        u_shift = u - np.max(u)               # stabilise exponentials
        a = np.exp(u_shift)
        a /= a.sum()

        # full convolution a*a
        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # log‑sum‑exp weights (smooth max)
        shift = max_c                     # prevents overflow
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp            # shape = 2n‑1

        # gradient w.r.t. a (derived analytically)
        full = _conv_fft(a[::-1], w)      # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule to obtain gradient w.r.t. logits u
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** iter_count)
        v_hat = v / (1 - beta2 ** iter_count)

        # step size that slowly decays
        lr = 0.04 / (np.sqrt(iter_count) + 1e-8)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation / mass‑transfer simulated annealing.
# ----------------------------------------------------------------------
def _local_mass_transfer(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small fraction of probability mass from one index to another.
    Accepts downhill moves and occasional uphill moves (linear cooling).
    """
    n = len(start_seq)
    rng = np.random.default_rng()

    cur = start_seq.copy()
    cur = cur / cur.sum()
    cur_max = np.max(_conv_fft(cur, cur))
    cur_val = 2.0 * n * cur_max

    best = cur.copy()
    best_val = cur_val

    start_time = time.time()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # pick source with non‑negligible mass
        i = rng.integers(0, n)
        if cur[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of source mass
        alpha = rng.random() * 0.3          # up to 30 % of the source mass
        delta = cur[i] * alpha
        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta

        # evaluate
        conv = _conv_fft(ns, ns)
        new_max = np.max(conv)
        new_val = 2.0 * n * new_max

        # linear cooling schedule
        remaining = deadline - time.time()
        total = deadline - start_time
        temperature = max(1e-12, remaining / total)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (0.01 * temperature)):
            cur = ns
            cur_val = new_val
            if new_val < best_val:
                best = ns.copy()
                best_val = new_val

    return best, best_val

    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(0, n)
        if cur[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (≤30 %) of the source mass
        alpha = rng.random() * 0.3
        delta = cur[i] * alpha
        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta

        new_val = 2.0 * n * np.max(_conv_fft(ns, ns))

        # linear cooling
        remaining = deadline - time.time()
        total = deadline - start_time
        temperature = max(1e-12, remaining / total)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (0.01 * temperature)):
            cur = ns
            cur_val = new_val
            if new_val < best_val:
                best = ns.copy()
                best_val = new_val

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a large pool of analytically generated seeds,
    a smooth Adam descent on a log‑sum‑exp surrogate, and a final mass‑transfer
    annealing phase.  The best sequence found is stored in the global variable
    ``height_sequence_1`` and also returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – moderate length gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if present (helps continuation runs)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Populate a diverse pool of seed vectors (all ℓ₁‑normalised)
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sym_exp_decay(n, rate=0.0015))

    # exponential decays
    for rate in np.logspace(-4, -2, num=14):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law families
    for p in np.linspace(0.3, 3.0, num=14):
        seeds.append(_power(n, p))

    # Gaussian bumps of varying width
    for sigma in np.linspace(200.0, 2000.0, num=12):
        seeds.append(_gaussian(n, sigma))

    # sinusoidal / cosine shapes
    seeds.append(_half_sine(n))
    seeds.append(_cosine_bump(n))

    # beta distributions (grid)
    alphas = np.linspace(0.3, 3.0, num=7)
    betas = np.linspace(0.3, 3.0, num=7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with diverse concentration
    for conc in (0.02, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear random shapes
    for nb in (3, 6, 9):
        for _ in range(5):
            seeds.append(_piecewise_linear(n, nb, rng))

    # previous best and a handful of its permutations
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Refine the top seeds with smooth Adam descent
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate among seeds = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - idx))
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx+1}] refinement budget ≈ {share:.1f}s")
        seq_opt, val_opt = _adam_smooth_opt(seed_vec, sub_deadline)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final polishing with a short mass‑transfer annealing stage
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final mass‑transfer annealing (remaining time) ---")
        seq_last, val_last = _local_mass_transfer(global_best_seq, deadline)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final anneal] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Evaluate with the provided ``evaluate_sequence`` and store result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
