"""
Propose a candidate step‑function that (hopefully) improves the current bound.

Strategy
--------
1. Build a diverse collection of simple “seed” distributions (uniform,
   power‑laws, beta draws, arcsine, triangular, Dirichlet samples …).  All
   seeds are normalised to L¹ = 1, because the evaluation function is
   homogeneous.

2. Pick the few best seeds (by the exact surrogate 2·n·max(conv)) and
   refine each of them with a fast *mirror‑Adam* optimiser.
   - The variables live in the log‑softmax space so that the positivity
     constraint is automatically satisfied.
   - At every iteration the exact sub‑gradient of the max‑convolution is
     used (the active index of the convolution).
   - Adam’s momentum allows larger steps than the plain exponentiated
     gradient used previously and usually reaches a better local minimum
     for a given time budget.

3. After the Adam phase a tiny random local search (move a tiny amount of
   mass from one coordinate to another) is performed to polish the
   solution.

The whole routine respects the global ``budget_s`` seconds limit and
stores the best sequence it has ever seen in the global variable
``height_sequence_1``.  The returned list can be used directly with the
provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  (sum(seq) is assumed =1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_seed(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p  (monotone increasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    a /= a.sum()
    return a


def _inv_power_seed(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^p  (monotone decreasing)."""
    i = np.arange(n, 0, -1, dtype=np.float64)
    a = np.power(i, p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (mirror‑descent in log‑softmax)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   lr0: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-12,
                   max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑Adam on the true max‑convolution objective.
    Returns the best sequence (as a probability vector) and its surrogate value.
    """
    # log‑softmax variables
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # current probabilities
        a = _softmax(v)

        # exact convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]               # Jacobian of a_i a_{k-i}

        # gradient w.r.t. log‑softmax variables (mirror map)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(v2_hat) + eps)
        v = v - lr * m_hat

        # occasional exact evaluation
        if it % 5000 == 0 or time.time() >= deadline:
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    # final check
    final_seq = _softmax(v)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random local refinement (mass transfers)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 120_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """Move a tiny amount of probability mass from one index to another."""
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    prev = globals().get("height_sequence_1")

    else:
