        if prev_arr.shape[0] == n:
            a = prev_arr.copy()
        else:
            # downsample or upsample
            if prev_arr.shape[0] > n:
                # downsample by averaging blocks
                factor = prev_arr.shape[0] // n
                a = prev_arr[:factor*n].reshape(n, factor).mean(axis=1)
            else:
                # upsample via linear interpolation
                a = np.interp(np.linspace(0, len(prev_arr)-1, n), np.arange(len(prev_arr)), prev_arr)
    else:
        # start from triangular shape
        a = _triangular(n)

    a = np.maximum(a, 0.0)

"""
search‑algorithm – alternating LP + exponentiated‑gradient polishing

The optimizer works on a non‑negative step‑function a = (a₀,…,a_{n‑1}) with
ℓ₁‑norm 1.  Starting from a simple seed (triangular or the previous best)
the following two‑phase loop is performed:

1️⃣  LP step – for the current a compute M = max((a∗a)).  Solve the linear
    program

        maximise   Σ bⱼ
        subject to  (a∗b)[k] ≤ M   (k = 0…2n‑2)
                    bⱼ ≥ 0 .

    The feasible set consists of all non‑negative vectors whose convolution
    with a never exceeds the present max‑value M.  The optimizer therefore
    pushes mass to indices where a is small.  The solution b is normalised
    to a probability vector g = b / Σb and mixed with a:

        a ← (1 – t)·a + t·g   (t ≈ 0.25).

    Repeating the LP step drives a towards a shape where the
    self‑convolution is uniformly low, i.e. it reduces the surrogate
    2·n·max(conv(a,a)).

2️⃣  After a few LP refinements (the budget usually permits ≤ 10–12 of them)
    a long mirror‑descent (exponentiated‑gradient) optimisation is run on the
    exact max‑convolution.  This “polishing” phase uses the sub‑gradient that
    corresponds to the (single) active index of the maximum and converges
    quickly because the LP already provides a good initial guess.

A very light simulated‑annealing pass is added at the end to escape any
remaining tiny local minima.  The final sequence is evaluated with the
provided ``evaluate_sequence`` function and stored in the global variable
``height_sequence_1``.  The whole routine respects the supplied time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP and sparse matrices)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """stable soft‑max, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vector assumed ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """exact objective for sum(seq)=1: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (used for an inexpensive seed)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_simple(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 5.0,
                             single_active: bool = True):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
       Works in log‑space; returns the best sequence seen."""
    a = start_seq.copy()
    # avoid exact zeros – log(0) is handled by a tiny epsilon
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if single_active:
            k = int(conv.argmax())
            active = np.array([k], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max (vectorised)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]

        if active.size:
            g = 2.0 * g / active.size   # averaged exact sub‑gradient

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing (light polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """Gaussian perturbation SA with linear temperature schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (maximise Σb subject to a∗b ≤ M)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray):
    """
    Build a sparse matrix A of shape (2n‑1, n) such that A @ b == a∗b.
    For each offset d (0…n‑1) the diagonal –d contains the constant a[d].
    """
    n = len(a)
    # offsets are 0, -1, -2, … , -(n‑1)
    offsets = -np.arange(n, dtype=np.int64)
    # each diagonal has the same constant value a[d] repeated n times
    data = np.tile(a[:, np.newaxis], (1, n)).astype(np.float64)
    return _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')


def _lp_maximize_sum(a: np.ndarray, M_val: float):
    """Solve  max Σb  s.t. (a∗b) ≤ M_val , b ≥ 0 .  Returns b (or None)."""
    if _opt is None or _sp is None:
        return None
    n = len(a)
    A = _build_conv_matrix(a)                     # (2n‑1) × n
    ub = np.full(2 * n - 1, M_val * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)             # maximise Σb  <=>  minimise -Σb
    bounds = [(0, None)] * n
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    except Exception as e:  # pragma: no cover
        print("[LP] linprog exception:", e)
        return None
    if not res.success:
        # Occasionally the LP can be infeasible due to rounding; ignore.
        print("[LP] linprog failed:", res.message)
        return None
    return res.x

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a list) and stores it in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size – 2000 gives a fast LP while still allowing a low bound
    # --------------------------------------------------------------
    n = 2000

    # --------------------------------------------------------------
    # Initialise from the previous best if possible, otherwise use a triangular seed
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape[0] == n:
            a = arr.copy()
        else:
            # simple down‑sample / up‑sample to the required length
            if arr.shape[0] > n:
                factor = arr.shape[0] // n
                a = arr[:factor * n].reshape(n, factor).mean(axis=1)
            else:
                a = np.interp(np.linspace(0, arr.shape[0] - 1, n),
                              np.arange(arr.shape[0]), arr)
    else:
        a = _triangular(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = _uniform(n)
    a = a / a.sum()

    print(f"Initial surrogate = {_objective_surrogate(a):.6f}")

    # --------------------------------------------------------------
    # 1. Alternating LP loop
    # --------------------------------------------------------------
    max_lp_iters = 12
    t = 0.25                     # mixing weight
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_lp_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()

        b = _lp_maximize_sum(a, M)
        if b is None or b.sum() == 0.0:
            print("[LP] No feasible solution – terminating LP phase.")
            break

        g = b / b.sum()          # normalise to a probability vector
        a = (1.0 - t) * a + t * g
        a = a / a.sum()           # enforce exact ℓ₁‑norm 1

        val = _objective_surrogate(a)
        print(f"[LP] iter {it:2d}: surrogate = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f" *** new best after LP (iter {it})")

    # --------------------------------------------------------------
    # 2. Long exponentiated‑gradient polishing (single‑active)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allow a generous number of EG iterations if time permits

        print("\n--- EG polishing (single‑active) ---")
        best_seq, best_val = _exp_gradient_opt_simple(best_seq, n, deadline,
                                                     max_iters=max_iters,
                                                     eta0=6.0,
                                                     single_active=True)

    # --------------------------------------------------------------
    # 3. Very brief SA polishing (helps escape minute local minima)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- SA polishing ---")
        best_seq, best_val = _simulated_annealing(best_seq, n, deadline,
                                                  rng,
                                                  max_attempts=80_000,
                                                  start_sigma=0.001,
                                                  T_start=5e-3,
                                                  T_end=1e-9)

    # --------------------------------------------------------------
    # 4. Final evaluation via the provided function
    # --------------------------------------------------------------
    seq_list = best_seq.tolist()
    true_val = None
    try:

        print(f"Surrogate (ℓ₁‑norm=1) ≈ {best_val:.6f}")
        print(f"evaluate_sequence   = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)

    # store for future runs
    globals()["height_sequence_1"] = seq_list
    return seq_list
