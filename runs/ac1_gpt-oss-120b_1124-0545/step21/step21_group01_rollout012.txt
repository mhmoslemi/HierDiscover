"""
Propose a candidate step‑function that improves the bound.
The algorithm is based on many random Dirichlet seeds,
followed by intensive exponentiated‑gradient (mirror‑descent)
iterations on the exact max‑convolution objective,
and a final lightweight random local refinement.
All heavy work uses a fast FFT‑based convolution; the final
value is evaluated with the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).   seq must sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution (integer heights)."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly minimising
    max(conv(a,a)).  Returns the best sequence found and its surrogate.
    """
    a = start_seq.copy()
    # work in log‑softmax space
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search (tiny hill‑climb)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of the mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search runs for at most ``budget_s`` seconds and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse pool of seeds (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))

    # a few Beta‑shaped draws
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, alpha, beta))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # small perturbations of the uniform distribution
    for sigma in (0.005, 0.01):
        seeds.append(_perturb(_uniform(n), rng, sigma=sigma))

    # ------------------------------------------------------------------
    # 1️⃣  Evaluate seeds and keep the best (surrogate objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short exponentiated‑gradient runs from each seed
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        quick_phase_end = time.time() + remaining * 0.20  # ~20% of remaining time
        print("\n--- Short EG refinements (per‑seed) ---")
        for idx, seed_vec in enumerate(seeds):
            if time.time() > quick_phase_end:
                break
            per_deadline = time.time() + min(30.0,
                                            (quick_phase_end - time.time()) / max(1, len(seeds) - idx))
            seq_opt, val_opt = _exp_gradient_opt(seed_vec / seed_vec.sum(),
                                                 n,
                                                 per_deadline,
                                                 max_iters=120_000,
                                                 eta0=5.0)
            print(f" EG seed #{idx+1:02d}: best surrogate = {val_opt:.6f}")
            if val_opt < best_val:
                best_val = val_opt
                best_seq = seq_opt.copy()
        print(f"[After short EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long exponentiated‑gradient polishing on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        long_deadline = time.time() + remaining * 0.70  # most of the rest
        # estimate iteration budget linearly from remaining wall‑clock time
        # (upper bounded to keep the loop manageable)
        max_iters = int(2_500_000 *
                        (long_deadline - time.time()) /
                        (deadline - start_time))
        max_iters = max(200_000, min(max_iters, 5_000_000))
        print("\n--- Long EG polishing ---")
        print(f"  allocated iterations approx.: {max_iters}")
        best_seq, best_val = _exp_gradient_opt(best_seq, n,
                                               long_deadline,
                                               max_iters=max_iters,
                                               eta0=5.0)
        print(f"[Long EG] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n,
                                                    deadline,
                                                    attempts=200_000,
                                                    sigma=0.02)
        print(f"[Local] best surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
