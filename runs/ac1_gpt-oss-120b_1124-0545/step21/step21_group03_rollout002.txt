"""
Improved optimiser for the step‑function “max‑convolution” problem.

Main ideas
----------
1. Build a diverse pool of analytic and random seeds (uniform, exponential,
   symmetric exponential, triangular, Gaussian, power‑law, beta, Dirichlet,
   piecewise‑linear, plus permutations of the previous best).

2. Keep the top few seeds according to the exact surrogate objective  
   (surrogate = 2·n·max(conv(seq,seq)) which equals the true evaluate_sequence
   value when the sequence is ℓ₁‑normalised).

3. For each top seed run a *continuation* optimisation based on a smooth
   approximation of the max‑operator.  The smooth max is realised by a
   log‑sum‑exp with an annealing schedule of the temperature (β).  The
   optimisation is performed in the log‑softmax (probability) space using
   SciPy’s L‑BFGS‑B (if available); otherwise we fall back to a long
   exponentiated‑gradient (mirror‑descent) run.

4. After the smoothing stage a final, very long exponentiated‑gradient run
   (exact sub‑gradient of the active diagonal) is executed on the current
   best sequence, followed by a short permutation‑based simulated annealing
   to escape possible ordering traps.

All work respects the total time budget (default 1000 s) and prints a short
log of progress.  The best sequence found is stored in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for L‑BFGS‑B smoothing (if present)
# ----------------------------------------------------------------------
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Exact max‑convolution value for a probability vector."""
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for an ℓ₁‑normalised vector:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    return 2.0 * n * _max_conv(seq)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = centre + 1 - np.abs(i - centre)
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in the log‑softmax domain.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    # small epsilon to avoid log(0)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # renormalise (softmax)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimisation via L‑BFGS‑B (if SciPy is present)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    minimise a smooth approximation of max(conv) using L‑BFGS‑B.
    Operates in the log‑softmax domain.
    """
    if _opt is None:
        # fallback – return the start vector unchanged
        return start_vec

    if beta_schedule is None:
        # a reasonably fine annealing schedule, ending at a very high β
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            2_000.0, 5_000.0, 10_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0, 10_000_000.0,
            20_000_000.0, 50_000_000.0, 100_000_000.0
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            # smooth max via log‑sum‑exp (stable)
            shift = float(np.max(conv))
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_max = shift + np.log(sum_exp) / beta_val
            obj = 2.0 * len(a) * smooth_max

            # gradient wrt a (smooth‑max weights)
            w = exp_beta / sum_exp                 # shape = len(conv)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            # gradient wrt log‑softmax parameters v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # early‑exit support
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a simple linear temperature schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,

    """

    The routine follows the three‑stage strategy described above.
    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Assemble a rich collection of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_exp_decay(n, rate=0.0021))
    seeds.append(_sym_exp_decay(n, rate=0.0021))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, sigma=n / 6.0))
    seeds.append(_gaussian(n, sigma=n / 10.0))

    # power‑law family
    for p in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    # beta family
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws with a wide range of concentrations
    for conc in (0.005, 0.01, 0.02, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random piecewise‑linear shapes
    for nb in (3, 6, 10):
        for _ in range(4):
            seeds.append(_piecewise_linear(n, nb, rng))

    # a handful of permutations of the previous best (if any)
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(6):
            seeds.append(np.random.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop (smooth‑max L‑BFGS + optional EG)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (~12 % each)
        remaining = deadline - time.time()
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] allocated budget ≈ {share:.1f}s")

        # ---- (a) smooth‑max L‑BFGS optimisation (if SciPy is available) ----
        if _opt is not None:
            seq_lb = _lbfgs_smooth_optimize(seed_vec,
                                            sub_deadline,
                                            beta_schedule=None,
                                            max_iter_per_beta=5000)
            val_lb = _surrogate(seq_lb)
        else:
            # fallback: a long exponentiated‑gradient run
            seq_lb, val_lb = _exp_gradient_opt(seed_vec,
                                               sub_deadline,
                                               max_iters=2_000_000,
                                               eta0=6.0)

        # ---- (b) short EG polishing (exact max) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lb,
                                               sub_deadline,
                                               max_iters=800_000,
                                               eta0=5.0)
        else:
            seq_eg, val_eg = seq_lb, val_lb

        # ---- (c) permutation‑based simulated annealing ----
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(seq_eg,
                                              sub_deadline,
                                              max_iters=150_000)
        else:
            seq_perm, val_perm = seq_eg, val_eg

        # pick the best among the three refinements
        local_best_seq, local_best_val = min(
            [(seq_lb, val_lb), (seq_eg, val_eg), (seq_perm, val_perm)],
            key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

        # early exit if we already beat the target bound
        if global_best_val <= 1.503:
            print("\nTarget bound reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_val > 1.503:
        print("\n--- Final long EG polishing (full remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=3_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (few seconds) – may shave a bit
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
