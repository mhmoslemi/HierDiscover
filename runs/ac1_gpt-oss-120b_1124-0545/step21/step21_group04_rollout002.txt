"""
propose_candidate – Adam‑based smooth‑max optimisation for the max‑convolution
objective.

The search proceeds as follows (all steps respect the 1000 s budget):

1. Choose a problem size (default n = 4000) and gather a diverse set of
   starters:
      – the previous best sequence (if available),
      – a uniform distribution,
      – exponential decays,
      – a U‑shaped power‑law (cheap 1‑D search),
      – a few random Dirichlet draws.

2. For each starter run a *quick* Adam optimiser on a log‑softmax
   parametrisation with a smoothed max (log‑sum‑exp) objective.  The
   schedule of smoothing parameters is `[1, 5, 20]`.  The best result among
   the starters is kept.

3. Refine the current best with a *long* Adam run using a finer schedule
   of β‑values up to 10 000 and a few thousand Adam steps per β.  The
   optimiser works directly on the logits `v` where `a = softmax(v)` so the
   vector stays on the probability simplex.

4. Optional Euclidean projected‑gradient polishing (sub‑gradient of the
   exact max‑convolution) gives a final local improvement.

5. A cheap 1‑D mixing with the uniform distribution is performed:
   `mix = λ·best + (1‑λ)·uniform` for λ∈[0,1] to see whether a convex
   combination can lower the true objective.

6. The best sequence found is evaluated with the supplied
   `evaluate_sequence` function, stored globally as ``height_sequence_1``
   and returned as a plain list of floats.

All heavy work (convolutions) is performed with a real FFT for
O(n log n) speed, and the algorithm checks the deadline frequently so it
always finishes before the 1000 s cut‑off.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Cheap 1‑D search for a good U‑shaped starter."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # coarse sweep
    for p in np.linspace(0.0, 2.0, 41):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq, best_p = val, seq, p

    # local refinement
    step = 0.3
    p_center = best_p if 'best_p' in locals() else 0.0
    for _ in range(6):
        if time.time() > deadline:
            break
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq


def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a L1‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    g /= max(1, len(active))
    return g


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule,
              steps_per_beta: int = 500,
              lr: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Optimise a soft‑max (log‑sum‑exp) approximation of max(conv(a,a)).
    The optimisation is performed on the logits `v` such that `a = softmax(v)`.
    Returns the best probability vector found and its (exact) surrogate value.
    """
    a = start_seq.astype(np.float64)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise Adam state
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    t = 0

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1

            a = _softmax(v)          # probability vector (∑a=1)
            conv = _conv_fft(a, a)   # full convolution, length 2n‑1
            max_conv = conv.max()
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()

            # ----- smooth max (log‑sum‑exp) -----
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp   # soft‑max weights, length 2n‑1

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]          # same shape as a

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                   # chain rule through softmax

            # ----- Adam update on v -----
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-8)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine combines several analytic starters,
    a short Adam‑based polishing, a long refinement, a tiny projected‑gradient
    polishing, and a final uniform‑mix sweep.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best (if compatible)
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s > 0:
                seq = seq / s
            else:
                seq = _uniform(n)
            candidates.append(seq)
        except Exception:
            pass

    # b) analytic starters
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.003))
    candidates.append(_exp_decay(n, rate=0.01))
    candidates.append(_best_power_law(n, deadline))

    # c) few random Dirichlet draws
    for _ in range(7):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2. Quick Adam polish for each starter (very cheap)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    beta_quick = [1.0, 5.0, 20.0]
    for idx, starter in enumerate(candidates):
        if time.time() > deadline - 40:
            break
        seq_opt, val_opt = _adam_opt(starter, n, deadline,
                                     beta_schedule=beta_quick,
                                     steps_per_beta=250,
                                     lr=0.5)
        print(f"[Scout {idx}] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # fallback
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 3. Long Adam refinement on the current best
    # ------------------------------------------------------------------
    beta_long = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                 1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
    seq_ref, val_ref = _adam_opt(best_seq, n, deadline,
                                 beta_schedule=beta_long,
                                 steps_per_beta=2500,
                                 lr=0.3)
    print(f"[Refine] surrogate after long Adam = {val_ref:.6f}")
    if val_ref < best_val:
        best_val, best_seq = val_ref, seq_ref.copy()

    # ------------------------------------------------------------------
    # 4. Optional projected‑gradient polishing (sub‑gradient of exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15:
        seq_pg, val_pg = _pgd_opt(best_seq, n, deadline,
                                 max_iters=150_000,
                                 eta0=0.6)
        print(f"[PGD] surrogate after polishing = {val_pg:.6f}")
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()

    # ------------------------------------------------------------------
    # 5. Small convex mix with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        uniform = _uniform(n)
        mix_best_val = best_val
        mix_best_seq = best_seq.copy()
        for lam in np.linspace(0.0, 1.0, 21):
            mixed = lam * best_seq + (1.0 - lam) * uniform
            val = _objective_surrogate(mixed)
            if val < mix_best_val:
                mix_best_val, mix_best_seq = val, mixed.copy()
        print(f"[Mix] best after mixing = {mix_best_val:.6f}")
        if mix_best_val < best_val:
            best_val, best_seq = mix_best_val, mix_best_seq.copy()

    # ------------------------------------------------------------------
    # 6. Final evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pylint: disable=broad-except
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
