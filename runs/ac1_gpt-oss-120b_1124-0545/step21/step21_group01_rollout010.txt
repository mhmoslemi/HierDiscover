"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The algorithm is a hybrid of many cheap heuristics and a few
gradient‑based refinements:

1. **Initial seeds** – uniform, arcsine, triangular plus a few
   Dirichlet draws (different concentration parameters).  The best
   seed becomes the current incumbent.

2. **Multiple short Exponentiated‑Gradient (EG) runs** – each run starts
   from a fresh random Dirichlet draw (or a perturbed incumbent) and
   performs mirror‑descent on the exact max‑convolution objective.
   The best result of all runs is kept.

3. **Variance‑minimising L‑BFGS** – a smooth surrogate
   ``∑ (conv(a,a) – avg)²`` is minimised with L‑BFGS‑B.  Reducing the
   variance of the auto‑convolution drives the max value down.

4. **Final local random walk** – a tiny stochastic walk on the simplex
   (move a small amount from one coordinate to another) finishes the
   search.

All stages respect the global 1000 s budget.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the L‑BFGS stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during search: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.005) -> np.ndarray:
    """Add small Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact ``max(conv)`` objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        # compute full convolution
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # index of the active max

        # sub‑gradient of max‑convolution (only entries that contribute to k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement on the simplex.
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk: move a small random amount from i to j and keep
    improvements.  Stops when ``attempts`` are exhausted or the deadline
    is reached.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:            # sanity check
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the *variance* of the auto‑convolution.
# The surrogate objective is   sum_k (conv_k – avg)² .
# ----------------------------------------------------------------------
def _lbfgs_variance_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        maxiter: int = 15000) -> tuple[np.ndarray, float]:
    """
    Minimise the L² distance between ``conv(a,a)`` and its average
    value (1/(2n‑1)).  Returns the best sequence (as a numpy array) and
    its exact surrogate value.
    """
    if _opt is None:
        # SciPy not available – fall back to the original sequence
        return start_seq.copy(), _objective_surrogate(start_seq)

    avg = 1.0 / (2 * n - 1)

    def fun(v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        residual = conv - avg
        obj = np.sum(residual ** 2)

        # gradient wrt a : 2 * conv(residual, rev_a)
        rev_a = a[::-1]
        grad_a = 2.0 * _conv_fft(residual, rev_a)[n - 1:n - 1 + n]

        # convert to log‑softmax gradient (mirror‑descent geometry)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

    v0 = np.log(start_seq + 1e-30)

    try:
        res = _opt.minimize(fun=fun,
                            x0=v0,
                            method='L-BFGS-B',
                            jac=True,
                            options={'maxiter': maxiter,
                                     'ftol': 1e-12,
                                     'disp': False})
        seq_opt = _softmax(res.x)
        val_opt = _objective_surrogate(seq_opt)
        return seq_opt, val_opt
    except Exception as e:                          # pragma: no cover
        print("[L‑BFGS variance] optimisation failed:", e)
        return start_seq.copy(), _objective_surrogate(start_seq)


# ----------------------------------------------------------------------
# Deterministic seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a
