def adam_opt(start_seq, n, deadline, max_iters, lr=1.0, eps=1e-12):

Initialize logit v = np.log(start_seq+eps).

Initialize m = zeros, v2= zeros.

Loop for iteration i:

- compute a = softmax(v).

- compute subgradient g (subgradient of max conv). g shape (n,). (However subgradient can break symmetry; but due to symmetry of a maybe g also symmetric.)

- compute grad_v = a * (g - np.dot(g, a))

- Update m = beta1*m + (1-beta1)*grad_v

- Update v2 = beta2*v2 + (1-beta2)*(grad_v**2)

- Compute bias-corrected m_hat = m/(1-beta1**i); v2_hat = v2/(1-beta2**i)

- v = v - lr * m_hat / (np.sqrt(v2_hat)+eps)

- Symmetrize v: v = (v + v[::-1]) /2

- Every few iterations evaluate surrogate and keep best solution.

Stop when deadline.

Return best_seq.

"""
propose_candidate – multi‑scale Adam / smooth‑max optimisation for the
max‑convolution objective.

Key ideas
---------
1. **Rich starter family** – In addition to the classic uniform,
   exponential and power‑law starters we add analytically motivated
   “sqrt‑difference’’ and “inverse‑sqrt’’ families that approximate the
   continuous solution  f(t)∝1/√t (which is known to give a flat
   self‑convolution in the interior of the interval).

2. **Coarse‑to‑fine optimisation** – First optimise a short version of the
   problem (n_coarse≈800) with many fast Adam steps.  The best coarse
   distribution is then up‑sampled to the target length (n=4000) and
   refined at full resolution.

3. **Adam on a smoothed max** – Instead of the sub‑gradient of the exact
   max‑convolution we use a differentiable soft‑max
   ``log‑sum‑exp`` approximation with a continuation on the temperature
   (parameter β).  The gradient of the smooth objective is exact and
   Adam’s adaptive step‑size gives faster convergence than the vanilla
   exponentiated‑gradient used previously.

4. **Strict symmetry** – After each update the log‑it­ernary vector is
   symmetrised (v ← (v+v[::-1])/2).  The optimal solution is known to be
   mirror‑symmetric, halving the effective search space and improving
   stability.

5. **Final L‑BFGS polishing** – If SciPy is available we run a few
   BFGS‑B steps on the same smooth objective with a very large β,
   obtaining a final fine‑grained improvement.

The routine respects the 1000 s wall‑clock limit by polling the deadline
frequently and by keeping all heavy work in NumPy (FFT based convolution
≈ O(n log n)).  The returned list is stored in the global variable
``height_sequence_1`` for the next call.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the final L‑BFGS polishing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a probability vector (sum = 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max utilities (log‑it­ernary → probability simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    return w / w.sum()


def _sqrt_diff(n: int) -> np.ndarray:
    """
    a_i ∝ (√(i+1)‑√i) + (√(n‑i)‑√(n‑i‑1)).
    This discretises f(t)=1/(2√t) on both ends (the classical flat‑conv
    construction).
    """
    i = np.arange(n, dtype=np.float64)
    left = np.sqrt(i + 1.0) - np.sqrt(i)
    right = np.sqrt(n - i) - np.sqrt(n - i - 1.0)
    w = left + right
    return w / w.sum()


def _inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1) + 1/√(n‑i)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    return w / w.sum()


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over the active set)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a))."""
    n = a.size
    conv = _conv_fft(a, a)
    mx = conv.max()
    # treat anything within 1e‑12 relative tolerance as active
    tol = max(mx * 1e-12, 1e-14)
    active = np.where(conv >= mx - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    # average over active indices
    g /= max(1, len(active))
    return g


# ----------------------------------------------------------------------
# Gradient of the smooth max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_objective, gradient_wrt_logits) for the soft‑max
    approximation of max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)                     # length 2n‑1
    shift = conv.max()
    # log‑sum‑exp of beta·(conv‑shift)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # w = soft‑max over the convolution entries (weights for the smooth max)
    w = exp_beta / sum_exp                     # shape (2n‑1,)

    # gradient of max(conv) w.r.t. a:
    rev_a = a[::-1]
    # full convolution rev_a * w (full length = 3n‑2)
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # length n, corresponds to ∂max/∂a

    # chain rule to logits (log‑softmax parametrisation)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return smooth_obj, grad_logits


# ----------------------------------------------------------------------
# Adam optimiser (with optional smooth‑max)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-12,
              eta0: float = 1.0,
              smooth_beta_schedule: List[float] | None = None) -> tuple[np.ndarray, float]:
    """
    Adam on the surrogate/​smooth surrogate.
    If ``smooth_beta_schedule`` is supplied the optimiser uses the smooth
    max with the given β values (continuation).  Between schedule steps the
    algorithm simply keeps iterating with the current β.
    """
    a = start_seq.copy()
    # initialise logits (log‑softmax variables)
    v = np.log(a + 1e-30)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # set up β progression
    if smooth_beta_schedule is None:
        # plain max‑convolution (β = ∞) → use sub‑gradient
        use_smooth = False
        beta_iter = iter([None])
    else:
        use_smooth = True
        beta_iter = iter(smooth_beta_schedule)

    cur_beta = next(beta_iter)
    # for a smooth phase we pre‑compute the objective at the first β
    if use_smooth:
        cur_obj, _ = _smooth_grad(_softmax(v), cur_beta)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Update β when a fraction of the total iterations has elapsed
        if use_smooth:
            # simple schedule: change β every 20 % of the allocated iterations
            if it % (max_iters // max(1, len(smooth_beta_schedule))) == 0:
                try:
                    cur_beta = next(beta_iter)
                except StopIteration:
                    pass

        a = _softmax(v)

        # enforce strict mirror symmetry (helps a lot for this problem)
        a = 0.5 * (a + a[::-1])

        if use_smooth:
            obj, grad = _smooth_grad(a, cur_beta)
        else:
            # exact sub‑gradient of the max‑convolution
            g = _subgradient_max_conv(a)
            # gradient w.r.t logits: a * (g - <g,a>)
            dot = np.dot(g, a)
            grad = a * (g - dot)
            obj = _objective_surrogate(a)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = eta0 / (np.sqrt(v2_hat) + eps)
        v = v - step * m_hat

        # keep logits symmetric as well (helps numerical stability)
        v = 0.5 * (v + v[::-1])

        # keep track of the best sequence seen so far
        if obj < best_val:
            best_val = obj
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Quick 1‑D search for the optimal power‑law exponent (used for a starter)
# ----------------------------------------------------------------------
def _best_power_law(n: int, deadline: float) -> np.ndarray:
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)
    # coarse grid
    for p in np.linspace(0.0, 2.0, 31):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # refine around the best exponent
    p_center = float(p)  # last p from loop, replaced later
    # find exact centre from stored best
    # (a tiny brute‑force refinement – cheap)
    step = 0.2
    for _ in range(5):
        if time.time() > deadline:
            break
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq


# ----------------------------------------------------------------------
# Upsampling from a coarse grid to the target size
# ----------------------------------------------------------------------
def _upsample(coarse: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(coarse) to target_n and normalisation."""
    x_old = np.linspace(0.0, 1.0, coarse.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    interpolated = np.interp(x_new, x_old, coarse)
    interpolated = np.maximum(interpolated, 0.0)
    s = interpolated.sum()
    if s == 0.0:
        return _uniform(target_n)
    return interpolated / s

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ₁‑normalised) that minimises
    ``evaluate_sequence``.  The routine combines:
      * a diverse set of analytic starters (including the sqrt‑difference family)
      * a coarse Adam optimisation on a reduced dimension (n≈800)
      * up‑sampling to the final dimension (n=4000)
      * a long Adam refinement with a continuation on the smooth‑max temperature
      * an optional L‑BFGS polishing phase (if SciPy is available)
    The best sequence found is stored in the global variable
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Target resolution – the previous best used 4000, keep the same
    # ------------------------------------------------------------------
    n_target = 4000

    # ------------------------------------------------------------------
    # 1. Build a small pool of starters (coarse dimension)
    # ------------------------------------------------------------------
    n_coarse = 800
    starters = []

    # a) analytic families (coarse)
    starters.append(_uniform(n_coarse))
    starters.append(_exp_decay(n_coarse, rate=0.002))
    starters.append(_power_law(n_coarse, p=0.5))
    starters.append(_sqrt_diff(n_coarse))
    starters.append(_inv_sqrt(n_coarse))

    # b) a few random Dirichlet draws (coarse)
    for _ in range(5):
        starters.append(rng.dirichlet(np.ones(n_coarse)))

    # c) optionally reuse the previous best (if it has the same length)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n_target,):
                # down‑sample to coarse grid for the first stage
                starters.append(_upsample(prev_arr, n_coarse))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Short Adam runs on the coarse grid (cheap exploration)
    # ------------------------------------------------------------------
    best_coarse_seq = None
    best_coarse_val = float("inf")
    for idx, seq0 in enumerate(starters):
        if time.time() > deadline - 30:
            break
        seq_opt, val_opt = _adam_opt(seq0, n_coarse, deadline,
                                     max_iters=60_000,
                                     eta0=0.8,
                                     smooth_beta_schedule=None)  # exact max‑conv
        print(f"[Coarse {idx}] surrogate = {val_opt:.6f}")
        if val_opt < best_coarse_val:
            best_coarse_val, best_coarse_seq = val_opt, seq_opt.copy()

    # ------------------------------------------------------------------
    # 3. Upsample to the final resolution and perform a long refinement
    # ------------------------------------------------------------------
    if best_coarse_seq is None:
        best_coarse_seq = _uniform(n_coarse)

    seq_fine = _upsample(best_coarse_seq, n_target)
    # ensure exact normalisation
    seq_fine = np.maximum(seq_fine, 0.0)
    seq_fine /= seq_fine.sum()

    # continuation schedule for the smooth‑max temperature β
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                  1_000.0, 2_000.0, 5_000.0, 20_000.0,
                  50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0]

    remaining_iters = 2_400_000
    seq_opt, val_opt = _adam_opt(seq_fine, n_target, deadline,
                                 max_iters=remaining_iters,
                                 eta0=0.9,
                                 smooth_beta_schedule=beta_sched)
    print(f"[Fine Adam] surrogate after long run = {val_opt:.6f}")
    if val_opt < best_coarse_val:
        best_seq = seq_opt.copy()
        best_val = val_opt
    else:
        best_seq = seq_fine.copy()
        best_val = best_coarse_val

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS polishing (smooth‑max, very high β)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5 and _opt is not None:
        def _smooth_obj_and_grad(logits):
            a = _softmax(logits)
            obj, grad = _smooth_grad(a, beta=2_000_000.0)   # almost exact max
            return obj, grad

        def _callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            init_logits = np.log(best_seq + 1e-30)
            res = _opt.minimize(fun=_smooth_obj_and_grad,
                                x0=init_logits,
                                method='L-BFGS-B',
                                jac=True,
                                callback=_callback,
                                options={'maxiter': 4000,
                                         'ftol': 1e-12,
                                         'disp': False})
            seq_lb = _softmax(res.x)
            val_lb = _objective_surrogate(seq_lb)
            print(f"[L‑BFGS] surrogate after polishing = {val_lb:.6f}")
            if val_lb < best_val:
                best_seq, best_val = seq_lb, val_lb
        except KeyboardInterrupt:
            pass
        except Exception as exc:   # pragma: no cover
            print("[L‑BFGS] exception:", exc)

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided oracle
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
