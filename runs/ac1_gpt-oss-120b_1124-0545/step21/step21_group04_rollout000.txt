def _smooth_obj_and_grad(logits: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_c = conv.max()
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits

def _lbfgs_optimize(start_seq, n, deadline, beta_schedule=None, max_iter_per_beta=5000):
    if _opt is None:
        return start_seq
    if beta_schedule is None:
        beta_schedule = [1,5,20,80,200,500,1000,2000,5000,10000,20000,50000,100000,200000,500000,1_000_000]

    v = np.log(start_seq + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            obj, grad = _smooth_obj_and_grad(v_local, n, beta_val)
            return obj, grad

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'gtol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The approach improves on the previous
implementation by

* adding several analytic starters (uniform, exponential, power‑law,
  cosine‑U, random Dirichlet);
* selecting the best few starters;
* refining them with a large‑scale Adam optimiser on a smooth
  log‑sum‑exp approximation of the max‑convolution;
* polishing the result with a higher‑iteration L‑BFGS run (if SciPy is
  available);
* performing a few random perturbations of the final logits to escape
  local minima.

All optimisation respects a 1000 s wall‑time budget, repeatedly checks
the deadline, and finally evaluates the true objective with
``evaluate_sequence``.  The best sequence is stored globally as
``height_sequence_1`` and returned as a plain list of floats.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based linear convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (probability simplex representation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families (all return a probability vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _cosine_u(n: int) -> np.ndarray:
    """U‑shaped cosine: high at the ends, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 + np.cos(2.0 * np.pi * i / (n - 1))
    # shift to be non‑negative; the formula already is ≥0
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L1‑norm 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) objective and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad_logits) where obj is the smooth approximation of
    2·n·max(conv(a,a)) and grad_logits is the gradient w.r.t. the logits
    (a = softmax(logits)).
    """
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_c = conv.max()
    # stable log‑sum‑exp
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weights for the smooth max
    w = exp_beta / sum_exp

    # gradient w.r.t. a (see derivation in the analysis)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]           # slice the linear part

    # chain‑rule: d obj / d logits = a * (grad_a - <grad_a,a>)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   iters_per_beta: int = 30_000,
                   lr: float = 0.05,
                   betas: tuple = (0.9, 0.999),
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth objective.
    Returns the best probability vector found and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 10_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
        ]

    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0
    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad = _smooth_obj_and_grad(v, n, beta)

            # Adam update
            m = betas[0] * m + (1.0 - betas[0]) * grad
            v_hat = betas[1] * v_hat + (1.0 - betas[1]) * (grad * grad)

            m_hat = m / (1.0 - betas[0] ** t)
            v_hat_corr = v_hat / (1.0 - betas[1] ** t)

            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # occasional exact evaluation (every 5000 steps)
            if t % 5000 == 0:
                a = _softmax(v)
                val = _objective_surrogate(a)
                if val < best_val:
                    best_val, best_seq = val, a.copy()

    # final check
    final_a = _softmax(v)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val, best_seq = final_val, final_a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth objective (requires SciPy)
# ----------------------------------------------------------------------
def _lbfgs_optimize(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    beta_schedule=None,
                    max_iter_per_beta: int = 8_000):
    """
    Successively runs L‑BFGS‑B on the smooth objective for an increasing
    sequence of β values.  Returns a probability vector.
    """
    if _opt is None:
        # SciPy unavailable – fall back to the input sequence unchanged
        return start_seq

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 10_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
        ]

    v = np.log(start_seq + 1e-30)  # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            obj, grad = _smooth_obj_and_grad(v_local, n, beta_val)
            return obj, grad

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={
                                    'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'gtol': 1e-12,
                                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
