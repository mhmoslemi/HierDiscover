"""
Propose a step‑function (non‑negative vector that sums to 1) that
minimises the “evaluate_sequence’’ objective.

Approach
--------
1. Gather a diverse pool of deterministic seeds (uniform, arcsine,
   triangular, β‑distributions, cosine‑shaped, power‑law edges …).
2. Scan a few one‑parameter families (γ‑arcsine, edge‑power, cosine‑power,
   shifted arcsine) on a fine grid – this cheap exhaustive scan often
   finds a better shape than the generic Dirichlet / Gaussian‑mixture
   seeds used previously.
3. Starting from the best vector seen so far, run a short
   exponentiated‑gradient mirror‑descent on the *exact* max‑convolution
   (sub‑gradient of the ℓ∞‑norm).  The step‑size schedule is a bit more
   aggressive than the reference implementation.
4. Finish with a lightweight stochastic local‑search that moves a small
   random amount of mass between two coordinates while always keeping
   the vector normalised.
5. Return the best sequence (as a plain list).  The routine respects the
   overall time budget (default 1000 s) and stores the result in the
   global variable ``height_sequence_1'' for possible warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (linear full convolution via real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using a real FFT (fast for n≈4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq is assumed normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax helper (used by the exponentiated‑gradient routine)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed families (all return a *probability* vector, i.e. sum=1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density (β=α=0.5)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a /= a.sum()
    return a


def _arcsine_gamma(n: int, gamma: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)(n‑½‑i))^{-γ}."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 0.5) * (n - 0.5 - i)) ** gamma
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _arcsine_shift(n: int, shift: float) -> np.ndarray:
    """Shifted arcsine: add a constant to the denominator for extra smoothing."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + shift)
    a /= a.sum()
    return a


def _edge_power(n: int, p: float) -> np.ndarray:
    """Power‑law decay from both edges: a_i ∝ (i+1)^{-p} + (n‑i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    a /= a.sum()
    return a


def _cosine_power(n: int, power: float) -> np.ndarray:
    """|cos|‑shaped profile: a_i ∝ |cos(π·(i+½)/n)|^{power}."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(math.pi * (i + 0.5) / n)) ** power
    # avoid exact zeros (they would stay at 0 forever)
    a += 1e-30
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1} (continuous version, then discretised)."""
    if n == 1:
        return np.array([1.0])
    t = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(t + eps, alpha - 1.0) * np.power(1.0 - t + eps, beta - 1.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = _uniform(n)
    else:
        a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    v = rng.dirichlet(np.full(n, conc))
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return _uniform(n)
    v /= v.sum()
    return v.astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the genuine max‑convolution.
    Uses the exact sub‑gradient at the (single) active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute convolution and locate an index attaining the maximum
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of (a*a)[k] w.r.t. a:
        # for each i that can appear in a_i * a_{k-i}
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        g[idx_i] = 2.0 * a[rev]                # derivative of a_i * a_rev

        # step‑size schedule (aggressive early, then 1/√t decay)
        eta = eta0 / math.sqrt(it)

        # mirror‑descent update in log‑space (exponentiated‑gradient)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny local random walk (hill climbing on the simplex)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 150_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic walk: repeatedly move a random fraction of mass
    from one coordinate to another, keeping improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # a random fraction of the source mass (scaled by sigma)
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:          # safety (should not happen)
            continue
        val = _objective(new_seq)
        if val < best_val:
            best_seq, best_val = new_seq, val

    return best_seq, best_val

# Helper for triangular shape (used only if the name is already in globals)
# ----------------------------------------------------------------------
def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a
