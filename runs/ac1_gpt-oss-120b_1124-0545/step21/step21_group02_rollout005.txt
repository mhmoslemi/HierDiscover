"""
search for a step‑function that minimises `evaluate_sequence`.

The algorithm consists of three stages:

1. **Linear‑programming spreading (LP)**
   – With the current distribution `a` we compute its worst auto‑convolution
     value `M = max(a*a)`.
   – We maximise the total mass `∑b` of a new vector `b` under the linear
     constraints `a * b ≤ M` (component‑wise convolution) using `scipy.linprog`.
   – After normalising `b` to unit sum we keep the better of `b` or the
     midpoint `(a+b)/2`.  Repeating a few times spreads the mass and typically
     lowers the surrogate value `2·n·max(conv)`.

2. **Exact mirror‑descent polishing (exponentiated‑gradient)**
   – On the simplex we perform sub‑gradient steps for the true objective
     `max(a*a)`.  The sub‑gradient of the active convolution entry is obtained
     analytically, and the iteration is carried out in log‑space to stay on the
     probability simplex.

3. **Optional smooth L‑BFGS refinement**
   – A smoothed version of the max‑convolution (log‑sum‑exp) is optimised with
     `scipy.optimize.minimize` (L‑BFGS‑B) if time permits.

All intermediate results and timings are printed.  The best sequence found is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import math
import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)), n = len(seq).  Exact if sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient mirror descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    `start_seq` must already sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)           # work in log‑space (probability simplex)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        # soft‑max to stay on the simplex
        m = np.max(log_a)
        e = np.exp(log_a - m)
        a = e / e.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming spreading step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n matrix A with A[k, j] = a_{k‑j} (0 if out of range).
    This matrix represents the linear constraints   a * b ≤ M .
    """
    n = len(a)
    m = 2 * n - 1
    # each diagonal d (offset = d) is constant a[d] repeated n times
    diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    A = sp.diags(diags, offsets, shape=(m, n), format='csr')
    return A


def _lp_spread(seq: np.ndarray,
               n: int,
               deadline: float,
               max_lp_iters: int = 6) -> np.ndarray:
    """
    Perform a few LP iterations that try to increase the total mass
    while keeping the cross‑convolution with the current vector below its
    current maximum `M`.  The best sequence obtained during the loop is
    returned.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for lp_it in range(max_lp_iters):
        if time.time() > deadline - 5.0:
            break

        # current bound M
        M = _conv_fft(a, a).max()

        # build constraint matrix for a * b ≤ M
        A = _build_conv_matrix(a)
        m = 2 * n - 1
        b_ub = np.full(m, M, dtype=np.float64)

        # maximise sum(b)  ⇔ minimise –sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve LP
        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'disp': False})

        if not res.success:
            print(f"[LP] iteration {lp_it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no improvement possible – we are already optimal for this M
            print(f"[LP] iteration {lp_it} – no increase in total mass, stop.")
            break

        # normalise to unit sum
        g = b / sum_b

        # evaluate candidates
        val_g = _objective_surrogate(g)
        if val_g < best_val:
            best_val, best_seq = val_g, g.copy()
            a = g.copy()
            print(f"[LP] iter {lp_it}: direct candidate improved surrogate = {val_g:.6f}")
            continue

        # try the midpoint (a+g)/2
        mid = (a + g) / 2.0
        mid = mid / mid.sum()
        val_mid = _objective_surrogate(mid)
        if val_mid < best_val:
            best_val, best_seq = val_mid, mid.copy()
            a = mid.copy()
            print(f"[LP] iter {lp_it}: midpoint improved surrogate = {val_mid:.6f}")
            continue

        # if neither improves, stop early
        print(f"[LP] iter {lp_it}: no improvement, stop.")
        break

    return best_seq


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Smooth optimisation of max‑convolution via log‑sum‑exp,
    solved with L‑BFGS‑B in the log‑softmax space.
    """
    if not hasattr(opt, "minimize"):
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_f = float(beta)

        def fun(v_local):
            a = np.exp(v_local - np.max(v_local))   # soft‑max (numerically stable)
            a = a / a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_f * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_f)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # transform gradient to log‑space (mirror‑descent formula)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = opt.minimize(fun=fun,
                               x0=v,
                               method='L-BFGS-B',
                               jac=True,
                               callback=callback,
                               options={'maxiter': max_iter_per_beta,
                                        'disp': False,
                                        'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:
            print(f"[L‑BFGS] exception at β={beta_f}: {exc}")
            break

    a = np.exp(v - np.max(v))
    a = a / a.sum()
    return a

    else:
        best_seq = seq_eg
        best_val = val_eg

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided secure function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # Store for the next run (incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
