    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid random‑seed / DE / L‑BFGS / smoothed sub‑gradient optimizer.

The goal is to minimise the true objective
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows (all phases respect a total 1000 s budget)

1.  Assemble a diverse pool of probability vectors (uniform, Dirichlet,
    sparse Dirichlet, tiny Gaussian‑mixture, Fourier‑softmax and a few
    manual “edge‑heavy’’ seeds).  Keep the best one according to the
    cheap surrogate 2·n·max(conv).

2.  Run SciPy’s differential evolution on the two low‑dimensional
    parametric families that performed best in earlier experiments
    (mixture of Gaussians and Fourier‑softmax).  The resulting vectors
    replace the pool‑best if they improve the surrogate.

3.  Refine the current best with L‑BFGS on the log‑sum‑exp smoothed
    version of the objective (the same code as in the reference run).

4.  Perform a long “smoothed sub‑gradient’’ descent on the *exact*
    max‑convolution.  For a given vector a we compute the convolution,
    pick the (few) indices where it is maximal, build a weight vector
    w (either a uniform weight on the active indices or a soft‑max
    weighting when the active set is a singleton) and obtain the
    sub‑gradient via a single FFT convolution.  The update is a
    projected additive step on the simplex (clip‑negative‑and‑renormalise).

5.  Finally a short stochastic polishing loop (perturb → short
    sub‑gradient optimisation) searches the neighbourhood of the
    current best.

All phases share the global variable ``height_sequence_1`` and the
function returns the best sequence found.
"""

import time
import math
import numpy as np

# optional SciPy – used for differential evolution
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (no denominator – the sum is kept equal to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the objective for a simplex‑normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (clip‑negative & renorm)
# ----------------------------------------------------------------------
def _simplex_project(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k random positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, later normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K : 2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K : 3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)  # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for kk in range(K):
        a += w[kk] * np.exp(-((i - mu[kk]) ** 2) / (2.0 * sigma[kk] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """
    Build a probability vector by applying a softmax to a Fourier series.
    a_i ∝ exp( Σ_k θ_k φ_k(i) ).
    """
    logits = basis @ theta
    e = np.exp(logits - np.max(logits))
    a = e / e.sum()
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, then project back to the simplex."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    return _simplex_project(noisy)


# ----------------------------------------------------------------------
# Smoothed sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _smooth_subgradient_opt(start_seq: np.ndarray,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 4.0,
                            beta_single: float = 12.0,
                            max_active: int = 12) -> tuple[np.ndarray, float]:
    """
    Projected gradient–descent on the exact max‑convolution.
    * start_seq* must already be a probability vector.
    The step size is η₀ / √t.
    When the active set (indices attaining the maximum) is a single
    element we fall back to a soft‑max weighting with temperature controlled
    by *beta_single* – this gives a useful descent direction even at a
    strict local maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        maxc = np.max(conv)

        # indices where the max is attained (within a tiny tolerance)
        active = np.where(conv >= maxc - 1e-15)[0]

        if active.size == 0:          # should never happen
            active = np.array([np.argmax(conv)], dtype=int)

        # keep the active set manageable
        if active.size > max_active:
            # keep a deterministic subset (first few) – randomness would add bias
            active = active[:max_active]

        # Build the weight vector w (same length as conv)
        w = np.zeros_like(conv)
        if active.size == 1:
            # smooth logarithmic approximation of the max
            shift = maxc
            exp_beta = np.exp(beta_single * (conv - shift))
            w = exp_beta / exp_beta.sum()
        else:
            w[active] = 1.0 / float(active.size)   # uniform over the active set

        # Gradient w.r.t. a (derived in the reference implementation)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # Simple diminishing step‑size
        eta = eta0 / math.sqrt(t)

        # Additive move followed by projection onto the simplex
        a = _simplex_project(a - eta * grad)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return the best step‑function (list of non‑negative heights) found
    within the allocated time.  The result is also stored in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    n = 4000   # problem size (same as the reference run)

    # ------------------------------------------------------------------
    # 0️⃣  Gather a possible previous solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Diverse initial pool (all vectors are already ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    # dense Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse Dirichlet (support sizes 80 … 200)
    for k in (80, 120, 160, 200):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # tiny Gaussian‑mixture seeds (8 components)
    K_mix = 8
    for _ in range(4):
        params = np.concatenate([
            rng.random(K_mix),                     # weights (will be normalised)
            rng.random(K_mix),                     # means
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n, K=K_mix))

    # Fourier‑softmax seeds (20 basis functions)
    D_fourier = 20
    i_grid = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis = []
    for k in range(1, D_fourier // 2 + 1):
        basis.append(np.sin(2.0 * np.pi * k * i_grid))
        basis.append(np.cos(2.0 * np.pi * k * i_grid))
    basis = np.column_stack(basis)   # (n, D_fourier)
    for _ in range(6):
        theta = rng.normal(scale=1.0, size=D_fourier)
        seeds.append(_fourier_softmax(theta, basis))

    # a couple of ad‑hoc “edge‑heavy’’ seeds (more mass at the borders)
    def _edge_heavy():
        a = np.full(n, 1e-12, dtype=np.float64)
        # allocate a quarter of the total mass to the first/last 5 % of locations
        edge_len = max(1, n // 20)          # ~5 %
        a[:edge_len] = 0.2 / edge_len
        a[-edge_len:] = 0.2 / edge_len
        a[edge_len : -edge_len] = 0.6 / (n - 2 * edge_len)
        return a

    seeds.append(_edge_heavy())

    # ------------------------------------------------------------------
    # 2️⃣  Pick the best seed according to the cheap surrogate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating initial seeds (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Global DE in the two low‑dimensional parametric families
    # ------------------------------------------------------------------
    # ---- Gaussian‑mixture DE -------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution on Gaussian mixture (8 comps) ---")
        bounds = [(0.0, 1.0)] * K_mix + [(0.0, 1.0)] * K_mix + [(0.001, 0.5)] * K_mix

        def de_obj_gauss(p):
            seq = _mixture_gaussian(p, n, K=K_mix)
            return _objective_surrogate(seq)

        try:
            result = _opt.differential_evolution(
                de_obj_gauss,
                bounds,
                seed=int(seed),
                popsize=16,
                maxiter=65,
                polish=False,
                updating='deferred',
                workers=1,
                init='latinhypercube',
                disp=False,
                callback=lambda xk, convergence: time.time() > deadline,
            )
            seq_gauss = _mixture_gaussian(result.x, n, K=K_mix)
            val_gauss = _objective_surrogate(seq_gauss)
            print(f"[DE‑Gauss] surrogate = {val_gauss:.6f}")
            if val_gauss < best_val:
                best_val, best_seq = val_gauss, seq_gauss.copy()
        except Exception as e:  # pragma: no cover
            print("[DE‑Gauss] failed:", e)

    # ---- Fourier‑softmax DE -------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution on Fourier‑softmax ---")
        bounds = [(-6.0, 6.0)] * D_fourier

        def de_obj_fourier(th):
            seq = _fourier_softmax(th, basis)
            return _objective_surrogate(seq)

        try:
            result = _opt.differential_evolution(
                de_obj_fourier,
                bounds,
                seed=int(seed + 1),
                popsize=14,
                maxiter=55,
                polish=False,
                updating='deferred',
                workers=1,
                init='latinhypercube',
                disp=False,
                callback=lambda xk, convergence: time.time() > deadline,
            )
            seq_four = _fourier_softmax(result.x, basis)
            val_four = _objective_surrogate(seq_four)
            print(f"[DE‑Fourier] surrogate = {val_four:.6f}")
            if val_four < best_val:
                best_val, best_seq = val_four, seq_four.copy()
        except Exception as e:  # pragma: no cover
            print("[DE‑Fourier] failed:", e)

    # ------------------------------------------------------------------
    # 4️⃣  Local refinement with smooth L‑BFGS (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of current best ---")
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45

        # Re‑use the helper from the reference code (unchanged)
        def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                                   n_loc: int,
                                   deadline_loc: float,
                                   beta_schedule=None,
                                   max_iter_per_beta: int = 15000):
            if _opt is None:
                return start_vec.copy()
            if beta_schedule is None:
                beta_schedule = [
                    1.0, 5.0, 20.0, 80.0,
                    200.0, 500.0, 1000.0,
                    2000.0, 5000.0,
                    20000.0, 50000.0,
                    100000.0, 200000.0, 500000.0,
                    1000000.0, 2000000.0, 5000000.0,
                    10000000.0,
                ]

            v = np.log(start_vec + 1e-30)   # log‑softmax variables

            for beta in beta_schedule:
                if time.time() > deadline_loc:
                    break
                beta_val = float(beta)

                def fun(v_loc):
                    a = np.exp(v_loc - np.max(v_loc))
                    a = a / a.sum()
                    conv = _conv_fft(a, a)
                    max_c = np.max(conv)
                    shift = max_c
                    exp_beta = np.exp(beta_val * (conv - shift))
                    sum_exp = exp_beta.sum()
                    obj = 2.0 * n_loc * (shift + np.log(sum_exp) / beta_val)

                    w = exp_beta / sum_exp
                    rev_a = a[::-1]
                    full = _conv_fft(rev_a, w)
                    grad_a = 2.0 * full[n_loc - 1 : n_loc - 1 + n_loc]

                    dot = np.dot(grad_a, a)
                    grad_v = a * (grad_a - dot)
                    return obj, grad_v

                def callback(_):
                    if time.time() > deadline_loc:
                        raise KeyboardInterrupt()

                try:
                    res = _opt.minimize(fun=fun,
                                        x0=v,
                                        method='L-BFGS-B',
                                        jac=True,
                                        callback=callback,
                                        options={'maxiter': max_iter_per_beta,
                                                 'disp': False,
                                                 'ftol': 1e-12})
                    v = res.x
                except KeyboardInterrupt:
                    break
                except Exception as e:  # pragma: no cover
                    print(f"[L‑BFGS] exception {e} at β={beta_val}")
                    break

            a_opt = np.exp(v - np.max(v))
            a_opt = a_opt / a_opt.sum()
            return a_opt

        seq_lbfgs = _lbfgs_smooth_optimize(
            best_seq, n,
            sub_deadline,
            beta_schedule=[
                1.0, 5.0, 20.0, 80.0,
                200.0, 500.0, 1000.0,
                2000.0, 5000.0,
                20000.0, 50000.0,
                100000.0, 200000.0, 500000.0,
                1000000.0, 2000000.0, 5000000.0,
                10000000.0,
            ],
            max_iter_per_beta=25000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("  >>> Updated global best after L‑BFGS")

    # ------------------------------------------------------------------
    # 5️⃣  Exact smoothed sub‑gradient optimisation (the new ingredient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Smoothed sub‑gradient optimisation (exact max) ---")
        sub_deadline = deadline - 3.0
        time_left = sub_deadline - time.time()
        max_iters = int(min(2_500_000,
                            max(200_000,
                                time_left / (deadline - start_time) * 2_500_000)))
        seq_sg, val_sg = _smooth_subgradient_opt(
            best_seq,
            sub_deadline,
            max_iters=max_iters,
            eta0=4.0,
            beta_single=12.0,
            max_active=12,
        )
        print(f"[SubGrad] surrogate = {val_sg:.6f}")
        if val_sg < best_val:
            best_val, best_seq = val_sg, seq_sg.copy()
            print("  >>> Updated global best after sub‑gradient")

    # ------------------------------------------------------------------
    # 6️⃣  Stochastic polishing (perturb + short sub‑gradient)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        cand = _perturb(best_seq, rng, sigma=0.0012)
        short_deadline = time.time() + 0.45
        cand_seq, cand_val = _smooth_subgradient_opt(
            cand,
            short_deadline,
            max_iters=120_000,
            eta0=2.5,
            beta_single=10.0,
            max_active=8,
        )
        print(f"[Pert‑SG] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"  >>> New best after perturb‑SG = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
