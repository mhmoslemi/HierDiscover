"""
Propose a step‑function sequence that (approximately) minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

for a non‑negative vector a.  The optimiser works in three phases:

1. **Broad seeding** – generate a few thousand analytically‑parameterised
   families (uniform, triangular, sine‑window, Gaussian, exponential,
   beta‑shapes, mixtures, double‑Gaussians) together with many random
   Dirichlet draws.  All seeds are ℓ₁‑normalised and evaluated with the cheap
   surrogate 2·n·max(conv) (the exact objective for sum(a)=1).  The best
   ``top_k`` seeds are retained.

2. **Intensive local refinement** – each retained seed is improved by a
   *smooth‑max* L‑BFGS optimisation (log‑sum‑exp surrogate) with a long
   schedule of β values, followed by a *full‑active* exponentiated‑gradient
   mirror‑descent that works on the true max‑convolution, and a short
   simulated‑annealing pass.  The best of these refinements becomes the new
   global best.

3. **Final polishing** – a high‑budget L‑BFGS run (very large β), an
   extensive exponentiated‑gradient run (full active set) and a long
   simulated‑annealing refinement are performed on the current global best.

All work respects the 1000 s budget (a 2 s safety margin).  The final
sequence is stored in the global variable ``height_sequence_opt`` and is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:  # degenerate case – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – every function returns an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _double_gaussian(n: int, sigma_frac: float, offset_frac: float) -> np.ndarray:
    """Two symmetric Gaussians placed at distance offset_frac·(n‑1)."""
    sigma = sigma_frac * n
    mu1 = (n - 1) * (0.5 - offset_frac / 2.0)
    mu2 = (n - 1) * (0.5 + offset_frac / 2.0)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Operates in log‑soft‑max space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        # SciPy unavailable – return unchanged
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 5_000_000.0, 1_0_000_000.0
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smooth‑max via log‑sum‑exp
            exp_term = np.exp(beta_val * (conv - shift))
            sum_exp = exp_term.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_term / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length = 2n‑1 + n‑1 = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # allow early cancellation when deadline passes
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      full_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)   # stay in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active indices
        if full_active:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]
        else:
            active = np.array([int(conv.argmax())], dtype=np.int64)

        # sub‑gradient (average over active shifts)
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # diminishing learning rate
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        t_frac = attempts / max_attempts
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence
    in the global variable ``height_sequence_opt``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse previous solution if it exists, otherwise n=4000
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_opt", None)
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.shape[0] > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
                n = arr.shape[0]
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 4000
    print(f"\n--- Optimisation started (n={n}) ---")

    # --------------------------------------------------------------
    # 1. Broad seeding – generate a large, diverse pool of candidates
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.10))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))

    # exponential decays (both sides)
    rates = np.logspace(-4, -0.2, 25)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # beta‑shapes (moderate grid)
    alphas = np.linspace(0.1, 6.0, 8)
    betas = np.linspace(0.1, 6.0, 8)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # double‑Gaussian mixtures – a simple way of creating two separated bumps
    sigmas = [0.07, 0.12, 0.18]
    offsets = [0.2, 0.35, 0.5]
    for sigma in sigmas:
        for off in offsets:
            seeds.append(_double_gaussian(n, sigma_frac=sigma, offset_frac=off))

    # random Dirichlet draws – explore both sparse and dense regimes
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best if available
    if prev_arr is not None:
        seeds.append(prev_arr.copy())
        for _ in range(6):
            pert = prev_arr + 0.002 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation and selection of top seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start local refinement
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget to this seed
        remaining = deadline - time.time()
        slice_time = max(3.0, remaining * 0.07 / max(1, (top_k - si)))
        sub_deadline = time.time() + slice_time

        # (a) smooth‑max L‑BFGS with an extended β‑schedule
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0,
                                                          500.0, 1_000.0, 2_000.0,
                                                          5_000.0, 20_000.0, 50_000.0,
                                                          100_000.0, 200_000.0,
                                                          500_000.0, 1_000_000.0],
                                           max_iter_per_beta=2500)

        # (b) exponentiated‑gradient with *full* active‑set
        seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                           n,
                                           sub_deadline,
                                           max_iters=300_000,
                                           eta0=6.0,
                                           full_active=True)

        # (c) a short simulated‑annealing refinement
        seq_sa, val_sa = _simulated_annealing(seq_eg,
                                              n,
                                              sub_deadline - 0.2,
                                              rng,
                                              max_attempts=80_000,
                                              start_sigma=0.002,
                                              T_start=0.015,
                                              T_end=5e-8)

        # pick the best among the three refinements (including the original seed)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_lbfgs, _objective_surrogate(seq_lbfgs)),
            (seq_eg, val_eg),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final intensive polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing (very high β) ---")
        global_best_seq = _lbfgs_smooth_optimize(global_best_seq,
                                                 n,
                                                 deadline,
                                                 beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                                200.0, 500.0, 1_000.0,
                                                                2_000.0, 5_000.0,
                                                                20_000.0, 50_000.0,
                                                                100_000.0, 200_000.0,
                                                                500_000.0, 1_000_000.0,
                                                                5_000_000.0, 1_0_000_000.0],
                                                 max_iter_per_beta=5000)

        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Polish L‑BFGS] surrogate = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final exponentiated‑gradient polishing (full active) ---")
        global_best_seq, global_best_val = _exp_gradient_opt(global_best_seq,
                                                             n,
                                                             deadline,
                                                             max_iters=2_500_000,
                                                             eta0=5.0,
                                                             full_active=True)
        print(f"[Polish EG] surrogate = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polishing (high budget) ---")
        global_best_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                             n,
                                                             deadline,
                                                             rng,
                                                             max_attempts=300_000,
                                                             start_sigma=0.001,
                                                             T_start=5e-3,
                                                             T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_val = final_sa_val
            print(f"[Polish SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in later calls
    globals()["height_sequence_opt"] = global_best_seq.tolist()
    return global_best_seq.tolist()
