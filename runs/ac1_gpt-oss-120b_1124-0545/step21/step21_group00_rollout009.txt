    Construct a simple decreasing polynomial distribution a_i ∝ (1 - i/(n-1))^p.

    2·n·max(conv(a,a)) / (sum(a)^2) .

    ``evaluate_sequence``.  The construction is an analytical polynomial
    decreasing profile a_i ∝ (1 - i/(n-1))^p with p=3, normalised to sum=1.
    """
    # ------------------------------------------------------------------
    # Parameters – they can be changed without affecting correctness.
    # ------------------------------------------------------------------
    n = 4000                # length of the step function (same as reference)
    p = 3                   # polynomial exponent; p=3 yields a very low score

    # ------------------------------------------------------------------
    # Build the decreasing polynomial on a uniform grid.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)               # indices 0 … n‑1
    # unnormalised values: ((n‑1‑i)/(n‑1))**p  (i.e. (1 - i/(n‑1))**p)
    vals = ((n - 1 - i) / (n - 1)) ** p

    # Normalise to a probability vector (sum = 1).  This also guarantees
    # non‑negativity and satisfies the evaluator’s scaling invariance.
    vals /= vals.sum()

    # Convert to a plain Python list – the evaluator expects a mutable list.
    seq = vals.tolist()

    # ------------------------------------------------------------------
    # Optional sanity check: compute and display the true objective.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq)
        print(f"[Constructed] n={n}, p={p}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        # In the unlikely event the evaluator raises, we still return the
        # sequence (the driver will treat the failure as a very large score).
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # Store the sequence globally as required by the competition framework.
    globals()["height_sequence_1"] = seq
    return seq
