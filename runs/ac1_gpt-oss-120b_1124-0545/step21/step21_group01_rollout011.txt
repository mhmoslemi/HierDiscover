"""
Search for non‑negative step‑function coefficients that minimise
evaluate_sequence.  The algorithm builds a large, diverse pool of
probability vectors (uniform, arcsine, beta, Dirichlet, random
perturbations and convex combinations), keeps the best few, then
refines them with exponentiated‑gradient mirror‑descent and a fast
random‑local search.  Additional time is spent permuting the current
best vector – a cheap way of exploring the huge space of orderings.
The best vector is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""
import time
import math
import numpy as np
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# --------- low‑level utilities ----------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for n≈few thousand)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)).  Works for sequences that sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# --------- seed generators ----------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a

def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# --------- exponentiated‑gradient polish ----------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution using a sub‑gradient."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max entry: derivative w.r.t a_i involves the two
        # indices that contribute to the k‑th entry of the convolution.
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # each pair (i, rev_idx) appears once; derivative 2*a[rev_idx]
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            a = _softmax(v_local)

            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]

            grad_a = 2.0 * full[n - 1:n - 1 + n]      # d/d a of weighted sum

            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):

                                method='L-BFGS-B',

                                         'disp': False,
                                         'ftol': 1e-12})

        except KeyboardInterrupt:
            break
        except Exception as exc:  # pragma: no cover
            print("[L‑BFGS] exception:", exc)
            break

    return _softmax(v)

def _permute_search(base_seq: np.ndarray,

                    deadline: float,

smoothing (if SciPy is available), a cheap permutation refinement and a
final random‑local walk are performed.  The best sequence is stored in
the global variable ``height_sequence_1`` and returned as a plain list.

# ----------------------------------------------------------------------
# Low‑level numerical utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for the official objective when ``seq`` sums to 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors that sum to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient polish (mirror‑descent on the true max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 250_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Run sub‑gradient descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry of the convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                           max_iter_per_beta=12_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("   >>> Updated global best (L‑BFGS)")

    # ------------------------------------------------------------------
    # 4️⃣  Random permutation refinement – cheap but can reshuffle mass.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random permutation refinement ---")
        seq_perm, val_perm = _permute_search(best_seq, n, deadline,
                                             max_attempts=300_000)
        print(f"[Perm] surrogate = {val_perm:.6f}")
        if val_perm < best_val:
            best_val, best_seq = val_perm, seq_perm.copy()
            print("   >>> Updated global best (perm)")

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny random local search.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final local random search ---")
        seq_local, val_local = _local_random_search(best_seq, n, deadline,
                                                    attempts=400_000,
                                                    sigma=0.015)
        print(f"[Local] surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
