def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _arcsine(n: int) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)

def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2

        right = left[::-1]

        right = left[:-1][::-1]

def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    Layout (K = 5):
        w_0 … w_{K‑1}      (non‑negative)
        μ_0 … μ_{K‑1}      (in [0,1])
        σ_0 … σ_{K‑1}      (positive)
    The resulting vector is normalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            exp_beta = np.exp(beta_val * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)                     # weights over 2n‑1 entries

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]          # d/d a of Σ w·conv

            grad_a = grad_weighted * (2.0 * n)                  # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            it_total += 1
            eta = 0.30 / math.sqrt(it_total)                    # tuned constant
            log_a = log_a - eta * grad_log

            if it_total % 5000 == 0:
                cur_val = _objective_surrogate(_softmax(log_a))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(log_a).copy()

    final_seq = _softmax(log_a)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val

# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
