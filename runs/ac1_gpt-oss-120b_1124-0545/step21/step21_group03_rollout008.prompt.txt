def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _exp_decay(n: int, rate: float) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)

    idx = np.arange(1, n + 1, dtype=np.float64)

def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a simple linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val               # a temperature proportional to the objective
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights for the smooth max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

            if time.time() > deadline:
                raise KeyboardInterrupt()

                                x0=v,

            break

            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a
