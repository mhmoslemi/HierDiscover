"""
Propose a better step‑function sequence.

This implementation improves on the previous reference by
* performing a finer search of the power‑law exponent (grid + local refinement),
* adding very sparse Dirichlet starts (α=0.1 gives mass close to the ends),
* using an Adam optimizer on a *smooth* max‑convolution objective.
  The smoothness is controlled by a schedule of β values (1 → 10⁵), which
  gradually turns the smooth‑max into the true ℓ∞‑norm.
* optional L‑BFGS smoothing (if SciPy is available) is kept as a final polish.

All optimisation is performed on the probability simplex (the vector is
represented via a soft‑max of unconstrained log‑weights).  The routine keeps
track of the best exact (non‑smooth) objective seen during the run and finally
returns the best sequence found before the deadline.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the final L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Exact surrogate (used for fast comparisons)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence already normalised to
    L₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max – gradient w.r.t. the *probability* vector a
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """Smooth max‑convolution objective (value) and its gradient w.r.t. a.
    Uses log‑sum‑exp smoothing with parameter β."""
    n = a.shape[0]
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_c = conv.max()
    # smooth max = shift + logsumexp(beta*(conv-shift))/beta
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. a
    w = exp_beta / sum_exp                     # weights, length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 2n‑1
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice of length n
    return obj, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_total_iters: int = 300_000,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam on the smooth max‑convolution. Returns best exact (non‑smooth)
    objective seen and the corresponding sequence."""
    # β schedule – we start very smooth and end with a very sharp approximation.
    beta_schedule = [0.5, 2.0, 10.0, 50.0, 200.0, 1_000.0,
                     5_000.0, 20_000.0, 100_000.0, 500_000.0]

    # Number of iterations per β – roughly equal share.
    iters_per_beta = max_total_iters // len(beta_schedule)

    log_a = np.log(start_seq + 1e-30)          # log‑softmax parameters
    m = np.zeros(n, dtype=np.float64)          # first moment
    v = np.zeros(n, dtype=np.float64)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    it = 0

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            it += 1

            # current probability vector
            a = _softmax(log_a)

            # exact objective (used for bookkeeping)
            conv = _conv_fft(a, a)
            exact_val = 2.0 * n * np.max(conv)
            if exact_val < best_val:
                best_val = exact_val
                best_seq = a.copy()

            # smooth objective + gradient
            _, grad_a = _smooth_obj_and_grad(a, beta)

            # gradient w.r.t. log‑softmax parameters (chain rule)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr0 / (np.sqrt(it) + 1.0)
            log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # Final exact evaluation of the last iterate (in case it improved)
    a = _softmax(log_a)
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()

    return best_seq, best_val

# Optional L‑BFGS smoothing (identical to the reference implementation)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Smooth‑max optimisation via L‑BFGS‑B (requires SciPy)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Refined power‑law search (grid + local refinement)

def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped distribution (p ≈ optimal) using a fine grid."""
    # coarse scan
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)
    p_vals = np.linspace(0.0, 5.0, 101)    # step = 0.05
    for p in p_vals:
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # local refinement around best p
    # start from the p that gave the current best
    p_center = p_vals[np.argmin([_objective_surrogate(_power_law(n, p)) for p in p_vals])]
    step = 0.05
    for _ in range(6):
        if time.time() > deadline:
            break
        improved = False
        for delta in (-step, step):
            p = p_center + delta
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
                improved = True
        if not improved:
            step *= 0.5
        else:
            # keep the step size for the next refinement round
            pass
    return best_seq

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
