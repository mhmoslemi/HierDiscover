    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set indices
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            active = np.where(conv >= max_val - tolerance * max_val)[0]

        # compute subgradient g with vectorized approach
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)

def _search_mix(global_seq, top_seqs, n, deadline):
    best_seq = global_seq.copy()
    best_val = _objective_surrogate(best_seq)
    rng = np.random.default_rng(0)
    # try random convex combos of up to 3 sequences
    seqs = [global_seq] + top_seqs
    while time.time() < deadline:
        # random number of components 2 or 3
        k = rng.integers(2, len(seqs)+1)
        idxs = rng.choice(len(seqs), size=k, replace=False)
        weights = rng.random(k)
        weights = weights / weights.sum()
        mix = np.zeros(n)
        for w, idx in zip(weights, idxs):
            mix += w * seqs[idx]

        val = _objective_surrogate(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()

  * ranks them by the cheap surrogate 2·n·max(conv(a,a)) (note that the
    true objective is scale‑invariant, so we always keep ∑a = 1);
  * refines the best *top_k* seeds with a mirror‑descent optimiser that
    works on the exact max‑convolution.  The mirror‑descent uses an
    Adam‑style adaptive stepsize in log‑space (i.e. exponentiated‑gradient
    on the simplex).  Afterwards a smooth‑max LBFGS polishing (if SciPy
    is available) and a short simulated‑annealing (SA) run are applied.
  * the resulting best sequence is further improved by line‑search
    mixing it with the other top seeds.
  * a final long‑run Adam‑EG polish finishes the search.

All steps respect the global time budget (default 1000 s) and the best
sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (for LBFGS and Kaiser window)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.special import i0 as _i0
except Exception:  # pragma: no cover
    _i0 = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (Adam‑style) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_adam(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           single_active: bool = True,
                           tol_rel: float = 1e-12):
    """
    Mirror‑descent (exponentiated‑gradient) with Adam‑style adaptive step.
    Works in log‑space and respects the simplex constraints.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)               # initialise log‑softmax variables
    m = np.zeros_like(log_a)               # first moment
    v = np.zeros_like(log_a)               # second moment
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx = np.arange(n, dtype=np.int64)     # reused index vector

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # --- active set -------------------------------------------------
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps_abs = max_val * tol_rel
            active = np.where(conv >= max_val - eps_abs)[0]

        # --- sub‑gradient of the max ------------------------------------
        g = np.zeros_like(a)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size          # exact averaged sub‑gradient

        # --- Adam update in log‑space -----------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = eta0 / (np.sqrt(v_hat) + eps)

        log_a = log_a - lr * m_hat
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1200):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0,
            100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
            5_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                       # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.004,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mixing of top candidates (grid search over convex weights)
# ----------------------------------------------------------------------
def _mix_with_top(global_seq: np.ndarray,
                  top_seqs: list[np.ndarray],
                  n: int,
                  deadline: float):
    """
    Try simple convex mixes of the current global best with each of the
    top candidates.  A grid of weights w∈{0,0.1,…,1} is examined.
    Returns the best mixed sequence seen.
    """
    best_seq = global_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for other in top_seqs:
        for w in np.linspace(0.0, 1.0, 11):
            mix = w * global_seq + (1.0 - w) * other
            # each component already sums to 1 → mix also sums to 1
            val = _objective_surrogate(mix)
            if val < best_val:
                best_val = val
                best_seq = mix.copy()
            if time.time() > deadline:
                return best_seq, best_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in the
    global variable ``height_sequence_1``, and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    margin = 5.0                               # safety margin for the deadline
    deadline = start_time + budget_s - margin

    # --------------------------------------------------------------
    # Problem size (moderate for fast FFTs)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if available (helps continuity)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()    # normalised copy
            else:
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation – a very diverse family sweep
    # --------------------------------------------------------------
    seeds = []

    # Basic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_kaiser_window(n, beta=6.0))
    seeds.append(_kaiser_window(n, beta=12.0))

    # Beta‑shape grid (moderately dense)
    alphas = np.linspace(0.1, 8.0, 12)
    betas = np.linspace(0.1, 8.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Gaussian widths
    for sigma_frac in np.linspace(0.07, 0.35, 12):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays (log‑spaced)
    for rate in np.logspace(-5, -0.3, 14):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 5.5, 14):
        seeds.append(_power_law(n, p))

    # Logistic families
    for scale in [0.5, 1.0, 2.0]:
        for mid in [n * 0.3, n * 0.5, n * 0.7]:
            seeds.append(_logistic_shape(n, scale, mid))

    # Random Dirichlet draws (sparse to dense)
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform + exponential (different mixing weights)
    for rate in np.logspace(-5, -0.5, 6):
        exp_seq = _exp_decay(n, rate)
        for alpha in (0.0, 0.1, 0.3, 0.6):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Perturbations of the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            pert = prev_arr + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        per_seed_budget = remaining * 0.15
        sub_deadline = time.time() + per_seed_budget

        # ---- (a) Adam‑EG (single active) ---------------------------------
        seq_eg, val_eg = _exp_gradient_opt_adam(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=350_000,
                                                eta0=6.0,
                                                single_active=True)

        # ---- (b) L‑BFGS smoothing (if time permits) --------------------
        seq_lbfgs = seq_eg
        val_lbfgs = val_eg
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0,
                                                              5_000_000.0],
                                               max_iter_per_beta=1800)
            val_lbfgs = _objective_surrogate(seq_lbfgs)

        # ---- (c) short simulated annealing -------------------------------
        seq_sa, val_sa = seq_lbfgs, val_lbfgs
        if time.time() < sub_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  sub_deadline,
                                                  rng,
                                                  max_attempts=120_000,
                                                  start_sigma=0.003,
                                                  T_start=0.015,
                                                  T_end=1e-8)

        # ---- (d) pick the best among the four versions --------------------
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Mix with other top seeds (simple convex interpolation)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mixing with top seeds ---")
        mix_seq, mix_val = _mix_with_top(global_best_seq, top_seeds, n, deadline)
        if mix_val < global_best_val:
            global_best_seq = mix_seq
            global_best_val = mix_val
            print(f"[Mix] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final long‑run Adam‑EG polishing (single‑active)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam‑EG polishing (single‑active) ---")
        final_seq, final_val = _exp_gradient_opt_adam(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=5_000_000,
                                                      eta0=5.0,
                                                      single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final brief SA polishing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final short SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=100_000,
                                                         start_sigma=0.001,
                                                         T_start=5e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
