"""
Propose a step‑function sequence that (hopefully) beats the previous
record (≈ 1.5085) for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

All numbers are non‑negative and the ℓ₁‑norm is irrelevant (the
objective is homogeneous), so we keep the vectors ℓ₁‑normalised.
The procedure consists of three phases

1. **Massive seeding** – generate several thousand candidates from
   analytic families (beta‑shapes, exponentials, power‑laws, Dirichlet)
   and keep the best few according to the *exact* surrogate
   `2·n·max(conv)`.

2. **Heavy continuous optimisation** – for each top seed run a
   continuation L‑BFGS optimisation that smooths the “max’’ by a
   log‑sum‑exp with an ever‑increasing β, followed by a long
   exponentiated‑gradient mirror‑descent (single active index).  The
   best result among all seeds is retained.

3. **Final polishing** – a high‑budget simulated‑annealing run that
   accepts occasional worsening moves.  The final sequence is stored
   in the global variable ``height_sequence_1`` and returned.

All heavy work respects the supplied time budget (default 1000 s) and
regularly checks the deadline.  The implementation uses only NumPy
(and SciPy if available) and runs on a single thread.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑distribution shape on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (left‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i))  (right‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (power‑law, i starts from 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)   # stay in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active indices -----
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient -------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning rate -----
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        # SciPy not available – simply return the start vector
        return start_vec

    if beta_schedule is None:
        # a geometric progression from modest to huge β
        beta_schedule = np.geomspace(1.0, 1e7, num=12)

    # initialise log‑softmax variables
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # smooth approximation of max by log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # full conv of rev_a and w (length = 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # callback only to enforce deadline
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        t_frac = attempts / max_attempts
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous solution if available,
    # otherwise start with n=4000.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0 and prev_arr.sum() > 0.0:
                n = prev_arr.shape[0]
                prev_norm = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    print(f"\n--- Optimisation started (n={n}) ---")

    # ------------------------------------------------------------------
    # 1. Very broad seeding – thousands of candidates
    # ------------------------------------------------------------------
    seeds = []

    # ----- deterministic families -----
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # ----- beta‑shape grid (dense, focusing on edge‑heavy shapes) -----
    alphas = np.concatenate((
        np.geomspace(0.05, 0.5, 7, endpoint=False),
        np.linspace(0.5, 5.0, 9, endpoint=False),
        np.geomspace(5.0, 25.0, 7)
    ))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # ----- exponential decays (both sides) -----
    rates = np.geomspace(1e-4, 0.5, 30)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # ----- power‑law families -----
    ps = np.linspace(0.0, 6.0, 20)
    for p in ps:
        seeds.append(_power(n, p))

    # ----- random Dirichlet draws (wide range of concentrations) -----
    for conc in (0.0005, 0.001, 0.003, 0.006, 0.01, 0.02, 0.05, 0.1):
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ----- perturb previous best (if any) -----
    if prev_arr is not None:
        seeds.append(prev_norm)
        for _ in range(12):
            pert = prev_norm + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0:
                seeds.append(pert / pert.sum())

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds; keep the very best ones
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)      # exact surrogate (no extra cost)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavily optimise the best few seeds with continuation L‑BFGS
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float('inf')
    # allocate a modest slice of the remaining budget to each seed
    for si, seed_vec in enumerate(top_seeds[:5]):   # only the first few
        if time.time() >= deadline:
            break
        rem = deadline - time.time()
        # give each L‑BFGS run at most 60 s or a fraction of the remaining budget
        sub_deadline = time.time() + min(60.0, rem * 0.4)

        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                           beta_schedule=np.geomspace(1.0, 1e7, num=12),
                                           max_iter_per_beta=6000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        print(f"[L‑BFGS seed {si+1}/{len(top_seeds[:5])}] surrogate = {val_lbfgs:.6f}")

        if val_lbfgs < global_best_val:
            global_best_val = val_lbfgs
            global_best_seq = seq_lbfgs.copy()
            print(f"    *** New global best after L‑BFGS = {global_best_val:.6f}")

    # If for some reason no L‑BFGS run succeeded, fall back to the very best seed
    if global_best_seq is None:
        global_best_seq = top_seeds[0].copy()
        global_best_val = _objective_surrogate(global_best_seq)

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient refinement (single active index)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Intensive EG refinement (single‑active) ---")
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq.copy(),
                                           n,
                                           deadline,
                                           max_iters=2_500_000,
                                           eta0=6.0,
                                           single_active=True)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final SA polishing (high budget) ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              rng,
                                              max_attempts=500_000,
                                              start_sigma=0.001,
                                              T_start=5e-3,
                                              T_end=1e-9)
        if val_sa < global_best_val:
            global_best_seq = seq_sa
            global_best_val = val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
