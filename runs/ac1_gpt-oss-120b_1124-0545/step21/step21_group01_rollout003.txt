def _adam_mirror_descent(start_seq, n, deadline, top_k=7):
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    eta = 0.1
    best_seq = None
    best_val = np.inf
    t = 0
    rng = np.random.default_rng(12345) // maybe use seed.
    while time.time() < deadline:
        t += 1
        a = _softmax(log_a)   # ensure sum 1
        conv = _conv_fft(a, a)
        # compute top indices
        top_k_local = min(top_k, conv.size)
        idx = np.argpartition(-conv, top_k_local-1)[:top_k_local]
        # weight vector: uniform on these indices
        w = np.zeros_like(conv)
        w[idx] = 1.0 / top_k_local
        # gradient of weighted sum
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
        # gradient w.r.t log a
        grad_log = a * (grad - np.dot(grad, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = eta * m_hat / (np.sqrt(v_hat) + epsilon)
        log_a = log_a - update

        # occasional random perturbation
        if t % 5000 == 0:
            # add tiny noise
            log_a += 0.01 * rng.normal(size=n)
            # re-normalize
            a = _softmax(log_a)
            log_a = np.log(a + 1e-30)

        # check current surrogate bound (max conv)
        cur_val = 2.0 * n * np.max(conv)  # surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # early break if best_val already < target? not needed.

    return best_seq, best_val

3️⃣  A light‑weight random local walk that transfers a small amount of
    mass between two coordinates and keeps any improvement.

The whole run respects the 1000 s budget, updates the global variable
``height_sequence_1`` with the best sequence found and finally prints the
exact value reported by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (equal to evaluate_sequence when sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k weights[k]·(a*a)[k] efficiently via FFT.
    ``weights`` has length 2·n‑1.
    """
    n = len(a)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, weights)
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# ADAM mirror‑descent on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_mirror_descent(start_seq: np.ndarray,
                         deadline: float,
                         top_k: int = 7,
                         eta: float = 0.12,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8,
                         perturb_interval: int = 8000,
                         perturb_scale: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent in log‑softmax space using ADAM.
    The gradient is taken w.r.t. a uniform weight on the ``top_k``
    largest convolution entries (a smooth surrogate of the true max).
    """
    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)           # initialise in log‑softmax space
    m = np.zeros_like(log_a)                    # first moment
    v = np.zeros_like(log_a)                    # second moment
    best_seq = None
    best_val = np.inf
    it = 0
    rng = np.random.default_rng(123456)

    while time.time() < deadline:
        it += 1
        # ------- current point -------
        a = _softmax(log_a)                     # a lives on the simplex
        conv = _conv_fft(a, a)                  # full convolution (len 2n‑1)
        # ------- top‑k indices -------
        tk = min(top_k, conv.size)
        idx = np.argpartition(-conv, tk - 1)[:tk]
        # ------- weight vector -------
        w = np.zeros_like(conv)
        w[idx] = 1.0 / tk                        # uniform weight on the chosen entries
        # ------- gradient of the weighted sum -------
        grad = _weighted_gradient(a, w)          # ∇_a Σ w_k·conv[k]
        # ------- mirror‑gradient -------
        grad_log = a * (grad - np.dot(grad, a))   # d/ d log_a of the objective
        # ------- ADAM updates -------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step

        # occasional random jitter to break symmetry / escape flats
        if it % perturb_interval == 0:
            log_a += perturb_scale * rng.normal(size=n)

        # keep the best exact surrogate seen so far
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    # In the unlikely case the loop never entered (deadline already passed)
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random local refinement (move a small amount from i to j)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 120_000,
                         sigma: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Simple stochastic walk on the simplex: repeatedly transfer a tiny
    fraction of mass from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng(999)
    n = len(start_seq)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method combines a fast ADAM mirror‑descent
    on a smoothed max‑convolution surrogate with a final random local
    refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size (may be overridden by the caller)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load a previously stored solution (if it matches the current size)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                init_seq = arr / arr.sum()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        init_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 0️⃣  Quick evaluation of a handful of random Dirichlet seeds
    # ------------------------------------------------------------------
    best_seq = init_seq.copy()
    best_val = _objective_surrogate(best_seq)
    print("\n--- Seed evaluation ---")
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            val = _objective_surrogate(vec)
            print(f" Dirichlet(conc={conc:4.2f}) surrogate = {val:.6f}")
            if val < best_val:
                best_seq, best_val = vec, val
                print("   >>> New seed best")
    print(f"[Seeds] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Main ADAM mirror‑descent phase (dominant part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_deadline = time.time() + remaining * 0.85
        print("\n--- ADAM mirror‑descent (smooth max surrogate) ---")
        seq_adam, val_adam = _adam_mirror_descent(best_seq,
                                                  phase_deadline,
                                                  top_k=9,
                                                  eta=0.12,
                                                  perturb_interval=7000,
                                                  perturb_scale=0.017)
        print(f"[ADAM] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam

    # ------------------------------------------------------------------
    # 2️⃣  Final local stochastic refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq,
                                                    deadline,
                                                    attempts=150_000,
                                                    sigma=0.02)
        print(f"[Local] best surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_seq, best_val = seq_local, val_local

    # ------------------------------------------------------------------
    # 3️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the result for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
