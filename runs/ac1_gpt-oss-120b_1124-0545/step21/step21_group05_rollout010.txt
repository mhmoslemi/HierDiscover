def _project_simplex(v):
    # v is numpy array
    # returns projection onto simplex {x >= 0, sum = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved step‑function optimiser.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0

After normalising the vector to sum = 1 the objective is simply
`2·n·max(conv(a,a))`.  The optimiser works in three stages:

1️⃣  generate a very large pool of analytically parametrised seeds
    (uniform, triangular, sine‑windows, Gaussians, exponentials,
    power‑laws, beta‑shapes and many Dirichlet draws).  All seeds are
    ℓ₁‑normalised; the 6 best (according to the surrogate objective)
    are retained.

2️⃣  each retained seed is locally refined by (a) a deterministic ordering,
    (b) a short permutation‑hill‑climb and (c) a mirror‑descent (exponentiated
    gradient) optimiser that uses the *exact* sub‑gradient of the active
    convolution entry (the diagonal term is treated correctly).  The best
    refinement becomes the new global best.

3️⃣  a long exponentiated‑gradient polishing run is performed on the global
    best until the time budget expires.  Random perturbations are repeatedly
    tried in the remaining time to escape possible local minima.

All heavy lifting is performed with an FFT‑based convolution; the code
keeps a small safety margin and stores the final sequence in the global
variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser with exact sub‑gradient
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (guaranteed feasibility).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑parameters (softmax will enforce sum=1 and non‑negativity)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active convolution index

        # --- exact sub‑gradient of max(conv) w.r.t a ---
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        # off‑diagonal entries contribute twice
        g[idx_i] = 2.0 * a[rev_idx]

        # diagonal (i == rev_idx) must be counted only once
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent update in log‑space
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)               # <g, a>
        grad_log = a * (g - dot)         # ∇_log objective
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two entries or adds a tiny Gaussian perturbation.
    Accepts worsening moves according to a simple annealing schedule.
    Returns the best sequence observed.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                         # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a large pool of analytic seeds,
    keeps the few best, refines each with permutation hill‑climbing and
    exponentiated‑gradient mirror descent, and finally spends the remaining
    time polishing the global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off between expressivity
    # and computational cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helpful for continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of analytical seeds
    # ------------------------------------------------------------------
    seeds = []

    # a few deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 50):
        seeds.append(_sine_window(n, gamma))

    # Gaussian – sweep sigma fraction
    for sigma_frac in np.linspace(0.05, 0.35, 25):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians – sweep centre and width
    for mu_frac in np.linspace(0.15, 0.85, 12):
        for sigma_frac in np.linspace(0.05, 0.30, 10):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0.5, 60):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweeps
    for p in np.linspace(0.0, 6.0, 30):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.1, 10.0, 12)
    betas  = np.geomspace(0.1, 10.0, 12)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # many random Dirichlet draws with a variety of concentration parameters
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(8):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        # ensure positivity and ℓ₁‑normalisation
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))               # we will invest more time in few seeds
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # give ~10 % of what remains to this seed (but never less than 15 s)
        share = max(15.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) deterministic ordering (descending) – a cheap heuristic
        seq_desc = np.sort(seed_vec)[::-1]

        # b) short permutation hill‑climb on the original seed
        seq_hc, val_hc = _perm_hill_climb(seed_vec, n,
                                          sub_deadline,
                                          rng,
                                          max_iters=140_000)

        # c) exponentiated‑gradient polish on three variants
        variants = [seed_vec, seq_desc, seq_hc]
        local_best_seq = None
        local_best_val = np.inf

        var_deadline_start = time.time()
        for vi, var in enumerate(variants):
            if time.time() >= sub_deadline:
                break
            # split remaining time equally among the remaining variants
            time_left = sub_deadline - time.time()
            var_share = time_left / (len(variants) - vi)
            var_deadline = time.time() + var_share
            seq_opt, val_opt = _exp_gradient_opt(var, n,
                                                 var_deadline,
                                                 max_iters=600_000,
                                                 eta0=6.0)
            if val_opt < local_best_val:
                local_best_val = val_opt
                local_best_seq = seq_opt

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (exponentiated gradient) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=3_500_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for final eval
        # generate a modestly perturbed copy of the current best
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 8.0          # short burst
        seq_tmp, val_tmp = _exp_gradient_opt(perturb, n,
                                             sub_deadline,
                                             max_iters=400_000,
                                             eta0=6.0)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
