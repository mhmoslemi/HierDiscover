def proj_simplex(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)

def projected_subgradient(start, n, deadline, max_iters=5_000_000, eta0=1.0):
    a = start.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # Compute convolution to find max offset
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # gradient g of max conv w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k - i
        g[i] = 2.0 * a[rev_idx]
        # step size
        eta = eta0 / math.sqrt(it)
        # gradient descent step
        a = a - eta * g
        # project onto simplex
        a = proj_simplex(a)
        # compute objective
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best = a.copy()
    return best, best_val

def proj_simplex(v):
    """Project onto probability simplex (sum=1, v>=0) using sorting O(n log n)."""
    # reference: https://arxiv.org/abs/1101.6081
    v = np.asarray(v, dtype=np.float64)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    # find rho: last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        theta = 0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

# Project onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Reference: Duchi et al. “Efficient Projections onto the L1‑Ball”.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform vertex
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)          # stay on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` function.  The best sequence is kept in the global
    variable ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the previous run.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Load previous solution or initialise an exponential‑decay shape.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
    else:
        i = np.arange(n, dtype=np.float64)
        rate = 0.003
        a = np.exp(-rate * i)
        seq0 = a / a.sum()

    # Ensure we are on the probability simplex
    seq0 = _proj_simplex(seq0)

    print(f"[Init] surrogate = {_objective_surrogate(seq0):.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent with a few random restarts.
    # ------------------------------------------------------------------
    # reserve ~70 % of the total budget for this phase
    subgrad_deadline = min(deadline - 30.0, start_time + 0.70 * budget_s)

    best_seq = seq0.copy()
    best_val = _objective_surrogate(best_seq)

    # First deterministic run
    seq_a, val_a = _proj_subgradient_opt(best_seq, n, subgrad_deadline,
                                         max_iters=800_000, eta0=2.0)
    print(f"[Subgrad‑run0] best surrogate = {val_a:.6f}")
    if val_a < best_val:
        best_seq, best_val = seq_a, val_a

    # Additional stochastic restarts (as many as time permits)
    restart = 0
    while time.time() < subgrad_deadline and restart < 4:
        restart += 1
        # Dirichlet with small concentration produces a sparse random start
        dirichlet_alpha = np.full(n, 0.1)
        pert = rng.dirichlet(dirichlet_alpha)
        # Blend a little with the current best to stay near a good basin
        start = 0.6 * best_seq + 0.4 * pert
        start = _proj_simplex(start)

        seq_r, val_r = _proj_subgradient_opt(start, n, subgrad_deadline,
                                             max_iters=300_000, eta0=2.0)
        print(f"[Subgrad‑restart {restart}] surrogate = {val_r:.6f}")
        if val_r < best_val:
            best_seq, best_val = seq_r, val_r

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=2_000_000,
                                           eta0=5.0)
        print(f"[EG] surrogate after polishing = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS smoothing of a log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 5.0:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_seq, best_val = seq_lb, val_lb

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided secure function.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
