"""
propose_candidate
-----------------

Search for a non‑negative step‑function (height vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution.  The routine loads any
previously discovered best vector stored in the global variable
``height_sequence_1`` (if present), normalises it to ℓ₁‑norm 1 and then
performs two intensive refinement phases that respect the overall
1000 s budget:

1. **Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution**  
   – single‑active shift (the current worst shift) – long run (up to a
   few million iterations) with a decaying learning rate.

2. **Simulated‑annealing fine‑tuning** – Gaussian‑like transfers of a
   tiny amount of mass between random indices, accepting worsening
   moves with an exponential temperature schedule.

Both phases keep the vector on the probability simplex, guaranteeing
non‑negativity and scale‑invariance.  The best vector found is stored
in ``height_sequence_1`` and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Single‑active shift for speed.  Returns the best sequence (ℓ₁‑normalised)
    encountered and its surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # initialise log‑weights safely
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and worst shift
        conv = _conv_fft(a, a)
        k = int(conv.argmax())                     # single active shift

        # gradient of max‑convolution w.r.t. a (ignoring constant 2·n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        # g[idx] = a[k - idx]  (vectorised)
        g[idx] = a[k - idx]

        # learning‑rate schedule
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Simple SA: transfer a tiny amount of probability mass between two
    random coordinates.  Temperature and perturbation size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        # pick donor index with some mass
        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # amount to move: a small fraction of a[i]
        amt = sigma * rng.random() * a[i]          # 0 ≤ amt ≤ sigma·a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        # evaluate
        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    The function loads any previously computed ``height_sequence_1``,
    runs a long exponentiated‑gradient descent followed by simulated‑annealing
    and finally returns the best sequence (as a Python list).
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Initialise – use previous solution if available, otherwise uniform
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            s = arr.sum()
            if s <= 0.0:
                raise ValueError
            seq0 = arr / s                     # ℓ₁‑normalise
            n = seq0.shape[0]
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    print(f"\n=== Optimisation started – n = {n} ===")
    print(f"Initial surrogate value: {_surrogate_obj(seq0):.6f}")

    # --------------------------------------------------------------
    # Phase‑1: long exponentiated‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate roughly 70 % of the remaining budget to EG
    eg_deadline = time.time() + max(5.0, 0.70 * remaining)
    print("\n--- Phase 1: Exponentiated‑gradient (long) ---")
    seq_eg, val_eg = _exp_gradient_opt_long(seq0, eg_deadline, eta0=5.0,
                                            max_iters=5_000_000)
    print(f"[EG] surrogate = {val_eg:.6f}")

    # --------------------------------------------------------------
    # Phase‑2: simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_deadline = time.time() + remaining
        print("\n--- Phase 2: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(seq_eg, sa_deadline, rng,
                                                     max_attempts=500_000,
                                                     T_start=0.02,
                                                     T_end=1e-8,
                                                     sigma_start=0.003)
        print(f"[SA] surrogate = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_eg, val_eg

    # --------------------------------------------------------------
    # Choose the best of the two refinements (they are the same object
    # hierarchy, but we keep the minimum just in case)
    # --------------------------------------------------------------
    if val_sa < val_eg:
        best_seq = seq_sa
        best_val = val_sa
    else:
        best_seq = seq_eg
        best_val = val_eg

    print(f"\n=== Final surrogate after polishing = {best_val:.6f} ===")

    # --------------------------------------------------------------
    # True evaluation (might be slightly different due to rounding)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
