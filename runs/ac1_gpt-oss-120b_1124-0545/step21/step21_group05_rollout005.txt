"""
search_step_function.py

A compact, time‑aware optimiser for the “step‑function” challenge.

The optimiser builds a diverse pool of analytic shapes, keeps the best few,
and refines each seed with a *smooth max‑convolution* mirror‑descent that
optimises the log‑sum‑exp surrogate of the peak convolution.  The smooth
objective spreads the gradient over many convolution entries (instead of the
single active entry used by a plain sub‑gradient), allowing the algorithm to
flatten the convolution profile more effectively.  After a global polishing
run, a short permutation‑based hill‑climb is applied for a final tweak.

All arithmetic is performed in the probability simplex (∑aᵢ = 1, aᵢ ≥ 0);
the objective `2·n·max(conv(a,a))` equals the evaluation function for any
scale of the vector, so the optimiser works directly on the surrogate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – full linear convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    """Gaussian with adjustable centre (mu_frac∈[0,1])."""
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Normalisation & quick surrogate helper
# ----------------------------------------------------------------------
def _norm_and_surrogate(vec: np.ndarray) -> tuple[np.ndarray, float]:
    """Clip to ≥0, normalise to sum = 1 and return the surrogate value."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return None, np.inf
    vec = vec / s
    return vec, _objective_surrogate(vec)


# ----------------------------------------------------------------------
# Mirror‑descent on the *smooth* max‑convolution (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             betas: list[float] | None = None,
                             max_iters_per_beta: int = 10_000,
                             eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth surrogate
        f_β(a) = 2·n·(1/β)·log Σₖ exp(β·convₖ(a,a))
    using exponentiated‑gradient (mirror descent) in log‑probability space.
    As β→∞ the surrogate approaches the true max‑convolution.
    """
    if betas is None:
        betas = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            2_000.0, 5_000.0, 20_000.0, 80_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    a = np.copy(start_seq)
    # initialise log‑probability safely
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for beta in betas:
        if time.time() > deadline:
            break

        for it in range(1, max_iters_per_beta + 1):
            if time.time() > deadline:
                break

            # rebuild a from log_a (softmax keeps the simplex constraints)
            a = _softmax(log_a)

            # ----- smooth max -------------------------------------------------
            conv = _conv_fft(a, a)                # length = 2n‑1
            max_c = np.max(conv)                  # shift for numerical stability
            diff = conv - max_c                    # ≤ 0
            # weight of each convolution entry in the smooth max
            exp_beta = np.exp(beta * diff)        # max term becomes 1
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                # softmax over conv entries

            # ----- gradient wrt a --------------------------------------------
            rev_a = a[::-1]
            # convolution of rev_a with the weight vector
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # extract the relevant slice

            # ----- mirror‑descent update (log‑space) -------------------------
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)         # gradient on the simplex

            eta = eta0 / math.sqrt(it)            # diminishing step size
            log_a = log_a - eta * grad_log

            # ----- keep best (periodic check) --------------------------------
            if it % 250 == 0:
                cur_val = _objective_surrogate(_softmax(log_a))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(log_a).copy()

    # final evaluation
    final_seq = _softmax(log_a)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based hill‑climbing (used only for a final tweak)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        # swap move (dominant)
        if rng.random() < 0.6:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        else:
            sigma = 0.0008 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm proceeds in three stages:
      1.  Generate a large, diverse pool of analytic shapes (uniform,
          triangular, sine / cosine windows, Gaussians, exponentials,
          power‑law, beta‑shapes and random Dirichlet draws).  Each shape is
          normalised to sum = 1 and evaluated cheaply using the surrogate.
      2.  Keep the top‑K seeds (default K = 12) and refine each of them with
          the smooth max‑convolution mirror‑descent optimiser
          – a fast first‑order method that spreads the gradient over the
          whole convolution profile.
      3.  Perform a long polishing run on the overall best seed, followed
          by a short permutation‑based hill‑climb for a final tweak.

    The routine respects the overall ``budget_s`` time budget (default
    1000 s) and stores the best sequence in the global variable
    ``height_sequence_1`` for possible continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Keep a few seconds as a safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension – 4000 offers a good balance of expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation (deterministic families + random Dirichlet)
    # ------------------------------------------------------------------
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n))

    # Sine windows – dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 40):
        seeds.append(_sine_window(n, gamma))

    # Gaussian – sweep sigma fraction
    for sigma_frac in np.linspace(0.04, 0.30, 25):
        seeds.append(_gaussian(n, sigma_frac))

    # Shifted Gaussians – sweep centre and width
    for mu_frac in np.linspace(0.15, 0.85, 12):
        for sigma_frac in np.linspace(0.04, 0.25, 10):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # Exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0.5, 55):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Power‑law
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # Beta‑shapes – log‑space grid
    alphas = np.geomspace(0.1, 10.0, 12)
    betas  = np.geomspace(0.1, 10.0, 12)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # Random Dirichlet draws (various concentrations)
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previous best (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds – keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec, val = _norm_and_surrogate(vec)
        if vec is not None:
            evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of each top seed with the smooth‑max optimiser
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # Allocate roughly 8 % of the *remaining* time to this seed
        share = remaining * 0.08 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] smooth‑max refinement budget ≈ {share:.1f}s")

        seq_opt, val_opt = _smooth_exp_gradient_opt(seed_vec,
                                                    n,
                                                    sub_deadline,
                                                    max_iters_per_beta=8_000,
                                                    eta0=6.0)

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt, val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long smooth‑max polishing ---")
        final_seq, final_val = _smooth_exp_gradient_opt(global_best_seq,
                                                        n,
                                                        deadline,
                                                        max_iters_per_beta=12_000,
                                                        eta0=5.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Polish] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short permutation hill‑climb for a final tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0  # ≤20 s
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq, global_best_val = seq_last, val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
