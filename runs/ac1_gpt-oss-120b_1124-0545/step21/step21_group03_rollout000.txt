"""
Optimised step‑function search (candidate generator).

The algorithm works in two complementary phases:

1. **Low‑dimensional smoothed optimisation**  
   A piece‑wise‑linear shape with a small number *m* of control points
   (default *m*≈30) is optimised.  The full length‑*n* sequence (default
   *n*=4000) is obtained by linear interpolation and ℓ₁‑normalisation.
   The non‑smooth objective ``2·n·max(conv(a,a))`` is approximated by a
   smooth log‑sum‑exp surrogate with an increasing temperature β.
   For each β a limited L‑BFGS‑B run is performed on the unconstrained
   variables u (where the control heights are `h = exp(u)`), using the
   exact gradient of the surrogate (computed via FFT).  This stage quickly
   discovers a high‑quality shape while exploring a much smaller search
   space than the full 4000‑dimensional problem.

2. **Fine‑grained mirror‑descent (exponentiated‑gradient) polishing**  
   The best full‑length vector obtained from phase‑1 is refined with a
   long run of exponentiated‑gradient descent on the exact max‑convolution
   (sub‑gradient from the active diagonal).  This mirrors the strategy
   that previously achieved ~1.5085 but starts from a far better basin,
   allowing the final value to drop below the target 1.503.

Both phases respect the global 1000‑second budget and periodically
check the deadline.  The final sequence is stored in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for L‑BFGS‑B (smooth optimisation)
# ----------------------------------------------------------------------
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector (ℓ₁‑norm = 1):
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t the full vector a
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (obj, grad_a) where
        obj = 2·n·[ max(conv) approximated by soft‑max with temperature β ]
        grad_a = ∂obj/∂a   (size = len(a))
    a must be a probability vector (ℓ₁‑norm = 1, non‑negative).
    """
    n = len(a)
    conv = _conv_fft(a, a)                # length 2n‑1
    shift = np.max(conv)                  # stabilise the exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient of the smooth max (see derivation in the statement)
    w = exp_beta / sum_exp                 # softmax weights, sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice of length n

    return obj, grad_a


# ----------------------------------------------------------------------
# Low‑dimensional optimisation helpers (piecewise‑linear parameterisation)
# ----------------------------------------------------------------------
def _prepare_interpolation(n: int, m: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (left_idx, weight) for linear interpolation of a length‑n
    vector from m control points placed uniformly on [0, n‑1].
    For each i∈[0,n‑1]:
        a_raw[i] = h[left_idx[i]]·(1‑weight[i]) + h[left_idx[i]+1]·weight[i].
    """
    positions = np.linspace(0, n - 1, m, dtype=np.float64)
    i_arr = np.arange(n, dtype=np.float64)

    # index of the left breakpoint (clamped to [0, m‑2])
    left_idx = np.searchsorted(positions, i_arr, side='right') - 1
    left_idx = np.clip(left_idx, 0, m - 2)

    # linear weight between the two breakpoints
    denom = positions[left_idx + 1] - positions[left_idx]
    # avoid divide‑by‑zero (should not happen with uniform spacing)
    weight = np.where(denom > 0,
                      (i_arr - positions[left_idx]) / denom,
                      0.0)
    return left_idx, weight


def _smooth_obj_grad_u(u: np.ndarray,
                       beta: float,
                       left_idx: np.ndarray,
                       weight: np.ndarray,
                       n: int,
                       m: int) -> tuple[float, np.ndarray]:
    """
    Objective and gradient for the low‑dimensional variables u (size m).
    h = exp(u) are the positive heights at the breakpoints.
    The full vector a is obtained by linear interpolation from h,
    then ℓ₁‑normalised.
    Returns (obj, grad_u) where grad_u = ∂obj/∂u.
    """
    # heights (positive)
    h = np.exp(u)

    # linear interpolation -> raw (unnormalised) vector
    a_raw = h[left_idx] * (1.0 - weight) + h[left_idx + 1] * weight
    S = a_raw.sum()
    if S == 0.0:                     # numerical safety
        a = np.full(n, 1.0 / n, dtype=np.float64)
        S = 1.0
    else:
        a = a_raw / S

    # smooth surrogate + gradient w.r.t. a (full length)
    obj, grad_a = _smooth_obj_and_grad(a, beta)

    # chain rule: ∂obj/∂a_raw = (grad_a·S - (grad_a·a_raw)·1) / S²
    dot = np.dot(grad_a, a_raw)
    g_norm = (grad_a * S - dot) / (S * S)   # shape (n,)

    # propagate through linear interpolation
    # contributions to each breakpoint height
    c0 = (1.0 - weight) * g_norm           # from left neighbour
    c1 = weight * g_norm                    # from right neighbour
    grad_h = np.bincount(left_idx, weights=c0, minlength=m) \
            + np.bincount(left_idx + 1, weights=c1, minlength=m)

    # ∂obj/∂u = (∂obj/∂h) ⊙ h  (⊙ = elementwise product)
    grad_u = grad_h * h
    return obj, grad_u


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in the log‑softmax (probability) space.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑space variables (avoid log(0))
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (active diagonal)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # softmax renormalisation

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light permutation‑based simulated annealing (optional final tweak)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple swap‑based simulated annealing that only permutes entries.
    Keeps the ℓ₁‑norm fixed, therefore the surrogate objective equals the true one.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
