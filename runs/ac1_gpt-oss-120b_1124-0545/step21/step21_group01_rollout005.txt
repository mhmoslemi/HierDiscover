def propose_candidate(seed=123, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # Build seeds
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Add a few Dirichlet draws
    for conc in (0.1, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Add previous solution if exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum()>0:
            seeds.append(arr / arr.sum())

    best_seq = None
    best_val = np.inf

    # Evaluate seeds
    print("\n--- Seed evaluation ---")
    for idx, s in enumerate(seeds):
        if time.time()>deadline: break
        seq = s.copy()
        seq = seq / seq.sum()
        val = _objective_surrogate(seq)
        print(f" Seed {idx+1}/{len(seeds)}: surrogate={val:.6f}")
        if val < best_val:
            best_val = val; best_seq = seq.copy()
    print(f"[Init] best surrogate={best_val:.6f}")

    # Allocate half time for Adam smoothing across seeds, then remaining for polishing
    # For each seed run a short Adam smoothing portion.
    time_per_seed = (deadline - time.time()) * 0.5 / len(seeds)
    for idx, seq in enumerate(seeds):
        sub_deadline = time.time() + time_per_seed
        # if insufficient time break
        if time.time()>deadline: break
        print(f"\n--- Adam smoothing on seed {idx+1} ---")
        seq_s, val_s = _adam_smoothing(seq, n, sub_deadline)
        print(f"    Adam best surrogate = {val_s:.6f}")
        if val_s < best_val:
            best_val, best_seq = val_s, seq_s.copy()
            print("    >>> Updated global best (Adam)")

    # Remaining time for EG polishing and possible local refinement
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated gradient polishing (exact) ---")
        polishing_deadline = deadline - 5.0
        remaining_time = polishing_deadline - time.time()
        max_iters = int(2_500_000 * remaining_time / (deadline - start_time))
        max_iters = max(250_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           max_iters=max_iters, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("    >>> Updated global best (EG)")

    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=120_000, sigma=0.02)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("    >>> Updated global best (local)")

    # Evaluate with official evaluate_sequence
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ~ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as e:
        print("Error in evaluate_sequence:", e)

    # Store global
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

   space, uses a schedule of increasing smoothing parameters (β) and a
   standard Adam step‑size schedule.  The best exact surrogate value seen
   during the run is retained.

2. **Exact exponentiated‑gradient polishing** – a sub‑gradient descent
   directly on the true max‑convolution (as in the reference code).  This
   phase finishes the optimisation with a large, time‑adapted number of
   iterations.

3. **Tiny random local search** – a final cheap stochastic walk on the
   simplex to squeeze any remaining improvement.

The final sequence (normalised to sum = 1) is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.  All steps are
timed so that the total running time never exceeds the supplied budget
(≤ 1000 s).  Only the standard library and NumPy are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    surrogate of   2·n·max(conv(a,a)).
    Returns the best sequence (as a probability vector) and its exact
    surrogate value seen during the run.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # start from a fairly low smoothing, finish with a very sharp one
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # a modest but fixed number of inner iterations per β
        max_iter_beta = 4000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n-1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15  # a modest learning rate that works well in practice
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasionally evaluate the *exact* surrogate to keep the best
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored in the global variable ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size (default 4000)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                # Normalise to a probability vector
                prev_vec = arr / arr.sum()
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # a few Dirichlet draws with different concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Initial evaluation of seeds, keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam smoothing on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth max) ---")
        remain = deadline - time.time()
        adam_deadline = time.time() + remain * 0.55
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣  Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0
        remaining_time = polishing_deadline - time.time()
        # scale the allowed number of iterations to the remaining fraction
        max_iters = int(2_500_000 *
                        remaining_time /
                        (deadline - start_time))
        max_iters = max(250_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=120_000,
                                                sigma=0.02)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
