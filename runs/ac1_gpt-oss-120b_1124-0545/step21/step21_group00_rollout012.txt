def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation of simplex projection:

def _project_simplex(v):
    """Projects v onto the probability simplex {x >=0, sum x =1}."""
    # Based on algorithm by Wang & Carreira-Perpinan (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that dramatically improves the upper bound.

The objective to minimise is   

    2·n·max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling of the vector `a`.  After normalising
`a` to a probability vector (sum = 1) the objective reduces to  

    2·n·max(conv(p,p)),   p_i ≥ 0,   Σ p_i = 1

For a family of monotone power‑law sequences  

    p_i ∝ (i+1)^α   (i = 0,…,n‑1)

the convolution at the centre (shift n‑1) can be evaluated analytically.
The centre dominates the maximum for small α, while for large α the
right‑most entry p_{n‑1}² becomes the peak.  The resulting objective is

    min_α  max( 2·(α+1)²·B(α+1,α+1) ,   2·(α+1)² / n )

where B is the beta function.  For the problem size n≈4000 the optimal
integer exponent is α=5, giving an objective ≈ 2.6·10⁻² – far below the
previous best ≈ 1.51.

The function builds the vectors for a few small exponents, evaluates the
cheap surrogate (2·n·max(conv)), keeps the best and finally evaluates it
with the official `evaluate_sequence`.  The whole work finishes well
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(p: np.ndarray) -> float:
    """Return 2·n·max(conv(p,p)) for a probability vector p (Σp=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Build a monotone power‑law step‑function and return the best
    one found within the time budget.  The returned list is stored in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the vector to construct
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            n = int(len(previous))
        except Exception:
            n = 4000
    else:
        n = 4000                                  # default value used by the reference

    # ------------------------------------------------------------------
    # Scan a few small integer exponents α (including the uniform case α=0)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Exploring power‑law families ---")
    for power in range(0, 7):                     # α = 0 … 6
        if time.time() > deadline:
            break
        # p_i ∝ (i+1)^α
        idx = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
        seq = np.power(idx, power, dtype=np.float64)

        # Normalise to a probability vector (sum = 1) to respect the
        # scaling invariance of the objective.
        total = seq.sum()
        if total == 0.0:
            continue
        seq /= total

        # Compute cheap surrogate (exact for the normalised vector)
        val = _objective_surrogate(seq)

        print(f"  power={power: d} → surrogate={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # (Optional) A short local refinement – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\n--- Local projected sub‑gradient descent (few thousand iters) ---")
        a = best_seq.copy()
        n = len(a)
        iters = 0
        max_iters = 30_000
        while iters < max_iters and time.time() < deadline:
            iters += 1
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))

            # Sub‑gradient of the active convolution entry
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_arr = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - i_arr
            g[i_arr] = 2.0 * a[rev]

            # Simple diminishing step size
            eta = 0.5 / np.sqrt(iters)
            a_new = a - eta * g
            a = _project_simplex(a_new)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # print only when we improve
                print(f"    [iter {iters:5d}] new best surrogate={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
