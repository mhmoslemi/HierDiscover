def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000

    # function to normalize vector to sum 1
    def normalize(v):
        v = np.maximum(v, 0)
        total = v.sum()
        if total == 0:
            v = np.full_like(v, 1.0/v.size)
        else:
            v = v / total
        return v

    # define list of initial vectors
    init_vectors = []
    # previous solution if present
    prev = globals().get('height_sequence_1')
    if isinstance(prev, list):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            init_vectors.append(normalize(arr))

    # uniform
    init_vectors.append(np.full(n, 1.0/n))

    # exponential decays
    for rate in [0.001, 0.002, 0.003, 0.005, 0.01]:
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * i)
        init_vectors.append(normalize(a))

    # random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        a = rng.gamma(alpha, 1.0, size=n)
        init_vectors.append(normalize(a))

    # random sparse: choose k random indices (k small)
    for k in [10, 20, 50]:
        a = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a[idx] = vals
        init_vectors.append(normalize(a))

    best_seq = None
    best_val = np.inf

    # iterate through init vectors, perform local optimization
    for base_seq in init_vectors:
        if time.time() > deadline:
            break
        # perform exponentiated gradient for a limited number of steps
        seq_opt, val_opt = _exp_gradient_opt(base_seq, n, deadline, max_iters=300_000, eta0=3.0)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt

        # after polished, maybe apply short simulated annealing
        if time.time() < deadline - 30:
            seq_ann, val_ann = _value_anneal(seq_opt, n, deadline - 5, rng, max_iters=200_000)
            if val_ann < best_val:
                best_val = val_ann
                best_seq = seq_ann

    # final L-BFGS smoothing if time permits
    if time.time() < deadline - 5 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_val = val_lb
            best_seq = seq_lb

    # scaling for evaluate_sequence
    final_seq = best_seq.tolist()
    try:
        true_val = evaluate_sequence(final_seq)
        print(f"[Result] best surrogate = {best_val:.6f}, true value = {true_val:.6f}")
    except Exception as e:
        print("[Error] evaluate_sequence exception:", e)

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
search_step_function.py

A multi‑restart stochastic optimiser for the step‑function problem.
The optimiser combines:
  * a diverse set of random initialisations (uniform, exponential, Dirichlet…)
  * a fast exponentiated‑gradient mirror‑descent on the exact max‑convolution
  * a short simulated‑annealing polishing phase
  * an optional high‑precision L‑BFGS refinement on a smooth log‑sum‑exp
    approximation of the max‑convolution.

All phases respect the time budget (default 1000 s).  The best sequence found
is stored in the global variable ``height_sequence_1`` for warm‑starting the
next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real signals)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq));  n = len(seq).  Exact for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on the exact max‑conv.)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on max‑convolution.
    ``start_seq`` must sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)          # work in log‑space

    n = len(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - i_vals
        g[i_vals] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing (swap + mass‑transfer) polish
# ----------------------------------------------------------------------
def _value_anneal(start_seq: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the simplex.
    Returns the best sequence found and its surrogate value.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        # 70 % mass‑transfer, 30 % simple swap
        if rng.random() < 0.7:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] == 0.0:
                continue
            delta = cur_seq[i] * rng.random() * 0.5   # move up to 50 % of source
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)

        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Smooth max via log‑sum‑exp and L‑BFGS‑B refinement
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt  # type: ignore
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           iterations_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises a smooth approximation of max‑convolution using L‑BFGS‑B.
    Operates in log‑softmax space (mirrors the simplex).
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         20_000.0, 50_000.0, 100_000.0,
                         200_000.0, 500_000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)       # log‑softmax parameters

    n = len(start_vec)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # current probability vector
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max = m + log(sum exp(beta*(conv-m))) / beta
            exp_term = np.exp(beta * (conv - m))
            sum_exp = exp_term.sum()
            obj = 2.0 * n * (m + np.log(sum_exp) / beta)

            # gradient w.r.t. a (derived analytically)
            w = exp_term / sum_exp                # soft‑weights for each index
            rev_a = a[::-1]
            # full convolution of rev_a with w (length = 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice central n entries

            # transform gradient to log‑softmax space (mirror‑descent formula)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': iterations_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Helper: normalise a raw vector to a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    tot = v.sum()
    if tot == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / tot

        4. final high‑precision L‑BFGS smoothing (if time permits),
        5. return the best sequence found.
    The globally best sequence is kept in ``height_sequence_1`` for the next call.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the previous run if possible.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Assemble a set of initial vectors.
    # ------------------------------------------------------------------
    candidates = []

    # warm‑start from previous solution
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(_normalise(arr))
        except Exception:
            pass

    # deterministic baselines
    candidates.append(np.full(n, 1.0 / n))                        # uniform
    i = np.arange(n, dtype=np.float64)
    for rate in [0.001, 0.002, 0.003, 0.005, 0.01]:
        a = np.exp(-rate * i)
        candidates.append(_normalise(a))

    # random Dirichlet draws (different concentration parameters)
    for alpha in [0.05, 0.1, 0.5, 1.0, 2.0]:
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # sparse random starts (few non‑zero entries)
    for k in [5, 10, 20, 50]:
        a = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.random(k)
        candidates.append(_normalise(a))

    # ------------------------------------------------------------------
    # 2. Loop over candidates, apply cheap mirror‑descent and anneal.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    for idx, cand in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"[Stage 1] candidate {idx+1}/{len(candidates)} – start surrogate = {_objective_surrogate(cand):.6f}")

        # a) exponentiated‑gradient polishing (fast, exact max‑conv)
        seq_eg, val_eg = _exp_gradient_opt(cand, deadline,
                                            max_iters=200_000,
                                            eta0=5.0)
        print(f"    EG → surrogate {val_eg:.6f}")

        # b) short simulated‑annealing polish (if time left)
        if time.time() < deadline - 30.0:
            seq_ann, val_ann = _value_anneal(seq_eg, deadline, rng,
                                             max_iters=150_000)
            print(f"    anneal → surrogate {val_ann:.6f}")
            cur_seq, cur_val = seq_ann, val_ann
        else:
            cur_seq, cur_val = seq_eg, val_eg

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
            print(f"   ** new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional high‑precision L‑BFGS smoothing on the current best.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, deadline,
                                        beta_schedule=None,
                                        iterations_per_beta=5000)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_val = val_lb
            best_seq = seq_lb
            print(f"   ** improved by L‑BFGS → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the secure function and bookkeeping.
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No feasible sequence generated.")

    final_list = best_seq.tolist()
    true_val = None
    try:
        true_val = evaluate_sequence(final_list)
        print(f"[Result] surrogate = {best_val:.6f}, real evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = final_list
    return final_list
