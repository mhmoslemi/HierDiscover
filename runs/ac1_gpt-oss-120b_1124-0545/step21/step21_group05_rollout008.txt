"""
optimised_step_function_v2.py

A self‑contained optimiser for the step‑function challenge.

Strategy
--------
1.  Initialise a diverse pool of candidate distributions (Dirichlet draws,
    a few analytic families and the previous best if available).  All vectors
    are ℓ₁‑normalised (sum = 1).

2.  For each candidate perform a *large* simulated‑annealing permutation
    search.  Swaps preserve the ℓ₁‑norm, therefore only the ordering of the
    multiset matters.  The temperature is linearly lowered from 0.5·value
    to 1e‑6·value while the surrogate objective
        `S(a)=2·n·max(conv(a,a))`
    is evaluated after every swap via a fast FFT‑based convolution.

3.  The best ordering from the annealing stage is polished with
    exponentiated‑gradient mirror descent on the exact non‑smooth objective.
    The update uses the sub‑gradient at the active convolution index and a
    decreasing step size `η₀/√t`.  The best sequence seen during the run is
    retained.

4.  The current global best (over all seeds) is finally polished for the
    remaining time with a longer EG run and a lightweight permutation tweak.

All heavy work respects the 1000 s budget.  The final sequence is stored in
the global variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities (FFT based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact value when the vector is ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a probability vector from a symmetric Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact non‑smooth objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on max‑convolution.  Returns the best sequence seen."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # stay in the interior of the simplex
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only indices that actually
        # contribute to the maximal convolution entry receive a non‑zero gradient)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing permutation optimiser
# ----------------------------------------------------------------------
def _permute_anneal(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Large‑scale simulated‑annealing on the ordering of a fixed multiset.
    Only swaps are performed, therefore the ℓ₁‑norm stays equal to 1.
    """
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule – linear in iteration index
    T_start = 0.5 * cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a simple swap
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (faster convergence on a smooth surrogate)
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000):
    """Smooth max‑convolution optimisation via L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smoothed max using log‑sum‑exp
            shifted = conv - max_c
            exp_beta = np.exp(beta_val * shifted)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in the original script)
            w = exp_beta / sum_exp                 # soft‑weights over convolution indices
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # pick the n relevant entries

            # map gradient to log‑space (mirror‑descent geometry)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Run a single seed through the whole pipeline
# ----------------------------------------------------------------------
def _process_seed(seed_vec: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    For a given initial probability vector:
        1. heavy simulated‑annealing on the ordering,
        2. exponentiated‑gradient polishing,
        3. optional L‑BFGS smoothing.
    Returns the best sequence and its surrogate value.
    """
    # ------------------------------------------------------------------
    # 1) permutation simulated‑annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate ~30 % of the remaining budget for the annealing phase
    anneal_deadline = time.time() + 0.30 * remaining
    seq_perm, val_perm = _permute_anneal(seed_vec, n,
                                         anneal_deadline,
                                         rng,
                                         max_iters=300_000)

    # ------------------------------------------------------------------
    # 2) exponentiated‑gradient polishing (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_eg, val_eg = _exp_gradient_opt(seq_perm, n, deadline,
                                           max_iters=600_000,
                                           eta0=7.0)
    else:
        seq_eg, val_eg = seq_perm, val_perm

    # ------------------------------------------------------------------
    # 3) optional L‑BFGS smoothing (fast, often <1 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline,
                                           max_iter_per_beta=2000)
        val_lbfgs = _surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # pick best of the three refinements
    candidates = [(seq_perm, val_perm),
                  (seq_eg, val_eg),
                  (seq_lbfgs, val_lbfgs)]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val
