def _exp_gradient_opt(seq, n, deadline, max_iters=..., eta0=3.0):
    a = np.copy(seq)   # ensures sum 1
    log_a = np.log(a+1e-30)
    # best record...
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient g for each i:
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k - idx_i
        # g_i = 2 * a_rev_idx
        g[idx_i] = 2.0 * a[rev_idx]
        # update
        eta = eta0 / math.sqrt(it+1)
        log_a -= eta * g
        a = _softmax(log_a)  # ensures sum=1
        # track best
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a step‑function minimising ``evaluate_sequence``.

The algorithm
----------------
1. Choose a moderate problem size (default ``n = 8000`` – larger than the
   reference run – to reduce discretisation error).
2. Build a diverse pool of *probability* vectors:
   * uniform,
   * an arcsine‑type law,
   * random symmetric Beta densities,
   * full‑ and sparse Dirichlet draws,
   * mixtures of many Gaussians,
   * Fourier‑softmax parametrisations.
   All vectors are normalised to sum 1.
3. Evaluate the cheap surrogate ``2·n·max(conv(a,a))`` for each seed and keep
   the best few (≤5).
4. Run a L‑BFGS optimiser on the smooth *log‑sum‑exp* surrogate,
   gradually sharpening the smooth max (β → ∞).  
   The optimiser works on the log‑softmax representation, guaranteeing the
   simplex constraints.
5. Polish the resulting candidate with an *exponentiated‑gradient* (mirror‑
   descent) that directly minimises the true max‑convolution objective.
6. Perform cheap stochastic perturbations of the current best solution and
   re‑run a short EG polish; accept any improvement.
7. Finally evaluate the best candidate with the official ``evaluate_sequence``
   and store it in the global ``height_sequence_1`` variable.

All phases respect the overall 1000 s budget.  The routine returns a plain
list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – required only for the L‑BFGS phase.
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Surrogate objective (scale‑invariant):
        2·n·max(conv(a,a))
    Assumes ``a`` is already normalised to sum 1.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law: 1/√(x(1‑x))."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = 1.0 / np.sqrt(i * (1.0 - i) + eps)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta density discretisation."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    # avoid infinities at the boundaries
    eps = 1e-12
    i = np.clip(i, eps, 1.0 - eps)
    a = (i ** (alpha - 1.0)) * ((1.0 - i) ** (beta - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    a /= a_sum
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, later normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """
    Build a probability vector by applying a softmax to a Fourier
    series: a_i ∝ exp( Σ_k θ_k φ_k(i) ).
    """
    logits = basis @ theta
    return _softmax(logits)


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000):
    """L‑BFGS‑B on the smooth surrogate (variables = log‑softmax)."""
    if _opt is None:
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0,
                         20_000.0, 50_000.0,
                         100_000.0, 200_000.0,
                         500_000.0, 1_000_000.0,
                         2_000_000.0, 5_000_000.0,
                         10_000_000.0]

    # initialise log‑softmax variables
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # shape (2n‑1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 3.0):
    """
    Mirror‑descent on the true max‑convolution.
    Operates on the log‑softmax representation (exponentiated gradient).
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]          # each unordered pair appears twice

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Perturbation helper
# ----------------------------------------------------------------------
def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise and re‑normalise onto the simplex."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid stochastic / gradient based search for the step‑function
    minimising ``evaluate_sequence``.
    The best vector found is stored in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a larger n reduces discretisation error but costs a
    # little more per iteration.  8000 works comfortably within the budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Retrieve the previous solution (if it matches our n)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]
    seeds.append(_arcsine(n))

    # Dirichlet seeds with a range of concentration parameters
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse Dirichlet (support sizes)
    for k in (80, 150, 300):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # Random symmetric Beta shapes (α,β) drawn from log‑uniform ranges
    for _ in range(20):
        alpha = 10 ** rng.uniform(-1.5, 1.5)   # ~[0.03, 30]
        beta = 10 ** rng.uniform(-1.5, 1.5)
        seeds.append(_beta_shape(n, alpha, beta))

    # Gaussian‑mixture seeds (K = 12 components)
    K_mix = 12
    for _ in range(8):
        params = np.concatenate([
            rng.random(K_mix),                     # raw weights (will be normalised)
            rng.random(K_mix),                     # means in [0,1]
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n, K=K_mix))

    # Fourier‑softmax seeds (20 basis functions)
    D_fourier = 20
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis = []
    for k in range(1, D_fourier // 2 + 1):
        basis.append(np.sin(2.0 * np.pi * k * i))
        basis.append(np.cos(2.0 * np.pi * k * i))
    basis = np.column_stack(basis)   # (n, D_fourier)

    for _ in range(6):
        theta = rng.normal(scale=1.0, size=D_fourier)
        seeds.append(_fourier_softmax(theta, basis))

    # If we have a previous solution, make a few perturbed copies
    if prev_arr is not None:
        for _ in range(4):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation and keep the top few candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    top_candidates = []      # (value, vector)

    print("\n--- Evaluating initial seeds (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # guarantee a proper simplex vector
        vec = vec / max(vec.sum(), 1e-30)
        val = _objective_surrogate(vec)
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
        top_candidates.append((val, vec.copy()))

    # keep only the five best seeds for the expensive refinement steps
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  L‑BFGS smoothing of the top candidates (if SciPy is available)
    # ------------------------------------------------------------------
    if _opt is not None:
        remaining = deadline - time.time()
        if remaining > 30.0:        # ensure we have enough time left
            sub_deadline = time.time() + remaining * 0.45
            print("\n--- L‑BFGS smoothing of top seeds ---")
            for rank, (val0, seed_vec) in enumerate(top_candidates, 1):
                if time.time() > sub_deadline:
                    break
                print(f"[L‑BFGS] start smoothing seed #{rank} (surrogate={val0:.6f})")
                seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                                   max_iter_per_beta=20000)
                val_lbfgs = _objective_surrogate(seq_lbfgs)
                print(f"[L‑BFGS] result surrogate = {val_lbfgs:.6f}")
                if val_lbfgs < best_val:
                    best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
                    print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact mirror‑descent (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:

        time_left = polish_deadline - time.time()

        max_iters = int(min(3_000_000,

        print("\n--- Exponentiated‑gradient polishing (full) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polish_deadline,
                                           max_iters=max_iters,
                                           eta0=4.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Stochastic refinement: perturb + short EG polish
    # ------------------------------------------------------------------
    sigma = 0.0025
    while time.time() < deadline - 2.0:

                                                short_deadline,
                                                max_iters=120_000,
                                                eta0=2.5)
        print(f"[Pert‑EG] surrogate = {cand_val:.6f}")
        if cand_val < best_val:

            print(f"  >>> New best after perturb‑EG = {best_val:.6f}")
        # slowly reduce the perturbation scale

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
