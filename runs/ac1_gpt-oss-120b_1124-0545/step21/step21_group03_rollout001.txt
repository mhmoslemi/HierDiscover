"""
Improved optimisation for the step‑function problem.

The pipeline:

1. **Rich seeding** – build a large, diverse pool of ℓ₁‑normalised vectors.
   Analytic families (exponential, symmetric exponential, power‑law,
   beta, linear ramps, quadratic ramps) are scanned over a fine grid of
   parameters, together with random Dirichlet draws and permutations of the
   previous best solution.

2. **Quick surrogate ranking** – evaluate the exact surrogate
   `2·n·max(conv(a,a))` for every seed and keep the best few (top‑k).

3. **Smooth max‑convolution optimisation** – for each top seed run a
   schedule of log‑sum‑exp smoothings with increasing temperature `β`
   and optimise the smoothed objective with SciPy’s L‑BFGS‑B.  The variable
   lives in the log‑softmax (probability) domain, guaranteeing non‑negativity
   and the ℓ₁‑constraint.

4. **Mirror‑descent polishing** – a long run of exponentiated‑gradient (EG)
   (mirror‑descent on the exact max‑convolution) refines the L‑BFGS result.

5. **Fine stochastic tweak** – a short simulated‑annealing phase that
   moves a small amount of mass between two positions (not only swaps).

All stages respect a global 1000 s deadline; the best vector found is stored
in the global variable `height_sequence_1` and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for the L‑BFGS smoothing stage.
# ----------------------------------------------------------------------
try:                                  # pragma: no cover
    import scipy.optimize as _opt
except Exception:                    # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective for ℓ₁‑normalised vectors (exact!)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a vector with sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns an ℓ₁‑normalised vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_ramp(n: int, increasing: bool = True) -> np.ndarray:
    """Linear ramp from 0 to 1 (or reverse)."""
    xs = np.arange(n, dtype=np.float64)
    if increasing:
        a = xs
    else:
        a = n - 1 - xs
    a = a - a.min()
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _quadratic_ramp(n: int, increasing: bool = True) -> np.ndarray:
    xs = np.arange(n, dtype=np.float64)
    if increasing:
        a = xs ** 2
    else:
        a = (n - 1 - xs) ** 2
    a = a - a.min()
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (sub‑gradient)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a)

    n = len(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                       # projection onto simplex

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimisation using L‑BFGS‑B (log‑softmax space).
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of max(conv) via L‑BFGS‑B."""
    if _opt is None:
        # SciPy not available – fall back to the start vector.
        return start_vec.copy(), _surrogate(start_vec)

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0
        ]

    # work in log‑softmax space to enforce positivity & simplex constraint
    v = np.log(start_vec + 1e-30)

    n = len(start_vec)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a_sum = a.sum()
            a /= a_sum                      # softmax → ℓ₁‑normalised

            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                    # shift to avoid overflow
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max value
            smooth_max = shift + np.log(sum_exp) / beta
            obj = 2.0 * n * smooth_max

            # weights for smooth max (softmax over conv entries)
            w = exp_beta / sum_exp

            # gradient of smooth max w.r.t a:
            # ∂/∂a_j (2 n Σ_i w_i Σ_k a_k a_{i‑k}) = 2 n Σ_i w_i (a_{i‑j} + a_{i‑j})
            # which reduces to 2 * (conv(rev_a, w))[segment].
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1: n - 1 + n]

            # projection onto the tangent of the simplex (log‑softmax trick)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if deadline passed
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:               # pragma: no cover
            print(f"[L‑BFGS] exception {exc} at β={beta}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a, _surrogate(a)


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (continuous swaps).
# ----------------------------------------------------------------------
def _mass_transfer_anneal(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from one index to another.
    Accepts moves that improve the surrogate or with a temperature
    proportional to the current objective.
    """
    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to move (fraction of source mass)
        delta = delta_frac * rng.random() * cur[i]
        if delta <= 0.0:
            continue

        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta

        # maintain non‑negativity (numerical safety)
        if ns[i] < 0.0:
            continue

        val = _surrogate(ns)

        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
