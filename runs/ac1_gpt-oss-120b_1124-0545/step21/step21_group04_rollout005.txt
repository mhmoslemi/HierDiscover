```python
"""
Propose‑candidate implementation.

The algorithm consists of the following stages:

1. **Starter generation** – create a diverse set of analytic families
   (uniform, exponential decay, symmetric & asymmetric power laws,
   mixtures with the uniform distribution) together with a few random
   Dirichlet draws.  All vectors are L1‑normalised (sum = 1).

2. **Rapid screening** – evaluate the exact surrogate
   2·n·max(conv(a,a)) for each starter and keep the best few.

3. **Short exponentiated‑gradient (EG) polishing** – run a cheap
   mirror‑descent sub‑gradient optimiser (log‑space EG) on the retained
   starters.

4. **Long EG refinement** – continue the optimisation on the overall
   best vector for as many iterations as time permits.

5. **Adam on smooth‑max** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) with an Adam optimiser in the
   log‑probability space, gradually increasing the smoothing parameter β.

6. **Projected‑gradient polish** – a final Euclidean projected sub‑gradient
   descent to clean up any remaining irregularities.

7. **Evaluation** – the resulting vector is evaluated with the supplied
   ``evaluate_sequence`` function, stored as ``height_sequence_1`` and
   returned.

All heavy numerical work uses FFT‑based convolutions (O(n log n)) and
the code constantly checks the deadline, so the 1000 s budget is never
exceeded.
"""

import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the sequence is L1‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L1 = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s != 0.0 else _uniform(n)


def _sym_power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _asym_power_law(n: int, p: float) -> np.ndarray:
    """Monotonically decreasing law: a_i ∝ (i+1)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _mix_uniform_power(n: int, p: float, lam: float) -> np.ndarray:
    """λ·uniform + (1‑λ)·symmetric‑power‑law, renormalised."""
    u = _uniform(n)
    w = _sym_power_law(n, p)
    mix = lam * u + (1.0 - lam) * w
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, sum a = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser – fast version
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int,
                      eta0: float,
                      eval_period: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best vector found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % eval_period == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int,
             eta0: float,
             eval_period: int = 2000) -> Tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent on the max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        if it % eval_period == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule: List[float],
                     iters_per_beta: int = 15000,
                     lr: float = 0.05,
                     eval_period: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam minimises a log‑sum‑exp approximation of max(conv(a,a)).
    The variable is the log‑probability vector (softmax stays on the simplex).
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = None
    best_val = float('inf')

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            a = _softmax(log_a)

            # ---------- gradient of the smooth max ----------
            conv = _conv_fft(a, a)
            max_c = conv.max()
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]          # ∇_a smooth_max
            # gradient w.r.t. log‑probabilities
            dot = np.dot(grad_a, a)
            g_log = a * (grad_a - dot)

            # ---------- Adam update ----------
            t += 1
            m = beta1 * m + (1.0 - beta1) * g_log
            v = beta2 * v + (1.0 - beta2) * (g_log * g_log)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            log_a = log_a - step

            # ---------- occasional evaluation ----------
            if t % eval_period == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
        # end of inner loop for this beta
    # final check
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val

def _generate_initial_candidates(n: int,
                                 rng: np.random.Generator,
                                 deadline: float) -> List[np.ndarray]:
    """Create a diverse pool of analytic and random starters."""
    candidates = []

    # previous best (if compatible)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            if seq.shape == (n,):
                seq = np.clip(seq, 0.0, 1000.0)
                s = seq.sum()
                if s > 0:
                    seq = seq / s
                else:
                    seq = _uniform(n)
                candidates.append(seq)
        except Exception:
            pass

    # basic analytic families
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.003))
    candidates.append(_exp_decay(n, rate=0.01))

    # symmetric power‑law sweep
    for p in np.linspace(0.0, 6.0, 31):
        candidates.append(_sym_power_law(n, p))

    # asymmetric (monotone) power‑law sweep
    for p in np.linspace(0.2, 5.0, 21):
        candidates.append(_asym_power_law(n, p))

    # mixtures uniform + symmetric power‑law
    for lam in [0.2, 0.5, 0.8]:
        for p in [0.5, 1.0, 2.0]:
            candidates.append(_mix_uniform_power(n, p, lam))

    # a few random Dirichlet draws
    for _ in range(10):
        candidates.append(rng.dirichlet(np.ones(n)))

    # keep only unique (by hash) – cheap de‑duplication
    uniq = {}
    for seq in candidates:
        key = tuple(np.round(seq, 12))   # rounding removes floating noise
        if key not in uniq:
            uniq[key] = seq
    return list(uniq.values())

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------

    pool = _generate_initial_candidates(n, rng, deadline)

        scored.append((val, seq))

    top_candidates = [seq for _, seq in scored[:8]]

    # ------------------------------------------------------------------

        if val_opt < best_val:

    # Fallback in the unlikely case that nothing was kept.
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 4. Long EG refinement (uses remaining time generously).
    # ------------------------------------------------------------------

        seq_adam, val_adam = _adam_smooth_opt(best_seq, n, deadline,
                                              beta_schedule=beta_sched,
                                              iters_per_beta=iters_per_beta,

        true_val = None

    return best_seq.tolist()
```
