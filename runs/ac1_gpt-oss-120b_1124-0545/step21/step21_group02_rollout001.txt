def _exp_gradient_multi(start_seq, n, deadline, max_iters= million, eta0=5.0):
   a = start_seq.copy()
   log_a = np.log(a + 1e-30)
   best_seq = a.copy()
   best_val = _objective_surrogate(a)
   it=0
   while time.time() < deadline and it < max_iters:
       it+=1
       conv = _conv_fft(a,a)
       max_val = conv.max()
       # active indices:
       eps=1e-12
       active = np.where(conv >= max_val - eps)[0]
       # compute subgradient:
       g = np.zeros_like(a)
       for k in active:
           i_start = max(0, k-(n-1))
           i_end = min(n-1, k)
           idx_i = np.arange(i_start, i_end+1)
           rev_idx = k - idx_i
           g[idx_i] += 2.0 * a[rev_idx]
       # optionally average:
       if len(active) > 1:
           g = g / len(active)
       eta = eta0 / math.sqrt(it)
       log_a -= eta * g
       a = _softmax(log_a)
       cur_val = _objective_surrogate(a)
       if cur_val < best_val:
           best_val = cur_val
           best_seq = a.copy()
   return best_seq, best_val

"""
Optimisation of the step‑function evaluation.

The programme proceeds in three stages:

1. **Broad initial sweep** – a collection of analytically motivated
   shapes (exponential decay, power‑law, symmetric Beta, sine,
   semi‑circular) and the previously stored best sequence are normalised
   to a unit L₁‑norm and their exact surrogate
   2·n·max(conv(a,a)) is evaluated.  The best among them is kept as a
   warm start.

2. **Mirror‑descent on the exact max‑convolution** – exponentiated‑gradient
   (multiplicative‑weights) updates are performed on the log‑probabilities.
   The sub‑gradient is computed from *all* active constraints
   (all indices where the convolution attains its maximum) to obtain a
   smoother descent direction.  An adaptive step size η₀/√t is used and the
   best iterate is stored.  The loop runs until the time budget is
   exhausted.

3. **Optional smooth optimisation** – if time permits a sequence of
   increasingly sharp log‑sum‑exp approximations of the max‑convolution is
   minimised with L‑BFGS‑B, exploiting the exact gradient of the smooth
   surrogate.

The final sequence is returned (and stored in the global variable
``height_sequence_1``) and its true value is printed using the provided
``evaluate_sequence`` routine.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1 : 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active maxima)
# ----------------------------------------------------------------------
def _subgradient_multi(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    For a probability vector a (sum=1) and an array of indices where the
    convolution attains its maximal value, return the (averaged) sub‑gradient.
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] += 2.0 * a[rev_idx]

    if active_idx.size > 0:
        g /= float(active_idx.size)          # average over active constraints
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            max_iters: int = 10_000_000,
                            eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Multiplicative‑weights (mirror‑descent) on the piecewise‑linear
    max‑convolution objective.  Returns the best sequence seen and its
    surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # treat indices that are numerically equal to the maximum as active
        eps = 1e-12 * max_val
        active = np.where(conv >= max_val - eps)[0]

        g = _subgradient_multi(a, active)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional random “shake” to escape plateaus
        if no_improve >= 200_000 and (deadline - time.time()) > 30.0:
            # re‑centre around the current best and add small Gaussian noise
            log_a = np.log(best_seq + 1e-30)
            log_a += rng.normal(scale=0.5, size=n)
            a = _softmax(log_a)
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth optimisation via log‑sum‑exp and L‑BFGS‑B
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt   # type: ignore
except Exception:                        # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine a probability vector using a smooth log‑sum‑exp surrogate."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                       # stabilise exponentials
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (derived analytically)
            w = exp_beta / sum_exp               # weights of each convolution entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # mirror‑descent gradient (w.r.t v)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Helper: generate a collection of analytic initial candidates
# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int,
                                 rng: np.random.Generator,
                                 previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a list of diverse normalised vectors (sum = 1) that serve as
    starting points for the fine‑grained optimiser.
    """
    cand = []

    # 0 – previous best, if available and correct size
    if previous is not None and previous.shape == (n,):
        cand.append(np.maximum(previous, 0.0) / previous.sum())

    # 1 – exponential decay (several rates)
    rates = np.logspace(-4, -1, num=12)   # 0.0001 … 0.1
    i = np.arange(n, dtype=np.float64)
    for rate in rates:
        a = np.exp(-rate * i)
        a = a / a.sum()
        cand.append(a)

    # 2 – symmetric power‑law (edge‑heavy when p>0, centre‑heavy when p<0)
    p_vals = np.linspace(-1.5, 2.5, num=11)
    centre = (n - 1) / 2.0
    d = np.abs(i - centre)                     # distance to centre
    for p in p_vals:
        a = (d + 1.0) ** p
        a = a / a.sum()
        cand.append(a)

    # 3 – symmetric Beta distribution (α = β) – avoids true 0/1 endpoints
    alphas = np.array([0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0])
    x = (i + 0.5) / n                         # strictly in (0,1)
    for a in alphas:
        pdf = (x ** (a - 1)) * ((1.0 - x) ** (a - 1))
        pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
        # clip extreme values (evaluate_sequence caps at 1000)
        pdf = np.clip(pdf, 0.0, 1000.0)
        pdf = pdf / pdf.sum()
        cand.append(pdf)

    # 4 – half‑sine (smooth bell)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = a / a.sum()
    cand.append(a)

    # 5 – discrete semicircle (Wigner‑type)
    x_centered = (i - centre) / centre
    a = np.sqrt(np.maximum(0.0, 1.0 - x_centered ** 2))
    a = a / a.sum()
    cand.append(a)

    # 6 – a few random Dirichlet draws (light exploration)
    for _ in range(5):
        a = rng.random(n) ** 2          # bias towards small numbers
        a = a / a.sum()
        cand.append(a)

    return cand

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the previously stored result
    # ------------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Broad sweep over analytically motivated shapes
    # ------------------------------------------------------------------
    candidates = _generate_initial_candidates(n, rng, prev_arr)
    best_seq = None
    best_surrogate = float("inf")
    for idx, seq in enumerate(candidates):
        val = _objective_surrogate(seq)
        if val < best_surrogate:
            best_surrogate = val
            best_seq = seq.copy()
        # early exit if we run out of time
        if time.time() > deadline:
            break

    if best_seq is None:
        # fallback – uniform distribution (never happens)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_surrogate = _objective_surrogate(best_seq)

    print(f"[Sweep] best surrogate from analytic families = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    seq_eg, val_eg = _exp_gradient_opt_multi(best_seq, n, deadline, rng,
                                            max_iters=8_000_000, eta0=10.0)
    print(f"[EG] surrogate after mirror‑descent = {val_eg:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional L‑BFGS refinement on a smooth surrogate
    # ------------------------------------------------------------------
    final_seq = seq_eg
    final_val = val_eg
    if time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(seq_eg, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < val_eg:
            final_seq = seq_lb
            final_val = val_lb

    # ------------------------------------------------------------------
    # 4. Final true evaluation (security‑checked)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(final_seq.tolist())
        print(f"[Result] surrogate = {final_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # store for the next run
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
