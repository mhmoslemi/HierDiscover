"""
Search for a non‑negative step‑function (height sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The code follows a three‑stage strategy:

1. **Seeding** – generate many diverse ℓ₁‑normalised candidates from analytic
   families (uniform, triangular, sine‑window, Gaussian, beta, power‑law,
   exponential decays, flat‑top, mixtures) and a few random Dirichlet draws.
   The surrogate objective 2·n·max(conv) (identical to the true metric when
   Σa = 1) is evaluated quickly with an FFT‑based convolution; the best
   ``TOP_K`` seeds are kept.

2. **Local refinement** – each seed is independently improved by three
   complementary procedures:
   * Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution,
     using the full active set of binding indices.
   * Smooth‑max L‑BFGS‑B optimisation (log‑sum‑exp surrogate) with an aggressive
     schedule of β parameters.
   * Simulated‑annealing with a long cooling schedule.
   The best result among the three refinements updates the global optimum.

3. **Polishing** – the current global best undergoes a prolonged exponentiated‑gradient
   run (full active set) followed by a high‑budget simulated‑annealing pass and,
   finally, a very fine‑grained pairwise “mass‑transfer’’ hill‑climb that moves a
   tiny amount of probability between two coordinates.  The resulting vector
   is returned and stored in the global variable ``height_sequence_1`` for
   reuse.

All work respects the supplied time budget (default 1000 s).  The implementation
uses only NumPy and optionally SciPy (if available).  No file or network I/O is
performed.
"""

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (identical to true metric when Σa = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) (left‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i)) (right‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _flat_top(n: int, width_frac: float = 0.2) -> np.ndarray:
    """Uniform mass on a centred block of width width_frac·n."""
    w = max(1, int(round(width_frac * n)))
    start = (n - w) // 2
    a = np.zeros(n, dtype=np.float64)
    a[start:start + w] = 1.0
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      use_full_active: bool = True):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space so that a stays on the probability simplex
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if use_full_active:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]
        else:
            active = np.array([int(conv.argmax())], dtype=np.int64)

        # sub‑gradient for each active shift
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            # average over active set (any convex combination works)
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # learning‑rate schedule (decreasing ~ 1/√t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 1e7, 2e7, 5e7, 1e8
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.005,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fine‑grained pairwise hill‑climb (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def _pairwise_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 400_000,
                         delta_frac: float = 0.004):
    """
    Randomly select a pair (i, j) and move a tiny amount of mass from i
    to j (keeping Σa constant). Accept only improving moves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        i, j = rng.integers(n, size=2)
        if i == j:
            attempts += 1
            continue
        if a[i] == 0.0:
            attempts += 1
            continue
        # amount to transfer – a small fraction of the source mass
        delta = delta_frac * a[i] * rng.random()
        if delta <= 0.0:
            attempts += 1
            continue
        ns = a.copy()
        ns[i] -= delta
        ns[j] += delta
        # (sum unchanged by construction)
        val = _objective_surrogate(ns)
        if val < cur_val:
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previous solution if available; otherwise use 4000.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 4000
    else:
        n = prev_arr.shape[0]

    print(f"\n--- Optimisation started (n={n}) ---")

    # ------------------------------------------------------------------
    # 1. Broad seeding – analytic families + random Dirichlet draws
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_flat_top(n, width_frac=0.1))
    seeds.append(_flat_top(n, width_frac=0.3))

    # beta‑skewed families
    alphas = np.linspace(0.2, 6.0, 12)
    betas = np.linspace(0.2, 6.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (both sides)
    rates = np.logspace(-4, -0.2, 25)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # power‑law families
    ps = np.linspace(0.0, 4.0, 12)
    for p in ps:
        seeds.append(_power(n, p))

    # mixtures with uniform
    for r in rates[::5]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for a_mix in (0.2, 0.4, 0.6):
            seeds.append(_mix_shapes(_uniform(n), exp_l, a_mix))
            seeds.append(_mix_shapes(_uniform(n), exp_r, a_mix))

    # random Dirichlet draws (explores sparse/dense regimes)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a portion of the remaining budget for this seed
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(3.0, remaining * 0.12 / max(1, (top_k - si)))

        # ---- (a) Exponentiated‑gradient with full active set ----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=250_000,
                                           eta0=6.0,
                                           use_full_active=True)

        # ---- (b) Smooth‑max L‑BFGS refinement ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0,
                                                              2_000_000.0, 5_000_000.0],
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ---- (c) Simulated annealing (moderate budget) ----
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=180_000,
                                                  start_sigma=0.002,
                                                  T_start=0.01,
                                                  T_end=1e-8)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among the three refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – intensive EG + SA + pairwise hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive EG polishing (full active) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0,
                                                 use_full_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=600_000,
                                                          start_sigma=0.0008,
                                                          T_start=0.003,
                                                          T_end=1e-10)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Pairwise hill‑climb fine‑tuning ---")
        final_hc_seq, final_hc_val = _pairwise_hill_climb(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=500_000,
                                                          delta_frac=0.003)
        if final_hc_val < global_best_val:
            global_best_seq = final_hc_seq
            global_best_val = final_hc_val
            print(f"[Hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
