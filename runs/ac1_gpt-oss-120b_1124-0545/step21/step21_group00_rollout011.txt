def _lbfgs_fourier(start_theta, basis, n, deadline, beta_schedule):
    if _opt is None:
        return start_theta
    theta = start_theta.copy()
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(th):
            a = _softmax(basis @ th)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            # compute gradient wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]   # length n
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)      # gradient wrt logits
            grad_theta = basis.T @ grad_v
            return obj, grad_theta
        try:
            res = _opt.minimize(fun, theta, method='L-BFGS-B', jac=True, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            theta = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print("[L‑BFGS‑Fourier] exception:", e)
            break
    return theta

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The algorithm is a very simple but
time‑budget aware stochastic search:

1.  Generate a large pool of random *probability* vectors.
    – uniform, Dirichlet (various concentrations), sparse Dirichlet,
    – Gaussian‑mixture, and a Fourier‑softmax family.
2.  Keep the best candidate according to the fast surrogate
    ``2·n·max(conv(seq,seq))``.
3.  Refine this candidate with a smooth L‑BFGS optimisation of a
    log‑sum‑exp approximation to the max‑convolution.
4.  Polish the result with a long exponentiated‑gradient (mirror‑descent)
    optimisation that works on the true (non‑smoothed) objective.
5.  Return the best sequence found; also store it in the global
    variable ``height_sequence_1`` for possible reuse.

All heavy work respects the 1000 s overall budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for the smooth L‑BFGS stage
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 3.0):
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000):
    """L‑BFGS‑B on the smooth surrogate (variables = log‑softmax)."""

    if _opt is None:
        return start_vec.copy()          # nothing to do without SciPy

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Stochastic + gradient based search for the step‑function that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    n = 4000                               # problem size

    # --------------------------------------------------------------
    # Retrieve the previous solution if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    best_seq = None
    best_val = np.inf
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                best_seq = arr / arr.sum()
                best_val = _objective_surrogate(best_seq)
                print(f"[init] previous best surrogate = {best_val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1️⃣ Generate a large diverse pool of random seeds
    # --------------------------------------------------------------
    seeds = []   # list of probability vectors

    # a few deterministic seeds
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform

    # Dirichlet draws with many concentrations
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Sparse Dirichlet (supports of size 80 … 180)
    for k in (80, 120, 160, 200):
        for _ in range(2):
            idx = rng.choice(n, size=k, replace=False)
            w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
            a = np.zeros(n, dtype=np.float64)
            a[idx] = w
            seeds.append(a)

    # Tiny Gaussian‑mixture seeds (8 components)
    K_mix = 8
    for _ in range(6):
        params = np.concatenate([
            rng.random(K_mix),                     # weights (will be normalised)
            rng.random(K_mix),                     # means
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        # build mixture
        w = np.maximum(params[0:K_mix], 0.0)
        mu = np.clip(params[K_mix:2 * K_mix], 0.0, 1.0)
        sigma = np.clip(params[2 * K_mix:3 * K_mix], 0.001, 0.5)
        if w.sum() > 0.0:
            w = w / w.sum()
        else:
            w = np.full(K_mix, 1.0 / K_mix)

        i = np.arange(n, dtype=np.float64) / (n - 1.0)
        a = np.zeros(n, dtype=np.float64)
        for kk in range(K_mix):
            a += w[kk] * np.exp(-((i - mu[kk]) ** 2) / (2.0 * sigma[kk] ** 2))
        a = a / a.sum()
        seeds.append(a)

    # Fourier‑softmax random seeds (moderate dimensionality)
    D_fourier = 80                     # 40 sine + 40 cosine terms
    K_freq = D_fourier // 2
    i_norm = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis_rows = []
    for kk in range(1, K_freq + 1):
        basis_rows.append(np.sin(2.0 * np.pi * kk * i_norm))
        basis_rows.append(np.cos(2.0 * np.pi * kk * i_norm))
    basis = np.column_stack(basis_rows)   # shape (n, D_fourier)

    for _ in range(30):
        theta = rng.normal(scale=1.0, size=D_fourier)
        logits = basis @ theta
        a = _softmax(logits)
        seeds.append(a)

    # --------------------------------------------------------------
    # 2️⃣ Evaluate the huge seed pool (fast surrogate) and keep best
    # --------------------------------------------------------------
    print("\n--- Evaluating random seeds (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        if (idx + 1) % 200 == 0:
            print(f"[seed {idx+1:04d}] current best surrogate = {best_val:.6f}")

    print(f"[Init] best surrogate after random pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Local refinement with smooth L‑BFGS (log‑sum‑exp)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of the best seed ---")
        # allocate roughly half of the remaining time for smoothing
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45
        seq_lbfgs = _lbfgs_smooth_optimize(
            best_seq, n, sub_deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                           200.0, 500.0, 1000.0,
                           2000.0, 5000.0,
                           20000.0, 50000.0,
                           100000.0, 200000.0,
                           500000.0, 1000000.0,
                           2000000.0, 5000000.0,
                           10000000.0],
            max_iter_per_beta=20000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Exact mirror‑descent (exponentiated‑gradient) polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polish_deadline = deadline - 3.0
        time_left = polish_deadline - time.time()
        # allocate a generous number of iterations proportionally to remaining time
        max_iters = int(min(5_000_000,
                            max(200_000,
                                time_left / (deadline - start_time) * 2_500_000)))
        seq_eg, val_eg = _exp_gradient_opt(
            best_seq, n,
            polish_deadline,
            max_iters=max_iters,
            eta0=3.0,
        )
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best after EG = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ A few cheap stochastic perturb‑and‑polish passes
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # small random perturbation of the current best
        perturb_sigma = 0.001
        noisy = best_seq + perturb_sigma * rng.normal(size=best_seq.shape)
        noisy = np.maximum(noisy, 0.0)
        noisy = noisy / noisy.sum()
        # short EG polish (few tens of thousands iters)
        short_deadline = time.time() + 0.5
        cand_seq, cand_val = _exp_gradient_opt(
            noisy, n,
            short_deadline,
            max_iters=120_000,
            eta0=2.0,
        )
        print(f"[Pert‑EG] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"  >>> New best after perturb‑EG = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6️⃣ Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
