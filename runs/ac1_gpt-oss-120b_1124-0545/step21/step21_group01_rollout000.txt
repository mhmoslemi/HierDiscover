def _coarse_evolutionary_search(n, m, deadline):
    rng = np.random.default_rng()
    dim = m
    # initialize mean as zeros (logits)
    mu = np.zeros(dim)
    sigma = 1.0  # step size
    best_seq = None
    best_val = np.inf
    # evaluate initial uniform coarse:
    init = np.full(dim, 1.0 / dim)
    init_full = _upsample(init, n)
    best_seq = init_full
    best_val = _objective_surrogate(best_seq)
    # initial best coarse representation:
    best_logits = mu.copy()
    # loop
    iteration = 0
    while time.time() < deadline:
        # population size dynamic perhaps based on remaining time/time per eval
        lam = 12  # could be 12 per iteration
        # sample offspring
        candidates = []
        vals = []
        for _ in range(lam):
            logits = mu + sigma * rng.normal(size=dim)
            probs = _softmax(logits)
            full = _upsample(probs, n)
            val = _objective_surrogate(full)
            candidates.append(logits)
            vals.append(val)
        # find best among candidates
        min_idx = int(np.argmin(vals))
        cand_best_logits = candidates[min_idx]
        cand_best_val = vals[min_idx]
        # if improvement:
        if cand_best_val < best_val:
            best_val = cand_best_val
            best_seq = _upsample(_softmax(cand_best_logits), n)
            best_logits = cand_best_logits.copy()
            # increase sigma
            sigma *= 1.2
        else:
            # decrease sigma
            sigma *= 0.85
        # move mu towards best candidate? Use simple update: mu = cand_best_logits maybe
        mu = cand_best_logits.copy()
        iteration += 1
        if iteration % 20 == 0:
            print(f"[Coarse ES] iter {iteration}, best surrogate = {best_val:.6f}, sigma={sigma:.4f}")
    return best_seq, best_val

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  

The search proceeds in several loosely coupled stages:

1️⃣  Initialise a good seed (previous solution, uniform, arcsine,
    Beta‑densities, Dirichlet draws, tiny perturbations) and keep the best.

2️⃣  **Coarse evolutionary search** – optimise a low‑dimensional
    representation (m≈200 “bins”).  A simple (1+λ)‑ES works on the
    logits of a softmax, repeatedly up‑sampling the coarse vector to the
    full size (n=4000) and evaluating the cheap surrogate  
    2·n·max(conv(a,a)).  The best coarse shape is up‑sampled to the full
    resolution.

3️⃣  **L‑BFGS smoothing** – starting from the best full‑size vector,
    minimise a smooth log‑sum‑exp surrogate of the max‑convolution
    (variables live in log‑softmax space).  The schedule of β values is
    aggressive, allowing the optimiser to approach the true max.

4️⃣  **Exact exponentiated‑gradient polishing** – a mirror‑descent step
    on the true max‑convolution, using the sub‑gradient at the active
    index.  Many iterations are performed until the time budget is
    exhausted.

5️⃣  **Local random refinement** – tiny stochastic moves on the simplex
    (move a random fraction of mass from one coordinate to another)
    keep any improvement.

All stages respect the overall 1000 s budget.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the L‑BFGS optimiser)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  (seq must sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _upsample(coarse: np.ndarray, n: int) -> np.ndarray:
    """
    Upsample a coarse probability vector (length m) to length n by
    piecewise‑constant replication, then renormalise to sum‑to‑1.
    """
    m = len(coarse)
    reps = int(math.ceil(n / m))
    up = np.repeat(coarse, reps)[:n]
    total = up.sum()
    if total <= 0.0:
        # fallback – uniform
        up = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        up = up / total
    return up


def _compress_to_coarse(full: np.ndarray, m: int) -> np.ndarray:
    """
    Aggregate a full probability vector (length n) to m bins by averaging,
    then renormalise.
    """
    n = len(full)
    reps = int(math.ceil(n / m))
    padded = np.pad(full, (0, reps * m - n), mode="constant")
    reshaped = padded.reshape(m, reps)
    coarse = reshaped.mean(axis=1)
    s = coarse.sum()
    if s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    return coarse / s


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors of length n)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimise the smooth max‑convolution objective using L‑BFGS‑B.
    Variables live in the log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1000000.0, 2000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp over the 2n‑1 convolution entries
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # d/d a of weighted sum
            grad_a = grad_a * (2.0 * n)               # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient at the (single) active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max_{i+j=k} a_i a_j
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Coarse (1+λ)‑ES on a low‑dimensional softmax representation
# ----------------------------------------------------------------------
def _coarse_evolutionary_search(full_n: int,
                                coarse_m: int,
                                deadline: float,
                                rng: np.random.Generator,
                                init_coarse: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Simple (1+λ) evolution strategy on the logits of a softmax of length ``coarse_m``.
    Each candidate is up‑sampled to length ``full_n`` and evaluated with the cheap surrogate.
    Returns the best full‑size vector found and its surrogate value.
    """
    # Parameters
    lam = 12                     # offspring per generation
    sigma = 1.0                  # mutation step‑size (logit space)
    sigma_min, sigma_max = 0.05, 5.0

    # Initialise μ (logits).  If an initial coarse distribution is supplied,
    # take its log‑softmax as the starting point, else start from zeros.
    if init_coarse is not None:
        mu = np.log(init_coarse + 1e-30)
    else:
        mu = np.zeros(coarse_m, dtype=np.float64)

    # Evaluate the current μ
    probs = _softmax(mu)
    best_full = _upsample(probs, full_n)
    best_val = _objective_surrogate(best_full)

    it = 0
    while time.time() < deadline:
        candidates = []
        cand_vals = []
        for _ in range(lam):
            logits = mu + sigma * rng.normal(size=coarse_m)
            probs_c = _softmax(logits)
            full_c = _upsample(probs_c, full_n)
            val_c = _objective_surrogate(full_c)
            candidates.append(logits)
            cand_vals.append(val_c)

        # Find best offspring in this generation
        idx_best = int(np.argmin(cand_vals))
        best_off_logit = candidates[idx_best]
        best_off_val = cand_vals[idx_best]

        # Update σ according to the 1/5‑success rule (very coarse)
        if best_off_val < best_val:
            sigma = min(sigma * 1.2, sigma_max)
            mu = best_off_logit.copy()          # move the centre
            best_val = best_off_val
            best_full = _upsample(_softmax(best_off_logit), full_n)
        else:
            sigma = max(sigma * 0.85, sigma_min)

        it += 1
        if it % 20 == 0:
            print(f"[Coarse ES] iter {it:04d}, sigma={sigma:.4f}, best surrogate={best_val:.6f}")

        # early stop if we are very close to the deadline
        if time.time() > deadline:
            break

    return best_full, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size (default 4000)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))

    # a range of Beta shapes
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0),
                        (0.3, 0.3), (0.3, 1.0), (1.0, 0.3)]:
        seeds.append(_beta_distribution(n, alpha, beta))

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # small perturbations of uniform
    for _ in range(4):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.005))

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate seeds and keep the best (surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Coarse evolutionary search (low‑dimensional)
    # ------------------------------------------------------------------
    coarse_m = 200                     # number of bins for the coarse representation

            full_n=n,
            coarse_m=coarse_m,
            deadline=sub_deadline,
            rng=rng,

    true_val = None

        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
