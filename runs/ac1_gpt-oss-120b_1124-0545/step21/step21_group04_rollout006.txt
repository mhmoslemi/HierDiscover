def _lbfgs_smooth_optimize(start_a, n, deadline):
    if _opt is None:
        return start_a
    v = np.log(start_a + 1e-30)
    beta_schedule = [0.5, 2, 10, 30, 100, 300, 1000, 3000, 1e4, 3e4, 1e5, 3e5, 1e6, 3e6, 1e7]
    for beta in beta_schedule:
        if time.time()>deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient w.r.t. v (logits)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time()>deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': 4000, 'disp': False, 'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L-BFGS] exception {e} at beta={beta_val}")
            break
    a_opt = _softmax(v)
    a_opt = _symmetrize(a_opt)
    a_opt = _proj_simplex(a_opt)
    return a_opt
```

"""
Propose a step‑function (non‑negative vector on the probability simplex) that
minimises the supplied ``evaluate_sequence`` objective.

The algorithm works in three phases:

1. **Candidate generation** – a small collection of analytic families
   (uniform, symmetric exponential and power‑law) together with a bunch of
   random Dirichlet draws (different concentration parameters) are created.
   All candidates are forced to be symmetric, normalised to sum 1 and their
   surrogate objective ``2·n·max(conv(a,a))`` is evaluated.  The best few
   (default 5) are kept for the next phase.

2. **Smooth L‑BFGS optimisation** – each selected candidate is optimised with
   a smooth approximation of the max‑convolution (log‑sum‑exp).  The smoothing
   parameter β is increased on a schedule (from 0.5 up to 1e7).  SciPy’s
   L‑BFGS‑B optimiser is used (fallback to an exponentiated‑gradient if SciPy is
   unavailable).  After each β‑stage the vector is projected back onto the
   simplex and symmetrised.

3. **Projected sub‑gradient polishing** – the best result from phase 2 is
   refined by a simple projected‑gradient descent on the exact (non‑smooth)
   objective.  The gradient is the average sub‑gradient of all active
   convolution entries (those attaining the current maximum).  The update is
   symmetrised and projected onto the simplex after each step.

The whole procedure respects the 1000 s time budget by frequently checking the
deadline.  The final sequence is stored in the global variable
``height_sequence_1`` for warm‑starting future runs and is returned as a plain
list of floats.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the smooth L‑BFGS phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the ℓ₁‑norm is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector (ℓ₁‑norm 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_sym_decay(n: int, rate: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-rate·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Simple ℓ₁‑simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Enforce symmetry (a[i] = a[n‑1‑i])
# ----------------------------------------------------------------------
def _symmetrize(x: np.ndarray) -> np.ndarray:
    return (x + x[::-1]) / 2.0


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (average over all active indices)
# ----------------------------------------------------------------------
def _subgradient_avg_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = (conv >= max_val - tol).astype(np.float64)   # length 2n‑1
    count = active.sum()
    if count == 0.0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    full = _conv_fft(rev_a, active)                     # length 2n‑1
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent on max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_avg_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _symmetrize(a)
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) – fallback when SciPy is missing
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 100_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution using the averaged sub‑gradient."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_avg_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_a: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Optimise a smooth approximation of max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        # SciPy not available – fall back to exponentiated‑gradient
        seq, _ = _exp_gradient_opt_multi(start_a, n, deadline,
                                         max_iters=200_000, eta0=5.0)
        return seq

    if beta_schedule is None:
        beta_schedule = [
            0.5, 2.0, 10.0, 30.0,
            100.0, 300.0, 1_000.0, 3_000.0,
            10_000.0, 30_000.0, 100_000.0,
            300_000.0, 1_000_000.0, 3_000_000.0,
            10_000_000.0
        ]

    v = np.log(start_a + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. logits v (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a_opt = _softmax(v)
    a_opt = _symmetrize(a_opt)
    a_opt = _proj_simplex(a_opt)
    return a_opt


# ----------------------------------------------------------------------
# Simple 1‑D power‑law search (coarse + local refinement)
# ----------------------------------------------------------------------
def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped sequence with good surrogate value."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # coarse grid
    for p in np.linspace(0.0, 2.5, 31):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # local refinement around the best p
    p_center = np.linspace(0.0, 2.5, 31)[np.argmin([
        _objective_surrogate(_power_law(n, p)) for p in np.linspace(0.0, 2.5, 31)
    ])]
    step = 0.2
    for _ in range(5):
        if time.time() > deadline:
            break
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ₁‑normalised) that minimises
    ``evaluate_sequence``.  The routine respects the allotted time budget,
    stores the best sequence globally as ``height_sequence_1`` and returns it
    as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin so we finish before the strict deadline
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Problem size – reuse the size of a previously found sequence if
    # available, otherwise start with n = 4000.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # --------------------------------------------------------------
    # 1. Generate a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # a) previously‑found solution (if compatible)
    if prev is not None:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            if seq.shape == (n,):
                seq = np.clip(seq, 0.0, 1000.0)
                if seq.sum() > 0:
                    seq = seq / seq.sum()
                else:
                    seq = _uniform(n)
                candidates.append(seq)
        except Exception:
            pass

    # b) Analytic families
    candidates.append(_uniform(n))
    candidates.append(_exp_sym_decay(n, rate=0.003))
    candidates.append(_exp_sym_decay(n, rate=0.008))
    candidates.append(_best_power_law(n, deadline))

    # c) Random Dirichlet draws with various concentration parameters
    for alpha in (0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            seq = rng.dirichlet(np.full(n, alpha))
            candidates.append(seq)

    # --------------------------------------------------------------
    # 2. Evaluate cheap surrogate and keep the best few (top_k)
    # --------------------------------------------------------------
    evaluated = []
    for seq in candidates:
        seq = _symmetrize(seq)
        s = seq.sum()
        if s == 0.0:
            continue
        seq = seq / s
        val = _objective_surrogate(seq)
        evaluated.append((val, seq))
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [seq for _, seq in evaluated[:top_k]]

    # --------------------------------------------------------------
    # 3. Detailed optimisation of the top candidates
    # --------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    for idx, start_seq in enumerate(top_candidates):
        if time.time() > deadline - 30:
            break

        # ---------- smooth L‑BFGS ----------
        seq_opt = _lbfgs_smooth_optimize(start_seq, n, deadline)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS {idx}] surrogate after smoothing = {val_opt:.6f}")

        # ---------- projected‑gradient polishing ----------
        if time.time() < deadline - 10:
            seq_pg, val_pg = _pgd_opt(seq_opt, n, deadline,
                                      max_iters=200_000,
                                      eta0=1.0)
            print(f"[PGD {idx}] surrogate after polishing = {val_pg:.6f}")
        else:
            seq_pg, val_pg = seq_opt, val_opt

        # ---------- (optional) short exponentiated‑gradient ----------
        if _opt is None and time.time() < deadline - 5:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_pg, n, deadline,
                                                    max_iters=50_000,
                                                    eta0=5.0)
            if val_eg < val_pg:
                seq_pg, val_pg = seq_eg, val_eg
                print(f"[EG {idx}] surrogate after extra EG = {val_eg:.6f}")

        # keep the best across all restarts
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()

    # --------------------------------------------------------------
    # 4. Fallback in the unlikely case that nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # --------------------------------------------------------------
    # 5. Final evaluation with the provided function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Result] evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run (enables incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
