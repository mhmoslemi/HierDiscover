"""
Propose a step‑function sequence that lowers the evaluation metric.

Algorithm
---------
1. **Seed generation** – create a diverse pool of ℓ₁‑normalised vectors:
   uniform, exponential (and symmetric), power‑law, beta‑shape,
   random Dirichlet draws and piecewise‑linear profiles.  The previous
   best sequence (if present) and a few of its permutations are also
   inserted.

2. **Surrogate ranking** – the objective
        2·n·max(conv(p,p))
   (with p = a/∑a, hence independent of scaling) is cheap to compute.
   The best few seeds are kept for refinement.

3. **Smooth exponential‑gradient descent** – minimise a *smooth*
   approximation of the max‑convolution:
        fβ(p)=max(conv) + 1/β·log Σ exp(β·(conv−max(conv))).
   The gradient w.r.t. p is obtained analytically (see the code).  The
   optimisation runs in the log‑softmax domain, guaranteeing non‑negativity
   and unit ℓ₁‑norm.  A schedule of increasing β (more smoothing → true
   max) and a decreasing step‑size ηt=η0/√t are used.

4. **Permutation annealing** – after gradient polishing a simple
   simulated‑annealing on the ordering of the coefficients (swap moves
   only) is applied for a long period.  The temperature follows a linear
   schedule.

5. **Mass‑transfer hill‑climbing** – a final short phase moves a small
   amount of mass between two random positions; swaps are kept if they
   improve the surrogate.

6. The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned as a plain list.  Throughout, the
   remaining time budget is respected; if the deadline is reached the
   current best is returned.

The whole pipeline runs under the 1000 s budget and never produces
negative or NaN entries, so it is safe to feed the result directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(p,p))  (p sums to 1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth exponential‑gradient optimiser (log‑softmax domain)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             deadline: float,
                             max_iters: int = 1_500_000,
                             eta0: float = 5.0,
                             beta_schedule=None) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of the max‑convolution.
    Operates on log‑probabilities to keep the vector non‑negative
    and ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0
        ]

    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                     # stay in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    per_beta = max(1, max_iters // len(beta_schedule))

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(per_beta):
            if time.time() > deadline:
                break

            it += 1
            conv = _conv_fft(a, a)
            max_val = conv.max()
            # smooth weights (numerically stable because diff ≤ 0)
            diff = conv - max_val
            exp_term = np.exp(beta * diff)
            w = exp_term / exp_term.sum()

            # gradient of the smooth max (see derivation in the docstring)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad = 2.0 * full[len(a) - 1: len(a) - 1 + len(a)]

            eta = eta0 / math.sqrt(it)
            log_a -= eta * grad
            a = np.exp(log_a)
            s = a.sum()
            if s == 0.0:
                a = _uniform(len(a))
                log_a = np.log(a + 1e-30)
            else:
                a /= s

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Swap‑based simulated annealing on the ordering of the coefficients.
    The surrogate is used for speed.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small mass‑transfer hill‑climber (move a fraction of weight)
# ----------------------------------------------------------------------
def _pairwise_transfer_opt(start_seq: np.ndarray,
                           deadline: float,
                           max_iters: int = 100_000,
                           step_frac: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a tiny amount of mass from one index to another.
    Accepts the move only if the surrogate improves.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * step_frac * np.random.rand()
        a[i] -= eps
        a[j] += eps

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        else:
            # revert
            a[i] += eps
            a[j] -= eps

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure follows the outline described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a good speed/quality trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present (helps continuation runs)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diversified pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    for rate in (0.001, 0.002, 0.004):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear shapes
    for nb in (3, 5, 8):
        for _ in range(3):
            seeds.append(_piecewise_linear(n, nb, rng))

    # power‑law variants
    for p in (0.5, 0.8, 1.0, 1.5):
        seeds.append(_power(n, p))

    # beta‑shaped families
    for _ in range(8):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # include the previous best and a few of its permutations
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(np.random.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top‑k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (≈12 % each)
        remaining = deadline - time.time()
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # (a) smooth exponential‑gradient descent
        seq_sg, val_sg = _smooth_exp_gradient_opt(
            seed_vec,
            sub_deadline,
            max_iters=int(150_000),
            eta0=6.0
        )

        # (b) permutation annealing on the polished vector
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(
                seq_sg,
                sub_deadline,
                max_iters=int(300_000)
            )
        else:
            seq_perm, val_perm = seq_sg, val_sg

        # pick the local best
        local_seq, local_val = min(
            [(seq_sg, val_sg), (seq_perm, val_perm)],
            key=lambda x: x[1]
        )
        print(f"[Seed {si+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long smooth‑gradient polishing (uses all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long smooth‑gradient polishing ---")
        final_seq, final_val = _smooth_exp_gradient_opt(
            global_best_seq,
            deadline,
            max_iters=500_000,
            eta0=5.0
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final SG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final permutation annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final permutation annealing ---")
        seq_last, val_last = _perm_anneal(
            global_best_seq.copy(),
            deadline,
            max_iters=200_000
        )
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short mass‑transfer hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(20.0, deadline - time.time())
        print("\n--- Local mass‑transfer hill climbing ---")
        seq_final, val_final = _pairwise_transfer_opt(
            global_best_seq.copy(),
            short_deadline,
            max_iters=100_000
        )
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final transfer] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
