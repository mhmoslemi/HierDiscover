def _smooth_eg(start_seq, deadline, beta_schedule, max_iters_per_beta=20000, eta0=4.0):
    a = np.copy(start_seq)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    total_iters = 0.0
    n = len(a)

    for beta in beta_schedule:
        iters = max_iters_per_beta
        for it in range(1, iters+1):
            if time.time() > deadline:
                return best_seq, best_val
            conv = _conv_fft(a, a)
            shift = conv.max()
            diff = conv - shift
            # compute softmax weights w
            # The exponent may underflow for large negative diff, but that's fine; we treat small values as zero
            exp_terms = np.exp(beta * diff)
            sum_exp = exp_terms.sum()
            if sum_exp == 0 or np.isnan(sum_exp):
                # fallback to hard max
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_terms / sum_exp

            # gradient
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad = 2.0 * full[n-1:n-1+n]

            # step size
            eta = eta0 / math.sqrt(total_iters + it + 1)

            # exponentiate update - clip grad to avoid overflow of exponent
            grad_clip = np.clip(grad, -50.0, 50.0)
            a = a * np.exp(-eta * grad_clip)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        total_iters += iters
    return best_seq, best_val

"""
Advanced smoothing‑EG optimiser for the step‑function “max‑convolution”
problem.

The programme builds a diverse collection of ℓ₁‑normalised seeds,
keeps the best few and refines each of them with a *smoothed*
exponentiated‑gradient descent.  The smoothing is a log‑sum‑exp
approximation of the max‑convolution, controlled by a schedule of
inverse temperatures (β).  The optimisation proceeds in three stages:

1. **Seed generation & selection** – analytic families (uniform,
   exponential, symmetric‑exponential, power‑law, beta, Dirichlet,
   piecewise‑linear, bimodal exponential) together with the previous
   best solution are generated, normalised, and ranked by the exact
   surrogate 2·n·max(conv).

2. **Smoothed EG refinement** – for each top‑k seed we run a mirror‑
   descent in the simplex using multiplicative updates
   `a_i ← a_i·exp(‑η·∇_i)`.  The gradient ∇_i is the exact gradient of
   the β‑smoothed max‑convolution (obtained by a short weighted
   convolution).  β is increased through a prescribed schedule so that
   the method starts with a smooth landscape and ends with the true
   non‑smooth objective.

3. **Final polishing** – the globally best refined vector receives a
   long high‑β EG run, an optional L‑BFGS pass on the same smoothed
   objective (if SciPy is present) and a short permutation‑based
   simulated‑annealing tweak.

All heavy work uses FFT‑based linear convolution (`_conv_fft`) which
runs in O(n log n).  The optimiser respects the 1000 s budget, prints
progress and stores the final sequence in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time

# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ  (i from 1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on the discrete index set."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    pts = np.sort(rng.choice(n, size=num_breaks, replace=False))
    pts = np.concatenate(([0], pts, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(pts))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, pts, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exp(n: int, rate1: float, pos1: int,
                rate2: float, pos2: int) -> np.ndarray:
    """Sum of two shifted exponentials."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate1 * np.abs(i - pos1)) + np.exp(-rate2 * np.abs(i - pos2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _smooth_eg(start_seq: np.ndarray,
               deadline: float,
               beta_schedule,
               max_iters_per_beta: int = 15_000,
               eta0: float = 4.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the β‑smoothed max‑convolution.
    Uses multiplicative updates (a_i ← a_i·exp(‑η·g_i)) and normalises.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    total_iters = 0

    for beta in beta_schedule:
        for it in range(1, max_iters_per_beta + 1):
            if time.time() > deadline:
                return best_seq, best_val

            # 1) convolution of a with itself
            conv = _conv_fft(a, a)
            shift = conv.max()
            diff = conv - shift

            # 2) soft‑max weights   w_k ∝ exp(β·(conv_k−shift))
            exp_terms = np.exp(beta * diff)          # all ≤ 1, max entry = 1
            sum_exp = exp_terms.sum()
            if sum_exp == 0 or np.isnan(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_terms / sum_exp

            # 3) gradient of the smoothed max (see L‑BFGS implementation)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad = 2.0 * full[n - 1:n - 1 + n]        # size n

            # 4) step‑size (inverse‑sqrt schedule)
            eta = eta0 / math.sqrt(total_iters + it + 1.0)

            # 5) multiplicative update – clip gradient to keep exp stable
            grad_clip = np.clip(grad, -50.0, 50.0)
            a = a * np.exp(-eta * grad_clip)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)          # fallback to uniform
            else:
                a /= s

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        total_iters += max_iters_per_beta
    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that repeatedly swaps two entries.
    Accepts uphill moves with a linearly decreasing temperature.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3_000) -> np.ndarray:
    """
    Minimises the β‑smoothed objective with L‑BFGS‑B.
    Operates in log‑softmax space (a_i = exp(v_i) / Σexp(v)).
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            # transform gradient to log‑space (softmax manifold)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine follows the three‑stage plan
    described in the module docstring and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power‑of‑two eases FFT internal padding.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1. Build a rich pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_sym_exp_decay(n, rate=0.0015))

    # random Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # power‑law families
    for p in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_power(n, p))

    # beta‑shaped families
    for _ in range(10):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # piecewise‑linear shapes
    for nb in (3, 6, 10):
        for _ in range(3):
            seeds.append(_piecewise_linear(n, nb, rng))

    # bimodal exponential mixtures
    for _ in range(6):
        rate1 = rng.uniform(0.001, 0.004)
        rate2 = rng.uniform(0.001, 0.004)
        pos1 = rng.integers(0, n // 2)
        pos2 = rng.integers(n // 2, n)
        seeds.append(_double_exp(n, rate1, pos1, rate2, pos2))

    # include previous best (if any) and a few noisy perturbations
    previous = globals().get("height_sequence_1")
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.ndim == 1:
            if arr.shape[0] != n:
                # resample by linear interpolation to current length
                xs_old = np.linspace(0.0, 1.0, arr.shape[0])
                xs_new = np.linspace(0.0, 1.0, n)
                arr = np.interp(xs_new, xs_old, arr)
            arr = np.maximum(arr, 0.0)
            if arr.sum() == 0.0:
                arr = _uniform(n)
            else:
                arr = arr / arr.sum()
            seeds.append(arr)
            # a few noisy copies
            for _ in range(4):
                noisy = arr + rng.normal(scale=0.01, size=n)
                noisy = np.maximum(noisy, 0.0)
                if noisy.sum() == 0.0:
                    noisy = _uniform(n)
                else:
                    noisy = noisy / noisy.sum()
                seeds.append(noisy)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seed produced.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [v for v, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top seeds (smoothed EG + optional L‑BFGS)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate roughly 60 % of total time to seed‑wise refinements
    seed_time_budget = (deadline - time.time()) * 0.60
    per_seed_budget = max(5.0, seed_time_budget / top_k)   # at least a few seconds

    beta_sched_refine = [5, 20, 80, 200, 800, 2000, 8000, 20000]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_deadline = min(time.time() + per_seed_budget, deadline)
        print(f"\n[Seed {idx+1}] refining (budget ≈ {per_seed_budget:.1f}s)")

        # (a) smoothed EG
        seq_eg, val_eg = _smooth_eg(seed_vec,
                                    sub_deadline,
                                    beta_schedule=beta_sched_refine,
                                    max_iters_per_beta=14_000,
                                    eta0=4.0)

        # (b) optional L‑BFGS smoothing (if time permits)
        if time.time() < sub_deadline and _opt is not None:
            seq_lb = _lbfgs_smooth_optimize(seq_eg,
                                            sub_deadline,
                                            max_iter_per_beta=2000)
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_eg, val_eg

        # (c) final short permutation anneal
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(seq_lb,
                                             sub_deadline,
                                             max_iters=150_000)
        else:
            seq_perm, val_perm = seq_lb, val_lb

        # pick the best of the three refinements
        candidates = [(seq_eg, val_eg), (seq_lb, val_lb), (seq_perm, val_perm)]
        local_seq, local_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing (long high‑β EG run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final polishing phase ---")
        beta_sched_final = [10, 50, 200, 800, 3000, 10000,
                            50000, 200000, 800000]
        # allocate the remaining time to this stage
        final_seq, final_val = _smooth_eg(global_best_seq,
                                          deadline,
                                          beta_schedule=beta_sched_final,
                                          max_iters_per_beta=25_000,
                                          eta0=3.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional L‑BFGS refinement on the very last sequence
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        try:
            seq_lb = _lbfgs_smooth_optimize(global_best_seq,
                                            deadline,
                                            beta_schedule=[1e5, 5e5, 1e6, 2e6, 5e6],
                                            max_iter_per_beta=1500)
            val_lb = _objective_surrogate(seq_lb)
            if val_lb < global_best_val:
                global_best_seq = seq_lb
                global_best_val = val_lb
                print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print("L‑BFGS final stage exception:", e)

    # ------------------------------------------------------------------
    # 6. Very short permutation tweak (few seconds) – may shave a bit
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = min(time.time() + 20.0, deadline)
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print("\n=== Finished ===")
    print(f"Surrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val}")

    # expose for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
