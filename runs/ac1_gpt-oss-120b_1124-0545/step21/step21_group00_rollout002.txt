def _lbfgs_obj(v, beta, n):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta*(conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp)/beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

"""
Hybrid stochastic‑global / local optimiser for the step‑function problem.

The algorithm proceeds as follows

1.  Initialise a pool of diverse normalised seeds (uniform, Dirichlet,
    sparse Dirichlet, Gaussian mixtures, Fourier‑softmax and a copy of the
    previous best if it exists).

2.  Evaluate the cheap surrogate   f(a) = 2·n·max(conv(a,a))   (the true
    objective is f(a)/ (∑a)², however we keep ∑a = 1, which makes the two
    identical). Keep the best seed.

3.  Global optimisation on the two low‑dimensional parametric families
    (Gaussian mixture, Fourier‑softmax) using SciPy's *differential‑evolution*.
    The objective is the same surrogate, the parameter bounds are chosen to
    keep the generated vectors non‑negative and normalised internally.

4.  Refine the current best with a smooth log‑sum‑exp surrogate and L‑BFGS‑B.
    A schedule of increasing β smoothes the max‑operator; the optimiser works
    in the log‑softmax space to keep the iterate on the probability simplex.

5.  Polishing with the exact max‑convolution objective by exponentiated‑gradient
    (mirror‑descent).  The step‑size follows ηₜ = η₀ / √t with a larger initial
    η₀ (5.0) than in the reference run.

6.  A cheap stochastic hill‑climber: small mass transfers between random
    positions are accepted only if they improve the surrogate.

7.  Finally the candidate is evaluated with the provided ``evaluate_sequence``
    and stored in the global variable ``height_sequence_1``.

All heavy work respects a global deadline (budget – 5 s safety margin).
The implementation uses only top‑level helper functions – no closures or
lambda‑expressions – and works with a single CPU (SciPy uses one thread by
default).  The code is self‑contained and ready to be called by the evaluation
environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global placeholders for deadline checks (required for callbacks)
# ----------------------------------------------------------------------
_GLOBAL_DEADLINE = None          # for DE callbacks
_GLOBAL_OPT_DEADLINE = None      # for L‑BFGS callbacks


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Surrogate of the true objective for a probability vector:
    2·n·max(conv(seq,seq)).
    """
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed factories (all vectors are *not* normalised internally)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet on `k` randomly chosen positions."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric families (all factories return *unnormalised* vectors)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a density from a mixture of K Gaussian windows.
    params layout: [w_0..w_{K-1}, μ_0..μ_{K-1}, σ_0..σ_{K-1}]
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """Fourier‑series → logits → softmax."""
    logits = basis @ theta
    return _softmax(logits)


# ----------------------------------------------------------------------
# L‑BFGS optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, beta: float, n: int):
    """
    Smooth surrogate and its gradient.
    Variables are the log‑softmax parameters `v`.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # smooth max via log‑sum‑exp with parameter β
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights for each entry of conv
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_time_callback(_):
    """Interrupt L‑BFGS when the global optimisation deadline is reached."""
    if _GLOBAL_OPT_DEADLINE is not None and time.time() > _GLOBAL_OPT_DEADLINE:
        raise KeyboardInterrupt()
    return False


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000):
    """L‑BFGS‑B on the smooth surrogate (variables = log‑softmax)."""
    global _GLOBAL_OPT_DEADLINE
    _GLOBAL_OPT_DEADLINE = deadline

    # import lazily – SciPy may be unavailable in some environments
    try:
        import scipy.optimize as _opt
    except Exception:          # pragma: no cover
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0,
                         20_000.0, 50_000.0,
                         100_000.0, 200_000.0,
                         500_000.0, 1_000_000.0,
                         2_000_000.0, 5_000_000.0,
                         10_000_000.0]

    v = np.log(start_vec + 1e-30)   # start in log‑softmax space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(_lbfgs_obj,
                                v,
                                args=(beta, n),
                                method='L-BFGS-B',
                                jac=True,
                                callback=_lbfgs_time_callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] unexpected exception:", e)
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the true max‑convolution objective."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Differential‑evolution helper functions (no closures)
# ----------------------------------------------------------------------
def _de_obj_gauss(params: np.ndarray, n: int, K: int) -> float:
    seq = _mixture_gaussian(params, n, K)
    return _objective_surrogate(seq, n)


def _de_obj_fourier(theta: np.ndarray, n: int, basis: np.ndarray) -> float:
    seq = _fourier_softmax(theta, basis)
    return _objective_surrogate(seq, n)


def _de_time_callback(xk, convergence):
    """DE callback – stop when the global deadline has been passed."""
    if _GLOBAL_DEADLINE is not None and time.time() > _GLOBAL_DEADLINE:
        return True
    return False


# ----------------------------------------------------------------------
# Simple hill‑climber on the surrogate
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                n: int,
                rng: np.random.Generator,
                deadline: float,
                max_moves: int = 200_000) -> np.ndarray:
    """Random mass transfers that are kept only if they improve the surrogate."""
    best = seq.copy()
    best_val = _objective_surrogate(best, n)
    moves = 0
    while moves < max_moves and time.time() < deadline:
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # transfer at most 2 % of the source mass
        eps = rng.random() * 0.02 * best[src]
        if eps <= 0.0:
            continue
        cand = best.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        val = _objective_surrogate(cand, n)
        moves += 1
        if val < best_val:
            best = cand
            best_val = val
    return best

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The best sequence found is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final official evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size (same as the reference run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve the previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_norm = None
    if previous is not None and isinstance(previous, list) and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        s = arr.sum()
        if s > 0.0:
            prev_norm = arr / s

    # ------------------------------------------------------------------
    # 1️⃣ Initialise a diverse pool of seeds (all normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_norm is not None:
        seeds.append(prev_norm)

    # dense Dirichlet with several concentrations
    for conc in (0.02, 0.1, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse Dirichlet (supports 80 … 160)
    for k in (80, 120, 160):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # tiny Gaussian‑mixture seeds (8 components)
    K_mix = 8
    for _ in range(6):
        params = np.concatenate([
            rng.random(K_mix),                     # raw weights
            rng.random(K_mix),                     # means
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n, K_mix))

    # Fourier‑softmax seeds (20 basis functions)
    D_fourier = 20
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis_parts = []
    for k in range(1, D_fourier // 2 + 1):
        basis_parts.append(np.sin(2.0 * np.pi * k * i))
        basis_parts.append(np.cos(2.0 * np.pi * k * i))
    basis = np.column_stack(basis_parts)   # shape (n, D_fourier)

    for _ in range(6):
        theta = rng.normal(scale=1.0, size=D_fourier)
        seeds.append(_fourier_softmax(theta, basis))

    # a few perturbed uniforms
    for _ in range(3):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.005))

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate the seed pool (surrogate) and keep the best so far
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Evaluating initial seeds ---")
    for idx, raw in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = raw / raw.sum()           # ensure sum = 1
        val = _objective_surrogate(vec, n)
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Global DE on the two parametric families
    # ------------------------------------------------------------------
    try:
        import scipy.optimize as _opt
    except Exception:               # pragma: no cover
        _opt = None

    # ---- Gaussian mixture DE ------------------------------------------------
    global _GLOBAL_DEADLINE
    _GLOBAL_DEADLINE = deadline
    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution on Gaussian mixture ---")
        bounds = [(0.0, 1.0)] * K_mix + [(0.0, 1.0)] * K_mix + [(0.001, 0.5)] * K_mix
        try:
            result = _opt.differential_evolution(
                _de_obj_gauss,
                bounds,
                args=(n, K_mix),
                seed=int(seed),
                popsize=20,
                maxiter=60,
                polish=False,
                updating='deferred',
                workers=1,
                init='latinhypercube',
                disp=False,
                callback=_de_time_callback,
            )
            seq = _mixture_gaussian(result.x, n, K_mix)
            val = _objective_surrogate(seq, n)
            print(f"[DE‑Gauss] surrogate = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        except Exception as e:      # pragma: no cover
            print("[DE‑Gauss] failed:", e)

    # ---- Fourier‑softmax DE ------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution on Fourier‑softmax ---")
        bounds = [(-6.0, 6.0)] * D_fourier
        try:
            result = _opt.differential_evolution(
                _de_obj_fourier,
                bounds,
                args=(n, basis),
                seed=int(seed + 1),
                popsize=15,
                maxiter=55,
                polish=False,
                updating='deferred',
                workers=1,
                init='latinhypercube',
                disp=False,

            seq = _fourier_softmax(result.x, basis)
            val = _objective_surrogate(seq, n)
            print(f"[DE‑Fourier] surrogate = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        except Exception as e:      # pragma: no cover
            print("[DE‑Fourier] failed:", e)

    # ------------------------------------------------------------------
    # 4️⃣ Local refinement with smooth L‑BFGS (log‑sum‑exp)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        sub_deadline = time.time() + remaining * 0.45
        print("\n--- L‑BFGS smoothing of current best ---")
        seq_lbfgs = _lbfgs_smooth_optimize(
            best_seq, n, sub_deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                           200.0, 500.0, 1_000.0,
                           2_000.0, 5_000.0,
                           20_000.0, 50_000.0,
                           100_000.0, 200_000.0,
                           500_000.0, 1_000_000.0,
                           2_000_000.0, 5_000_000.0,

            max_iter_per_beta=30000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs, n)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("  >>> Updated global best after L‑BFGS")

    # ------------------------------------------------------------------
    # 5️⃣ Exact max‑convolution polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polish_deadline = deadline - 5.0
        time_left = polish_deadline - time.time()

            max_iters=max_iters,
            eta0=5.0,

            print("  >>> Updated global best after EG")

    # 6️⃣ cheap stochastic hill‑climbing passes
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # short perturb‑and‑accept loop
        candidate = _perturb(best_seq, rng, sigma=0.0015)
        candidate = candidate / candidate.sum()
        candidate = _hill_climb(candidate, n, rng,
                                deadline - 1.0,
                                max_moves=50_000)
        val = _objective_surrogate(candidate, n)
        print(f"[Hill‑Climb] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, candidate.copy()
            print("  >>> New best after hill‑climb")
        # break if we are about to exceed the deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # 7️⃣ Final official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
