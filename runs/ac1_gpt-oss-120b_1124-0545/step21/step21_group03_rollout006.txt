"""
Improved optimiser for the step‑function problem.

The algorithm works as follows:

1. **Seed construction** – a diverse collection of ℓ₁‑normalised vectors is
   generated (uniform, exponential, symmetric exponential, power‑law,
   beta‑shaped, Dirichlet draws, random piecewise‑linear, and permutations of
   the previous best solution).

2. **Coarse evaluation** – the cheap surrogate 2·n·max(conv) is computed for all
   seeds and the best few (top‑k) are kept.

3. **Smooth Adam optimisation** – each top‑seed is refined by Adam applied to
   a *smooth* surrogate of the max‑convolution.  The smooth surrogate is the
   log‑sum‑exp (with a growing temperature β), whose gradient can be computed
   exactly via two FFT‑based convolutions.  The schedule `β = [1,5,20,80,300,
   1e3,3e3,1e4,3e4,1e5]` is used, with a few thousand Adam steps at each β.

   Adam works in the log‑softmax space (the variables are the logarithms of the
   probabilities), guaranteeing non‑negativity and the ℓ₁‑normalisation
   constraint.

4. **Optional L‑BFGS smoothing** – if SciPy is available, a few L‑BFGS‑B
   iterations are performed on the same smooth objective (continuation on β).

5. **Permutation‑based simulated annealing** – a short swap‑annealing stage
   removes any remaining ordering artefacts.

6. The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned.  The programme respects the total
   time‑budget (default 1000 s) by checking the deadline frequently.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for a final L‑BFGS smoothing (if available).
# ----------------------------------------------------------------------
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth objective and its gradient (log‑softmax variables)
# ----------------------------------------------------------------------
def _smooth_obj_and_gradient(v: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (shift, grad_v) where `shift` = max(conv(a,a)) and
    `grad_v` is the gradient of the smoothed objective
    (with the log‑sum‑exp smoothing of parameter `beta`)
    w.r.t. the log‑softmax variables `v`.
    """
    n = len(v)
    # probability vector
    a = np.exp(v)
    a_sum = a.sum()
    a /= a_sum

    # convolution a * a
    conv = _conv_fft(a, a)
    shift = np.max(conv)                     # = max_{k} (a*a)[k]

    diff = conv - shift                      # non‑positive, zero at the active index
    expb = np.exp(beta * diff)               # safe because diff ≤ 0
    sum_exp = expb.sum()
    w = expb / sum_exp                       # soft‑max weights over the full convolution

    # gradient w.r.t. a (see derivation in the statement)
    a_rev = a[::-1]
    full = _conv_fft(a_rev, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]    # centred part

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)              # chain rule for log‑softmax

    return shift, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   beta_schedule: list[float],
                   steps_per_beta: int = 2000,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    Returns the best sequence (ℓ₁‑norm = 1) found and its surrogate value.
    """
    n = len(start_seq)
    v = np.log(start_seq + 1e-30)
    v -= np.max(v)                 # re‑centre for numerical stability

    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    t = 0

    # best so far
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq, best_val

            t += 1
            shift, grad_v = _smooth_obj_and_gradient(v, beta)

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

            # keep the log‑variables centred to avoid overflow
            v -= np.max(v)

            # current surrogate value (exact, because sum(a)=1)
            cur_val = 2.0 * n * shift
            if cur_val < best_val:
                # reconstruct the current probability vector a
                a = np.exp(v)
                a /= a.sum()
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Minimise a smoothed version of max(conv) using L‑BFGS‑B.
    Operates in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 300.0,
            1_000.0, 3_000.0, 10_000.0,
            30_000.0, 100_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            expb = np.exp(beta_val * (conv - shift))
            sum_exp = expb.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            w = expb / sum_exp
            a_rev = a[::-1]
            full = _conv_fft(a_rev, w)
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure builds a diverse set of seeds, refines the best few with a
    smooth Adam optimiser (continuation on the temperature β), optionally
    smooths further with L‑BFGS‑B, and finishes with a short permutation‑based
    simulated annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0           # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse length of previous solution if present.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            n = int(len(previous))
        except Exception:
            n = 4000
    else:
        n = 4000

    # ------------------------------------------------------------------
    # 1. Build a rich pool of seeds (ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_sym_exp_decay(n, rate=0.0015))
    seeds.append(_power_law(n, p=0.7))
    seeds.append(_power_law(n, p=1.2))
    seeds.append(_beta_shape(n, alpha=0.5, beta=1.5))
    seeds.append(_beta_shape(n, alpha=1.5, beta=0.5))

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear random shapes
    for nb in (3, 5, 8):
        for _ in range(3):
            seeds.append(_piecewise_linear(n, nb, rng))

    # include the previous best and a handful of its permutations
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,):
                seeds.append(prev / prev.sum())
                for _ in range(5):
                    seeds.append(np.random.permutation(prev) / np.sum(prev))
        except Exception:
            pass

    # random perturbations of the previous best (log‑space noise)
    if previous is not None:
        try:
            base = np.asarray(previous, dtype=np.float64)
            if base.shape == (n,):
                v = np.log(base + 1e-30)
                for _ in range(4):
                    noise = rng.normal(scale=0.15, size=n)
                    v2 = v + noise
                    a = np.exp(v2)
                    a /= a.sum()
                    seeds.append(a)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop – Adam on smoothed max‑convolution
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    beta_sched = [1.0, 5.0, 20.0, 80.0, 300.0,
                  1_000.0, 3_000.0, 10_000.0, 30_000.0, 100_000.0]

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget
        remaining = deadline - time.time()
        # each seed gets roughly an equal share (but never less than 10 s)
        share = max(10.0, remaining / max(1, (top_k - si)))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        seq_adam, val_adam = _adam_optimize(seed_vec,
                                            sub_deadline,
                                            beta_schedule=beta_sched,
                                            steps_per_beta=2500,
                                            lr=0.05)

        # optional L‑BFGS smoothing if time permits
        if time.time() < sub_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_adam,
                                            sub_deadline,
                                            beta_schedule=beta_sched,
                                            max_iter_per_beta=2000)
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_adam, val_adam

        # quick permutation annealing (few seconds)
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(seq_lb,
                                              sub_deadline,
                                              max_iters=100_000)
        else:
            seq_perm, val_perm = seq_lb, val_lb

        # select best of the three refinements
        candidates = [(seq_adam, val_adam), (seq_lb, val_lb), (seq_perm, val_perm)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long Adam polishing (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final long Adam polishing (full remaining budget) ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                               deadline,
                                               beta_schedule=beta_sched,
                                               steps_per_beta=4000,
                                               lr=0.04)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak – may shave a tiny amount
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 15.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
