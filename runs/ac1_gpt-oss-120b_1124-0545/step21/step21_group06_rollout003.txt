"""
Improved optimiser for the step‑function bound.

The algorithm works in three phases:

1. **Initialisation** – start from the best known global sequence
   (`height_sequence_1`) if available, otherwise sample a few
   Dirichlet candidates and keep the one with the smallest exact
   max‑convolution (the true objective when the ℓ₁‑norm is
   √(2 n)).

2. **Iterative LP‑mixing** – at each iteration we fix the current
   sequence `a`, compute `M = max(conv(a,a))` and solve a linear
   programme

        maximise   Σ b_j
        subject to  conv(a,b) ≤ M   (element‑wise)
                    b_j ≥ 0

   (the convolution constraint is a Toeplitz matrix and is built
   with a sparse diagonal representation).  The solution `b` is
   re‑scaled to the required ℓ₁‑norm `√(2 n)` and mixed with `a`
   using a convex combination `a ← (1‑t)·a + t·b`.  Only updates that
   actually decrease the objective are kept.  A few
   exponentiated‑gradient (mirror‑descent) steps are performed after
   each mix to clean up numerical noise.

3. **Polishing** – after the LP stage the current best sequence is
   refined by a long exponentiated‑gradient run (single‑active
   index for speed) and a final high‑budget simulated‑annealing
   pass.  The final sequence is returned and stored as
   ``height_sequence_1`` for possible reuse.

All work respects the supplied time budget (default 1000 s) and
only uses NumPy and SciPy (no file or network I/O).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – required for the LP step

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent) on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered (unit ℓ₁‑norm) and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    # surrogate = 2·n·max(conv) for unit‑sum vectors
    best_val = 2.0 * n * _conv_fft(a, a).max()

    log_a = np.log(a + 1e-30)   # stay in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active indices -----
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient -------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning rate -----
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * _conv_fft(a, a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline (unit‑sum).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * _conv_fft(a, a).max()
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = 2.0 * n * _conv_fft(ns, ns).max()
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sparse Toeplitz matrix for the convolution constraint (A·b ≤ M)
# ----------------------------------------------------------------------
def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A b)[k] = (a * b)[k].
    Each diagonal offset d≥0 contains the constant a[d].
    """
    n = a.shape[0]
    # each diagonal has n entries – we use np.full to create them
    data = [np.full(n, val, dtype=np.float64) for val in a]
    offsets = np.arange(n, dtype=np.intc)   # 0 … n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget
    and stores the best found sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – we target a modest length that keeps the LP cheap.
    # If a previous solution exists we reuse its length (capped at 2500).
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = min(2500, arr.shape[0])
            a0 = arr[:n].copy()
        except Exception:
            n = 2500
            a0 = None
    else:
        n = 2500
        a0 = None

    target_sum = math.sqrt(2.0 * n)   # the normalisation required by evaluate_sequence

    # ------------------------------------------------------------------
    # Initialise a: use previous sequence if available, otherwise
    # draw a few Dirichlet candidates and keep the best.
    # ------------------------------------------------------------------
    if a0 is not None and a0.sum() > 0.0:
        a = a0
    else:
        # generate a handful of random Dirichlet draws
        candidates = [_uniform(n)]
        for conc in (0.005, 0.02, 0.05):
            for _ in range(4):
                candidates.append(rng.dirichlet(np.full(n, conc)))
        # pick the one with smallest max‑convolution
        best_val = np.inf
        best_vec = None
        for vec in candidates:
            vec = np.maximum(vec, 0.0)
            vec = vec / vec.sum()
            conv = _conv_fft(vec, vec)
            val = conv.max()                     # unit‑sum case → true objective = val * (target_sum**2) / (2n)
            # we compare on the unit‑sum surrogate 2·n·val, which is monotone in val
            if val < best_val:
                best_val = val
                best_vec = vec
        a = best_vec

    # Scale to the required ℓ₁‑norm
    a = a * (target_sum / a.sum())
    a = np.maximum(a, 0.0)   # safeguards against tiny negatives from rounding

    # Helper to compute the true objective (max‑convolution) for a given a
    def _objective(seq: np.ndarray) -> float:
        """Return max(conv(seq, seq)) – the value of evaluate_sequence after normalisation."""
        return float(_conv_fft(seq, seq).max())

    # ------------------------------------------------------------------
    # Keep track of the best overall sequence
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_val = _objective(best_seq)
    print(f"\n--- Optimisation started (n={n}) ---")
    print(f"Initial objective (max‑conv) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # LP‑mixing loop
    # ------------------------------------------------------------------
    max_lp_iters = 25               # hard cap on LP rounds
    t_mix = 0.25                    # convex‑combination weight
    lp_iter = 0

    while time.time() < deadline - 8.0 and lp_iter < max_lp_iters:
        lp_iter += 1
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Iter {lp_iter:2d}] New global best max‑conv = {best_val:.6f}")

        # ---- Build the sparse convolution matrix for the current a ----
        if sp is None or opt is None:
            print("SciPy sparse/optimise not available – skipping LP stage.")
            break
        A = _conv_constraint_matrix(a)

        # ---- Linear programme: maximise Σ b_j  subject to A b ≤ M, b ≥ 0 ----
        M = cur_val
        c = -np.ones(n)                       # minimise -Σb ⇒ maximise Σb
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        # Solve with the high‑performance HiGHS interior‑point method
        try:
            res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs',
                              options={'presolve': True})
        except Exception as e:   # pragma: no cover
            print(f"Linear programme exception: {e}")
            break

        if not res.success:
            print(f"LP failed at iteration {lp_iter}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0.0:
            print("LP returned zero‑sum vector – aborting LP phase.")
            break
        # Re‑scale b to the required ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # ---- Mix a and g (accept only if improvement) ----
        a_mix = (1.0 - t_mix) * a + t_mix * g
        mix_val = _objective(a_mix)

        if mix_val < cur_val:
            a = a_mix
            # tiny renormalisation to guard against round‑off drift
            a = a * (target_sum / a.sum())
            print(f"[Iter {lp_iter:2d}] Mix improved max‑conv to {mix_val:.6f}")
        else:
            # no improvement – reduce mixing weight and try again
            t_mix *= 0.5
            print(f"[Iter {lp_iter:2d}] Mix did not improve (mix_val={mix_val:.6f}); t reduced to {t_mix:.4f}")
            if t_mix < 1e-4:
                print("Mixing weight too small – terminating LP loop.")
                break

        # ---- Quick EG clean‑up (few hundred iterations) ----
        sub_dead = min(deadline, time.time() + 2.0)
        a_norm = a / a.sum()
        a_ref, _ = _exp_gradient_opt(a_norm, n,
                                     sub_dead,
                                     max_iters=150_000,
                                     eta0=5.0,
                                     single_active=False)
        a = a_ref * target_sum   # re‑scale back
        # loop continues

    # ------------------------------------------------------------------
    # Final intensive polishing – EG (single‑active) then SA
    # ------------------------------------------------------------------
    # Final EG (large budget)
    if time.time() < deadline - 6.0:
        print("\n--- Final intensive EG polishing (single‑active) ---")
        sub_dead = time.time() + max(5.0, (deadline - time.time()) * 0.45)
        a_norm = a / a.sum()
        a_ref, _ = _exp_gradient_opt(a_norm, n,
                                     sub_dead,
                                     max_iters=2_500_000,
                                     eta0=6.0,
                                     single_active=True)
        a = a_ref * target_sum
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Final EG] improved to {best_val:.6f}")

    # Final SA (high budget)
    if time.time() < deadline - 2.0:
        print("\n--- Final SA polishing (high budget) ---")
        sub_dead = time.time() + max(5.0, (deadline - time.time()) * 0.45)
        a_norm = a / a.sum()
        a_sa, _ = _simulated_annealing(a_norm, n,
                                       sub_dead,
                                       rng,
                                       max_attempts=300_000,
                                       start_sigma=0.001,
                                       T_start=5e-3,
                                       T_end=1e-9)
        a = a_sa * target_sum
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Final SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest max‑conv = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()


# ----------------------------------------------------------------------
# Analytic family helpers (used only for the very first random seeds)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n entries (unit sum)."""
    return np.full(n, 1.0 / n, dtype=np.float64)
