"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The algorithm proceeds in several stages:

1.   Initialise a diverse pool of seed vectors (uniform, arcsine,
     triangular, several Beta‑distributions, Dirichlet samples and the
     previous best if available).  Keep the best seed.

2.   Extensive random search in the 2‑parameter Beta family and
     random convex combinations of the base seeds.

3.   A short L‑BFGS optimisation of the smooth log‑sum‑exp surrogate.

4.   (Optional) A linear‑programming refinement inspired by the classic
     “LP‑direction’’ method: maximise the ℓ₁‑mass of a partner vector g
     under the constraint ‖f∗g‖∞ ≤ M, then mix f and the normalised g.

5.   A fast exponentiated‑gradient polish (mirror–descent on the exact
     max‑convolution).

6.   Simulated‑annealing style local search that repeatedly moves a
     small amount of mass between two coordinates, accepting worse
     moves with a temperature schedule.

All stages respect the global time budget (default 1000 s) and the best
vector encountered is stored in the global variable ``height_sequence_1``.
The returned list is already normalised to sum to 1, which makes the
surrogate 2·n·max(conv) equal to the true evaluate_sequence() value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for global optimisation & LP)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # d/d a of weighted sum
            grad_a = grad_a * (2.0 * n)               # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search (tiny, many moves)
# ----------------------------------------------------------------------
def _simulated_annealing(seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         sigma_start: float = 0.05,
                         sigma_end: float = 0.001,
                         temp_start: float = 0.001,
                         temp_end: float = 1e-6) -> tuple[np.ndarray, float]:
    cur = seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 1.0
        sigma = sigma_start * (1 - frac) + sigma_end * frac
        temp = temp_start * (1 - frac) + temp_end * frac

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = sigma * cur[i] * rng.random()
        if delta <= 0.0 or cur[i] - delta < 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_val = _objective_surrogate(new)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probabilistic acceptance
            if rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val

    return best, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑direction step)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 5,
               mix_factor: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Implements the LP direction described in the problem statement:
    maximise Σ g_j s.t. conv(f, g) ≤ M (pointwise) and g ≥ 0.
    Then mix f with the normalised optimiser.
    """
    if _opt is None or not hasattr(_opt, "linprog"):
        return seq, _objective_surrogate(seq)

    best = seq.copy()
    best_val = _objective_surrogate(best)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        f = best
        conv_f = _conv_fft(f, f)
        M = float(conv_f.max())

        L = 2 * n - 1
        # Build dense A_ub matrix (each column is a shifted copy of f)
        A = np.empty((L, n), dtype=np.float64)
        for col in range(n):
            A[col:col + n, col] = f
        b_ub = np.full(L, M)

        # maximise sum(g) -> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=(0, None),
                               method='highs',
                               options={'presolve': True})
        except Exception as e:  # pragma: no cover
            print("[LP] linprog error:", e)
            break

        if not res.success or res.x is None:
            # No feasible LP or other issue
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()  # renormalise to unit ℓ₁

        # convex combine
        new_f = (1.0 - mix_factor) * f + mix_factor * g
        new_val = _objective_surrogate(new_f)

        if new_val < best_val - 1e-12:
            best, best_val = new_f, new_val
            print(f"[LP] iteration {it+1}: improved value {best_val:.6f}")
        else:
            # no improvement – stop early
            break

    return best, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search runs for at most ``budget_s`` seconds and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse pool of seed vectors (all sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, a, b))

    # Dirichlet draws at several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few perturbed uniform vectors
    for _ in range(4):

    if prev_arr is not None:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Beta‑parameter sweep
    # ------------------------------------------------------------------
    if time.time() < deadline:

            alpha = rng.uniform(0.1, 5.0)
            beta = rng.uniform(0.1, 5.0)
            seq = _beta_distribution(n, alpha, beta)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"   improved to {best_val:.6f} (α={alpha:.3f}, β={beta:.3f})")

    # ------------------------------------------------------------------
    # 3️⃣  Random convex mixtures of the base seeds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random mixtures of base shapes ---")
        base_mat = np.stack(seeds, axis=0)          # shape (m, n)
        m = base_mat.shape[0]
        trials = 3000
        for _ in range(trials):
            if time.time() > deadline:

            w = rng.dirichlet(np.ones(m))
            seq = np.dot(w, base_mat)
            seq = seq / seq.sum()
            val = _objective_surrogate(seq)

    # ------------------------------------------------------------------

    return best_seq.tolist()
