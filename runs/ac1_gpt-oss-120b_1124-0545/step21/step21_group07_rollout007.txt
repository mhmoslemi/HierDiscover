def _adam_optimize_max(start_seq, n, deadline, max_iters=500_000, lr0=0.5):
    v = np.log(start_seq+1e-30)
    m = np.zeros_like(v)
    vhat = np.zeros_like(v)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = None
    best_val = np.inf
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        maxc = conv.max()
        # active set: indices within tolerance
        tol = maxc * 1e-12
        active = np.where(conv >= maxc - tol)[0]
        # subgradient g
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size
        # gradient w.r.t v (mirror descent)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)
        # Adam updates
        m = beta1*m + (1-beta1)*grad_v
        vhat = beta2*vhat + (1-beta2)*(grad_v*grad_v)
        m_hat = m/(1-beta1**t)
        vhat_hat = vhat/(1-beta2**t)
        lr = lr0 / (np.sqrt(t))
        v = v - lr * m_hat/(np.sqrt(vhat_hat)+eps)
        # check record
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq if best_seq is not None else start_seq, best_val

"""
Propose a non‑negative step‑function that minimizes the provided evaluation
function.  The algorithm works as follows:

1.  Generate a large, diverse set of analytic “seed’’ sequences (uniform,
    triangular, sine, Kaiser, Gaussian, exponential, power‑law, beta‑shaped,
    logistic, arcsine, parabolic, raised‑cosine, and random Dirichlet draws).
    All seeds are ℓ₁‑normalised.

2.  Quickly rank the seeds by the exact surrogate objective
    ``2·n·max(conv(a,a))`` (the same quantity that ``evaluate_sequence``
    computes after normalisation).  Keep the best *top_k* seeds.

3.  Refine each top seed using a mirror‑descent Adam optimiser that works in
    log‑softmax space and uses the exact sub‑gradient of the max‑convolution
    (single‑active or multi‑active).  The optimiser respects the simplex
    constraints and runs until the allocated sub‑deadline expires.

4.  Run a short simulated‑annealing (SA) phase from each refined candidate
    to escape possible local minima.

5.  Keep the best candidate found so far.  After the multi‑start phase,
    allocate the remaining time to a long final Adam polishing (single‑active)
    followed by a brief SA polish.

All operations are vectorised and use FFT‑based convolution for speed.
The routine respects the 1000 s budget, stores the best sequence in the
global variable ``height_sequence_1`` and returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))  (assumes sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _parabolic_window(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - ((i - c) / c) ** 2
    a = np.maximum(a, 0.0) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey‑style window: cosine taper controlled by alpha∈[0,1]."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    # taper left side
    left = i < alpha * (n - 1) / 2.0
    a[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / (alpha * (n - 1)) - 1)))
    # taper right side
    right = i > (n - 1) * (1 - alpha / 2.0)
    a[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    # Fallback to uniform if SciPy's i0 is not present
    try:
        from scipy.special import i0 as _i0
    except Exception:  # pragma: no cover
        return _uniform(n)

    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i·(n‑1‑i)) (zero at the very ends)."""
    if n < 3:
        return _uniform(n)
    i = np.arange(1, n - 1, dtype=np.float64)  # avoid zero denominator
    a_mid = 1.0 / np.sqrt(i * (n - 1 - i))
    a = np.concatenate(([0.0], a_mid, [0.0]))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture (1‑α)·a + α·b, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (mirror‑descent)
# ----------------------------------------------------------------------
def _adam_optimize_max(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 600_000,
                       lr0: float = 0.8,
                       single_active: bool = True):
    """
    Mirror‑descent Adam optimiser that minimises the exact max‑convolution.
    Parameters
    ----------
    start_seq: np.ndarray
        Initial ℓ₁‑normalised probability vector (sum = 1).
    n: int
        Length of the vector.
    deadline: float
        Absolute time (seconds since epoch) after which the optimiser stops.
    max_iters: int
        Upper bound on iteration count.
    lr0: float
        Initial learning‑rate scaling factor.
    single_active: bool
        If True use only the (single) maximum index; otherwise use all
        indices within a tiny tolerance of the maximum.
    Returns
    -------
    best_seq, best_val : (np.ndarray, float)
        Best sequence found and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v_hat = np.zeros_like(v)              # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector
        conv = _conv_fft(a, a)
        max_c = conv.max()

        # active set for the sub‑gradient
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            tol = max_c * 1e-12
            active = np.where(conv >= max_c - tol)[0]

        # sub‑gradient of max (averaged if multiple active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # mirror‑descent (log‑softmax) gradient
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update on the log‑variables
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    # final safety evaluation
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """Simple SA with Gaussian perturbations and linear cooling."""
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    for scale in [0.5, 1.0, 2.0, 4.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # Arcsine‑type shape (often very flat autocorrelation)
    seeds.append(_arcsine_shape(n))

    # Random Dirichlet draws (various concentrations)
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform with exponential (different alphas)
    exp_rates = np.logspace(-5, -0.5, 7)
    for r in exp_rates:
        exp_seq = _exp_decay(n, r)
        for alpha in (0.02, 0.10, 0.25, 0.40):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a modest slice of the remaining budget to each seed
        remaining_time = deadline - time.time()
        # give at most 12 % of remaining time per seed, but at least a few seconds
        sub_deadline = time.time() + max(5.0, 0.12 * remaining_time)

        # ---- (a) Adam optimiser on the exact max (single‑active) ----
        seq_adam, val_adam = _adam_optimize_max(
            seed_vec.copy(),
            n,
            sub_deadline,
            max_iters=400_000,
            lr0=1.0,
            single_active=True,
        )

        # ---- (b) Short simulated annealing ----------------------
        # give a little extra budget if time permits
        if time.time() < sub_deadline - 1.0:
            seq_sa, val_sa = _simulated_annealing(
                seq_adam,
                n,
                sub_deadline - 1.0,
                rng,
                max_attempts=120_000,
                start_sigma=0.002,
                T_start=0.012,
                T_end=5e-9,
            )
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # Compare candidates (original seed, Adam, SA)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long‑run polishing (Adam + SA)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing (single‑active) ---")
        final_seq, final_val = _adam_optimize_max(
            global_best_seq,
            n,
            deadline,
            max_iters=5_000_000,
            lr0=0.6,
            single_active=True,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=150_000,
            start_sigma=0.001,
            T_start=4e-3,
            T_end=1e-9,
        )
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
