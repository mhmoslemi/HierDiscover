"""
Propose a step‑function sequence that (approximately) minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

where n = len(seq).  The algorithm performs many independent restarts,
each consisting of a quick stochastic annealing phase followed by
exponentiated‑gradient (mirror‑descent) polishing and an optional L‑BFGS
smooth optimisation.  The best sequence seen over all restarts is
returned.  The implementation respects the 1000 s time budget and keeps
all helper functions at module level (no closures, no lambdas).
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact if sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by mirror‑descent)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient mirror‑descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_500_000,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on max‑convolution.
    ``start_seq`` must already sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (Kullback–Leibler geometry)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing on the simplex (swap + mass transfer moves)
# ----------------------------------------------------------------------
def _value_anneal(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing on the simplex.
    Returns the best sequence found and its surrogate value.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule: linear from cur_val down to 1e‑6·cur_val
    T_start = cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.7:          # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] == 0.0:
                continue
            # move at most half of the source mass
            delta = cur_seq[i] * rng.random() * 0.5
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta
        else:                           # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)

        # Metropolis acceptance
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS polishing (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt   # type: ignore
    import scipy.sparse as _sp      # type: ignore
except Exception:  # pragma: no cover
    _opt = None
    _sp = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2_000) -> np.ndarray:
    """
    Optimise a smooth approximation of max‑convolution (log‑sum‑exp)
    using L‑BFGS‑B. Operates in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabilise exponentials
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # mirror‑descent gradient (projected onto simplex)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Run a single optimisation pipeline (anneal → EG → optional L‑BFGS)
# ----------------------------------------------------------------------
def _run_optimization(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    From a starting probability vector (sum=1) run value‑annealing,
    exponentiated‑gradient polishing and a final L‑BFGS smoothing
    (if time permits).  Returns the best sequence and its surrogate.
    """
    # 1. Quick annealing to escape very poor starts
    anneal_deadline = deadline - 30.0          # leave time for polishing
    seq_ann, val_ann = _value_anneal(start_seq, n, anneal_deadline, rng,
                                     max_iters=300_000)

    # 2. Exact mirror‑descent polishing
    seq_eg, val_eg = _exp_gradient_opt(seq_ann, n, deadline,
                                       max_iters=1_200_000, eta0=5.0)

    # 3. Optional L‑BFGS refinement
    best_seq, best_val = (seq_eg, val_eg)
    if time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(seq_eg, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_seq, best_val = seq_lb, val_lb

    return best_seq, best_val

# ----------------------------------------------------------------------

    """

    ``evaluate_sequence``.  The routine:

      * loads any previously stored best,
      * performs many independent restarts (random Dirichlet initialisation),
      * each restart runs a short stochastic annealing, a long
        exponentiated‑gradient mirror‑descent polishing and optional L‑BFGS,
      * keeps the globally best sequence,
      * finally evaluates it with the provided secure function.

    The whole search respects the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size – a fixed length that works well in practice.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        seq0 = np.maximum(seq0, 0.0)
        if seq0.sum() == 0.0:

        else:
            seq0 = seq0 / seq0.sum()
        best_seq = seq0.copy()
        best_val = _objective_surrogate(best_seq)
        print(f"[Prev] surrogate of previous best = {best_val:.6f}")
    else:

        i = np.arange(n, dtype=np.float64)
        rate = 0.003

        seq0 = a / a.sum()
        best_seq = seq0.copy()
        best_val = _objective_surrogate(best_seq)
        print(f"[Init] surrogate of exponential start = {best_val:.6f}")

    # --------------------------------------------------------------

        sub_deadline = time.time() + min(150.0, time_left * 0.5)

        # keep the best overall

            best_seq = seq_cand.copy()

        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
