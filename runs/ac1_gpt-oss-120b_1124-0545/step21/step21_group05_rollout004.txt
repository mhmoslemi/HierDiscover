def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
```

def _proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
optimised_step_function_v2.py

Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The strategy is:

1.  Build a modest pool of analytic shapes (uniform, triangular,
    sine‑window, Gaussian, shifted‑Gaussian, exponential, symmetric
    exponential, power‑law, beta‑shapes and a few Dirichlet draws).  All
    vectors are ℓ₁‑normalised (sum = 1) – this is exactly the scale‑invariant
    form of the original problem.

2.  Keep the best *K* (default 5) seeds according to the cheap surrogate
    ``2·n·max(conv(p,p))`` – this surrogate equals the true objective when
    ``sum(p)=1`` and therefore can be evaluated with a single FFT.

3.  For every seed run a **long exponentiated‑gradient (mirror descent)**
    optimisation on the true max‑convolution.  The update works in log‑space
    (soft‑max) and therefore automatically stays feasible (non‑negative,
    unit‑sum).  A generous iteration budget (up to the remaining time) is
    allocated, with a decreasing stepsize ``η = η₀/√t``.  The best sequence
    over all seeds becomes the global best.

4.  A short **L‑BFGS** smoothing phase (only if SciPy is available) is run
    on the smooth log‑sum‑exp approximation of the max‑convolution.  This
    often refines the solution beyond what pure EG can achieve.

5.  Finally a tiny **additive sub‑gradient** pass with Euclidean projection
    onto the probability simplex provides a high‑precision fine‑tune.

6.  The resulting sequence is evaluated with the provided
    ``evaluate_sequence`` function and stored in the global variable
    ``height_sequence_1`` for possible continuation runs.

The whole routine respects the overall 1000 s budget; the heavy work is
concentrated on the exponentiated‑gradient step which is the most effective
for reducing the max‑convolution term.
"""

# ----------------------------------------------------------------------
# FFT‑based full convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(p)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(p: np.ndarray) -> float:
    """2·n·max(conv(p,p)) for a probability vector p (sum(p)=1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    """Gaussian with adjustable mean (mu_frac∈[0,1] gives position)."""
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Normalisation & surrogate helper
# ----------------------------------------------------------------------
def _norm_and_surrogate(vec: np.ndarray) -> tuple[np.ndarray, float]:
    """Clip, normalise to sum = 1, return the surrogate value."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return None, np.inf
    vec = vec / s
    return vec, _objective_surrogate(vec)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the
    active index).  Works in log‑space, i.e. a = softmax(log_a)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Additive sub‑gradient with Euclidean projection onto the simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup
    w = w / w.sum()
    return w


def _additive_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 500_000,
                           eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent on max‑convolution with projection onto
    the simplex.  Good for a final high‑precision polish."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a = _proj_simplex(a - eta * g)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule=None,
                      max_iter_per_beta: int = 1500):
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth approximation of max
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (from smoothed max)
            w = exp_beta / sum_exp                     # weights over indices
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # convert to gradient w.r.t. v (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The routine builds a small diverse pool of analytic
    shapes, refines the best few with a long exponentiated‑gradient run,
    optionally smooths with L‑BFGS, and finishes with an additive sub‑gradient
    polish.  The best sequence is stored in the global variable
    ``height_sequence_1`` for possible continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            prev_arr = arr / arr.sum()

    # ------------------------------------------------------------------
    # 1. Seed generation (analytic families + a few random draws)
    # ------------------------------------------------------------------
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian_shift(n, sigma_frac=0.12, mu_frac=0.3))
    seeds.append(_gaussian_shift(n, sigma_frac=0.12, mu_frac=0.7))
    seeds.append(_exp_decay(n, rate=0.02))
    seeds.append(_sym_exp_decay(n, rate=0.02))
    seeds.append(_power(n, p=2.0))
    seeds.append(_power(n, p=1.0))
    seeds.append(_beta_shape(n, alpha=0.5, beta=0.5))   # arcsine‑type shape
    seeds.append(_beta_shape(n, alpha=0.8, beta=0.8))

    # A few Dirichlet draws (low concentration favours sparsity)
    for conc in (0.03, 0.07, 0.15):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the top K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec, val = _norm_and_surrogate(vec)
        if vec is not None:
            evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    K = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:K]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:K]):
        print(f"  #{i+1}: {v:.8f}")

    # ------------------------------------------------------------------
    # 3. Long exponentiated‑gradient polishing on each top seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.8f}")

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # Allocate a fair share of the remaining budget (≈80 % of what's left)
        remaining = deadline - time.time()
        share = remaining * 0.80 / max(1, (K - si))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] EG polishing budget ≈ {share:.1f}s")
        polished_seq, polished_val = _exp_gradient_opt(seed_vec,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=10_000_000,
                                                       eta0=12.0)
        if polished_val < global_best_val:
            global_best_seq, global_best_val = polished_seq, polished_val
            print(f"    *** New global best (EG) = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS smoothing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        sub_deadline = deadline - 5.0   # leave a few seconds for final tweak
        print("\n--- L‑BFGS smoothing ---")
        smoothed_seq = _lbfgs_smooth_opt(global_best_seq, n, sub_deadline)
        smoothed_val = _objective_surrogate(smoothed_seq)
        if smoothed_val < global_best_val:
            global_best_seq, global_best_val = smoothed_seq, smoothed_val
            print(f"    *** New global best (L‑BFGS) = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 5. Final additive sub‑gradient fine‑tune (tiny budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 25.0
        print("\n--- Final additive sub‑gradient polish ---")
        final_seq, final_val = _additive_gradient_opt(global_best_seq,
                                                      n,
                                                      short_deadline,
                                                      max_iters=200_000,
                                                      eta0=0.3)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"    *** New global best (additive) = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, "
              f"true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
