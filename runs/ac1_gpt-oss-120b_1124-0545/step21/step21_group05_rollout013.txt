def _smooth_obj(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # length n
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

A second‑generation optimiser for the “step‑function” challenge.

Strategy
--------
1.  Build a very small pool of analytically‑parameterised seeds.
    The dominant family is a one‑sided exponential distribution
    (a_i ∝ exp(‑rate·i)).  Its rate is tuned with a fast 1‑D
    search (coarse grid + golden‑section) to obtain the best
    surrogate value (2·n·max(conv(a,a))).

2.  A high‑precision smoothing of the max‑convolution is performed
    with SciPy’s L‑BFGS‑B on the log‑softmax parametrisation.
    The smoothing parameter β is increased geometrically; for each
    β the optimiser runs up to 5 000 iterations (or stops early
    when the time budget is exhausted).

3.  Finally a long exponentiated‑gradient (mirror‑descent) polishing
    run refines the solution on the exact non‑smoothed objective.

All steps respect a global 1000 s budget (a few seconds are left as a
safety margin).  The best sequence found is stored in the global
variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# optional SciPy import – used only for the L‑BFGS phase
try:                         # pragma: no cover
    import scipy.optimize as _opt
except Exception:           # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Global placeholder for the deadline – accessed by the L‑BFGS callback
# ----------------------------------------------------------------------
_global_deadline = None


# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the exact value when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector already normalised to sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """One‑sided exponential a_i ∝ exp(‑rate·i), i=0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _double_exp(n: int, rate: float) -> np.ndarray:
    """
    Symmetric sum of two one‑sided exponentials:
        a_i ∝ exp(-rate·i) + exp(-rate·(n‑1‑i)).
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# 1‑D rate optimisation for the exponential family
# ----------------------------------------------------------------------
def _best_exponential_rate(n: int, deadline: float) -> float:
    """
    Returns the rate (positive float) giving the smallest surrogate value
    for the exponential family.  A coarse log‑grid (200 points) is scanned,
    then a golden‑section refinement (≈30 iterations) is performed around the
    best candidate.
    """
    # coarse search – log‑space from 1e‑5 to 0.01 (covers the region of interest)
    rates = np.logspace(-5, -2, 200)          # 1e‑5 … 1e‑2
    best_rate = rates[0]
    best_val = np.inf
    for r in rates:
        if time.time() > deadline:
            break
        seq = _exp_decay(n, r)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_rate = r

    # narrow interval around the best candidate (factor 2 on each side)
    lo = max(best_rate * 0.3, 1e-7)
    hi = best_rate * 3.0
    phi = (math.sqrt(5) - 1) / 2  # 0.618...

    for _ in range(30):
        if time.time() > deadline:
            break
        c = hi - phi * (hi - lo)
        d = lo + phi * (hi - lo)
        seq_c = _exp_decay(n, c)
        val_c = _objective_surrogate(seq_c)
        seq_d = _exp_decay(n, d)
        val_d = _objective_surrogate(seq_d)

        if val_c < val_d:
            hi = d
        else:
            lo = c

    # final selection – take the middle of the final interval
    final_rate = (lo + hi) / 2.0
    return final_rate


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient at the active index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed max‑convolution – L‑BFGS optimiser
# ----------------------------------------------------------------------
def _smooth_obj(v, n, beta):
    """
    Smoothed objective and its gradient for a given β.
    Returns (obj, grad_v) where v are the log‑softmax parameters.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth approximation of max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt the underlying probabilities a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]           # slice the central n entries

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                    # chain rule via log‑softmax
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5_000):
    """Iterative L‑BFGS optimisation of the smoothed max‑convolution."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    # callback to abort when the global budget expires
    def _callback(_):
        if time.time() > deadline:
            raise KeyboardInterrupt()

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(fun=_smooth_obj,
                                x0=v,
                                args=(n, float(beta)),
                                method='L-BFGS-B',
                                jac=True,
                                callback=_callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine builds a few analytic seeds (exponential, double‑exponential,
    uniform), fine‑tunes the exponential rate, refines the best seed with
    L‑BFGS smoothing and a final exponentiated‑gradient polishing run.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0
    _global_deadline = deadline

    # ------------------------------------------------------------------
    # Problem dimension – modest but expressive
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Generate analytically‑parameterised seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) uniform (baseline)
    seeds.append(_uniform(n))

    # b) exponential – rate tuned by a cheap 1‑D search
    best_rate = _best_exponential_rate(n, deadline)
    print(f"[Rate search] best exponential rate ≈ {best_rate:.6e}")
    seeds.append(_exp_decay(n, best_rate))

    # c) double‑exponential (symmetric) with the same rate – often improves flatness
    seeds.append(_double_exp(n, best_rate))

    # d) a few random Dirichlet draws (to diversify the landscape)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of the seeds – keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
        print(f"  seed surrogate = {val:.6f}")

    print(f"\n[Init] best surrogate from seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS smoothing on the best seed (if SciPy is available)
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        seq_smooth = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                            max_iter_per_beta=5_000)
        val_smooth = _objective_surrogate(seq_smooth)
        print(f"[L‑BFGS] smoothed surrogate = {val_smooth:.6f}")
        if val_smooth < best_val:
            best_seq, best_val = seq_smooth, val_smooth

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[EG polishing] starting long mirror‑descent run")
        seq_final, val_final = _exp_gradient_opt(best_seq, n, deadline,
                                                 max_iters=2_500_000,
                                                 eta0=6.5)
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
