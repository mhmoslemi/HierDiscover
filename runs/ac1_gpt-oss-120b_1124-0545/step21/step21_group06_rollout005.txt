```python
def _grad_max_conv(a, active_indices):
    n = len(a)
    grad = np.zeros_like(a)
    for k in active_indices:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev = k - idx
        grad[idx] += a[rev]
    # gradient of max(conv) w.r.t a is 2 * grad / |active|
    if len(active_indices) > 0:
        grad = 2.0 * grad / len(active_indices)
    return grad
```

```python
def _pairwise_swap_opt(a, n, deadline, rng, max_iter=200000, init_eps=0.01, final_eps=0.0001):
    best = a.copy()
    best_val = _objective_surrogate(a)
    cur = a.copy()
    cur_val = best_val
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        eps = init_eps * (1 - (it-1)/(max_iter-1)) + final_eps * ((it-1)/(max_iter-1))
        # choose source i with mass > eps
        i = rng.choice(n, p=cur)  # probability proportional to weight (so likely bigger)
        # choose target j uniformly among others
        j = rng.integers(0, n-2)
        if j >= i:
            j += 1
        # delta as random fraction of cur[i] * eps factor?
        delta = eps * cur[i] * rng.random()
        if delta <= 0 or cur[i] < delta:
            continue
        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta
        # ensure non-negative
        if ns[i] < 0:
            continue
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val
        # optionally accept worse with small probability?
    return best, best_val
```

```python
def _proj_simplex(v):
    # projection onto simplex (sum=1, v>=0)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simulated_annealing(start_seq, n, deadline, rng, max_attempts=200_000, start_sigma=0.005, T_start=0.02, T_end=1e-7):
    # same as before but compute objective each time
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0:
            attempts += 1
            continue
        ns /= s
        val = _objective(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val
        attempts += 1
    return best, best_val

def _pairwise_swap_opt(start_seq, n, deadline, rng, max_iters=200_000, eps_start=0.01, eps_end=0.0001):
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        eps = eps_start * (1 - (it-1)/(max_iters-1)) + eps_end * ((it-1)/(max_iters-1))
        # Choose source index proportional to its weight (so larger values more likely)
        i = rng.choice(n, p=a)
        # Choose a different target uniformly
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # delta is random fraction up to eps of source weight
        delta = eps * a[i] * rng.random()
        if delta <= 0 or a[i] < delta:
            continue
        ns = a.copy()
        ns[i] -= delta
        ns[j] += delta
        # ensure non-neg
        if ns[i] < 0:
            continue
        val = _objective(ns)
        if val < cur_val:
            a = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val
    return best, best_val

def _generate_seeds(n, rng, prev_arr=None):
    seeds = []
    seeds.append(np.full(n, 1.0 / n))
    # triangular
    i = np.arange(n)
    c = (n-1)/2.0
    tri = 1.0 - np.abs(i - c) / c
    tri = np.maximum(tri, 0)
    seeds.append(tri/tri.sum())
    # sine window
    sine = np.sin(np.pi*(i+1)/(n+1))
    seeds.append(sine/ sine.sum())
    # Hann window (cosine)
    hann = 0.5 - 0.5*np.cos(2*np.pi*i/(n-1))
    seeds.append(hann/ hann.sum())
    # Gaussian
    for sigma_frac in (0.12, 0.20, 0.30):
        sigma = sigma_frac*n
        mu = (n-1)/2.0
        gauss = np.exp(-((i - mu)**2)/(2*sigma**2))
        seeds.append(gauss/ gauss.sum())
    # exponential left/right
    for rate in np.logspace(-4, -0.2, 20):
        exp_l = np.exp(-rate*i)
        exp_r = np.exp(-rate*(n-1-i))
        seeds.append(exp_l/exp_l.sum())
        seeds.append(exp_r/exp_r.sum())
    # power law
    for p in np.linspace(0.0, 5.0, 12):
        idx = np.arange(1, n+1)
        pw = 1.0 / (idx**p)
        seeds.append(pw/pw.sum())
    # beta shapes
    alphas = np.linspace(0.2, 6.0, 9)
    betas = np.linspace(0.2, 6.0, 9)
    for a in alphas:
        for b in betas:
            i1 = np.arange(1, n+1)
            beta = (i1**(a-1))*((n-i1+1)**(b-1))
            seeds.append(beta / beta.sum())
    # Dirichlet draws
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            d = rng.dirichlet(np.full(n, conc))
            seeds.append(d)
    # mixtures with uniform
    for rate in np.logspace(-4,-0.3,10):
        exp_l = np.exp(-rate*i)
        exp_r = np.exp(-rate*(n-1-i))
        for alpha in (0.2,0.5,0.8):
            mix = (1-alpha)*np.full(n,1.0/n) + alpha*exp_l/exp_l.sum()
            seeds.append(mix/mix.sum())
            mix = (1-alpha)*np.full(n,1.0/n) + alpha*exp_r/exp_r.sum()
            seeds.append(mix/mix.sum())
    # if prev exists
    if prev_arr is not None and prev_arr.shape[0]==n:
        seeds.append(prev_arr.copy())
        for _ in range(5):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0)
            s=pert.sum()
            if s>0:
                seeds.append(pert/ s)
    # ensure unique (optional but not necessary)
    # Convert all to float (already)
    return seeds

def _lbfgs_smooth_opt(start_vec, n, deadline, beta_schedule=None, max_iter_per_beta=1500):
    if _opt is None:
        return start_vec
    if beta_schedule is None:
        beta_schedule = [1.0,5.0,20.0,80.0,200.0,500.0,1_000.0,5_000.0,20_000.0,100_000.0]
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time()>deadline:
            break
        beta_val=float(beta)
        def fun(v_loc):
            a = _softmax(v_loc)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0*n*(shift + np.log(sum_exp)/beta_val)
            # gradient w.r.t a
            w = exp_beta / sum_exp
            # gradient w.r.t a is 2 * conv of reversed a with w
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]  # slice of length n
            # project onto simplex via log-softmax chain rule
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(x):
            if time.time()>deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta, 'ftol':1e-12, 'disp':False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break
    return _softmax(v)

* a bespoke pair‑wise “swap” hill‑climber.

The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list.  All helpers are
top‑level functions – no closures or lambdas are used – and the code
conforms to the 1000 s per‑run budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective and gradient utilities
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv(seq,seq))/sum(seq)²."""
    n = len(seq)
    s = seq.sum()
    if s <= 0.0:
        return float('inf')
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max() / (s * s)


def _gradient_max_conv(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    For every active index k we add the vector (a_{k-i}) to the gradient.
    The returned gradient already contains the factor 2 and the averaging
    over the active set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += a[rev]
    if active.size > 0:
        g = 2.0 * g / active.size
    return g


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 5.0):
    """
    Exponentiated‑gradient (mirror descent) on the exact objective,
    using *all* active indices (not a single one).  Operates in log‑space
    to keep the iterates non‑negative and ℓ₁‑normalised.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the *objective* (2·n·max(conv))
        grad = _gradient_max_conv(a, active) * (2.0 * n)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _proj_gradient_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000,
                       eta0: float = 0.4):
    """
    Projected (Euclidean) gradient descent on the exact objective.
    After each step the iterate is projected onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = _gradient_max_conv(a, active) * (2.0 * n)  # gradient of objective
        eta = eta0 / math.sqrt(it)

        a = a - eta * grad
        a = _proj_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_opt(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule=None,
                      max_iter_per_beta: int = 1500):
    """
    Continuation L‑BFGS on a log‑sum‑exp smoothing of the max‑convolution.
    Operates in log‑softmax space so that non‑negativity and the simplex
    constraint are automatically satisfied.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 5_000.0, 20_000.0,
                         100_000.0, 500_000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)       # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_loc):
            a = _softmax(v_loc)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple geometric‑cooling SA that perturbs the whole vector with
    Gaussian noise, re‑normalises and occasionally accepts worsened
    candidates.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling from T_start down to T_end
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        attempts += 1

    return best_seq, best_val


def _pairwise_swap_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_iters: int = 200_000,
                       eps_start: float = 0.01,
                       eps_end: float = 0.0001):
    """
    Very cheap hill‑climber that repeatedly moves a tiny amount of mass
    from a source index to a different target index.  The magnitude of
    the move decays linearly from eps_start to eps_end.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        eps = eps_start * (1 - (it - 1) / (max_iters - 1)) + eps_end * ((it - 1) / (max_iters - 1))

        # source proportional to its weight (so larger entries move more often)
        i = rng.choice(n, p=a)
        # uniform random distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # random fraction of the source mass, limited by eps
        delta = eps * a[i] * rng.random()
        if delta <= 0.0 or a[i] < delta:
            continue

        ns = a.copy()
        ns[i] -= delta
        ns[j] += delta

        # sanity check (should never be negative)
        if ns[i] < 0.0:
            continue

        val = _objective(ns)
        if val < cur_val:
            a = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (deterministic families + random draws)
# ----------------------------------------------------------------------
def _generate_seeds(n: int,
                    rng: np.random.Generator,
                    prev_arr: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # 1) Simple analytic families
    i = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    seeds.append(uniform)

    # triangular
    c = (n - 1) / 2.0
    tri = 1.0 - np.abs(i - c) / c
    tri = np.maximum(tri, 0.0)
    seeds.append(tri / tri.sum())

    # sine window
    sine = np.sin(np.pi * (i + 1) / (n + 1))
    seeds.append(sine / sine.sum())

    # Hann window (cosine bump)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    seeds.append(hann / hann.sum())

    # Gaussian bumps
    for sigma_frac in (0.12, 0.20, 0.30):
        sigma = sigma_frac * n
        mu = (n - 1) / 2.0
        gauss = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
        seeds.append(gauss / gauss.sum())

    # Exponential decays (both sides)
    rates = np.logspace(-4, -0.2, 20)
    for r in rates:
        exp_l = np.exp(-r * i)
        exp_r = np.exp(-r * (n - 1 - i))
        seeds.append(exp_l / exp_l.sum())
        seeds.append(exp_r / exp_r.sum())

    # Power‑law tails
    for p in np.linspace(0.0, 5.0, 12):
        idx = np.arange(1, n + 1, dtype=np.float64)
        pw = 1.0 / (idx ** p)
        seeds.append(pw / pw.sum())

    # Beta‑shapes (discrete analogue of Beta(α,β))
    alphas = np.linspace(0.2, 6.0, 9)
    betas = np.linspace(0.2, 6.0, 9)
    for a in alphas:
        for b in betas:
            ib = np.arange(1, n + 1, dtype=np.float64)
            beta = (ib ** (a - 1.0)) * ((n - ib + 1) ** (b - 1.0))
            if beta.sum() > 0:
                seeds.append(beta / beta.sum())

    # Random Dirichlet draws (exploring both dense and sparse regimes)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            d = rng.dirichlet(np.full(n, conc))
            seeds.append(d)

    # Mixtures of uniform with an exponential bump
    for r in np.logspace(-4, -0.3, 10):
        exp_l = np.exp(-r * i)
        exp_r = np.exp(-r * (n - 1 - i))
        for α in (0.2, 0.5, 0.8):
            mix_l = (1 - α) * uniform + α * (exp_l / exp_l.sum())
            mix_r = (1 - α) * uniform + α * (exp_r / exp_r.sum())
            seeds.append(mix_l / mix_l.sum())
            seeds.append(mix_r / mix_r.sum())

    # Perturb previous best (if any) – yields a local neighbourhood
    if prev_arr is not None and prev_arr.shape[0] == n:
        seeds.append(prev_arr.copy())
        for _ in range(5):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The three‑phase strategy described at the top
    of the file is used.  The best sequence discovered is stored in the
    global variable ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Retrieve previously found vector (if any) and decide on a problem size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0 and prev_arr.sum() > 0.0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    n = 4000 if prev_arr is None else int(prev_arr.shape[0])
    print(f"\n=== Optimisation start (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Generate a broad set of seeds and evaluate them quickly
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng, prev_arr)
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after coarse scan:")
    for idx, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{idx + 1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2. Multi‑start local refinement
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] global best = {global_best_val:.6f}")

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a modest slice of the remaining budget to this seed
        remaining = deadline - time.time()
        per_seed = max(5.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + per_seed

        # --------------------------------------------------------------
        # (a) Projected gradient descent
        # --------------------------------------------------------------
        pg_seq, pg_val = _proj_gradient_opt(seed_vec, n, sub_deadline,
                                            max_iters=150_000,
                                            eta0=0.4)

        # --------------------------------------------------------------
        # (b) Exponentiated gradient (full active set)
        # --------------------------------------------------------------
        if time.time() < sub_deadline:
            eg_seq, eg_val = _exp_gradient_opt_full(pg_seq, n,
                                                    sub_deadline,
                                                    max_iters=200_000,
                                                    eta0=6.0)
        else:
            eg_seq, eg_val = pg_seq, pg_val

        # --------------------------------------------------------------
        # (c) Optional smooth L‑BFGS refinement (if SciPy is available)
        # --------------------------------------------------------------
        if _opt is not None and time.time() < sub_deadline:
            try:
                lb_seq = _lbfgs_smooth_opt(eg_seq, n, sub_deadline)
                lb_val = _objective(lb_seq)
            except Exception:
                lb_seq, lb_val = eg_seq, eg_val
        else:
            lb_seq, lb_val = eg_seq, eg_val

        # --------------------------------------------------------------
        # (d) Simulated annealing
        # --------------------------------------------------------------
        if time.time() < sub_deadline:
            sa_seq, sa_val = _simulated_annealing(lb_seq, n, sub_deadline, rng,
                                                  max_attempts=120_000,
                                                  start_sigma=0.004,
                                                  T_start=0.018,
                                                  T_end=5e-8)
        else:
            sa_seq, sa_val = lb_seq, lb_val

        # --------------------------------------------------------------
        # (e) Pair‑wise swap hill‑climber
        # --------------------------------------------------------------
        if time.time() < sub_deadline:
            sw_seq, sw_val = _pairwise_swap_opt(sa_seq, n, sub_deadline, rng,
                                                max_iters=120_000,
                                                eps_start=0.008,
                                                eps_end=5e-5)
        else:
            sw_seq, sw_val = sa_seq, sa_val

        # Choose the best among all intermediate results
        candidates = [
            (seed_vec, _objective(seed_vec)),
            (pg_seq, pg_val),
            (eg_seq, eg_val),
            (lb_seq, lb_val),
            (sa_seq, sa_val),
            (sw_seq, sw_val)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si + 1}/{top_k}] local best = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polishing – allocate the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing: long projected gradient ---")
        final_seq, final_val = _proj_gradient_opt(global_best_seq, n,
                                                  deadline,
                                                  max_iters=400_000,
                                                  eta0=0.3)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Polish PGD] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing: long exponentiated gradient ---")
        final_seq, final_val = _exp_gradient_opt_full(global_best_seq, n,
                                                      deadline,
                                                      max_iters=600_000,
                                                      eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Polish EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing: long simulated annealing ---")
        final_seq, final_val = _simulated_annealing(global_best_seq, n,
                                                    deadline,
                                                    rng,
                                                    max_attempts=300_000,
                                                    start_sigma=0.001,
                                                    T_start=4e-3,
                                                    T_end=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Polish SA] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing: swap hill‑climber ---")
        final_seq, final_val = _pairwise_swap_opt(global_best_seq, n,
                                                  deadline,
                                                  rng,
                                                  max_iters=250_000,
                                                  eps_start=0.005,
                                                  eps_end=1e-5)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Polish Swap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Official evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
