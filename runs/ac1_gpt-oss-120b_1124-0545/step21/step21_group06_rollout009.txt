def _exp_gradient_full(start_seq: np.ndarray, n: int, deadline: float, max_iters: int = 1000000, eta0: float = 5.0, epsilon_factor: float = 1e-12):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active mask
        eps = max_val * epsilon_factor
        active = (conv >= max_val - eps).astype(np.float64)  # mask 0/1
        # subgradient via convolution
        a_rev = a[::-1]
        # compute full convolution of active mask and a_rev
        grad_full = _conv_fft(active, a_rev)
        # extract slice n-1:n-1+n
        grad = 2.0 * grad_full[n-1:n-1+n]
        # learning rate
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
Improved optimiser for the step‑function bound.

Main ideas
----------
* Parameter‑free exponentiated‑gradient (mirror‑descent) on the exact
  max‑convolution, **using all active constraints**.  The sub‑gradient is
  obtained via a *single* FFT (convolution of the active‑mask with the
  reversed current iterate), which makes each iteration cheap (≈2 FFTs).

* A simple momentum term is added to the mirror‑gradient to stabilise the
  updates and allow larger effective step‑sizes.

* Several analytic families (Beta, exponential, power‑law, Dirichlet)
  are used for seeding; the best few seeds are refined with the EG
  optimiser for as long as the time budget allows.

* The whole search runs inside the 1000 s deadline, constantly updating
  a global best sequence.  At the end the best sequence is evaluated
  with the provided ``evaluate_sequence`` and stored in the global variable
  ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – only used for a final L‑BFGS polishing (fallback).
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (full active set, momentum)
# ----------------------------------------------------------------------
def _exp_gradient_full(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 1_000_000,
                       eta0: float = 8.0,
                       eps_factor: float = 1e-9,
                       momentum: float = 0.9):
    """
    Mirror‑descent on the exact max‑convolution.
    Uses all (almost) active constraints, computed via a convolution
    of the active‑mask with the reversed iterate.
    Returns the best sequence encountered and its surrogate value.
    """
    a = np.copy(start_seq)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (mirror‑descent)
    log_a = np.log(a + 1e-30)

    # momentum accumulator
    g_avg = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution a*a and active set ---
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * eps_factor
        active_mask = (conv >= max_val - eps).astype(np.float64)  # 0/1

        # --- sub‑gradient via convolution (active_mask * a_rev) ---
        a_rev = a[::-1]
        grad_full = _conv_fft(active_mask, a_rev)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]            # size n

        # --- momentum update ---
        if it == 1:
            g_avg = grad
        else:
            g_avg = momentum * g_avg + (1.0 - momentum) * grad

        # --- learning‑rate schedule ---
        eta = eta0 / math.sqrt(it)

        # --- mirror‑descent step ---
        log_a -= eta * g_avg
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional smooth (log‑sum‑exp) L‑BFGS refinement.
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iter: int = 8000):
    """
    Refine a sequence using L‑BFGS on a smooth approximation of the max.
    Only used if SciPy is available.
    """
    if _opt is None:
        return start_seq

    v = np.log(start_seq + 1e-30)

    def fun(v_local):
        a = _softmax(v_local)
        conv = _conv_fft(a, a)
        m = np.max(conv)
        # smooth max via log‑sum‑exp
        beta = 200_000.0
        exp_term = np.exp(beta * (conv - m))
        smooth_max = m + np.log(exp_term.sum()) / beta
        obj = 2.0 * n * smooth_max

        # gradient w.r.t. a (smooth max)
        w = exp_term / exp_term.sum()
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule for log‑softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v

    try:
        res = _opt.minimize(fun=fun,
                            x0=v,
                            method='L-BFGS-B',
                            jac=True,
                            bounds=[(None, None)] * len(v),
                            options={'maxiter': max_iter,
                                     'ftol': 1e-12,
                                     'disp': False})
        return _softmax(res.x)
    except Exception:  # pragma: no cover
        return start_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse previous solution if present, otherwise 4000.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size > 0 and prev_arr.sum() > 0.0:
                prev_arr = prev_arr / prev_arr.sum()
                n = prev_arr.shape[0]
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    print(f"\n=== Optimisation started (n={n}) ===")

    # --------------------------------------------------------------
    # 1. Build a large pool of seeds (analytic families + random draws)
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_beta_shape(n, 0.5, 0.5))                # arcsine‑like
    seeds.append(_beta_shape(n, 0.8, 0.8))
    seeds.append(_beta_shape(n, 1.2, 1.2))
    seeds.append(_exp_decay_left(n, 0.001))
    seeds.append(_exp_decay_left(n, 0.01))
    seeds.append(_exp_decay_right(n, 0.001))
    seeds.append(_exp_decay_right(n, 0.01))
    seeds.append(_power_law(n, 0.5))
    seeds.append(_power_law(n, 1.0))

    # finer beta grid around the promising region (0.2 … 2)
    alphas = np.linspace(0.2, 2.0, 13)
    betas = np.linspace(0.2, 2.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (wide range of concentrations)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best (if any) – encourages local exploration
    if prev_arr is not None:
        seeds.append(prev_arr.copy())
        for _ in range(6):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: surrogate = {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best from the best seed
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine each top seed with the full‑active EG optimiser
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget to this seed
        remaining = deadline - time.time()
        # ensure at least a couple of seconds per seed
        sub_deadline = time.time() + max(3.0, remaining * 0.12 / max(1, (top_k - si)))

        seq_opt, val_opt = _exp_gradient_full(seed_vec.copy(),
                                               n,
                                               sub_deadline,
                                               max_iters=800_000,
                                               eta0=9.0,
                                               eps_factor=1e-9,
                                               momentum=0.85)

        # Optional L‑BFGS polishing (only if time permits)
        if time.time() < sub_deadline and _opt is not None:
            seq_opt = _lbfgs_refine(seq_opt, n, sub_deadline, max_iter=4000)
            val_opt = _objective_surrogate(seq_opt)

        print(f"[Seed {si+1}/{top_k}] refined surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final intensive polishing on the current global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive EG polishing (full active set) ---")
        final_seq, final_val = _exp_gradient_full(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=2_000_000,
                                                  eta0=7.0,
                                                  eps_factor=1e-9,
                                                  momentum=0.9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline and _opt is not None:
        print("\n--- Final L‑BFGS polishing (smooth max) ---")
        final_seq = _lbfgs_refine(global_best_seq, n, deadline, max_iter=12000)
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
