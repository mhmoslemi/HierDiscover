def _adam_optimize(init_seq, n, deadline, betas, iters_per_beta, lr):
    # init log parameters
    v = np.log(init_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = init_seq.copy()
    best_val = _objective(init_seq)

    for beta in betas:
        for step in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(v)
            # compute objective and gradient
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # obj not needed for update; but we compute grad_v
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1**t)
            v_hat_hat = v_hat / (1 - beta2**t)
            v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)

            if step % 100 == 0:
                cur_seq = _softmax(v)
                cur_val = _objective(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()
    # final best
    return best_seq, best_val

"""
Random‑restart Adam optimizer for the step‑function “max‑convolution” problem.

The pipeline:
  1. Build a few diverse initial probability vectors (previous best,
     exponential decay, arcsine‑like, uniform, Dirichlet draws).
  2. For each start run a smooth‑max Adam optimisation on the exact
     convolution (log‑sum‑exp surrogate).  A schedule of increasing β
     makes the surrogate converge to the true max.
  3. Keep the best sequence (exact objective 2·n·max(conv)/(∑a)²).
  4. (Optional) a few iterations of the sub‑gradient mirror‑descent
     (exponentiated‑gradient) polish the winner.
  5. Return the best sequence; it is stored as ``height_sequence_1`` for
     the next invocation.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the same quantity returned by evaluate_sequence)
# ----------------------------------------------------------------------
def _exact_objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  Works for any non‑negative seq."""
    n = len(seq)
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Softmax (log‑space parametrisation)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient w.r.t. log‑softmax parameters
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad_v) where
        obj  = 2·n·(shift + log Σ exp(beta·(conv‑shift))/beta)
        grad_v = ∂obj/∂v  (v are the log‑softmax parameters)
    """
    a = _softmax(v)                     # probability vector, Σa = 1
    conv = _conv_fft(a, a)              # length 2·n‑1
    shift = np.max(conv)                # for numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)

    # smooth approximation of the max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see analysis in the description)
    w = exp_beta / sum_exp               # soft weights over the convolution entries
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: dobj/dv = (da/dv)·grad_a  ;  da/dv = diag(a) - a aᵀ
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(init_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   betas: list[float],
                   iters_per_beta: int,
                   lr: float = 0.05):
    """
    Adam optimiser in log‑softmax space on the smooth max surrogate.
    Returns the best exact (non‑smooth) sequence seen.
    """
    # initialise log‑parameters (add tiny epsilon for safety)
    v = np.log(init_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = init_seq.copy()
    best_val = _exact_objective(best_seq)

    for beta in betas:
        for step in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val

            t += 1
            obj, grad_v = _smooth_obj_and_grad(v, n, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat_hat = v_hat / (1.0 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)

            # occasional exact objective check (every 200 steps)
            if step % 200 == 0:
                cur_seq = _softmax(v)
                cur_val = _exact_objective(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

    # one final exact check
    cur_seq = _softmax(v)
    cur_val = _exact_objective(cur_seq)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated‑gradient) polishing – works on the true max.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient on the active entry).
    The sequence must sum to 1.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _exact_objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _exact_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Strategy:
      * several diverse starts (previous best, exponential, arcsine, uniform,
        random Dirichlet vectors);
      * Adam optimisation on a smooth max surrogate with a β‑schedule;
      * final mirror‑descent polishing;
    Returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # tiny safety margin

    # --------------------------------------------------------------
    # problem size – keep the same length as the historic run
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # load previous solution if it exists
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if isinstance(prev_seq, list) and len(prev_seq) == n:
        seq_prev = np.asarray(prev_seq, dtype=np.float64)
        seq_prev = np.maximum(seq_prev, 0.0)
        total = seq_prev.sum()
        if total > 0:
            seq_prev = seq_prev / total
            best_seq = seq_prev.copy()
            best_val = _exact_objective(best_seq)
            print(f"[Init] loaded previous best ≈ {best_val:.6f}")
        else:
            best_seq = None
            best_val = float("inf")
    else:
        best_seq = None
        best_val = float("inf")

    # --------------------------------------------------------------
    # construct a pool of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    if best_seq is not None:
        init_candidates.append(best_seq)

    i = np.arange(n, dtype=np.float64)

    # exponential decay
    rate = 0.003
    exp_vec = np.exp(-rate * i)
    init_candidates.append(exp_vec / exp_vec.sum())

    # arcsine‑like (large mass near the ends)
    eps = 1e-12
    arc = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + eps)
    init_candidates.append(arc / arc.sum())

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with different concentration parameters
    for shape in (0.3, 0.7, 1.5, 3.0):
        rand = rng.gamma(shape=shape, scale=1.0, size=n)
        init_candidates.append(rand / rand.sum())

    # --------------------------------------------------------------
    # optimisation loop over the candidates
    # --------------------------------------------------------------
    for idx, init_seq in enumerate(init_candidates):
        if time.time() > deadline - 10.0:
            break   # leave a little slack

        print(f"[Candidate {idx + 1}/{len(init_candidates)}] start")

        # ---- Adam on smooth max ----
        betas = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0
        ]
        # give each candidate roughly the same budget; 20 000 steps total
        iters_per_beta = max(1, int(20000 // len(betas)))
        seq_opt, val_opt = _adam_optimize(
            init_seq, n, deadline, betas, iters_per_beta, lr=0.05
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"  [Adam] new best ≈ {best_val:.6f}")

        # ---- optional polishing with exponentiated‑gradient (if time) ----
        if time.time() < deadline - 30.0:
            seq_pol, val_pol = _exp_gradient_opt(
                best_seq, n, deadline, max_iters=600_000, eta0=5.0
            )
            if val_pol < best_val:
                best_val = val_pol
                best_seq = seq_pol.copy()
                print(f"  [EG] new best ≈ {best_val:.6f}")

    # --------------------------------------------------------------
    # final safety check – if nothing succeeded fall back to uniform
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _exact_objective(best_seq)

    print(f"[Result] best surrogate (exact) = {best_val:.6f}")

    # --------------------------------------------------------------
    # evaluate with the provided secure function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[True] evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised:", exc)

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
