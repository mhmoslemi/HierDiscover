"""
step‑function optimizer – diversified search

The routine builds a very large pool of analytically‑generated seeds
(including uniform, triangular, a wide family of window functions,
Beta‑shapes, exponential/Dirichlet draws and random mixtures).  All
seeds are ℓ₁‑normalised and ranked by the exact surrogate
2·n·max(conv(a,a)).  The best *top_k* seeds are refined independently
by two complementary mirror‑descent style optimisers:

    • Adam‑accelerated exponentiated‑gradient on the exact non‑smooth
      max‑convolution (single‑active sub‑gradient).  This respects the
      simplex constraints via the log‑softmax parametrisation.

    • A short plain EG polish (single‑active) for extra robustness.

The locally best candidates are collected, the global best is kept,
and finally a long simulated‑annealing run (hundreds of thousands of
perturbations) explores the neighbourhood of the current optimum.

All work respects the global “budget_s” time limit: the algorithm
continuously checks the deadline and gracefully terminates early.
The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window (fallbacks to uniform if SciPy i0 unavailable)."""
    try:
        from scipy.special import i0 as _i0
    except Exception:
        return _uniform(n)

    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dirichlet_draw(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt_simple(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 5.0,
                             single_active: bool = True):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if single_active:
            k = int(conv.argmax())
            active = np.array([k], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]

        if active.size:
            g = 2.0 * g / active.size

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """
    Adam‑accelerated exponentiated‑gradient on the exact max‑convolution.
    Works in log‑softmax space, guaranteeing a∈Δⁿ.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient for the single active index k
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        rev_idx = k - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g[mask] = a[rev_idx[mask]]
        g = 2.0 * g

        # Adam updates on log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the global time budget, stores the best found sequence
    in the global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(123456)
    start_time = time.time()
    # keep a safety margin (2 s) for final evaluation / printing
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (chosen as a compromise between resolution and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (provides continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr.copy()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Generate a massive pool of diverse seeds
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_cosine_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Kaiser window – sweep a wide range
    for beta in np.linspace(0.0, 14.0, 15):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian shapes – many sigma values
    for sigma_frac in np.linspace(0.04, 0.60, 20):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays – log‑spaced rates
    for rate in np.logspace(-5, -0.1, 30):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 8.0, 25):
        seeds.append(_power(n, p))

    # Beta‑shapes – coarse grid (kept for reference)
    alphas_grid = np.linspace(0.01, 12.0, 12)
    betas_grid = np.linspace(0.01, 12.0, 12)
    for a in alphas_grid:
        for b in betas_grid:
            seeds.append(_beta_shape(n, a, b))

    # Random Beta‑shapes – dense sampling in log‑space
    for _ in range(500):
        a = 10 ** rng.uniform(-2.5, 1.5)   # roughly 0.003 – 30
        b = 10 ** rng.uniform(-2.5, 1.5)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – various concentration parameters
    for conc in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(4):
            seeds.append(_dirichlet_draw(n, conc, rng))

    # Mixtures of uniform with an exponential tail
    for rate in np.logspace(-5, -0.2, 12):
        exp_seq = _exp_decay(n, rate)
        for mix in [0.05, 0.10, 0.20, 0.35]:
            seeds.append(_mix_shapes(_uniform(n), exp_seq, mix))

    # Small perturbations of the previous best (if any)
    if prev_arr is not None:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(8):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best from the very first top seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        # aim to spend roughly equal time on the remaining seeds,
        # but keep some slack for the final polishing.
        sub_deadline = time.time() + remaining * 0.75 / max(1, (top_k - si))

        # ---- (a) Adam‑accelerated EG ----------
        seq_opt, val_opt = _adam_opt(seed_vec.copy(),
                                     n,
                                     sub_deadline,
                                     max_iters=400_000,
                                     lr=0.15)

        # ---- (b) brief plain EG polish (single‑active) -----
        if time.time() < sub_deadline:
            seq_opt, val_opt = _exp_gradient_opt_simple(seq_opt,
                                                        n,
                                                        sub_deadline,
                                                        max_iters=200_000,
                                                        eta0=7.0,
                                                        single_active=True)

        # keep track of the best seen so far
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {si+1}/{top_k}] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run polishing with simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Simulated Annealing polishing ---")
        final_seq, final_val = _simulated_annealing(global_best_seq,
                                                    n,
                                                    deadline,
                                                    rng,
                                                    max_attempts=500_000,
                                                    start_sigma=0.0015,
                                                    T_start=0.015,
                                                    T_end=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final SA] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
