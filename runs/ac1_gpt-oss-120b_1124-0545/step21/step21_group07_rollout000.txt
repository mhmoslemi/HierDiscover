"""
step‑function optimizer – ADAM + SA (large n)

This implementation searches for a non‑negative sequence that minimises
``evaluate_sequence``.  It uses a larger problem size (n = 8192) than the
reference solution and a simple Adam optimiser in the log‑softmax (mirror
descent) space, followed by a short simulated‑annealing polish.  The
algorithm proceeds as follows:

1.  Build a diverse set of analytically generated seed vectors (uniform,
    triangular, window families, exponential/power‑law, Dirichlet draws,
    mixtures, etc.) and ℓ₁‑normalise them.
2.  Rank the seeds by the exact surrogate   2·n·max(conv(a,a))   and keep
    the best few.
3.  For each top seed run Adam on the exact max‑convolution (single‑active
    sub‑gradient).  The optimiser works in log‑probability space,
    guaranteeing non‑negativity and unit sum.
4.  Optionally improve the result with a brief simulated‑annealing run.
5.  After all seeds have been processed, perform a final long Adam polish
    on the current best sequence and a final SA polish.
6.  Evaluate the final sequence with the provided ``evaluate_sequence`` and
    store it in the global variable ``height_sequence_1``.

All helper functions are defined at top level; no closures or lambda
expressions are used.  The routine respects the overall time budget
(`budget_s`, default 1000 s) and returns a Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution, ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq, seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window; falls back to uniform if SciPy i0 is unavailable."""
    try:
        from scipy.special import i0 as _i0  # type: ignore
    except Exception:  # pragma: no cover
        _i0 = None
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (single‑active sub‑gradient)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-12,
              single_active: bool = True,
              tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in log‑softmax space.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)        # log‑probabilities
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        cur_val = 2.0 * n * max_val

        # keep best
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # active indices for sub‑gradient
        if single_active:
            active_idx = np.array([int(conv.argmax())])
        else:
            eps_active = max_val * tol
            active_idx = np.where(conv >= max_val - eps_active)[0]

        # sub‑gradient of the max (averaged over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active_idx:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active_idx.size:
            g = 2.0 * g / active_idx.size

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(t)  # simple decay
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing (gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.005,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA that perturbs the probability vector with Gaussian noise.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------

            prev_arr = None

    # Beta shapes (coarse grid)

            pert = np.maximum(pert, 0.0)
            s = pert.sum()

    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------

        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        # fallback to uniform if something went wrong
        best_vec = _uniform(n)
        best_val = _objective_surrogate(best_vec)
        globals()["height_sequence_1"] = best_vec.tolist()
        return best_vec.tolist()

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"\n--- Seeding (n={n}) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  Seed #{i+1}: surrogate = {v:.6f}")
    print(f"Keeping top {top_k} seeds for refinement.\n")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # give each seed a fraction of the remaining budget
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        # ---- Adam optimisation (single‑active) ----
        seq_adam, val_adam = _adam_opt(seed_vec.copy(),
                                       n,
                                       sub_deadline,
                                       max_iters=500_000,
                                       lr0=0.6,
                                       beta1=0.9,
                                       beta2=0.999,
                                       eps=1e-12,
                                       single_active=True)

        # ---- Short simulated annealing ----
        if time.time() < sub_deadline - 0.3:
            seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                  n,
                                                  sub_deadline - 0.2,
                                                  rng,
                                                  max_attempts=80_000,
                                                  start_sigma=0.0018,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # ---- Choose best of the candidates ----
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq.copy()
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_opt(global_best_seq.copy(),
                                         n,
                                         deadline,
                                         max_iters=800_000,
                                         lr0=0.4,
                                         beta1=0.9,
                                         beta2=0.999,
                                         eps=1e-12,
                                         single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief Simulated Annealing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq.copy(),
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.001,
                                                         T_start=0.004,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
