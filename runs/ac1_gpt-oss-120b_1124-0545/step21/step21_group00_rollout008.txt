"""
Hybrid optimiser for the step‑function problem.

The algorithm proceeds in four stages:

1. **Diverse seeds** – a small pool of normalised vectors (uniform,
   Dirichlet with several concentrations, sparse Dirichlet, the previous
   solution if any) is built and the best seed is kept.

2. **Exponentiated‑gradient (mirror descent)** – the current best seed is
   refined by a long run of mirror‑descent on the *exact* max‑convolution
   objective.  The update is performed in log‑softmax space with a
   diminishing stepsize η_t = η₀ / √t (η₀ = 4.0).  The loop stops as soon
   as the global time budget is exhausted.

3. **L‑BFGS polishing** – a smooth log‑sum‑exp surrogate of the max is
   optimised with SciPy's L‑BFGS‑B (if SciPy is available).  A very
   aggressive schedule of β‑values is used and the optimisation runs for
   as long as there is time left after stage 2.

4. **Stochastic greedy perturbations** – while a few seconds remain we
   repeatedly move a tiny amount of probability mass from a random index
   to another one; improvements are kept.  This cheap hill‑climber is
   able to exploit tiny local gains that the previous continuous methods
   may have missed.

All intermediate vectors are always normalised to unit ℓ₁‑norm, stay
non‑negative, and the best vector found is stored in the global variable
``height_sequence_1``.  The routine returns the final vector as a plain
list of floats, ready to be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution of `seq`."""
    return float(np.max(_conv_fft(seq, seq)))


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate used during the search:
       2·n·max(conv(seq,seq))   (seq is assumed to sum to 1)."""
    n = len(seq)
    return 2.0 * n * _conv_max(seq)


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a full‑length Dirichlet with total concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet on a random support of size `k`."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 5_000_000,
                      eta0: float = 4.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the true max‑convolution.
    The variables live in log‑softmax space; the stepsize is η₀/√t.
    Returns the best sequence seen and its objective value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # log‑softmax parameters
    n = len(a)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # compute convolution and the active shift
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # stepsize schedule
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update in log‑space
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS polishing on a smooth surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 30_000):
    """L‑BFGS‑B on the smooth log‑sum‑exp surrogate (variables = log‑softmax)."""
    try:
        import scipy.optimize as _opt
    except Exception:                               # pragma: no cover
        return start_vec.copy()

    n = len(start_vec)

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0, 10_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            mx = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = mx
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                      # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                  # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={
                    'maxiter': max_iter_per_beta,
                    'ftol': 1e-12,
                    'disp': False,
                })
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:                               # pragma: no cover
            break

    return _softmax(v)

            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Long mirror‑descent (exponentiated‑gradient) polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # We give the bulk of the remaining time to EG
        eg_deadline = time.time() + remaining * 0.70
        max_iters = 5_000_000                     # generous upper bound; the loop respects the deadline
        print("\n--- Exponentiated‑gradient phase ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           max_iters=max_iters,
                                           eta0=4.0)
        print(f"[EG] best objective = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  L‑BFGS polishing on the smooth surrogate (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # allocate the remaining time (≈30 % of the budget) to L‑BFGS
        lbfgs_deadline = time.time() + (deadline - time.time()) * 0.30
        print("\n--- L‑BFGS polishing ---")
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                           lbfgs_deadline,
                                           max_iter_per_beta=30_000)
        val_lbfgs = _objective(seq_lbfgs)
        print(f"[L‑BFGS] objective = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Cheap stochastic perturbations (hill‑climbing)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Stochastic perturbation phase ---")
        step_eps = 0.02                          # start with a reasonably sized move
        while time.time() < deadline - 2.0:
            # random move of a small mass ε from i to j
            eps = step_eps * (0.5 ** rng.random())   # randomise the magnitude a little
            i = rng.integers(0, n)
            if best_seq[i] < eps:
                continue
            j = rng.integers(0, n)
            cand = best_seq.copy()
            cand[i] -= eps
            cand[j] += eps
            # sanity check – non‑negativity is guaranteed by the eps test
            val_cand = _objective(cand)
            if val_cand < best_val:
                best_val, best_seq = val_cand, cand.copy()
                print(f"[perturb] improved objective = {best_val:.6f}")
                # if we find an improvement we reset the step size a bit
                step_eps = max(step_eps * 0.98, 0.001)
            # slowly decay the candidate step size
            if rng.random() < 0.01:
                step_eps = max(step_eps * 0.9, 0.001)

    # --------------------------------------------------------------
    # 6️⃣  Final official evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[Warning] evaluate_sequence raised:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
