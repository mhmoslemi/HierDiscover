"""
Improved optimiser for the “step‑function” challenge.

The idea is simple:
  1. Initialise from the best known solution (or an exponential decay).
  2. Refine the probability vector using a *smooth* approximation of the
     max‑convolution (log‑sum‑exp) and SciPy’s L‑BFGS‑B optimiser.
     We run a long schedule of increasing β values so that the smoothed
     objective approaches the true objective 2·n·max(conv(a,a)).
  3. If SciPy is not available, fall back to a plain exponentiated‑gradient
     (mirror‑descent) optimiser.
  4. Finally evaluate the candidate with the supplied ``evaluate_sequence``
     and keep the best one seen before the time budget runs out.

The optimiser works entirely in the probability simplex (the vector is
normalised to sum‑to‑one).  Scaling does not change the true objective,
so we can safely work with normalised vectors throughout the optimisation.
"""

from __future__ import annotations
import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used only for the L‑BFGS‑B polishing stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt   # type: ignore
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Numerically stable soft‑max (operates on log‑probabilities)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Return a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser for the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on ``max(conv(a,a))``.
    ``start_seq`` must be a probability vector (sum = 1).
    Returns the best vector seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to stay on the simplex
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # current convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active lag (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional progress report (without printing too often)
        if it % 200_000 == 0:
            print(f"[EG] it={it:,} best_surrogate={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS polishing of a smooth approximation of max‑conv.
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule: List[float] | None = None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smooth (log‑sum‑exp) surrogate of ``max(conv(a,a))``
    using L‑BFGS‑B.  Operates in the log‑probability domain.

    Parameters
    ----------
    start_vec : np.ndarray
        Starting probability vector (must sum to 1).
    deadline : float
        Absolute Unix time at which the optimisation must stop.
    beta_schedule : list of float, optional
        Sequence of inverse‑temperature parameters.  Larger β → tighter
        approximation of the max.  If omitted a default schedule is used.
    max_iter_per_beta : int, optional
        Maximum L‑BFGS iterations for each β value.

    Returns
    -------
    np.ndarray
        Refined probability vector (still sum‑to‑one).
    """
    if _opt is None:
        # SciPy not available – return the unchanged vector.
        return start_vec

    if beta_schedule is None:
        # progressively larger β; the last value is deliberately huge.
        beta_schedule = [
            1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1_000.0,
            3_000.0, 10_000.0, 30_000.0, 100_000.0,
            300_000.0, 1_000_000.0, 3_000_000.0,
            10_000_000.0, 30_000_000.0, 100_000_000.0
        ]

    # work in log‑probability space
    v = np.log(start_vec + 1e-30)

    n = len(start_vec)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local: np.ndarray) -> Tuple[float, np.ndarray]:
            """Smooth max‑convolution objective and its gradient w.r.t. v."""
            a = _softmax(v_local)                     # probability vector
            conv = _conv_fft(a, a)                    # full convolution
            m = np.max(conv)
            # stabilise exponentials by subtracting the maximum
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of  max(conv) = shift + log(sum_exp)/β
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in the analysis)
            w = exp_beta / sum_exp                    # weight for each lag
            rev_a = a[::-1]
            # full = conv(rev_a, w) ; length = 3·n‑2
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # convert gradient w.r.t. a into gradient w.r.t. log‑probabilities
            # (mirror‑descent formula)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return float(obj), grad_v

        def callback(xk: np.ndarray) -> bool:  # pragma: no cover
            # L‑BFGS calls the callback after every iteration.
            # If we are out of time we raise a KeyboardInterrupt.
            if time.time() > deadline:
                raise KeyboardInterrupt()
            return False

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
            cur_obj = _objective_surrogate(_softmax(v))
            print(f"[L‑BFGS] β={beta_val:.0e} surrogate={cur_obj:.6f}")
        except KeyboardInterrupt:
            print("[L‑BFGS] deadline reached – stopping early.")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] unexpected exception {e!r} – aborting schedule.")
            break

    # Final probability vector
    return _softmax(v)


# ----------------------------------------------------------------------
# Helper to build a useful set of starting vectors
# ----------------------------------------------------------------------
def _build_initial_vectors(n: int,
                           rng: np.random.Generator,
                           seed: int) -> List[np.ndarray]:
    """
    Produce a modest collection of diverse probability vectors.
    The list always contains the previous best (if available) and a
    handful of analytic families (exponential, power‑law, logarithmic,
    uniform, and a couple of random Dirichlet draws).
    """
    init_vecs: List[np.ndarray] = []

    # 1. previous solution (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0.0:
                    init_vecs.append(arr / s)
        except Exception:
            pass   # ignore malformed previous data

    # 2. exponential families (rates chosen to span the interesting region)
    for rate in [0.001, 0.002, 0.0035, 0.005, 0.008]:
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * i)
        init_vecs.append(a / a.sum())

    # 3. power‑law families
    for p in [0.7, 1.0, 1.3, 1.6]:
        i = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (i ** p)
        init_vecs.append(a / a.sum())

    # 4. logarithmic decay
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.log(i + 1.0)
    init_vecs.append(a / a.sum())

    # 5. uniform
    init_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 6. a few random Dirichlet initialisations (concentration < 1 -> sparse)
    for _ in range(2):
        a = rng.gamma(shape=0.5, scale=1.0, size=n)
        init_vecs.append(a / a.sum())

    # ensure we have something even if everything else failed
    if not init_vecs:
        init_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    return init_vecs

    # ------------------------------------------------------------------

        # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        best_seq = init_vecs[0].copy()

        except Exception:
            best_true_val = float("inf")

    # ------------------------------------------------------------------
    # Store the best sequence for the next run and return it.
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"\n[Final] best true value = {best_true_val:.6f}")
    return best_seq.tolist()
