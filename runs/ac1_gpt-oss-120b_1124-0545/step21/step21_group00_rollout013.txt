def _lbfgs_optimize(start_vec, n, deadline, max_iter_per_beta=30000):

    v = np.log(start_vec + 1e-30)
    beta_sched = [1.0,5.0,20.0,80.0,200.0,500.0,1000.0,2000.0,5000.0,20000.0,50000.0,100000.0,200000.0,500000.0,1e6,2e6,5e6,1e7]

    for beta in beta_sched:
        if time.time() > deadline: break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)  # sum=1
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max = shift + log(sum_exp)/beta
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp  # shape 2n-1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1: n-1 + n]  # length n

            # gradient w.r.t v (logits before softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            return obj, grad_v

        def callback(xk):
            # optional: early stop
            return time.time() > deadline

        try:
            res = _opt.minimize(fun=fun, x0=v, method='L-BFGS-B', jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'disp': False})
            v = res.x
        except Exception as e:
            # If abort due to time, break
            break
    return _softmax(v)

"""
Hybrid random‑seed + smooth L‑BFGS + long exponentiated‑gradient
optimization targeting a low L∞‑norm of the self‑convolution.
We use a longer vector (default n=8000) to get a tighter theoretical
lower bound and allocate most of the 1000 s budget to a high‑precision
mirror‑descent (EG)‐polish.
"""

import time, math, numpy as np

try:
    import scipy.optimize as _opt
except Exception:
    _opt = None

# ------- convolution utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a probability vector."""
    n = len(seq)
    return float(2.0 * n * np.max(_conv_fft(seq, seq)))

# ------- seed generators ------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a

def _exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    return a / a_sum

def _beta_shape(n: int, a: float, b: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.beta(a, b, size=n)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s

def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s

# ------- L‑BFGS optimisation on smooth max ------------------------------------
def _lbfgs_smooth(start_vec: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iter_per_beta: int = 30000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp smooth approximation of max(conv)."""
    if _opt is None:
        return start_vec.copy()

    # schedule of inverse temperatures
    beta_schedule = [1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0,
                     2000.0, 5000.0,
                     20000.0, 50000.0,
                     100000.0, 200000.0,
                     500000.0, 1e6, 2e6,
                     5e6, 1e7]

    v = np.log(start_vec + 1e-30)   # variables before softmax

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_b = np.exp(beta_val * (conv - shift))
            sum_exp = exp_b.sum()
            # smooth max = shift + log(sum_exp)/beta
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_b / sum_exp                 # weight for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1: n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            return time.time() > deadline

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)

# ------- Mirror‑descent (exponentiated‑gradient) on exact max -----------------
def _exp_gradient(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 4_000_000,
                  eta0: float = 3.0) -> (np.ndarray, float):
    """Exponentiated‑gradient mirror descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # build sub‑gradient
        g = np.zeros_like(a)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        rev_vals = k - i_vals
        g[i_vals] = a[rev_vals]

        e = np.exp(log_a - max_log)

        if it % 1000 == 0:
            cur = _objective(a)
            if cur < best_val:
                best_val = cur
                best_seq = a.copy()

"""
Hybrid random‑seed + smooth L‑BFGS + long exponentiated‑gradient (EG)
optimization for the step‑function problem.

The algorithm:

1.  Build a large, diverse pool of probability vectors (uniform,
    Dirichlet with many concentrations, sparse Dirichlet, exponential,
    Beta‑shaped, and perturbed copies).

2.  Keep the best few according to the cheap surrogate
    2·n·max(conv(a,a)) (the objective after normalising the vector).

3.  Refine the current best with a long L‑BFGS run on a
    log‑sum‑exp smooth approximation of the max–convolution.
    The optimiser works directly in the log‑softmax space
    (variables v = log a) and the gradient is obtained analytically.

4.  Polish the refined candidate with a large‑scale exponentiated‑gradient
    (mirror‑descent) that minimises the *exact* max‑convolution.
    The update is
        log_a ← log_a – η·g ,   a ← softmax(log_a)
    where g is the sub‑gradient of the active convolution entry.

5.  Perform a few cheap perturb‑and‑short‑EG passes to escape possible
    plateaus.

6.  Return the best sequence (as a plain list) and store it in the
    global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for L‑BFGS
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:                      # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)),                                  \
    valid for any non‑negative vector that sums to 1."""
    n = len(seq)
    return float(2.0 * n * np.max(_conv_fft(seq, seq)))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, a: float, b: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.beta(a, b, size=n)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# L‑BFGS optimisation on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable object required by scipy.optimize.minimize."""
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray):
        """Return (objective, gradient) for the current beta."""
        a = _softmax(v)                              # probability vector
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        # smooth max = shift + log(∑ exp(beta·(conv‑shift))) / beta
        exp_b = np.exp(self.beta * (conv - shift))
        sum_exp = exp_b.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # gradient w.r.t. a (see derivation in the analysis)
        w = exp_b / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[self.n - 1: self.n - 1 + self.n]

        # chain rule through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


def _lbfgs_smooth(start_vec: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iter_per_beta: int = 25000) -> np.ndarray:
    """Run L‑BFGS on the log‑sum‑exp surrogate."""
    if _opt is None:
        return start_vec.copy()

    beta_schedule = [1.0, 5.0, 20.0, 80.0,
                     200.0, 500.0, 1000.0,
                     2000.0, 5000.0,
                     20000.0, 50000.0,
                     100000.0, 200000.0,
                     500000.0, 1e6, 2e6,
                     5e6, 1e7]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        obj_fun = _LbfgsObjective(n, beta)
        try:
            res = _opt.minimize(fun=obj_fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except Exception:                # pragma: no cover
            break
    return _softmax(v)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) on the true max
# ----------------------------------------------------------------------
def _exp_gradient(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 4_000_000,
                  eta0: float = 3.0) -> (np.ndarray, float):
    """
    Mirror‑descent on the true max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # build sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = a[rev]                     # a_{k‑i}
        if k % 2 == 0:                      # diagonal terms get doubled
            mid = k // 2
            if i_start <= mid <= i_end:
                g[mid] += a[mid]

        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        # projection back onto the simplex (softmax)
        m = np.max(log_a)
        e = np.exp(log_a - m)
        a = e / e.sum()

        # occasional progress check
        if it % 1000 == 0:
            cur = _objective(a)
            if cur < best_val:
                best_val = cur
                best_seq = a.copy()
    return best_seq, best_val

    # 4️⃣  Exact EG polishing (most of the remaining time)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polish_deadline = deadline - 3.0
        time_left = polish_deadline - time.time()
        # allocate up to a few million iterations depending on the time left
        max_iters = int(min(5_000_000,
                            max(300_000,
                                time_left / (deadline - start_time) * 4_000_000)))
        seq_eg, val_eg = _exp_gradient(best_seq, n,
                                       polish_deadline,
                                       max_iters=max_iters,
                                       eta0=3.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print("  >>> Updated global best after EG")

    # --------------------------------------------------------------
    # 5️⃣  Light stochastic refinement (perturb‑and‑short‑EG)
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        cand = _perturb(best_seq, rng, sigma=0.001)
        short_deadline = time.time() + 0.4
        cand_seq, cand_val = _exp_gradient(cand, n,
                                           short_deadline,
                                           max_iters=80_000,
                                           eta0=2.5)
        print(f"[Pert‑EG] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_seq, best_val = cand_seq, cand_val
            print(f"  >>> New best after perturb‑EG = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
