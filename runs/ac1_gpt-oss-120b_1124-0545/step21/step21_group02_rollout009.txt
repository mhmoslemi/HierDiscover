def _adam_optimize(start_seq, n, deadline, max_iters=...):
    a = start_seq
    log_a = np.log(a+1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1=0.9
    beta2=0.999
    epsilon=1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t=0
    while time.time() < deadline and t<max_iters:
        t +=1
        # compute conv and argmax
        conv = _conv_fft(a, a)
        k = np.argmax(conv)
        # gradient wrt a
        grad_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1)
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]  # includes factor 2

        # adjust to log gradient: grad_log = a * (grad_a - dot)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a = log_a - 0.1 * m_hat / (np.sqrt(v_hat) + epsilon)

        # project back to simplex
        a = _softmax(log_a)

        # evaluate surrogate
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    n = 4000

    # get previous solution if exists
    prev_seq = globals().get('height_sequence_1')
    candidates = []

    if prev_seq is not None:
        arr = np.asarray(prev_seq, dtype=np.float64)
        if arr.shape == (n,):
            seq = np.maximum(arr, 0.0)
            s = seq.sum()
            if s > 0:
                seq = seq / s
                candidates.append(seq)

    # generate exponential rates
    rates = np.linspace(0.001, 0.010, 10)
    for r in rates:
        seq = np.exp(-r * np.arange(n, dtype=np.float64))
        seq = seq / seq.sum()
        candidates.append(seq)
        # symmetric
        seq_sym = np.exp(-r * np.arange(n, dtype=np.float64)) + np.exp(-r * (np.arange(n, dtype=np.float64)[::-1]))
        seq_sym = seq_sym / seq_sym.sum()
        candidates.append(seq_sym)

    # power laws
    ps = np.linspace(0.5, 3.0, 10)
    idx = np.arange(1, n+1, dtype=np.float64)
    for p in ps:
        seq = idx ** (-p)
        seq = seq / seq.sum()
        candidates.append(seq)
        # symmetric
        seq_sym = seq + seq[::-1]
        seq_sym = seq_sym / seq_sym.sum()
        candidates.append(seq_sym)

    # random Dirichlet draws
    for _ in range(10):

        seq = rng.dirichlet(np.full(n, alpha))
        candidates.append(seq)

    # Evaluate candidates via surrogate and keep best few
    best_seq = None
    best_val = float('inf')
    for seq in candidates:
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq

    print(f"[Init] Best surrogate among candidates = {best_val:.6f}")

    # Multi-start Adam optimization

    cand_vals = [(seq, _objective_surrogate(seq)) for seq in candidates]

    # schedule for Adam runs

            best_val_opt = val_opt
            best_seq_opt = seq_opt

    globals()["height_sequence_1"] = best_seq_opt.tolist()

def _adam_optimize(start_seq, n, deadline, max_iters=500000, lr=0.1):
   a = start_seq.copy()
   log_a = np.log(a + 1e-30)
   m = np.zeros_like(log_a)
   v = np.zeros_like(log_a)
   beta1 = 0.9; beta2 = 0.999; eps = 1e-8
   best_seq = a.copy()
   best_val = _objective_surrogate(a)
   t = 0
   while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # gradient w.r.t a
        grad_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # range inclusive
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]  # compute 2*a_{k-i}
        # subgradient for log space
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)
        # optional clip small for numerical stability?
        # compute surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
   return best_seq, best_val

"""
search for a height sequence that minimises evaluate_sequence

The algorithm mixes inexpensive global exploration (parameter sweeps,
random Dirichlet draws, an optional previous solution) with a
mirror‑descent optimiser on the exact max‑convolution.  The optimiser
uses Adam in log‑space, which usually converges faster than the plain
exponentiated‑gradient used previously.  After the Adam runs a short
value‑annealing refinement is performed if time permits.

All helpers are defined at module level – no closures or lambdas are
used – and the function `propose_candidate` returns a plain list of
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple surrogate: 2·n·max(conv(seq,seq))  (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – assumes seq sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax (mirror‑descent on the probability simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (mirror‑descent)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the exact max‑convolution.
    `start_seq` must already sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # 1) locate the active index of the max‑convolution
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # 2) sub‑gradient of conv_k w.r.t. a (g_i = 2·a_{k-i})
        grad_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        grad_a[idx_i] = 2.0 * a[rev_idx]

        # 3) map to gradient w.r.t. log‑variables (mirror‑descent)
        dot = np.dot(grad_a, a)                # ⟨∇f(a), a⟩
        grad_log = a * (grad_a - dot)          # ∇_log f = a·(∇_a f – ⟨∇_a f,a⟩)

        # 4) Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection back to the simplex
        a = _softmax(log_a)

        # 6) keep the best seen sequence
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing on the simplex (swap + mass transfer)
# ----------------------------------------------------------------------
def _value_anneal(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated–annealing with two move types (swap, mass‑transfer).
    Operates on probability vectors.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.7:      # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] == 0.0:
                continue
            delta = cur_seq[i] * rng.random() * 0.5    # move at most 50 % of source
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta
        else:                        # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)

        # Metropolis acceptance
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generators (always produce a probability vector)
# ----------------------------------------------------------------------
def _exp_decay(rate: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(rate: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    return a / a.sum()


def _power_law(p: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _sym_power_law(p: float, n: int) -> np.ndarray:
    a = _power_law(p, n)
    a = a + a[::-1]
    return a / a.sum()

    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    candidates.append(arr / s)
        except Exception:
            pass

    # deterministic families
    for rate in np.linspace(0.001, 0.010, 12):
        candidates.append(_exp_decay(rate, n))
        candidates.append(_sym_exp_decay(rate, n))

    for p in np.linspace(0.5, 3.0, 12):
        candidates.append(_power_law(p, n))
        candidates.append(_sym_power_law(p, n))

    # random Dirichlet draws (different concentration levels)
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.5)
        cand = rng.dirichlet(np.full(n, alpha))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # 2. evaluate cheap surrogate and keep the best few
    # ------------------------------------------------------------------
    # evaluate each candidate (fast, uses FFT)
    cand_vals = []
    for seq in candidates:
        val = _objective_surrogate(seq)
        cand_vals.append((seq, val))

    # sort by surrogate value (lower is better)
    cand_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_candidates = [c for c, _ in cand_vals[:top_k]]

    best_seq = top_candidates[0]
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate among pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror‑descent Adam refinement (run several starts)
    # ------------------------------------------------------------------
    for idx, start_seq in enumerate(top_candidates):
        now = time.time()
        if now >= deadline - 30.0:      # keep a little time for polishing
            break
        # each Adam run gets a slice of the remaining budget
        remaining = deadline - now
        run_deadline = now + min(remaining * 0.25, 200.0)   # ≤200 s per run
        seq_opt, val_opt = _adam_optimize(start_seq, n, run_deadline,
                                          max_iters=250_000, lr=0.15)
        print(f"[Adam #{idx}] best surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt

    # ------------------------------------------------------------------
    # 4. Final short value‑annealing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_ann, val_ann = _value_anneal(best_seq, n, deadline, rng,
                                         max_iters=150_000)
        print(f"[Anneal] after final anneal = {val_ann:.6f}")
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann

    # ------------------------------------------------------------------
    # 5. Final evaluation with the secure function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # store for next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
