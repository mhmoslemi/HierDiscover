def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1 # margin
    n = 4000

    # Init from previous solution
    init = globals().get("height_sequence_1")
    if isinstance(init, list) and len(init)==n:
        a = np.asarray(init, dtype=np.float64)
        # Clip negatives & scale to sum 1
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0/n, dtype=np.float64)
        else:
            a = a / a.sum()
    else:
        # start with uniform
        a = np.full(n, 1.0/n, dtype=np.float64)

    # Compute initial surrogate
    def surrogate(x):
        conv = _conv_fft(x, x)
        return float(2.0 * n * np.max(conv))

    cur_val = surrogate(a)
    best_a = a.copy()
    best_val = cur_val

    # Parameters
    max_iters = 300_000   # will be limited by time
    # Determine per-iteration cost approx? We'll just loop while time < deadline
    it = 0
    # initial temperature related to cur_val
    T_start = cur_val * 0.1
    T_end = cur_val * 1e-4

    # Step size initial: second
    step_start = 0.01  # max delta fraction of current a_i
    step_end = 1e-5

    while time.time() < deadline and it < max_iters:
        it += 1
        # linear schedules
        T = T_start + (T_end - T_start) * (it / max_iters)
        max_step = step_start + (step_end - step_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # ensure delta <= a[i] and also <= max_step
        max_delta = min(a[i], max_step)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # propose new a
        a[i] -= delta
        a[j] += delta
        # compute surrogate
        val = surrogate(a)

        improve = val < cur_val
        if improve or rng.random() < np.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_val = val
            if improve:
                # keep
                pass
        else:
            # revert
            a[i] += delta
            a[j] -= delta
            # cur_val unchanged

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"Iter {it}: new best surrogate {best_val:.6f}")

    # final evaluate sequence true function
    seq = best_a.tolist()
    try:
        true_val = evaluate_sequence(seq)
        print(f"Final true evaluate_sequence value: {true_val}")
    except Exception as exc:
        print("Error evaluating:", exc)

    globals()["height_sequence_1"] = seq
    return seq
```

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The optimiser works as follows

1. Load the best known candidate (``height_sequence_1``) if it exists,
   otherwise start from the uniform distribution.
2. Re‑optimise the candidate with a **Adam** optimiser in the
   log‑softmax space.  The objective is the exact surrogate
   ``2·n·max(conv(a,a))`` (the value returned by ``evaluate_sequence``
   when the vector sums to 1).  The Adam update uses the sub‑gradient of
   the active convolution entry, i.e. the gradient of the max‑convolution.
3. To avoid being trapped in a local basin, several short Adam runs are
   launched from randomly perturbed initial points.  The best result of
   all runs is kept.
4. The final candidate is evaluated with the provided ``evaluate_sequence``
   function and stored in the global variable ``height_sequence_1`` for
   possible continuation runs.

The whole procedure respects the allotted time budget (default 1000 s)
and prints a few progress messages.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector whose sum is 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # degenerate case – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """
    Compute the surrogate value 2·n·max(conv(a,a)) for a vector that already
    satisfies sum(a)=1 and a_i≥0.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient of the active lag)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 1_500_000,
              lr: float = 0.12,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- sub‑gradient of the active convolution entry -----------------
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                     # ∂ max(conv) / ∂ a_i

        # Gradient w.r.t. log‑softmax parameters
        dot_ag = np.dot(a, g)
        grad_log = a * (g - dot_ag)                      # ∂f/∂log a_i

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(log_a)                              # remains on simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (every 200 k iterations)
        if it % 200_000 == 0:
            print(f"   [Adam] it={it:,}, best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The search starts from the previously saved best sequence (if any) and
    refines it with several Adam optimisation runs, each run possibly starting
    from a randomly perturbed version of the current best.
    The final best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise with the previous solution if it exists; otherwise uniform.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / a0.sum()
        print("[Init] Loaded previous solution.")
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("[Init] No previous solution – starting from uniform.")

    cur_best_seq = a0.copy()
    cur_best_val = _objective_surrogate(cur_best_seq)
    print(f"[Init] surrogate = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Adam optimisation – several short runs from perturbed starts.
    # ------------------------------------------------------------------
    run = 0
    while time.time() < deadline and run < 8:          # at most eight restarts
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget (but at least 5 s)
        run_budget = min(remaining * 0.20, max(5.0, remaining - 60.0))
        sub_deadline = time.time() + run_budget

        # starting point for this run
        if run == 0:
            start_seq = cur_best_seq.copy()
        else:
            # random perturbation in log‑space (small Gaussian noise)
            log_start = np.log(cur_best_seq + 1e-30)
            log_start += 0.04 * rng.normal(size=n)      # modest jitter
            start_seq = _softmax(log_start)

        print(f"\n[Run {run+1}] start surrogate = {_objective_surrogate(start_seq):.6f}")

        seq, val = _adam_opt(start_seq,
                             n,
                             sub_deadline,
                             max_iters=800_000,
                             lr=0.12)

        if val < cur_best_val:
            cur_best_val = val
            cur_best_seq = seq.copy()
            print(f"[Run {run+1}] NEW global best surrogate = {cur_best_val:.6f}")

        run += 1

    # ------------------------------------------------------------------
    # Final long Adam polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing with Adam ---")
        final_seq, final_val = _adam_opt(cur_best_seq,
                                         n,
                                         deadline,
                                         max_iters=2_000_000,
                                         lr=0.10)
        if final_val < cur_best_val:
            cur_best_seq, cur_best_val = final_seq, final_val
            print(f"[Final] improved to {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for continuation runs
    globals()["height_sequence_1"] = cur_best_seq.tolist()
    return cur_best_seq.tolist()
