```
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 5_000_000,
                           eta0: float = 5.0,
                           top_k: int = 3) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    iteration = 0
    while iteration < max_iters and time.time() < deadline:
        iteration += 1
        conv = _conv_fft(a, a)
        # Get indices of top_k maximum values
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            # use argpartition for efficiency
            # get top_k indices
            # we want sorted descending maybe not needed
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            ks = idxs.tolist()
        # compute gradient vector
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            # contributions from ordered pairs
            g[idx_i] += 2.0 * a[rev_idx]
        # step size schedule
        eta = eta0 / math.sqrt(iteration)
        log_a -= eta * g
        # apply softmax renormalization
        # we can use stable softmax.
        # For numerical stability, subtract max.
        m = np.max(log_a)
        e = np.exp(log_a - m)
        a = e / e.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
Search for a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

*   Generate a diverse pool of probability vectors (uniform, full‑length
    Dirichlet draws, sparse Dirichlet draws, tiny Gaussian‑mixture
    approximations).
*   Keep the best seed according to the fast surrogate ``2·n·max(conv)``.
*   Refine every seed with a large‑scale exponentiated‑gradient (“mirror
    descent”) iteration that works directly on the true max‑convolution
    objective.  In each step the gradient is formed by the *top‑k* most
    violated convolution entries (instead of only the absolute maximum);
    this gives a stronger descent direction.
*   After each refinement keep the best sequence seen.
*   Perform a final long‑running refinement starting from the current
    global best.
*   Return the best sequence found (scaled to sum = 1).  The function also
    stores the sequence in ``globals()[\"height_sequence_1\"]`` for
    later runs.

The implementation respects the 1000 s budget, prints progress, and
contains only top‑level helper functions (no closures or ``lambda``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k random positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 5.0,
                            top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.

    Parameters
    ----------
    start_seq : np.ndarray
        Starting probability vector (must sum to 1, non‑negative).
    n : int
        Length of the sequence.
    deadline : float
        Unix time after which the routine stops.
    max_iters : int, optional
        Upper bound on the number of iterations.
    eta0 : float, optional
        Initial step‑size; actual step at iteration t is eta0/√t.
    top_k : int, optional
        Number of largest convolution entries used when building the
        sub‑gradient (k=1 reproduces the original code).

    Returns
    -------
    best_seq : np.ndarray
        Best sequence found during the run.
    best_val : float
        Surrogate objective of the best sequence.
    """
    a = start_seq.copy()
    # avoid log(0)
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)

        # determine the indices that currently have the largest values
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            # ``np.argpartition`` is O(n) and gives the indices of the
            # ``top_k`` entries (unsorted is fine).
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            ks = idxs.tolist()

        # build the sub‑gradient (ordered‑pair contribution)
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # stable softmax
        m = np.max(log_a)
        e = np.exp(log_a - m)
        a = e / e.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:

    rng = np.random.default_rng(seed)

    n = 4000  # length of the step function

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of initial seeds
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    # Include the sequence from a previous run if it exists.
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            seeds.append(prev_arr / prev_arr.sum())

    # Full‑length Dirichlet draws with a range of concentration parameters.
    for conc in (0.02, 0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse Dirichlet draws – support sizes from 30 up to 400.
    for k in (30, 60, 90, 150, 250, 350):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # Tiny Gaussian‑mixture seeds (8 components) – useful for creating
    # smooth bumps on top of a mostly flat background.
    K_mix = 8
    for _ in range(4):
        params = np.concatenate([
            rng.random(K_mix),                     # (un‑normalised) weights
            rng.random(K_mix),                     # means in [0,1]
            rng.uniform(0.001, 0.4, K_mix)         # standard deviations
        ])
        # Build a probability vector from the mixture.
        w = np.maximum(params[0:K_mix], 0.0)
        if w.sum() > 0.0:
            w = w / w.sum()
        else:
            w = np.full(K_mix, 1.0 / K_mix)
        mu = np.clip(params[K_mix:2 * K_mix], 0.0, 1.0)
        sigma = np.clip(params[2 * K_mix:3 * K_mix], 0.001, 0.4)

        i = np.arange(n, dtype=np.float64) / (n - 1.0)
        a = np.zeros(n, dtype=np.float64)
        for comp in range(K_mix):
            a += w[comp] * np.exp(-((i - mu[comp]) ** 2) / (2.0 * sigma[comp] ** 2))
        s = a.sum()
        if s > 0.0:
            seeds.append(a / s)
        else:
            seeds.append(_uniform(n))

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate the seed pool using the fast surrogate and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Evaluating initial seeds (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Refine every seed with exponentiated‑gradient (top‑k gradient).
    # ------------------------------------------------------------------
    # We'll spend roughly half of the remaining budget on systematic
    # refinement; the rest will be used for stochastic perturb‑and‑polish.
    total_budget = deadline - time.time()

    # Number of seeds that we will try to improve – limited by time.
    max_seeds_to_refine = 12
    seeds_to_refine = seeds[:max_seeds_to_refine]

    for s_idx, seed_vec in enumerate(seeds_to_refine):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate a small chunk of the refinement budget to this seed
        sub_deadline = time.time() + min(refinement_budget / max_seeds_to_refine,
                                         remaining * 0.4)

        print(f"\n[refine {s_idx+1}/{len(seeds_to_refine)}] start")
        seq_ref, val_ref = _exp_gradient_opt_multi(seed_vec, n,
                                                   sub_deadline,
                                                   max_iters=2_500_000,
                                                   eta0=5.0,
                                                   top_k=3)
        print(f"[refine {s_idx+1}] surrogate = {val_ref:.6f}")
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"  >>> new global best = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

                                                     short_deadline,
                                                     max_iters=300_000,
                                                     eta0=4.0,
                                                     top_k=5)
        print(f"[pert‑eg] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"  >>> new best after perturb‑eg = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final long optimisation from the best point.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_deadline = deadline - 1.0
        print("\n--- Final long‑run exponentiated‑gradient refinement ---")
        final_seq, final_val = _exp_gradient_opt_multi(best_seq, n,
                                                       final_deadline,
                                                       max_iters=5_000_000,
                                                       eta0=6.0,
                                                       top_k=5)
        print(f"[final] surrogate = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq.copy()
            print(f"  >>> updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with ``evaluate_sequence``.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

            seed_vec, n, sub_deadline,
            max_iters=2_500_000,
            eta0=5.0,
            top_k=3
        )
        print(f"[refine {s_idx+1}] surrogate = {val_ref:.6f}")
        if val_ref < best_val:
            best_val = val_ref
            best_seq = seq_ref.copy()
            print(f"  >>> new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Stochastic perturb‑and‑polish passes.
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        candidate = _perturb(best_seq, rng, sigma=0.001)
        short_deadline = time.time() + 0.6
        cand_seq, cand_val = _exp_gradient_opt_multi(
            candidate, n, short_deadline,
            max_iters=300_000,
            eta0=4.0,
            top_k=5
        )
        print(f"[pert‑eg] surrogate = {cand_val:.6f}")
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"  >>> new best after perturb‑eg = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final long‑run optimisation from the current best.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_deadline = deadline - 1.0
        print("\n--- Final long‑run exponentiated‑gradient refinement ---")
        final_seq, final_val = _exp_gradient_opt_multi(
            best_seq, n, final_deadline,
            max_iters=5_000_000,
            eta0=6.0,
            top_k=5
        )
        print(f"[final] surrogate = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq.copy()
            print(f"  >>> updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with ``evaluate_sequence``.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
