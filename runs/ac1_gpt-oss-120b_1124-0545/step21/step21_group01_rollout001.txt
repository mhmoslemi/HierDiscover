"""
Propose a step‑function that (empirically) beats the previous 1.5085 bound.

The search works in three stages:

1️⃣  **Diverse seeding** – a large pool of random draws:
    * uniform, arcsine, triangular
    * Dirichlet draws with many concentrations (very sparse to almost uniform)
    * Gaussian‑mixture families with 3‑8 components (weights, means, sigmas)

2️⃣  **Local refinement** – each seed is polished with
    * L‑BFGS‑B on a smooth log‑sum‑exp surrogate of the max‑convolution
    * an exact exponentiated‑gradient (mirror descent) on the true max‑convolution
    * a cheap stochastic walk that moves a tiny amount of mass between two coordinates

3️⃣  **Restart loop** – the whole pipeline is repeated many times (until the
    1000 s budget is exhausted) with fresh random seeds.  The best sequence
    found over all restarts is kept, stored in ``height_sequence_1`` and
    returned.

All vectors are normalised to sum to 1, therefore
    evaluate_sequence(seq) == 2·n·max(conv(seq,seq))
for the returned ``seq``.  The code uses only NumPy (and SciPy if available)
and stays well below the time budget.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the global L‑BFGS optimiser)
# ----------------------------------------------------------------------
try:                     # pragma: no cover

except Exception:       # pragma: no cover

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate / exact objective (the sum of the vector is 1 → exact = surrogate)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  ``seq`` must sum to 1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple deterministic seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Random Dirichlet draw (sparsity controlled by concentration)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Gaussian‑mixture family (variable number of components)
# ----------------------------------------------------------------------
def _mixture_gaussian_general(params: np.ndarray, n: int, K: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    Layout (3·K numbers):
        w_0 … w_{K‑1}      (non‑negative, later normalised)
        μ_0 … μ_{K‑1}      (clipped to [0,1])
        σ_0 … σ_{K‑1}      (clipped to [0.001,0.5])
    The resulting vector is normalised to sum to 1.
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# L‑BFGS smoothing on the log‑sum‑exp approximation of the max‑convolution
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iter_per_beta: int = 20_000) -> np.ndarray:
    """
    Optimise the smooth max‑convolution objective using L‑BFGS‑B.
    Variables live in the log‑softmax space.
    """
    if _opt is None:
        # SciPy not available – fall back to the start vector
        return start_vec.copy()

    # initialise in log‑softmax space
    v = np.log(start_vec + 1e-30)

    # a relatively aggressive β‑schedule (more iterations per β)
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                  1000.0, 2000.0, 5000.0, 10_000.0,
                  20_000.0, 50_000.0, 100_000.0, 200_000.0,
                  500_000.0, 1_000_000.0]

    for beta in beta_sched:
        if time.time() > deadline:
            break
        beta_f = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # probability vector
            conv = _conv_fft(a, a)                     # full convolution
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (shifted for stability)
            shift = max_c
            exp_beta = np.exp(beta_f * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_f)

            # gradient w.r.t. a (weighted sum of conv entries)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # d/d a of Σ w·conv
            grad_a = grad_a * (2.0 * n)               # outer factor 2·n

            # map to gradient in log‑softmax coordinates
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient at the (single) active index.
    Returns the best sequence encountered and the corresponding surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 50_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pic a tiny chunk of mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq, n)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s`` seconds
    and stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if it matches the current size)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    best_seq = None
    best_val = np.inf
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                arr = arr / arr.sum()
                best_seq = arr.copy()
                best_val = _objective_surrogate(arr, n)
                print("[Init] Loaded previous best surrogate =", best_val)
        except Exception:   # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Build a rich pool of deterministic and random seeds
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Dirichlet draws with a variety of concentrations (very sparse → dense)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # A handful of random Gaussian‑mixture seeds (variable K)
    for _ in range(6):
        K = rng.integers(3, 9)                     # 3 … 8 components
        params = np.concatenate([
            rng.random(K),                         # raw weights
            rng.random(K),                         # means in [0,1]
            rng.uniform(0.001, 0.4, K)             # sigmas
        ])
        seeds.append(_mixture_gaussian_general(params, n, K))

    # ------------------------------------------------------------------
    # 1️⃣  Evaluate seeds and keep the best one
    # ------------------------------------------------------------------
    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec, n)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Main restart loop (random seeds → L‑BFGS → EG → local search)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 8.0:           # not enough time for a full restart
            break

        restart += 1
        print(f"\n--- Restart #{restart} (remaining {remaining:.1f}s) ---")

        # a) random Dirichlet seed (concentration drawn adaptively)
        conc = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2, 0.5],
                          p=[0.05, 0.10, 0.15, 0.25, 0.25, 0.20])
        seed_vec = _random_dirichlet(n, conc, rng)

        # b) L‑BFGS smoothing (takes about 50‑60% of the remaining budget)
        sub_deadline = time.time() + remaining * 0.55
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs, n)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("   >>> Updated global best (L‑BFGS)")

        # c) Exact exponentiated‑gradient polishing (≈30% of the remaining time)
        polishing_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=800_000,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

        # d) Final cheap random walk (the rest of the time)
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                deadline,
                                                attempts=30_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 3️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
