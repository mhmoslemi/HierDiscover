"""
propose_candidate – global‐search + mirror descent + hill climbing.

The function builds many analytic “starter” sequences (uniform,
exponential, power‑law, Beta‑like) and a few random Dirichlet draws.
Each starter is polished by a short exponentiated‑gradient (mirror‑descent)
run that follows the averaged sub‑gradient of the max‑convolution.
The best candidate is then refined with a long EG pass, followed by a
lightweight stochastic hill‑climbing phase that swaps tiny amounts of
mass between two positions if it improves the surrogate objective.
If SciPy is present, a final L‑BFGS optimisation on a smooth
log‑sum‑exp approximation of the max is performed.
All work respects the 1000 s budget (checked frequently).  The final
sequence (normalised to ℓ₁‑norm = 1) is stored globally as
``height_sequence_1`` and returned as a plain Python list – ready for
``evaluate_sequence``.
"""

import time
import math
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L1‑norm 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _exp_symmetric(n: int, rate: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-rate·i)+exp(-rate·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑like density on {0,…,n‑1} with parameters a,b>0."""
    i = np.arange(1, n + 1, dtype=np.float64)  # 1‑based for nicer formula
    w = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance to collect *active* indices
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    # average over the number of active positions
    g /= max(1, len(active))
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the averaged sub‑gradient."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent on the max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            5_000_000.0, 20_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max = shift + logsumexp / beta
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp          # soft weights for the smooth max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)    # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Quick 1‑D search for the optimal power‑law exponent
# ----------------------------------------------------------------------
def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped sequence whose surrogate is locally best."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # coarse grid
    for p in np.linspace(0.0, 2.5, 21):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # local refinement
    p_center = None
    for step in [0.2, 0.1, 0.05, 0.02]:
        if time.time() > deadline:
            break
        # use current best p as centre
        if p_center is None:
            # retrieve centre from coarse grid
            grid = np.linspace(0.0, 2.5, 21)
            p_center = grid[np.argmin([_objective_surrogate(_power_law(n, pp)) for pp in grid])]
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq

                                  max_iters=250_000,
                                  eta0=1.0)
        print(f"[PGD] surrogate = {val_pg:.6f}")
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()

    # --------------------------------------------------------------
    # 5.  Hill‑climbing perturbations (tiny mass swaps)
    # --------------------------------------------------------------
    # Very cheap local search – works well for shaving the last few percent.
    while time.time() < deadline - 5:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small fraction of mass from i to j
        delta = (rng.random() - 0.5) * 0.02  # up to ±1 % of a_i
        # keep within the simplex
        delta = np.clip(delta, -best_seq[j], best_seq[i])
        if abs(delta) < 1e-12:
            continue
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_surrogate(cand)
        if val < best_val:
            best_seq, best_val = cand, val
            print(f"[Hill] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 6.  Final L‑BFGS smoothing (if SciPy available)
    # --------------------------------------------------------------
    if time.time() < deadline - 2 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
        if val_lb < best_val:
            best_seq, best_val = seq_lb, val_lb

    # --------------------------------------------------------------
    # 7.  True evaluation via the supplied function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Result] evaluation raised:", exc)
        true_val = None

    # Store globally for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
