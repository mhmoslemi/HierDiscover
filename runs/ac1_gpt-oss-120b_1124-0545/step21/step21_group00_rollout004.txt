def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4000
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Load previous best if exists: height_sequence_1
    prev_seq = globals().get('height_sequence_1')
    candidates = []
    if prev_seq is not None:
        arr = np.asarray(prev_seq, dtype=np.float64)
        if arr.size == n:
            cand = arr/arr.sum()
            candidates.append(cand)
    
    # uniform
    candidates.append(_uniform(n))
    
    # Dirichlet draws
    conc_levels = [0.01, 0.05, 0.1, 0.5, 1.0]
    for conc in conc_levels:
        for _ in range(30): # total 150
            candidates.append(_random_dirichlet(n, conc, rng))

    # Sparse Dirichlet
    for k in [50, 80, 120]:
        for _ in range(10):
            candidates.append(_sparse_dirichlet(n, k, rng))

    # Gaussian mixture random draws
    K_mix = 12
    for _ in range(300):
        params = np.concatenate([
            rng.random(K_mix),                     # weights
            rng.random(K_mix),                     # mus
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        candidates.append(_mixture_gaussian(params, n, K=K_mix))

    # Fourier softmax draws
    D_fourier = 30
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis_list = []
    for k in range(1, D_fourier//2 + 1):
        basis_list.append(np.sin(2.0*np.pi*k*i))
        basis_list.append(np.cos(2.0*np.pi*k*i))
    basis = np.column_stack(basis_list)

    for _ in range(300):
        theta = rng.normal(scale=1.0, size=D_fourier)
        candidates.append(_fourier_softmax(theta, basis))

    # Evaluate all candidates quickly
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating candidates ---")
    for idx, seq in enumerate(candidates):
        if time.time() > deadline:
            break
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
        # optional print progress at intervals
        if idx % 100 == 0:
            print(f"Eval {idx}/{len(candidates)}: current best {best_val:.6f}")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # Stage: LBFGS smoothing
    if _opt is not None and time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.4
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                          max_iter_per_beta=30000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[LBFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()

"""
Hybrid random‑search + exponentiated‑gradient polishing.

1. Generate a large, diverse pool of candidate step‑functions:
   * uniform, sine‑shaped, exponential‑decay, triangular.
   * Dirichlet draws with several concentration parameters.
   * Sparse Dirichlet draws (limited support).
   * Random Gaussian‑mixture models (K≈12).
   * Random Fourier‑softmax models (≈30 basis functions).

2. Evaluate the cheap surrogate  2·n·max(conv)  for every candidate and
   keep the best one.

3. Refine the current best with a few runs of exponentiated‑gradient
   (mirror‑descent) on the exact max‑convolution objective, using several
   step‑size schedules and a simple perturb‑and‑re‑polish loop.

All work respects the global 1000 s budget; the final sequence is stored
in the global variable ``height_sequence_1`` and returned as a ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq, seq)).
    The input is expected to be a probability vector (sum≈1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Seed generation (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_shape(n: int) -> np.ndarray:
    """Half‑wave sine, positive on (0,π)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(i - mid) / mid
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exponential_decay(n: int, rate: float = 0.001) -> np.ndarray:
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc > 0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet on a random support of size k."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, later normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """
    Build a probability vector by applying a softmax to a Fourier series.
    a_i ∝ exp( Σ_k θ_k φ_k(i) ).
    """
    logits = basis @ theta
    return _softmax(logits)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, keep non‑negative, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 3.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
