"""
step‑function optimiser – extended version

The optimiser builds a larger and more diverse seed bank (including
symmetric exponential/power families, U‑shaped beta distributions,
mixtures with the uniform window and sparse Dirichlet draws) and then
refines the best few seeds.

Two optimisation stages are used:

1.  Exponentiated‑gradient (EG) mirror‑descent in log‑softmax space.
    The sub‑gradient is computed on the *exact* max‑convolution;
    a decreasing step‑size η_t = η₀ / √t is used.

2.  A continuation L‑BFGS optimiser on a smooth log‑sum‑exp
    approximation of the max.  The β‑schedule is dense (up to 10⁷) and
    each β gets a generous iteration budget, allowing the optimiser
    to converge close to the true non‑smooth optimum.

A short simulated‑annealing (SA) polish follows each L‑BFGS run.
Finally a long EG polishing round is performed on the globally best
sequence.

All sequences are kept on the probability simplex (sum = 1) which
guarantees that the surrogate 2·n·max(conv) equals the true
evaluate_sequence objective.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vector already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) for a simplex vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta shape (α,β) on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α)+b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _sym_exponential(n: int, rate: float) -> np.ndarray:
    """U‑shaped exponential: a_i ∝ exp(-rate·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * np.minimum(i, n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power(n: int, p: float, delta: float = 1e-6) -> np.ndarray:
    """U‑shaped power law: a_i ∝ (i+δ)^‑p + (n‑1‑i+δ)^‑p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) + (n - 1 - i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse random vector (positive entries only)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = rng.random(len(idx))
    s = a.sum()
    if s == 0.0:
        a[0] = 1.0
        s = 1.0
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = True,
                      tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent in log‑space using a sub‑gradient of the exact max."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # keep away from log(0)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # which indices are (almost) active?
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * tol
            active = np.where(conv >= max_val - eps)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    v = np.log(start_vec + 1e-30)      # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            w = exp_beta / sum_exp                # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.02,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """Simple SA with Gaussian perturbations."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within ``budget_s`` seconds.
    The best found sequence is stored in the global variable
    ``height_sequence_1`` and also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – keep it modest so that many FFTs fit in the budget
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution (if any) to enable continuity
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation – a very diverse collection
    # --------------------------------------------------------------
    seeds = []

    # base analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # Kaiser window (different betas)
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        # we fall back to uniform if SciPy is missing – no harm
        seeds.append(_uniform(n))  # placeholder, already added earlier

    # Gaussian shapes (wide to narrow)
    for sigma_frac in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays (log‑spaced)
    for rate in np.logspace(-5, -0.2, 30):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails (monotone)
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power(n, p))

    # Symmetric exponential (U‑shaped)
    for rate in np.logspace(-5, -0.2, 15):
        seeds.append(_sym_exponential(n, rate))

    # Symmetric power‑law (U‑shaped)
    for p in np.linspace(0.5, 4.0, 15):
        seeds.append(_sym_power(n, p))

    # Beta shapes (full grid, also very spiky extremes)
    alphas = np.concatenate((np.logspace(-3, -1, 6), np.linspace(0.2, 5.0, 8), np.linspace(5.0, 10.0, 4)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Logistic profiles
    for scale in [0.5, 1.0, 2.0, 5.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # Random Dirichlet draws (wide concentration range)
    conc_vals = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
    for conc in conc_vals:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse random vectors (different sparsities)
    for k in [8, 16, 32, 64]:
        seeds.append(_sparse_random(n, k, rng))

    # Mixtures of uniform with a U‑shaped exponential (various mix ratios)
    for rate in [1e-3, 5e-3, 1e-2, 5e-2]:
        exp_seq = _sym_exponential(n, rate)
        for mix in [0.05, 0.15, 0.30, 0.45]:
            seeds.append(_mix_shapes(_uniform(n), exp_seq, mix))

    # Perturb previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            pert = prev_arr + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    # keep the best few for full optimisation
    evaluated.sort(key=lambda x: x[1])
    top_k = min(25, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Global best initialisation
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair slice of the remaining budget
        remaining = deadline - time.time()
        # give a slightly larger share to earlier seeds
        share = remaining * (0.8 / (top_k - si) + 0.2 / top_k)
        sub_deadline = time.time() + share

        # ---- (a) exponentiated‑gradient (exact max) ----
        seq_eg, val_eg = _exp_gradient_opt(start_seq=seed_vec.copy(),
                                            n=n,
                                            deadline=sub_deadline,
                                            max_iters=400_000,
                                            eta0=7.0,
                                            single_active=True,
                                            tol=1e-12)

        # ---- (b) L‑BFGS smoothing ----------------------
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_opt(start_vec=seq_eg,
                                          n=n,
                                          deadline=sub_deadline,
                                          beta_schedule=None,
                                          max_iter_per_beta=2000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # ---- (c) short SA polish -----------------------
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(start_seq=seq_lbfgs,
                                                  n=n,
                                                  deadline=hill_deadline,
                                                  rng=rng,
                                                  max_attempts=120_000,
                                                  start_sigma=0.002,
                                                  T_start=0.015,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick best of the four candidates for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1:02d}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long‑run polishing on the global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (exact max) ---")
        final_seq, final_val = _exp_gradient_opt(start_seq=global_best_seq,
                                                 n=n,
                                                 deadline=deadline,
                                                 max_iters=5_000_000,
                                                 eta0=6.0,
                                                 single_active=True,
                                                 tol=1e-12)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(start_seq=global_best_seq,
                                                          n=n,
                                                          deadline=deadline,
                                                          rng=rng,
                                                          max_attempts=150_000,
                                                          start_sigma=0.001,
                                                          T_start=5e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. True evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
