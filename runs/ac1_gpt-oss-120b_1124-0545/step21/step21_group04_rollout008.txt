Given sequence a (list of floats). The function does:

- Ensure sequence is list; elements non-negative; clamp to [0,1000]; sum_a = sum (clipped) then if sum_a < 0.01 => return inf.
- Compute n = len(sequence)
- Compute convolution of sequence with itself (full conv) b = np.convolve(sequence, sequence)
- Compute max_b = max(b)
- Return 2 * n * max_b / (sum_a**2)

def _adam_opt(start_seq, n, deadline, beta_schedule=None, iters_per_beta=2000):
    if beta_schedule is None: beta_schedule = [1, 5, 20, 80, 200, 500, 1000, 2000, 5000, 20000]
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    t = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t+=1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            # compute smooth weights
            maxc = conv.max()
            # For numerical stability, ensure shift across steps
            shift = maxc
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp
            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # grad wrt softmax logits
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1-beta1) * grad_v
            v2 = beta2 * v2 + (1-beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1**t)
            v2_hat = v2 / (1 - beta2**t)

            # current surrogate (max conv)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val

"""
Propose a high‑quality step‑function (non‑negative vector on the simplex) that
minimises the supplied ``evaluate_sequence`` objective.

The algorithm works in three stages:

1. **Initialisation** – build a small portfolio of analytic starters
   (uniform, exponential decay, power‑law, arcsine, sinusoidal) together with a
   few Dirichlet random draws (and the previous best if available).

2. **Rough polishing** – each starter is quickly refined by the exponentiated‑
   gradient (mirror‑descent) optimiser that works directly on the non‑smooth
   max‑convolution objective.

3. **Fine optimisation** – the current best candidate is then optimised with a
   continuation‑Adam scheme on a *smooth* approximation of the max‑convolution.
   The smoothness is controlled by an increasing schedule of the softmax‐β
   parameter; for each β a short Adam run is performed.  The Adam update works
   on the log‑softmax parametrisation (v‑variables) which automatically enforces
   the positivity and ℓ¹‑normalisation constraints.

Optionally a short projected‑gradient polishing and an L‑BFGS smoothing phase
are performed if time permits.  The best sequence found during the whole run is
returned (and stored globally as ``height_sequence_1``) before the time budget
expires.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the final L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector is ℓ¹‑normalised to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence with ℓ¹ norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max utilities (used by exponentiated‑gradient and Adam)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0..n‑1, then normalised to ℓ¹=1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _arcsine_dist(n: int) -> np.ndarray:
    """Discrete version of the arcsine beta(½,½) density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n - i))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _sin_dist(n: int) -> np.ndarray:
    """Half‑wave sinusoidal shape: a_i ∝ sin(π·(i+½)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    if active.size > 0:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the averaged sub‑gradient."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Euclidean projected (sub‑)gradient descent on the max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule: List[float] | None = None,
              iters_per_beta: int = 3_000) -> tuple[np.ndarray, float]:
    """
    Continuation‑Adam on the smooth max‑convolution.
    The loss is 2·n·(β⁻¹·log Σ exp(β·conv)).
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         20_000.0, 50_000.0, 100_000.0]

    v = np.log(start_seq + 1e-30)          # logits (softmax parametrisation)
    m = np.zeros_like(v)                  # 1st moment
    v2 = np.zeros_like(v)                 # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            max_c = conv.max()
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp            # weights for the smooth max

            # gradient w.r.t. a  (same expression as in the L‑BFGS helper)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. softmax logits (v)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update (learning‑rate decays like 1/√t)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)
            lr = 0.02 / math.sqrt(t)
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

            # current (non‑smooth) objective – used for book‑keeping
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2_000) -> np.ndarray:
    """Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Quick 1‑D search for the optimal power‑law exponent
# ----------------------------------------------------------------------
def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped sequence whose surrogate is locally best."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # coarse grid
    for p in np.linspace(0.0, 2.0, 21):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # local refinement
    p_center = 0.0
    # pick the best p from the coarse grid
    vals = [_objective_surrogate(_power_law(n, p)) for p in np.linspace(0.0, 2.0, 21)]
    p_center = np.linspace(0.0, 2.0, 21)[int(np.argmin(vals))]

    step = 0.2
    for _ in range(5):
        if time.time() > deadline:
            break
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ¹‑normalised) that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time budget,
    stores the best sequence globally as ``height_sequence_1`` and returns it as
    a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse the length of the previously stored best vector,
    # otherwise fall back to 4000.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, (list, tuple)):
        n = len(prev)
    else:
        n = 4000

    # ------------------------------------------------------------------
    # 1. Gather initial candidates
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # a) previous best, if compatible
    if prev is not None and len(prev) == n:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s > 0.0:
                seq = seq / s
            else:
                seq = _uniform(n)
            candidates.append(seq)
        except Exception:
            pass

    # b) analytic starters
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.003))
    candidates.append(_best_power_law(n, deadline))
    candidates.append(_arcsine_dist(n))
    candidates.append(_sin_dist(n))

    # c) a few random Dirichlet draws
    for _ in range(4):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2. Short EG polish for each starter (cheap, 1 × 10⁵ iterations max)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for idx, seq0 in enumerate(candidates):
        if time.time() > deadline - 30:
            break
        seq_opt, val_opt = _exp_gradient_opt_multi(seq0, n, deadline,
                                                   max_iters=100_000,
                                                   eta0=3.0)
        print(f"[Init {idx}] EG surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # Fallback in the very unlikely case that nothing succeeded
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 3. Fine optimisation with continuation Adam (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15:
        seq_fine, val_fine = _adam_opt(best_seq, n, deadline,
                                       beta_schedule=[10, 30, 100, 300,
                                                      1_000, 3_000, 10_000,
                                                      30_000, 100_000],
                                       iters_per_beta=3_000)
        print(f"[Adam] surrogate after fine = {val_fine:.6f}")
        if val_fine < best_val:
            best_val, best_seq = val_fine, seq_fine.copy()

    # ------------------------------------------------------------------
    # 4. Optional projected‑gradient polishing (cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        seq_pg, val_pg = _pgd_opt(best_seq, n, deadline,
                                  max_iters=150_000,
                                  eta0=0.8)
        print(f"[PGD] surrogate after projected‑gradient = {val_pg:.6f}")
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()

    # ------------------------------------------------------------------
    # 5. Optional L‑BFGS smoothing (if SciPy available)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()

    # ------------------------------------------------------------------
    # 6. Final evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Result] evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
