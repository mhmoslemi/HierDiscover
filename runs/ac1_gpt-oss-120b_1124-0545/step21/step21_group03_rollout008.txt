def smooth_grad_step(log_a, a, n, beta, lr):
    conv = _conv_fft(a, a)
    max_conv = conv.max()
    exp_val = np.exp(beta * (conv - max_conv))
    sum_exp = exp_val.sum()
    w = exp_val / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    # Project gradient to tangent of simplex under log parameterization
    dot = np.dot(grad_a, a)
    grad_log = a * (grad_a - dot)
    # update log_a
    log_a -= lr * grad_log
    # map back
    a = np.exp(log_a)
    a /= a.sum()
    return a, log_a
```

def adam_optimize(log_a, a, n, beta, num_iters, lr):
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, num_iters+1):
        # compute gradient
        # ... as above to get grad_log
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr_t = lr / (np.sqrt(v_hat) + eps)
        log_a -= lr_t * m_hat
        # renorm
        a = np.exp(log_a)
        a_sum = a.sum()

            log_a = np.log(a)
        else:
            a /= a_sum
        # maybe early break if time exceeded
    return a, log_a
```

def _smooth_objective_and_gradient(a, n, beta):
    conv = _conv_fft(a, a)
    max_c = conv.max()
    exp_beta = np.exp(beta * (conv - max_c))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    # gradient w.r.t log-prob after renorm
    dot = np.dot(grad_a, a)
    grad_log = a * (grad_a - dot)
    return obj, grad_log
```

def _adam_optimize(start_a, beta_schedule, deadline, max_total_iters):
    a = start_a.copy()
    # ensure positive
    a = np.maximum(a, 1e-30)
    log_a = np.log(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    for beta in beta_schedule:
        # determine iterations per beta based on total allowed time
        iters = 2000  # or dynamic
        for i in range(iters):
            if time.time()>deadline:
                return a
            t += 1
            obj, grad = _smooth_objective_and_gradient(a, len(a), beta)
            # update moments
            m = beta1 * m + (1-beta1) * grad
            v = beta2 * v + (1-beta2) * (grad*grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr_t = 0.5 / (np.sqrt(v_hat) + eps)   # base lr normalized
            log_a -= lr_t * m_hat
            # compute a
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / len(a))
                log_a = np.log(a)
                continue
            a /= a_sum
    return a

def _mass_move_opt(seq, deadline, max_iters):
    n = len(seq)
    cur = seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val
    T_start = cur_val
    T_end = cur_val*1e-6
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        T = T_start + (T_end - T_start)*(it/max_iters)
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i==j: continue
        # propose move
        delta = cur[i] * (np.random.rand()*0.2)  # up to 20% of mass at i
        if delta <= 0: continue
        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta
        # ensure non-negativity
        if ns[i] < 0:
            ns[i] = 0.0
        # compute surrogate
        val = _objective_surrogate(ns)
        if val < cur_val or (T>0 and np.random.rand() < math.exp(-(val-cur_val)/T)):
            cur = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val
    return best, best_val

"""
Propose a step‑function (non‑negative sequence) minimizing

    evaluate_sequence(sequence)

The algorithm expands on the previous optimiser:

1. **Extensive seed search** – analytic families (uniform,
   exponential, symmetric‑exponential, power‑law, beta, Dirichlet,
   random piecewise‑linear) are sampled densely.  Every candidate is
   ℓ₁‑normalised and the cheap surrogate
   `2·n·max(conv(seq,seq))` is evaluated; the best few seeds are kept.

2. **Two‑phase refinement** – for each top seed we first run a long
   exponentiated‑gradient (mirror‑descent) optimisation on the exact
   max‑convolution, then a smooth‑max descent (log‑sum‑exp) with a
   schedule of increasing smoothing parameters (β).  Both updates work
   in the log‑softmax space, guaranteeing non‑negativity and unit
   ℓ₁‑norm.

3. **Permutation‑based simulated annealing** – a short but aggressive
   swap‑based annealing removes any remaining ordering artefacts.

4. **Final polishing** – the globally best sequence undergoes a final
   long EG run and a brief mass‑transfer annealing.

The whole procedure respects the 1000 s wall‑time budget and stores the
best sequence in the global variable ``height_sequence_1`` for possible
continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    The iterate lives in the log‑softmax (probability) space.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate) with simple decay.
# ----------------------------------------------------------------------
def _smooth_descent_opt(start_seq: np.ndarray,
                        deadline: float,
                        beta_schedule: list[float] | None = None,
                        iters_per_beta: int = 500) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        2·n·(max_c + log Σ exp(β·(c−max_c))/β)
    where c = conv(a,a).  Operates in the log‑softmax space.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    a = np.maximum(start_seq, 1e-30).astype(np.float64)
    a /= a.sum()
    log_a = np.log(a + 1e-30)

    n = len(a)
    iteration = 0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            iteration += 1

            # ---- smooth objective & gradient w.r.t. a ----
            conv = _conv_fft(a, a)
            max_c = conv.max()
            exp_beta = np.exp(beta * (conv - max_c))          # shift for stability
            sum_exp = exp_beta.sum()
            # smooth objective (unused except for possible debugging)
            # obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

            w = exp_beta / sum_exp                               # soft‑max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]                  # ∂obj/∂a

            # Project gradient onto simplex tangent (log‑softmax domain)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # ---- learning‑rate schedule (≈ 1/√t) ----
            lr = 0.5 / math.sqrt(iteration)

            # ---- update log‑probabilities ----
            log_a -= lr * grad_log
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = _uniform(n)
                log_a = np.log(a)
            else:
                a /= a_sum

            # track best surrogate value
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Turn analytic families into a modest seed pool (dense sampling)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator,
                    previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # basic families
    seeds.append(_uniform(n))

    # exponential families (log‑spaced rates)
    for rate in np.logspace(-4, -1, num=12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law
    for p in np.linspace(0.2, 4.0, num=12):
        seeds.append(_power(n, p))

    # beta families – systematic grid plus a few random draws
    for a in np.linspace(0.2, 3.0, num=9):
        for b in np.linspace(0.2, 3.0, num=5):
            seeds.append(_beta_shape(n, a, b))
    for _ in range(20):
        a = rng.uniform(0.1, 3.5)
        b = rng.uniform(0.1, 3.5)
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet (various concentrations)
    for conc in (0.02, 0.1, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random piecewise‑linear shapes
    for nb in (3, 6, 9):
        for _ in range(6):
            seeds.append(_piecewise_linear(n, nb, rng))

    # include previous best and a few permutations of it
    if previous is not None and previous.shape == (n,):
        prev = np.maximum(previous, 0.0)
        total = prev.sum()
        if total > 0:
            prev = prev / total
            seeds.append(prev)
            for _ in range(5):
                seeds.append(np.random.permutation(prev))

    # normalise & filter
    final = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        total = s.sum()
        if total <= 0.0:
            continue
        s = s / total
        final.append(s)
    return final

    """

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin

    # --------------------------------------------------------------
    # problem size – 4000 balances quality and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    prev_global = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev_global, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev_global, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = arr
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Build a rich pool of seeds
    # --------------------------------------------------------------
    print("[Stage 1] generating seed pool …")
    seed_pool = _generate_seeds(n, rng, prev_vec)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # --------------------------------------------------------------
    evaluated: list[tuple[np.ndarray, float]] = []
    for vec in seed_pool:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Main refinement loop (EG → smooth → permutation)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate time slices – give the very best seed a larger budget
    remaining = max(0.0, deadline - time.time())
    if top_k > 0:
