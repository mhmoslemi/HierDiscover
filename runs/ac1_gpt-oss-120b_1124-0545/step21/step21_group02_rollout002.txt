def project_simplex(v):
    # Project onto {x >=0, sum(x)=1}
    # Sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = softmax(log_a)
    best_val = surrogate
    t = 0
    while time.time() < deadline:
        t += 1
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
    return best_seq, best_val

"""
Propose a step‑function (non‑negative sequence that sums to 1) with a very small
max‑convolution value.  The algorithm is a multi‑start stochastic mirror‑descent
using Adam on the log‑probability (soft‑max) parametrisation, followed by a
smooth‑max L‑BFGS polishing stage if time permits.

The objective that is optimised is the surrogate
    J(a) = 2·n·max(conv(a,a))          (a_i ≥ 0, Σ a_i = 1)
which is exactly the value returned by `evaluate_sequence` up to the harmless
scale‑invariance of the problem.  The optimisation works entirely on the
simplex, the constraints are enforced by the soft‑max mapping.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (mirrors the simplex ⇢ ℝⁿ) and surrogate objective
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact 2·n·max(conv(seq,seq)) when Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimisation in log‑probability space (mirror‑descent)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   base_lr: float = 0.8,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the exact max‑convolution.
    Parameters
    ----------
    start_seq : np.ndarray, shape (n,)
        Probability vector (non‑negative, sums to 1).
    n : int
        Length of the vector.
    deadline : float
        Unix time when the optimisation must stop.
    base_lr : float
        Initial learning‑rate multiplier (scaled by 1/√t internally).
    Returns
    -------
    best_seq, best_val
        The best probability vector seen and its surrogate value.
    """
    # initialise log‑variables
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    # current state
    a = _softmax(log_a)
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    t = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) compute convolution and active index
        # ------------------------------------------------------------------
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # 2) sub‑gradient of the active lag (mirror‑descent formulation)
        # ------------------------------------------------------------------
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. log‑probabilities (chain rule through softmax)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # ------------------------------------------------------------------
        # 3) Adam update in log‑space
        # ------------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr / math.sqrt(t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # 4) map back to the simplex and evaluate
        # ------------------------------------------------------------------
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional smooth‑max L‑BFGS polishing (already used in the reference code)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt  # type: ignore
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise a smooth approximation of the max‑convolution (log‑sum‑exp)
    using L‑BFGS‑B. Operates in log‑softmax space.
    If `scipy.optimize` is unavailable the input vector is returned unchanged.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # translate to the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The strategy:

    1. Gather a handful of diverse initial probability vectors
       (previous best, exponential decay, uniform, sparse Dirichlet).
    2. For each start run Adam‑mirror‑descent on the exact max‑convolution
       (sub‑gradient of the active lag).  The runs share the total time budget.
    3. Keep the best sequence seen across all starts.
    4. If time permits, polish the best with the smooth‑max L‑BFGS routine.
    5. Return the final sequence (list of floats) and store it globally for
       the next invocation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final `evaluate_sequence` call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the previous run.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Assemble a collection of start vectors (all sum to 1)
    # ------------------------------------------------------------------
    starts = []

    # Previous solution, if any
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
        seq0 = np.maximum(seq0, 0.0)
        total = seq0.sum()
        if total > 0.0:
            starts.append(seq0 / total)

    # Exponential decay (deterministic rate)
    i = np.arange(n, dtype=np.float64)
    rate = 0.003
    exp_seq = np.exp(-rate * i)
    starts.append(exp_seq / exp_seq.sum())

    # Uniform distribution
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Sparse Dirichlet draws (α < 1 → many near‑zeros)
    for alpha in (0.2, 0.5):
        gam = rng.gamma(shape=alpha, scale=1.0, size=n)
        if gam.sum() > 0:
            starts.append(gam / gam.sum())

    # Random exponential with random rate
    rand_rate = rng.uniform(0.001, 0.01)
    rand_exp = np.exp(-rand_rate * i)
    starts.append(rand_exp / rand_exp.sum())

    # ------------------------------------------------------------------
    # 2. Run Adam optimisation on each start (time shared)
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float('inf')
    n_starts = len(starts)
    # allocate roughly equal time to each start
    time_per_start = (deadline - time.time()) / max(1, n_starts)

    print(f"[Info] {n_starts} start vectors, ~{time_per_start:.1f}s per start.")

    for idx, seq in enumerate(starts):
        start_deadline = time.time() + time_per_start
        if time.time() >= deadline:
            break
        # Adam optimisation
        seq_opt, val_opt = _adam_optimize(seq, n, start_deadline,
                                          base_lr=0.8)
        print(f"[Adam] start {idx+1}/{n_starts}: best surrogate = {val_opt:.6f}")

        if val_opt < best_overall_val:
            best_overall_val = val_opt
            best_overall_seq = seq_opt.copy()

    if best_overall_seq is None:
        # fallback – very unlikely
        best_overall_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_overall_val = _objective_surrogate(best_overall_seq)

    # ------------------------------------------------------------------
    # 3. Optional L‑BFGS polishing (smooth‑max) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_overall_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after polishing = {val_lb:.6f}")
        if val_lb < best_overall_val:
            best_overall_seq = seq_lb
            best_overall_val = val_lb

    # ------------------------------------------------------------------
    # 4. Final evaluation with the provided secure function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"[Result] surrogate = {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # Store for the next run (incremental improvement)
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
