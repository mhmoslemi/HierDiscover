def _exp_gradient_multi_active(start_seq, n, deadline, max_iters=500_000, eta0=5.0, active_tol=1e-6):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a,a)
        max_val = conv.max()
        eps = max_val * active_tol
        active = np.where(conv >= max_val - eps)[0]
        # compute subgradient as sum over active
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]

        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
propose_candidate – hybrid LP / mirror‑descent optimizer

The routine builds a moderate set of analytic seeds (uniform,
triangular, exponential, power‑law, Gaussian, Kaiser, logistic,
Dirichlet draws).  All seeds are ℓ₁‑normalised.  The best few seeds are
refined by three stages:

1. **Linear‑program improvement** – for a candidate `a` we solve
   ``max 1·b  s.t. a∗b ≤ M`` (where ``M = max(a∗a)``) using a dense
   Toeplitz matrix and SciPy’s HiGHS LP solver.  The optimal vector
   `b` is scaled back to unit sum, which provably reduces the maximal
   convolution if ``sum(b) > 1``.

2. **Exponentiated‑gradient (mirror descent)** on the exact
   ``max(a∗a)`` using a *multi‑active* sub‑gradient (all convolution
   entries within a tiny tolerance of the maximum).  This is cheap
   (one FFT per iteration) and runs for many‐hundred‑thousands of
   steps.

3. **Continuation L‑BFGS** on a smooth log‑sum‑exp surrogate of the
   max, followed by a short simulated‑annealing polish.

All stages respect the global time budget (default 1000 s) and keep
the sequence non‑negative with ℓ₁‑norm 1.  The best sequence found is
saved in the global variable ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP, L‑BFGS, Kaiser window)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:                               # pragma: no cover
    _opt = None

try:
    from scipy.linalg import toeplitz
except Exception:                               # pragma: no cover
    toeplitz = None

try:
    from scipy.special import i0 as _i0
except Exception:                               # pragma: no cover
    _i0 = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window; falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Linear‑program improvement (primal formulation)
# ----------------------------------------------------------------------
def _lp_improve(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 1,
                tol: float = 1e-9):
    """
    One‑step (or few‑step) LP improvement:

    maximise   1·b
    subject to  a∗b ≤ M   (full linear convolution)
               b ≥ 0

    The optimal b is scaled back to unit ℓ₁‑norm, which yields a
    strictly smaller maximal convolution whenever sum(b) > 1.
    """
    if _opt is None or toeplitz is None:
        # SciPy unavailable – report no improvement.
        return start_seq.copy(), _objective_surrogate(start_seq)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()

        # Build Toeplitz matrix T such that T·b = a∗b (full convolution)
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
        T = toeplitz(col, row)          # shape (2n‑1, n)

        c = -np.ones(n, dtype=np.float64)           # minimise –sum(b)
        ub = M * np.ones(2 * n - 1, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        res = _opt.linprog(c,
                           A_ub=T,
                           b_ub=ub,
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True, 'time_limit': max(0, deadline - time.time())})

        if not res.success:
            # Infeasible or other failure – abort LP stage.
            break

        b = res.x
        s = b.sum()

        if s <= 1.0 + tol:
            # No improvement possible.
            break

        a = b / s
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    return best_seq, best_val

def _exp_gradient_multi_active(start_seq: np.ndarray,

                               deadline: float,
                               max_iters: int = 300_000,
                               eta0: float = 6.0,
                               active_tol: float = 1e-8):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Uses all indices whose convolution value is within ``active_tol`` of the
    current maximal value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * active_tol
        active = np.where(conv >= max_val - eps)[0]

        # Sub‑gradient (averaged over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 800):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                       # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                 # weights over each conv index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 5e-8):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1``, and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a balance between expressivity and speed
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Load a previous solution if present (helps continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – diverse analytic families
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_exp_decay(n, rate=0.004))
    seeds.append(_power(n, p=1.5))
    seeds.append(_power(n, p=3.0))
    seeds.append(_logistic_shape(n, scale=2.0, midpoint=n * 0.35))
    seeds.append(_logistic_shape(n, scale=2.0, midpoint=n * 0.65))

    for beta in [2.0, 4.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    for conc in [0.002, 0.01, 0.05, 0.2]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(4):
            pert = base + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement of the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of remaining budget for this seed
        remaining = deadline - time.time()
        # roughly 12 % of the remaining time per remaining seed
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - si))

        print(f"\n[Seed {si+1}/{top_k}] start LP improvement")
        seq_lp, val_lp = _lp_improve(seed_vec.copy(),
                                     n,
                                     sub_deadline,
                                     max_iters=3)   # a few LP passes

        # exponentiated‑gradient (multi‑active)
        if time.time() < sub_deadline:
            print("[Seed] EG refinement (multi‑active)")
            seq_eg, val_eg = _exp_gradient_multi_active(seq_lp.copy(),
                                                        n,
                                                        sub_deadline,
                                                        max_iters=300_000,
                                                        eta0=8.0,
                                                        active_tol=1e-7)
        else:
            seq_eg, val_eg = seq_lp.copy(), val_lp

        # smooth L‑BFGS refinement
        if time.time() < sub_deadline:
            print("[Seed] L‑BFGS continuation")
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               max_iter_per_beta=1000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # short SA polish
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            print("[Seed] SA polish")
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=80_000,
                                                  start_sigma=0.002,
                                                  T_start=0.012,
                                                  T_end=1e-8)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among the candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_lp, val_lp),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run polishing (EG → SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (multi‑active) ---")
        final_seq, final_val = _exp_gradient_multi_active(global_best_seq,
                                                          n,
                                                          deadline,
                                                          max_iters=5_000_000,
                                                          eta0=6.0,
                                                          active_tol=1e-8)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.001,
                                                         T_start=5e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
