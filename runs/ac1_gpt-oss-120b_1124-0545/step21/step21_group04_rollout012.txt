def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1 # margin

    n = 4000 # can be changed

    # Load previous best
    prev = globals().get("height_sequence_1")
    candidates = []
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.sum() > 0:
                    arr = arr/arr.sum()
                else:
                    arr = _uniform(n)
                candidates.append(arr)
        except Exception:
            pass

    # analytic families
    candidates.append(_uniform(n))
    for rate in [0.001, 0.003, 0.01, 0.05]:
        candidates.append(_exp_decay(n, rate))
    for p in [0.0, 0.2, 0.5, 1.0, 1.5, 2.0]:
        candidates.append(_power_law(n, p))
    # cosine power
    for p in [0.5, 1.0, 2.0, 4.0]:
        candidates.append(_cosine_power(n, p))

    # beta distribution (U shape)
    for a in [0.2,0.4,0.6,0.8]:
        for b in [0.2,0.4,0.6,0.8]:
            candidates.append(_beta_distribution(n, a, b))

    # random Dirichlet with various alphas
    alphas = [0.05,0.1,0.2,0.5,1.0,2.0,5.0]
    for alpha in alphas:
        for _ in range(3):
            candidates.append(rng.dirichlet(np.full(n, alpha)))

    # random shuffled version of uniform (i.e., permutation)
    perm = rng.permutation(n)
    candidates.append(_uniform(n)[perm])

    # evaluate all candidates quickly (surrogate)
    evals = []
    for idx, seq in enumerate(candidates):
        val = _objective_surrogate(seq)
        evals.append((val, seq))
    evals.sort(key=lambda x: x[0])
    top_k = 10
    top_candidates = [seq for (_, seq) in evals[:top_k]]
    best_seq = top_candidates[0].copy()
    best_val = _objective_surrogate(best_seq)

    # Now exponentiated gradient refinement on each top candidate but limited total iterations

    remaining = deadline - time.time()
    # allocate at most 0.6*budget for EG

    # Instead use fixed 200k iterations per candidate but break if time exceed.

    for i, seq0 in enumerate(top_candidates):

        # use exponentiated gradient long run
        seq_opt, val_opt = _exp_gradient_opt_multi(seq0, n, deadline,

        print(f"[EG refine {i}] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # Optional L-BFGS if SciPy present
    if time.time() < deadline - 5 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()

    # Simulated annealing refinement
    # We'll run until deadline
    cur_seq = best_seq.copy()
    cur_val = best_val
    step_factor = 0.1 # initial fraction of mass move
    temp = 0.01 # temperature parameter for acceptance
    # schedule: decrease step_factor and temperature over time linearly
    while time.time() < deadline - 0.5:

        new_seq = cur_seq.copy()

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

"""
Propose a step‑function (non‑negative sequence) that minimizes the
``evaluate_sequence`` objective.

The algorithm is a multi‑stage heuristic:

1. **Generate a diverse pool of start vectors** – analytic families
   (uniform, exponential, power‑law, cosine‑U, Beta‑U) together with many
   random Dirichlet draws of different concentration.  All vectors are
   L1‑normalised to 1.

2. **Select the most promising few** (lowest surrogate value
   ``2·n·max(conv)``).

3. **Short mirror‑descent (exponentiated‑gradient) polish** of each of
   those candidates.

4. **Long mirror‑descent on the current best** – the loop stops
   automatically when the global deadline is reached.

5. **Optional smooth‑max L‑BFGS refinement** (uses SciPy if available).

6. **Simulated‑annealing local search** that moves a small amount of mass
   between two random positions; downhill moves are always accepted,
   uphill moves are accepted with a temperature‑dependent probability.
   Both the step size and the temperature decay over time.

7. Evaluate the final sequence with the supplied ``evaluate_sequence``,
   store it globally as ``height_sequence_1`` and return it.

All heavy work is done with FFT‑based convolution, which makes a single
objective evaluation O(n log n) with a small constant factor.
The function respects the 1000 s budget by checking the deadline at
regular intervals.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (∑=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s != 0.0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _cosine_power(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shape constructed from a cosine:
    a_i ∝ (1 + cos(2π·i/(n‑1)))**p   (high values at the two ends,
    low in the centre).
    """
    i = np.arange(n, dtype=np.float64)
    w = (1.0 + np.cos(2.0 * np.pi * i / (n - 1))) ** p
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on the points (i+0.5)/n,  i=0..n‑1.
    For α,β<1 this yields a U‑shaped distribution.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid NaNs for α<1 or β<1 at the boundaries
    w = np.where(
        (x > 0) & (x < 1),
        np.power(x, alpha - 1) * np.power(1 - x, beta - 1),
        0.0,
    )
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, ∑a = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance: a few ULPs are enough
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    if len(active) > 0:
        g /= len(active)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a decaying step size η_t = η₀ / √t.
    Returns the best sequence encountered during the run and its surrogate.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Euclidean projected sub‑gradient descent."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Smooth‑max L‑BFGS‑B optimisation (requires SciPy)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing local search (mass‑exchange moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_step: float = 0.10,
                         init_temp: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly moves a random fraction of mass from one
    index to another.  Down‑hill moves are always accepted; uphill moves
    are accepted with probability exp(-Δ/Temp).  Both step size and
    temperature decay exponentially with the elapsed fraction of the
    time budget.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    start = time.time()
    total_budget = deadline - start

    while time.time() < deadline - 0.2:
        # fraction of time already spent (0 → 1)
        elapsed = (time.time() - start) / total_budget

        # exponential decay of step size and temperature
        step = max(1e-5, init_step * (0.5 ** elapsed))
        temp = max(1e-7, init_temp * (0.7 ** elapsed))

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a small amount ε from i to j
        max_eps = cur_seq[i]
        if max_eps <= 0.0:
            continue
        eps = step * max_eps * rng.random()
        if eps <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        new_val = _objective_surrogate(new_seq)

        if new_val < cur_val:
            cur_seq, cur_val = new_seq, new_val
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_val - cur_val) / temp):
                cur_seq, cur_val = new_seq, new_val

    return cur_seq, cur_val

            break
        seq_opt, val_opt = _exp_gradient_opt_multi(
            seq0, n, deadline, max_iters=80_000, eta0=5.0)
        print(f"[Short EG {idx}] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient refinement of the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 40:
        seq_fine, val_fine = _exp_gradient_opt_multi(
            best_seq, n, deadline, max_iters=2_500_000, eta0=5.0)
        print(f"[Long EG] surrogate = {val_fine:.6f}")
        if val_fine < best_val:
            best_val, best_seq = val_fine, seq_fine.copy()

    # ------------------------------------------------------------------
    # 5. Optional L‑BFGS smoothing (if SciPy is present)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing local‑search refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              init_step=0.12, init_temp=0.012)
        print(f"[SA] surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()

    # ------------------------------------------------------------------
    # 7. Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
