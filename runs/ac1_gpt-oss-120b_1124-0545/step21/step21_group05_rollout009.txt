    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = a.copy()

    t=0

        # compute conv and max idx

        max_idx = int(np.argmax(conv))
        # compute gradient g
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)

        g[idx] = 2.0 * a[rev]

        dot = np.dot(a, g)
        grad_log = a * (g - dot)

        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val

"""
Optimise a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The strategy is:

  1.  Build many candidate sequences:
        – a few analytic families (uniform, exponential, Gaussian, …)
        – a large batch of random monotone vectors (sorted random numbers)
        – Dirichlet‑style draws with various concentration levels.
      All candidates are ℓ₁‑normalised.

  2.  Evaluate the cheap surrogate 2·n·max(conv(a,a)) for every candidate,
      keep the best few (default 6).

  3.  Refine each of those seeds with a *mirrored‑gradient* optimiser
      (Adam on the log‑weights) that works on the exact max‑convolution.
      The optimiser runs for a time‑budget that is shared among the seeds.

  4.  A final long Adam run on the currently best sequence uses all
      remaining time.

  5.  As a last check the monotone‑sorted version of the final sequence
      is also examined – the better of the two is returned.

The optimiser works directly on the `log`‑parameterisation (softmax ↔ probability
simplex), guaranteeing non‑negativity and unit sum.  Adam’s adaptive step‑size
yields faster convergence than the plain exponentiated‑gradient used earlier.
The whole routine respects the 1000 s overall budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    Returned length = len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for sum == 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a vector with sum == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (mirror‑descent in log‑space)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr: float = 0.55,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities (softmax) to minimise
    2·n·max(conv(a,a)).  Returns the best sequence seen and its surrogate.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)               # initialise log‑softmax argument
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * np.max(conv))

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # convolution and index of the active (max) entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        #   g_i = 2*a_{max_idx-i} for i in the overlap interval
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = max_idx - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[partner]

        # gradient w.r.t. log‑softmax parameters (mirror‑descent form)
        dot = np.dot(a, g)
        grad_log = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to the simplex
        a = _softmax(log_a)

        # surrogate value for this iterate
        cur_val = float(2.0 * n * np.max(conv))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    `evaluate_sequence`.  Returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin of a few seconds so we always have time for the final eval
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a good compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1.  Build a huge pool of candidate vectors (all ℓ₁‑normalised)
    # ------------------------------------------------------------------
    candidates = []

    # Analytic families – a few representative parameters
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.001))
    candidates.append(_exp_decay(n, rate=0.01))
    candidates.append(_exp_decay(n, rate=0.05))
    candidates.append(_gaussian(n, sigma_frac=0.10))
    candidates.append(_gaussian(n, sigma_frac=0.18))
    candidates.append(_beta_shape(n, alpha=0.5, beta=5.0))
    candidates.append(_beta_shape(n, alpha=2.0, beta=2.0))

    # Random monotone vectors – sorted uniform samples
    num_random = 2000
    for _ in range(num_random):
        # generate unsorted random numbers, sort descending, normalise
        vec = np.sort(rng.random(n))[::-1]
        s = vec.sum()
        if s == 0.0:
            continue
        candidates.append(vec / s)

    # Dirichlet‑style draws with several concentration parameters
    conc_vals = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 80
    for conc in conc_vals:
        for _ in range(draws_per_conc):
            # Dirichlet is equivalent to normalising independent Gamma draws
            vec = rng.gamma(shape=conc, scale=1.0, size=n)
            # enforce monotonicity (helps with the centre peak)
            vec = np.sort(vec)[::-1]
            s = vec.sum()
            if s == 0.0:
                continue
            candidates.append(vec / s)

    # ------------------------------------------------------------------
    # 2.  Quick surrogate evaluation – keep the top K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in candidates:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3.  Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refinement of each top seed with Adam (time‑shared)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate an equal share of the *remaining* time to this seed
        remaining = deadline - time.time()
        # reserve a fraction (≈30 %) for the final polishing run
        share = remaining * 0.70 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # ensure the seed starts monotone decreasing (often helpful)
        seed_desc = np.sort(seed_vec)[::-1]

        seq_opt, val_opt = _adam_opt(seed_desc, n, sub_deadline,
                                     max_iters=250_000, lr=0.55)

        print(f"[Seed {si+1}] best surrogate after Adam = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final long Adam polishing on the overall best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final polishing (full remaining time) ---")
        seq_final, val_final = _adam_opt(global_best_seq, n, deadline,
                                         max_iters=3_000_000, lr=0.50)
        if val_final < global_best_val:
            global_best_seq, global_best_val = seq_final, val_final
            print(f"[Final] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Check the sorted‑descending version (often very competitive)
    # ------------------------------------------------------------------
    sorted_desc = np.sort(global_best_seq)[::-1]
    val_desc = _objective_surrogate(sorted_desc)
    if val_desc < global_best_val:
        global_best_seq = sorted_desc
        global_best_val = val_desc
        print(f"[Sorted] better surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7.  Real evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
