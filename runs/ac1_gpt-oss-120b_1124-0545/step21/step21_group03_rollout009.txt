"""
Simulated‑annealing refinement of a step‑function.

The function loads the previously best known sequence (global
``height_sequence_1``) – if none is found it falls back to a uniform
distribution – and then performs a dedicated annealing phase that
transfers mass between two positions while keeping the ℓ₁‑norm equal
to one.  The convolution is updated *exactly* by an O(n) slice
operation, so millions of cheap moves can be tried inside the 1000 s
budget.  The temperature and the maximal transferred fraction are
cooled linearly, allowing large exploratory moves early and fine
tuning later.  After the annealing we optionally run a short L‑BFGS
smooth‑max optimisation (if SciPy is available) and finally evaluate
the sequence with the provided ``evaluate_sequence`` routine.

The returned list is stored in ``height_sequence_1`` for possible
continuation runs.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (used by the FFT convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution via real FFT (O(n log n))
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max optimiser (optional, needs SciPy)
# ----------------------------------------------------------------------
try:                                   # pragma: no cover
    import scipy.optimize as _opt
except Exception:                     # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Refine ``start_vec`` (ℓ₁‑normalised) by minimising a
    log‑sum‑exp smooth approximation of the maximum convolution.
    """
    if _opt is None:
        return start_vec

    # schedule of smoothness parameters (larger → closer to true max)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0, 20_000.0,
        50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]

    # work in log‑softmax space to enforce non‑negativity and ℓ₁=1
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max weights)
            w = exp_beta / sum_exp                     # (2n‑1)‑vector
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length = 2n‑1
            grad_a = 2.0 * full[:len(a)]

            # back‑prop to log‑softmax variables
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method loads the previous best sequence,
    then performs a large‑scale simulated‑annealing on the ℓ₁‑normalised
    vector (mass‑transfer moves) while keeping the convolution exact via
    in‑place slice updates.  Optionally a short L‑BFGS fine‑tuning is
    applied before the final evaluation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # Load previous solution (if any) or fall back to uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0:
            a = arr.copy()
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.shape[0]

    # normalise to ℓ₁ = 1 and enforce non‑negativity
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total

    # ------------------------------------------------------------------
    # Initial convolution and current best record
    # ------------------------------------------------------------------
    conv = _conv_fft(a, a)                     # length = 2n‑1
    cur_max = conv.max()
    best_seq = a.copy()
    best_max = cur_max

    print(f"[Init] n={n}, initial surrogate = {2.0 * n * cur_max:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing on the probability simplex
    # ------------------------------------------------------------------
    # temperature schedule (linear)
    T_start = cur_max * 0.3   # start relatively warm
    T_end   = cur_max * 1e-5  # finish almost frozen

    # maximal transferred fraction (large → coarse, small → fine)
    frac_start = 0.5
    frac_end   = 0.02

    accepted = 0
    it = 0
    # we will periodically recompute the convolution from scratch
    recompute_every = 5000

    while time.time() < deadline:
        # coverage fraction of the total allowed time
        elapsed = time.time() - start_time
        total_allowed = deadline - start_time
        fraction = min(elapsed / total_allowed, 1.0)

        # linear cooling for both temperature and step size
        T = T_start * (1.0 - fraction) + T_end * fraction
        max_frac = frac_start * (1.0 - fraction) + frac_end * fraction

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        # choose a transfer amount δ (a[i] is the source)
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # keep a copy for possible reversion
        a_before = a.copy()
        conv_before = conv  # we will modify in‑place, keep a reference

        # apply the mass transfer
        a[i] -= delta
        a[j] += delta

        # ----- exact convolution update (O(n)) -----
        # linear part   – 2·δ·(shift_j – shift_i)
        conv[i:i + n] -= 2.0 * delta * a_before
        conv[j:j + n] += 2.0 * delta * a_before
        # quadratic part – δ² at 2i and 2j, –2δ² at i+j
        d2 = delta * delta
        conv[2 * i] += d2
        conv[2 * j] += d2
        conv[i + j] -= 2.0 * d2

        new_max = conv.max()

        # ---- Metropolis acceptance test ----
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ) / T)
            prob = math.exp(-(new_max - cur_max) / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            accepted += 1
            # record a new global best if improvement
            if new_max < best_max:
                best_max = new_max
                best_seq = a.copy()
                print(f"[Iter {it:>7}] new best surrogate = {2.0 * n * best_max:.6f}")
        else:
            # reject – revert both a and conv
            a[:] = a_before
            # undo the convolution update (reverse the signs)
            conv[i:i + n] += 2.0 * delta * a_before
            conv[j:j + n] -= 2.0 * delta * a_before
            conv[2 * i] -= d2
            conv[2 * j] -= d2
            conv[i + j] += 2.0 * d2

        it += 1

        # occasional full recomputation to avoid drift from rounding errors
        if accepted > 0 and accepted % recompute_every == 0:
            conv = _conv_fft(a, a)
            cur_max = conv.max()

    # ------------------------------------------------------------------
    # Optional L‑BFGS polishing (smooth‑max) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        try:
            seq_lb = _lbfgs_smooth_optimize(best_seq, deadline)
            conv_lb = _conv_fft(seq_lb, seq_lb)
            val_lb = 2.0 * n * conv_lb.max()
            if val_lb < 2.0 * n * best_max:
                best_seq = seq_lb
                best_max = conv_lb.max()
                print(f"[L‑BFGS] improved surrogate = {val_lb:.6f}")
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] skipped due to exception:", e)

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {2.0 * n * best_max:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose the best sequence for possible continuation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
