"""
Optimiser for the step‑function attack.

We search for a non‑negative vector a with Σa = 1 that minimises

    E(a) = 2·n·max(conv(a, a))

where conv is the (linear) convolution.  The algorithm proceeds in three
phases:

1. **Seed generation** – a set of analytically‑parameterised probability
   vectors (uniform, triangular, sine‑window, Gaussian, shifted‑Gaussian,
   exponential, symmetric‑exponential, power‑law, Beta, random Dirichlet)
   is built and the best few (by the exact surrogate) are kept.

2. **Local optimisation** – each seed is refined with a bespoke Adam
   optimiser that works in log‑softmax space (mirror‑descent).  The optimiser
   minimises the true nonsmooth objective via sub‑gradients of the active
   convolution entry.  The per‑seed budget is a fraction of the remaining
   time; the best local result becomes the global best.

3. **Final polishing** – the current global best is polished for the rest
   of the time with the same Adam optimiser (more iterations, smaller step).

All heavy work uses an FFT‑based convolution (O(n log n)).  No file or
network access is performed.  The final sequence is stored in the global
variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (efficient for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ₁‑normalised vector:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _norm_and_surrogate(vec: np.ndarray):
    """Clip to non‑negative, normalise to sum = 1 and return the surrogate."""
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return None, np.inf
    vec = vec / s
    return vec, _objective_surrogate(vec)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient in log‑softmax space)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum for the nonsmooth objective
    2·n·max(conv(a, a)).  The parameter vector is the log‑softmax (logits)
    to keep feasibility (a ≥ 0, Σa = 1).  Returns the best sequence seen and
    its surrogate value.
    """
    a = np.copy(start_seq)
    n = len(a)
    log_a = np.log(a + 1e-30)               # initialise logits
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # helper: sub‑gradient of the active convolution entry w.r.t. a
    def _subgrad(a_vec: np.ndarray, max_idx: int) -> np.ndarray:
        g = np.zeros_like(a_vec)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a_vec[rev_idx]
        return g

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute convolution and locate the active diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient w.r.t. a
        g_a = _subgrad(a, max_idx)

        # transform to gradient w.r.t. logits (mirror‑descent identity)
        dot = np.dot(g_a, a)
        g_logits = a * (g_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_logits
        v = beta2 * v + (1.0 - beta2) * (g_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        # occasional evaluation of the surrogate (every few thousand iters)
        if it % 5000 == 0 or it == max_iters:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

    The routine builds a modest set of analytic seeds, refines each with an
    Adam optimiser (sub‑gradient of max‑convolution) and finishes with a
    long polishing run on the current global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – 4000 works well for the time budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Seed generation (deterministic families + few random draws)
    # --------------------------------------------------------------
    seeds = []

    # deterministic families (coarse grids)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.05, 0.30, 20):
        seeds.append(_gaussian(n, sigma_frac))

    for mu_frac in np.linspace(0.15, 0.85, 8):
        for sigma_frac in np.linspace(0.05, 0.25, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    for rate in np.logspace(-5, 0.4, 35):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    for p in np.linspace(0.0, 5.0, 20):
        seeds.append(_power(n, p))

    alphas = np.geomspace(0.1, 5.0, 12)
    betas = np.geomspace(0.1, 5.0, 12)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws (different concentrations)
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include a previous solution if present (useful for continuation)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0:
                prev_arr = prev_arr / s
                seeds.append(prev_arr)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec_norm, val = _norm_and_surrogate(vec)
        if vec_norm is None:
            continue
        evaluated.append((vec_norm, val))

    if not evaluated:
        raise RuntimeError("No valid seeds were generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Local refinement of each top seed (Adam optimiser)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a share of the remaining budget (~10 % each)
        remaining = deadline - time.time()
        share = remaining * 0.10 / max(1, (top_k - idx))
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx + 1}] optimisation budget ≈ {share:.1f}s")
        seq_opt, val_opt = _adam_optimize(seed_vec,
                                           sub_deadline,
                                           max_iters=150_000,
                                           lr=0.6)

        print(f"[Seed {idx + 1}] best surrogate after Adam = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_budget = deadline - time.time()
        print("\n--- Final polishing (Adam) ---")
        print(f"Remaining time ≈ {final_budget:.1f}s")
        seq_final, val_final = _adam_optimize(global_best_seq,
                                               deadline,
                                               max_iters=1_500_000,
                                               lr=0.5)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate the true objective (the only call to evaluate_sequence)
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
