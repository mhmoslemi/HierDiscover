def _local_random_search_improved(start_seq, n, deadline):
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    rng = np.random.default_rng()
    total_attempts = 0
    # Use schedule: start with sigma = 0.02, then 0.008, 0.003, 0.001
    sigma_sched = [(0.02, 200_000), (0.008,200_000), (0.003,200_000), (0.001,200_000), (0.0005,200_000)]
    for sigma, max_attempts in sigma_sched:
        attempts = 0
        while attempts < max_attempts and time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = sigma * best_seq[i] * rng.random()
            if delta <= 0.0:
                continue
            new_seq = best_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            if new_seq[i] < 0.0: # could happen due to floating error
                continue
            val = _objective_surrogate(new_seq)
            if val < best_val:
                best_seq = new_seq
                best_val = val
                # Reset attempts for this sigma? maybe continue search
            attempts += 1
            total_attempts += 1
```

def _lbfgs_fun(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    grad_a = grad_a * (2.0 * n)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v
```

"""
Propose a step‑function (non‑negative sequence) that lowers the objective

    2·n·max(conv(seq,seq))  ==  evaluate_sequence(seq)

The algorithm builds a diverse pool of seeds (uniform, arcsine, Beta,
Dirichlet, mixtures of Gaussians) and keeps the best one.  It then runs

* a low‑dimensional differential‑evolution search in a Gaussian‑mixture
  family,
* a limited L‑BFGS optimisation of the smooth log‑sum‑exp surrogate,
* a long exponentiated‑gradient (mirror‑descent) phase that directly
  minimises the exact max‑convolution,
* a multi‑phase random local refinement with decreasing step sizes.

All stages respect the global 1000 s budget; the best sequence found is
stored in the global variable ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for DE and L‑BFGS)
# ----------------------------------------------------------------------
try:                                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:                       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Global used by DE callback (no closures, no lambdas)
# ----------------------------------------------------------------------
DE_DEADLINE = None


def _de_callback(xk, convergence):      # pragma: no cover
    """Terminate DE early when the global deadline is passed."""
    return (DE_DEADLINE is not None) and (time.time() > DE_DEADLINE)


# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (used for optimisation; equals the true objective
# when the sequence sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)).  ``seq`` must already be normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all outputs are normalised to sum‑1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1} (normalised)."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gaussian mixture family (parameterised by K components)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int, K: int = 5) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    Layout (K components):
        w_0 … w_{K‑1}      (non‑negative, then normalised)
        μ_0 … μ_{K‑1}      (in [0,1])
        σ_0 … σ_{K‑1}      (positive, bounded)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """Return (obj, grad) for the log‑sum‑exp surrogate (variables live in log‑space)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # objective = 2·n·(shift + log(sum_exp)/beta)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp                     # weight vector over 2n‑1 entries
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # convolution of rev_a with w
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # d/d a of Σ w·conv
    grad_a = grad_a * (2.0 * n)                # outer factor 2·n

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)                # chain rule through softmax
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimise the smooth surrogate (log‑sum‑exp) using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        try:
            res = _opt.minimize(fun=_lbfgs_smooth_obj_grad,
                                x0=v,
                                args=(n, beta_val),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:   # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient of the true max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_seq, best_val).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Soft‑max mirror descent (smooth max) – custom implementation
# ----------------------------------------------------------------------
def _smooth_mirror_descent(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           iters_per_beta: int = 15000) -> tuple[np.ndarray, float]:
    """
    Gradient descent in log‑softmax space using the smooth
    log‑sum‑exp approximation of the max‑convolution.
    Returns (best_seq, best_val) with respect to the exact surrogate.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    log_a = np.log(start_seq + 1e-30)
    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    it_total = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            exp_beta = np.exp(beta_val * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)                     # weights over 2n‑1 entries

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]          # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)                  # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            it_total += 1
            eta = 0.30 / math.sqrt(it_total)                    # tuned constant
            log_a = log_a - eta * grad_log

            if it_total % 5000 == 0:
                cur_seq = _softmax(log_a)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

    final_seq = _softmax(log_a)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search with decreasing step size
# ----------------------------------------------------------------------
def _local_random_search_multi(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               schedule=None) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.
    ``schedule`` is a list of (sigma, max_attempts) tuples.
    """
    if schedule is None:
        schedule = [(0.02, 200_000),
                    (0.008, 200_000),
                    (0.003, 200_000),
                    (0.001, 200_000),
                    (0.0005, 200_000)]

    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for sigma, max_attempts in schedule:
        attempts = 0
        while attempts < max_attempts and time.time() < deadline:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = sigma * best_seq[i] * rng.random()
            if delta <= 0.0:
                continue
            new_seq = best_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            if new_seq[i] < 0.0:
                continue
            val = _objective_surrogate(new_seq)
            if val < best_val:
                best_seq, best_val = new_seq, val
            attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the overall ``budget_s`` second timeout.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # --------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # Load a previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]
    if prev_arr is not None:
        seeds.append(prev_arr)

    # Dirichlet draws – a wide range of concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Beta draws – random shapes
    for _ in range(40):
        a = 10 ** rng.uniform(-2, 1)      # α in [0.01, 10]
        b = 10 ** rng.uniform(-2, 1)      # β in the same range
        seeds.append(_beta_distribution(n, a, b))

    # Mixtures of two Betas
    for _ in range(12):
        a1 = 10 ** rng.uniform(-2, 1)
        b1 = 10 ** rng.uniform(-2, 1)
        a2 = 10 ** rng.uniform(-2, 1)
        b2 = 10 ** rng.uniform(-2, 1)
        w1 = rng.random()
        w2 = 1.0 - w1
        seq = w1 * _beta_distribution(n, a1, b1) + w2 * _beta_distribution(n, a2, b2)
        seq = seq / seq.sum()
        seeds.append(seq)

    # Slight perturbations of uniform
    for _ in range(4):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.005))

    # --------------------------------------------------------------
    # 1️⃣  Evaluate seeds and keep the best
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Global optimisation in a low‑dimensional Gaussian‑mixture family
    # --------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution (Gaussian mixture) ---")
        global DE_DEADLINE
        DE_DEADLINE = deadline
        Kmix = 10                                   # more components than before
        bounds = [(0.0, 1.0)] * Kmix + [(0.0, 1.0)] * Kmix + [(0.001, 0.5)] * Kmix

        def de_objective(params):
            seq = _mixture_gaussian(params, n, K=Kmix)
            return _objective_surrogate(seq)

        try:
            result = _opt.differential_evolution(
                de_objective,
                bounds,
                seed=int(seed),
                popsize=12,
                maxiter=40,
                polish=False,
                updating='deferred',
                init='latinhypercube',
                workers=1,
                callback=_de_callback,
                disp=False
            )
            seq_de = _mixture_gaussian(result.x, n, K=Kmix)
            val_de = _objective_surrogate(seq_de)
            print(f"[DE] surrogate = {val_de:.6f}")
            if val_de < best_val:
                best_val, best_seq = val_de, seq_de.copy()
                print("   >>> Updated global best (DE)")
        except Exception as e:                       # pragma: no cover
            print("[DE] exception:", e)
        finally:
            DE_DEADLINE = None

    # --------------------------------------------------------------
    # 3️⃣  L‑BFGS smoothing (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing stage ---")
        remain = deadline - time.time()
        sub_deadline = time.time() + remain * 0.45   # allocate ~45 % of remaining time
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                           max_iter_per_beta=20000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("   >>> Updated global best (L‑BFGS)")

    # --------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polishing (uses the true max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 2.0          # keep a tiny margin for final steps
        time_left = polishing_deadline - time.time()
        # Scale iteration budget with the amount of time left (≈ 1 M iters for 1 ks)
        max_iters = int(1_000_000 * time_left / 1000.0)
        max_iters = max(250_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 5️⃣  Final tiny random local refinement with decreasing sigma
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement (multi‑phase) ---")
        seq_local, val_local = _local_random_search_multi(best_seq, n, deadline)
        print(f"[Local] exact surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 6️⃣  Official evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
