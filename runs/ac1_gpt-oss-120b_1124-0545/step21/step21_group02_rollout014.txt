Pseudo:

def smooth_obj_and_grad(logits, beta, n):
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    # compute softmax of scaled conv (beta)
    shift = max_c  # for numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    Z = exp_beta.sum()
    # smooth approximation of max: smooth_max = shift + np.log(Z)/beta
    obj = 2.0 * n * (shift + np.log(Z) / beta)
    # gradient w.r.t a: w = exp_beta / Z (shape 2n-1)
    w_prob = exp_beta / Z
    rev_a = a[::-1]
    # convolution rev_a * w_prob gives something shape len=3n-2; we need indices n-1 .. n-1+n-1
    full = _conv_fft(rev_a, w_prob)
    grad_a = 2.0 * full[n-1:n-1+n]  # size n
    # gradient wrt logits: a * (grad_a - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits

def _lp_step(a, n, K, epsilon):
    conv = _conv_fft(a, a)
    max_val = np.max(conv)
    # get indices sorted descending
    idxs = np.argsort(-conv)[:K]
    # maybe also include near-zero indices? Not needed.

    # Build A_ub
    rows = []
    cols = []
    data = []
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row)
            cols.append(j)
            data.append(a[i])
        # M coefficient
        rows.append(row)
        cols.append(n)  # M variable
        data.append(-1.0)

    A_ub = csr_matrix((data, (rows, cols)), shape=(K, n+1))
    b_ub = np.zeros(K)

    # Equality sum b = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([1.0])

    # Objective: minimize M
    c = np.zeros(n+1)
    c[n] = 1.0

    # bounds for b: [0,1], for M: [0, None]
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        print(f"[LP] Linprog failed: {res.message}")
        return a, max_val
    b_opt = res.x[:n]
    # Normalize due to potential rounding errors
    b_opt = b_opt / b_opt.sum()
    return b_opt, np.max(_conv_fft(b_opt, b_opt))

            a = b_opt

"""
Improved optimizer for the step‑function bound.

Main ideas
---------
* Initialise from several diverse starting points (previous best, a few
  exponential decays, Dirichlet draws).
* For each start run a tiny *linear‑program* refinement that solves:
        minimise M  subject to   (a * b)[k] ≤ M   (k in the most
        critical shifts),   Σ b = 1 , b ≥ 0 .
  The LP is tiny (≈30 constraints) and quickly pushes the sequence
  away from the current worst convolution peaks.
* Follow the LP with a long mirror‑descent (exponentiated‑gradient)
  optimisation on the exact max‑convolution – the same update that
  the reference solution used, but we give it far more iterations.
* A final smooth L‑BFGS polishing (log‑sum‑exp surrogate) is performed
  if time permits.
* The best sequence found (by the cheap surrogate
  `2·n·max(conv)`) is evaluated with the provided `evaluate_sequence`
  and stored globally for the next run.

The whole routine respects the 1000 s budget, prints useful progress
information and returns the best list of non‑negative heights.
"""

import math
import time
import numpy as np
import scipy.optimize as _opt   # type: ignore
import scipy.sparse as _sp     # type: ignore

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact if sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max utilities (mirror descent on the probability simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _mirror_descent_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 5_000_000,
                        eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient mirror‑descent on max‑convolution.
    `start_seq` must already sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to stay on the simplex
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # classic diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny linear‑program refinement (active‑set of the worst shifts)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray,
               n: int,
               deadline: float,
               K: int = 30,
               max_iter: int = 10) -> tuple[np.ndarray, float]:
    """
    Given a probability vector `seq`, solve a small LP that looks for a new
    vector `b` with a smaller max‑convolution against the *fixed* `seq`.
    Only the K largest convolution entries are kept as constraints.
    Returns the improved sequence (or the original one if the LP fails).
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # indices of the K largest convolution values
        idxs = np.argpartition(-conv, K - 1)[:K]

        # build sparse matrix for constraints  a * b (at those shifts) - M <= 0
        rows, cols, data = [], [], []
        for row, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # coefficients for the b‑variables
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(row)
                cols.append(j)
                data.append(a[i])
            # coefficient for the auxiliary variable M
            rows.append(row)
            cols.append(n)          # last column is M
            data.append(-1.0)

        A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(K, n + 1))
        b_ub = np.zeros(K)

        # equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # objective: minimise M
        c = np.zeros(n + 1)
        c[n] = 1.0

        bounds = [(0, None)] * n + [(0, None)]   # b_j >=0 , M >=0

        res = _opt.linprog(c,
                           A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds,
                           method='highs')
        if not res.success:
            # If the LP fails we simply stop refining.
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b_opt = res.x[:n]
        # re‑normalise (numerical noise)
        b_opt = b_opt / b_opt.sum()
        new_val = _objective_surrogate(b_opt)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_seq = b_opt
            a = b_opt
            print(f"[LP] improvement to {best_val:.6f}")
        else:
            # no progress – stop early
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp surrogate and L‑BFGS polishing
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise a smooth approximation of max‑convolution (log‑sum‑exp)
    using L‑BFGS‑B. Operates in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabilise the exponentials
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (analytical)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # project gradient onto the simplex (mirror‑descent formula)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
