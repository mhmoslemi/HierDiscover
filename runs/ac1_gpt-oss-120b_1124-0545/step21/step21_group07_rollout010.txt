"""
Propose candidate for the step‑function attack.

The algorithm:
1. Generate a diverse set of analytic seeds (uniform, beta‑shapes,
   inverse‑sqrt, mixtures, random Dirichlet draws).
2. Evaluate the cheap surrogate 2·n·max(conv) and keep the best few.
3. For each top seed run a smooth‑max optimisation using Adam on the
   log‑softmax parametrisation.  The smooth‑max is a log‑sum‑exp
   approximation with a schedule of increasing β, allowing us to
   approach the true max while retaining a differentiable objective.
4. Finish with a long exponentiated‑gradient polishing (single‑active
   sub‑gradient) to drive the exact max down further.
5. Return the best sequence found (as a plain list) and store it in
   the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution discretised on indices 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt_product(n: int) -> np.ndarray:
    """U‑shaped distribution a_i ∝ 1/√((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64)
    j = n - 1 - i
    a = 1.0 / np.sqrt((i + 1.0) * (j + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt_sum(n: int) -> np.ndarray:
    """U‑shaped distribution a_i ∝ 1/√(i+1) + 1/√(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    draw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return draw / draw.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam on log‑softmax variables)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     max_iter_per_beta: int = 2000,
                     lr: float = 0.8,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution using Adam.
    Work in log‑softmax space to keep the iterates inside the simplex.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0,
            100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
            2_000_000.0,
        ]

    v = np.log(start_seq + 1e-30)       # log‑softmax parameters
    m = np.zeros_like(v)
    v_exp = np.zeros_like(v)
    t_iter = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # smooth‑max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # obj = 2·n·(shift + log(sum_exp)/beta)
            # (we only need gradient, not obj)

            # gradient w.r.t a
            w = exp_beta / sum_exp                # weights over each conv index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_exp = beta2 * v_exp + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v_exp / (1.0 - beta2 ** t_iter)

            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (single‑active) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 6.0,
                      single_active: bool = True):
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Operates in log‑space and respects the simplex.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max (vector‑averaged)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        eta = eta0 / np.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0,
            100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
            2_000_000.0,
        ]

    v = np.log(start_seq + 1e-30)           # log‑softmax parameters
    m = np.zeros_like(v)
    v_exp = np.zeros_like(v)
    t_iter = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = conv.max()

            # smooth‑max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_exp = beta2 * v_exp + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v_exp / (1.0 - beta2 ** t_iter)

            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 6.0,
                      single_active: bool = True):
    """
    Mirror‑descent (exponentiated‑gradient) using the exact max‑convolution.
    Operates in log‑space to stay on the simplex.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max (averaged over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        eta = eta0 / np.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable ``height_sequence_1``,
    and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Load a compatible previous solution, if any
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                s = arr.sum()
                if s > 0.0:
                    prev_arr = arr / s
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Analytic seed generation
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_beta_shape(n, 0.5, 0.5))          # arcsine‑like U‑shape
    seeds.append(_beta_shape(n, 0.3, 0.3))
    seeds.append(_beta_shape(n, 0.2, 0.2))
    seeds.append(_beta_shape(n, 0.1, 0.1))
    seeds.append(_inverse_sqrt_product(n))
    seeds.append(_inverse_sqrt_sum(n))

    for conc in [0.02, 0.05, 0.1, 0.3]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for shape in [_beta_shape(n, 0.5, 0.5), _inverse_sqrt_product(n)]:
        for alpha in [0.1, 0.25, 0.45]:
            seeds.append(_mix_shapes(_uniform(n), shape, alpha))

    if prev_arr is not None:
        base = prev_arr
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate ranking of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine each top seed (Adam → exact‑max EG)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.15 / max(1, (top_k - si))

        print(f"\n[Seed {si+1}/{top_k}] optimisation window ≈ {sub_deadline - time.time():.1f}s")

        # (a) Adam on the smooth max
        seq_adam = _adam_smooth_opt(seed_vec.copy(),
                                    n,
                                    sub_deadline,
                                    beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                   200.0, 500.0, 1_000.0,
                                                   2_000.0, 5_000.0,
                                                   20_000.0, 50_000.0,
                                                   1e5, 5e5, 1e6],
                                    max_iter_per_beta=2500,
                                    lr=0.8)
        val_adam = _objective_surrogate(seq_adam)

        # (b) Exact‑max polishing with exponentiated‑gradient
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                              n,
                                              sub_deadline,
                                              max_iters=1_500_000,
                                              eta0=6.0,
                                              single_active=True)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # choose the best among the three candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_eg, val_eg),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")
        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Global long‑run EG polishing (single‑active)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Global EG polishing (single‑active) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=6_000_000,
                                                 eta0=5.0,
                                                 single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
