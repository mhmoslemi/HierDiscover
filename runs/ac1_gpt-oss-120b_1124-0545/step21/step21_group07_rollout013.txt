def _adam_opt(start_seq, n, deadline, max_iters=1_000_000, lr=0.1, beta1=0.9, beta2=0.999):
    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())
        # compute subgradient g for this active idx
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        rev_idx = max_idx - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g[mask] = a[rev_idx[mask]]
        g = 2.0 * g

        # Mirror descent gradient (log-space)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)

"""
height_sequence optimizer – Adam + SA refinement

The algorithm proceeds in three stages:

1. **Broad seeding** – a large family of analytically‑generated vectors
   (uniform, windows, beta‑shapes, exponentials, power‑laws, Gaussians,
   Dirichlet draws, mixtures, …).  All candidates are ℓ₁‑normalised and
   quickly ranked by the surrogate objective
   ``2·n·max(conv(a,a))`` (exact max‑convolution via an FFT).

2. **Adam mirror‑descent** – starting from the best seed we optimise the
   exact non‑smooth objective using a sub‑gradient of the active
   convolution entry.  The iterates live in log‑space (v = log a) and are
   updated with the Adam optimizer while the simplex constraint is
   enforced by a soft‑max.  The schedule is purely time‑driven, so the
   optimiser runs until the global deadline is reached.

3. **Fine‑grained simulated‑annealing** – a final SA stage makes tiny
   Gaussian perturbations of the current best vector.  The temperature
   decays linearly, allowing the algorithm to escape shallow local minima
   and to polish the solution.

All steps respect the 1000 s wall‑clock budget, log progress, and store
the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple soft‑max (stable) – maps logits to a probability simplex
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for sum‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – return ℓ₁‑normalised vectors (sum == 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 5_000_000,
              lr: float = 0.03,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """
    Mirror‑descent with Adam on the exact max‑convolution.
    Operates in log‑space (v = log a) and keeps a on the simplex
    via soft‑max.
    Returns the best sequence seen.
    """
    v = np.log(start_seq + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)

        # full convolution & active index
        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        idx = np.arange(n, dtype=np.int64)
        rev_idx = k - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = a[rev_idx[mask]]
        g = 2.0 * g                     # exact sub‑gradient for the max

        # mirror‑descent gradient (log‑space)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam updates in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing (Gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.02,
                         T_end: float = 1e-9):
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linearly decreasing temperature
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
