def subgradient_step(a, eta):
    # compute conv and argmax
    conv = _conv_fft(a, a)  # shape length 2n-1
    k = int(np.argmax(conv))
    # compute gradient
    g = np.zeros_like(a)
    i_start = max(0, k-(n-1))
    i_end = min(n-1, k)
    idx_i = np.arange(i_start, i_end+1)
    rev_idx = k - idx_i
    g[idx_i] = 2 * a[rev_idx]
    # gradient descent step
    a_new = a - eta * g
    # project onto simplex
    a_new = project_simplex(a_new)
    return a_new

"""
Search for a non‑negative step‑function that minimises evaluate_sequence().
The method builds a rich set of analytic seeds (beta, power‑law,
arcsine/uniform mixtures, Dirichlet draws), selects the best one under the
exact surrogate 2·n·max(conv) and then runs a long exponentiated‑gradient
mirror‑descent phase (sub‑gradient of the true max‑convolution).  A final
tiny random local search cleans up the result.  The best sequence found is
saved in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# Optional SciPy import for L‑BFGS smoothing (may be absent)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  Assumes seq sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------
# Simple probability‑vector generators (all return normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a

def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    a /= a.sum()
    return a

def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1} (α,β > 0)."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a

def _power_law_distribution(n: int, exponent: float) -> np.ndarray:
    """U‑shaped power‑law a_i ∝ ((i+0.5)(n‑0.5‑i))^{-exponent}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.power(i * (n - i), -exponent)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    Exponentiated‑gradient mirror‑descent that minimises the exact objective.
    ``start_seq`` must already be normalised (sum = 1).
    Returns the best sequence seen and its surrogate value.

    best_seq = a.copy()

        i_start = max(0, k - (n - 1))

def _local_random_search(start_seq: np.ndarray,

    one coordinate to another and keep improvements.

def _lbfgs_smooth_optimize(start_vec: np.ndarray,

                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12_000):

    Optimise the smooth max‑convolution surrogate using L‑BFGS‑B.
    Variables live in the log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (weighted sum of convolutions)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a *= 2.0 * n                     # outer factor 2·n

            # transform to gradient w.r.t. log‑softmax variables
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):

                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)

    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

    a /= a.sum()
    return a

    a /= a.sum()
    return a

        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Optional L‑BFGS smoothing on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.30
        print("\n--- L‑BFGS smoothing (optional) ---")
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                           max_iter_per_beta=20_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("   >>> Updated global best (L‑BFGS)")

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (main workhorse)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 3.0    # keep a few seconds for the final step
        total_budget = deadline - start_time
        remaining_fraction = (polishing_deadline - time.time()) / total_budget
        max_iters = int(2_000_000 * remaining_fraction)
        max_iters = max(200_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=10.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n, deadline,
                                                    attempts=200_000,
                                                    sigma=0.02)
        print(f"[Local] surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
