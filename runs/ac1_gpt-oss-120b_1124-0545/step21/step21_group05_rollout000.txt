"""
Optimiser for the step‑function challenge.

Approach
--------
1. **Seed generation** – start from a handful of analytically‑parameterised
   families (uniform, triangular, sine‑window, Gaussian, power‑law) and from
   the previously found best sequence (if any).  All seeds are ℓ₁‑normalised
   and have the same length `n`.

2. **Smooth gradient descent** – we minimise a *smooth* surrogate of the
   true objective 2·n·max(conv(a,a)).  For a temperature β the surrogate is

       Lβ(a) = 2·n·log( Σk exp(β·convk) ) / β ,

   whose gradient can be written in closed form using one extra convolution.
   By increasing β progressively we approach the true max‑norm while keeping
   the optimisation landscape smooth.

   The optimisation is performed in the log‑softmax (mirror‑descent) space,
   guaranteeing non‑negativity and the ℓ₁‑constraint without an explicit
   projection step.

3. **Budget allocation** – a small fraction of the total time is spent
   refining each seed (≈10 % of the remaining budget).  The best seed is
   then polished for the rest of the time with a long run of the smooth
   descent.

4. **Final evaluation** – the best sequence is handed to `evaluate_sequence`
   and stored in the global variable `height_sequence_1` for possible
   continuation runs.

The algorithm deliberately uses a smoother gradient (full softmax) instead
of the single‑index sub‑gradient used in the baseline; this gives a more
accurate descent direction and, together with longer optimisation time,
typically pushes the bound below the previous 1.5085 value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(a,a)) for a ℓ₁‑normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth gradient descent on the soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 6.0,
                         beta0: float = 1.0,
                         beta_mul: float = 2.0,
                         beta_step: int = 100_000,
                         beta_max: float = 1_000_000.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution objective.

    Parameters
    ----------
    start_seq : np.ndarray
        ℓ₁‑normalised starting vector (length n).
    n : int
        Problem dimension.
    deadline : float
        Unix timestamp after which the routine must stop.
    max_iters : int, optional
        Upper bound on iteration count.
    eta0 : float, optional
        Initial learning‑rate factor.
    beta0 : float, optional
        Starting temperature for the soft‑max.
    beta_mul : float, optional
        Multiplicative factor to raise β every ``beta_step`` iterations.
    beta_step : int, optional
        Interval (in iterations) at which β is increased.
    beta_max : float, optional
        Upper bound for β.

    Returns
    -------
    best_seq, best_val : (np.ndarray, float)
        The best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    # keep everything in the log‑softmax space
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    beta = beta0
    it = 0
    last_improve_it = 0
    rng = np.random.default_rng(12345)

    while it < max_iters and time.time() < deadline:
        it += 1

        # increase temperature on schedule
        if it % beta_step == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # convolution a*a
        conv = _conv_fft(a, a)                # length 2n‑1
        max_c = np.max(conv)

        # soft‑max weights on the convolution entries
        w = np.exp(beta * (conv - max_c))     # max entry becomes 1.0
        w_sum = w.sum()
        w = w / w_sum                         # probabilities, sum=1

        # gradient of the smoothed objective:
        #   ∂/∂a  ( Σ_k w_k conv_k ) = 2 * (rev_a * w)  (central slice)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]    # slice of length n

        # Mirror‑descent update (projected onto simplex automatically)
        eta = eta0 / math.sqrt(it)
        # remove the component along the all‑ones direction to keep Σa = 1
        grad_adj = grad - np.dot(grad, a)
        log_a = log_a - eta * grad_adj
        a = _softmax(log_a)

        # current surrogate value
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve_it = it

        # occasional random perturbation to escape plateaus
        if it % 200_000 == 0 and (it - last_improve_it) > 150_000:
            # add a tiny isotropic noise in log‑space
            log_a += 0.02 * rng.standard_normal(n)
            a = _softmax(log_a)   # re‑normalise
            last_improve_it = it   # reset counter

    return best_seq, best_val

    """

    start_time = time.time()

    # Problem size – we use a larger dimension than the baseline.
    # ------------------------------------------------------------------

    # Load previous solution (if any) and up‑sample it to the current n.
    # ------------------------------------------------------------------

    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.2))
    seeds.append(_gaussian(n, sigma_frac=0.10))
    seeds.append(_power_law(n, p=0.7))

    # ------------------------------------------------------------------
    # A few random Dirichlet draws (different concentration parameters)
    # ------------------------------------------------------------------
    for conc in (0.05, 0.2, 1.0):
        for _ in range(3):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # ------------------------------------------------------------------
    # Quick refinement of each seed (≈10 % of the remaining time per seed)
    # ------------------------------------------------------------------
    best_global_seq = None
    best_global_val = np.inf

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break

        # allocate a modest share of the remaining budget
        remaining = deadline - time.time()
        # ensure at least a few seconds per seed
        share = max(5.0, remaining * 0.10 / max(1, len(seeds) - idx))
        sub_deadline = time.time() + share

        # Normalise explicitly (guards against accidental rounding errors)
        seed_vec = np.maximum(seed_vec, 0.0)
        seed_vec = seed_vec / seed_vec.sum()

        print(f"\n[Seed {idx + 1}/{len(seeds)}] start – allocated ~{share:.1f}s")

        # Short smooth‑gradient run
        seq_opt, val_opt = _smooth_gradient_opt(seed_vec,
                                                n,
                                                sub_deadline,
                                                max_iters=250_000,
                                                eta0=6.0,
                                                beta0=1.0,
                                                beta_mul=2.0,
                                                beta_step=50_000,
                                                beta_max=200_000.0)

        print(f"[Seed {idx + 1}] surrogate after refinement = {val_opt:.6f}")

        if val_opt < best_global_val:
            best_global_val = val_opt
            best_global_seq = seq_opt.copy()
            print(f"    *** New global best = {best_global_val:.6f}")

    # ------------------------------------------------------------------
    # Final long polishing on the current best seed
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # Fallback – should not happen, but guard against it
        best_global_seq = _uniform(n)

    if time.time() < deadline:
        print("\n--- Final polishing phase ---")
        final_seq, final_val = _smooth_gradient_opt(best_global_seq,
                                                    n,
                                                    deadline,
                                                    max_iters=2_200_000,
                                                    eta0=5.0,
                                                    beta0=10.0,
                                                    beta_mul=2.0,
                                                    beta_step=100_000,
                                                    beta_max=1_000_000.0)
        if final_val < best_global_val:
            best_global_seq, best_global_val = final_seq, final_val
            print(f"[Final] improved to {best_global_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate with the provided function (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_global_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_global_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_global_seq.tolist()
    return best_global_seq.tolist()
