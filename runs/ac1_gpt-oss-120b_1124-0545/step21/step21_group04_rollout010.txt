def adam_opt(seq, start_logit, beta_schedule, it_per_beta, lr):
    v = start_logit
    m = np.zeros_like(v)
    vvar = np.zeros_like(v)
    best_seq = None
    best_val = inf
    for beta in beta_schedule:
        for it in range(it_per_beta):
            if time.time()>deadline: return best_seq or seq
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = conv.max()
            # compute smooth max obj and gradient
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_obj = 2.0 * n * (shift + np.log(sum_exp)/beta)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)    # length = 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient w.r.t logits:
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1-beta1) * grad_v
            vvar = beta2 * vvar + (1-beta2) * (grad_v**2)
            m_hat = m / (1 - beta1_pow)
            v_hat = vvar / (1 - beta2_pow)
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)
            # track best using actual surrogate (max conv) not smooth
            cur_max = conv.max()
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.  The algorithm mixes several
strategies that proved useful for the previous best solution:

1.  A rich set of analytic starters (uniform, exponential decay, power‑law
    U‑shapes, symmetric Beta‑shapes) and a few random Dirichlet draws.
2.  A fast Adam‑accelerated mirror‑descent on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  The continuation schedule on the smoothing
    parameter β starts from β≈1 and ends at β≈5·10⁴.
3.  A short stochastic local‑search that transfers a tiny amount of mass
    between two coordinates if it improves the exact surrogate.
4.  (Optional) an L‑BFGS‑B refinement of the smooth objective if SciPy is
    available.
5.  Final evaluation with ``evaluate_sequence`` and storage of the result in
    ``height_sequence_1`` for the next run.

All helpers are defined at module level, no closures are used and everything
runs inside the 1000 s budget.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the final L‑BFGS refinement.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level FFT‑based convolution utilities (real FFT, O(n log n)).
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (for the probability‑simplex parametrisation).
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑shape: a_i ∝ (i+1)^{α-1} (n‑i)^{α-1}."""
    i = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    w = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (alpha - 1.0))
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1).
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence already L1‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (averaged over all active ks).
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    g /= max(1, len(active))
    return g


# ----------------------------------------------------------------------
# Adam‑accelerated mirror descent on a smooth max‑convolution.
# ----------------------------------------------------------------------
def _adam_opt(start_a: np.ndarray,
              deadline: float,
              beta_schedule: List[float],
              iters_per_beta: int = 4000,
              lr: float = 0.6) -> tuple[np.ndarray, float]:

    Adam on the log‑softmax parametrisation with a log‑sum‑exp smooth max.
    Returns the best sequence (L1‑normalised) seen during the run and its
    exact surrogate value.
    """
    n = len(start_a)
    # initialise logits (log‑softmax variables)
    log_v = np.log(start_a + 1e-30)

    # Adam state
    m = np.zeros_like(log_v)
    v2 = np.zeros_like(log_v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(log_v)

            # ----- smooth max (log‑sum‑exp) -----
            conv = _conv_fft(a, a)                # length 2n‑1
            max_c = conv.max()
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                # weights for the smooth max

            # gradient of smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

            # chain rule to logits
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_log * grad_log)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v2 / (1.0 - beta2 ** t)

            log_v = log_v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- keep track of the **exact** surrogate -----
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        # slowly decay learning rate after each β
        lr *= 0.7
        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max) – more aggressive than the Adam run.
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_a: np.ndarray,
                           deadline: float,
                           beta_schedule: List[float] = None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Smooth‑max optimisation with L‑BFGS‑B (log‑softmax variables)."""
    if _opt is None:
        return start_a

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0, 50000.0]

    v = np.log(start_a + 1e-30)          # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = conv.max()
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth = shift + np.log(sum_exp) / beta_val

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule (log‑softmax)
            return 2.0 * len(a) * smooth, grad_v

            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:

                                x0=v,
                                method='L-BFGS-B',
                                jac=True,

            break
        except Exception as e:  # pragma: no cover
