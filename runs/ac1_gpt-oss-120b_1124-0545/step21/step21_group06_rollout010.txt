```
def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv
```

```
def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0:
        return np.full_like(v, 1.0 / len(v))
    return e / s
```

   double‑exponential, edge‑power, Gaussians, β‑shapes, power‑laws,
   mixtures) plus random Dirichlet draws are generated, ℓ₁‑normalised and
   evaluated with an FFT‑based surrogate (the same objective for a
   normalised vector). The best *top_k* seeds are kept.

2. **Local refinement** – every seed undergoes three complementary
   procedures:
        *exponentiated‑gradient* mirror‑descent on the exact max‑convolution,
        a *smooth‑max* L‑BFGS optimisation (log‑sum‑exp surrogate) and a
        *simulated‑annealing* pass.
   The best variant updates the global optimum.

3. **Polishing** – a longer exponentiated‑gradient run (all active
   indices) and a high‑budget simulated‑annealing finish the search.

All work respects a 1000 s wall‑time budget (2 s safety margin). The final
sequence is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) (left‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i)) (right‑hand exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    """Symmetric sum of left and right exponentials: high on both edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_cosine(n: int) -> np.ndarray:
    """U‑shaped cosine: a_i ∝ 1 + cos(2π i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2.0 * np.pi * i / (n - 1))
    # a is already non‑negative (range [0,2])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) + (n-i)^(-p) – large at the two ends."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space (KL‑mirror)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active indices -----
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient -------
        g_tot = np.zeros_like(a)
        # Loop over active indices – typically a handful
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            # factor 2 accounts for symmetry of the objective (2·n·max)
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning rate -----
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0               # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – reuse previous solution if it exists.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
                n = int(arr.shape[0])
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    print(f"\n--- Optimisation started (n={n}) ---")

    # --------------------------------------------------------------
    # 1. Broad seeding – analytic families + random draws
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_u_cosine(n))

    # beta‑shape grid (coarse)
    alphas = np.linspace(0.2, 6.0, 13)
    betas = np.linspace(0.2, 6.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential families (single‑sided)
    rates = np.logspace(-4, -0.2, 25)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # double‑exponential (U‑shaped)
    for r in rates:
        seeds.append(_double_exponential(n, r))

    # edge‑power family
    for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
        seeds.append(_edge_power(n, p))

    # power‑law families (monotone decreasing)
    for p in np.linspace(0.0, 5.0, 13):
        seeds.append(_power(n, p))

    # mixtures with uniform – adds extra flexibility
    for r in rates[::5]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for a_mix in (0.1, 0.3, 0.5, 0.7):
            seeds.append(_mix_shapes(_uniform(n), exp_l, a_mix))
            seeds.append(_mix_shapes(_uniform(n), exp_r, a_mix))
            seeds.append(_mix_shapes(_uniform(n), _double_exponential(n, r), a_mix))
            seeds.append(_mix_shapes(_uniform(n), _u_cosine(n), a_mix))

    # random Dirichlet draws – explore sparse and dense regimes
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best (if any) – encourage local exploration
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr.copy()
        seeds.append(base)
        for _ in range(12):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds generated before deadline.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start local refinement on top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget
        remaining = deadline - time.time()
        sub_budget = max(4.0, remaining * 0.13 / max(1, (top_k - si)))
        sub_deadline = time.time() + sub_budget

        # ----- (a) Exponentiated‑gradient, multi‑active for quality -----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=300_000,
                                           eta0=7.0,
                                           single_active=False)

        # ----- (b) L‑BFGS smooth‑max refinement (more iterations) -----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0],
                                               max_iter_per_beta=3500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ----- (c) Simulated‑annealing refinement -----
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=200_000,
                                                  start_sigma=0.0025,
                                                  T_start=0.018,
                                                  T_end=5e-8)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among the three refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final polishing – intensive EG + final SA
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive EG polishing (all active) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_200_000,
                                                 eta0=6.5,
                                                 single_active=False)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=350_000,
                                                          start_sigma=0.001,
                                                          T_start=4e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
