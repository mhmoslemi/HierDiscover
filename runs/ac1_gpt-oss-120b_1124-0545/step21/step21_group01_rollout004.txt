assistantfinal**Solution Overview**

The evaluation function is homogeneous – scaling the whole vector does not change the score – therefore we can restrict the search to probability vectors (`sum = 1`).  
The score is  

\[
\text{score}(a)=\frac{2n\;\max_k (a*a)_k}{(\sum_i a_i)^2}=2n\;\max_k (a*a)_k ,
\qquad a_i\ge 0,\;\sum a_i=1 .
\]

Finding a vector that minimises the infinity‑norm of the auto‑convolution is a non‑convex problem.  
The program below combines several complementary optimisation phases that have proven to be effective for this class of problems:

1. **Diverse deterministic & random seeds** – a small pool of well‑known shapes (uniform, arcsine, triangular, beta‑laws, Dirichlet draws) is evaluated and the best seed is kept.

2. **Adam optimiser on a smooth surrogate** – the non‑smooth `max` is replaced by a log‑sum‑exp with a temperature β.  
   Starting with a low temperature the optimiser performs many Adam steps, after each temperature increase it is “re‑initialised” at the current best point.  
   This stage is the core driver of improvement.

3. **Exact exponentiated‑gradient polish** – after the smooth phase we run the sub‑gradient method that directly minimises the true max‑convolution (the same method used in the reference implementation) but with far more iterations, thanks to the remaining time budget.

4. **Tiny random local search** – a last‑minute hill‑climb that moves a small amount of mass between two coordinates if the move improves the exact objective.

All stages respect the global 1000 s limit and store the best sequence in the global variable `height_sequence_1`.  
The code is deliberately written without nested closures (as required) and contains extensive logging.

```python
"""
propose_candidate
=================

A self‑contained optimiser for the “max‑auto‑convolution” problem.
The algorithm consists of

* a deterministic/randome seed pool,
* Adam optimisation on a log‑sum‑exp surrogate with an increasing
  temperature schedule,
* a long exponentiated‑gradient polish on the exact objective,
* a final lightweight random local hill‑climb.

The best vector found (as a plain list of floats) is stored in the
global variable ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, softmax, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """
    Exact surrogate for a probability vector:
        2 * n * max(conv(seq, seq))
    The function works for any non‑negative vector (it normalises internally).
    """
    s = seq.sum()
    if s <= 0.0:
        return np.inf
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Seed generators (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_smooth_optimize(start_vec: np.ndarray,
                          n: int,

    Adam optimisation in the log‑softmax space using a smooth
    approximation of the max‑convolution (log‑sum‑exp with temperature β).
    Returns the best exact surrogate value seen.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0,
        ]

    # initialise log‑probabilities (they automatically stay on the simplex)
    log_v = np.log(start_vec + 1e-30)
    m = np.zeros_like(log_v)
    v = np.zeros_like(log_v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq, n)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1

            a = _softmax(log_v)                     # current probability vector
            conv = _conv_fft(a, a)                   # full auto‑convolution
            shift = np.max(conv)                     # stabilise the soft‑max
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / exp_beta.sum()            # soft‑weights over 2n‑1 entries

            # gradient of  Σ w_k (a*a)_k  w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length ≈ 3n‑2
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)        # outer factor 2·n

            # map to the log‑softmax (mirror‑descent) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate schedule – larger at the beginning,
            # slowly decaying with the iteration count
            lr = 0.3 / (math.sqrt(t) + 1.0)
            log_v -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation (every few hundred steps)
            if (t % 500) == 0:
                cur_val = _objective_surrogate(a, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

        # after finishing a temperature, re‑centre the log‑variables
        # at the best sequence found so far – this avoids drift.
        log_v = np.log(best_seq + 1e-30)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 1_500_000,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient at the active lag.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random local hill‑climb
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 60_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Small stochastic walk on the simplex: move a tiny random amount
    from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq, n)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm respects the overall 1000 s budget and stores the best
    construction in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # Load a previously stored solution (if dimensions match)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    # few beta‑shaped families
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, alpha, beta))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # small perturbations of uniform
    for _ in range(4):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.004))

    # --------------------------------------------------------------
    # 1️⃣  Evaluate seeds and keep the current best
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec, n)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Lightweight random Dirichlet search (quick global sweep)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Random Dirichlet sweep ---")
        trials = int(10_000 * remaining / (budget_s))
        for t in range(trials):
            if time.time() > deadline:
                break
            conc = rng.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            cand = _random_dirichlet(n, conc, rng)
            val = _objective_surrogate(cand, n)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"  [RND] new best surrogate = {best_val:.6f} (trial {t})")

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation on the smooth surrogate (core phase)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation on smooth surrogate ---")
        remain = deadline - time.time()
        sub_deadline = time.time() + remain * 0.60      # 60 % of remaining time
        seq_adam, val_adam = _adam_smooth_optimize(best_seq, n, sub_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polish (sub‑gradient of true max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 3.0
        # give it the bulk of the remaining budget
        max_iters = int(2_000_000 *
                        max(0.0, polishing_deadline - time.time()) /
                        max(1.0, deadline - start_time))
        max_iters = max(300_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           max_iters=max_iters, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 5️⃣  Final tiny random local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random local refinement ---")
        attempts = int(120_000 *
                       max(0.0, deadline - time.time()) /
                       max(1.0, deadline - start_time))
        attempts = max(20_000, attempts)
        seq_local, val_local = _local_random_search(best_seq, n, deadline,
                                                    attempts=attempts,
                                                    sigma=0.02)
        print(f"[Local] exact surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 6️⃣  Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
