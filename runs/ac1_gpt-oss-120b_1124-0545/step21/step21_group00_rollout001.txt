def _lbfgs_fun(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)  # length len(a)+len(w)-1 = 2n-1? Actually w length=2n-1; but conv length = len(w)

"""
Hybrid stochastic‑global + deterministic‑local search for a step‑function
with minimal max‑auto‑convolution.

The search proceeds in four stages:

1. **Seed generation** – a wide collection of manually‑crafted shapes
   (uniform, Dirichlet, sparse Dirichlet, Gaussian mixture,
   Fourier‑softmax, single sine window, triangular window,
   single Gaussian, discrete Beta / arcsine shapes) and the previous
   best sequence (if present) are normalised to sum = 1.

2. **Global selection** – the surrogate objective
   `S(a)=2·n·max(conv(a,a))` (which equals the true objective after
   normalisation) is evaluated for every seed; the best seed is kept.

3. **Deterministic refinement** – a long L‑BFGS optimisation on the
   smooth log‑sum‑exp approximation of the max is performed on the best
   seed.  The optimisation works in the log‑softmax space, i.e. the
   variables are the logits of a probability vector.

4. **Exact polishing** – an exponentiated‑gradient (mirror‑descent)
   algorithm directly minimises the true max‑convolution objective.
   After polishing a cheap hill‑climbing loop (tiny random perturbations
   with immediate acceptance of improvements) tries to squeeze a little
   more out of the remaining budget.

The best sequence found is stored in the global variable
``height_sequence_1`` and also returned as a plain ``list[float]``.
All numbers are non‑negative and the routine respects the 1000 s budget.
"""

import time
import math
import numpy as np

# optional SciPy – used for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a probability vector of length n:
    2·n·max(conv(seq, seq)).  (Value equals `evaluate_sequence` when sum(seq)=1.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k random positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      – weights (non‑negative, later normalised)
        μ_0 … μ_{K-1}      – means in [0,1]
        σ_0 … σ_{K-1}      – std‑devs >0
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """
    Build a probability vector by applying a softmax to a Fourier series:
    a_i ∝ exp( Σ_k θ_k φ_k(i) ).
    """
    logits = basis @ theta
    return _softmax(logits)


def _single_sine_window(n: int) -> np.ndarray:
    """Half‑sine window: a_i ∝ sin(π (i+0.5)/n)."""
    w = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _triangular_window(n: int) -> np.ndarray:
    """Symmetric triangle (pyramid) shape."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = 1.0 - np.abs(i - half) / half
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _single_gaussian_window(n: int, sigma_frac: float = 0.2) -> np.ndarray:
    """Single Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # handle possible numerical issues when α or β < 1
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _lbfgs_fun(v: np.ndarray, n: int, beta: float):
    """Objective & gradient for the smooth surrogate (log‑sum‑exp of conv)."""
    a = _softmax(v)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)              # length 2n‑1
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth approximation of the max
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000):
    """
    L‑BFGS‑B on the smooth surrogate (variables = log‑softmax).
    The optimisation stops early if the global deadline is exceeded.
    """
    if _opt is None:
        return start_vec.copy()          # nothing to do without SciPy

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         2000000.0, 5000000.0,
                         10000000.0]

    v = np.log(start_vec + 1e-30)   # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(_lbfgs_fun,
                                v,
                                args=(n, float(beta)),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 3.0):
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (all pairs i+j=k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # each unordered pair contributes twice, diagonal only once
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing perturbation loop
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                rng: np.random.Generator,
                deadline: float,
                init_sigma: float = 0.0015,
                min_sigma: float = 1e-5,
                decay: float = 0.995):
    """
    Repeatedly perturb the current sequence and keep any move that improves
    the surrogate objective.  Sigma is geometrically decayed.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    sigma = init_sigma

    while time.time() < deadline and sigma >= min_sigma:
        cand = _perturb(best_seq, rng, sigma=sigma)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq, best_val = cand, cand_val
        sigma *= decay

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm follows the stages described in the module docstring.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0
    n = 4000                     # work length (as used in the reference run)

    # --------------------------------------------------------------
    # Retrieve a previous best if it exists
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds
    # --------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    # Dirichlet (various concentrations)
    for conc in (0.02, 0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse Dirichlet (support sizes)
    for k in (80, 120, 200):
        for _ in range(2):
            seeds.append(_sparse_dirichlet(n, k, rng))

    # Small Gaussian‑mixture seeds
    K_mix = 8
    for _ in range(6):
        params = np.concatenate([
            rng.random(K_mix),                     # raw weights (will be normalised)
            rng.random(K_mix),                     # means
            rng.uniform(0.001, 0.5, K_mix)         # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n, K=K_mix))

    # Fourier‑softmax seeds (moderate dimension)
    D_fourier = 40
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    basis = []
    for k in range(1, D_fourier // 2 + 1):
        basis.append(np.sin(2.0 * np.pi * k * i))
        basis.append(np.cos(2.0 * np.pi * k * i))
    basis = np.column_stack(basis)   # shape (n, D_fourier)
    for _ in range(5):
        theta = rng.normal(scale=1.0, size=D_fourier)
        seeds.append(_fourier_softmax(theta, basis))

    # Hand‑crafted shape seeds
    seeds.append(_single_sine_window(n))
    seeds.append(_triangular_window(n))
    seeds.append(_single_gaussian_window(n, sigma_frac=0.2))

    # Discrete Beta / arcsine‑like seeds
    for _ in range(12):
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        seeds.append(_beta_distribution(n, alpha, beta))

    # --------------------------------------------------------------
    # 2️⃣  Evaluate all seeds (fast surrogate) and keep the best so far
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    seed_vals = []

    print("\n--- Evaluating initial seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        seed_vals.append((val, vec.copy()))
        print(f"[seed {idx+1:03d}] surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Local refinement with smooth L‑BFGS (log‑sum‑exp)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of current best ---")
        # allocate roughly half of the remaining budget for smoothing
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45
        # more aggressive schedule (still safe)
        beta_sched = [1.0, 5.0, 20.0, 80.0,
                      200.0, 500.0, 1000.0,
                      2000.0, 5000.0,
                      20000.0, 50000.0,
                      100000.0, 200000.0,
                      500000.0, 1000000.0]   # stop a bit earlier
        seq_lbfgs = _lbfgs_smooth_optimize(
            best_seq,
            n,
            sub_deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=30000,   # a bit more iterations per beta
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("  >>> Updated global best after L‑BFGS")

    # --------------------------------------------------------------
    # 4️⃣  Exact mirror‑descent (exponentiated‑gradient) polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polish_deadline = deadline - 3.0
        time_left = polish_deadline - time.time()
        # give a generous iteration budget proportional to remaining time
        max_iters = int(min(3_000_000,
                            max(200_000,
                                time_left / (deadline - start_time) * 2_500_000)))
        seq_eg, val_eg = _exp_gradient_opt(
            best_seq,
            n,
            polish_deadline,
            max_iters=max_iters,
            eta0=3.0,
        )
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("  >>> Updated global best after EG")

    # --------------------------------------------------------------
    # 5️⃣  Simple hill‑climbing perturbation passes
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # use a geometric decay of sigma to explore finer neighbourhoods
        cand_seq, cand_val = _hill_climb(best_seq, rng,
                                         deadline - 2.0,
                                         init_sigma=0.0010,
                                         min_sigma=5e-6,
                                         decay=0.990)
        if cand_val < best_val:
            best_val, best_seq = cand_val, cand_seq.copy()
            print(f"[Hill‑climb] new best surrogate = {best_val:.6f}")

        # abort if time exhausted
        if time.time() > deadline - 2.0:
            break

    # --------------------------------------------------------------
    # 6️⃣  Final official evaluation with ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
