def _subgradient_max_conv(a):
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev = k - idx
        # rev within 0..n-1
        g[idx] += 2.0 * a[rev]
    g /= max(1, len(active))
    return g

def _adam_optimize(start_seq, n, deadline, max_iters, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    logit = np.log(a+1e-30)
    # enforce symmetry at start
    logit = (logit + logit[::-1]) / 2.0
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(logit)
        # enforce symmetry:
        a = (a + a[::-1]) / 2.0
        # re-project to simplex? sum stays 1 else hyper small diff due to averaging.
        # Might not sum exactly 1; renorm:
        a = a / a.sum()
        # compute surrogate and maybe track best
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
        # compute subgradient wrt a
        g_a = _subgradient_max_conv(a)
        # note: if max is zero? seldom.
        # gradient w.r.t logits
        grad_logit = a * (g_a - np.dot(a, g_a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logit
        v = beta2 * v + (1 - beta2) * (grad_logit ** 2)

"""
propose_candidate – symmetry‑aware Adam + L‑BFGS refinement.

The optimiser works directly on the probability vector a (a_i ≥ 0,
∑a_i = 1) using a soft‑max parametrisation.
A mirrored (palindromic) shape is enforced at every iteration
because the optimal step function is known to be symmetric.
The inner loop uses an Adam optimiser on the exact non‑smooth
objective ``2·n·max(conv(a,a))`` (the surrogate of evaluate_sequence
when the L1‐norm is 1).  After a cheap multi‑restart phase a
longer refinement is performed on the best candidate, followed by
a smooth‑max L‑BFGS stage (if SciPy is present).  The best sequence
found is stored globally as ``height_sequence_1`` and returned as a
plain list of floats.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real‑valued input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy).real[:L]
    return conv


# ----------------------------------------------------------------------
# Simple analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    w /= w.sum()
    return w


def _prod_inv_power(n: int, p: float) -> np.ndarray:
    """
    U‑shaped law a_i ∝ ((i+1)*(n-i))^{-p}.
    This concentrates mass near the two ends more aggressively than
    the simple sum‑law used above.
    """
    i = np.arange(n, dtype=np.float64)
    w = ((i + 1.0) * (n - i)) ** (-p)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Soft‑max utilities (used by the Adam optimiser)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the L¹‑norm is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) for a sequence already normalised
    to L¹‑norm 1.  This equals the value returned by evaluate_sequence.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (average over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance – a few ULPs are enough
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    if active.size > 0:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation, with symmetry enforcement
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.10,
                   beta1: float = 0.90,
                   beta2: float = 0.999,
                   eps_adam: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using Adam on the exact max‑convolution objective.
    The iterate lives in the log‑probability space; after each update
    the logits are symmetrised (logits ↔ reverse) so that the resulting
    probability vector stays (approximately) palindromic.
    """
    a = start_seq.copy()
    # initialise logits – add a tiny constant for numerical safety
    logit = np.log(a + 1e-30)

    # enforce exact symmetry on the logits before the first iteration
    logit = (logit + logit[::-1]) * 0.5

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ------------ probability vector and symmetry ------------
        a = _softmax(logit)
        # enforce symmetry in the probability vector
        a = (a + a[::-1]) * 0.5
        a /= a.sum()                     # tiny re‑normalisation

        # ------------ keep the best surrogate observed ------------
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

        # ------------ gradient of the surrogate (w.r.t a) ----------
        g_a = _subgradient_max_conv(a)

        # ------------ translate to gradient w.r.t. logits ----------
        # dL/dlogit = a * (g_a - <a,g_a>)
        dot = np.dot(a, g_a)
        grad_logit = a * (g_a - dot)

        # ------------ Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------------ re‑impose symmetry on the logits ------------
        logit = (logit + logit[::-1]) * 0.5

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑softmax variables)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        # a fairly dense schedule reaching a very sharp approximation
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    # optimisation variables are logits (log‑softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_max = shift + np.log(sum_exp) / beta_val

            obj = 2.0 * n * smooth_max

            # gradient w.r.t. a (see the derivation in the original code)
            w = exp_beta / sum_exp          # softmax‑weights over convolution entries
            rev_a = a[::-1]
            # convolution of rev_a with w gives the gradient for each a_i
            full = _conv_fft(rev_a, w)      # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ₁‑normalised) minimising
    ``evaluate_sequence``.  A multi‑restart Adam optimiser with enforced
    symmetry is combined with a final L‑BFGS smoothing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – the previous best used 4000, we keep the same.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1.  Build a modest collection of starter sequences
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            if seq.shape == (n,):
                seq = np.clip(seq, 0.0, 1000.0)
                s = seq.sum()
                if s > 0.0:
                    seq = seq / s
                else:
                    seq = _uniform(n)
                candidates.append(seq)
        except Exception:
            pass

    # b) a few analytic families
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.001))
    candidates.append(_exp_decay(n, rate=0.005))
    candidates.append(_power_law(n, p=0.5))
    candidates.append(_power_law(n, p=0.9))
    candidates.append(_power_law(n, p=1.2))

    # c) product‑inverse‑power law (more aggressive U‑shape)
    for p in (0.3, 0.6, 0.9, 1.2):
        candidates.append(_prod_inv_power(n, p))

    # d) mixtures of a power‑law and the uniform law
    base = _power_law(n, p=0.8)
    for alpha in (0.2, 0.5, 0.8):
        mix = alpha * base + (1.0 - alpha) * _uniform(n)
        mix /= mix.sum()
        candidates.append(mix)

    # e) a handful of random Dirichlet draws
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2.  Short Adam polish for each starter (cheap exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, seq0 in enumerate(candidates):
        if time.time() > deadline - 30:
            break
        # 30 k iterations are enough to get a reasonable local optimum
        seq_opt, val_opt = _adam_optimize(seq0, n, deadline,
                                          max_iters=30_000,
                                          lr=0.20)
        print(f"[Init {idx}] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # Fallback – should never happen
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 3.  Long Adam refinement on the current best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        remaining_iters = 500_000
        seq_fine, val_fine = _adam_optimize(best_seq, n, deadline,
                                            max_iters=remaining_iters,
                                            lr=0.10)
        print(f"[Long] surrogate = {val_fine:.6f}")
        if val_fine < best_val:
            best_val, best_seq = val_fine, seq_fine.copy()

    # ------------------------------------------------------------------
    # 4.  Optional L‑BFGS smoothing (smooth‑max) if time permits
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 5:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                        max_iter_per_beta=5000)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()

    # ------------------------------------------------------------------
    # 5.  Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Result] evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next round (incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
