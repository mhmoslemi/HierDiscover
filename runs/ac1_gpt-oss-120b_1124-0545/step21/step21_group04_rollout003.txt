"""
Propose a better step‑function sequence.

The search works in three phases:

1️⃣  **Analytic families & random draws** – Generate many candidate
    probability vectors (uniform, exponential decay, power‑law, mixtures of a
    power‑law with a uniform baseline and random Dirichlet draws).  Their
    exact surrogate objective `2·n·max(conv(a,a))` (which equals the true
    `evaluate_sequence` value when the L1‑norm is 1) is measured and the best
    few are kept.

2️⃣  **Adam optimiser on a smooth max** –  Starting from each of the top
    candidates we run an Adam optimiser on a *smooth* approximation of the
    max‑convolution (log‑sum‑exp with a continuation schedule of β).  The
    optimisation is performed on the logits of a soft‑max parameterisation,
    guaranteeing non‑negativity and the simplex constraint at every step.
    The optimiser keeps the best iterate according to the exact surrogate.

3️⃣  **Final long Adam run** –  The overall best candidate is refined with a
    longer Adam run (using the remaining time budget).  The final sequence
    is evaluated with the provided `evaluate_sequence` and stored globally
    as ``height_sequence_1`` for the next iteration.

All heavy work is done with FFT‑based convolutions (`_conv_fft`), which run
in O(n log n) and are fast enough to allow millions of iterations within the
1000 s budget.
"""

# Convolution utilities (FFT based)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points (L1‑norm = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1, then normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law a_i ∝ (i+1)^{-p} + (n‑i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    return w / s if s != 0 else _uniform(n)


def _mix_power_uniform(n: int, p: float, lam: float) -> np.ndarray:
    """Mixture: lam·power_law(p) + (1‑lam)·uniform."""
    return lam * _power_law(n, p) + (1.0 - lam) * _uniform(n)


# ----------------------------------------------------------------------
# Soft‑max utilities (parameterise the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s != 0 else np.full_like(v, 1.0 / len(v))


# ----------------------------------------------------------------------
# Exact surrogate objective (used for bookkeeping)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L1=1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * len(seq) * conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_obj_grad(logits: np.ndarray, n: int, beta: float):
    """
    Returns (smooth_obj, grad_logits) where the smooth objective is
    2·n·(max_c + log_sum_exp/β) and grad_logits is the gradient on the
    log‑softmax parametrisation.
    """
    a = _softmax(logits)                     # probability vector, sum=1
    conv = _conv_fft(a, a)                   # length 2n‑1
    max_c = conv.max()

    # smooth max via log‑sum‑exp, shifted to avoid overflow
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # weights = softmax over the β‑scaled convolution values
    w = exp_beta / sum_exp                    # length 2n‑1, sum=1

    # gradient of Σ_i w_i·c_i w.r.t. a: 2·rev(a) * w
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice of length n

    # gradient through the soft‑max (logits → a)
    dot = grad_a @ a
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------
def _adam_opt(v0: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule,
              max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution.
    Returns the best probability vector (according to the exact surrogate)
    and its surrogate value.
    """
    v = v0.copy()
    m = np.zeros_like(v)          # first moment
    u = np.zeros_like(v)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # schedule handling – we move to the next β after a fixed number of iters
    schedule_len = len(beta_schedule)
    iters_per_beta = max(1, max_iters // schedule_len)
    beta_idx = 0
    beta = beta_schedule[beta_idx]

    best_a = _softmax(v)
    best_val = _objective_surrogate(best_a)

    while time.time() < deadline and t < max_iters:
        # possibly change β
        if (t > 0) and (t % iters_per_beta == 0) and (beta_idx + 1 < schedule_len):
            beta_idx += 1
            beta = beta_schedule[beta_idx]

        obj, grad_v = _smooth_obj_grad(v, n, beta)
        t += 1

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_v
        u = beta2 * u + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        u_hat = u / (1.0 - beta2 ** t)

        lr = 0.05  # base learning rate (tuned empirically)
        v -= lr * m_hat / (np.sqrt(u_hat) + eps)

        # occasional exact evaluation (every 800 steps) – cheap compared to the
        # inner work, but gives a reliable stopping criterion.
        if t % 800 == 0:
            a_cur = _softmax(v)
            val_cur = _objective_surrogate(a_cur)
            if val_cur < best_val:
                best_val = val_cur
                best_a = a_cur.copy()

    # final check
    a_final = _softmax(v)
    val_final = _objective_surrogate(a_final)
    if val_final < best_val:
        best_val, best_a = val_final, a_final.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (normalised to L¹=1) that minimizes
    the provided ``evaluate_sequence`` objective.  The method combines a
    broad analytic‑family sweep, several random Dirichlet starts and a
    continuation‑Adam optimiser on a smooth max‑convolution proxy.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer framework
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – keep compatible with any existing best sequence.
    # ------------------------------------------------------------------
    n = 4000
    # if a previous solution with another size exists we could upscale/downscale,
    # but for simplicity we stay with the same dimension.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 1. Build a rich pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best, if compatible
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.clip(arr, 0.0, 1000.0)
                s = arr.sum()
                candidates.append(arr / s if s > 0 else _uniform(n))
        except Exception:
            pass

    # b) deterministic families
    candidates.append(_uniform(n))
    candidates.append(_exp_decay(n, rate=0.003))
    for p in np.linspace(0.0, 2.0, 9):          # 0,0.25,…,2
        candidates.append(_power_law(n, p))

    # c) mixtures of power‑law + uniform (explore a 2‑D grid)
    for p in np.linspace(0.2, 4.0, 9):          # broader p‑range
        for lam in np.linspace(0.0, 1.0, 6):    # weight of the power‑law part
            candidates.append(_mix_power_uniform(n, p, lam))

    # d) random Dirichlet draws – give a diverse set of shapes
    for _ in range(12):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the top few
    # ------------------------------------------------------------------
    evaluated = []
    for seq in candidates:
        val = _objective_surrogate(seq)
        evaluated.append((val, seq))
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [seq for _, seq in evaluated[:top_k]]
    print(f"[Stage‑1] Best surrogate among raw families = {evaluated[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam refinement on the smooth max objective
    # ------------------------------------------------------------------
    # continuation schedule for β (smoothness parameter)
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1000.0, 2000.0, 5000.0, 20000.0]

    best_seq = None
    best_val = float('inf')

    # allocate roughly equal time slices to the top candidates
    remaining_time = deadline - time.time()
    per_candidate_time = remaining_time / max(1, len(top_candidates))

    for idx, seq0 in enumerate(top_candidates):
        cand_deadline = time.time() + per_candidate_time * 0.9  # a tiny safety margin
        # initialise logits from the candidate (add a tiny epsilon to avoid log(0))
        v0 = np.log(seq0 + 1e-30)

        # rough budget for this candidate – 200 k iterations is usually enough
        # for a good local optimum, but we may limit it further by the time left.
        max_iters = 250_000
        # adjust max_iters if we are running out of time
        est_iter = int((cand_deadline - time.time()) / 0.0012)
        max_iters = min(max_iters, max(30_000, est_iter))

        seq_opt, val_opt = _adam_opt(v0, n, cand_deadline,
                                    beta_schedule, max_iters)
        print(f"[Refine {idx}] surrogate after Adam = {val_opt:.6f}")

        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # ------------------------------------------------------------------
    # 4. Final long Adam run on the current best (if we still have time)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    if time.time() < deadline - 1.0:
        v0 = np.log(best_seq + 1e-30)
        # use the remaining budget for a thorough refinement
        rem_iters = int((deadline - time.time()) / 0.0012)
        seq_fine, val_fine = _adam_opt(v0, n, deadline,
                                       beta_schedule, max_iters=rem_iters)
        print(f"[Fine‑tune] final surrogate after long Adam = {val_fine:.6f}")
        if val_fine < best_val:
            best_val, best_seq = val_fine, seq_fine.copy()

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run (helps incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
