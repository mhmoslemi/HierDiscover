"""
Optimiser for the step‑function bound – LP + Mirror descent

The optimiser works in three phases:

1️⃣ **Seeding** – generate a handful of deterministic families
   (uniform, triangular, sine‑window, Gaussian, beta‑shapes,
   left/right exponentials) and a few random Dirichlet draws.
   All candidates are ℓ₁‑normalised and the best one (by the exact
   surrogate 2·n·max(conv)) is kept.

2️⃣ **Iterative LP improvement** – given a current vector *f* we compute
   *M = max(conv(f,f))*.
   The linear program

        maximise    Σ g_j
        subject to  f * g ≤ M   (point‑wise)
                    g_j ≥ 0

   has a cheap sparse formulation (the convolution matrix of *f* is
   the constraint matrix).  The LP enlarges the ℓ₁‑mass of *g* while
   keeping the same convolution bound *M*.  After solving we rescale
   *g* to have unit ℓ₁‑norm and replace

        f ← (1−t)·f + t·g          (t=0.5)

   which strictly reduces the bound in practice.  The loop runs as
   long as time permits (≈30 LP‑iterations for n=2000).

3️⃣ **Polishing** – a few hundred thousand steps of exponentiated‑gradient
   mirror‑descent on the exact max‑convolution (full‑gradient, not the
   smooth surrogate) are performed from the best LP‑result.

The method respects the 1000 s budget, stores the best sequence in the
global variable ``height_sequence_1`` and finally returns it as a list
of floats.
"""

import time
import math
import numpy as np

# optional imports ---------------------------------------------------------
try:
    from scipy.optimize import linprog
    import scipy.sparse as sp
except Exception:                 # pragma: no cover
    linprog = None
    sp = None

# -------------------------------------------------------------------------
# basic numeric utilities
# -------------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate when Σ seq = 1 : 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# -------------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# -------------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# -------------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑conv.
# -------------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)   # stay safely in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active indices (those attaining the max)
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of max‑convolution
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # learning rate schedule
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# -------------------------------------------------------------------------
# Build the convolution matrix for a given left vector f (sparse)
# -------------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """Return a sparse CSR matrix A where A @ g = conv(f,g)."""
    n = f.shape[0]
    N = n * n
    rows = np.empty(N, dtype=np.int32)
    cols = np.empty(N, dtype=np.int32)
    data = np.empty(N, dtype=np.float64)

    idx = 0
    arange_n = np.arange(n, dtype=np.int32)
    for i in range(n):
        rows[idx:idx + n] = i + arange_n          # k = i + j
        cols[idx:idx + n] = arange_n              # column = j
        data[idx:idx + n] = f[i]                  # coeff = f_i
        idx += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    stores the best found sequence as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # tiny safety margin

    # --------------------------------------------------------------
    # problem size – a modest size makes the LP cheap and still gives
    # a good bound (the theoretical limit is ~1 for any n).
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = arr.shape[0]
            if n > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 2000                     # a size that balances speed & quality
    else:
        n = prev_arr.shape[0]

    print(f"\n--- Optimisation started (n={n}) ---")

    # --------------------------------------------------------------
    # 1. Seed generation
    # --------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.35))

    # beta families (incl. U‑shapes)
    for alpha, beta in [(0.2, 0.2), (0.2, 1.0), (1.0, 0.2), (0.5, 0.5), (1.0, 1.0)]:
        seeds.append(_beta_shape(n, alpha, beta))

    # left/right exponentials
    for rate in [0.0003, 0.0005, 0.001, 0.002, 0.005]:
        seeds.append(_exp_decay_left(n, rate))
        seeds.append(_exp_decay_right(n, rate))

    # random Dirichlet draws – explore sparse / dense regimes
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_arr is not None:
        seeds.append(prev_arr)

    # ----------------------------------------------------------------
    # 2. Fast evaluation of seeds, keep the best one
    # ----------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        seq = vec / s
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    print(f"[Init] best surrogate from seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Iterative LP improvement (if SciPy is available)
    # --------------------------------------------------------------
    if linprog is not None and sp is not None:
        f = best_seq.copy()
        global_best_seq = f.copy()
        global_best_val = best_val

        max_lp_iters = 30                     # safeguard for the time budget
        lp_iter = 0
        while time.time() < deadline and lp_iter < max_lp_iters:
            conv_f = _conv_fft(f, f)
            M = conv_f.max()

            # build the (sparse) convolution matrix A where A @ g = conv(f,g)
            A = _build_conv_matrix(f)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            # linear program: maximise Σ g_j  <=> minimise -Σ g_j
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n)]

            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})
            except Exception as exc:            # pragma: no cover
                print(f"[LP] exception {exc}")
                break

            if not res.success:
                print(f"[LP] failed: {res.message}")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g <= 0.0:
                print("[LP] produced zero ℓ₁‑mass, aborting")
                break

            # normalise g to unit ℓ₁‑norm
            g_norm = g / sum_g

            # convex combination – t=0.5 gives a solid descent step
            t = 0.5
            f_new = (1.0 - t) * f + t * g_norm

            val_new = _objective_surrogate(f_new)
            print(f"[LP iter {lp_iter+1}] surrogate = {val_new:.6f}")

            if val_new < global_best_val:
                global_best_val = val_new
                global_best_seq = f_new.copy()
                print(f"   *** new global best = {global_best_val:.6f}")

            # prepare for next iteration
            f = f_new
            lp_iter += 1

        # --------------------------------------------------------------
        # 4. Final polishing with exponentiated‑gradient (full gradient)
        # --------------------------------------------------------------
        if time.time() < deadline:
            print("\n--- Polishing with exponentiated‑gradient ---")
            final_seq, final_val = _exp_gradient_opt(
                global_best_seq,
                n,
                deadline,
                max_iters=600_000,
                eta0=6.0,
                single_active=False
            )
            if final_val < global_best_val:
                global_best_seq = final_seq
                global_best_val = final_val
                print(f"[Polish EG] improved to {global_best_val:.6f}")

        best_seq = global_best_seq
    else:
        # fallback – use the best seed directly
        best_seq = best_seq

    # -----------------------------------------------------------------
    # 5. Final normalisation & evaluation
    # -----------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / s

    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {_objective_surrogate(best_seq):.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
