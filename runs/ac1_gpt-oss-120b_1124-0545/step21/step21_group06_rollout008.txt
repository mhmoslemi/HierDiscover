Goal
----
Minimise  2·n·max(conv(a,a)) / (∑a)²   over non‑negative vectors a.
The objective is homogeneous, therefore all vectors are ℓ₁‑normalised

Main ideas

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (used for fast seeding & all local optimisation)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalized vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    """a_i = (1 - cos(2πi/(n‑1))) * α  ; normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) * alpha
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) (decay from the left)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·(n‑1‑i)) (decay from the right)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a point from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      eps_rel: float = 1e-12):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)   # stay in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # --- active set (all indices within eps_rel of the maximum) ---
        eps = max_val * eps_rel
        active = np.where(conv >= max_val - eps)[0]

        # --- sub‑gradient for max(conv) ---
        # For each active k we add the shifted copy of a.
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx_i
            g[idx_i] += a[rev]

        g = 2.0 * g / max(active.size, 1)   # average over the active set

        # --- learning‑rate (standard diminishing) ---
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4_000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:                 # pragma: no cover
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0,
                         5_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0,
                         1_000_000.0]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # log‑sum‑exp safely
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp               # weights summing to 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # middle n entries

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_x):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.02,
                         T_end: float = 1e-8):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny coordinate‑swap hill‑climber (deterministic local search)
# ----------------------------------------------------------------------
def _coordinate_swap_hill_climb(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               step: float = 1e-5,
                               max_passes: int = 5):
    """
    Repeatedly moves a tiny amount of probability mass from any index i
    to any j (i≠j) if the surrogate value improves.
    Very cheap – runs until time expires or a few passes are finished.
    """
    a = start_seq.copy()
    best_val = _objective_surrogate(a)

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # random order of (i,j) pairs helps escaping shallow plateaus
        idx = np.arange(n)
        rng = np.random.default_rng()
        rng.shuffle(idx)
        for i in idx:
            if a[i] < step:
                continue
            for j in idx:
                if i == j:
                    continue
                # propose moving 'step' from i to j
                a[i] -= step
                a[j] += step
                val = _objective_surrogate(a)
                if val < best_val:
                    best_val = val
                    improved = True
                else:
                    # revert
                    a[i] += step
                    a[j] -= step
            if time.time() > deadline:
                break
        if not improved:
            break
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # keep a safety margin (2 s) for the final evaluation
    deadline = t0 + budget_s - 2.0

    # --------------------------------------------------------------
    # Use previous solution if it exists, otherwise pick a default size
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    if prev_arr is None:
        n = 4000
    else:
        n = prev_arr.shape[0]

    print(f"\n--- Optimisation started (n={n}) ---")

    # --------------------------------------------------------------
    # 1.  Seeding – build a rich library of candidate shapes
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_window(n, alpha=1.0))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))

    # beta‑shapes (grid)
    alphas = np.linspace(0.2, 8.0, 12)
    betas = np.linspace(0.2, 8.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # monotone exponentials
    rates = np.logspace(-4, -0.2, 25)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # power‑law families
    ps = np.linspace(0.0, 5.0, 12)
    for p in ps:
        seeds.append(_power_law(n, p))

    # random Dirichlet draws – sample many different sparsities
    concs = (0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2)
    for conc in concs:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures with the uniform vector (adds asymmetry)
    for r in rates[::5]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for alpha_mix in (0.15, 0.35, 0.55):
            seeds.append(_mix_shapes(_uniform(n), exp_l, alpha_mix))
            seeds.append(_mix_shapes(_uniform(n), exp_r, alpha_mix))

    # perturb previous best (if any)
    if prev_arr is not None:
        base = prev_arr.copy()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds could be generated before the deadline.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate scan:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start local refinement on the top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # time budget for this seed (roughly 8 % of remaining time)
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(4.0, remaining * 0.08 / max(1, top_k - si))

        # ---- (a) exponentiated‑gradient on the exact max (full active set) ----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=300_000,
                                           eta0=7.0,
                                           eps_rel=1e-10)

        # ---- (b) smooth‑max L‑BFGS refinement ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              100_000.0, 200_000.0,
                                                              500_000.0, 1_000_000.0],
                                               max_iter_per_beta=4_000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ---- (c) simulated annealing -------------------------------------------------
        hill_deadline = sub_deadline - 0.5
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=200_000,
                                                  start_sigma=0.002,
                                                  T_start=0.018,
                                                  T_end=2e-8)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # ---- (d) tiny coordinate‑swap local improvement -------------------------------
        if time.time() < hill_deadline:
            seq_swap, val_swap = _coordinate_swap_hill_climb(seq_sa,
                                                             n,
                                                             hill_deadline,
                                                             step=5e-6,
                                                             max_passes=3)
        else:
            seq_swap, val_swap = seq_sa, val_sa

        # pick the best among the refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
            (seq_swap, val_swap)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final polishing – intensive runs on the current global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive EG polishing (full active set) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=6.0,
                                                 eps_rel=1e-12)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing (high‑β, many iters) ---")
        final_seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                               200.0, 500.0, 1_000.0,
                                                               2_000.0, 5_000.0,
                                                               20_000.0, 50_000.0,
                                                               100_000.0, 200_000.0,
                                                               500_000.0, 1_000_000.0,
                                                               2_000_000.0],
                                                max_iter_per_beta=6_000)
        final_val_lbfgs = _objective_surrogate(final_seq_lbfgs)
        if final_val_lbfgs < global_best_val:
            global_best_seq = final_seq_lbfgs
            global_best_val = final_val_lbfgs
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=400_000,
                                                          start_sigma=0.001,
                                                          T_start=4e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
