def _adam_opt(seq, deadline, max_iters, beta_schedule, lr=0.2):

  v = np.log(seq + 1e-30)
  m = np.zeros_like(v)
  v2 = np.zeros_like(v) # second moment
  t = 0

  for beta in beta_schedule:

    for it in range(max_iters_per_beta):

     if time.time() > deadline: break

     # compute gradient of smooth objective at current beta
     a = np.exp(v); a /= a.sum()
     conv = _conv_fft(a, a)
     max_c = np.max(conv)
     shift = max_c
     exp_beta = np.exp(beta * (conv - shift))
     sum_exp = exp_beta.sum()
     # smooth objective (not needed)
     w = exp_beta / sum_exp
     rev_a = a[::-1]
     full = _conv_fft(rev_a, w)   # length = 3n-2
     grad_a = 2.0 * full[n-1:n-1+n]   # correct slice
     # project gradient to simplex tangent
     dot = np.dot(grad_a, a)
     grad_v = a * (grad_a - dot)

     # Adam updates
     t += 1
     m = beta1 * m + (1 - beta1) * grad_v
     v2 = beta2 * v2 + (1 - beta2) * (grad_v**2)
     m_hat = m / (1 - beta1**t)
     v2_hat = v2 / (1 - beta2**t)
     v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-8)

  after loops, return final a (exp(v)/sum)

"""Hybrid optimiser for the step‑function problem.

The optimiser works in three phases:

1. **Rich seed generation** – a large, diverse pool of analytically
   parametrised families (uniform, several exponentials, symmetric
   exponentials, power‑laws, beta PDFs, Dirichlet draws with a wide
   range of concentration parameters and sparse “spike’’ vectors) is
   built.  Every candidate is ℓ₁‑normalised (∑aᵢ = 1) and its exact
   surrogate – 2·n·max(conv(a,a)) – is evaluated.  The best *k* seeds
   are retained for full optimisation.

2. **Intensive mirror‑descent + L‑BFGS smoothing** – each retained seed
   is polished with a long run of exponentiated‑gradient (mirror‑descent
   in the log‑softmax domain) on the exact max‑convolution.  After the
   EG phase a smoothed‑max L‑BFGS (log‑sum‑exp surrogate) is applied
   while time permits.  The result with the lowest surrogate becomes
   the current global best.

3. **Final refinement** – the current global best is fed to a final,
   long EG run that uses almost the whole remaining budget, followed
   by a short permutation‑swap annealing and a very brief mass‑transfer
   annealing to escape any remaining ordering traps.

All heavy operations (full convolution) are performed with a real FFT
implementation (`_conv_fft`).  The routine respects a global 1000 s
budget, checks the deadline frequently and stores the best sequence in
the global name ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# optional L‑BFGS support
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    cen = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - cen))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _spike_distribution(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries, each = 1/k, placed at random positions."""
    if k <= 0:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent in the log‑softmax domain.
    Returns the best sequence seen and its surrogate value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact max‑convolution and its active diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see problem statement)
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decreasing like 1/sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # projection onto the simplex

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Minimise a smoothed version of max(conv) using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max
            w = exp_beta / sum_exp                     # shape 2n‑1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[len(a) - 1: len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Permutation‑swap annealing (swap two entries)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser by swapping two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]
        val = _surrogate(ns)

        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny mass‑transfer annealing (move a bit of mass from i to j)
# ----------------------------------------------------------------------
def _mass_transfer_anneal(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from one index to another."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] == 0.0:
            continue
        # take a random fraction (0‑1) of the mass at i
        frac = rng.random() * 0.5          # at most 50 % of the source mass
        delta = cur_seq[i] * frac
        ns = cur_seq.copy()
        ns[i] -= delta
        ns[j] += delta

        # renormalise (should stay on the simplex automatically)
        val = _surrogate(ns)

        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    The routine respects the global time budget and stores the best
    sequence in the global name ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    # problem size – 4000 provides a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution for continuation (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    # many exponential rates (including very slow decay)
    for rate in (0.0002, 0.0004, 0.0006, 0.0008, 0.0010,
                 0.0015, 0.0020, 0.0030, 0.0040):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law shapes
    for p in (0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_power_law(n, p))

    # beta PDFs – a few random draws
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with a wide concentration range (including tiny values)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse spike vectors (k active entries)
    for k in (5, 10, 20, 40, 80, 160):
        seeds.append(_spike_distribution(n, k, rng))

    # include previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(8):
            seeds.append(np.random.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair share of the remaining time
        remaining = deadline - time.time()
        share = remaining * 0.13 / max(1, (top_k - si))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # (a) exponentiated‑gradient polishing
        seq_eg, val_eg = _exp_gradient_opt(seed_vec,
                                           sub_deadline,
                                           max_iters=2_200_000,
                                           eta0=7.0)

        # (b) optional L‑BFGS smoothing
        if time.time() < sub_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_eg,
                                            sub_deadline,
                                            max_iter_per_beta=3000)
            val_lb = _surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_eg, val_eg

        # (c) permutation annealing
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(seq_lb,
                                             sub_deadline,
                                             max_iters=300_000)
        else:
            seq_perm, val_perm = seq_lb, val_lb

        # pick the best local result
        local_best_seq, local_best_val = min(
            [(seq_eg, val_eg), (seq_lb, val_lb), (seq_perm, val_perm)],
            key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use virtually all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (full remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=4_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Short permutation + mass‑transfer tweaks
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 30.0
        seq_perm, val_perm = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=100_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[Final perm] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_mt, val_mt = _mass_transfer_anneal(global_best_seq.copy(),
                                                short_deadline,
                                                max_iters=80_000)
        if val_mt < global_best_val:
            global_best_seq = seq_mt
            global_best_val = val_mt
            print(f"[Final mass‑transfer] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
