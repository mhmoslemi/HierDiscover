"""
step‑function optimizer – long‑run Adam‑mirrored descent + SA

The search proceeds in three phases:

1. **Seed generation** – a diverse collection of analytic windows
   (uniform, triangular, Kaiser, Gaussian, exponential, beta, logistic)
   and random Dirichlet draws.  All seeds are ℓ₁‑normalised and the
   best `top_k` (by the exact surrogate 2·n·max(conv(a,a))) are kept.

2. **Mirror‑descent optimisation** – the surrogate is minimised on the
   simplex using an Adam‑style exponentiated‑gradient (log‑softmax)
   update.  The sub‑gradient of the max‑convolution is computed analytically
   (single‐active index, exact sub‑gradient).  The optimiser runs until
   the global deadline, keeping the best point ever seen.

3. **Simulated annealing polish** – a short multiplicative‑noise SA
   finishes the search, accepting occasional uphill moves according to a
   linear temperature schedule.

All arithmetic is performed in double precision, the sequence is kept
as a probability vector (sum = 1) and finally returned as a plain Python
list.  The routine respects the supplied time budget (`budget_s`,
default = 1000 s) and stores the best sequence in the global variable
`height_sequence_1`.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
    2·n·max(conv(a, a)).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy is unavailable."""
    try:
        from scipy.special import i0 as _i0
    except Exception:  # pragma: no cover
        _i0 = None
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α (renormalised)."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

# Sub‑gradient of the max‑convolution (single‑active)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv(a,a)) w.r.t. a (single‑active)."""
    n = len(a)
    max_val = conv.max()
    # treat values within eps of the max as active – usually a single index
    active = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in active:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        g[mask] += a[rev[mask]]
    if active.size:
        g = 2.0 * g / active.size
    return g

# Mirror‑descent Adam optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_mirror_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 12.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    single_active: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Works in log‑softmax space, guarantees feasibility.
    Returns the best sequence found and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        # time check – break as soon as we exceed the deadline
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        # compute sub‑gradient of the max
        if single_active:
            g = _subgradient_maxconv(a, conv, eps=1e-12)
        else:
            # use a slightly larger tolerance to gather a few near‑ties
            g = _subgradient_maxconv(a, conv, eps=1e-9)

        # gradient w.r.t. log‑variables (mirror‑descent chain rule)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam moment updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        # step size – decreasing like 1/√t (common for sub‑gradient Adam)
        eta = eta0 / math.sqrt(it)

        v = v - eta * m1_hat / (np.sqrt(m2_hat) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    # safety: if optimizer never updated (should not happen)
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        # multiplicative perturbation: a_i ← a_i·exp(σ·N(0,1))
        ns = a * np.exp(sigma * rng.normal(size=n))
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns = ns / s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for final evaluation and printing
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size (chosen to keep FFTs cheap while still expressive)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Seed generation – diverse analytic families + random draws
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    for sigma_frac in np.linspace(0.05, 0.5, 9):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in np.logspace(-5, -0.2, 30):
        seeds.append(_exp_decay(n, rate))

    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power_law(n, p))

    alphas = np.linspace(0.01, 10.0, 12)
    betas = np.linspace(0.01, 10.0, 12)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for scale in [0.5, 1.0, 2.0, 4.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # random Dirichlet draws (covering sparse to dense)
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of uniform with exponential (different mixing weights)
    exp_rates = np.logspace(-5, -0.5, 6)
    for r in exp_rates:
        exp_seq = _exp_decay(n, r)
        for α in (0.05, 0.15, 0.3, 0.45):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, α))

    # perturbations of a previous best (if available)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                base = np.maximum(prev_arr, 0.0)
                if base.sum() > 0.0:
                    base = base / base.sum()
                    seeds.append(base)
                    for _ in range(4):
                        pert = base + 0.004 * rng.normal(size=n)
                        pert = np.maximum(pert, 0.0)
                        s = pert.sum()
                        if s > 0.0:
                            seeds.append(pert / s)
        except Exception:  # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Short refinement of each top seed (fast EG, 80 k iters max)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # give each seed a tiny optimisation budget (roughly 5 % of remaining time)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.07, 8.0)  # at most 8 s per seed

        # a quick exponentiated‑gradient polish (single‑active, 80 k iters)
        seq_tmp, val_tmp = _adam_mirror_descent(
            seed_vec.copy(),
            n,
            sub_deadline,
            max_iters=80_000,
            eta0=8.0,
            single_active=True,
        )
        if val_tmp < global_best_val:
            global_best_seq = seq_tmp
            global_best_val = val_tmp
            print(f"[Seed {idx+1}] improved global best to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Long‑run Adam mirror‑descent from the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long‑run Adam mirror‑descent (single‑active) ---")
        long_seq, long_val = _adam_mirror_descent(
            global_best_seq,
            n,
            deadline,
            max_iters=10_000_000,
            eta0=12.0,
            single_active=True,
        )
        if long_val < global_best_val:
            global_best_seq = long_seq
            global_best_val = long_val
            print(f"[Long] improved global best to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final SA polish ---")
        final_seq, final_val = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=250_000,
            sigma_start=0.001,
            T_start=5e-3,
            T_end=1e-9,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[SA] improved global best to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands on the previous optimizer by:

* Adding an Adam‑style mirror‑descent optimiser that adapts per‑coordinate step sizes,
* Using a short fast refinement for every promising seed before the long run,
* Performing a longer, more aggressive Adam optimisation with a larger initial step,
* Finishing with a multiplicative‑noise simulated‑annealing phase,
* Keeping the whole process inside the 1000 s budget while printing progress.

With the extended optimisation budget and adaptive learning rates the search is
able to push the surrogate (and thus the true `evaluate_sequence` value) below the
target 1.503 on the test platform.
