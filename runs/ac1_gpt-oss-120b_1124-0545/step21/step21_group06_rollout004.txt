"""
Optimiser for the step‑function bound (target < 1.503).

The algorithm builds a rich library of analytic “windows” (Gaussian,
exponential, beta‑shape, Kaiser, Hann, Hamming, Blackman, U‑shaped,
etc.) together with many random Dirichlet draws and shuffled variants.
All candidates are ℓ₁‑normalised and scored by a cheap surrogate
``2·n·max(conv(a,a))``.

The top‐k seeds are refined by a three‑stage pipeline:

1. cheap exponentiated‑gradient (EG) with a *single* active index
   (fast descent toward a good region);
2. smooth‑max L‑BFGS optimisation (log‑sum‑exp surrogate) with a
   long β‑schedule;
3. a *full‑active* EG run that directly follows the exact max‑convolution
   sub‑gradient (slow but accurate).

After all seeds are processed the current global best undergoes an
intensive polishing stage (full‑active EG → L‑BFGS → high‑budget
simulated‑annealing).  The final sequence is returned and stored in the
global variable ``height_sequence_1`` for reuse.

The implementation respects the 1000 s budget (2 s safety margin),
uses only NumPy (SciPy is optional for L‑BFGS) and runs on a single core.
"""

import time
import math
import numpy as np

# optional SciPy import for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (fast, assumes ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    s = v.sum()
    if s == 0.0:
        return v
    return v / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return _normalize(a)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return _normalize(a)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )
    return _normalize(a)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    a = np.kaiser(n, beta)
    return _normalize(a)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return _normalize(a)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return _normalize(a)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    return _normalize(a)


def _U_shape(n: int, rate: float) -> np.ndarray:
    """High at both ends, low in the middle (U‑shape)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * dist)
    return _normalize(a)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _normalize(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _normalize(a)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return _normalize(mixed)


def _shuffle_shape(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    perm = rng.permutation(len(a))
    return a[perm].copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient accumulation
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev = k - idx_i
            g_tot[idx_i] += a[rev]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()          # keep on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            conv = _conv_fft(a, a)
            shift = conv.max()
            # smooth max via log‑sum‑exp (numerically stable)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # derivative of softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] unexpected error at β={beta_val}: {e}")
            break

    a_opt = np.exp(v)
    a_opt /= a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """Simple SA with Gaussian perturbations and a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        tfrac = attempts / max_attempts
        T = T_start * (1.0 - tfrac) + T_end * tfrac
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0   # small safety gap

    # ------------------------------------------------------------------
    # Determine problem size (reuse previous solution if present)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = arr.shape[0]
            if n > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 4000                     # modest default size
    print(f"\n=== Optimisation started (n={n}) ===")

    # ------------------------------------------------------------------
    # 1. Generate a rich pool of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_gaussian(n, sigma_frac=0.35))

    # Kaiser windows – a handful of betas
    for beta in (2.0, 5.0, 8.0, 12.0, 20.0):
        seeds.append(_kaiser_window(n, beta))

    # exponential decays (left / right) & U‑shape
    rates = np.logspace(-4, -0.2, 28)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))
        seeds.append(_U_shape(n, r))

    # power‑law family
    for p in np.linspace(0.0, 4.5, 12):
        seeds.append(_power_law(n, p))

    # beta‑shape grid (coarse)
    alphas = np.linspace(0.2, 6.0, 12)
    betas = np.linspace(0.2, 6.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform with exponentials (adds asymmetry)
    for r in rates[::5]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for w in (0.2, 0.4, 0.6):
            seeds.append(_mix_shapes(_uniform(n), exp_l, w))
            seeds.append(_mix_shapes(_uniform(n), exp_r, w))

    # random Dirichlet draws (various concentration parameters)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # shuffled variants of a few strong bases
    strong_bases = [seeds[0], seeds[3], seeds[6], seeds[11]]
    for base in strong_bases:
        for _ in range(3):
            seeds.append(_shuffle_shape(base, rng))

    # perturb previous best (if available)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr.copy())
        for _ in range(6):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(_normalize(pert))

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of seed pool
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        vec = _normalize(vec)
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate screening:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement (fast EG → L‑BFGS → full‑active EG → SA)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of remaining time to this seed
        remaining = deadline - time.time()
        fraction = 0.12 / max(1, top_k - si)   # roughly equal sharing
        sub_deadline = time.time() + max(3.0, remaining * fraction)

        # (a) quick EG with SINGLE active (speed)
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=180_000,
                                           eta0=8.0,
                                           single_active=True)

        # (b) smooth max L‑BFGS (more accurate)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[
                                                   1.0, 5.0, 20.0, 80.0, 200.0,
                                                   500.0, 1_000.0, 2_000.0,
                                                   5_000.0, 20_000.0,
                                                   50_000.0, 100_000.0,
                                                   200_000.0, 500_000.0,
                                                   1_000_000.0, 5_000_000.0
                                               ],
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # (c) full‑active EG for precise descent
        if time.time() < sub_deadline:
            seq_full_eg, val_full_eg = _exp_gradient_opt(seq_lbfgs.copy(),
                                                         n,
                                                         sub_deadline,
                                                         max_iters=120_000,
                                                         eta0=5.0,
                                                         single_active=False)
        else:
            seq_full_eg, val_full_eg = seq_lbfgs, val_lbfgs

        # (d) a light SA pass
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_full_eg,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=130_000,
                                                  start_sigma=0.003,
                                                  T_start=0.015,
                                                  T_end=5e-8)
        else:
            seq_sa, val_sa = seq_full_eg, val_full_eg

        # pick best among the refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_full_eg, val_full_eg),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1:02d}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f" *** New global best = {global_best_val:.6f}")

        # early exit if we beat the target early
        if global_best_val < 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 5. Polishing stage on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Polishing: full‑active EG (long run) ---")
        polish_deadline = time.time() + min(200.0, deadline - time.time() - 100.0)
        polished_seq, polished_val = _exp_gradient_opt(global_best_seq.copy(),
                                                       n,
                                                       polish_deadline,
                                                       max_iters=2_000_000,
                                                       eta0=6.0,
                                                       single_active=False)
        if polished_val < global_best_val:
            global_best_seq = polished_seq
            global_best_val = polished_val
            print(f"[Polish EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Polishing: high‑budget L‑BFGS ---")
        lbfgs_deadline = time.time() + min(150.0, deadline - time.time() - 80.0)
        global_best_seq = _lbfgs_smooth_optimize(global_best_seq,
                                                 n,
                                                 lbfgs_deadline,
                                                 beta_schedule=[
                                                     1.0, 5.0, 20.0, 80.0,
                                                     200.0, 500.0, 1_000.0,
                                                     2_000.0, 5_000.0,
                                                     20_000.0, 50_000.0,
                                                     100_000.0, 200_000.0,
                                                     500_000.0, 1_000_000.0,
                                                     5_000_000.0,
                                                     1e7
                                                 ],
                                                 max_iter_per_beta=3500)
        global_best_val = _objective_surrogate(global_best_seq)
        print(f"[Polish L‑BFGS] surrogate = {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (very high budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=400_000,
                                                          start_sigma=0.001,
                                                          T_start=5e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function and store result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Completed ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
