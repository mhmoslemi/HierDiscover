import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y=None):
    if y is None:
        y = x
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _softmax(z):
    m = np.max(z)
    e = np.exp(z - m)
    s = e.sum()
    if s == 0:
        return np.full_like(z, 1.0/len(z))
    return e/s

def _objective(seq, n):
    # Ensure non‑negative, normalised
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total <= 0:
        return np.inf
    a = seq / total
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

def _uniform(n):
    return np.full(n, 1.0/n, dtype=np.float64)

def _arcsine(n):
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0/np.sqrt((i+0.5)*(n-0.5-i)+eps)
    a /= a.sum()
    return a

def _triangular(n):
    half = n//2
    if n%2==0:
        left = np.arange(1, half+1)
        right = left[::-1]
    else:
        left = np.arange(1, half+2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def _beta_distribution(n, alpha, beta):
    # draws β(α,β) density on grid [0,...,n-1]
    i = np.arange(n, dtype=np.float64)/(n-1.0)
    eps = 1e-12
    a = np.power(i+eps, alpha-1.0) * np.power(1.0-i+eps, beta-1.0)
    a /= a.sum()
    return a

def _mixture_beta_uniform(n, w, alpha, beta):
    """ convex combination w * Beta(alpha,beta) + (1-w) * Uniform """
    return w*_beta_distribution(n, alpha, beta) + (1.0-w)*_uniform(n)

def _mixture_two_beta(n, w, alpha1, beta1, alpha2, beta2):
    """ w * Beta(alpha1,beta1) + (1-w) * Beta(alpha2,beta2) """
    return w*_beta_distribution(n, alpha1, beta1) + (1.0-w)*_beta_distribution(n, alpha2, beta2)

def _exp_gradient_opt(start_seq, n, deadline, max_iters=200_000, eta0=2.0):
    # Mirror‑descent on the exact max‑convolution.
    a = np.maximum(start_seq, 0.0)
    if a.sum() == 0:
        a = _uniform(n)
    a = a / a.sum()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective(a, n)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # gradient for max position k
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)
        cur = _objective(a, n)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
    return best_seq, best_val

def _random_local_search(start_seq, n, deadline, attempts=100_000, sigma=0.02):
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective(best_seq, n)
    attempts_done = 0
    while attempts_done < attempts and time.time() < deadline:
        attempts_done += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective(new_seq, n)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

"""
Propose a candidate step‑function that minimises ``evaluate_sequence``.

Strategy
--------
0.  Load a previously‑found sequence (if it matches the current length).
1.  Build a rich pool of analytic seeds:
    • uniform, arcsine, triangular,
    • a dense grid of symmetric Beta(α,α) distributions,
    • random convex mixtures of a Beta with the uniform distribution,
    • random convex mixtures of two independent Betas.
2.  Keep the best few (elite) seeds.
3.  For each elite seed run a dedicated exponentiated‑gradient (mirror‑descent)
    on the *exact* max‑convolution objective for the time that remains.
4.  Finish with a lightweight random local search that moves a tiny amount of
    mass from one coordinate to another.
5.  Return the globally best sequence and store it in the global variable
    ``height_sequence_1`` for the next call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    if y is None:
        y = x
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact – equal to ``evaluate_sequence`` for a normalised vector)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, n: int) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)), assuming seq is non‑negative."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total <= 0.0:
        return np.inf
    a = seq / total
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1} (normalised)."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a /= a.sum()
    return a


def _mixture_beta_uniform(n: int, w: float, alpha: float, beta: float) -> np.ndarray:
    """Convex combination w·Beta(α,β) + (1‑w)·Uniform (already normalised)."""
    return w * _beta_distribution(n, alpha, beta) + (1.0 - w) * _uniform(n)


def _mixture_two_beta(
    n: int, w: float, alpha1: float, beta1: float, alpha2: float, beta2: float
) -> np.ndarray:
    """Convex combination w·Beta(α₁,β₁) + (1‑w)·Beta(α₂,β₂)."""
    return w * _beta_distribution(n, alpha1, beta1) + (1.0 - w) * _beta_distribution(
        n, alpha2, beta2
    )


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = np.maximum(start_seq, 0.0)
    if a.sum() == 0.0:
        a = _uniform(n)
    a = a / a.sum()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective(a, n)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of (a*a)[k] w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _random_local_search(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    attempts: int = 120_000,
    sigma: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small random amount
    from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective(best_seq, n)

    att = 0
    while att < attempts and time.time() < deadline:
        att += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective(new_seq, n)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search runs for at most ``budget_s`` seconds and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size (default 4000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)

                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣  Gather an initial pool of deterministic seeds
    # ------------------------------------------------------------------
    seed_seqs = [_uniform(n), _arcsine(n), _triangular(n)]
    if prev_arr is not None:
        seed_seqs.append(prev_arr)

    # ------------------------------------------------------------------
    # 1️⃣  Systematic search over symmetric Beta(α,α) densities
    # ------------------------------------------------------------------
    print("\n--- Structured Beta grid search ---")
    best_seq = None
    best_val = np.inf

    alphas = np.concatenate(
        [
            np.linspace(0.05, 0.5, 10, endpoint=False),

        ]
    )
    for a in alphas:
        if time.time() > deadline:
            break
        seq = _beta_distribution(n, a, a)
        val = _objective(seq, n)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        seed_seqs.append(seq)

    print(f"[Beta grid] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random mixtures of a Beta and the uniform distribution
    # ------------------------------------------------------------------
    print("\n--- Random Beta‑Uniform mixture search ---")
    mix_trials = 1200
    for t in range(mix_trials):
        if time.time() > deadline:
            break
        w = rng.random()

        beta = 10 ** rng.uniform(-2, 1)   # β ∈ [0.01, 10]
        seq = _mixture_beta_uniform(n, w, alpha, beta)
        val = _objective(seq, n)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        if t % 200 == 0:
            seed_seqs.append(seq)

    # ------------------------------------------------------------------
    # 3️⃣  Random mixtures of two independent Beta distributions

    for t in range(double_trials):
        if time.time() > deadline:
            break
        w = rng.random()
        a1 = 10 ** rng.uniform(-2, 1)
        b1 = 10 ** rng.uniform(-2, 1)
        a2 = 10 ** rng.uniform(-2, 1)
        b2 = 10 ** rng.uniform(-2, 1)
        seq = _mixture_two_beta(n, w, a1, b1, a2, b2)
        val = _objective(seq, n)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        if t % 200 == 0:
            seed_seqs.append(seq)

    # ------------------------------------------------------------------
    # 4️⃣  Keep the globally best from the initial stage
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq, n)
    print(f"\n[Overall init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Pick a handful of elite seeds for dedicated polishing
    # ------------------------------------------------------------------

    elite.sort(key=lambda p: p[0])
    elite = elite[:5]  # keep at most five promising starts

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
