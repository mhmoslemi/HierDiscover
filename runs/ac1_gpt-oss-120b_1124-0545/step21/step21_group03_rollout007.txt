    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c

"""
Improved optimiser for the non‑negative step‑function problem.

The idea is simple:
  1. Generate a diverse pool of ℓ₁‑normalised candidate vectors (analytic
     families, Dirichlet draws, and permutations of the best known solution).
  2. Keep the few best seeds according to the exact surrogate
     ``2·n·max(conv(seq,seq))``.
  3. Starting from the single best seed, run a *smooth‑max* Adam optimiser
     directly in the log‑softmax (probability) space.  The smooth‑max
     (log‑sum‑exp) approximates the hard max, providing a well‑behaved gradient
     that respects the non‑negativity and ℓ₁ constraints.
  4. After the smooth optimisation finishes, perform a short permutation‑
     based simulated‑annealing phase to escape possible ordering traps.
  5. Evaluate the final vector with the supplied ``evaluate_sequence`` and
     store it globally as ``height_sequence_1`` for continuation runs.

All heavy work (full linear convolution) is performed with a real FFT.
The algorithm respects the global time budget (default 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (real input → real output)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with all parameters = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    """Coarse piecewise‑linear shape with ``num_breaks`` interior break points."""
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (log‑softmax mirror descent)
# ----------------------------------------------------------------------
def _smooth_max_adam(start_seq: np.ndarray,
                     deadline: float,
                     lr: float = 0.12,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of max(conv).
    Works in the log‑softmax space so that iterates are automatically
    non‑negative and ℓ₁‑normalised.
    """
    n = len(start_seq)
    # initialise log‑probabilities
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0

    # track the best (lowest surrogate) encountered
    a = np.exp(v)
    a /= a.sum()
    best_seq = a.copy()
    best_sur = _objective_surrogate(best_seq)

    # schedule for the smooth‑max temperature β (from very smooth → almost exact)
    beta_schedule = [1.0, 10.0, 100.0, 1_000.0, 10_000.0, 100_000.0, 1_000_000.0]
    # allocate roughly equal iteration budget to each β value
    # we will simply loop over the schedule and run as many steps as time permits
    for beta in beta_schedule:
        # inner loop – run until β changes or time runs out
        while time.time() < deadline:
            # ---- compute a, objective and gradient of smooth‑max -----------------
            a = np.exp(v)
            a_sum = a.sum()
            if a_sum == 0.0:               # numerical safety
                a = _uniform(n)
                a_sum = 1.0
            a /= a_sum

            conv = _conv_fft(a, a)
            # shift by the (exact) maximum to avoid overflow
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth objective (exact surrogate would be with beta → ∞)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient w.r.t. a (see derivation in the original template)
            w = exp_beta / sum_exp                     # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3·n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. log‑probability vector v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ---- Adam update ----------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

            # ---- keep track of best surrogate ------------------------------------
            cur_sur = _objective_surrogate(a)
            if cur_sur < best_sur:
                best_sur = cur_sur
                best_seq = a.copy()

            # ---- progress check --------------------------------------------------
            if time.time() >= deadline:
                break

    return best_seq, best_sur


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The optimiser builds a modest seed pool, picks the best seed, and spends the
    majority of the time running a smooth‑max Adam optimiser followed by a
    short permutation annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem dimension – keep it at 4000 (the same size as the benchmark)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            n = prev_arr.shape[0]
            # normalise if needed
            if prev_arr.sum() == 0.0:
                prev_arr = _uniform(n)
            else:
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            n = 4000
            prev_arr = None
    else:
        n = 4000
        prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    # several exponential rates (including a very gentle decay)
    for rate in (0.0005, 0.001, 0.0021, 0.004):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law shapes
    for p in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_power_law(n, p))

    # random beta shapes
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear coarse shapes
    for breaks in (2, 4, 6):
        for _ in range(4):
            seeds.append(_piecewise_linear(n, breaks, rng))

    # include previous best and a few permutations of it
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            seeds.append(np.random.permutation(prev_arr))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:

        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))          # we will focus on the very best seeds
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Pick the overall best seed and run a long Adam optimisation
    # ------------------------------------------------------------------
    best_seed = top_seeds[0].copy()
    best_surrogate = _objective_surrogate(best_seed)
    print(f"\n[Init] best seed surrogate = {best_surrogate:.6f}")

    # run Adam on the best seed for the bulk of the remaining budget
    if time.time() < deadline - 30.0:   # keep a few seconds for the final anneal
        print("\n--- Adam optimisation (smooth‑max) ---")
        best_seq, best_surrogate = _smooth_max_adam(best_seed, deadline)
        print(f"[Adam] best surrogate after smooth optimisation = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 4. Very short permutation annealing (a few seconds) – may shave a bit
    # ------------------------------------------------------------------
    if time.time() < deadline:

        anneal_deadline = time.time() + min(20.0, sec_left)
        print("\n--- Final permutation annealing ---")
        best_seq, best_surrogate = _perm_anneal(best_seq, anneal_deadline,
                                               max_iters=80_000)
        print(f"[Final perm] surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 5. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
