        base_shapes.append(_power(n, p))

    # random Dirichlet
    for conc in [0.02,0.05,0.1,0.2,0.5,1.0,2.0,5.0]:
        for _ in range(3):
            base_shapes.append(_random_dirichlet(n, conc, rng))

    # Choose e.g., the power shape with p=1 (harmonic) as candidate.
    selected = [_power(n, 1.0), _exp_decay(n,0.001)]  # etc
    for shape in selected:

        shape = shape / shape.sum()
        for _ in range(100):  # 100 random permutations per shape
            perm = rng.permutation(shape)
            base_shapes.append(perm)

    # also generate some uniform but with small random perturbation
    for _ in range(20):
        pert = _uniform(n) + rng.normal(scale=0.001, size=n)
        pert = np.maximum(pert, 0)
        pert = pert / pert.sum()
        base_shapes.append(pert)

    # If previously saved sequence exists, add it and perturbed versions
    prev = globals().get('height_sequence_1')
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        arr = np.maximum(arr,0)
        if arr.shape[0] != n:
            # rescale/resample to length n via interpolation
            arr = np.interp(np.linspace(0, len(arr)-1, n), np.arange(len(arr)), arr)
        arr = arr/arr.sum()
        base_shapes.append(arr)
        for _ in range(10):
            noisy = arr + 0.001 * rng.normal(size=n)
            noisy = np.maximum(noisy,0)
            noisy = noisy / noisy.sum()
            base_shapes.append(noisy)

    # Quick surrogate evaluation and keep top K

    for vec in base_shapes:

"""
optimised_step_function_v2.py

Search for a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The method

1. builds a wide variety of analytic families (uniform, triangular,
   sine‑window, Gaussian, exponential, power‑law, beta‑shape) and a large
   number of random permutations of a slowly decaying power‑law.  Every
   candidate is ℓ₁‑normalised and its *surrogate* value
   `2·n·max(conv(a,a))` (exact for `sum(a)=1`) is computed with a fast FFT.

2. The best *K* (default 30) candidates are kept as *seeds*.

3. Each seed is refined with a three‑stage local optimiser:
      – exponentiated‑gradient mirror‑descent on the exact max‑convolution
        (sub‑gradient on the active index);
      – short permutation‑based simulated‑annealing hill‑climbing;
      – a final long exponentiated‑gradient run.
   The best refined seed becomes the current global optimum.

4. The remaining time is spent polishing the global optimum with a very
   long exponentiated‑gradient run.

All steps respect the overall 1000 s budget; the final sequence is stored
in the global variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    """Gaussian with adjustable mean (mu_frac∈[0,1] gives position in the array)."""
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Helper for maintaining the pool of best seeds
# ----------------------------------------------------------------------
def _seed_key(item):
    """Key function for sorting (value, vector) tuples."""
    return item[0]


def _add_candidate(vec: np.ndarray, pool: list, top_k: int, deadline: float):
    """
    Normalise ``vec`` to ℓ₁‑norm 1, compute its surrogate value and, if it
    belongs to the best ``top_k`` candidates, insert it into ``pool``.
    """
    if time.time() > deadline:
        return
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return
    vec = vec / s
    val = _objective_surrogate(vec)
    if len(pool) < top_k:
        pool.append((val, vec))
        pool.sort(key=_seed_key)
    else:
        # keep only the best ``top_k`` items
        if val < pool[-1][0]:
            pool[-1] = (val, vec)
            pool.sort(key=_seed_key)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient on the
    active index).  Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep the simplex constraint automatically satisfied
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing optimiser (kept from the baseline)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt  # pragma: no cover
except Exception:       # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break

    return _softmax(v)

# Main entry – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine builds a large pool of analytically parametrised shapes,
    adds many random permutations of a slowly‑decaying power‑law, keeps the
    best few according to the exact surrogate, and then refines each with
    exponentiated‑gradient + permutation hill‑climbing.  A final long EG run
    consumes the remaining time.  The best sequence is stored in the global
    variable ``height_sequence_1`` for future continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a trade‑off between expressivity and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
            else:
                # interpolate to the required length
                x_old = np.linspace(0, 1, len(arr))
                x_new = np.linspace(0, 1, n)
                prev_arr = np.interp(x_new, x_old, arr)
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation (analytic families + random permutations)
    # ------------------------------------------------------------------
    TOP_K = 30                     # how many seeds we retain
    seed_pool = []                 # list of (value, vector) tuples

    # Simple analytic families
    _add_candidate(_uniform(n), seed_pool, TOP_K, deadline)
    _add_candidate(_triangular(n), seed_pool, TOP_K, deadline)

    # Sine windows – moderate grid
    for gamma in np.linspace(0.2, 4.0, 30):
        _add_candidate(_sine_window(n, gamma), seed_pool, TOP_K, deadline)

    # Gaussian – modest sweep
    for sigma_frac in np.linspace(0.05, 0.35, 20):
        _add_candidate(_gaussian(n, sigma_frac), seed_pool, TOP_K, deadline)

    # Shifted Gaussians – coarse sweep over centre/width
    for mu_frac in np.linspace(0.15, 0.85, 8):
        for sigma_frac in np.linspace(0.05, 0.30, 8):
            _add_candidate(_gaussian_shift(n, sigma_frac, mu_frac),
                           seed_pool, TOP_K, deadline)

    # Exponential decay – log‑space sweep
    for rate in np.logspace(-5, 0.5, 50):
        _add_candidate(_exp_decay(n, rate), seed_pool, TOP_K, deadline)
        _add_candidate(_sym_exp_decay(n, rate), seed_pool, TOP_K, deadline)

    # Power‑law – wide sweep
    for p in np.linspace(0.0, 6.0, 25):
        _add_candidate(_power(n, p), seed_pool, TOP_K, deadline)

    # Beta‑shape – systematic grid
    alphas = np.geomspace(0.1, 10.0, 12)
    betas  = np.geomspace(0.1, 10.0, 12)
    for a in alphas:
        for b_ in betas:
            _add_candidate(_beta_shape(n, a, b_), seed_pool, TOP_K, deadline)

    # Random Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(4):
            _add_candidate(_random_dirichlet(n, conc, rng),
                           seed_pool, TOP_K, deadline)

    # Random permutations of a slowly decaying power‑law.
    # These permutations explore many different orderings of the same multiset.
    base_perm_shape = _power(n, 0.3)   # very gentle decay → rich set of permutations
    PERM_COUNT = 2000
    for _ in range(PERM_COUNT):
        perm = rng.permutation(base_perm_shape)
        _add_candidate(perm, seed_pool, TOP_K, deadline)

    # Include the previous best and noisy copies if available
    if prev_arr is not None:
        _add_candidate(prev_arr, seed_pool, TOP_K, deadline)
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            _add_candidate(noisy, seed_pool, TOP_K, deadline)

    print("\n--- Top seeds after surrogate evaluation (keep {} best) ---".format(TOP_K))
    for idx, (val, _) in enumerate(seed_pool[:TOP_K]):
        print(f"  Seed #{idx+1:2d}: surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = seed_pool[0][1].copy()
    global_best_val = seed_pool[0][0]
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of each top seed
    # ------------------------------------------------------------------
    for si, (seed_val, seed_vec) in enumerate(seed_pool[:TOP_K]):
        if time.time() >= deadline:
            break

        # allocate a modest fraction of the remaining budget to this seed
        remaining = deadline - time.time()
        # we aim to spend roughly an equal share on the remaining seeds
        share = remaining * 0.10 / max(1, (TOP_K - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s (surrogate {seed_val:.6f})")

        # a) exponentiated‑gradient (exact max) – moderate budget
        seq_eg, val_eg = _exp_gradient_opt(seed_vec, n, sub_deadline,
                                           max_iters=250_000,
                                           eta0=8.0)

        # b) short permutation hill‑climb – uses the same sub‑deadline
        seq_perm, val_perm = _perm_hill_climb(seq_eg, n,
                                              sub_deadline,
                                              rng,
                                              max_iters=120_000)

        # c) optional L‑BFGS smoothing if time permits
        if time.time() < sub_deadline:
            seq_smooth = _lbfgs_smooth_optimize(seq_perm, n, sub_deadline,
                                                max_iter_per_beta=2000)
            val_smooth = _objective_surrogate(seq_smooth)
        else:
            seq_smooth, val_smooth = seq_perm, val_perm

        # pick the best among the three refinements
        candidates = [(seq_eg, val_eg),
                      (seq_perm, val_perm),
                      (seq_smooth, val_smooth)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (consume the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=4_500_000,
                                                 eta0=7.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Tiny final permutation tweak (very short budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 22.0   # ~20 s
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq, global_best_val = seq_last, val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
