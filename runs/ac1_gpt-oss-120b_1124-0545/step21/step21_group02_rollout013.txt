"""
Optimise a non‑negative step‑function for the evaluation function:

    2·n·max(conv(seq,seq)) / (sum(seq)^2)

The search consists of three stages:

1. **Fast parametric sweep** – generate a diverse set of candidate
   probability vectors from simple analytic families (exponential,
   symmetric exponential, Gaussian, power‑law, polynomial decay, symmetric
   Beta).  The surrogate objective (2·n·max(conv)) is evaluated with an
   FFT‑based convolution (exact when the vector sums to 1).  The best few
   candidates are kept.

2. **Exponentiated‑gradient polishing** – starting from the best candidate,
   run mirror–descent on the exact max‑convolution using the sub‑gradient
   at the active index.  The updates are performed in log‑space, guaranteeing
   the simplex constraints (non‑negative, sum = 1).  Iterations continue
   until time runs out.

3. **Optional L‑BFGS smoothing** – if SciPy is available and time permits,
   minimise a smooth log‑sum‑exp approximation of the max‑convolution,
   gradually increasing the smoothing parameter β (continuation).

The final sequence is stored in the global ``height_sequence_1`` variable
so that subsequent runs can initialise from it.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact if sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max (mirror‑descent projection onto probability simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on max‑convolution.
    ``start_seq`` must already sum to 1.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep a on the simplex
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Optional L‑BFGS smoothing (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------

except Exception:   # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise a smooth approximation of max‑convolution using L‑BFGS‑B.
    Operates in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabilise the exponentials
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (analytical)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # mirror‑descent (gradient in log‑space)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):

                raise KeyboardInterrupt()

                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")

    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    return a / a.sum()

    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _shape_poly_decrease(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n-1))^γ , monotone decreasing."""
    i = np.arange(n, dtype=np.float64) / (n - 1)

    return a / a.sum()

    """Simple symmetric triangle (linear rise‑fall)."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = np.where(i <= half, i + 1, n - i)
    return a / a.sum()


def _shape_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    try:
        from scipy.stats import beta as beta_dist
        x = np.linspace(0.0, 1.0, n)
        a = beta_dist.pdf(x, alpha, alpha)

        B = gamma(alpha) * gamma(alpha) / gamma(2.0 * alpha)
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0)) / B
    return a / a.sum()
