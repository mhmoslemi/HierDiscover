def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, later normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

                     n: int,

                     eta0: float = 3.0):

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g[idx_i] = 2.0 * a[rev_idx]

        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
