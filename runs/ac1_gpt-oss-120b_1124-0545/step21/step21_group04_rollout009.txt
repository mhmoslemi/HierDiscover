def _smooth_obj_and_grad(logits, beta, n):
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    shift = max_val
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max
    # gradient wrt a:
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    # gradient wrt logits:
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return obj, grad_logits

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The optimiser works in several stages:

1.   Build a small pool of starter sequences (uniform, exponential,
     U‑shaped power‑law, previous best and a few Dirichlet draws).
2.   Run a brief exponentiated‑gradient (mirror‑descent) polish for
     each starter.
3.   Refine the current best with a long exponentiated‑gradient run.
4.   Apply a multi‑stage Adam optimiser on a smooth log‑sum‑exp
     surrogate of the max‑convolution.
5.   Run an extended L‑BFGS smoothing phase (if SciPy is available).
6.   Perform a cheap stochastic perturbation/hill‑climbing search.
7.   Return the best sequence found and store it globally for the next
     run.
"""

import math
import time
from typing import List

import numpy as np

# Optional SciPy import – used only for the L‑BFGS smoothing phase.
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability‑simplex utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s != 0.0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    return w / s if s != 0.0 else _uniform(n)


def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped sequence whose surrogate value is locally best."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)
    best_p = 0.0

    # coarse grid
    for p in np.linspace(0.0, 2.0, 21):
        if time.time() > deadline:
            break
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq, best_p = val, seq, p

    # local refinement
    step = 0.2
    for _ in range(5):
        if time.time() > deadline:
            break
        for p in (best_p - step, best_p + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, best_p = val, seq, p
        step *= 0.5
    return best_seq


# ----------------------------------------------------------------------
# Surrogate objective (exact when the L¹‑norm equals 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L¹ = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over the active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance – a few ULPs are enough
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]

    if active.size == 0:
        # No active indices – return zero gradient (should not happen)
        return np.zeros_like(a)

    # build weight vector w where active entries get equal weight
    w = np.zeros_like(conv, dtype=np.float64)
    w[active] = 1.0 / active.size

    # gradient wrt a: 2 * (rev_a * w) restricted to the original length
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 2n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent using the averaged sub‑gradient."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max utilities (log‑sum‑exp) – used by Adam & L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray,
                         beta: float,
                         n: int) -> tuple[float, np.ndarray]:
    """Smooth version of 2·n·max(conv) together with gradient w.r.t. logits."""
    a = _softmax(logits)
    conv = _conv_fft(a, a)
    max_val = conv.max()

    # smooth max via log‑sum‑exp
    shift = max_val
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    # gradient w.r.t. a (weights of the smooth max)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 2n‑1
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # shape (n,)

    # gradient w.r.t. logits: a * (grad_a - <grad_a,a>)
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)

    return obj, grad_logits


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize_multi(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         beta_schedule,
                         steps_per_beta: int = 2000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate.  Returns the best sequence seen."""
    v = np.log(start_seq + 1e-30)           # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    eps = 1e-8

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            obj, grad = _smooth_obj_and_grad(v, beta, n)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat_hat = v_hat / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat_hat) + eps)
            v = v - step
            a = _softmax(v)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Very cheap stochastic local‑search (hill‑climbing)
# ----------------------------------------------------------------------
def _local_perturb_search(seq: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass between random positions; keep improvements."""
    a = seq.copy()
    best = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline:
        # decreasing step size
        scale = max(1e-6, 0.02 * (0.9995 ** it))
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if i == j:
            continue
        delta = min(a[i], scale * np.random.rand())
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best = cur_val, a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
        it += 1
    return best, best_val
