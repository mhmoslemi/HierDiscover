```
def evaluate_sequence(sequence: list[float]) -> float:
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01: return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

"""
Fast stochastic optimizer for the step‑function bound.

The idea is simple:
* keep the whole convolution vector `C = a * a` (linear convolution)
* propose cheap *mass‑transfer* moves (move a fraction of probability mass
  from index i to j) and update `C` *incrementally* in O(n) time
  using the exact update formula
    C_k ← C_k + 2·Δ·(a_{k‑j} – a_{k‑i})               (Δ > 0)
    plus the three quadratic correction terms at 2i, 2j and i+j.
* accept the move with a standard simulated‑annealing rule.
  Because the update is O(n) we can perform many millions of moves
  inside the 1000 s budget, far more than the previous FFT‑based approach.
* after the annealing phase a short mirror‑descent polishing (the same
  exponentiated‑gradient used before) is applied with a few thousand
  FFT‑based steps to clean up the remaining noise.
* the best sequence found (as a probability vector, i.e. sum = 1) is
  returned and stored in the global ``height_sequence_1`` for the next run.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: stable soft‑max (mirror‑descent projection)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Incremental convolution update for a mass‑transfer move
# ----------------------------------------------------------------------
def _transfer_update(a: np.ndarray,
                     C: np.ndarray,
                     i: int,
                     j: int,
                     delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return new (a, C) after moving `delta` mass from index i to j.
    The update uses the exact formula derived from
    C'ₖ = Cₖ + 2·Δ·(a_{k‑j} – a_{k‑i})
    plus the three quadratic correction terms.
    """
    n = a.shape[0]

    # copy – cheap (32 KB / 64 KB) and safe for rejection
    a_new = a.copy()
    C_new = C.copy()

    # slice updates (use the *old* a for the linear term)
    C_new[i:i + n] -= 2.0 * delta * a
    C_new[j:j + n] += 2.0 * delta * a

    # three isolated quadratic corrections
    C_new[2 * i] += delta * delta
    C_new[2 * j] += delta * delta
    C_new[i + j] -= 2.0 * delta * delta

    # modify the probability vector
    a_new[i] -= delta
    a_new[j] += delta

    return a_new, C_new


# ----------------------------------------------------------------------
# Optional mirror‑descent polishing (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 3.0) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
    Works on the probability simplex (sum = 1).  Stops when the deadline
    is reached.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        # exact convolution (FFT, O(n log n))
        conv = np.fft.irfft(np.fft.rfft(a, n * 2) *
                            np.fft.rfft(a, n * 2))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # sub‑gradient of the active peak
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        inds = np.arange(i_start, i_end + 1)
        rev = k - inds
        g[inds] = 2.0 * a[rev]

        it = max_iters
        eta = eta0 / math.sqrt(max_iters + 1)
        log_a = log_a - eta * g
        a = _softmax(log_a)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Strategy:
      1. initialise from the previous best or an exponential decay,
      2. massive stochastic annealing with cheap O(n) updates,
      3. short exponentiated‑gradient polishing,
      4. return the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # problem size (same as in the reference run)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. initialise
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        a = np.asarray(prev, dtype=np.float64)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= total
    else:
        # exponential‑decay start, then normalise to a probability vector
        i = np.arange(n, dtype=np.float64)
        rate = 0.003
        a = np.exp(-rate * i)
        a /= a.sum()

    # initial full convolution (once)
    C = np.convolve(a, a)                # length 2n‑1
    cur_val = 2.0 * n * C.max()
    best_a = a.copy()
    best_val = cur_val

    print(f"[Init] objective = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 2. simulated‑annealing with incremental updates
    # --------------------------------------------------------------
    anneal_end = deadline - 30.0          # keep 30 s for polishing
    total_anneal_time = anneal_end - start_time
    it = 0
    reporting_interval = 200_000

    while time.time() < anneal_end:
        it += 1

        # linear temperature schedule (high → low)
        elapsed = time.time() - start_time
        frac = elapsed / total_anneal_time
        T_start = cur_val
        T_end = cur_val * 1e-6
        T = T_start + (T_end - T_start) * frac

        # ---- move selection -------------------------------------------------
        if rng.random() < 0.75:                     # mass‑transfer move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] == 0.0:
                continue
            # move at most half of the source mass (random fraction)
            delta = a[i] * rng.random() * 0.5
            # generate candidate vectors
            a_cand, C_cand = _transfer_update(a, C, i, j, delta)
        else:                                        # occasional swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            a_cand = a.copy()
            a_cand[i], a_cand[j] = a_cand[j], a_cand[i]
            # recompute conv for swaps (rare, therefore OK)
            C_cand = np.convolve(a_cand, a_cand)

        cand_val = 2.0 * n * C_cand.max()

        # ---- Metropolis acceptance -------------------------------------------
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            a, C, cur_val = a_cand, C_cand, cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Iter {it}] new best = {best_val:.6f}")

        # periodic full recompute to avoid drift from rounding errors
        if it % 100_000 == 0:
            C = np.convolve(a, a)
            cur_val = 2.0 * n * C.max()
            # also re‑normalise just in case
            a /= a.sum()

        # occasional progress report
        if it % reporting_interval == 0:
            print(f"[Iter {it}] cur = {cur_val:.6f}, best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. short mirror‑descent polishing (exact max‑convolution)
    # --------------------------------------------------------------
    polish_deadline = deadline - 5.0
    if time.time() < polish_deadline:
        best_a = _exp_gradient_opt(best_a, n, polish_deadline,
                                   max_iters=200_000, eta0=3.0)
        best_val = 2.0 * n * np.max(np.convolve(best_a, best_a))
        print(f"[Polish] objective after mirror‑descent = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. final evaluation with the provided secure function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
