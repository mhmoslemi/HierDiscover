"""
Propose a step‑function (non‑negative sequence) with a small
evaluate_sequence() score.

The strategy
----------------
1. **Seeding** – a few analytically‑motivated shapes
   (uniform, geometric with ε≈1/n, random Dirichlet draws) together
   with the previous best solution (if any) are generated and ℓ₁‑normalised.

2. **Mirror‑descent (exponentiated‑gradient)** – each seed is polished by a
   long run of exponentiated‑gradient on the exact max‑convolution.
   The iterates live in the log‑softmax domain, guaranteeing non‑negativity
   and unit ℓ₁‑norm.  The step size η_t = η₀ / √t with η₀=5 is used.

3. **Local random perturbations** – the current best vector is repeatedly
   perturbed by sampling a Dirichlet distribution whose mean is the current
   vector (concentration gradually increased).  Any improvement is kept.

4. **Permutation annealing** – a final short simulated‑annealing phase
   swaps two entries at a time; this can escape ordering traps that the
   gradient step cannot.

All heavy operations (full convolution) are performed via a real FFT
implementation (`_conv_fft`).  The routine respects the global time budget
(`budget_s`, default 1000 s) and stores the best sequence in the global
variable ``height_sequence_1`` for possible continuation runs.

The returned list yields a surrogate value (2·n·max(conv)) that is also
the true evaluate_sequence() value because the vector is ℓ₁‑normalised
(sum = 1).  The algorithm is deliberately simple yet aggressive enough
to improve the previous best 1.508514 to below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of the result is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geometric_seq(n: int, eps: float) -> np.ndarray:
    """
    a_i ∝ (1‑eps)^{i}  with scale chosen so that sum_{i=0}^{n‑1} a_i = 1.
    eps is the decay per step (≈1/n gives a good starting shape).
    """
    if eps <= 0.0:
        return _uniform(n)
    q = 1.0 - eps
    i = np.arange(n, dtype=np.float64)
    a = eps * (q ** i)                      # un‑normalised geometric tail
    total = 1.0 - q ** n                    # exact truncated sum
    if total <= 0.0:
        total = 1.0
    a /= total
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (exponentiated‑gradient).
    Returns the best sequence seen during the run and its surrogate value.
    """
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active diagonal (active index = max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # renormalise => softmax

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that swaps two entries at a time.
    Uses a linear temperature schedule from the current objective
    down to 1e‑6·objective.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
