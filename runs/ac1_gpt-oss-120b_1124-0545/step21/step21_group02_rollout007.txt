"""
Propose a new step‑function sequence.

The optimiser is a two‑phase procedure:

1.  **Adam‑optimisation of a smooth max‑convolution**  
    We minimise a log‑sum‑exp surrogate of `max(a*a)` while staying on
    the probability simplex (the vector is stored in log‑space and
    transformed by a softmax).  A geometric schedule of β‑values
    (smoothness) is used and a fixed budget of Adam steps is spread
    over the schedule.  The best vector seen (according to the exact
    max‑convolution) is retained.

2.  **Exact max‑convolution mirror‑descent polishing**  
    Using the sub‑gradient of the true max‑convolution we run a classic
    exponentiated‑gradient (mirror descent) sweep for a further *few*
    hundred thousand iterations.  This phase often reduces the final
    bound by a tiny amount.

Multiple random restarts are performed (previous best, exponential
decay, a Dirichlet draw) and the best overall vector is returned.
All numbers are non‑negative and the vector is automatically normalised
to sum to 1, which leaves the evaluation invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (real data, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)).  (n = len(seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_steps: int = 120_000,
                   lr0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best sequence seen (as a probability vector) and its
    surrogate value.
    """
    eps = 1e-30
    a = np.clip(start_seq, eps, None)
    a = a / a.sum()                     # ensure sum = 1
    log_a = np.log(a)                   # log‑space (simple softmax)
    m = np.zeros_like(log_a)            # Adam first moment
    v = np.zeros_like(log_a)            # Adam second moment
    beta1, beta2 = 0.9, 0.999
    pow1, pow2 = beta1, beta2
    t = 0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # smoothness schedule (geometric increase)
    beta_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0,
                     2048.0, 8192.0, 32768.0]

    steps_done = 0
    lr = lr0

    for beta in beta_schedule:
        steps_for_this_beta = max_steps // len(beta_schedule)
        for _ in range(steps_for_this_beta):
            if steps_done >= max_steps or time.time() > deadline:
                break

            # ----- exact convolution (needed for both objective & gradient) -----
            conv = _conv_fft(a, a)               # length = 2n-1
            max_conv = conv.max()
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

            # ----- smooth‑max gradient (log‑sum‑exp) -----
            shift = max_conv                       # stabilise exp
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                 # soft weights, length = 2n‑1

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice of length n

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)          # gradient w.r.t. log‑softmax variables

            # ----- Adam update in log‑space -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            pow1 *= beta1
            pow2 *= beta2
            m_hat = m / (1.0 - pow1)
            v_hat = v / (1.0 - pow2)

            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _softmax(log_a)                    # stays on the simplex

            steps_done += 1

        if time.time() > deadline:
            break
        # slowly decay learning‑rate when smoothness grows
        lr *= 0.7

    # final check (in case the loop exited early)
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (sub‑gradient) polishing
# ----------------------------------------------------------------------
def _mirror_descent_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400_000,
                        eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / mirror‑descent on the *exact* max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())

        # sub‑gradient of the active entry (only the contributing pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    # final sanity check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()
    return best_seq, best_val

        # split the remaining time evenly among the remaining candidates + polishing
        available = remaining / (len(candidates) - idx)
        polish_deadline = time.time() + min(available, 5.0)  # at least a few seconds
        seq_polish, val_polish = _mirror_descent_opt(seq_adam, n, polish_deadline,
                                                     max_iters=max_iters_mirror,
                                                     eta0=8.0)
        print(f"    [Polish] surrogate after mirror‑descent = {val_polish:.6f}")

        # keep the best overall
        if val_polish < best_val_global:
            best_val_global = val_polish
            best_seq_global = seq_polish.copy()

    # ------------------------------------------------------------------
    # Final evaluation with the safe function
    # ------------------------------------------------------------------
    true_val = None
    if best_seq_global is not None:
        try:
            true_val = evaluate_sequence(best_seq_global.tolist())
            print(f"[Result] surrogate = {best_val_global:.6f}, "
                  f"true evaluate_sequence = {true_val:.6f}")
        except Exception as exc:  # pragma: no cover
            print("[Warning] evaluate_sequence raised an exception:", exc)

    # store for the next iteration (the global variable is used by the runner)
    globals()["height_sequence_1"] = best_seq_global.tolist() if best_seq_global is not None else []

    return best_seq_global.tolist() if best_seq_global is not None else []
