"""
Propose a step‑function whose ℓ₁‑normalised shape minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s total)
1️⃣  Generate a diverse pool of ℓ₁‑normalised seeds:
   - uniform, exponential, symmetric exponential,
   - discrete arcsine (β=½,½), a few Beta(α,β) families,
   - Dirichlet draws with several concentration values,
   - noisy perturbations of the current best solution (if any)
   - a simple interpolated version of the previous global solution.

2️⃣  Keep the best few seeds w.r.t. the exact surrogate
   ``2·n·max(conv(a,a))`` (identical to the true objective for ℓ₁‑norm = 1).

3️⃣  Refine each seed with a *smoothed‑max* optimisation:
   - Work in the log‑softmax (log‑probability) space so that
     non‑negativity and the ℓ₁‑constraint are enforced automatically.
   - For a temperature β, the smooth max is
       f_β(a) = (2·n/β)·log∑_k exp(β·conv_k(a,a)).
   - Gradient w.r.t. the logits is obtained analytically:
       g = a·(∇_a f_β – ⟨∇_a f_β , a⟩) .
   - Optimise f_β with the ADAM optimizer, gradually increasing β
     (1 → 5 → 20 → 80 → 200 → 5 000 → 50 000).  This yields a
     high‑quality approximation of the true max‑convolution objective.

4️⃣  After the smoothed optimisation, run a short *permutation‑annealing*
   (swap two entries) to escape ordering traps.

5️⃣  Keep the globally best sequence across all seeds, run a final long
   ADAM polishing with the remaining time budget, and finish with a
   final permutation tweak.

All heavy convolutions are performed with a real FFT (``_conv_fft``) for
O(n log n) speed.  The code respects the total time budget, prints
intermediate surrogate values and finally stores the best sequence in
the global variable ``height_sequence_1`` for possible continuation runs.

import time

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (ℓ₁‑norm = 1): 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) – heavy at the ends, light in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _noisy_uniform(n: int, scale: float, rng: np.random.Generator) -> np.ndarray:
    base = _uniform(n)
    noise = rng.normal(scale=scale, size=n)
    a = np.maximum(base + noise, 0.0)
    s = a.sum()
    return a / s if s > 0 else base


def _interpolate_to(vec: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of an existing vector to a different length."""
    old_n = len(vec)
    if old_n == n_new:
        return vec / vec.sum()
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, n_new)
    new = np.interp(xs_new, xs_old, vec)
    s = new.sum()
    return new / s if s > 0 else _uniform(n_new)


# ----------------------------------------------------------------------
# Smoothed‑max objective and its gradient (log‑softmax parameterisation)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(logits: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (obj, grad_logit) where
    obj = 2·n·(max(conv) + log∑exp(beta·(conv−max))/beta)
    grad_logit = ∂obj/∂logits  (log‑softmax space)
    """
    # a = softmax(logits)
    a = np.exp(logits)
    a_sum = a.sum()
    a = a / a_sum

    conv = _conv_fft(a, a)
    max_c = np.max(conv)                     # shift for numerical stability
    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_terms = np.exp(beta * shifted)       # max term becomes 1
    sum_exp = exp_terms.sum()
    obj = float(2.0 * len(a) * (max_c + np.log(sum_exp) / beta))

    # softmax weights for the smooth max
    w = exp_terms / sum_exp

    # gradient w.r.t. a : 2 * (rev_a * w)  (convolution)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3·n‑2
    grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

    # chain rule to logits (log‑softmax)
    dot = np.dot(grad_a, a)
    grad_logit = a * (grad_a - dot)
    return obj, grad_logit


# ----------------------------------------------------------------------
# ADAM optimiser on the smoothed‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   beta_schedule: list[float] | None = None,
                   max_iters_per_beta: int = 4000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with ADAM on the smooth max‑convolution objective.
    Operates on logits = log a (softmax) to keep a ≥ 0 and ∑a = 1.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 2000.0, 5000.0, 20000.0, 50000.0]

    n = len(start_seq)
    # initialise logits
    logits = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # ADAM hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    t = 0  # global step counter

    for beta in beta_schedule:
        # reset first‑moment / second‑moment accumulators for each β
        m.fill(0.0)
        v.fill(0.0)
        for _ in range(max_iters_per_beta):
            if time.time() >= deadline:
                return best_seq, best_val
            t += 1
            obj, grad = _smooth_obj_and_grad(logits, beta)

            # ADAM update in logits space
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate scaling – larger β needs smaller steps
            lr = 0.05 / (math.sqrt(beta) + 1e-8)

            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate the exact surrogate (cheap – uses FFT)
            a = np.exp(logits)
            a /= a.sum()
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # end of current β
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Swaps two entries repeatedly; accepts worsening moves with a linear
    temperature schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        nxt = cur_seq.copy()
        nxt[i], nxt[j] = nxt[j], nxt[i]

        nxt_val = _objective_surrogate(nxt)
        if nxt_val < cur_val or (T > 0.0 and np.random.rand() < math.exp(-(nxt_val - cur_val) / T)):
            cur_seq = nxt
            cur_val = nxt_val
            if nxt_val < best_val:
                best_seq = nxt.copy()
                best_val = nxt_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects a total runtime budget
    and stores the best found sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – 5 000 balances quality and speed.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Build a pool of seeds (all ℓ₁‑normalised)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_sym_exp_decay(n, rate=0.0015))
    seeds.append(_arcsine_shape(n))

    # a few Beta shapes (grid)
    for a in (0.4, 0.8, 1.5):
        for b in (0.4, 0.8, 1.5):
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with diverse concentrations
    for conc in (0.02, 0.07, 0.2, 0.8, 1.5):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # noisy uniform variants – small random bumps
    for scale in (0.01, 0.03, 0.07):
        seeds.append(_noisy_uniform(n, scale=scale, rng=rng))

    # incorporate the previous global solution (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.ndim == 1:
            # adapt length if needed
            if prev_arr.shape[0] != n:
                prev_arr = _interpolate_to(prev_arr, n)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = prev_arr.sum()
            if s > 0.0:
                prev_arr = prev_arr / s
                seeds.append(prev_arr)
                # a few random permutations of the previous best
                for _ in range(3):
                    seeds.append(np.random.permutation(prev_arr))

    # --------------------------------------------------------------
    # Quick surrogate evaluation – keep the best few seeds
    # --------------------------------------------------------------
    evaluated: list[tuple[np.ndarray, float]] = []
    for vec in seeds:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s <= 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # Main refinement loop (ADAM on smoothed max + local annealing)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a roughly equal share of the remaining budget
        remaining = deadline - time.time()
        share = remaining * 0.13 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # ---- (a) ADAM optimisation on smoothed max -----------------
        seq_adam, val_adam = _adam_optimize(seed_vec,
                                            sub_deadline,
                                            max_iters_per_beta=5000)

        # ---- (b) short permutation anneal -------------------------
        if time.time() < sub_deadline:
            seq_perm, val_perm = _perm_anneal(seq_adam,
                                             sub_deadline,
                                             max_iters=150_000)
        else:
            seq_perm, val_perm = seq_adam, val_adam

        # select the best local result
        local_best_seq, local_best_val = min(
            [(seq_adam, val_adam), (seq_perm, val_perm)],
            key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final long ADAM polishing with the full remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long ADAM polishing (full remaining budget) ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                              deadline,
                                              max_iters_per_beta=8000)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final ADAM] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Very short final permutation tweak (a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Evaluate the true objective and store the result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
