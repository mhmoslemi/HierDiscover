"""
Improved search for a non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The optimiser works in three stages:

1.  Initialise a small collection of analytic “U‑shaped’’ sequences
    (exponential decay, power‑law, arcsine) and a few random Dirichlet draws.
    All candidates are symmetrised and normalised to ℓ₁‑norm 1.

2.  For each starter run a modest exponentiated‑gradient (mirror‑descent)
    optimisation on the exact max‑convolution (sub‑gradient) – this
    quickly removes the most obvious spikes.

3.  Polish the best survivor with a long smooth‑max L‑BFGS run.
    The smooth max is the log‑sum‑exp approximation to the maximum;
    a schedule of increasing β (temperature) focuses the optimisation
    on the true peak.  The L‑BFGS routine is given a generous iteration
    budget (5000 per β) and stops as soon as the global deadline is
    reached.

4.  (Optional) perform cheap random perturbations of the current best
    sequence and run a short EG polish – useful for escaping shallow
    local minima.

All vectors are kept on the probability simplex via the softmax
parameterisation.  Symmetry aₖ = aₙ₋₁₋ₖ is enforced after every major
step because the objective is invariant under reversal and symmetric
solutions tend to be superior.

The function returns a plain list of floats ready for `evaluate_sequence`.

# Optional SciPy import – used for the L‑BFGS smoothing phase.

# ----------------------------------------------------------------------
# Fast FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple analytic starters
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: (i+1)⁻ᵖ + (n‑i)⁻ᵖ."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑type shape: aᵢ ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return w / w.sum()


# ----------------------------------------------------------------------
# Softmax utilities (mirrored‑descent variables)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a≥0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # treat values within ~1 e‑12 · max_val as active
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    g /= max(1, len(active))
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 200_000,
                            eta0: float = 5.0) -> np.ndarray:
    """
    Mirror‑descent using the averaged sub‑gradient.
    Returns the best sequence seen during the run.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_max_conv(a)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, beta: float, n: int):
    """Return (smooth‑max objective, gradient w.r.t. logits v)."""
    a = _softmax(v)                     # probability vector, sum = 1
    conv = _conv_fft(a, a)
    max_c = conv.max()
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth max = shift + log(sum_exp)/beta
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)          # chain‑rule through softmax
    return obj, grad_v

        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    # initialise logits
    v = np.log(start_seq + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, beta_val, n)

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:           # pragma: no cover
            print("[L‑BFGS] exception:", e)
            break

    return _softmax(v)

# Helper: quick 1‑D power‑law scan
# ----------------------------------------------------------------------
def _best_power_law(n: int, deadline: float) -> np.ndarray:
    """Return a power‑law shaped vector with locally minimal surrogate."""
    best_seq = _uniform(n)
    best_val = _objective_surrogate(best_seq)

    # coarse sweep
    for p in np.linspace(0.0, 2.0, 21):
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # local refinement
    p_center = float(np.linspace(0.0, 2.0, 21)[np.argmin([
        _objective_surrogate(_power_law(n, p)) for p in np.linspace(0.0, 2.0, 21)
    ])])
    step = 0.2
    for _ in range(6):
        if time.time() > deadline:
            break
        for p in (p_center - step, p_center + step):
            if p < 0.0:
                continue
            seq = _power_law(n, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq, p_center = val, seq, p
        step *= 0.5
    return best_seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ₁‑normalised) that minimises
    the evaluation function.  The routine respects the given time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it compatible with previous runs.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Assemble a modest pool of starters.
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best as a starter (if it has the right length)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            p = np.asarray(prev, dtype=np.float64)
            if p.shape == (n,):
                p = np.clip(p, 0.0, 1000.0)
                s = p.sum()
                if s > 0:
                    p = p / s
                else:
                    p = _uniform(n)
                candidates.append(p)
        except Exception:
            pass

    # b) a few analytically motivated families
    candidates.append(_exp_decay(n, rate=0.0025))
    candidates.append(_uniform(n))
    candidates.append(_best_power_law(n, deadline))
    candidates.append(_arcsine_shape(n))

    # c) a handful of random Dirichlet draws
    for _ in range(4):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2. Short exponentiated‑gradient polish for each starter.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for i, seq0 in enumerate(candidates):
        if time.time() > deadline - 30:
            break
        # normalise and enforce symmetry
        seq = np.maximum(seq0, 0.0)
        if seq.sum() == 0.0:
            seq = _uniform(n)
        else:
            seq = seq / seq.sum()
        seq = (seq + seq[::-1]) * 0.5

        # modest EG run (fast) – 100 k iterations max
        seq = _exp_gradient_opt_multi(seq, n, deadline,
                                      max_iters=100_000,
                                      eta0=5.0)
        seq = (seq + seq[::-1]) * 0.5

        val = _objective_surrogate(seq)
        print(f"[Init {i}] surrogate after EG = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()

    # ------------------------------------------------------------------
    # 3. Heavy L‑BFGS polishing of the current best.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        best_seq = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                          max_iter_per_beta=5_000)
        best_seq = (best_seq + best_seq[::-1]) * 0.5
        best_val = _objective_surrogate(best_seq)
        print(f"[L‑BFGS] surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random local perturbations (if time permits).
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 2:
        # generate a small random move that respects the simplex
        pert = best_seq.copy()
        # pick a few indices and transfer a tiny amount of mass
        idxs = rng.integers(0, n, size=8)
        deltas = rng.normal(scale=1e-3, size=8)
        for idx, d in zip(idxs, deltas):
            pert[idx] = max(0.0, pert[idx] + d)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()
        pert = (pert + pert[::-1]) * 0.5

        # a brief EG polish to settle the noisy direction
        pert = _exp_gradient_opt_multi(pert, n, deadline,
                                       max_iters=50_000,
                                       eta0=5.0)
        pert = (pert + pert[::-1]) * 0.5
        val = _objective_surrogate(pert)
        if val < best_val:
            best_val, best_seq = val, pert.copy()
            print(f"[Random] improved surrogate = {val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final true evaluation and export.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[Result] evaluation failed:", exc)
        true_val = None

    # make the sequence globally available for the next round
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
