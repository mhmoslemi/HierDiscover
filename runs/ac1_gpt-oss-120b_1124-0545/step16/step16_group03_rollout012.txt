def _exp_gradient_opt(start_seq, n, deadline, max_iters, step_factor=0.5):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = step_factor / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
A fast, robust optimizer for the “step‑function” challenge.

The algorithm consists of three phases:

1. **Diverse seed pool** – Analytic families (uniform, triangular,
   Gaussian, power, exponential, cosine, semi‑circular, beta‑shapes)
   together with random Dirichlet draws and a shuffled version of each.
   The 20 best seeds (according to the exact surrogate
   2·n·max(conv)) are retained.

2. **Exponentiated‑gradient (mirror‑descent) polishing** – Starting from
   each retained seed we run a long mirror‑descent on the true
   non‑smooth objective (sub‑gradient of the max‑convolution).  The
   step size ηₜ = 0.5/√t works well in practice.  The best sequence
   among all runs is kept.

3. **Simulated‑annealing fine‑tuning** – A final annealing walk in
   log‑softmax space explores the neighbourhood of the current best
   solution for a remaining time budget, allowing occasional uphill
   moves to escape shallow local minima.

All operations respect the global time budget (default 1000 s).  The
final sequence is returned as a plain list of floats and stored in the
global ``height_sequence_1`` variable for possible reuse in later runs.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_shape(n: int) -> np.ndarray:
    """Half‑cosine: 0 at the ends, 1 in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _semicircle_shape(n: int) -> np.ndarray:
    """Semicircle (square‑root) shape."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0  # range [-1, 1]
    a = np.sqrt(np.clip(1.0 - x ** 2, 0.0, None))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^α·(n‑i)^β   (zero at the edges)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** alpha) * ((n - i + 1) ** beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser – exact surrogate
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     step_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution surrogate.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # early‑stopping helper
    stagnant = 0
    last_improve = best_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = step_factor / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            stagnant = 0
            last_improve = cur_val
        else:
            stagnant += 1

        # simple stagnation stop‑condition
        if stagnant >= 20_000:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (log‑softmax space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Random walk in log‑softmax space with a linear cooling schedule.
    Accepts uphill moves with probability exp(-(Δ)/T).
    """
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_cosine_shape(n))
    seeds.append(_semicircle_shape(n))

    for sigma in (0.08, 0.15, 0.22, 0.30):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 1.5):
        seeds.append(_power(n, p))

    for rate in (0.002, 0.005, 0.01, 0.02):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # beta‑shapes (U‑like, bell‑like, etc.)
    for a, b in ((0.5, 0.5), (0.2, 0.2), (0.2, 2.0), (2.0, 0.2), (1.0, 1.0)):
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws – several concentrations, plus permutations
    concs = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in concs:
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                 # raw
            seeds.append(np.sort(raw)[::-1])                 # sorted descending
            seeds.append(_alternating_order_from_vec(raw))    # alternating
            seeds.append(raw[rng.permutation(n)])            # random permutation

    # Previous solution (if available)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            clean.append(v / s)
        else:
            clean.append(_uniform(n))
    return clean
