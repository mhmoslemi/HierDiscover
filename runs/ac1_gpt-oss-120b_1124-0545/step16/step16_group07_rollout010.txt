"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the official ``evaluate_sequence`` objective.

Algorithm
---------

1. **Seed generation** – a mixture of analytic families (uniform,
   beta‑shapes, power‑laws, Gaussians) together with many random
   Dirichlet draws (different concentrations) are created and ℓ₁‑normalised.

2. **Coarse selection** – the surrogate objective ``2·n·max(conv)`` (which
   equals the true objective for ℓ₁‑normalised vectors) is evaluated for all
   seeds; the best seed is kept.

3. **Smoothed gradient descent** – we minimise a *smooth* approximation of the
   max‑convolution:
   ```
   f(a) = (1/β)·log Σ_k exp(β·conv_k(a))
   ```
   The gradient is
   ```
   ∂f/∂a = 2·correlation(a, softmax(β·conv))
   ```
   and is applied in the log‑softmax (exponentiated‑gradient) space.
   β is increased gradually from 1 to 50 so that the optimisation first
   sees a smooth landscape and later focuses on the true max.

4. **Fine‑tuning** – the classical sub‑gradient EG optimiser (exact max)
   and an Adam‑style optimiser are run for a short while starting from the
   best point found in step 3.

5. **Local random walk** – a final simulated‑annealing‑style polishing step
   adds Gaussian noise to the logits and keeps any improvement.

All phases respect the 1000 s wall‑clock budget.  The best sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 500_000,
                             eta0: float = 0.7,
                             beta_start: float = 1.0,
                             beta_end: float = 50.0) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated‑gradient descent.
    Optimises f(a) = (1/β)·log Σ exp(β·conv) by gradient descent
    in the log‑softmax space.  β grows linearly from ``beta_start`` to
    ``beta_end`` over the iterations.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)  # log‑parameters for the simplex
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- convolution of a with itself -----
        conv = _conv_fft(a, a)                     # length 2n-1
        max_conv = conv.max()

        # ----- softmax weight over conv entries (smooth max) -----
        beta = beta_start + (beta_end - beta_start) * it / max_iters
        # use max_conv for numerical stability
        w_raw = np.exp(beta * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum                       # shape (2n-1,)

        # ----- gradient = 2·correlation(a, w) -----
        # correlation for lag 0..n-1 = valid part of conv(a_rev, w)
        a_rev = a[::-1]
        corr_full = _conv_fft(a_rev, w)            # length 3n-2
        # extract the *valid* slice (indices n-1 … 2n-2)
        grad = 2.0 * corr_full[n - 1: n - 1 + n]    # shape (n,)

        # ----- exponentiated‑gradient update in log‑space -----
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad
        a = _softmax(log_a)                        # stays on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[SmoothEG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent (still in log‑softmax space) using the exact
    max‑convolution sub‑gradient.
    Returns the best sequence seen and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam]   it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search in log‑softmax space.
    Gaussian noise (annealed) is added to the logits; improvements are kept.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[Local]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – use the same length as the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Use previous solution if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation.
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(30):
        a = 10 ** rng.uniform(-1, 1)   # between 0.1 and 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.08, 0.5)
        seeds.append(_gaussian(n, sigma))

    # Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # Normalise and discard degenerate vectors.
    # ------------------------------------------------------------------
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Rank seeds by the exact surrogate and keep the best.
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec

    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed exponentiated‑gradient optimisation.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    max_iters_smooth = int(min(600_000, remaining * 2500))   # allow up to ~600 k it.
    print(f"\n--- Smoothed EG phase (max_iters≈{max_iters_smooth}) ---")
    seq_smooth, val_smooth = _smooth_exp_gradient_opt(best_seed, n, deadline,
                                                      max_iters=max_iters_smooth,
                                                      eta0=0.7,
                                                      beta_start=1.0,
                                                      beta_end=50.0)
    print(f"[SmoothEG] surrogate = {val_smooth:.6f}")

    global_best_seq = seq_smooth.copy()
    global_best_val = val_smooth

    # ------------------------------------------------------------------
    # 4. Classical sub‑gradient EG fine‑tuning (optional, short).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_eg = int(min(400_000, remaining * 2500))
        print(f"\n--- Classic EG fine‑tuning (max_iters≈{max_iters_eg}) ---")
        seq_eg, val_eg = _exp_gradient_opt(global_best_seq, n, deadline,
                                           max_iters=max_iters_eg, eta0=0.6)
        if val_eg < global_best_val:
            global_best_seq = seq_eg.copy()
            global_best_val = val_eg
            print(f"[Classic EG] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Adam‑style refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_adam = int(min(300_000, remaining * 2500))
        print(f"\n--- Adam‑style refinement (max_iters≈{max_iters_adam}) ---")
        seq_adam, val_adam = _adam_opt(global_best_seq, n, deadline,
                                       max_iters=max_iters_adam, lr0=0.5)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Fast stochastic local search polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random walk polishing (budget≈{remaining:.1f}s) ---")
        seq_loc, val_loc = _local_random_walk(global_best_seq, n, deadline)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
