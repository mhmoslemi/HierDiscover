"""
Fast local improvement for the step‑function problem.

The algorithm starts from the best known sequence (the global variable
`height_sequence_1`).  It then repeatedly performs a “mass‑inflation’’ linear
program (LP) that finds a partner distribution `g` with the largest possible
ℓ₁‑mass while respecting the current convolution bound `M = max(f*f)`.
Instead of the fixed mixing factor used before, the new code computes the
optimal convex combination `t ∈ [0,1]` that minimizes the maximal entry of
the mixed convolution

    ( (1‑t)·f + t·ĝ ) * ( (1‑t)·f + t·ĝ ),

where `ĝ = g/∑g`.  This simple line‑search yields a strictly larger
reduction of `max(conv)` per iteration.  The LP‑mass‑inflation loop is run
until no improvement is observed or the time budget expires.

After the LP phase a short exponentiated‑gradient (mirror‑descent) polishing
stage is applied to clean up any remaining sub‑optimality.

The whole routine respects a 1000 s wall‑clock budget, prints progress
information and stores the final sequence in the global variable
`height_sequence_1`.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# LP utilities (mass‑inflation)
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows, cols = [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows.append(np.full(length, k, dtype=np.int32))
        cols.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    offset = rows - cols                    # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Solve
        maximise   Σ g_j
        s.t.      g ≥ 0,
                  (f * g)_k ≤ M  for all k.
    Returns (g_opt, sum(g_opt)).
    """
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximise Σg  → minimise -Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is the default method for recent SciPy installations
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        # fallback – return the current vector (no improvement)
        g_opt = f.copy()
    return g_opt, g_opt.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # AdaGrad‑style decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same as the reference implementation
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution (if any); otherwise start uniformly.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            f = np.asarray(previous, dtype=np.float64)
            if f.size == n and np.sum(f) > 0:
                f = f / np.sum(f)
            else:
                raise ValueError
        except Exception:
            f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to compute the surrogate (2·n·max(conv)) – identical to
    # evaluate_sequence when the vector is ℓ₁‑normalised.
    # ------------------------------------------------------------------
    def _surrogate(v):
        return 2.0 * n * np.max(_conv_fft(v, v))

    best_seq = f.copy()
    best_val = _surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. LP‑mass‑inflation loop with optimal mixing factor.
    # ------------------------------------------------------------------
    rows, cols, offset = _get_conv_pattern(n)
    lp_iter = 0
    while time.time() < deadline and lp_iter < 15:
        lp_iter += 1
        conv_f = _conv_fft(best_seq, best_seq)
        M = conv_f.max()

        # solve the LP that finds the “heaviest’’ g compatible with the current bound
        g, sum_g = _solve_lp_max_sum(best_seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            # No extra mass can be injected → we are at a stationary point.
            print("[LP] No mass‑inflation possible, terminating LP loop.")
            break

        # Normalise g to unit ℓ₁‑mass
        g_norm = g / sum_g

        # Pre‑compute the three convolutions required for a fast line‑search
        conv_g = _conv_fft(g_norm, g_norm)
        conv_fg = _conv_fft(best_seq, g_norm)

        # ----- line search for the optimal mixing weight t ∈ [0,1] -----
        # coarse grid
        t_candidates = np.linspace(0.0, 1.0, 21)
        best_t = 0.0
        best_M = M
        for t in t_candidates:
            if t == 0.0:
                cur_M = M
            else:
                # (1‑t)²·conv_f + t²·conv_g + 2·t·(1‑t)·conv_fg
                comb = ((1.0 - t) ** 2) * conv_f + (t ** 2) * conv_g + (2.0 * t * (1.0 - t)) * conv_fg
                cur_M = comb.max()
            if cur_M < best_M:
                best_M = cur_M
                best_t = t

        # refined search around best_t (if improvement found)
        if best_t not in (0.0, 1.0):
            t_low = max(0.0, best_t - 0.05)
            t_high = min(1.0, best_t + 0.05)
            t_refined = np.linspace(t_low, t_high, 21)
            for t in t_refined:
                comb = ((1.0 - t) ** 2) * conv_f + (t ** 2) * conv_g + (2.0 * t * (1.0 - t)) * conv_fg
                cur_M = comb.max()
                if cur_M < best_M:
                    best_M = cur_M
                    best_t = t

        if best_t == 0.0:
            print(f"[LP] Iter {lp_iter}: mixing weight 0 → no improvement, stop.")
            break

        # Update the sequence with the optimal mixing weight
        new_seq = (1.0 - best_t) * best_seq + best_t * g_norm
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()               # safety normalisation

        new_val = _surrogate(new_seq)
        print(f"[LP] Iter {lp_iter}: t={best_t:.4f}, max conv ↓ {M:.6e} → {best_M:.6e}, "
              f"surrogate ↓ {best_val:.6f} → {new_val:.6f}")

        if new_val + 1e-12 < best_val:
            best_seq = new_seq
            best_val = new_val
        else:
            # If mixing did not improve, we stop early.
            print("[LP] Mixing did not improve surrogate, terminating LP loop.")
            break

    # ------------------------------------------------------------------
    # 2. Final exponentiated‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to the EG step
        max_iters = int(min(4_000_000, remaining * 0.08 * 1e5))
        print(f"\n[ExpGrad] polishing with up to {max_iters:,} iterations")
        polished_seq, polished_val = _exp_gradient_opt(best_seq, n, deadline,
                                                      max_iters=max_iters)
        if polished_val < best_val:
            best_seq = polished_seq
            best_val = polished_val
            print(f"[ExpGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
