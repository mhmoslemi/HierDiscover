"""
Propose a non‑negative step‑function (vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy is deliberately simple and focuses the whole 1000 s budget
on a (1) large diverse seed pool, (2) a short L‑BFGS refinement that
minimises a smooth log‑sum‑exp surrogate of the max‑convolution and finally
(3) a long projected sub‑gradient descent that directly reduces the true
max‑convolution.  All operations stay on the probability simplex
(∑a = 1) so the surrogate and the official evaluator coincide.
The implementation uses only NumPy (SciPy is optional for L‑BFGS) and
contains no closures – all helpers are top‑level functions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (∑=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = float(np.max(conv))
    return 2.0 * n * max_c  # denominator is 1 because ∑seq = 1


# ----------------------------------------------------------------------
# Analytic families (always l1‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    """Shape sin²(π·x) on [0,1] (zero at the ends)."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Refine a probability vector with L‑BFGS‑B on a smooth max‑conv."""
    try:
        import scipy.optimize as _opt
    except Exception:  # pragma: no cover
        raise RuntimeError("SciPy optimiser is required for L‑BFGS refinement.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)                # full convolution
            m = np.max(conv)
            shift = m
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max
            w = exp_beta / sum_exp                # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on the true objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient for the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)            # classic diminishing step size
        a -= lr * g
        a = _proj_simplex(a)                 # stay on the probability simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Short exponentiated‑gradient polish on the exact objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in logit space with a simple cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    The procedure is:
        1. Generate a rich pool of normalised candidates (analytic families,
           many Dirichlet draws, optionally the previous best).
        2. Keep the best few according to the cheap surrogate.
        3. Run a short L‑BFGS refinement on the best seed (if SciPy is
           available) to obtain a smoother starting point.
        4. Run a long projected sub‑gradient descent – the work‑horse that
           directly reduces the true max‑convolution.
        5. Optionally polish with exponentiated‑gradient and simulated‑annealing.
    The final sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between granularity and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed pool construction
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sin2(n))
    for sigma in (0.10, 0.15, 0.20):
        seeds.append(_gaussian(n, sigma))
    for p in (0.3, 0.7, 1.5, 3.0):
        seeds.append(_power(n, p))

    # Dirichlet draws with a wide range of concentration parameters
    conc_list = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in conc_list:
        for _ in range(12):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw order
            seeds.append(raw[rng.permutation(n)])  # shuffled version

    # include previous best if it exists and has the right length
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                s = arr.sum()
                if s > 0.0:
                    seeds.append(arr / s)
        except Exception:
            pass

    # normalise all seeds robustly (protect against NaNs / negatives)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    seeds = clean

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (cheap surrogate)
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Initial seed ranking (surrogate) ---")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Short L‑BFGS refinement on the best raw seed
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    try:
        # give the L‑BFGS phase ~10 % of the total budget
        lbfgs_deadline = start_time + budget_s * 0.10
        best_seq = _lbfgs_smooth_optimize(best_seq,
                                          n,
                                          lbfgs_deadline,
                                          beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                         200.0, 500.0, 1_000.0,
                                                         2_000.0, 5_000.0,
                                                         20_000.0, 50_000.0],
                                          max_iter_per_beta=4000)
        best_val = _objective_surrogate(best_seq)
        print(f"[L‑BFGS] surrogate after refinement = {best_val:.6f}")
    except Exception as e:
        print("[L‑BFGS] skipped (SciPy missing or error):", e)

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient descent – heavy lifting
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of the remaining time to sub‑gradient steps
        max_iters = int(min(3_000_000, remaining * 5_000))  # crude estimate
        max_iters = max(200_000, max_iters)                # guarantee some work
        print(f"\n--- Sub‑gradient descent (max_iters={max_iters}) ---")
        best_seq, best_val = _subgradient_opt(best_seq,
                                              n,
                                              deadline,
                                              max_iters=max_iters)
        print(f"[SubGrad] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 100_000))
        max_iters = max(100_000, max_iters)
        print("\n--- Exponentiated‑gradient polish ---")
        best_seq, best_val = _exp_gradient_opt(best_seq,
                                               n,
                                               deadline,
                                               max_iters=max_iters)
        print(f"[ExpGrad] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing walk
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 80_000))
        max_iters = max(100_000, max_iters)
        print("\n--- Simulated‑annealing refinement ---")
        best_seq, best_val = _simulated_annealing_opt(best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_iters)
        print(f"[SA] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
