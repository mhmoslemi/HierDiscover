assistantfinal**Solution Overview**

The optimizer searches for a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimises  

```
E(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```

(`conv` is the full linear convolution).  
The value of `E` does **not** change under a global scaling of `a`, therefore the
search is carried out on the ℓ₁‑simplex (`∑a = 1`).  

The programme follows three complementary ideas

1. **Very diverse seeds** – deterministic windows, random Dirichlet draws,
   sparse patterns, and a *spectral seed* obtained by taking a random unit‑
   magnitude Fourier spectrum and inverting it.  
2. **Continuation L‑BFGS** on a smooth log‑sum‑exp approximation of the max‑convolution.
   The schedule for the smoothing parameter `β` is longer than in the reference
   code and a larger number of L‑BFGS iterations is allowed.
3. **Exact mirror‑descent (exponentiated‑gradient)** on the true objective.
   Several long restarts are performed; the best result is polished once more
   with a high‑β L‑BFGS run and a cheap stochastic “mass‑transfer’’ local search.

All work is bounded by the given time budget (`budget_s`).  The best vector is
stored globally as `height_sequence_1` for warm‑starts in the next round.

```python
"""
search for a step‑function (non‑negative vector) that minimises
evaluate_sequence(sequence).  The algorithm builds a huge pool of seeds
(including a spectral seed), refines the best candidates with continuation
L‑BFGS on a smooth surrogate of the max‑convolution, then runs several
exponentiated‑gradient (mirror‑descent) optimisations on the exact objective,
followed by a tiny stochastic local search.  The final vector is returned
as a plain list of floats and stored globally as ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; result lies on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(v, 1.0 / v.size)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    return a / a.sum()


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet(conc,…,conc) – already ℓ₁‑normalised."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    Choose a random subset of size m = max(2, int(frac * n)),
    put equal mass 1/m on each chosen index, zero elsewhere.
    """
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift."""
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Spectral seed – random unit‑magnitude Fourier coefficients
# ----------------------------------------------------------------------
def _random_phase_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a vector whose discrete Fourier transform has unit magnitude
    (random phase) for all non‑DC frequencies.
    After an inverse FFT we clip negatives and renormalise.
    """
    # rfft size = n
    half = n // 2
    # DC component (frequency 0) – give a tiny positive value to keep sum >0
    dc = 1e-3
    # random phases for the remaining frequencies (except Nyquist when n even)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=half - 1)
    spec = np.empty(half + 1, dtype=np.complex128)
    spec[0] = dc
    spec[1:-1] = np.exp(1j * phases)
    if n % 2 == 0:
        spec[-1] = 0.0            # Nyquist term forced real (set to zero)
    a = np.fft.irfft(spec, n=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a list of diverse ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01):
        seeds.append(_exponential_decay(n, lam))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian family
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law
    for p in np.linspace(0.0, 4.0, 21):
        seeds.append(_power_law(n, p))

    # Beta family
    for a in np.linspace(0.5, 3.0, 7):
        for b in np.linspace(0.5, 3.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns (different sparsities)
    for frac in (0.05, 0.1, 0.15, 0.2):
        for _ in range(6):
            seeds.append(_random_sparse(n, frac, rng))

    # Low‑frequency curves
    for nf in (2, 3, 4, 5):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Spectral random‑phase seeds – they often give a flatter self‑convolution
    for _ in range(12):
        seeds.append(_random_phase_seed(n, rng))

    # Pairwise mixtures of already‑generated seeds
    base = np.arange(len(seeds))
    for _ in range(400):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of a few promising seeds
    for _ in range(30):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include the previous solution (if any)
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean and enforce ℓ₁‑normalisation
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for max‑convolution
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (obj, grad) where grad is the derivative w.r.t. logits v.
    """
    a = _softmax(v)                 # probability vector (∑a = 1)
    conv = _conv_fft(a, a)           # full convolution, length 2n‑1
    shift = np.max(conv)             # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp           # soft‑max weights over the lag index
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)       # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule: list[float] | None = None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0, 2_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)     # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun,
                                x0=v,
                                method="L-BFGS-B",
                                jac=True,
                                callback=callback,
                                options={"maxiter": max_iter_per_beta,
                                         "disp": False,
                                         "ftol": 1e-12})
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline hit during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} ({beta_val})")
            break
    return _softmax(v)               # return a probability vector


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum = 1 ⇒ 2·n·max)."""
    n = a.size
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv   # because sum(a)=1


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns g where g_i = 2·a_{k‑i} for any lag k attaining the maximum.
    """
    n = a.size
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))   # lag where the maximum is attained
    g = np.zeros_like(a)

    # i ranges such that 0 ≤ i < n and 0 ≤ idx‑i < n
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(start_a: np.ndarray,
                           deadline: float,
                           max_iters: int = 500_000,
                           eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best vector found and its objective value.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)          # diminishing step‑size
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)               # stays on the simplex

        if (it & 0x3FFFF) == 0:            # every 262 144 iterations
            cur = _objective_exact(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[EG] it={it:,} best={cur:.6f}")

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(start_a: np.ndarray,
                  deadline: float,
                  iters: int = 30_000,
                  step_frac: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a small fraction of mass from one index to another.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456)

    for it in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, a.size)
        j = rng.integers(0, a.size)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand                     # walk from the improvement
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` under the
    1000 s budget.  The best sequence is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size (kept identical to the reference implementation)
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start (n={n}, budget≈{budget_s}s) ===")

    # ------------------------------------------------------------------
    # Collect seeds (warm‑start with the previous solution if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} raw seeds")

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds – keep the best few
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # Keep a modest pool of the top‑k seeds for L‑BFGS refinement
    seeds.sort(key=_objective_exact)
    top_k = min(15, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline before finishing L‑BFGS runs.")
            break
        remaining = deadline - time.time()
        # allocate a per‑beta budget (more iterations than the reference code)
        per_beta = max(1200, int(min(5000, remaining * 0.06 * 1e5)))
        print(f"\n--- L‑BFGS run #{idx+1} (≤{per_beta} iters per β) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=per_beta)
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple exponentiated‑gradient restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        runs = 10
        iters_per_run = int(min(600_000, remaining * 0.13 / runs))
        print("\n--- Exponentiated‑gradient restarts ---")
        for r in range(runs):
            if time.time() > deadline:
                break
            # random Dirichlet start (concentration drawn uniformly)
            conc = rng.uniform(0.05, 2.0)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_exp, val_exp = _exp_gradient_optimize(start_vec,
                                                      deadline,
                                                      max_iters=iters_per_run,
                                                      eta0=2.0)
            print(f"[EG restart {r+1}] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – a long EG run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_500_000, remaining * 0.25 * 1e5))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from current best) ---")
            seq_exp, val_exp = _exp_gradient_optimize(global_best_seq,
                                                      deadline,
                                                      max_iters=max_iters,
                                                      eta0=2.5)
            print(f"[Long EG] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(50_000, remaining * 45))
        if iters > 0:
            print("\n--- Stochastic local search (mass transfers) ---")
            seq_loc, val_loc = _local_search(global_best_seq,
                                             deadline,
                                             iters=iters,
                                             step_frac=0.012)
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final polishing with very high β values
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(3000, remaining * 0.06 * 1e5)))
        print("\n--- Final polishing L‑BFGS (very high β) ---")
        seq_final = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            beta_schedule=[500_000.0,
                                                           1_000_000.0,
                                                           5_000_000.0,
                                                           10_000_000.0],
                                            max_iter_per_beta=per_beta)
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate best ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible warm‑starts in the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```
