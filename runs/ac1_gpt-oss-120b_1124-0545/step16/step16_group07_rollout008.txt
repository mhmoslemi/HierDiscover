def _smooth_max_opt(start_seq, n, deadline, p=8, max_iters=2000000, lr0=0.1):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)  # compute maxconv *2n

    # Adam:
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # conv and w
        conv = _conv_fft(a, a)
        max_val = np.max(conv)  # for tracking
        w = conv ** (p-1)
        S = np.sum(conv ** p)
        factor = (S ** (1.0/p)) / S   # = S^(1/p - 1)
        # compute correlation:
        w_rev = w[::-1]
        corr = _conv_fft(a, w_rev)
        # extract slice:
        offset = 2 * n - 2
        grad = 2.0 * factor * corr[offset:offset + n]
        # Adam update in Euclidean space:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / np.sqrt(it) / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection onto simplex:
        a = _project_simplex(a_new)
        # compute surrogate objective:
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optionally print
        # optionally decays: maybe adjust p over time.
    return best_seq, best_val
```

Project onto simplex: Write a function:

```
def _project_simplex(v):
    """Compute projection of v onto probability simplex (nonnegative sum=1)"""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    """
    Euclidean projection of v onto simplex {x >=0, sum(x)=1}
    Returns projected vector.
    """
    # From Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # All entries <= 0, project to uniform distribution
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_max_opt(start_seq, n, deadline, p=6, max_iters=1_000_000, lr0=0.5):
    a = start_seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = _objective_surrogate(a)
    # adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        # compute gradient
        w = conv ** (p - 1)
        S = np.sum(conv ** p)
        if S == 0:
            break
        factor = (S ** (1.0 / p)) / S  # S^(1/p-1)
        # correlation a and w (valid)
        w_rev = w[::-1]
        corr_full = _conv_fft(a, w_rev)
        offset = len(w) - 1
        grad = 2.0 * factor * corr_full[offset: offset + n]
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection onto simplex
        a = _project_simplex(a_new)
        # evaluate surrogate
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        if it % 20000 == 0:
            print(f"[Smooth] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The optimiser consists of
three phases:

1. **Rich multi‑start seeding** – a large set of analytically‑derived
   probability vectors (uniform, beta‑shapes, power‑laws, Gaussians,
   Dirichlet draws and a half‑sinusoid) are generated and normalised to
   ℓ₁‑norm 1.  The best seed (lowest surrogate ``2·n·max(conv)``) is kept.

2. **Smooth‑max Adam optimisation** – the non‑smooth ``max`` is
   approximated by an ℓₚ‑norm (p≈8).  The exact gradient of the
   smoothed objective is computed via FFT‑based convolution/correlation
   and used in an Adam optimiser in Euclidean space with a projection
   onto the probability simplex.  This phase quickly drives the
   surrogate down.

3. **Refinement** – a short exponentiated‑gradient (mirror‑descent) run
   on the best vector from phase 2 and a final stochastic local‑walk
   (Gaussian perturbations in log‑space) polish the result.

All phases respect a 1000 s wall‑clock budget; the best sequence is stored
in the global ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x) = 1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not cond.any():
        # All entries are non‑positive → project onto uniform simplex
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sin_shape(n: int) -> np.ndarray:
    # half‑sinusoid, all entries non‑negative
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    p: int = 8,
                    max_iters: int = 2_000_000,
                    lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the ℓₚ‑norm of the convolution (p>1) –
    a smooth surrogate for the max‑convolution.  The iterate is
    projected onto the probability simplex after each step.
    Returns the best sequence seen and its surrogate value
    ``2·n·max(conv)``.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = _objective_surrogate(a)

    # Adam moments in Euclidean coordinates
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) full self‑convolution
        conv = _conv_fft(a, a)

        # 2) smooth‑max gradient: p‑norm of conv
        conv_p = np.power(conv, p)
        S = np.sum(conv_p)
        if S == 0.0:
            # degenerate – stop early
            break
        factor = math.pow(S, 1.0 / p - 1.0)          # = S^{1/p-1}
        w = np.power(conv, p - 1)                    # conv^{p‑1}

        # 3) correlation a ⋆ w (valid part, length n)
        w_rev = w[::-1]
        corr_full = _conv_fft(a, w_rev)              # length 3n‑2
        offset = len(w) - 1                          # = 2n‑2
        corr = corr_full[offset: offset + n]         # shape (n,)

        # gradient of the smooth objective
        grad = 2.0 * factor * corr

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / math.sqrt(it)) / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # project back onto the simplex
        a = _project_simplex(a_new)

        # surrogate (exact for ℓ₁‑normalised vectors)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 20000 == 0:
            print(f"[Smooth‑max] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact ``max`` objective.
    Works in log‑softmax space, guaranteeing the simplex constraint.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search in log‑softmax space.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal = _softmax(logits + sigma * rng.normal(size=n))
        proposal_val = _objective_surrogate(proposal)

        if proposal_val < cur_val:
            cur_seq = proposal
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[Local] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val

# Main proposer

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted wall‑clock budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – keep the same length used by the reference solution
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve the previous solution (if any) – treat it as a seed
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (large diverse pool)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(40):
        a = 10 ** rng.uniform(-2, 2)   # between 0.01 and 100
        b = 10 ** rng.uniform(-2, 2)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(30):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(25):
        sigma = rng.uniform(0.05, 0.5)
        seeds.append(_gaussian(n, sigma))

    # sine‑shape
    seeds.append(_sin_shape(n))

    # Dirichlet draws with a wide range of concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise all seeds to ℓ₁‑norm 1 and discard pathological ones
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the best raw seed (surrogate = 2·n·max(conv))
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec
    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max Adam optimisation (the core improvement phase)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        # Estimate a safe iteration budget (about 1500 it/s for the double‑FFT)
        max_iters = int(min(2_000_000, remaining * 1500))
        print(f"\n--- Smooth‑max Adam phase (max_iters≈{max_iters}) ---")
        seq_smooth, val_smooth = _smooth_max_opt(
            best_seed, n, deadline, p=8, max_iters=max_iters, lr0=0.5
        )
        print(f"[Smooth‑max] surrogate = {val_smooth:.6f}")
        if val_smooth < best_seed_val:
            best_seed = seq_smooth
            best_seed_val = val_smooth

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient refinement (mirror‑descent on the exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_eg = int(min(2_000_000, remaining * 3000))
        print(f"\n--- Exponentiated‑gradient refinement (max_iters≈{max_iters_eg}) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seed, n, deadline, max_iters=max_iters_eg, eta0=0.6)
        print(f"[ExpGrad] refined surrogate = {val_eg:.6f}")
        if val_eg < best_seed_val:
            best_seed = seq_eg
            best_seed_val = val_eg

    # ------------------------------------------------------------------
    # 5. Fast stochastic local walk (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (budget≈{remaining:.1f}s) ---")
        seq_loc, val_loc = _local_random_walk(best_seed, n, deadline)
        print(f"[Local] final surrogate = {val_loc:.6f}")
        if val_loc < best_seed_val:
            best_seed = seq_loc
            best_seed_val = val_loc

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seed.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_seed_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence globally for possible warm‑starts in future runs
    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
