"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The optimiser works in the log‑softmax (logit) space and minimises a
smooth approximation of the max‑convolution using L‑BFGS‑B.  The
approximation is a log‑sum‑exp with a temperature λ that is increased
geometrically (continuation method).  Gradients are computed analytically
via FFT‑based convolutions, which makes each iteration O(n log n).  The
best sequence found is polished with a short simulated‑annealing walk
and finally stored in the global variable ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as opt


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # Guard against degenerate case (should never happen with our logits)
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient in logit space
# ----------------------------------------------------------------------
def _smooth_max_and_grad(logits: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, gradient_wrt_logits) where smooth_max is the
    λ‑smoothed approximation of max(conv(a,a)) and the gradient is
    computed analytically.
    """
    n = len(logits)

    # ---- map logits -> probability vector a (simplex) ----
    a = _softmax(logits)                     # a_i ≥ 0, ∑a = 1

    # ---- convolution a*a (full linear) ----
    conv_a = _conv_fft(a, a)                 # length 2n‑1

    # ---- smooth max via log‑sum‑exp ----
    # shift to avoid overflow/underflow
    shift = np.max(conv_a)
    # exp(lam * (conv - shift)) ∈ (0,1]
    exp_term = np.exp(lam * (conv_a - shift))
    sum_exp = np.sum(exp_term)
    smooth_max = shift + np.log(sum_exp) / lam

    # ---- weights w_k = softmax_{λ}(conv_a) ----
    w = exp_term / sum_exp                   # length 2n‑1, sum(w)=1

    # ---- gradient ∂smooth_max/∂a ----
    #   ∂c_k/∂a_j = 2 * a_{k-j}
    #   ∂smooth_max/∂a_j = Σ_k w_k * 2 * a_{k-j}
    # This is a discrete cross‑correlation of w with a (reversed)
    a_rev = a[::-1]
    conv_wa_rev = _conv_fft(w, a_rev)       # length (2n‑1)+n‑1 = 3n‑2
    # Extract the needed n entries: g_a[j] = 2 * conv_wa_rev[n-1-j]
    idx = (n - 1) - np.arange(n)            #  n‑1, n‑2, … , 0
    g_a = 2.0 * conv_wa_rev[idx]

    # ---- chain rule through the softmax (logits -> a) ----
    # ∂smooth_max/∂z = a * (g_a - ⟨a, g_a⟩)
    avg = np.dot(a, g_a)                    # scalar
    grad_logits = a * (g_a - avg)

    return smooth_max, grad_logits


def _lbfgs_obj(logits: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """Interface for scipy.optimize.minimize (value, gradient)."""
    val, grad = _smooth_max_and_grad(logits, lam)
    return val, grad


# ----------------------------------------------------------------------
# Local stochastic walk for polishing
# ----------------------------------------------------------------------
def _local_random_walk(start_seq: np.ndarray,
                       deadline: float,
                       sigma_start: float = 0.12,
                       sigma_end: float = 1e-5) -> np.ndarray:
    rng = np.random.default_rng(947832)
    cur_seq = start_seq.copy()
    best_seq = cur_seq.copy()
    # surrogate for fast comparison (exact max, no scaling)
    cur_val = np.max(_conv_fft(cur_seq, cur_seq))

    start = time.time()
    total = deadline - start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        prog = elapsed / total
        sigma = sigma_start * (1 - prog) + sigma_end * prog

        # propose via logit perturbation
        logits = np.log(cur_seq + 1e-30)
        logits_prop = logits + sigma * rng.normal(size=len(cur_seq))
        prop_seq = _softmax(logits_prop)

        prop_val = np.max(_conv_fft(prop_seq, prop_seq))
        if prop_val < cur_val - 1e-12:          # accept only strict improvement
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < np.max(_conv_fft(best_seq, best_seq)):
                best_seq = cur_seq.copy()

        it += 1
        if it % 2000 == 0:
            print(f"[Polish] iter={it:5d} best_smooth_max={np.max(_conv_fft(best_seq, best_seq)):.9f}")

    return best_seq


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – we keep the same size as the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load the previous solution if present.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.all(arr >= 0) and arr.sum() > 0.0:
                init_vec = arr / arr.sum()
            else:
                init_vec = np.full(n, 1.0 / n, dtype=np.float64)
        except Exception:
            init_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init_vec = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    # ------------------------------------------------------------------
    # 1. Prepare a small pool of seeds (uniform + few random Dirichlet).
    #    They are converted to logits for optimisation.
    # ------------------------------------------------------------------
    seed_vectors = [init_vec]

    # a few random Dirichlet draws to diversify the starting points
    for conc in (0.02, 0.1, 0.5, 1.0):
        draw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        draw /= draw.sum()
        seed_vectors.append(draw)

    # ------------------------------------------------------------------
    # 2. Run a continuation of smoothed‑max minimisation with L‑BFGS‑B.
    # ------------------------------------------------------------------
    best_a = None
    best_smooth = np.inf
    # geometric schedule for λ (temperature)
    lam_schedule = [5.0, 15.0, 45.0, 135.0, 405.0]  # last value ≈ 400 → tight approx.

    for idx, seed_vec in enumerate(seed_vectors):
        if time.time() > deadline:
            break

        # initialise logits from the seed
        logits = np.log(seed_vec + 1e-30)

        print(f"\n--- Optimisation start: seed #{idx + 1} (λ schedule length {len(lam_schedule)}) ---")
        for lam in lam_schedule:
            if time.time() > deadline:
                break
            max_iter = 150  # modest number of L‑BFGS‑B iterations per λ
            print(f"[L‑BFGS] λ={lam:.1f}  max_iter={max_iter}")

            res = opt.minimize(fun=_lbfgs_obj,
                               x0=logits,
                               args=(lam,),
                               jac=True,
                               method='L-BFGS-B',
                               options={'maxiter': max_iter,
                                        'disp': False,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12})

            logits = res.x  # warm‑start next λ value
            # keep track of the best smoothed max seen so far
            smooth_val, _ = _smooth_max_and_grad(logits, lam)
            if smooth_val < best_smooth:
                best_smooth = smooth_val
                best_a = _softmax(logits)
                print(f"   >>> New best smooth max = {best_smooth:.9f}")

    if best_a is None:
        # fallback – should never happen
        best_a = init_vec.copy()

    # ------------------------------------------------------------------
    # 3. Polishing with a simple simulated‑annealing‑like random walk.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Polishing phase (budget ≈ {remaining:.1f}s) ---")
        best_a = _local_random_walk(best_a, deadline)

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_a.tolist())
        print(f"\n=== Finished ===\nSmooth max ≈ {best_smooth:.9f}, true evaluate_sequence = {true_val:.9f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_a.tolist()
    return best_a.tolist()
