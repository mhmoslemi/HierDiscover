"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution.  The objective is scale‑invariant,
so we keep the vector ℓ₁‑normalised (sum = 1) and work on the simplex via a
soft‑max (log‑space) parametrisation.  The optimiser consists of three
phases:

1. **Diverse seed generation** – analytic windows (sine, Kaiser, Tukey,
   Gaussian, power‑law, beta‑shapes, etc.), random Dirichlet / exponential
   draws and low‑frequency cosine curves.  All seeds are ℓ₁‑normalised and the
   five best (by the exact surrogate 2·n·max(conv)) are retained.

2. **Adam‑mirrored descent** – we minimise the exact max‑convolution (sub‑
   gradient) in log‑softmax space using the Adam optimiser.  This yields fast
   early progress and a high‑precision final refinement because the step size
   is automatically adapted.  Each seed is polished for a time‑limited chunk
   of Adam steps; the best resulting sequence is kept.

3. **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the max is
   minimised with SciPy’s L‑BFGS‑B, using an increasing β‑schedule up to 1e⁶
   and generous iteration limits.  The resulting vector is finally polished
   with a short Adam pass.

The routine respects the global ``budget_s`` time limit (default 1000 s),
stores the best sequence in the global variable ``height_sequence_1`` and
returns it as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple simplex helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (scale‑invariant, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming np.sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Adam optimiser on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,

                   max_iters: int = 300_000,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution objective.
    The variables live in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    Returns (best_sequence, best_surrogate_value).
    """
    # initialise
    a = start_seq.copy()
    v = np.log(a + 1e-30)               # logits
    m = np.zeros_like(v)                # first moment
    v_hat = np.zeros_like(v)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # probability vector
        a = _softmax(v)

        # exact max‑convolution and its sub‑gradient
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute sub‑gradient g_i = 2 * a_{max_idx-i} whenever the pair exists
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # dot = <g, a> = 2 * conv[max_idx]
        dot = np.dot(g, a)

        # gradient of the exact objective O = 2 * n * max(conv)
        # (the factor 2*n is absorbed into the step‑size via lr)
        grad_v = a * (g - dot) * (2.0 * n)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # keep the best seen vector
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if t % 100_000 == 0:
                print(f"[Adam] it={t:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The optimisation variable lives in log‑softmax space.
    Returns a probability vector (ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0, 1_000_000.0]

    # Initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth approximation of the max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient w.r.t a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 123) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    stores the best found sequence in the global variable ``height_sequence_1``,
    and returns it as a ``list[float]``.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final call

    # ------------------------------------------------------------------
    # 1. Problem size (chosen as a compromise between quality and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum() if arr.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Seed generation (large, diverse pool)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic families – coarse grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.08, 0.26, 12):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 2.0, 12):
        seeds.append(_power(n, p))

    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Randomised seeds
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    for nf in (2, 3, 4, 5):
        for _ in range(5):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    seeds.append(_alternating_dirichlet(n, rng))

    # Include the previous solution if available
    if prev_arr is not None:
        seeds.append(prev_arr)

    # ------------------------------------------------------------------
    # 4. Evaluate seeds (cheap exact surrogate) and keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values among seeds:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 5. Initialise global best
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Adam polishing on the top seeds (majority of remaining time)
    # ------------------------------------------------------------------
    # Decide how many seconds to devote to Adam phases.
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # Pick a seed for a new Adam run.
        # Alternate between the current best and a random top‑seed.
        if np.random.rand() < 0.4:
            seed_seq = best_seq.copy()
        else:
            seed_seq = rng.choice(top_seeds).copy()

        # Determine a time budget for this chunk (≈ 25 % of what is left)
        chunk_budget = min(remaining * 0.25, 200.0)
        chunk_deadline = time.time() + chunk_budget

        # Run Adam (use a generous iteration cap, but will stop early by deadline)
        seq, val = _adam_optimize(seed_seq,
                                   n,
                                   chunk_deadline,
                                   max_iters=500_000,
                                   lr=0.04)

        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[Adam] improved surrogate to {best_val:.6f}")

        # If we made an improvement, we may afford a short extra polishing run
        if best_val < val and remaining > 30.0:
            extra_deadline = time.time() + min(remaining * 0.10, 20.0)
            seq2, val2 = _adam_optimize(best_seq,
                                        n,
                                        extra_deadline,
                                        max_iters=200_000,
                                        lr=0.02)
            if val2 < best_val:
                best_val = val2
                best_seq = seq2
                print(f"[Adam‑extra] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Continuation L‑BFGS fine‑tuning (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        print("\n--- L‑BFGS fine‑tuning ---")
        beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                      500.0, 1_000.0, 2_000.0, 5_000.0,
                      10_000.0, 20_000.0, 50_000.0,
                      100_000.0, 200_000.0, 500_000.0, 1_000_000.0]
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=beta_sched,
                                           max_iter_per_beta=15000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = seq_lbfgs
            print(f"[L‑BFGS] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final short Adam polish (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Final short Adam polish ---")
        final_deadline = time.time() + remaining
        seq_final, val_final = _adam_optimize(best_seq,
                                              n,
                                              final_deadline,
                                              max_iters=150_000,
                                              lr=0.02)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final
            print(f"[Final Adam] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
