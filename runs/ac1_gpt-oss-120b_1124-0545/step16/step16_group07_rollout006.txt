"""
Propose a step‑function (list of non‑negative heights) that minimises
the evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1️⃣  **Rich seed generation** – a variety of analytic families
    (uniform, beta, power‑law, Gaussian, random Dirichlet) plus the
    previously best sequence are generated and ℓ₁‑normalised.

2️⃣  **Smoothed exponentiated‑gradient descent** – the surrogate
    objective `max(conv)` is approximated by a soft‑max (temperature τ)
    which yields a smooth gradient.  The update is performed in
    log‑space (mirror‑descent w.r.t. KL divergence).  τ is annealed from
    a small value to a larger one, allowing the optimiser to first move
    in a smooth landscape and later focus on the true peak.

3️⃣  **Fine‑tuning** – (i) a classic sub‑gradient EG run on the exact
    max (identical to the reference method) and (ii) a short stochastic
    local‑walk (simulated‑annealing‑like) polish the best vector.

All phases respect the global 1000 s wall‑clock budget.  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             eta0: float = 0.5,
                             tau0: float = 0.05,
                             tau_max: float = 15.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a *smoothed* version
    of the max‑convolution.  The smoothing is performed with a
    soft‑max of temperature τ that is annealed from τ₀ to τ_max.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    L = 2 * n - 1                      # length of full self‑convolution
    N = _next_pow2(3 * n - 1)          # size sufficient for a*w convolutions

    start_time = time.time()

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- 1. compute the exact self‑convolution -----------------
        FA = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(FA * FA, n=N)[:L]

        # ----- 2. temperature schedule --------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        tau = tau0 + (tau_max - tau0) * progress

        # ----- 3. smooth weights (soft‑max over the convolution) -----
        # shift by the maximum for numerical stability
        max_c = conv_full.max()
        exp_arg = tau * (conv_full - max_c)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / L)
        else:
            w = w / w_sum

        # ----- 4. gradient of the smoothed objective ------------------
        FW = np.fft.rfft(w, n=N)
        conv_aw_full = np.fft.irfft(FA * FW, n=N)
        S2 = conv_aw_full[:n]                     # Σ_i a_i w_{j‑i}

        FW_rev = np.fft.rfft(w[::-1], n=N)
        conv_rev_full = np.fft.irfft(FA * FW_rev, n=N)
        S1 = conv_rev_full[L - 1 - np.arange(n)]  # Σ_i a_i w_{i+j}

        g = S1 + S2                               # sub‑gradient of smooth max

        # ----- 5. mirror‑descent update in log‑space -----------------
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)                       # stays on the simplex

        # ----- 6. keep the best iterate --------------------------------
        if (it % 2000) == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothEG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient on the *exact* max‑convolution
    (sub‑gradient of the active entry).  This is the method used in the
    reference solution; we run it for a short second phase to refine
    the vector obtained from the smoothed stage.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        if (it % 200_000) == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[ExactEG]   it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search that adds Gaussian noise to the logits.
    A slowly decreasing variance implements a simulated‑annealing schedule.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        # propose a new point
        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if (it % 2000) == 0:
            print(f"[Local]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val

            max_iters=max_iters,
            eta0=0.8,
        )
        print(f"[Phase 2] best surrogate = {val_phase2:.6f}")
    else:
        seq_phase2, val_phase2 = seq_phase1, val_phase1

    # ------------------------------------------------------------------
    # 5. Phase 3 – stochastic local‑walk polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase3_deadline = time.time() + remaining
        print(f"\n--- Phase 3: local random walk (budget≈{remaining:.1f}s) ---")
        seq_phase3, val_phase3 = _local_random_walk(
            seq_phase2,
            n,
            phase3_deadline,
            sigma_start=0.12,
            sigma_end=1e-5,
        )
        print(f"[Phase 3] best surrogate = {val_phase3:.6f}")
    else:
        seq_phase3, val_phase3 = seq_phase2, val_phase2

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_phase3.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_phase3:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = seq_phase3.tolist()
    return seq_phase3.tolist()
