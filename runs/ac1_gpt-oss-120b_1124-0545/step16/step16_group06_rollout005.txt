    # seeding

    # Uniform

    # previous
    if "height_sequence_1" in globals():
        prev = globals()["height_sequence_1"]
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape[0]==n and np.sum(arr)>0:
            seeds.append((arr/arr.sum(), "previous"))

    # parametric families: exps, revexps, power, beta, sin, parabola
    # for each param set compute and store.

    for c in np.linspace(0.5, 3.0, 20):

        seeds.append((_exp_family(n,c), f"exp{c:.3f}"))
        seeds.append((_rev_exp_family(n,c), f"revexp{c:.3f}"))

    # Power sweep
    for p in np.linspace(-0.5,5,30):
        if time.time()>deadline: break
        seeds.append((_power(n,p), f"power{p:.3f}"))

    # Beta sweep coarse
    alphas=np.linspace(0.1,5,7)
    betas=np.linspace(0.1,5,7)
    for a_ in alphas:
        for b_ in betas:
            if time.time()>deadline: break
            seeds.append((_beta_shape(n,a_,b_), f"beta{a_:.2f}_{b_:.2f}"))

    # Sin family
    for p in np.linspace(0.5,3,12):
        if time.time()>deadline: break
        vec = np.sin(np.pi * np.arange(1,n+1) / (n+1)) ** p
        # shift to positive and normalise
        vec = np.maximum(vec,0)
        seeds.append((vec/vec.sum(), f"sin{p:.2f}"))

    # Parabola (quadratic) shape concave
    idx=np.arange(n)
    vec = 1 - ((2*idx/(n-1))-1)**2  # concave parabola, max at center, 0 at ends
    vec = np.maximum(vec,0)

    # Random Dirichlet
    for conc in [0.01,0.02,0.05,0.1,0.2,0.5,1.0,2.0,5.0]:
        for _ in range(15):
            if time.time()>deadline: break
            vec = _random_dirichlet(n, conc, rng)
            seeds.append((vec, f"dirichlet_{conc:.3f}"))

    # Sorted random Dirichlet for monotonic shapes
    for conc in [0.1,0.2,0.5,1.0]:
        for _ in range(10):
            if time.time()>deadline: break
            vec = _random_dirichlet(n, conc, rng)
            vec = np.sort(vec)[::-1]  # decreasing
            seeds.append((vec/vec.sum(), f"sorted_{conc:.3f}"))

    # Mirror symmetric shape: generate half and mirror
    for conc in [0.2,0.5,1.0]:
        for _ in range(5):
            if time.time()>deadline: break
            half = _random_dirichlet((n+1)//2, conc, rng)
            full = np.concatenate([half, half[-2 if n%2 else -1::-1]])  # reflect
            full = full[:n]
            seeds.append((full/full.sum(), f"symmetric_{conc:.3f}"))

    # Evaluate each seed objective
    seed_objs = [(vec, objective(vec), tag) for vec, tag in seeds]

    seed_objs.sort(key=lambda x: x[1])

    best_val = objective(best_seq)
    print(f"Initial best surrogate val: {best_val:.6f}")

        print(f"L-BFGS on seed #{i+1}")

        a = start_seq.copy()
        v = np.log(a + 1e-30)  # logits

            # compute active gradient
            conv = _conv_fft(a, a)
            k = int(np.argmax(conv))

            g = np.zeros_like(a)
            i_start = max(0, k - (n-1))

            a = _softmax(v)
            cur_val = objective(a)
            if cur_val < best_val:

                best_seq = a.copy()
        return best_seq, best_val

    if time.time()<deadline:

        if val_adam < best_val:
            best_seq = seq_adam.copy()
            best_val = val_adam
            print(f"Adam improved to {best_val:.6f}")

    def hill_climb(start_seq, n, deadline, rng, sigma0=0.4, max_iters=4_000_000):
        v = np.log(start_seq + 1e-30)
        cur_seq = start_seq.copy()
        cur_val = objective(cur_seq)
        best_seq = cur_seq.copy()
        best_val = cur_val
        sigma = sigma0
        temperature = 1.0
        sigma_decay = 0.99996
        temp_decay = 0.99997
        no_improve = 0
        for it in range(1, max_iters+1):
            if time.time()>deadline:
                break
            v_prop = v + sigma * rng.normal(size=n)
            a_prop = _softmax(v_prop)
            val_prop = objective(a_prop)
            accept = False
            if val_prop < cur_val:
                accept = True
            else:
                prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
                if rng.random() < prob:

            if accept:

                cur_seq = a_prop
                cur_val = val_prop
                no_improve = 0
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()
            else:
                no_improve += 1
            sigma *= sigma_decay
            temperature *= temp_decay
            if no_improve > 250_000:

                no_improve = 0
        return best_seq, best_val

    if time.time()<deadline:

"""
Optimise a non‑negative step function (list of heights) to minimise

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm runs entirely in the given time budget (~1000 s).

The pipeline
--------------
1️⃣  Build a large pool of analytic / random seeds (uniform, exponential,
    beta, power‑law, sinusoidal, parabola, Dirichlet draws, sorted / symmetric
    variants).  Each seed is ℓ₁‑normalised, i.e. sum = 1.

2️⃣  Keep the *k* best seeds (by the surrogate 2·n·max(conv)) and run
    a smooth L‑BFGS‑B optimisation from each.  The surrogate is a
    log‑sum‑exp approximation of the max‑convolution with an increasing
    temperature schedule (β ranging up to 1e⁶).  The result is again a
    probability vector.

3️⃣  Polish the current best with an Adam‑style exponentiated‑gradient
    optimiser that works directly on the exact max‑convolution (sub‑gradient
    of the active convolution entry).  The parameters follow a
    √‑time‑decay and the first‑/second‑moment estimates are kept on the
    log‑itself (softmax parametrisation).

4️⃣  Perform a long simulated‑annealing / hill‑climbing stage that perturbs
    the logits with Gaussian noise; improvements are always taken and
    occasional uphill moves are accepted with a temperature schedule.

5️⃣  Finish with a short projected sub‑gradient sweep.

All iterates stay on the simplex (via softmax or Euclidean projection),
so the denominator in the true evaluator cancels out and the surrogate
objective equals the true one.  The routine reports progress, the best
surrogate value and finally the exact ``evaluate_sequence`` score, stores
the sequence in the global ``height_sequence_1`` and returns it as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised sequence (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective with L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 2000.0, 8000.0, 30000.0,
                         100000.0, 300000.0, 1e6]

    # log‑softmax parametrisation
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth version)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 2_000_000,
              alpha0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on logits (v = log a) using the exact sub‑gradient
    of 2·n·max(conv). Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient w.r.t. a (exact for the active entry)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                 # factor 2 comes from derivative

        # gradient w.r.t. logits (softmax Jacobian)
        mu = np.dot(g, a)
        grad_v = a * (g - mu) * (2.0 * n)            # include scalar 2·n

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        step = alpha0 / (math.sqrt(t) + 1e-6)        # slowly decaying step size
        v = v - step * m_hat / (np.sqrt(v2_hat) + eps)

        a = _softmax(v)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (baseline, short run)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.4,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb the logits with Gaussian noise, accepting improvements (or
    probabilistic uphill moves).  Works on the exact surrogate.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996
    temp_decay = 0.99997
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 250_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time.
    Returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives a finer discretisation)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Helper to compute the surrogate (exact) objective for a normalised
    # sequence (sum = 1).  The surrogate equals the true evaluate result.
    # ------------------------------------------------------------------
    def surrogate(a: np.ndarray) -> float:
        return _surrogate(a)

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution, if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (analytic families + random draws)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    seed_pool = []

    # Uniform baseline
    seed_pool.append((_uniform(n), "uniform"))

    # Previous best if available
    if prev_arr is not None:
        seed_pool.append((prev_arr, "previous"))

    # Exponential families (forward & reverse)
    for c in np.linspace(0.5, 3.0, 20):
        if time.time() > deadline:
            break
        seed_pool.append((_exp_family(n, c), f"exp_{c:.2f}"))
        seed_pool.append((_rev_exp_family(n, c), f"revexp_{c:.2f}"))

    # Power‑law family
    for p in np.linspace(-0.5, 5.0, 30):
        if time.time() > deadline:
            break
        seed_pool.append((_power(n, p), f"power_{p:.2f}"))

    # Beta‑shape family (coarse grid)
    alphas = np.linspace(0.1, 5.0, 7)
    betas = np.linspace(0.1, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            if time.time() > deadline:
                break
            seed_pool.append((_beta_shape(n, a_, b_), f"beta_{a_:.2f}_{b_:.2f}"))

    # Sinusoidal family (positive part only)
    for p in np.linspace(0.5, 3.0, 12):
        if time.time() > deadline:
            break
        vec = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
        vec = np.maximum(vec, 0)
        vec = vec / vec.sum()
        seed_pool.append((vec, f"sin_{p:.2f}"))

    # Concave parabola (peak in the middle, zero at ends)
    idx = np.arange(n, dtype=np.float64)
    parab = 1.0 - ((2.0 * idx / (n - 1)) - 1.0) ** 2
    parab = np.maximum(parab, 0)
    parab = parab / parab.sum()
    seed_pool.append((parab, "parabola"))

    # Random Dirichlet draws (various concentrations)
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 12
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            seed_pool.append((vec, f"dirichlet_{conc:.2f}"))

    # Sorted (decreasing) Dirichlet – gives monotone shapes
    for conc in [0.1, 0.2, 0.5, 1.0]:
        for _ in range(8):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            vec = np.sort(vec)[::-1]
            vec = vec / vec.sum()
            seed_pool.append((vec, f"sorted_{conc:.2f}"))

    # Symmetric construction (mirror half‑vector)
    for conc in [0.2, 0.5, 1.0]:
        for _ in range(5):
            if time.time() > deadline:
                break
            half = _random_dirichlet((n + 1) // 2, conc, rng)
            full = np.concatenate([half, half[-2 if n % 2 else -1::-1]])
            full = full[:n]
            full = full / full.sum()
            seed_pool.append((full, f"symmetric_{conc:.2f}"))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (by surrogate value)
    # ------------------------------------------------------------------
    seed_vals = [(vec, surrogate(vec), tag) for vec, tag in seed_pool]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _, _ in seed_vals[:top_k]]
    print(f"Generated {len(seed_vals)} seeds, keeping top {top_k}. "
          f"Best seed surrogate = {seed_vals[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS continuation from each promising seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 2000.0, 8000.0, 30000.0,
                     100000.0, 300000.0, 1e6]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=12000)
        val_opt = surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam exponentiated‑gradient polishing (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(2_500_000, remaining * 0.09 * 1e5))
        print(f"\n--- Adam optimisation (max_iters={max_iters}) ---")
        seq_adam, val_adam = _adam_opt(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=max_iters,
                                       alpha0=1.0)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(5_000_000, remaining * 0.08 * 1e5))
        print(f"\n--- Stochastic hill‑climbing (max_iters={max_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq,
                                         n,
                                         deadline,
                                         rng,
                                         sigma0=0.4,
                                         max_iters=max_iters)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(800_000, remaining * 0.05 * 1e5))
        if max_iters > 0:
            print("\n--- Final sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
