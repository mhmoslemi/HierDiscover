"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1.  Load the previous best sequence (if any) and fit a low‑dimensional
    Fourier model to its logarithm:
        a_i ∝ exp( c0 + Σ_{k=1}^{K} c_k·cos(2πk i/n) + d_k·sin(2πk i/n) )
    The model parameters live on the probability simplex via a soft‑max.

2.  Refine the Fourier parameters with a continuation L‑BFGS‑B optimiser
    on a *smooth* max‑convolution surrogate
        f_β(a) = 2·n·( max(conv)  +  log‑sum‑exp( β·(conv−max) ) / β )
    with an increasing schedule of β.  The objective gradients are
    obtained analytically (soft‑max → a, then convolution → gradient).

3.  Convert the final Fourier parameters back to a full probability
    vector (ℓ₁‑normalised).  Polish the result with three cheap heuristics
    that act directly on the true max‑convolution:
        * exponentiated‑gradient mirror descent
        * long projected sub‑gradient descent
        * pairwise logit hill‑climbing

4.  The best vector found is handed to ``evaluate_sequence`` once and
    stored in the global ``height_sequence_1`` for the next run.

All heavy work is done on the surrogate; only a few cheap passes use the
exact max‑convolution.  The routine respects the 1000 s budget and prints
progress information.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing (pairwise logit swaps)
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)   # small logit perturbation
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Fourier‑parameterised L‑BFGS optimiser (smooth max)
# ----------------------------------------------------------------------
# Global placeholders for the optimiser (used by the objective function)
_FOURIER_BASIS: np.ndarray | None = None
_FOURIER_BETA: float = 1.0
_FOURIER_N: int = 0


def _fourier_obj_fun(v: np.ndarray) -> tuple[float, np.ndarray]:
    """Smooth max‑convolution objective for a Fourier‑parameterised vector.

    The global variables ``_FOURIER_BASIS``, ``_FOURIER_BETA`` and
    ``_FOURIER_N`` must be set before calling this function.
    """
    global _FOURIER_BASIS, _FOURIER_BETA, _FOURIER_N
    # linear model → logits
    p = _FOURIER_BASIS @ v                     # shape (n,)
    # stable soft‑max → probability vector (ℓ₁‑normalised)
    maxp = np.max(p)
    exp_p = np.exp(p - maxp)
    a = exp_p / exp_p.sum()

    # full self‑convolution
    conv = _conv_fft(a, a)
    shift = np.max(conv)

    # smooth max via log‑sum‑exp
    exp_beta = np.exp(_FOURIER_BETA * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * _FOURIER_N * (shift + np.log(sum_exp) / _FOURIER_BETA)

    # gradient w.r.t. a (see derivation in the docstring above)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[_FOURIER_N - 1:_FOURIER_N - 1 + _FOURIER_N]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_p = a * (grad_a - dot)

    # and finally back to the Fourier parameters
    grad_v = _FOURIER_BASIS @ grad_p
    return obj, grad_v


def _lbfgs_fourier_opt(start_theta: np.ndarray,
                       n: int,
                       K: int,
                       deadline: float,
                       beta_schedule: list[float] | None = None,
                       max_iter_per_beta: int = 2000) -> np.ndarray:
    """Continuation L‑BFGS‑B on the Fourier model."""
    global _FOURIER_BASIS, _FOURIER_BETA, _FOURIER_N
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    _FOURIER_N = n

    # pre‑compute the Fourier basis (constant + cos + sin)
    i = np.arange(n, dtype=np.float64)
    cos_mat = np.cos(2.0 * np.pi * np.outer(np.arange(1, K + 1), i) / n)
    sin_mat = np.sin(2.0 * np.pi * np.outer(np.arange(1, K + 1), i) / n)
    _FOURIER_BASIS = np.vstack([np.ones((1, n)), cos_mat, sin_mat])

    v = start_theta.copy()

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        _FOURIER_BETA = float(beta)
        try:
            res = _opt.minimize(fun=_fourier_obj_fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:  # pragma: no cover
            print("[Fourier L‑BFGS] exception:", exc)
            break
    return v


def _theta_to_sequence(theta: np.ndarray) -> np.ndarray:
    """Transform Fourier parameters back to a probability vector."""
    global _FOURIER_BASIS
    p = _FOURIER_BASIS @ theta
    maxp = np.max(p)
    exp_p = np.exp(p - maxp)
    seq = exp_p / exp_p.sum()
    return seq

    log_start = np.log(start_seq + 1e-30)
    i = np.arange(n, dtype=np.float64)
    cos_mat = np.cos(2.0 * np.pi * np.outer(np.arange(1, K + 1), i) / n)
    sin_mat = np.sin(2.0 * np.pi * np.outer(np.arange(1, K + 1), i) / n)
    basis = np.vstack([np.ones((1, n)), cos_mat, sin_mat])          # shape (P, n)

    # solve basis.T @ theta ≈ log_start  (least‑squares)
    theta0, *_ = np.linalg.lstsq(basis.T, log_start, rcond=None)
    # add a tiny random jitter to escape flat regions
    theta0 += 0.02 * rng.normal(size=theta0.shape)

    # --------------------------------------------------------------
    # 4. Fourier continuation L‑BFGS optimisation (smooth max)
    # --------------------------------------------------------------
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  10000.0, 20000.0, 50000.0, 100000.0]

    # allocate a safe amount of iterations per β (capped to 4000)
    remaining = deadline - time.time()
    max_iter_per_beta = int(min(4000, remaining * 0.03 * 1e5))
    max_iter_per_beta = max(1500, max_iter_per_beta)

    print(f"\n--- Fourier L‑BFGS (n={n}, K={K}) ---")
    print(f"max_iter_per_beta = {max_iter_per_beta}")

    theta_opt = _lbfgs_fourier_opt(theta0, n, K, deadline,
                                  beta_schedule=beta_sched,
                                  max_iter_per_beta=max_iter_per_beta)

    seq = _theta_to_sequence(theta_opt)
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Fourier L‑BFGS] surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (exact max surrogate)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # aim for a few hundred thousand iterations; scale with time left
        max_exp_iters = int(min(800_000, remaining * 0.04 * 1e5))
        max_exp_iters = max(200_000, max_exp_iters)
        print(f"\n--- Exponentiated‑gradient polishing ({max_exp_iters} iters) ---")
        seq_exp, val_exp = _exp_gradient_opt(best_seq, n, deadline,
                                             max_iters=max_exp_iters)
        if val_exp < best_val:
            best_seq, best_val = seq_exp.copy(), val_exp
            print(f"[ExpGrad] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Long projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(1_200_000, remaining * 0.03 * 1e5))
        max_sub_iters = max(300_000, max_sub_iters)
        print(f"\n--- Projected sub‑gradient descent ({max_sub_iters} iters) ---")
        seq_sub, val_sub = _subgradient_opt(best_seq, n, deadline,
                                            max_iters=max_sub_iters)
        if val_sub < best_val:
            best_seq, best_val = seq_sub.copy(), val_sub
            print(f"[SubGrad] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final cheap hill‑climbing (pairwise logit swaps)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(400_000, remaining * 0.02 * 1e5))
        max_hc_iters = max(100_000, max_hc_iters)
        print(f"\n--- Hill‑climbing fine‑tuning ({max_hc_iters} iters) ---")
        seq_hc, val_hc = _hill_climbing_opt(best_seq, n, deadline,
                                            max_iters=max_hc_iters)
        if val_hc < best_val:
            best_seq, best_val = seq_hc.copy(), val_hc
            print(f"[HillClimb] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
