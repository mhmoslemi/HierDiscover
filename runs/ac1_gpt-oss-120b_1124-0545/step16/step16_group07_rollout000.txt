"""
Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑stage search:

1. **Rich seed pool** – uniform, previous best, beta‑shapes,
   power‑law, Gaussian, Dirichlet and sinusoidal families.
2. **Surrogate ranking** – evaluate the cheap surrogate
   `2·n·max(conv)` (identical to the true objective for L1‑normalised
   vectors) and keep the best few seeds.
3. **Smoothed Adam optimisation** – work in log‑softmax space.
   The non‑smooth max is replaced by the log‑sum‑exp
   ``s(α) = (1/α)·log Σ exp(α·conv)``.
   A linear schedule raises α from 0.5 to 200 while Adam
   (with decreasing learning rate) refines the logits.
4. **Local random walk** – a short simulated‑annealing‑like pass
   perturbs the logits with a Gaussian noise whose variance decays.
5. **Final evaluation** – the best sequence is evaluated with the
   official ``evaluate_sequence`` and stored as ``height_sequence_1``.

The whole procedure respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real → real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp on a 1‑D array."""
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(x - m))))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return L1‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Sinusoidal shape a_i ∝ sin(π·(i+0.5)/n)^power."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n) ** power
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _arcsine_shape(n: int) -> np.ndarray:
    """Arcsine (Beta(0.5,0.5)) shape a_i ∝ 1/sqrt(i·(n‑i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax space)
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     alpha_start: float = 0.5,
                     alpha_end: float = 200.0,
                     max_iters: int = 400_000,
                     lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective

        2·n· (1/α)·log Σ exp(α·conv(a,a))

    where a = softmax(logits).  Returns the best sequence (as a
    probability vector) seen and its *surrogate* value
    (2·n·max(conv)).  The true objective is evaluated only outside.
    """
    a = start_seq.copy()
    logits = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_surrogate = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase α from start to end
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # full convolution a*a
        conv = _conv_fft(a, a)                     # shape (2n‑1,)

        # smoothed soft‑max weights w_k ∝ exp(α·conv_k)
        # use log‑sum‑exp for numerical stability
        max_alpha_conv = np.max(alpha * conv)
        exp_term = np.exp(alpha * conv - max_alpha_conv)
        w = exp_term / exp_term.sum()               # shape (2n‑1,)

        # gradient w.r.t. a : 4·n· Σ_k w_k a_{k‑i}
        # compute Σ_k w_k a_{k‑i} via correlation:
        # h[i] = Σ_k w_k a_{k-i}   (i = 0..n‑1)
        # This equals central part of conv(w, a_rev)
        a_rev = a[::-1]
        corr = _conv_fft(w, a_rev)                 # length 3n‑2
        start = n - 1
        h = corr[start:start + n]                  # length n
        grad_a = 4.0 * n * h

        # convert to gradient on logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_logits = a * (grad_a - dot)

        # Adam update in logit space
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                   # decaying base LR
        logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

        # new probability vector
        a = _softmax(logits)

        # surrogate (exact max) – cheap to compute
        cur_surrogate = _objective_surrogate(a)
        if cur_surrogate < best_surrogate:
            best_surrogate = cur_surrogate
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SmoothAdam] it={it:7d} surrogate={best_surrogate:.6f}")

    return best_seq, best_surrogate


# ----------------------------------------------------------------------
# Fast local random walk (simulated‑annealing‑like)
# ----------------------------------------------------------------------
def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.12,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Stochastic local search in log‑softmax space.  Proposes Gaussian
    perturbations of the logits with a linearly decreasing variance.
    Returns the best sequence found and its *surrogate* value.
    """
    rng = np.random.default_rng(987654321)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1.0 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2_000 == 0:
            print(f"[LocalRW]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val

    The function returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – we stay with the size used in the reference.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Retrieve previous solution if present.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0.0:
            previous_vec = arr / arr.sum()

    # --------------------------------------------------------------
    # 1. Rich seed generation.
    # --------------------------------------------------------------
    seeds = [_uniform(n)]
    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws (log‑uniform in [0.1,10] for both parameters)
    for _ in range(40):
        a = 10 ** rng.uniform(-1.0, 1.0)   # [0.1,10]
        b = 10 ** rng.uniform(-1.0, 1.0)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(30):
        p = rng.uniform(0.0, 5.0)
        seeds.append(_power_law(n, p))

    # Gaussian draws
    for _ in range(25):
        sigma = rng.uniform(0.05, 0.45)
        seeds.append(_gaussian(n, sigma))

    # Dirichlet draws (low concentration favours sparsity)
    for conc in (0.02, 0.08, 0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Sinusoidal family (varying exponent)
    for power in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_sin_shape(n, power))

    # The classic arcsine‑shape (Beta(0.5,0.5))
    seeds.append(_arcsine_shape(n))

    # Normalise and filter degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation: {len(normed_seeds)} candidates created ---")

    # --------------------------------------------------------------
    # 2. Rank seeds by the exact surrogate (fast) and keep top K.
    # --------------------------------------------------------------
    K = 6
    surrogate_vals = [_objective_surrogate(vec) for vec in normed_seeds]
    top_idx = np.argsort(surrogate_vals)[:K]
    top_seeds = [normed_seeds[i] for i in top_idx]
    print(f"Best surrogate among seeds = {min(surrogate_vals):.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation loop – each top seed gets a time slice.
    # --------------------------------------------------------------
    global_best_seq = None
    global_best_true = np.inf

    # allocate time equally among the top seeds (with a tiny safety margin)
    remaining_seeds = len(top_seeds)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # time slice for this seed
        now = time.time()
        remaining_time = deadline - now
        # leave a margin for the following seeds and finalisation
        time_for_this = remaining_time / (remaining_seeds - idx)
        sub_deadline = now + time_for_this - 0.5   # 0.5 s safety

        # ----------------------------------------------------------
        # 3a. Smoothed Adam optimisation (major phase)
        # ----------------------------------------------------------
        max_iters = int(min(400_000, (sub_deadline - time.time()) * 2000))
        seq_opt, surrogate_opt = _smooth_adam_opt(
            seed_vec,
            n,
            sub_deadline,
            max_iters=max_iters,
        )
        # ----------------------------------------------------------
        # 3b. Local random‑walk polishing (remaining time)
        # ----------------------------------------------------------
        if time.time() < sub_deadline:
            seq_opt, surrogate_opt = _local_random_walk(
                seq_opt,
                n,
                sub_deadline,
                sigma_start=0.1,
                sigma_end=5e-6,
            )

        # ----------------------------------------------------------
        # 4. True evaluation
        # ----------------------------------------------------------
        try:
            true_val = evaluate_sequence(seq_opt.tolist())
        except Exception as exc:  # pragma: no cover
            true_val = np.inf
            print("Warning: evaluate_sequence raised:", exc)

        print(
            f"[Seed {idx + 1}/{K}] surrogate={surrogate_opt:.6f} "
            f"true={true_val:.6f}"
        )

        if true_val < global_best_true:
            global_best_true = true_val
            global_best_seq = seq_opt.copy()
            print(f"*** New global best true value: {global_best_true:.6f}")

    # --------------------------------------------------------------
    # 5. Fallback – if nothing succeeded, fall back to the best seed.
    # --------------------------------------------------------------
    if global_best_seq is None:
        # use the absolute best surrogate seed (no optimisation)
        best_idx = int(np.argmin(surrogate_vals))
        best_raw = normed_seeds[best_idx]
        global_best_seq = best_raw.copy()
        global_best_true = evaluate_sequence(global_best_seq.tolist())
        print("Fallback to raw best seed, true value =", global_best_true)

    # --------------------------------------------------------------
    # 6. Store and return.
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(
        f"\n=== Finished ===\nBest true evaluate_sequence = {global_best_true:.6f}"
    )
    return global_best_seq.tolist()
