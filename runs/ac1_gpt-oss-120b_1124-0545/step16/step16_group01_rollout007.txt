def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, math, numpy as np
    from scipy.optimize import linprog

    deadline = time.time() + budget_s - 2.0

    n = 2000   # chosen length

    rng = np.random.default_rng(seed)

    # create seed sequences
    seeds = []
    # families...
    # Dirichlet draws...
    # previous best transformation if any:
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, list) and len(prev) > 0:
            prev_arr = np.asarray(prev, dtype=np.float64)
            # downsample to length n using interpolation (simple slice)
            if prev_arr.size > n:
                # use linear interpolation with np.interp
                x_old = np.linspace(0, 1, prev_arr.size)
                x_new = np.linspace(0, 1, n)
                down = np.interp(x_new, x_old, prev_arr)
                seeds.append(down / down.sum())
            else:
                # pad with zeros if shorter
                pad = np.zeros(n)
                pad[:prev_arr.size] = prev_arr
                seeds.append(pad / pad.sum())
    except Exception:
        pass

    # Add analytic families etc.

    # Normalize each, ensure sum>0.

    # Evaluate surrogate, pick best top_k (maybe 5).
    candidates = [(s, _objective_surrogate(s)) for s in seeds]
    candidates.sort(key=lambda x: x[1])
    best_f = candidates[0][0].copy()
    best_val = candidates[0][1]

    # iterative LP+blend
    it = 0
    no_improve = 0
    while time.time() < deadline and it < 30 and no_improve < 5:
        it += 1
        # solve LP
        g, t_opt, ok = _lp_opt_step(best_f, deadline)
        if not ok:
            print("[LP] failed")
            break
        # current max conv for f:
        cur_val = _objective_surrogate(best_f)
        if t_opt * 2 * n >= cur_val - 1e-6:   # t_opt approx = max conv estimated from LP ; surrogate = 2*n*maxconv
            # not improvement
            no_improve += 1
            print(f"[LP] iteration {it}: no improvement (t={t_opt:.6e})")
            break
        # blend search
        cand_f, cand_val = _blend_improve(best_f, g, n, deadline)
        if cand_val < best_val - 1e-12:
            best_f = cand_f
            best_val = cand_val
            print(f"[Blend] iteration {it}: improved surrogate = {best_val:.6f}")
            no_improve = 0
        else:
            # maybe use g directly if better:
            val_g = _objective_surrogate(g)
            if val_g < best_val:
                best_f = g
                best_val = val_g
                print(f"[LP] iteration {it}: g itself better surrogate = {best_val:.6f}")
                no_improve = 0
            else:
                no_improve += 1

    # L-BFGS refinement if time left
    if time.time() < deadline:
        # Run L-BFGS with moderate iterations
        seq_opt = _lbfgs_smooth_optimize(best_f, n, deadline,
                                         beta_schedule=[1,5,20,80,200,500,1000,2000,5000,20000,50000,100000,200000],

"""
search for a non‑negative step‑function that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The strategy combines three complementary ideas:

1. **Diverse seed generation** – analytic families (uniform, triangular,
   Gaussian, power‑law, sin²) together with many Dirichlet draws are
   ℓ₁‑normalised. The best few seeds are kept as starting points.

2. **Alternating linear‑program + blending** – for a fixed ℓ₁‑normalised
   vector `f` we solve the LP

       minimise  t
       subject to  (f * g)_k ≤ t   for every convolution index k,
                   Σ g_j = 1, g_j ≥ 0

   which yields the partner distribution `g` giving the smallest possible
   maximal convolution with `f`.  A line‑search blend
   `(1‑λ)·f + λ·g` (λ∈[0,1]) is performed and the best blend replaces `f`.
   This step is repeated while it yields improvement.

3. **Refinement** – a projected sub‑gradient descent on the exact
   objective `2·n·max(conv)` followed by a short L‑BFGS optimisation of a
   smooth log‑sum‑exp surrogate pushes the peak a little further down.

All phases respect a global deadline (the 1000 s budget).  The final
sequence is returned as a plain list of floats and stored in the global
variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS refinement (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine a ℓ₁‑normalised vector with L‑BFGS‑B on a smooth max‑conv."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local, beta_local=beta_val):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)                # full conv
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_local * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_local)

            w = exp_beta / sum_exp                # soft‑weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            # If something goes wrong we just abort the current beta level
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Linear‑program step (minimise peak of f * g)
# ----------------------------------------------------------------------
def _lp_opt_step(f: np.ndarray, deadline: float):
    """
    Given a probability vector ``f`` (sum=1, non‑negative) solve

        minimise    t
        subject to   (f * g)_k ≤ t   for all k
                     Σ g_j = 1
                     g_j ≥ 0

    Returns (g, t, succeeded).
    """
    n = len(f)
    # Build the convolution matrix B where B_{k,j} = f_{k-j} (if index in range)
    i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)
    rows = (i + j).ravel()                             # length n*n
    cols = np.broadcast_to(j, (n, n)).ravel()
    data = np.repeat(f, n)                             # length n*n
    B = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Column for variable t (coefficient –1 in every inequality)
    t_col = sparse.csr_matrix((-np.ones(2 * n - 1, dtype=np.float64),
                               (np.arange(2 * n - 1), np.full(2 * n - 1, n))),
                              shape=(2 * n - 1, n + 1))

    A_ub = sparse.hstack([B, t_col])
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    # Equality Σ g_j = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    bounds = [(0, None)] * (n + 1)

    # Solve with HiGHS (fast for sparse problems)
    try:
        res = linprog(c,
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception as e:  # pragma: no cover
        print("[LP] exception:", e)
        return None, None, False

    if not res.success:
        # Infrequent failures (numerical); just give up this step
        print("[LP] failed:", res.message)
        return None, None, False

    g = res.x[:n]
    t = res.x[-1]

    # Numerical clean‑up
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s > 0.0:
        g /= s
    else:
        g = np.full_like(g, 1.0 / n)

    return g, t, True


# ----------------------------------------------------------------------
# Blending helper – line‑search between f and its LP partner g
# ----------------------------------------------------------------------
def _blend_improve(f: np.ndarray,
                   g: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Returns the best convex blend (1‑λ)·f + λ·g (λ∈[0,1]) and its surrogate
    value.  A small grid of λ values is examined.
    """
    best_seq = f.copy()
    best_val = _objective_surrogate(best_seq)

    # λ grid (skip λ=0 which is f itself)
    for lam in np.linspace(0.1, 1.0, 10):
        if time.time() > deadline:
            break
        cand = (1.0 - lam) * f + lam * g
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Standard sub‑gradient method for 2·n·max(conv(a,a)).
    Returns the best sequence seen and its surrogate value.
    """
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)        # classic diminishing step‑size
        a = a - lr * g
        a = _proj_simplex(a)             # stay on the probability simplex

    return best_seq, best_val

    """

    start_time = time.time()

    # ------------------------------------------------------------------

    # previous best (if any) – down‑sample or pad to length n

        except Exception:
            pass

    # normalise all seeds robustly
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 3. Pick a few promising seeds (cheap surrogate ranking)
    # ------------------------------------------------------------------
    evaluated = [(s, _objective_surrogate(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    # initialise current best
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Alternating LP + blend
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.time() < deadline and it < 30 and no_improve < 5:
        it += 1
        # Solve the LP for the present best sequence
        g, t_opt, ok = _lp_opt_step(best_seq, deadline)
        if not ok:
            break

        # Current max‑convolution of best_seq
        cur_val = _objective_surrogate(best_seq)

        # t_opt is the *minimum* possible max(conv(best_seq, g)).
        # If it does not improve the current peak we stop.
        if 2.0 * n * t_opt >= cur_val - 1e-9:
            no_improve += 1
            print(f"[LP] iteration {it}: no improvement (t={t_opt:.6e})")
            break

        # Find the best λ‑blend between best_seq and LP partner g
        blended, blended_val = _blend_improve(best_seq, g, n, deadline)

        if blended_val < best_val - 1e-12:
            best_seq = blended
            best_val = blended_val
            no_improve = 0
            print(f"[Blend] it={it} improved surrogate = {best_val:.6f}")
        else:
            # fallback: maybe g itself is already better
            g_val = _objective_surrogate(g)
            if g_val < best_val:
                best_seq = g
                best_val = g_val
                no_improve = 0
                print(f"[LP‑g] it={it} adopted g, surrogate = {best_val:.6f}")
            else:
                no_improve += 1

    # ------------------------------------------------------------------
    # 5. Sub‑gradient descent (fine‑grained optimisation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget to sub‑gradient steps
        max_iters = int(min(400_000, remaining * 200.0))  # crude scaling
        if max_iters > 0:
            print("\n--- Sub‑gradient descent (max_iters={}) ---".format(max_iters))
            seq_sg, val_sg = _subgradient_descent(best_seq, deadline, max_iters=max_iters)
            if val_sg < best_val:
                best_seq = seq_sg
                best_val = val_sg
                print(f"[SubGrad] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iter_per_beta = int(min(4000, remaining * 2.0))  # keep it quick
        max_iter_per_beta = max(1500, max_iter_per_beta)
        print("\n--- Final L‑BFGS polish ---")
        seq_lb = _lbfgs_smooth_optimize(best_seq,
                                        n,
                                        deadline,
                                        beta_schedule=[
                                            1.0, 5.0, 20.0, 80.0, 200.0,
                                            500.0, 1_000.0, 2_000.0,
                                            5_000.0, 20_000.0, 50_000.0,
                                            100_000.0, 200_000.0],
                                        max_iter_per_beta=max_iter_per_beta)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_seq = seq_lb
            best_val = val_lb
            print(f"[L‑BFGS] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
