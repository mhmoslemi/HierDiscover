def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 2000  # maybe adjust to 2500

    # initialize f uniform
    f = np.full(n, 1.0/n, dtype=np.float64)

    # ensure positivity epsilon
    eps = 1e-12
    f = np.maximum(f, eps)
    f /= f.sum()

    best_f = f.copy()
    best_val = _objective(f)

    # Build pattern matrix B
    i = np.arange(n)
    row_idx = (i[:,None] + i[None,:]).ravel()
    col_idx = np.tile(i, n)
    data_pat = np.ones_like(row_idx, dtype=np.float64)
    B = sparse.csr_matrix((data_pat, (row_idx, col_idx)), shape=(2*n-1, n))

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute M
        conv_f = _conv_fft(f, f)
        M = float(np.max(conv_f))

        # Build constraints matrix
        A = B.multiply(f)   # each column j multiplied by f_j

        b_vec = np.full(2*n-1, M, dtype=np.float64)

        # linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # call linprog
        # we can set bounds for each variable (0, None)
        bounds = [(0, None)] * n

        # solve
        res = opt.linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if not res.success:
            print(f"[Iter {iteration}] LP failed: {res.message}")
            break

        g = res.x
        # ensure g > 0 to avoid zero columns
        g = np.maximum(g, eps)

        # convex combination step size
        t = 0.2
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, eps)
        f_new /= f_new.sum()

        # compute objective
        val_new = _objective(f_new)

        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[Iter {iteration}] improved objective: {best_val:.6f}")

        # check improvement stagnation maybe break if little improvement
        if np.abs(best_val - val_new) < 1e-7:
            # optional break condition
            pass

        f = f_new

    # final refinement using exponentiated gradient (mirror descent) maybe
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(400_000, remaining * 0.1 * 1e5))
        if max_iters > 0:
            seq_exp, val_exp = _exp_gradient_opt(best_f, n, deadline, max_iters=max_iters)
            if val_exp < best_val:
                best_val = val_exp
                best_f = seq_exp
                print(f"[ExpGrad final] improved to {best_val:.6f}")

    # final evaluation
    true_val = evaluate_sequence(best_f.tolist())
    print(f"Final evaluate_sequence = {true_val:.6f}")

    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()

Propose a non‑negative step function via a Frank‑Wolfe (conditional
gradient) scheme on the exact max–convolution surrogate, followed by a
mirror‑descent polishing step.  The method works directly on the simplex
and uses a linear program to find a descent direction that maximises the
ℓ₁‑mass while keeping the convolution with the current iterate below its
current maximum.  The convex combination of the current iterate and the
LP‑solution is accepted only when it strictly improves the
2·n·max(conv) surrogate.  After the Frank‑Wolfe phase a few
exponentiated‑gradient iterations finish the job.  The routine respects
the global 1000 s deadline and returns the best sequence found.
"""
import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

# Exponentiated‑gradient polish (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Frank‑Wolfe (conditional gradient) step using a linear program
# ----------------------------------------------------------------------
def _fw_step(f: np.ndarray,
             B: _sp.csr_matrix,
             n: int,
             deadline: float,
             t: float = 0.2,
             eps: float = 1e-12) -> tuple[np.ndarray, float, bool]:
    """
    Perform one Frank‑Wolfe iteration.

    Returns (f_next, surrogate_val, improved_flag).
    """
    # current convolution max
    conv_f = _conv_fft(f, f)
    M = float(np.max(conv_f))

    # Linear programme: maximise sum(g)  subject to  B @ g <= M (rowwise)
    A = B.multiply(f)          # each column j scaled by f_j
    b_vec = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # minimise -sum(g)
    bounds = [(0.0, None)] * n

    res = _opt.linprog(c,
                       A_ub=A,
                       b_ub=b_vec,
                       bounds=bounds,
                       method='highs')
    if not res.success:
        # LP failed – treat as no progress.
        return f, _objective_surrogate(f), False

    g = res.x
    # enforce strict positivity (required for next LP)
    g = np.maximum(g, eps)

    # convex combination and renormalisation
    f_candidate = (1.0 - t) * f + t * g
    f_candidate = np.maximum(f_candidate, eps)
    f_candidate /= f_candidate.sum()

    surrogate_val = _objective_surrogate(f_candidate)

    improved = surrogate_val < _objective_surrogate(f)
    if improved:
        return f_candidate, surrogate_val, True
    else:
        # discard the move
        return f, _objective_surrogate(f), False

                                                 max_iters=max_exp_iters)
            if val_exp < best_surrogate:
                best_f = seq_exp
                best_surrogate = val_exp
                print(f"[ExpGrad] improved surrogate to {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_f.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
