"""
Improved optimizer for the step‑function bound.

The algorithm consists of three stages:

1. **Rich seed generation** – a collection of analytic families (uniform,
   beta‑shapes, power‑law, Gaussian, Dirichlet) together with the previous
   best solution is generated and ℓ₁‑normalised.

2. **Smoothed‑max mirror descent** – we minimise a smooth surrogate of the
   true objective:
        L(a) = log ∑ₖ exp(α·(a∗a)[k])
   (α is increased from a modest to a large value).  The gradient of L is a
   weighted sum of the individual convolution gradients, which can be obtained
   with two FFT‑based convolutions per iteration.  The optimisation lives in the
   log‑softmax (probability‑simplex) space, i.e. we maintain `log_a` and set
   `a = softmax(log_a)`.  A decaying step‑size ηₜ = η₀/√t is used.

3. **Fine polishing** – a fast stochastic local walk (Gaussian perturbation
   of the logits) is run for the remaining time budget.

The best sequence seen (according to the exact surrogate
`2·n·max(conv)`) is returned and stored in the global variable
`height_sequence_1`.  All numbers are non‑negative and the code respects the
1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (used for ranking & final evaluation)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n} (ℓ₁‑normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p (ℓ₁‑normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; σ = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc`` (ℓ₁‑normalised)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed‑max mirror‑descent optimiser
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    base_alpha: float = 5.0,
                    final_alpha: float = 120.0,
                    eta0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp surrogate of `max(conv)`.
    Works in log‑softmax space (a = softmax(log_a)).
    """
    log_a = np.log(start_seq + 1e-30)          # initialise logits
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    it = 0
    while time.time() < deadline:
        # current probabilities
        a = _softmax(log_a)

        # convolution a*a (full length = 2n-1)
        conv = _conv_fft(a, a)

        # exact surrogate for tracking
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Smooth] it={it:8d} surrogate={best_val:.6f}")

        # ---- smoothed gradient of log‑sum‑exp ----
        # linearly increase α from base_alpha to final_alpha
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = base_alpha + (final_alpha - base_alpha) * progress

        # softmax weights over convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()                # shape (2n‑1,)

        # gradient of Σₖ pₖ·convₖ w.r.t. a:
        #   ∂convₖ/∂a_j = 2·a_{k‑j} (zero when out‑of‑range)
        # we can compute g = 2·(p * a_rev) full convolution
        a_rev = a[::-1]                                 # reversed a, length n
        temp = _conv_fft(p, a_rev)                      # length 3n‑2
        grad = 2.0 * temp[(n - 1):(n - 1) + n] * alpha   # shape (n,)

        # step‑size schedule
        it += 1
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad                      # mirror‑descent update

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fast stochastic local walk (simulated‑annealing‑like)
# ----------------------------------------------------------------------
def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations of the logits, accepting only improvements.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1.0 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[Local]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Problem size (kept identical to the reference solution)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Pull the previous solution if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if isinstance(prev, list):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()          # ℓ₁‑normalise
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(30):
        a = 10 ** rng.uniform(-1, 1)          # 0.1 … 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.08, 0.5)
        seeds.append(_gaussian(n, sigma))

    # Dirichlet draws (low concentration → spikier vectors)
    for conc in (0.05, 0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Pick the best seed (according to the exact surrogate)
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec

    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max mirror‑descent optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Smoothed‑max mirror‑descent phase ---")
        seq_smooth, val_smooth = _smooth_max_opt(best_seed,
                                                 n,
                                                 deadline,
                                                 start_time,
                                                 base_alpha=5.0,
                                                 final_alpha=120.0,
                                                 eta0=0.20)
        print(f"[Smooth] surrogate = {val_smooth:.6f}")
        global_best_seq = seq_smooth.copy()
        global_best_val = val_smooth
    else:
        global_best_seq = best_seed.copy()
        global_best_val = best_seed_val

    # ------------------------------------------------------------------
    # 4. Final stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (≈ {remaining:.1f}s left) ---")
        seq_loc, val_loc = _local_random_walk(global_best_seq, n, deadline)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse in the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
