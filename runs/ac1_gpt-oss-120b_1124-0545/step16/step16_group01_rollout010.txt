"""
Optimisation of a non‑negative step‑function for the `evaluate_sequence`
objective.

The strategy is deliberately simple and fast:

1. **Initialisation** – Load the previous best solution (if any) and
   generate a small pool of analytic seeds (uniform, triangular, Gaussian,
   power‑law, sin²) together with a few Dirichlet draws.  Keep the best
   seed w.r.t. the exact surrogate `2·n·max(conv)` (the surrogate is equal
   to the official objective when the vector is ℓ₁‑normalised).

2. **Long sub‑gradient descent** – Starting from the best seed perform a
   projected sub‑gradient descent on the exact surrogate.  The sub‑gradient
   of `max(conv)` at the active shift is `g_i = 2·a_{s−i}` on the indices
   that actually contribute to the peak.  After each step the vector is
   projected onto the probability simplex (Euclidean projection).  A
   decreasing step size `η_t = 0.8 / sqrt(t)` is used and the current best
   vector is stored.

   The loop runs as long as time permits (up to a few hundred‑thousand
   iterations) and guarantees monotone improvement of the *recorded* best
   value.

3. **Final simulated‑annealing polish** – A short annealing walk in
   log‑probability space explores the neighbourhood of the best vector.
   The temperature and proposal scale are linearly annealed.

4. **Return** – The best vector found is evaluated with the official
   `evaluate_sequence` function, printed and stored in the global variable
   `height_sequence_1` for the next round.

The whole search respects the supplied time budget (default 1000 s) and
uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using a real FFT.
    Returns an array of length ``2*len(a)-1``.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex (sum=1, ≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    """Shape sin²(π·x) on [0,1] (zero at the ends)."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact surrogate (2·n·max(conv))
# ----------------------------------------------------------------------
def _subgradient_descent(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on ``2·n·max(conv)`` for an ℓ₁‑normalised
    vector.  Returns the best vector encountered and its surrogate value.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    # initial surrogate value
    conv = _conv_self_fft(a)
    best_val = float(2.0 * n * conv.max())

    last_improve = 0
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        # time check – leave a small safety margin for the final evaluation
        if time.time() > deadline:
            break

        conv = _conv_self_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = float(2.0 * n * conv[max_idx])

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] iteration {it:7d}, best = {best_val:.6f}")

        # sub‑gradient of the active entry (see comment in the original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule – a little more aggressive than the baseline
        lr = 0.8 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        # if we have not improved for a long while, inject a tiny random jitter
        if it - last_improve > 250_000:
            # jitter in log‑space to keep positivity
            log_a = np.log(a + 1e-30)
            log_a += 0.005 * rng.normal(size=n)
            a = np.exp(log_a)
            a = a / a.sum()
            last_improve = it  # reset stagnation counter

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Random walk in log‑probability space with a simple cooling schedule."""
    n = start_seq.shape[0]
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)

    cur_seq = start_seq.copy()
    cur_val = float(2.0 * n * _conv_self_fft(cur_seq).max())
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6          # temperature
        sigma = 0.15 * (1.0 - progress) + 1e-4      # proposal scale

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = np.exp(proposal_logits)
        proposal_seq = proposal_seq / proposal_seq.sum()
        proposal_val = float(2.0 * n * _conv_self_fft(proposal_seq).max())

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] iteration {it:7d}, best surrogate = {best_val:.6f}")

    return best_seq, best_val

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – keep the same size as the previous run (4000)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            n = arr.size
        except Exception:
            arr = None
            n = 4000
    else:
        arr = None
        n = 4000

    print(f"\n=== Optimisation start (n={n}) ===")

    # --------------------------------------------------------------
    # 1. Seed pool – a few analytically‑defined families + Dirichlet draws
    # --------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))
    seeds.append(_sin2(n))

    # a handful of Dirichlet samples with diverse concentration parameters
    conc_list = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in conc_list:
        for _ in range(4):
            d = _random_dirichlet(n, conc, rng)
            seeds.append(d)                     # raw draw
            seeds.append(np.sort(d)[::-1])      # descending order

    # include the previous best if we have one
    if arr is not None and arr.sum() > 0.0:
        seeds.append(arr / arr.sum())

    # normalise everything robustly (makes the simplex projection cheaper later)
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # --------------------------------------------------------------
    # 2. Pick the best raw seed w.r.t. the exact surrogate
    # --------------------------------------------------------------
    best_seed = None
    best_val = math.inf
    for s in seeds:
        val = float(2.0 * n * _conv_self_fft(s).max())
        if val < best_val:
            best_val = val
            best_seed = s
    print(f"[Init] best raw surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Long sub‑gradient descent (most of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        # we let the sub‑gradient loop decide how many iterations it can fit
        max_sub_iters = 1_200_000  # a generous upper bound
        print("\n--- Sub‑gradient descent ---")
        seq_sub, val_sub = _subgradient_descent(best_seed,
                                                deadline,
                                                max_iters=max_sub_iters)
        if val_sub < best_val:
            best_val = val_sub
            best_seed = seq_sub
            print(f"[SubGrad] improved surrogate to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final simulated‑annealing polish (short but useful)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa_iters = int(min(400_000, remaining * 0.04 * 1e5))
        if max_sa_iters > 0:
            print("\n--- Simulated‑annealing polish ---")
            seq_sa, val_sa = _simulated_annealing(best_seed,
                                                  deadline,
                                                  max_iters=max_sa_iters)
            if val_sa < best_val:
                best_val = val_sa
                best_seed = seq_sa
                print(f"[SA] final surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seed.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
