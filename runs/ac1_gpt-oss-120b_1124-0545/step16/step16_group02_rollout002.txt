def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = 1 << ((L - 1).bit_length())
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv
```

"""
Propose candidate step‑function via guided simulated‑annealing.

The algorithm starts from the best known solution (global variable
``height_sequence_1`` if present, otherwise a uniform seed).  It then
performs a long simulated‑annealing / hill‑climbing phase that repeatedly
moves a small amount of mass from an index that contributes strongly to
the current maximal convolution entry (identified via the sub‑gradient)
to a random other index.  The move size decays geometrically and the
temperature also decays, allowing early exploratory moves and later fine
tuning.  After the annealing run a short projected sub‑gradient polish
is applied.  The best sequence found is returned as a plain list of
floats and stored again in ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a single FFT (real‑valued).
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {w ≥ 0, sum(w)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Exact surrogate for L1‑normalised vectors:
        2 * n * max(conv(a,a))
    where conv is the full linear convolution.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * np.max(conv))

    Search for a non‑negative step‑function that minimises the official
    `evaluate_sequence` objective within the allocated time budget.

    start_time = time.time()
    deadline = start_time + budget_s - 2.0     # safety margin

    # 0. Starting point

    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple)) and len(previous) > 0:

        n = 4000

        print(f"No previous solution – starting from uniform (n={n})")

    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    cur_seq = best_seq.copy()
    cur_val = best_val

    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Guided simulated‑annealing / hill‑climbing
