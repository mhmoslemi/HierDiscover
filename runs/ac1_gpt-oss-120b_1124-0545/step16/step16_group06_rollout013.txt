"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1️⃣  **Seed** – use the best known sequence (``height_sequence_1``) or
    generate a small pool of analytic / random seeds and keep the best
    according to the surrogate ``2·n·max(conv)`` (the L1‑norm is irrelevant).

2️⃣  **Active‑set LP refinement** – given a current vector ``f`` we:
        • compute *M* = max(conv(f,f));
        • take only the most critical convolution indices (those within a few % of *M*);
        • solve a tiny linear programme  
              maximise   ∑ gᵢ               (i.e. maximise the ℓ₁‑norm)  
              subject to conv(f,g) ≤ M  on the selected indices and g ≥ 0,
          which yields a “direction” ``g`` that tries to spread the mass
          without breaking the tight constraints;
        • rescale ``g`` to have the same ℓ₁‑norm as ``f`` (namely 1);
        • line‑search on the convex combination ``(1‑t)·f + t·g`` (using the
          bilinearity of convolution) and keep the best ``t``.
        This loop is repeated while the objective improves (typically a few
        dozen iterations).  Only a handful of constraints are used, so the
        LP is tiny (≤30 rows) and solved in a fraction of a second.

3️⃣  **Fine polishing** – starting from the refined vector we run
    exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
    objective for a large number of cheap iterations, finishing with a
    short simulated‑annealing hill‑climber.

The whole search respects the 1000 s wall‑clock budget, never returns a
negative‑infinity value and stores the best sequence in the global
``height_sequence_1`` variable for future runs.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ₁‑normalised sequence: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    # decreasing exponential a_i ∝ r^i
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _power_law(n: int, p: float) -> np.ndarray:
    a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (soft‑max logits)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (works directly on logits)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.3,
                    max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Small‑scale annealing on the exact surrogate."""
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996
    temp_decay = 0.99998
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 150_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Active‑set linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 30,
               active_frac: float = 0.02,
               max_constraints: int = 30) -> np.ndarray:
    """
    Repeatedly improve ``f`` by solving a tiny LP that maximises the ℓ₁‑norm
    of a feasible partner ``g`` under the most critical convolution constraints.
    Returns the improved sequence (still ℓ₁‑normalised).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for the LP step.")

    cur_f = f.copy()
    cur_val = _objective_surrogate(cur_f)
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # 1) current maximum autocorrelation and active indices
        conv_ff = _conv_fft(cur_f, cur_f)
        M = conv_ff.max()
        # take indices whose value is within ``active_frac`` of M
        thresh = M * (1.0 - active_frac)
        active_idx = np.where(conv_ff >= thresh)[0]
        if len(active_idx) > max_constraints:
            # keep the top‑max ones only
            topk = np.argpartition(conv_ff[active_idx], -max_constraints)[-max_constraints:]
            active_idx = active_idx[topk]

        K = len(active_idx)
        if K == 0:
            break

        # 2) Build the LP A·g ≤ M, g ≥ 0
        # A has shape (K, n) and row k contains f[i] for all i+j = active_idx[k]
        A = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i runs from i_start … i_end; j = k - i
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k - i_vals
            A[row, j_vals] = cur_f[i_vals]

        b = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)                 # minimise –∑gᵢ
        bounds = [(0.0, None)] * n

        lp_res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                              method="highs", options={"presolve": True})
        if not lp_res.success:
            # fallback: stop refining
            break

        g = lp_res.x
        sum_g = g.sum()
        # if the LP cannot increase the ℓ₁‑norm, we are stuck
        if sum_g <= 1.0 + 1e-9:
            break

        # 3) Normalise to keep ℓ₁‑norm = 1
        g_norm = g / sum_g

        # 4) Find best convex combination (1‑t)·f + t·g_norm
        conv_fg = _conv_fft(cur_f, g_norm)
        conv_gg = _conv_fft(g_norm, g_norm)

        # grid search for t in (0,1]
        t_candidates = np.linspace(0.0, 1.0, 21)[1:]   # exclude 0 which gives current M
        best_t = 0.0
        best_M = M
        for t in t_candidates:
            comb = ((1 - t) ** 2) * conv_ff + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_gg
            cur_M = comb.max()
            if cur_M < best_M:
                best_M = cur_M
                best_t = t

        if best_t == 0.0:
            # no improvement possible with this direction
            break

        # update f
        cur_f = (1 - best_t) * cur_f + best_t * g_norm
        cur_val = 2.0 * n * best_M   # surrogate value
        # early exit if improvement is tiny
        if (cur_val - cur_val) / cur_val < 1e-8:
            break

    return cur_f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated wall‑clock budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the same length as the previous best.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Try to reuse the previous best if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seed_vec = arr / arr.sum()
            else:
                seed_vec = None
        except Exception:
            seed_vec = None
    else:
        seed_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed pool (few analytic families + Dirichlet draws)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best, if any
    if seed_vec is not None:
        candidate_pool.append((seed_vec, _objective_surrogate(seed_vec), ('previous',)))

    # a few exponential families
    for c in np.linspace(0.5, 3.0, 13):
        candidate_pool.append((_exp_family(n, c), _objective_surrogate(_exp_family(n, c)), ('exp', c)))
        candidate_pool.append((_rev_exp_family(n, c), _objective_surrogate(_rev_exp_family(n, c)), ('revexp', c)))

    # a few beta shapes
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            cand = _beta_shape(n, a, b)
            candidate_pool.append((cand, _objective_surrogate(cand), ('beta', a, b)))

    # random Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]
    for conc in concs:
        for _ in range(15):
            if time.time() > deadline:
                break
            cand = _random_dirichlet(n, conc, rng)
            candidate_pool.append((cand, _objective_surrogate(cand), ('dirichlet', conc)))

    # keep the top‑k seeds (by surrogate value)
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 2. Pick the current best (lowest surrogate) as the starting point.
    # ------------------------------------------------------------------
    cur_best_seq = top_seeds[0].copy()
    cur_best_val = _objective_surrogate(cur_best_seq)
    print(f"[Init] surrogate = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Active‑set LP refinement (the novel step)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Active‑set LP refinement ---")
        refined = _lp_refine(cur_best_seq, n, deadline,
                             max_iters=25,
                             active_frac=0.015,
                             max_constraints=30)
        refined_val = _objective_surrogate(refined)
        print(f"[LP‑refine] surrogate = {refined_val:.6f}")
        if refined_val < cur_best_val:
            cur_best_seq = refined
            cur_best_val = refined_val
            print(f"[Update] improved to {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (lots of cheap iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to EG
        max_exp_iters = int(min(4_000_000, remaining * 0.09 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(cur_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < cur_best_val:
            cur_best_seq = seq_exp
            cur_best_val = val_exp
            print(f"[ExpGrad] improved to {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climbing (light simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(2_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Stochastic hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(cur_best_seq,
                                         n,
                                         deadline,
                                         rng,
                                         sigma0=0.2,
                                         max_iters=max_hc_iters)
        if val_hc < cur_best_val:
            cur_best_seq = seq_hc
            cur_best_val = val_hc
            print(f"[HillClimb] improved to {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(cur_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = cur_best_seq.tolist()
    return cur_best_seq.tolist()
