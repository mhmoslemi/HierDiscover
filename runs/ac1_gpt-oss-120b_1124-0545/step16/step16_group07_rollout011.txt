"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of:

1. **Seed generation** – a diverse set of ℓ₁‑normalised vectors
   (uniform, Dirichlet, power‑law, Gaussian, Beta‑shapes, and a
   dedicated “inverse‑sqrt’’ shape).  The 5 best seeds according to the
   exact surrogate 2·n·max(conv) are kept.

2. **Smooth Adam optimisation** – a mirror‑descent on the *soft‑max*
   approximation of the max‑convolution.  The temperature τ is
   annealed (large → small) so that early iterations see a smooth
   landscape and later iterations converge to the true max.

3. **Simulated‑annealing polishing** – a stochastic walk on the
   log‑softmax (log‑probability) space.  Unlike the previous code it
   accepts uphill moves with a temperature schedule, allowing it to
   escape local minima.

4. **Final evaluation** – the best vector is stored in the global
   ``height_sequence_1`` variable and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√i (with a tiny shift to avoid division by zero)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_smooth(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 200_000,
                 lr0: float = 0.6,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-12,
                 tau0: float = 5e-3,
                 tau_end: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a *smooth* version of the max‑convolution.
    The smoothness is controlled by a temperature τ that is linearly
    annealed from ``tau0`` to ``tau_end``.
    The optimisation is performed in log‑softmax space, i.e. we keep a
    vector of logits ``log_a`` and obtain the candidate distribution via a
    softmax.  This guarantees non‑negativity and ℓ₁‑normalisation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + eps)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current temperature (linear anneal)
        tau = tau0 + (tau_end - tau0) * (it / max_iters)

        conv = _conv_fft(a, a)                     # length 2n‑1
        # soft‑max weighting of the convolution entries
        w = np.exp(conv / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient wrt a  (g = 2 * Σ_k w_k * a_{k-i})
        rev_w = w[::-1]                            # reversed weight vector
        grad_full = _conv_fft(a, rev_w)            # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # indices that give Σ_j w_{i+j} a_j
        g = 2.0 * grad_full[idx]                   # shape (n,)

        # gradient wrt logits (soft‑max Jacobian)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam moments (in log‑space)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size (decreasing like 1/√t)
        lr = lr0 / math.sqrt(it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50_000 == 0:
                print(f"[Adam‑smooth] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         sigma_start: float = 0.25,
                         sigma_end: float = 1e-5,
                         temp_start: float = 1e-3,
                         temp_end: float = 1e-6,
                         batch: int = 1) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style stochastic walk on the log‑softmax (log‑probability)
    space.  The proposal is a Gaussian perturbation of the logits; uphill moves
    are accepted with probability exp(-Δ/Temp).  The temperature and the
    perturbation width both decay linearly with the elapsed time.
    """
    rng = np.random.default_rng(20230314)

    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    start = time.time()
    total = deadline - start

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total

        sigma = sigma_start * (1.0 - progress) + sigma_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # propose a new point (log‑softmax perturbation)
        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)

        proposal_val = _objective_surrogate(proposal_seq)

        delta = proposal_val - cur_val
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # metropolis criterion
            if rng.random() < math.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 5_000 == 0:
            print(f"[SA] iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val
