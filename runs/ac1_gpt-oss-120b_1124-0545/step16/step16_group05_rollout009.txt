"""
search for a low‑objective non‑negative step‑function

Algorithm
---------

1. **Problem size** – use a moderate length (default n=4000) that gives a
   good trade‑off between the theoretical bound 2·n/(2n‑1)≈1 and the cost
   of FFT‑based operations.

2. **Rich seeding** – generate a diverse pool of ℓ₁‑normalised candidates:
   * classical windows (hann, hamming, blackman, kaiser)
   * exponential families (forward, reverse and a symmetric “U‑shape”)
   * power‑law families
   * beta‑shape families
   * random Dirichlet draws with a wide range of concentration parameters
   * the previous global best (if any).

   Only the `top_k` best seeds (according to the exact surrogate
   `2·n·max(conv)`) are kept.

3. **Smooth L‑BFGS** – each elite seed is refined with L‑BFGS‑B on the
   log‑sum‑exp surrogate of the max‑convolution.
   The β‑schedule is aggressive (up to 1e7) and we allow many inner
   iterations so that the surrogate is driven very close to the true max.

4. **Momentum exponentiated‑gradient polishing** – a mirror‑descent on the
   exact objective.  For the current iterate we
   * compute the full auto‑convolution,
   * collect *all* indices whose value lies within a relative tolerance
     of the maximum,
   * form the sub‑gradient as the average of the gradients of these
     active entries,
   * apply a momentum filter (β₁≈0.9) to the sub‑gradient and update the
     log‑variables.
   The loop runs until the time budget expires (the ``max_iters`` argument
   is set to a huge number; the routine stops automatically when the
   deadline is reached).

5. **Optional LP mass‑inflation** – for moderate n (≤4000) a few
   linear‑programming steps can still squeeze a small improvement.
   Each step solves a LP that maximises the ℓ₁‑mass of a vector `g`
   under the constraint `(f*g)_k ≤ M` where `M` is the current maximal
   convolution value.

6. **Stochastic restarts** – a few random perturbations of the best
   solution are fed back into a short L‑BFGS+EG pass to escape shallow
   minima.

7. **Final evaluation** – the best sequence found is stored in the global
   variable ``height_sequence_1`` and returned as a plain list.

The whole pipeline respects the supplied wall‑clock budget (default
1000 s) and prints progress information.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) assuming sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape_exp(n: int, c: float) -> np.ndarray:
    """U‑shaped sum of forward and reverse exponentials."""
    a = _exp_family(n, c) + _rev_exp_family(n, c)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _edges_uniform(n: int, frac: float) -> np.ndarray:
    """Two uniform blocks on the edges, middle zeroed."""
    m = max(1, int(frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:m] = 1.0 / m
    a[-m:] = 1.0 / m
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 2e4, 5e4,
            1e5, 2e5, 5e5,
            1e6, 5e6, 1e7
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max (weights = softmax over conv)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if time is over
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _sp_opt.minimize(fun=fun,
                                   x0=v,
                                   method='L-BFGS-B',
                                   jac=True,
                                   callback=callback,
                                   options={'maxiter': max_iter_per_beta,
                                            'disp': False,
                                            'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _exp_gradient_momentum(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 10_000_000,
                           momentum: float = 0.9,
                           rel_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution with momentum."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    vel = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_c = conv.max()
        tol = max_c * rel_tol + 1e-15
        active = np.where(conv >= max_c - tol)[0]

        # sub‑gradient averaged over all active entries
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            rev_idx = k - i_range
            g[i_range] += 2.0 * a[rev_idx]
        if len(active) > 1:
            g = g / len(active)

        # AdaGrad‑style decay (square‑root) with momentum
        eta = 0.6 / math.sqrt(it)
        vel = momentum * vel + (1.0 - momentum) * g
        log_a = log_a - eta * vel
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500_000 == 0:
                print(f"[ExpGrad‑M] it={it:9d} best={best_val:.6f}")

    return best_seq, best_val

_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """LP: maximise Σ g_j s.t. g ≥ 0, (f*g)_k ≤ M."""
    n = len(f)
    data = f[offset]                         # picks the appropriate coefficient
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 4) -> np.ndarray:
    """Perform a few LP “mass‑inflation’’ refinements."""
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1
        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break   # cannot increase ℓ₁‑mass

        g_scaled = g / sum_g                     # renormalise to sum = 1
        t = 0.5
        new_seq = (1 - t) * seq + t * g_scaled
        new_seq = new_seq / new_seq.sum()

        new_M = _conv_fft(new_seq, new_seq).max()
        if new_M < M:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            # try a smaller blend
            t = 0.25
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M:
                seq = new_seq
                print(f"[LP] improved (small t) max conv to {new_M:.6e}")
            else:
                break
    return seq

def _add_to_heap(heap, maxsize, entry):
    """Maintain a max‑heap of size ≤ maxsize on surrogate (lower is better)."""
    surrogate = entry[1]
    if len(heap) < maxsize:
        heapq.heappush(heap, (-surrogate, entry))
    else:
        if surrogate < -heap[0][0]:
            heapq.heapreplace(heap, (-surrogate, entry))

def propose_candidate(seed: int = 123,

    """

    ``evaluate_sequence``.  The routine respects the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good compromise for speed + LP feasibility
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load a previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (diverse families)
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    top_k = 12                     # number of seeds to keep for refinement
    heap = []                      # max‑heap storing (‑surrogate, (vec, meta))

    def consider(vec, meta):
        """Insert candidate into the heap if it improves the top‑k."""
        if time.time() > deadline:
            return
        surrogate = _objective_surrogate(vec)
        _add_to_heap(heap, top_k, (vec, surrogate, meta))

    # uniform baseline
    consider(_uniform(n), ('uniform',))

    # previous best
    if prev_arr is not None:
        consider(prev_arr, ('previous',))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        consider(win_func(n), (win_name,))

    # exponential families (forward, reverse, U‑shaped)
    cs = np.linspace(0.2, 10.0, 30)   # richer sweep
    for c in cs:
        consider(_exp_family(n, c), ('exp', c))
        consider(_rev_exp_family(n, c), ('revexp', c))
        consider(_u_shape_exp(n, c), ('ushape', c))

    # power‑law families
    for p in np.linspace(-0.5, 8.0, 35):
        consider(_power(n, p), ('power', p))

    # beta families
    alphas = np.linspace(0.2, 10.0, 15)
    betas = np.linspace(0.2, 10.0, 15)
    for a in alphas:
        for b in betas:
            consider(_beta_shape(n, a, b), ('beta', a, b))
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # edge‑uniform blocks
    for frac in [0.05, 0.10, 0.20, 0.30]:
        consider(_edges_uniform(n, frac), ('edges', frac))

    # Random Dirichlet draws (wide concentration range)
    concs = np.geomspace(0.01, 10.0, 12)
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            consider(_random_dirichlet(n, conc, rng), ('dirichlet', conc))

    # ------------------------------------------------------------------
    # 2. Extract the elite seeds
    # ------------------------------------------------------------------
    seed_vecs = [entry[0][0] for entry in heap]   # extract vectors
    print(f"Collected seeds, keeping top {len(seed_vecs)} for refinement.")

    # ------------------------------------------------------------------
    global_best_seq = seed_vecs[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    for idx, seed_vec in enumerate(seed_vecs):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None   # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Momentum exponentiated‑gradient polishing (majority of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the remaining bulk of time to EG polishing
        max_exp_iters = int(min(20_000_000, remaining * 1e5))  # generous upper bound
        print(f"\n--- Momentum EG polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_momentum(
            global_best_seq,
            n,
            deadline,
            max_iters=max_exp_iters,
            momentum=0.9,
            rel_tol=1e-12
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad‑M] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional linear‑programming mass‑inflation (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline and n <= 4000:
        print("\n--- Linear‑programming mass‑inflation refinement ---")
        seq_lp = _lp_inflation_step(global_best_seq, deadline, max_outer_iters=4)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Stochastic restarts (few quick perturbations)
    # ------------------------------------------------------------------
    restarts = 3
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{3 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_vec = _softmax(log_best + pert)

        # short L‑BFGS to re‑anchor
        seq_restart = _lbfgs_smooth_optimize(
            start_vec,
            n,
            deadline,
            max_iter_per_beta=3000,
            beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0, 500.0]
        )
        val_restart = _objective_surrogate(seq_restart)

        # brief EG polishing
        seq_restart, val_restart = _exp_gradient_momentum(
            seq_restart,
            n,
            deadline,
            max_iters=800_000,
            momentum=0.85,
            rel_tol=1e-12
        )
        if val_restart < global_best_val:
            global_best_seq = seq_restart.copy()
            global_best_val = val_restart
            print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final high‑β L‑BFGS pass (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final high‑β L‑BFGS fine‑tuning ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            max_iter_per_beta=4000,
            beta_schedule=[2e5, 5e5, 1e6, 5e6, 1e7, 5e7, 1e8]
        )
        val_final = _objective_surrogate(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final L‑BFGS] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
