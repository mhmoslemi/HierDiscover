"""
Propose a step‑function that minimises
    2·n·max(conv(sequence,sequence)) / (sum(sequence)²)

The method is a focused two‑stage optimisation:

1. **Diverse seed pool** – analytic families, random Dirichlet draws,
   sparse uniform blocks and the previous best solution.  All candidates
   are ℓ₁‑normalised (sum = 1) and the best surrogate (2·n·max(conv))
   is selected.

2. **Adam optimisation in log‑softmax space** – the selected seed is
   refined with a classic Adam optimiser that works directly on the
   exact surrogate (no smoothing).  Gradients of the active convolution
   entry are computed analytically; the optimiser respects the simplex
   constraint through the soft‑max parametrisation.  The optimisation
   runs until the time budget expires, keeping the best encountered
   vector.

3. **Local stochastic polishing** – a short random‑logit walk (simulated
   annealing style) finishes the search.

The whole routine respects the global 1000 s deadline, prints intermediate
progress, stores the best sequence in ``height_sequence_1`` and returns it
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Seed generators (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_opt(start_vec: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the exact surrogate (ℓ₁‑normalised)."""
    # Initialise logits (the optimisation variables)
    v = np.log(start_vec + 1e-30)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    best_vec = None
    best_val = np.inf

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        # ----- simplex point -----
        a = _softmax(v)                     # a ≥ 0, sum = 1

        # ----- surrogate value -----
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]            # max(conv)
        cur_obj = 2.0 * n * max_val         # surrogate = 2·n·max(conv)

        if cur_obj < best_val:
            best_val = cur_obj
            best_vec = a.copy()

        # ----- gradient of max(conv) w.r.t. a -----
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]        # factor 2 covers the two ordered pairs

        # ----- chain rule through soft‑max -----
        dot = np.dot(g, a)
        grad_v = a * (g - dot)             # ∂/∂logits

        # ----- Adam update -----
        # elementary step‑size schedule – slowly decreasing with √t
        lr = 0.5 / math.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_corr = m / (1.0 - beta1 ** t)
        v_corr = v_hat / (1.0 - beta2 ** t)

        v = v - lr * m_corr / (np.sqrt(v_corr) + eps)

    return best_vec, best_val


# ----------------------------------------------------------------------
# Simple stochastic polishing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _stochastic_polish(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Tiny random‑logit walk that keeps every improvement."""
    rng = np.random.default_rng()
    logits = np.log(start_vec + 1e-30)
    cur_seq = start_vec.copy()
    cur_val = _surrogate(cur_seq, n)

    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        progress = it / max_iters
        T = 0.05 * (1.0 - progress) + 1e-6
        sigma = 0.12 * (1.0 - progress) + 1e-4

        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _surrogate(prop_seq, n)

        if prop_val < cur_val or math.exp(-(prop_val - cur_val) / T) > rng.random():
            logits = prop_logits
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialise random generator and timing
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Problem size – 4000 gives a good trade‑off (as used before)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds (all ℓ₁‑normalised)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse uniform blocks – they often give interesting shapes
    for blk in (80, 120, 150):
        for _ in range(2):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # previous best solution (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size == n and np.sum(prev_arr) > 0:
                seeds.append(prev_arr / np.sum(prev_arr))
        except Exception:
            pass

    # normalise robustly (some generators may produce zeros)
    cleaned = []
    for vec in seeds:
        v = np.asarray(vec, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    seeds = cleaned

    # --------------------------------------------------------------
    # 3. Pick the most promising seed (lowest surrogate)
    # --------------------------------------------------------------
    surrogate_vals = [_surrogate(s, n) for s in seeds]
    best_idx = int(np.argmin(surrogate_vals))
    best_seed = seeds[best_idx]
    print(f"[Seed] Best initial surrogate = {surrogate_vals[best_idx]:.6f}")

    # --------------------------------------------------------------
    # 4. Adam optimisation (main workhorse)
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    # allocate the bulk of the time to Adam; keep a small fraction for polishing
    adam_iters = int(min(1_000_000, remaining_time * 400))  # rough scale
    adam_iters = max(200_000, adam_iters)                 # at least a decent run
    print(f"[Adam] Running up to {adam_iters:,} iterations")

    vec_adam, val_adam = _adam_opt(best_seed, n, deadline, max_iters=adam_iters)
    print(f"[Adam] Best surrogate after Adam = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 5. Stochastic polishing (optional, if time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polishing_iters = int(min(300_000, remaining * 300))
        if polishing_iters > 0:
            print(f"[Polish] Simulated‑annealing walk ({polishing_iters:,} iters)")
            vec_polish, val_polish = _stochastic_polish(vec_adam, n, deadline,
                                                        max_iters=polishing_iters)
            if val_polish < val_adam:
                vec_adam, val_adam = vec_polish, val_polish
                print(f"[Polish] Improved surrogate = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    final_seq = vec_adam.tolist()
    true_val = None
    try:
        true_val = evaluate_sequence(final_seq)
        print(f"\n=== Finished ===\nSurrogate ≈ {val_adam:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the result for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
