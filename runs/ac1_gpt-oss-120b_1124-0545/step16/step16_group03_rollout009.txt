def _adam_opt(start_seq, n, deadline, max_iters=500_000, beta1=0.9, beta2=0.999, eps=1e-8, beta_schedule=[...], steps_per_beta=...):
   v = np.log(start_seq + 1e-30)
   m = np.zeros_like(v)
   v_m = np.zeros_like(v)
   t = 0
   for beta in beta_schedule:
       if time.time()>deadline: break
       for i in range(steps_per_beta):
           if time.time()>deadline: break
           t += 1
           a = _softmax(v)
           conv = _conv_fft(a, a)
           # compute smooth max with this beta
           m_conv = np.max(conv)
           shift = m_conv
           exp_beta = np.exp(beta * (conv - shift))
           sum_exp = exp_beta.sum()
           # gradient of smooth max wrt a:
           w = exp_beta / sum_exp
           rev_a = a[::-1]
           full = _conv_fft(rev_a, w)
           grad_a = 2.0 * full[n-1:n-1+n]
           # gradient wrt v
           dot = np.dot(grad_a, a)
           grad_v = a * (grad_a - dot)
           # Adam update
           m = beta1 * m + (1-beta1) * grad_v
           v_m = beta2 * v_m + (1-beta2) * (grad_v**2)
           m_hat = m / (1 - beta1**t)
           v_hat = v_m / (1 - beta2**t)
           step = 0.1 / (np.sqrt(v_hat) + eps)
           v -= step * m_hat
   best_seq = _softmax(v)
   best_val = _objective_surrogate(best_seq)
   return best_seq, best_val
```

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

within the given time budget.  The algorithm is deliberately simple
yet aggressive:

1.  Build a diverse pool of ℓ₁‑normalised seeds (uniform, triangular,
    Gaussian, power‑law, exponential, Dirichlet draws, sparse blocks,
    previous solution if any).

2.  Pick the few best seeds (by the exact surrogate
    ``2·n·max(conv)``) and polish each of them with a *smooth*
    L‑BFGS optimiser that minimises a log‑sum‑exp approximation of the
    max‑convolution.  The β‑schedule is continued far beyond the
    original implementation and each β gets many more iterations.

3.  Take the overall best vector and run a *very long* exponentiated‑
    gradient (mirror‑descent on the simplex) on the *exact* max‑convolution
    surrogate.  The step‑size is  η = η₀ / √t  (η₀ = 1.0) and we
    periodically inject a tiny random jitter into the logits to escape
    shallow local minima.

4.  (Optional) a short hill‑climbing sweep to capture any remaining
    tiny improvements.

5.  Finally call the official ``evaluate_sequence`` on the best vector
    and store it globally as ``height_sequence_1``.

All heavy inner loops check the deadline so the function always returns
before the 1000 s limit.  The code uses only ``numpy`` and ``scipy`` (for
the L‑BFGS optimiser) which are available in the execution environment.
"""

# ----------------------------------------------------------------------

import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Dirichlet draws -------------------------------------------------------
    conc_list = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                             # raw
            seeds.append(np.sort(raw)[::-1])             # descending
            seeds.append(_sparse_uniform_block(n, 80, rng))   # a sparse block for variety

    # Sparse uniform blocks ------------------------------------------------
    for blk in (80, 100, 120, 150):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly ---------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iter_per_beta: int = 4000,
) -> np.ndarray:
    """
    Refine ``start_vec`` (already ℓ₁‑normalised) by minimising a
    log‑sum‑exp approximation of the max‑convolution.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1_000.0,
            2_000.0,
            5_000.0,
            10_000.0,
            20_000.0,
            50_000.0,
            100_000.0,
            200_000.0,
            500_000.0,
        ]

    # Log‑its initialise from the current vector
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)               # stays on the simplex
            conv = _conv_fft(a, a)               # full convolution
            shift = np.max(conv)                 # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (see original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # chain‑rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={
                    "maxiter": max_iter_per_beta,
                    "disp": False,
                    "ftol": 1e-12,
                },
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Long exponentiated‑gradient optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_long(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact max‑convolution.
    ``eta0`` controls the initial step size (η = eta0 / √t).
    A tiny random jitter is injected every 250 k iterations to escape
    shallow basins.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        # occasional progress report
        if it % 100_000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[ExpGrad] iter {it:,} – best surrogate {best_val:.6f}")

        # jitter to shake out of plateaus
        if it % 250_000 == 0:
            sigma = 0.05
            log_a += sigma * rng.normal(size=n)

    # final check
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Local pairwise hill‑climbing (tiny fine‑tuning)
# ----------------------------------------------------------------------
def _hill_climbing_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val

def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,

    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.

    """

    # 0. Initialisation & deadline
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # 1. Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Seed pool
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # Quick surrogate evaluation (exact for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 12
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS smoothing on the best few seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    beta_schedule = [
        1.0,
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1_000.0,
        2_000.0,
        5_000.0,
        10_000.0,
        20_000.0,
        50_000.0,
        100_000.0,
        200_000.0,
        500_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        remaining = deadline - time.time()
        # allocate a modest slice of the remaining budget to each seed
        max_iter_beta = int(min(8000, remaining * 0.03 * 1e5))
        max_iter_beta = max(3000, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx + 1} (max_iter/β = {max_iter_beta}) ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_schedule,
            max_iter_per_beta=max_iter_beta,
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Very long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_000_000, remaining * 0.45 * 1e5))
        max_exp_iters = max(500_000, max_exp_iters)
        print("\n--- Long exponentiated‑gradient polishing ---")
        seq_exp, val_exp = _exp_gradient_long(
            global_best_seq,
            n,
            deadline,
            max_iters=max_exp_iters,
            eta0=1.0,
        )
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp.copy(), val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Local hill‑climbing fine‑tuning (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(400_000, remaining * 0.15 * 1e5))
        max_hc_iters = max(100_000, max_hc_iters)
        print("\n--- Hill‑climbing fine‑tuning ---")
        seq_hc, val_hc = _hill_climbing_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=max_hc_iters,
        )
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc.copy(), val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
