def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The code builds a rich set of analytic seeds, selects the best one,
runs three optimisation phases and finally returns the best sequence
found before the wall‑clock deadline (≈ 1000 s).

Phases
------
0. Seed generation – uniform, power‑law, beta‑shape, Gaussian,
   random Dirichlet and the previous solution (if any).

1. Exponentiated‑gradient (EG) – mirror‑descent on the exact
   ‑max objective (sub‑gradient of the active convolution entry).

2. Smoothed‑max optimisation – a differentiable surrogate
   `softmax_α(conv)` is used.  The gradient of the surrogate
   (computed via a single cross‑correlation) is fed to an Adam‑style
   update, while the step‑size α of the soft‑max is increased slowly.
   This phase usually beats the plain EG because it uses information
   from *all* convolution entries, not only the active one.

3. Local random‑walk polishing – Gaussian perturbations of the logits
   (log‑softmax parametrisation) with a slowly decreasing variance.

All phases respect a global deadline; the current best vector is stored
in the global variable ``height_sequence_1`` so that successive runs can
warm‑start from it.
"""

import time
import math

# Low‑level numeric utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w ≥ 0 with sum(w) = 1 and ‖w‑v‖₂ minimal.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[EG]    it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam‑style mirror‑descent (still in log‑softmax space)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam moments (log‑space)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam]  it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000,
                    alpha_start: float = 0.2,
                    alpha_factor: float = 1.002,
                    lr0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Optimisation of a *smoothed* max‑convolution.
    Uses a soft‑max over the convolution entries with a temperature
    parameter α that is slowly increased; the gradient of the smooth
    surrogate is computed via a single cross‑correlation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Adam moments on the *real* (not log) parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    alpha = alpha_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- smoothed weights -----
        conv = _conv_fft(a, a)
        scaled = alpha * conv
        # numerical stability
        shift = np.max(scaled)
        w = np.exp(scaled - shift)
        ws = w.sum()
        w = w / ws if ws > 0 else np.full_like(w, 1.0 / w.size)

        # ----- gradient of the smooth surrogate -----
        # grad_j = 2 * Σ_i a_i * w_{i+j}
        # this is a cross‑correlation; we compute it via a single convolution
        w_rev = w[::-1]
        corr = np.convolve(a, w_rev)                     # length 3n‑2
        L = len(w)                                      # = 2n‑1
        grad = 2.0 * corr[L - 1 : L - 1 - n : -1]        # shape (n,)

        # ----- Adam update on the simplex (Euclidean) -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the probability simplex
        a = _project_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Smooth] it={it:7d} α={alpha:.4f} best={best_val:.6f}")

        # slowly sharpen the soft‑max
        alpha *= alpha_factor

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """Fast stochastic neighbourhood search (simulated‑annealing‑like)."""
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        # proposal in log‑softmax space
        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[Local]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val

    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.isfinite(arr).all() and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ---------------------------------------------------------
    # 1. Rich seed generation.
    # ---------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(30):
        a = 10 ** rng.uniform(-1, 1)      # between 0.1 and 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.08, 0.5)
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws – wide range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot > 0.0:
            normed_seeds.append(s / tot)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ---------------------------------------------------------
    # 2. Coarse ranking – keep the best raw seed.
    # ---------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec

    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ---------------------------------------------------------
    # 3. Exponentiated‑gradient optimisation (sub‑gradient on the true max).
    # ---------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    max_iters_eg = int(min(4_000_000, remaining * 3000))
    print(f"\n--- Exponentiated‑gradient phase (max_iters≈{max_iters_eg}) ---")
    seq_eg, val_eg = _exp_gradient_opt(best_seed, n, deadline, max_iters=max_iters_eg)
    print(f"[EG]    surrogate = {val_eg:.6f}")

    global_best_seq = seq_eg.copy()
    global_best_val = val_eg

    # ---------------------------------------------------------
    # 4. Smoothed‑max Adam optimisation – the new phase.
    # ---------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_smooth = int(min(500_000, remaining * 3000))
        print(f"\n--- Smoothed‑max optimisation phase (max_iters≈{max_iters_smooth}) ---")
        seq_smooth, val_smooth = _smooth_max_opt(
            global_best_seq, n, deadline, max_iters=max_iters_smooth,
            alpha_start=0.2, alpha_factor=1.002, lr0=0.4
        )
        if val_smooth < global_best_val:
            global_best_seq = seq_smooth.copy()
            global_best_val = val_smooth
            print(f"[Smooth] improved surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------
    # 5. Adam‑style fine‑tuning (short) – keep as a safety net.
    # ---------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_adam = int(min(300_000, remaining * 3000))
        print(f"\n--- Adam‑style fine‑tuning (max_iters≈{max_iters_adam}) ---")
        seq_adam, val_adam = _adam_opt(global_best_seq, n, deadline, max_iters=max_iters_adam)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam]  improved surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------
    # 6. Fast stochastic local search (simulated‑annealing‑like).
    # ---------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (budget≈{remaining:.1f}s) ---")
        seq_loc, val_loc = _local_random_walk(global_best_seq, n, deadline)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ---------------------------------------------------------
    # 7. Final evaluation with the official evaluator.
    # ---------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
