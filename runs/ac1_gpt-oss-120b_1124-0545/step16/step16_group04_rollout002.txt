def _simulated_annealing(start_a, n, deadline, max_iters=5000, rng=None):
   if rng is None: rng = np.random.default_rng()
   a_cur = start_a.copy()
   val_cur = _objective_exact(a_cur)
   best_a = a_cur.copy()
   best_val = val_cur
   for it in range(max_iters):
       if time.time() > deadline:
           break
       # Temperature schedule: exponential decay from T0 to Tf
       # T0=0.02, Tf=1e-6 maybe
       t_frac = it / max_iters
       T = 0.02 * (1e-6/0.02)**t_frac
       # propose neighbor
       a_cand = a_cur.copy()
       # select random fraction of indices to redistribute
       k = rng.integers(2, 10)
       idx = rng.choice(n, size=k, replace=False)
       total_mass = a_cand[idx].sum()
       # sample new distribution:
       new_weights = rng.dirichlet(np.full(k, 1.0))
       a_cand[idx] = total_mass * new_weights
       # optionally random small blur
       if rng.random() < 0.2:
           # blur via small kernel
           # we can compute convolution with kernel using np.convolve-fast
           kernel = np.array([0.2, 0.6, 0.2])
           a_cand = np.convolve(a_cand, kernel, mode='same')
           # ensure nonnegative
           a_cand = np.maximum(a_cand, 0.0)
           a_cand = a_cand / a_cand.sum()
       val_cand = _objective_exact(a_cand)
       delta = val_cand - val_cur
       if delta < 0 or rng.random() < np.exp(-delta / (T+1e-15)):
           a_cur = a_cand
           val_cur = val_cand
           if val_cand < best_val:
               best_val = val_cand
               best_a = a_cand.copy()
   return best_a, best_val

"""
Propose a step‑function that improves the current best bound (≈1.508534).
The search pipeline is:

1. Build a very diverse seed pool (analytic families + random draws).
2. Keep the best few seeds and run a continuation L‑BFGS on a

3. Refine the best vector with many iterations of exponentiated‑gradient
   (mirror descent) on the exact objective.
4. Perform a simple simulated‑annealing / mass‑redistribution local
   search to escape possible shallow minima.
5. Finish with a high‑β L‑BFGS polishing step.

The code respects the 1000 s budget, checks the deadline frequently,
and stores the final vector globally as ``height_sequence_1``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Modified Bessel function I0
    s = a.sum()
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i = 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(conc,…,conc) vector (already ℓ₁‑normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    Choose a random subset of size m = max(2, int(frac * n)),
    put equal mass 1/m on each chosen index, zero elsewhere.
    """
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Linear mixture a·(1‑α) + b·α, then renormalise."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    """Cyclic shift."""
    return np.roll(v, shift)

# Seed pool construction

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of diverse ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # Deterministic families (expanded parameter ranges)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    for lam in (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exponential_decay(n, lam))

    for g in np.linspace(0.3, 4.0, 12):
        seeds.append(_sine_window(n, gamma=g))

    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 14, 16):
        seeds.append(_kaiser_window(n, beta))

    for sigma in np.linspace(0.08, 0.55, 10):
        seeds.append(_gaussian(n, sigma))

    for p in np.linspace(0.0, 5.0, 26):
        seeds.append(_power_law(n, p))

    for a in np.linspace(0.5, 4.0, 8):
        for b in np.linspace(0.5, 4.0, 8):
            seeds.append(_beta_shape(n, a, b))

    # Random draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    for frac in (0.05, 0.1, 0.2, 0.35, 0.5):
        for _ in range(5):
            seeds.append(_random_sparse(n, frac, rng))

    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Pairwise mixtures
    base = np.arange(len(seeds))
    for _ in range(800):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of promising seeds
    for _ in range(40):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include previous solution if present
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean and normalise
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned

# Surrogate helpers for continuation L‑BFGS

def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (obj, grad) where grad is the derivative w.r.t. logits v.
    """
    a = _softmax(v)                     # probability vector on simplex
    conv = _conv_fft(a, a)               # full linear convolution
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a (soft‑max arguments)
    w = exp_term / sum_exp               # soft‑max weights over convolution indices
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]   # slice length n

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2000,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0,
            100000.0, 200000.0, 500000.0,
            1_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # final probability vector on the simplex


# ----------------------------------------------------------------------
# Exact objective & sub‑gradient (max‑convolution)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Returns a vector g with g_i = 2·a_{k‑i} for the lag k attaining the maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))          # lag attaining the maximum
    g = np.zeros_like(a)

    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g

# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns (best_a, best_obj).
    """
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val

# Simulated‑annealing local search (mass redistribution)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly redistributes mass among a small random set of indices.
    Accepts worsening moves with probability exp(-Δ/T).
    """
    if rng is None:
        rng = np.random.default_rng()
    a_cur = start_a.copy()
    val_cur = _objective_exact(a_cur)
    best_a = a_cur.copy()
    best_val = val_cur

    T_start = 0.02
    T_end = 1e-6

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # linear progress for temperature schedule
        frac = it / max_iters
        T = T_start * (T_end / T_start) ** frac

        # propose a redistribution among k random indices
        k = rng.integers(2, 9)        # 2…8 indices
        idx = rng.choice(n, size=k, replace=False)
        total = a_cur[idx].sum()
        if total <= 0.0:
            continue
        new_weights = rng.dirichlet(np.ones(k))
        a_cand = a_cur.copy()
        a_cand[idx] = total * new_weights

        # occasional gentle smoothing (small blur)
        if rng.random() < 0.15:
            kernel = np.array([0.2, 0.6, 0.2])
            a_cand = np.convolve(a_cand, kernel, mode='same')
            a_cand = np.maximum(a_cand, 0.0)
            a_cand = a_cand / a_cand.sum()

        val_cand = _objective_exact(a_cand)
        delta = val_cand - val_cur
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a_cur = a_cand
            val_cur = val_cand
            if val_cand < best_val:
                best_val = val_cand
                best_a = a_cand.copy()
    return best_a, best_val

        true_val = evaluate_sequence(global_best_seq.tolist())

            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible warm‑starts in the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
