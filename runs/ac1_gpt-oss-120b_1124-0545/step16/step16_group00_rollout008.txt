"""
Hybrid linear‑programming + exponentiated‑gradient optimiser.

1. Start from the best known sequence (global ``height_sequence_1``) or a simple
   analytic seed.
2. Repeatedly solve a linear programme  

        maximise   ∑ b_j
        subject to  a * b  ≤  M   (full linear convolution)
                    b ≥ 0

   where ``M`` is the current maximal autocorrelation of ``a``.
   The solution is scaled to sum = 1 and mixed with ``a`` using a small step
   size.  This guarantees a non‑increasing upper bound on
   ``2·n·max(conv(a,a))`` while keeping the dimensionality modest.
3. After a few LP steps a short exponentiated‑gradient (mirror‑descent) phase
   refines the shape, followed by a brief L‑BFGS smoothing of the
   ``log‑sum‑exp`` surrogate for the max‑convolution.
4. The best sequence found (non‑negative, normalised to sum = 1) is stored in
   ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
    import scipy.sparse as sp
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _opt = None
    sp = None
    _HAS_SCIPY = False


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Convolution matrix (used for the LP)
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray):
    """
    Return the (2n‑1) × n matrix A such that A @ b = conv(a, b) (full linear
    convolution).  The matrix is built as a sparse CSR matrix using the fact
    that each diagonal contains a constant coefficient from ``a``.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # offsets = 0, -1, -2, … , -(n‑1)
    offsets = -np.arange(n, dtype=int)
    # each diagonal consists of ``n`` repeated entries a[d]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(m, n), format="csr")
    return A


def _solve_lp_max_sum(a: np.ndarray, M: float, time_limit: float) -> np.ndarray:
    """
    Solve   maximise   1ᵀb   subject to   A @ b ≤ M   and   b ≥ 0,
    where A is the convolution matrix of ``a``.
    Returns the optimal vector ``b``.
    """
    if not _HAS_SCIPY:
        raise RuntimeError("SciPy is required for the LP phase")

    A = _conv_matrix(a)
    n = a.shape[0]
    m = A.shape[0]

    c = -np.ones(n, dtype=np.float64)                     # maximise sum(b)
    bounds = [(0.0, None)] * n
    b_ub = np.full(m, M, dtype=np.float64)               # all rows share the same bound

    # HiGHS accepts a time limit via options
    opts = {"presolve": True, "time_limit": time_limit}
    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options=opts,
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) polishing
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    The variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:          # pragma: no cover
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)                # full convolution
            max_c = np.max(conv)
            shift = max_c                         # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Analytic seed (triangular) – used when no previous solution is available
# ----------------------------------------------------------------------
def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a list of heights (sum = 1).  The best sequence found is stored in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000  # a compromise between quality and runtime

    # ------------------------------------------------------------------
    # 2. Initialise from previous solution if possible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        a = np.array(prev, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _triangular(n)
    else:
        a = _triangular(n)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. LP improvement phase (≈ 60 % of remaining time)
    # ------------------------------------------------------------------
    if _HAS_SCIPY:
        lp_fraction = 0.55
        lp_deadline = time.time() + (deadline - time.time()) * lp_fraction
        t_mix = 0.2                     # mixing coefficient (1‑t)·a + t·b
        iteration = 0
        while time.time() < lp_deadline:
            iteration += 1
            # current bound
            conv_a = _conv_fft(best_seq, best_seq)
            M = np.max(conv_a)

            # give the solver a modest slice of the remaining time
            time_for_lp = max(0.5, lp_deadline - time.time() - 0.5)
            try:
                b = _solve_lp_max_sum(best_seq, M, time_for_lp)
            except Exception as e:
                print(f"[LP #{iteration}] failed – {e}")
                break

            sum_b = b.sum()
            if sum_b <= 0:
                print("[LP] trivial solution, terminating LP phase")
                break
            b_scaled = b / sum_b

            # convex combination
            new_seq = (1.0 - t_mix) * best_seq + t_mix * b_scaled
            new_seq = np.maximum(new_seq, 0.0)
            new_seq /= new_seq.sum()

            new_val = _objective_surrogate(new_seq)
            if new_val < best_val - 1e-7:
                best_seq, best_val = new_seq, new_val
                print(f"[LP #{iteration}] improved surrogate → {best_val:.6f}")
                # keep mixing weight, maybe increase a bit
                t_mix = min(0.5, t_mix * 1.1)
            else:
                # no improvement – shrink step size
                t_mix *= 0.5
                if t_mix < 1e-4:
                    print("[LP] step size too small, exiting LP phase")
                    break
    else:
        print("[Info] SciPy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (≈ 30 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        eg_deadline = time.time() + remaining * 0.7
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline,
                                          max_iters=2_000_000, eta0=5.0)
        if val_eg < best_val:

    # ------------------------------------------------------------------
    # 5. L‑BFGS fine‑tuning (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS fine‑tuning ---")
        # a reduced schedule – fast but still powerful

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official scorer

    true_val = None

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
