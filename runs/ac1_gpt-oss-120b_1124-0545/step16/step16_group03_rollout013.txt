"""
Propose a non‑negative step‑function minimizing the objective
2·n·max(conv)/(sum²).

The search proceeds in several stages within a fixed time budget:

1.  **Seed pool** – a large diverse set of ℓ1‑normalised vectors:
    analytic families (uniform, triangular, Gaussian, power,
    exponential, double‑exponential), many Dirichlet draws with
    different concentrations and some smoothed variants, sparse uniform
    blocks and the previous best solution (if any).

2.  **Screening** – compute the cheap surrogate 2·n·max(conv) for all
    seeds (exact for ℓ1‑normalised vectors) and keep the best few
    (top‑k).

3.  **Continuation L‑BFGS** – for each top seed optimise a *smooth*
    approximation of the max‑convolution (log‑sum‑exp) with a schedule
    of increasing smoothing parameters β.  The optimiser works in the
    log‑softmax (simplex) space.

4.  **Exact mirror‑descent polishing** – a long exponentiated‑gradient
    (mirror‑descent) run on the current best candidate.  The step size
    decays as 1/√t and the run length is calibrated to the remaining
    time budget.

5.  **Simulated‑annealing refinement** – a random walk in logit space
    with a linear cooling schedule, accepting occasional uphill moves.

6.  **Projected sub‑gradient descent** – an aggressive sub‑gradient
    descent on the exact max‑convolution objective.

7.  **Local hill‑climbing** – cheap pairwise logit swaps for fine‑tuning.

8.  **Final evaluation** – call the official ``evaluate_sequence`` once
    and store the best sequence in the global variable
    ``height_sequence_1`` (so that later runs can reuse it).

All phases respect the global deadline (default 1000 s).  The code
uses only NumPy and SciPy (if available) and runs on the CPU.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Gaussian – a range of bandwidths
    for sigma_frac in np.linspace(0.05, 0.45, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Power law – a grid and some random exponents
    for p in np.linspace(0.0, 3.0, 7):
        seeds.append(_power(n, p))
    for _ in range(10):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Exponential families -------------------------------------------------
    for rate in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Dirichlet draws (different concentrations) --------------------------
    conc_list = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    draws_per_conc = 30
    for conc in conc_list:
        for _ in range(draws_per_conc):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])     # descending
            # smooth a raw draw with a small triangular kernel
            k = np.arange(-3, 4, dtype=np.float64)
            kernel = np.exp(-0.5 * (k / 1.5) ** 2)
            kernel /= kernel.sum()
            smooth = np.convolve(raw, kernel, mode='same')
            smooth = np.maximum(smooth, 0.0)
            if smooth.sum() > 0:
                seeds.append(smooth / smooth.sum())

    # Sparse uniform blocks ------------------------------------------------
    for blk in (80, 120, 160, 200):
        for _ in range(4):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly ---------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]

    # logits are the optimisation variables (softmax keeps us on the simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on simplex
            conv = _conv_fft(a, a)                # full convolution
            m = np.max(conv)
            shift = m                             # for numeric stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp                # soft weights, sum to 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # convolution of rev_a with w
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # gradient of each a_i

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.8 / math.sqrt(it)          # a slightly larger step‑size factor
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    momentum = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # simple momentum (Nesterov‑style) to accelerate
        momentum = 0.9 * momentum + g
        lr = 0.30 / math.sqrt(it)
        a -= lr * momentum
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Random walk in log‑softmax space with a linear cooling schedule."""
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing (pairwise logit swaps)
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Timing set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size – power‑of‑two for fast FFT
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2. Seed pool (previous best if available)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # ------------------------------------------------------------------
    # 3. Screen seeds with the cheap surrogate
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(80, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Initialise global best
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] current best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Continuation L‑BFGS on top seeds
    # ------------------------------------------------------------------
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                     500.0, 1_000.0, 2_000.0, 5_000.0,
                     10_000.0, 20_000.0, 50_000.0, 100_000.0]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing all L‑BFGS runs.")
            break

        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=2500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS {idx + 1}/{len(top_seeds)}] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq, best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

        # ------------------------------------------------------------------
        # 5a. Short exponentiated‑gradient polish after each L‑BFGS run
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(600_000, remaining * 5e5))
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_exp)
            if val_exp < best_val:
                best_seq, best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Long exponentiated‑gradient polishing (as much as time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 5e5))
        if max_iters < 20_000:
            break
        seq_exp, val_exp = _exp_gradient_opt(best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_iters)
        if val_exp < best_val:
            best_seq, best_val = seq_exp.copy(), val_exp
            print(f"[ExpGrad (loop)] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(2_000_000, remaining * 5e5))
        if max_sa > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_opt(best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=max_sa)
            if val_sa < best_val:
                best_seq, best_val = seq_sa.copy(), val_sa
                print(f"[SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_500_000, remaining * 5e5))
        if max_sub > 0:
            print("\n--- Long sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub)
            if val_sub < best_val:
                best_seq, best_val = seq_sub.copy(), val_sub
                print(f"[SubGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Local hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = int(min(800_000, remaining * 5e5))
        if max_hc > 0:
            print("\n--- Hill‑climbing fine‑tuning ---")
            seq_hc, val_hc = _hill_climbing_opt(best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_hc)
            if val_hc < best_val:
                best_seq, best_val = seq_hc.copy(), val_hc
                print(f"[HillClimb] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 10. Final L‑BFGS polish with an even larger β (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        try:
            extra_beta = beta_schedule + [200_000.0, 500_000.0]
            seq_final = _lbfgs_smooth_optimize(best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=extra_beta,
                                               max_iter_per_beta=3000)
            val_final = _objective_surrogate(seq_final)
            if val_final < best_val:
                best_seq, best_val = seq_final.copy(), val_final
                print(f"[Final L‑BFGS] improved to {best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print("[Final L‑BFGS] exception:", e)

    # ------------------------------------------------------------------
    # 11. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    seq_list = best_seq.tolist()
    true_val = None
    try:
        true_val = evaluate_sequence(seq_list)
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = seq_list
    return seq_list
