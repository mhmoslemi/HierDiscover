def propose_candidate(seed: int = 123, budget_s:int=1000, **kwargs):
    # Set random generator.
    rng = np.random.default_rng(seed)
    start=time.time()
    deadline=start+budget_s-2

    # Problem size n: choose 4096 for nice power-of-two.
    n=4096

    # Global best sequence variable.
    best_seq = None
    best_val = np.inf

    # Step 1. Seed generation.
    # Build list of candidate (vec, val)
    candidate_pool = []

    # baseline uniform
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # add windows
    windows = {
        'hann':_hann_window,
        'hamming':_hamming_window,
        'blackman':_blackman_window,
        'kaiser': lambda nn: _kaiser_window(nn, beta=14.0)
    }
    for name, func in windows.items():
        vec = func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (name,)))

    # exponential family grid
    for c in np.linspace(0.5,4.0,40):
        if time.time()>deadline: break
        vec = _exp_family(n,c)
        candidate_pool.append((vec, _objective_surrogate(vec), ('exp',c)))
        rev = _rev_exp_family(n,c)
        candidate_pool.append((rev, _objective_surrogate(rev), ('revexp',c)))

    # beta grid
    alphas = np.linspace(0.1,6.0,16)
    betas = np.linspace(0.1,6.0,16)
    for a in alphas:
        for b in betas:
            if time.time()>deadline: break
            vec = _beta_shape(n,a,b)
            candidate_pool.append((vec, _objective_surrogate(vec), ('beta',a,b)))

    # power family
    for p in np.linspace(-0.5,6.0,45):
        if time.time()>deadline: break
        vec = _power(n,p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('power',p)))

    # Dirichlet draws
    concs = [0.01,0.03,0.05,0.07,0.1,0.2,0.5,1.0,2.0,5.0]
    draws_per = 80
    for conc in concs:
        for _ in range(draws_per):
            if time.time()>deadline: break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # If previous global solution exists:
    if 'height_sequence_1' in globals():
        try:
            arr = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if arr.size == n and arr.sum()>0:
                v = arr/arr.sum()
                candidate_pool.append((v, _objective_surrogate(v), ('previous',)))
        except Exception:
            pass

    # Sort and keep top K
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, val, meta in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, top {top_k} kept.")

    # Step 2: Refine each seed with L-BFGS
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"Init best surrogate {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time()>deadline: break
        print(f"L-BFGS on seed {idx+1}/{len(top_seeds)}")
        opt_seq = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         max_iter_per_beta=12000)
        opt_val = _objective_surrogate(opt_seq)
        print(f"  after LBFGS surrogate {opt_val:.6f}")
        if opt_val < best_val:
            best_seq = opt_seq.copy()
            best_val = opt_val
            print("  new best!")

    # Step 3: Long EG polishing
    if time.time()<deadline:
        remaining = deadline - time.time()
        # allocate ~70% of remaining to EG
        max_iters = int(min(6_000_000, remaining * 0.07 * 1e5))
        print(f"EG polishing with up to {max_iters} iters")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        if val_eg < best_val:
            best_seq = seq_eg.copy()
            best_val = val_eg
            print(f" EG improved to {best_val:.6f}")

    # Step 4: Alternate LP inflation + EG cycles, up to time left

"""
Optimise a non‑negative step‑function to minimise

    2·n·max(conv(a,a)) / (sum(a)²)

The search proceeds in several stages, each respecting the global time
budget (default 1000 s).  The main ideas are:

1. **Very broad seeding** – uniform, DSP windows, exponential / beta /
   power‑law families, and many Dirichlet draws with a range of
   concentration parameters.  All candidates are ℓ₁‑normalised, and we
   keep the best few by the surrogate objective 2·n·max(conv).

2. **Smooth L‑BFGS** – each elite seed is refined using a log‑sum‑exp
   smooth approximation of the max‑convolution.  A schedule of β‑values
   drives the surrogate from very smooth (β≈1) to almost exact
   (β≈10⁶).  The optimiser works on the log‑softmax coordinates, so
   feasibility (non‑negativity and unit sum) is automatic.

3. **Long exponentiated‑gradient (mirror‑descent) polishing** – an exact
   sub‑gradient of the max‑convolution is applied in the log‑domain.
   A decreasing step‑size ηₜ≈c/√t (AdaGrad‑style) guarantees convergence.
   We allocate the bulk of the remaining time to this stage.

4. **Iterated LP‑mass‑inflation** – given the current iterate `f`, we
   solve the linear programme

        maximise   Σ g_j
        s.t.       g ≥ 0,
                   (f * g)_k ≤ M   ∀k,   with M = max(f*f),

   obtain a heavier ℓ₁‑mass distribution `g`, renormalise and mix it
   with `f`.  The mixture strictly reduces the peak of the convolution.
   The step is repeated a few times, each time followed by a short EG
   polishing.

5. **Stochastic restarts** – occasional Gaussian perturbations of the
   best solution are fed to a short L‑BFGS run and a modest EG pass.
   Improvements are kept.

All stages are guarded by the global deadline; if the time expires the
best sequence found so far is returned.  The final sequence is stored in
the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact 2·n·max(conv) for a unit‑ℓ₁‑normalised sequence."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    from scipy.special import i0
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy optimiser required for L‑BFGS.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5,
            1e6, 2e6, 5e6, 1e7,
        ]

    # work in the log‑softmax domain (always feasible)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp of the convolution
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # smoothing weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑domain EG)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # AdaGrad‑style stepsize
        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return sparsity pattern (row, col, offset) of the Toeplitz convolution."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        s.t.  g ≥ 0,
              (f * g)_k ≤ M   ∀k .
    Returns the optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # picks the correct coefficient f_i
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximise Σ g  → minimise –Σ g
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is fast for these medium‑sized LPs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 5) -> np.ndarray:
    """
    Perform a few LP “mass‑inflation’’ steps, each followed by a convex
    combination with the current iterate.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_scaled = g / sum_g                     # renormalise to sum = 1
        # try a couple of mixing factors
        for t in (0.5, 0.25, 0.125):
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M - 1e-12:
                seq = new_seq
                print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e} (t={t})")
                break
        else:
            break
    return seq

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects the supplied wall‑clock budget,
    stores the best sequence in the global variable ``height_sequence_1`` and
    returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well (reasonable FFT size & LP memory)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load a previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (very diverse families)
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # Baselines
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Classic DSP windows
    windows = {
        'hann': _hann_window,
        'hamming': _hamming_window,
        'blackman': _blackman_window,
        'kaiser': lambda nn: _kaiser_window(nn, beta=14.0)
    }
    for name, fn in windows.items():
        vec = fn(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (name,)))

    # Exponential families (dense sweep)
    for c in np.linspace(0.5, 4.0, 40):
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        candidate_pool.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        candidate_pool.append((rev, _objective_surrogate(rev), ('revexp', c)))

    # Beta family grid
    alphas = np.linspace(0.1, 6.0, 16)
    betas = np.linspace(0.1, 6.0, 16)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            candidate_pool.append((vec, _objective_surrogate(vec), ('beta', a, b)))

    # Power‑law family
    for p in np.linspace(-0.5, 6.0, 45):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('power', p)))

    # Random Dirichlet draws – a very rich source of candidates
    concs = [0.01, 0.03, 0.05, 0.07, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 80
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (by surrogate)
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement of elite seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None    # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(8_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_iters}) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        if val_eg < best_val:
            best_seq = seq_eg.copy()
            best_val = val_eg
            print(f"[ExpGrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Alternating LP mass‑inflation & short EG cycles
    # ------------------------------------------------------------------
    while time.time() < deadline - 5:
        # a) LP inflation
        seq_lp = _lp_inflation_step(best_seq, deadline, max_outer_iters=5)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < best_val:
            best_seq = seq_lp.copy()
            best_val = val_lp
            print(f"[LP] improved surrogate to {best_val:.6f}")

        # b) brief EG polishing after LP
        remaining = deadline - time.time()
        if remaining < 20:
            break
        max_iters = int(min(500_000, remaining * 0.04 * 1e5))
        seq_eg2, val_eg2 = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        if val_eg2 < best_val:
            best_seq = seq_eg2.copy()
            best_val = val_eg2
            print(f"[Post‑LP EG] improved to {best_val:.6f}")
        else:
            # No improvement – stop the alternation
            break

    # ------------------------------------------------------------------
    # 6. Stochastic restarts – perturb‑optimise cycles
    # ------------------------------------------------------------------
    restarts = 4
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{5 - restarts} ---")
        log_best = np.log(best_seq + 1e-30)
        pert = rng.normal(scale=0.15, size=n)
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS to re‑centre
        seq_restart = _lbfgs_smooth_optimize(
            start_vec,
            n,
            deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0],
            max_iter_per_beta=3000
        )
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < best_val:
            # modest EG polishing
            seq_restart, val_restart = _exp_gradient_opt(
                seq_restart,
                n,
                deadline,
                max_iters=300_000
            )
            if val_restart < best_val:
                best_seq = seq_restart.copy()
                best_val = val_restart
                print(f"[Restart] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
