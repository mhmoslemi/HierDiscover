"""
Propose a step‑function (list of non‑negative heights) that minimizes
the evaluation function 2·n·max(conv(a,a))/(sum(a)²).

Algorithm (≈1000 s budget)

0️⃣  Load the previous solution (if any) – it is a good starting point.
1️⃣  **Rich seed pool** – analytic families (uniform, exponential,
    beta, power‑law, sin‑shape, arcsine), extensive Dirichlet draws
    (conc. ranging from 1e‑3 to 2) and a few random perturbations of
    the uniform baseline.
2️⃣  **Screening** – keep the best *top_k* seeds (by exact objective).
3️⃣  **L‑BFGS smoothing** – starting from each seed run L‑BFGS‑B on a
    log‑sum‑exp (smooth‑max) surrogate.  The β‑schedule is long
    (up to 1 e⁶) and we give each run a generous iteration budget.
4️⃣  **Active‑set mirror descent** – from the best L‑BFGS result,
    repeatedly move the soft‑max logits in the direction that reduces
    the average sub‑gradient of all currently active (nearly‑tight)
    convolution constraints.
5️⃣  **Explicit max‑gradient polishing** – a few million steps of
    exponentiated‑gradient (mirror‑descent) on the exact max‑conv
    objective, with a 1/√t step‑size schedule.
6️⃣  **Stochastic hill‑climbing** – Gaussian perturbations of the
    logits with a slowly‑cooling temperature; uphill moves are allowed
    with Metropolis probability.
7️⃣  **Pairwise mass‑transfer local search** – randomly move a tiny
    amount of probability mass between two coordinates; accept only
    improving moves.
8️⃣  **Final sub‑gradient polishing** – short projected sub‑gradient
    run to clean up any remaining slack.

All iterates stay on the probability simplex, therefore the surrogate
objective equals the true value of ``evaluate_sequence``.
The routine respects the 1000 s wall‑clock budget and prints progress

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for length ~4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a sequence that sums to 1 (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    """Half‑sine shape (positive, symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the classical arcsine distribution."""
    return _beta_shape(n, 0.5, 0.5)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS on smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    The parameters are a probability vector (soft‑max of logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 10.0, 100.0, 1000.0, 5000.0,
                         20000.0, 100000.0, 500000.0, 1_000_000.0]

    # initialise logits (log‑softmax)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # max for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # soft‑max over shifts
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice the needed part

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Active‑set mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _active_set_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 500_000,
                    tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (soft‑max logits) that reduces the *average* sub‑gradient
    of all currently active constraints (shifts with convolution value
    within ``tol`` of the maximum).  This often pushes the peak down a
    little further after L‑BFGS.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    v = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol)[0]

        # Build weights for the active shifts (average)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / max(1, len(active))

        # gradient w.r.t. a for the averaged active constraints
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        eta = 0.5 / math.sqrt(it)
        v = v - eta * grad_v
        a = _softmax(v)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution (sub‑gradient of the active
    shift).  Step‑size η_t = 0.6 / sqrt(t) follows standard theory.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 1.0,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb the logits with Gaussian noise, accept improving moves and
    occasional uphill moves with a slowly cooling temperature.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.9999
    temp_decay = 0.99995
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 200_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val

# ----------------------------------------------------------------------
def _pairwise_local_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 500_000,
                        init_delta: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Randomly shift a tiny amount of probability mass from one coordinate
    to another; only accept strictly improving moves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    delta = init_delta

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] < delta:
            continue

        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        val_prop = _objective_surrogate(a_prop)
        if val_prop < best_val:
            a = a_prop
            best_val = val_prop
            best_seq = a.copy()
            delta *= 1.02               # slight increase when we win
        else:
            delta *= 0.998               # decay when we lose

        if delta < 1e-9:
            delta = init_delta

    return best_seq, best_val

def _grid_search_exponential(n: int, deadline: float) -> list:
    """Sweep exponential family over c ∈ [0.5, 3.0] (25 points)."""
    vals = []
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float) -> list:
    """Coarse sweep for beta shapes (α,β ∈ [0.1,5] on a 9×9 grid)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float) -> list:
    """Power‑law sweep p ∈ [-0.5,5] (31 points)."""
    vals = []
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    objective within the allotted wall‑clock budget.
    Returns the best sequence (list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size – we keep the historically‑successful size of 4000.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0️⃣  Load previous best (if any)
    # --------------------------------------------------------------
    prev_arr = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1️⃣  Rich seed generation
    # --------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous solution if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # analytic families
    candidate_pool.append((_sin_shape(n), _objective_surrogate(_sin_shape(n)), ('sin',)))
    candidate_pool.append((_arcsine_shape(n), _objective_surrogate(_arcsine_shape(n)), ('arcsine',)))

    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # random Dirichlet draws with a very wide span of concentration parameters
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # a few noisy perturbations of uniform (helps escape symmetry)
    for _ in range(20):
        if time.time() > deadline:
            break
        noisy = _uniform(n) + 0.05 * rng.random(n)
        noisy = _proj_simplex(noisy)
        candidate_pool.append((noisy, _objective_surrogate(noisy), ('noisy_uniform',)))

    # --------------------------------------------------------------
    # 2️⃣  Keep the most promising seeds
    # --------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k} for optimisation.")

    # --------------------------------------------------------------
    # 3️⃣  L‑BFGS continuation (smooth max) on each top seed
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] objective = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during L‑BFGS phase.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                             max_iter_per_beta=4000)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception:", e)

    # --------------------------------------------------------------
    # 4️⃣  Active‑set mirror descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Active‑set mirror descent ---")
        seq_act, val_act = _active_set_opt(global_best_seq, n, deadline,
                                           max_iters=500_000, tol=1e-9)
        if val_act < global_best_val:
            global_best_seq = seq_act.copy()
            global_best_val = val_act
            print(f"[Active‑set] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Exponentiated‑gradient polishing (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(2_000_000, remaining * 0.08 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Stochastic hill‑climbing (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(4_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Stochastic hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq, n, deadline,
                                         rng, sigma0=1.0, max_iters=max_hc_iters)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7️⃣  Pairwise mass‑transfer local search
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_pairwise_iters = int(min(800_000, remaining * 0.05 * 1e5))
        print(f"\n--- Pairwise local search (max_iters={max_pairwise_iters}) ---")
        seq_pair, val_pair = _pairwise_local_opt(global_best_seq, n, deadline,
                                                 rng, max_iters=max_pairwise_iters)
        if val_pair < global_best_val:
            global_best_seq = seq_pair.copy()
            global_best_val = val_pair
            print(f"[Pairwise] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8️⃣  Final projected sub‑gradient polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(500_000, remaining * 0.04 * 1e5))
        if max_sub_iters > 0:
            print("\n--- Final sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, deadline,
                                                max_iters=max_sub_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 9️⃣  Evaluate with the official evaluator and store the result
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
