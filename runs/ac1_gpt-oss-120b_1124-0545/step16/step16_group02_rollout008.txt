"""
A fast two‑stage optimizer for the autocorrelation (max‑convolution) problem.

Stage 1 – optional L‑BFGS refinement on a smooth log‑sum‑exp surrogate
          (uses SciPy if available).  This quickly brings a random seed
          close to a good local minimum.

Stage 2 – simulated‑annealing (SA) in the log‑probability (softmax) space.
          The SA accepts occasional uphill moves, allowing it to escape the
          basin reached by L‑BFGS and to keep shaving off the remaining
          fraction of the gap to the target bound (≈ 1.5).  The algorithm
          respects the overall time budget and always returns a feasible
          non‑negative ℓ₁‑normalised sequence.

All helper functions are top‑level; no file I/O is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n·log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq,seq)).
    For a ℓ₁‑normalised seq this equals the official evaluate_sequence.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple analytic seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform mass on the whole interval."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular window, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)                     # full convolution
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (stabilised)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # softmax weights over conv entries
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if time is over
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing optimizer (log‑probability space)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber with occasional uphill moves.
    Operates on log‑softmax parameters to stay on the simplex.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # annealing schedule (linear cooling)
        T = max(1e-9, 0.12 * (1.0 - it / max_iters))
        sigma = max(1e-8, 0.10 * (1.0 - it / max_iters))

        proposal_log = logit + sigma * rng.normal(size=n)
        proposal = _softmax(proposal_log)
        prop_val = _objective_surrogate(proposal)

        delta = prop_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # accept
            logit = proposal_log
            cur_seq = proposal
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 100_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the global time budget
    and returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution (if any) and build a small seed pool
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    seeds = [
        _uniform(n),
        _triangular(n),
    ]

    # a few random Dirichlet draws (different concentration)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # sparse random vectors (few non‑zeros)
    for k_frac in (0.010, 0.020, 0.035):
        k = max(1, int(round(k_frac * n)))
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.exponential(scale=1.0, size=k)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = vals
        a /= a.sum()
        seeds.append(a)

    if prev_arr is not None:
        seeds.append(prev_arr)

    # ------------------------------------------------------------------
    # 3. Pick the best seed according to the fast surrogate
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for s in seeds:
        val = _objective_surrogate(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"Initial surrogate best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional L‑BFGS refinement (if SciPy is present)
    # ------------------------------------------------------------------
    if _opt is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            # give L‑BFGS a modest slice of the remaining budget
            max_iter_per_beta = int(min(8_000, remaining * 2.0))
            print(f"Running L‑BFGS (max_iter_per_beta={max_iter_per_beta})")
            seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                            beta_schedule=None,
                                            max_iter_per_beta=max_iter_per_beta)
            val_lb = _objective_surrogate(seq_lb)
            print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
            if val_lb < best_val:
                best_val = val_lb
                best_seq = seq_lb.copy()
    else:
        print("SciPy not available – skipping L‑BFGS stage.")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing polishing (the main driver for the final gap)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        # Estimate feasible number of SA iterations (empirically ≈2000–3000/s)
        est_iters_per_sec = 2000
        max_iters = int(remaining * est_iters_per_sec)
        max_iters = max(200_000, max_iters)  # guarantee enough exploration
        print(f"Running simulated annealing (≈{max_iters:,} iters)")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, max_iters)
        print(f"[SA] best surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # ------------------------------------------------------------------
    # 7. Store and return the result
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
