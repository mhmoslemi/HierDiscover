def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy optimiser required for L‑BFGS.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights in the smoothed max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # ada‑grad‑style decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# cache for the sparse convolution matrix pattern for each n
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        subject to g ≥ 0,
                   (f * g)_k ≤ M   ∀k .
    Returns the optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # use HiGHS (sparse‑aware) – fast for our moderate size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        # fallback – return the current vector (no improvement possible)
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray, deadline: float,
                       max_outer_iters: int = 5) -> np.ndarray:
    """
    Perform a few “mass‑inflation’’ LP steps.
    Each iteration solves the LP described above, rescales the solution
    to unit ℓ₁‑mass and mixes it with the current iterate.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            # LP cannot increase ℓ₁‑mass – we are at a stationary point.
            break

        g_scaled = g / sum_g                     # renormalise to sum = 1
        t = 0.5                                   # mixing factor
        new_seq = (1 - t) * seq + t * g_scaled
        new_seq = new_seq / new_seq.sum()         # safety normalisation

        new_M = _conv_fft(new_seq, new_seq).max()
        if new_M < M:
            seq = new_seq
            # print a tiny progress line
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            # if no improvement with t=0.5 try a smaller blend
            t = 0.25
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M:
                seq = new_seq
                print(f"[LP] improved (small t) max conv to {new_M:.6e}")
            else:
                break
    return seq
