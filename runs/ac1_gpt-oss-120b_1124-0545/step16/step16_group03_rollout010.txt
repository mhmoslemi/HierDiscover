"""
Optimise a non‑negative step‑function (a probability vector) for the
objective

    2·n·max(conv(a,a)) / (sum(a)^2)

which is exactly ``evaluate_sequence`` when ``a`` is ℓ¹‑normalised.
The search consists of the following stages:

1. **Rich seed generation** – uniform, double‑exponential, a small set of
   Dirichlet draws, and – crucially – a *cosine window* family
   a_i ∝ exp(C·cos(2πi/(n‑1))) (implemented via a soft‑max).  Random
   two‑frequency combinations are also added.

2. **Fast surrogate evaluation** – the ℓ¹‑normalised objective is
   ``2·n·max(conv)`` and can be computed with an FFT‐based convolution.

3. **Selection of the best few seeds** – keep the top‑k candidates.

4. **Long mirror‑descent (exponentiated‑gradient) polishing** – a cheap
   sub‑gradient of the exact max‑convolution is applied in log‑space
   (mirror descent on the simplex).  This phase runs for as long as the
   allotted time permits and repeatedly restarts from the current best.

5. **Optional projected sub‑gradient refinement** – if time remains, a
   longer Euclidean projected sub‑gradient descent is performed.

The final sequence is stored in ``height_sequence_1`` and returned as a
plain list of ``float``.  All numbers are non‑negative; the evaluator
clips values >1000, which never occurs in the produced vectors.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

try:
    import scipy.optimize as _opt  # required only for optional L‑BFGS (unused here)
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by ``evaluate_sequence`` (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    """U‑shaped (mass at the edges) vector."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Cosine‑window families (log‑it space)
# ----------------------------------------------------------------------
def _cosine_window(n: int, coeffs: np.ndarray) -> np.ndarray:
    """
    Build a probability vector from cosine basis coefficients.

    logits[i] = Σ_k coeffs[k] * cos(2π k i / (n‑1)).
    The probabilities are a softmax of the logits.
    """
    i = np.arange(n, dtype=np.float64)[:, None]          # (n,1)
    k = np.arange(1, coeffs.size + 1, dtype=np.float64)  # (1,m)
    # cos(2π k i / (n‑1))
    angles = 2.0 * np.pi * i * k / (n - 1)
    logits = (coeffs * np.cos(angles)).sum(axis=1)
    return _softmax(logits)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent polishing on the exact max‑convolution surrogate.
    Operates on the log‑probability (logits) directly.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)          # classic diminishing stepsize
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Projected Euclidean sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialisation & time budget
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Problem size – 4000 gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Seed pool (including cosine‑window families)
    # --------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_double_exponential(n, rate=0.008))

    # a few Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution, if any
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size == n and np.sum(prev) > 0:
            seeds.append(prev / np.sum(prev))

    # --------------------------------------------------------------
    # 3. Cosine‑window seeds
    # --------------------------------------------------------------
    # single‑frequency cosine window (parameter C ≥ 0)
    for C in np.linspace(0.0, 8.0, 41):          # 0 … 8 in steps of 0.2
        seq = _cosine_window(n, np.array([C]))
        seeds.append(seq)

    # two‑frequency random combinations
    for _ in range(30):
        C1 = rng.uniform(0.0, 8.0)
        C2 = rng.uniform(-4.0, 4.0)            # sign allows different shapes
        seq = _cosine_window(n, np.array([C1, C2]))
        seeds.append(seq)

    # --------------------------------------------------------------
    # 4. Evaluate all seeds with the exact surrogate (fast)
    # --------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    best_seq, best_val = evaluated[0]
    print(f"\n[Seed] best surrogate among raw seeds = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Long exponentiated‑gradient polishing (mirrored descent)
    # --------------------------------------------------------------
    # The exponentiated‑gradient routine stops automatically when the
    # deadline is reached.
    while time.time() < deadline:
        remaining = deadline - time.time()
        # stop if very little time left for a meaningful iteration
        if remaining < 3.0:
            break

        # run a polishing phase from the current best sequence
        seq, val = _exp_gradient_opt(best_seq, n, deadline,
                                     max_iters=2_000_000)
        if val < best_val:
            best_seq, best_val = seq, val
            print(f"[ExpGrad] improved surrogate = {best_val:.6f}")

        # if the run exhausted the deadline, break out
        if time.time() >= deadline:
            break

        # occasional restart from a fresh random seed to escape possible plateaus
        # (but only if there is still time)
        if remaining > 30.0:
            # pick a random seed from the top‑few evaluated ones
            restart_vec = evaluated[rng.integers(0, top_k)][0]
            seq, val = _exp_gradient_opt(restart_vec, n, deadline,
                                         max_iters=500_000)
            if val < best_val:
                best_seq, best_val = seq, val
                print(f"[Restart‑ExpGrad] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Optional projected sub‑gradient refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("\n--- Projected sub‑gradient polishing ---")
            seq, val = _subgradient_opt(best_seq, n, deadline,
                                        max_iters=2_000_000)
            if val < best_val:
                best_seq, best_val = seq, val
                print(f"[SubGrad] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
