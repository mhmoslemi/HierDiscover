proj_simplex:

def proj_simplex(v):
   # from Duchi: sort descending, find rho where v_i - (sum_{j<=i} v_j - 1)/i > 0, compute theta, etc.
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(len(v)) + 1
   cond = u - cssv / ind > 0
   rho = np.where(cond)[0][-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

which is scale‑invariant and equal to the surrogate 2·n·max(conv) when
the vector is ℓ₁‑normalised.  The optimizer is a projected (simplex) gradient
descent on a smooth approximation of the ``max`` (log‑sum‑exp) of the
autocorrelation.  The algorithm:

1. Generate a diverse set of ℓ₁‑normalised seeds (uniform, Dirichlet,
   beta‑shapes, power‑law, Gaussian, previous best).
2. Keep the seed with the smallest surrogate value.
3. Run many iterations of projected gradient descent:
   – at each step compute the convolution a∗a (full linear via FFT);
   – replace the hard ``max`` by a smoothed version with temperature α;
   – the gradient of the smoothed loss is a correlation between the
     smoothing weights and a; it is obtained with a second FFT.
   – take a step in the Euclidean direction and project back onto the
     probability simplex.
   – α is slowly increased (annealed) so that the smooth loss approaches
     the true max.
4. A short stochastic local‑walk (Gaussian perturbations in log‑space)
   is performed to polish the solution.
5. The best sequence found is returned and stored in the global
   ``height_sequence_1`` variable.

The code stays well within the 1000 s budget and makes heavy use of
vectorised NumPy/FFT operations for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Duchi et al., 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    # If all are false (should not happen because sum(v) may be >1),
    # the last true index defaults to -1 → theta = 0.
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Gradient descent on smoothed max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  ( (1/α)·log Σ exp(α·conv) ) .
    This is 2·correlation(w, a) where w is the softmax over conv entries.
    """
    n = a.shape[0]
    # Compute softmax weights (log‑sum‑exp trick)
    max_c = np.max(conv)
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()               # shape (2n-1,)

    # Correlation = convolution with reversed a, then slice.
    grad_full = _conv_fft(w, a[::-1])          # length 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]    # shape (n,)
    return grad


def _projected_sgd(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   alpha_start: float = 5.0,
                   alpha_max: float = 50.0,
                   eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the smoothed max‑convolution.
    Returns the best sequence (ℓ₁‑normalised) and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    alpha = alpha_start
    # Speed‑estimate: we will adapt the inner loop to the remaining time.
    while time.time() < deadline:
        it += 1
        # current convolution and surrogate
        conv = _conv_fft(a, a)
        cur_val = float(2.0 * n * np.max(conv))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # gradient of the smooth approximation
        grad = _smooth_gradient(a, conv, alpha)

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _proj_simplex(a - eta * grad)

        # anneal α (slow increase)
        if it % 500 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.02)

        # occasional progress report
        if it % 5000 == 0:
            print(f"[SGD] it={it:6d}  surrogate={cur_val:.6f}  α={alpha:.2f}")

        # break if not enough time for another iteration (very rough guard)
        if time.time() + 0.015 > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small stochastic polishing (log‑softmax random walk)
# ----------------------------------------------------------------------
def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.12,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search in log‑softmax space.
    Gaussian noise on logits with a linearly decaying variance.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[Local] iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (keep the same n as the reference solution)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (few dozen candidates)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]
    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(30):
        a = 10 ** rng.uniform(-1, 1)   # 0.1 … 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.08, 0.5)
        seeds.append(_gaussian(n, sigma))

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot > 0.0:
            normed_seeds.append(s / tot)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Pick the best seed according to the exact surrogate.
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec
    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected SGD on smooth max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected SGD (smooth max) phase ---")
        seq_sgd, val_sgd = _projected_sgd(best_seed, n, deadline)
        print(f"[SGD] final surrogate = {val_sgd:.6f}")

        global_best_seq = seq_sgd.copy()
        global_best_val = val_sgd
    else:
        global_best_seq = best_seed.copy()
        global_best_val = best_seed_val

    # ------------------------------------------------------------------
    # 4. Fast stochastic local polishing (log‑softmax walk).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local random‑walk polishing ---")
        seq_loc, val_loc = _local_random_walk(global_best_seq, n, deadline)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
