def _hill_climb_opt(start_seq, n, deadline):
    a = start_seq.copy()
    conv = np.convolve(a, a)  # full conv length 2n-1
    cur_max = np.max(conv)
    cur_obj = 2.0 * n * cur_max
    best_a = a.copy()

    start = time.time()
    it = 0
    while time.time() < deadline:
        max_idx = int(np.argmax(conv))
        # Build gradient vector
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        # Choose i = idx_i with max gradient among those with a>0

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm works in three phases:

1. **Initialisation** – use the previous best sequence if it exists,
   otherwise fall back to a simple uniform distribution.
2. **Fine‑grained L‑BFGS** – run a few L‑BFGS‑B iterations on the smooth
   log‑sum‑exp surrogate of the max‑convolution (uses SciPy if available).
   This quickly reaches a good local optimum.
3. **Directed hill‑climbing** – repeatedly move a small amount of mass
   from the index that contributes most to the current convolution peak
   to an index that contributes least.  The convolution is updated in‑place
   in O(n) time using the exact formula for a rank‑2 change, so millions
   of cheap steps fit into the 1000 s budget.  The best vector seen is
   returned.

All numbers stay non‑negative, are kept ℓ₁‑normalised and the final
sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy optimiser ------------------------------------------------
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv) for ℓ₁‑normalised sequences (sum(seq)=1).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv))


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Refine a simplex‑constrained vector using L‑BFGS‑B on the
    log‑sum‑exp smooth approximation of the max‑convolution.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation backend not available")
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # logits → softmax stays on simplex

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = np.exp(v_local - np.max(v_local))
            a /= a.sum()                   # soft‑max
            conv = np.convolve(a, a)
            mx = np.max(conv)
            shift = mx
            expb = np.exp(beta * (conv - shift))
            sumexp = expb.sum()
            obj = 2.0 * n * (shift + np.log(sumexp) / beta)

            w = expb / sumexp              # soft‑weights
            rev_a = a[::-1]
            full = np.convolve(rev_a, w)   # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception", e)
            break

    a = np.exp(v - np.max(v))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Hill‑climbing optimiser ------------------------------------------------
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climbing on the exact max‑convolution.
    In each step a small amount of mass is moved from the index
    that most contributes to the current peak to an index that
    contributes the least.  Convolution updates are performed in‑place
    using the O(n) formula for a rank‑2 change.
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    conv = np.convolve(a, a)                     # full linear conv, length 2n‑1
    cur_obj = _objective_surrogate(a)
    best_a = a.copy()
    best_obj = cur_obj

    start_time = time.time()
    it = 0
    # hyper‑parameters for the step‑size schedule
    max_step_factor = 0.2    # up to 20 % of the source mass
    min_step_factor = 0.001  # never smaller than 0.1 %

    while time.time() < deadline:
        # active index of the convolution peak
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active peak (zero outside the support)
        g = np.zeros_like(a)
        i0 = max(0, max_idx - (n - 1))
        i1 = min(n - 1, max_idx)
        idx = np.arange(i0, i1 + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        # pick source index i (large gradient, positive mass)
        mask_src = a[idx] > 1e-15
        if np.any(mask_src):
            src_candidates = idx[mask_src]
            src_grads = g[src_candidates]
            i = src_candidates[np.argmax(src_grads)]
        else:
            i = rng.integers(0, n)

        # pick destination index j (small gradient)
        low_k = min(200, n)                     # consider the 200 lowest‑gradient slots
        low_idxs = np.argsort(g)[:low_k]
        # ensure j ≠ i
        if len(low_idxs) > 1:
            # random choice among the low‑gradient candidates
            j = rng.choice(low_idxs[low_idxs != i]) if i in low_idxs else rng.choice(low_idxs)
        else:
            j = (i + 1) % n

        # step‑size proportional to remaining time (linear cooling)
        remaining = deadline - time.time()
        total = deadline - start_time
        if total <= 0:
            break
        frac = max(0.0, remaining / total)
        step_factor = min_step_factor + frac * (max_step_factor - min_step_factor)

        delta = step_factor * a[i]               # amount moved from i → j
        if delta <= 0.0:
            # nothing to move – try another iteration
            continue

        # --- apply the exact convolution update --------------------------------
        #   conv' = conv - 2·δ·a (shift i) + 2·δ·a (shift j) + δ²·e_{2i}+δ²·e_{2j}-2δ²·e_{i+j}
        #   where e_k is a unit impulse at position k.
        # The slice a[i:i+n] is a shifted copy of a (zero‑padded outside the support).

        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        new_obj = _objective_surrogate(a)        # surrogate uses only a – not yet updated
        # compute the new max‑convolution value after the update
        new_max = np.max(conv)
        new_obj = 2.0 * n * new_max

        if new_obj < cur_obj - 1e-14:            # strict improvement
            # accept the move
            a[i] -= delta
            a[j] += delta
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # revert the convolution update (a is unchanged)
            conv[i:i + n] += 2.0 * delta * a
            conv[j:j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta

        it += 1

        # periodic clean‑up to avoid numerical drift
        if it % 50000 == 0:
            conv = np.convolve(a, a)
            cur_obj = _objective_surrogate(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            # renormalise just in case
            s = a.sum()
            if s != 1.0:
                a /= s
            print(f"[HillClimb] it={it:,d}  best surrogate={best_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # keep a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (4000 gives a good trade‑off between resolution and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from the previously best construction (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        a0 = np.asarray(prev, dtype=np.float64)
        if a0.size != n:
            # simple linear interpolation to the required length
            a0 = np.interp(np.linspace(0, a0.size - 1, n),
                           np.arange(a0.size), a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 /= s

    best_seq = a0.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement (if SciPy is present)
    # ------------------------------------------------------------------
    if _opt is not None:
        try:
            seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                            beta_schedule=None,
                                            max_iter_per_beta=4000)
            val_lb = _objective_surrogate(seq_lb)
            print(f"[L‑BFGS] surrogate = {val_lb:.6f}")
            if val_lb < best_val:
                best_seq = seq_lb.copy()
                best_val = val_lb
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] failed:", e)

    # ------------------------------------------------------------------
    # Phase 2 – Directed hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Hill‑Climb] start")
        seq_hc, val_hc = _hill_climb_opt(best_seq, n, deadline, rng)
        if val_hc < best_val:
            best_seq = seq_hc.copy()
            best_val = val_hc
        print(f"[Hill‑Climb] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true value = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
