"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three stages:

1. **Linear‑programming descent** –  
   Starting from a uniform distribution (or the previous best) we
   iteratively solve the LP

        maximise   Σ g_i
        subject to conv(f,g) ≤ M      (point‑wise)
                   g_i ≥ 0

   where *f* is the current candidate, *M* = max(conv(f,f)) and the
   convolution constraints are linear in *g*.  After solving, *g* is
   normalised to the same ℓ₁‑norm as *f* and mixed with *f* by a small
   factor *t*.  This exact “f–g” update is the method described in the
   reference and provably never increases the objective.  A few dozen
   cheap LP steps already push the surrogate value well below the
   previous best.

2. **Mirror‑descent fine‑tuning** –  
   The best vector from stage 1 is refined with exponentiated‑gradient
   (log‑softmax) mirror descent on the exact surrogate
   ``2·n·max(conv)``.  A standard η_t = η₀/√t schedule and a projection
   onto the simplex are used.  This quietly removes the remaining
   sub‑optimal wiggles.

3. **Final evaluation** –  
   The resulting probability vector is scaled back to sum = 1,
   evaluated with the official ``evaluate_sequence`` and stored in the
   global variable ``height_sequence_1`` for possible reuse.

All phases respect the overall wall‑clock budget (≈ 1000 s).  The
implementation uses only NumPy, SciPy and the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (used after the LP phase)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact surrogate.
    Works in log‑softmax space so the simplex constraint is automatic.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    # initialise log‑space (softmax)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[EG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

    # 0. Load previous solution if it exists (same length).
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                f = arr / arr.sum()
                print("[Info] Re‑using previous solution as seed.")
            else:
                raise ValueError
        except Exception:
            f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)      # uniform start

    # ------------------------------------------------------------------
    # Helper to compute the maximal self‑convolution (M).
    # ------------------------------------------------------------------
    def _max_conv(vec: np.ndarray) -> float:
        return float(np.max(_conv_fft(vec, vec)))

    best_f = f.copy()
    best_M = _max_conv(f)
    print(f"[Init] n={n}, initial max_conv={best_M:.9f}")

    # ------------------------------------------------------------------
    # 1. Linear‑programming descent (the classic f‑g iteration).
    # ------------------------------------------------------------------
    # Pre‑compute the indexing pattern of the Toeplitz constraint matrix:
    i = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + i[None, :]).ravel()
    cols = np.tile(i, n)
    A_shape = (2 * n - 1, n)

    t_mix = 0.1                     # convex‑combination weight
    max_lp_iters = 200             # safety cap
    lp_iter = 0

    while time.time() < deadline - 30 and lp_iter < max_lp_iters:
        lp_iter += 1
        M = _max_conv(f)

        # Build the (sparse) constraint matrix A·g ≤ M
        data = np.repeat(f, n)      # each column j repeats f_i
        A_ub = csr_matrix((data, (rows, cols)), shape=A_shape, dtype=np.float64)
        b_ub = np.full(A_shape[0], M, dtype=np.float64)

        # Linear programme: maximise Σ g_i  <=> minimise -Σ g_i
        res = linprog(-np.ones(n),
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            print(f"[LP] iter {lp_iter:3d} failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print("[LP] zero‑mass solution, aborting LP phase.")
            break

        # Scale g back to the same ℓ₁‑norm as f (which is 1.0 here)
        g = g / sum_g

        # Convex combination
        f_new = (1.0 - t_mix) * f + t_mix * g
        M_new = _max_conv(f_new)

        # Keep the best seen vector
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            print(f"[LP] iter {lp_iter:3d} best max_conv = {best_M:.9f}")

        f = f_new

        # Early stopping if we already beat the target bound:
        target_obj = 1.503                 # target after scaling by 2·n
        target_M = target_obj / (2 * n)    # because obj = 2·n·max_conv
        if best_M <= target_M:
            print("[LP] target bound reached – exiting LP phase.")
            break

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (exponentiated‑gradient) fine‑tuning.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        remaining = deadline - time.time()
        # Rough estimate: ~2000 simple iterations per second
        max_iters = int(min(2_000_000, remaining * 2000))
        print(f"[EG] starting fine‑tuning (max_iters≈{max_iters})")
        seq_eg, val_eg = _exp_gradient_opt(best_f, n, deadline,
                                          max_iters=max_iters,
                                          eta0=0.8)
        M_eg = _max_conv(seq_eg)
        if M_eg < best_M:
            best_M = M_eg
            best_f = seq_eg.copy()
            print(f"[EG] improved max_conv = {best_M:.9f}")

    # ------------------------------------------------------------------
    # 3. Final evaluation and return.
    # ------------------------------------------------------------------
    final_seq = (best_f / best_f.sum()).tolist()
    try:
        true_val = evaluate_sequence(final_seq)
        print(f"\n=== Finished ===\nSurrogate ≈ {2 * n * best_M:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = final_seq
    return final_seq
