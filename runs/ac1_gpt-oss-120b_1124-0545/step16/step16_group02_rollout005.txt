    top_seeds = [s for s,_ in seed_vals[:min(15, len(seed_vals))]]

    # L-BFGS on each top seed but allocate small iteration budget per seed
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0,
                     20000.0, 50000.0, 100000.0]

    # allocate time e.g., 0.3 remaining budget to L-BFGS (or dynamic).
    for i, seed_vec in enumerate(top_seeds):
        if time.time() > end:
            break
        remaining = end - time.time()
        # allocate per seed half of remaining / (remaining seeds) maybe.

        max_iter = int(min(8000, remaining * 5))

            global_best_seq = seq_opt
            global_best_val = val_opt

    # after L-BFGS, run exponentiated gradient for many iterations
    if time.time() < end:
        cur_seq = global_best_seq.copy()
        # allocate remaining for exponentiated gradient
        remaining = end - time.time()

            global_best_seq = seq_exp
            global_best_val = val_exp

    if time.time() < end:
        cur_seq = global_best_seq.copy()
        cur_val = _objective_surrogate(cur_seq)
        best_seq = cur_seq.copy()
        best_val = cur_val

        logit = np.log(cur_seq + 1e-30)
        # schedule: temperature and sigma start high, decline linearly
        # compute total SA steps approx based on time left
        remaining = end - time.time()
        # estimate iters per second:
        # we can just set target = 200_000
        target_iters = 300_000
        start_time = time.time()
        it = 0
        while time.time() < end and it < target_iters:
            it += 1
            fraction = it / target_iters
            sigma = 0.2 * (1 - fraction) + 0.001
            T = cur_val * 0.05 * (1 - fraction) + 1e-6
            proposal_log = logit + sigma * rng.normal(size=n)
            prop = np.exp(proposal_log)
            prop = prop / prop.sum()
            prop_val = _objective_surrogate(prop)
            delta = prop_val - cur_val
            if delta < 0 or rng.random() < math.exp(-delta / (T+1e-12)):
                cur_seq = prop
                cur_val = prop_val
                logit = proposal_log
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()
            if it % 50000 == 0:

        global_best_seq = best_seq
        global_best_val = best_val

    print(f"\n=== Final result ===\nSurrogate {global_best_val:.6f}, true {true_val:.6f}")
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

Propose a low‑autocorrelation step function.

The search proceeds in several stages, each respecting the global
time‑budget (≈ 1000 s):

1.  Build a large, diverse pool of ℓ₁‑normalised seed vectors
    (analytic windows, random Dirichlet / exponential draws,
    sparse / block‑uniform patterns, mixtures, …).  The previous
    solution (if any) is injected into the pool.

2.  Keep the best ~15 seeds (according to the smooth surrogate
    2·n·max(conv)).  Each seed is refined by a continuation L‑BFGS
    optimiser that minimises a log‑sum‑exp approximation of the
    max‑convolution.  The refinement is deliberately aggressive –
    many β‑levels and up to ~8000 L‑BFGS iterations per β.

3.  From the best result so far launch a number of noisy “restart”
    L‑BFGS runs (mirror‑descent in the log‑probability space) to
    explore neighbouring basins.

4.  A long exponentiated‑gradient (mirror‑descent) phase on the exact
    max‑convolution further reduces the objective.

5.  A simulated‑annealing walk in log‑probability space (random
    Gaussian perturbations with a linearly decreasing temperature)
    attempts to escape any remaining shallow minima.

6.  Finally a projected sub‑gradient descent (hundreds of thousands of
    iterations) gives a last polish.

All stages keep the current best vector; the best sequence found
before the deadline is returned as a plain list of floats, and also
saved in the global variable ``height_sequence_1`` for possible reuse

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective for a unit‑sum vector:
    2·n·max(conv(seq,seq)).
    This equals ``evaluate_sequence`` when ``sum(seq)=1``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a large, diverse pool of ℓ₁‑normalised vectors.
    """
    seeds: list[np.ndarray] = []

    # Analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian families -------------------------------------------------
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    seeds.append(_double_gaussian(n, sigma_frac=0.12, shift_frac=0.20))
    seeds.append(_double_gaussian(n, sigma_frac=0.10, shift_frac=0.30))
    seeds.append(_double_gaussian(n, sigma_frac=0.15, shift_frac=0.25))

    # Power law ---------------------------------------------------------
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta shape grid ---------------------------------------------------
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows -----------------------------------------------------
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet --------------------------------------------------
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential ------------------------------------------------
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves ---------------------------------------
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random vectors --------------------------------------------
    for k in (int(0.01 * n), int(0.03 * n), int(0.07 * n)):
        for _ in range(4):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors ---------------------------------------------
    for bf in (0.05, 0.15, 0.30):
        for _ in range(4):
            seeds.append(_block_uniform(n, bf, rng))

    # Mixtures -----------------------------------------------------------
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Special alternating‑extrema Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # The previous solution (if available)
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalisation sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Continuation L‑BFGS optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    # Initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient w.r.t a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing learning rate ≈ 1/√t
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val

# Projected sub‑gradient optimiser
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing optimiser (log‑probability space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random walk in log‑probability space with a decreasing temperature.
    Returns the best sequence encountered.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # Temperature and proposal‑scale schedules (linear decay)
    T0 = cur_val * 0.04 + 1e-6          # initial temperature proportional to objective
    sigma0 = 0.15                       # initial std‑dev in logit space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        frac = it / max_iters
        T = T0 * (1.0 - frac) + 1e-6
        sigma = sigma0 * (1.0 - frac) + 1e-6

        proposal_log = logit + sigma * rng.normal(size=n)
        prop = np.exp(proposal_log)
        prop = prop / prop.sum()
        prop_val = _objective_surrogate(prop)

        delta = prop_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            # accept
            logit = proposal_log
            cur_seq = prop
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if it % 50_000 == 0:
            print(f"[SA] it={it} best={best_val:.6f}")

    return best_seq, best_val

# ----------------------------------------------------------------------
# Noisy restart L‑BFGS (explores neighbouring basins)
# ----------------------------------------------------------------------
def _local_restart_opt(best_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule,
                       num_restarts: int = 20,
                       max_iter_per_beta: int = 4000) -> tuple[np.ndarray, float]:
    """
    Perform noisy restarts of L‑BFGS from the current best solution.
    Returns the best sequence among all restarts.
    """
    v0 = np.log(best_seq + 1e-30)
    best_local_seq = best_seq.copy()
    best_local_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    for r in range(num_restarts):
        if time.time() > deadline:
            break
        noise = 0.2 * rng.standard_normal(n)   # stronger perturbation
        v_start = v0 + noise
        start_vec = _softmax(v_start)
        seq_opt = _lbfgs_smooth_optimize(start_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_schedule,
                                         max_iter_per_beta=max_iter_per_beta)
        val_opt = _objective_surrogate(seq_opt)
        if val_opt < best_local_val:
            best_local_seq = seq_opt.copy()
            best_local_val = val_opt
            print(f"[Restart {r+1}] improved to {best_local_val:.6f}")

    return best_local_seq, best_local_val

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialise RNG and timing
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    t_start = time.time()
    deadline = t_start + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Load previous solution (if any)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list) and len(previous) == n:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 3. Generate seeds
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} candidate seeds.")

    # evaluate surrogate values (fast)
    evals = [(s, _objective_surrogate(s)) for s in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = min(15, len(evals))
    top_seeds = [vec for vec, _ in evals[:top_k]]

    print("\nTop seeds (surrogate):")
    for i, (_, v) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 4. Continuation L‑BFGS on the best few seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 1e5,
        2e5, 5e5, 1e6, 2e6, 5e6, 1e7,
    ]

    # We'll process only a small number of seeds with a generous iteration budget.
    max_lbfgs_seeds = 3
    for idx, seed_vec in enumerate(top_seeds[:max_lbfgs_seeds]):
        if time.time() > deadline:
            print("[Time] deadline reached before completing L‑BFGS seeds.")
            break
        remaining = deadline - time.time()
        # use a fairly large iteration count per β (but capped)
        max_iter = int(min(12000, remaining * 0.04))
        max_iter = max(6000, max_iter)
        print(f"\n--- L‑BFGS run on seed #{idx+1} (max_iter_per_beta={max_iter}) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=max_iter)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Noisy‑restart L‑BFGS from current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Noisy‑restart L‑BFGS phase ---")
        seq_restart, val_restart = _local_restart_opt(global_best_seq,
                                                      n,
                                                      deadline,
                                                      beta_schedule=beta_sched,
                                                      num_restarts=15,
                                                      max_iter_per_beta=4000)
        if val_restart < global_best_val:
            global_best_seq = seq_restart.copy()
            global_best_val = val_restart
            print(f"[Restart] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Exponentiated‑gradient polishing (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(2_000_000, remaining * 0.07))
        max_exp = max(500_000, max_exp)
        print(f"\n--- Exponentiated‑gradient polishing ({max_exp} iters) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(300_000, remaining * 0.07))
        max_sa = max(100_000, max_sa)
        print(f"\n--- Simulated‑annealing refinement ({max_sa} iters) ---")
        seq_sa, val_sa = _simulated_annealing_opt(global_best_seq,
                                                  n,
                                                  deadline,
                                                  max_iters=max_sa)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final long sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(800_000, remaining * 0.07))
        max_sub = max(200_000, max_sub)
        print(f"\n--- Final sub‑gradient polishing ({max_sub} iters) ---")
        seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_sub)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[SubGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 9. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
