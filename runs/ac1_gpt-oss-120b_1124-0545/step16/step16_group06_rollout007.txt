def propose_candidate(seed=123, budget_s=1000,...):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000

    # get previous sequence
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if np.any(arr<0): arr = np.abs(arr) # ensure nonneg?
        best_seq = arr / arr.sum()
    else:
        # if no previous, generate uniform seed
        best_seq = _uniform(n)

    best_val = _objective_surrogate(best_seq)

    # Initial L-BFGS smoothing from best_seq
    print("Initial L-BFGS smoothing")
    best_seq = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                      beta_schedule=_extended_beta_schedule(),
                                      max_iter_per_beta=3000)
    best_val = _objective_surrogate(best_seq)
    print(f"After init L-BFGS surrogate {best_val:.6f}")

    # Local refinement loops
    num_refine = 12
    for i in range(num_refine):
        if time.time() > deadline:
            break
        # noise scale decreasing
        scale = 0.2 * (0.9 ** i)
        v = np.log(best_seq + 1e-30) + scale * rng.normal(size=n)
        # run short L-BFGS warmup
        refined = _lbfgs_smooth_optimize(_softmax(v), n, deadline,
                                         beta_schedule=[1.0,5.0,20.0,80.0,200.0,500.0,1000.0],
                                         max_iter_per_beta=2000)
        val = _objective_surrogate(refined)
        print(f"Refine {i+1}/{num_refine} surrogate {val:.6f}")
        if val < best_val:
            best_seq = refined
            best_val = val
            print(f"New best surrogate {best_val:.6f}")

    # Exponentiated-gradient polishing (full EG) until deadline
    remaining = deadline - time.time()
    if remaining > 0:
        max_iters = int(min(5_000_000, remaining * 2000)) # 2000 iters per second
        print(f"Running EG polishing for up to {max_iters} iterations")
        best_seq, best_val = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iters)
        print(f"After EG best surrogate {best_val:.6f}")

    # Stochastic hill climbing if still time
    remaining = deadline - time.time()
    if remaining > 0:
        max_hc_iters = int(min(2_000_000, remaining * 1500))
        print(f"Running hill climbing for up to {max_hc_iters} iterations")
        best_seq, val_hc = _hill_climb_opt(best_seq, n, deadline, rng, sigma0=0.3, max_iters=max_hc_iters)
        if val_hc < best_val:
            best_seq = best_seq  # already updated inside function
            best_val = val_hc
            print(f"After hillclimb best surrogate {best_val:.6f}")

    # Final subgradient polishing short if time left
    remaining = deadline - time.time()
    if remaining > 0:
        max_sub_iters = int(min(500_000, remaining * 1000))
        print(f"Running final subgradient for up to {max_sub_iters} iterations")
        best_seq, val_sub = _subgradient_opt(best_seq, n, deadline, max_iters=max_sub_iters)
        if val_sub < best_val:
            best_seq = best_seq
            best_val = val_sub
            print(f"After subgradient best surrogate {best_val:.6f}")

    # Evaluate true objective
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as e:
        print("Error evaluating sequence:", e)
        true_val = None

    print(f"Final surrogate {best_val:.6f}, true value {true_val}")
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The search proceeds as follows:

1️⃣  Start from the previously found solution (if any) or from a uniform
    seed.
2️⃣  Perform a long L‑BFGS‑B run on a smooth log‑sum‑exp surrogate of the
    max‑convolution, with an extended schedule of β values (up to 1e7).
3️⃣  Refine the current best by many short L‑BFGS runs from perturbed
    logits (random Gaussian noise).  Each perturbation explores a new
    neighbourhood on the probability simplex.
4️⃣  Finish with a high‑β L‑BFGS pass (β up to 1e8) to tighten the
    approximation of the true max.
5️⃣  As a safety net, a short mirror‑descent (exponentiated‑gradient) pass
    is run if time permits.
6️⃣  The best sequence is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a ℓ₁‑normalised sequence
    (i.e. 2·n·max(conv)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) for the smoothed max‑convolution.
    Parameters are the logits v (softmax(v) = a).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)

    # log‑sum‑exp of the convolution
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimises the smoothed objective using L‑BFGS‑B.
    Returns a ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for L‑BFGS.")

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        try:
            res = _opt.minimize(fun=_lbfgs_obj,
                                x0=v,
                                args=(n, float(beta)),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception as e:  # pragma: no cover
            print("[L‑BFGS] exception", e, "at β =", beta)
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (softmax logits).
    Returns (best_seq, best_val).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
