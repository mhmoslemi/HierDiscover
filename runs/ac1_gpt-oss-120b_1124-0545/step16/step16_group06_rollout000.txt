"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a 4‑phase search adapted to the 1000 s budget.

1️⃣  **Rich seed pool** – many analytic families (uniform, exponential,
    beta, power‑law, symmetric‑power, cosine‑window) and random
    Dirichlet draws, each ℓ₁‑normalised (sum = 1).

2️⃣  **Smooth L‑BFGS** – minimise a log‑sum‑exp surrogate for the
    max‑convolution.  The surrogate uses an inverse‑temperature schedule
    β∈[1,…,10⁶]; the optimiser works in the log‑softmax (log‑probability)
    space and respects the simplex constraint automatically.
    Each seed is refined; the best refined seed is kept.

3️⃣  **Mirror‑descent polishing** – exponentiated‑gradient on the exact
    max‑convolution (sub‑gradient of the active convolution index) for a
    long time budget.  The iterates stay on the simplex via the softmax
    re‑parameterisation.

4️⃣  **Stochastic refinement** – simulated‑annealing on the soft‑max
    logits with a decaying Gaussian proposal; only moves that improve
    (or are accepted with a small temperature‑scaled probability) are
    kept.  After the annealing a short projected‑sub‑gradient pass is
    performed.

All heavy convolution work is performed with an FFT‑based routine
(`_conv_fft`) which is O(n log n) and therefore fast for n≈4000.
The final sequence is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real input, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a ℓ₁‑normalised vector (sum=1): 2·n·max(conv)."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ i^{-p} + (n‑i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int, p: float) -> np.ndarray:
    """Window proportional to (1+cos(π·i/(n‑1)))**p, symmetric, non‑negative."""
    i = np.arange(n, dtype=np.float64)
    # cos varies from 1 (i=0) to -1 (i=n-1); shift to non‑negative.
    w = (1.0 + np.cos(math.pi * i / (n - 1))) ** p
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective with L‑BFGS‑B.
    Works in the log‑softmax space, hence the ℓ₁‑normalisation is
    automatically enforced.
    """
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy is required for the L‑BFGS stage.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0
        ]

    # initialise logits (log‑softmax of the start vector)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            """Return smoothed objective and its gradient w.r.t. logits."""
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # for numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (sub‑gradient of the smoothed max)
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            # full convolution of rev_a with w gives needed gradient for each a_i
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            """Interrupt optimiser if time is up."""
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as exc:  # pragma: no cover
            print(f"[L‑BFGS] exception {exc} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax‑parametrised)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only the active index contributes)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize (standard EG schedule)
        eta = 0.7 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Stochastic hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the soft‑max logits.
    A Gaussian perturbation is added to the logits; moves are accepted
    if they improve the exact surrogate or with a temperature‑scaled
    probability otherwise.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = 0.25            # proposal scale
    temp = 0.02             # temperature for uphill moves
    sigma_decay = 0.99996
    temp_decay = 0.99998

    while time.time() < deadline:
        # propose a new point
        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            # metropolis criterion
            prob = math.exp(-(val_prop - cur_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        # anneal
        sigma *= sigma_decay
        temp *= temp_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (short polishing)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.4 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective (1000 s wall‑clock budget).
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin to guarantee returning before timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (kept identical to the previous best)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if any (kept for continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:  # pragma: no cover
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (fast – a few seconds)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # trivial baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # parametric families (coarse sweeps, interrupted by deadline)
    # exponential families
    for c in np.linspace(0.5, 3.0, 15):
        if time.time() > deadline:
            break
        candidate_pool.append((_exp_family(n, c), _objective_surrogate(_exp_family(n, c)), ('exp', c)))
        candidate_pool.append((_rev_exp_family(n, c), _objective_surrogate(_rev_exp_family(n, c)), ('revexp', c)))

    # beta family
    alphas = np.linspace(0.1, 5.0, 7)
    betas = np.linspace(0.1, 5.0, 7)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            candidate_pool.append((vec, _objective_surrogate(vec), ('beta', a, b)))

    # power‑law (monotone decreasing)
    for p in np.linspace(-0.5, 5.0, 21):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('power', p)))

    # symmetric power‑law (U‑shaped)
    for p in np.linspace(0.5, 3.0, 13):
        if time.time() > deadline:
            break
        vec = _sym_power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('sym_power', p)))

    # cosine window family (smooth dip in the centre)
    for p in np.linspace(0.2, 2.0, 10):
        if time.time() > deadline:
            break
        vec = _cosine_window(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('cosine', p)))

    # random Dirichlet draws
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 20
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2️⃣  Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = min(20, len(candidate_pool))
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k} for L‑BFGS.")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth L‑BFGS refinement
    # ------------------------------------------------------------------
    print("\n--- L‑BFGS refinement ---")
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during L‑BFGS runs.")
            break
        print(f"\nL‑BFGS run #{idx+1}/{top_k}")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=12000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"  surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"  ** new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Mirror‑descent polishing (exponentiated gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 60 % of the remaining time here
        exp_deadline = time.time() + remaining * 0.60
        print("\n--- Exponentiated‑gradient polishing ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, exp_deadline)
        print(f"  polished surrogate = {val_exp:.6f}")
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"  ** improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hc_deadline = time.time() + remaining * 0.70
        print("\n--- Stochastic hill‑climbing (annealing) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq, n, hc_deadline, rng)
        print(f"  hill‑climb surrogate = {val_hc:.6f}")
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"  ** improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final short projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.90  # leave a tiny slack
        max_iters = int(min(500_000, (remaining * 0.1) * 1000))
        if max_iters > 0:
            print("\n--- Final projected sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq, n, sub_deadline, max_iters=max_iters)
            print(f"  sub‑gradient surrogate = {val_sub:.6f}")
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"  ** improved global best to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true objective = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
