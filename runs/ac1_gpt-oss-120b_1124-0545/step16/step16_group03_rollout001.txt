"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm builds a large diverse seed pool (analytic families,
random Dirichlet draws, sparse blocks, arcsine, sine etc.), evaluates a
cheap ℓ₁‑normalised surrogate (2·n·max(conv) for sum‑one vectors) and
keeps the best few seeds.  Those seeds are refined by a cascade of
optimisers:

*   L‑BFGS on a smooth approximation of the max (log‑sum‑exp) with a
    continuation schedule for the smoothing parameter β.
*   Exponentiated‑gradient (mirror descent) directly on the exact surrogate.
*   Long projected sub‑gradient descent.
*   Simulated‑annealing walk in log‑softmax space.
*   Pairwise logit‑swap hill‑climbing.

Occasional random restarts and a simple moving‑average smoothing step are
added to escape plateaus.  All phases respect a global deadline (default
1000 s).  The final sequence is handed to the official ``evaluate_sequence``
function and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# SciPy is optional – L‑BFGS is used only if available.

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

# Analytic families (ℓ₁‑normalised)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine(n: int) -> np.ndarray:
    """Half‑sine shape, zero at the ends, maximum in the middle."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """Arcsine density on a discrete grid (ends are heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


def _double_block(n: int, block_size: int) -> np.ndarray:
    """Two blocks of equal mass placed at the left and right ends."""
    a = np.zeros(n, dtype=np.float64)
    a[:block_size] = 0.5 / block_size
    a[-block_size:] = 0.5 / block_size
    return a

def _alternating_order_from_vec(vec: np.ndarray) -> np.ndarray:
    """Place sorted values alternating from the left and right ends."""
    sorted_desc = np.sort(vec)[::-1]
    n = len(vec)
    out = np.empty_like(vec)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out

    for conc in conc_list:
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])      # desc
            seeds.append(_alternating_order_from_vec(raw))
            # random permutation
            seeds.append(raw[rng.permutation(n)])
            # thresholded sparsification
            th = np.quantile(raw, 0.8)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0:
                seeds.append(sparse / sparse.sum())

    # 4) Sparse uniform blocks ----------------------------------------------------
    for blk in (50, 100, 150, 250):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # 5) Previous best ------------------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # -----------------------------------------------------------------------
    # Normalise robustly ---------------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iter_per_beta: int = 2500,
) -> np.ndarray:
    """
    Refine ``start_vec`` (already ℓ₁‑normalised) with L‑BFGS‑B on the smooth
    max‑convolution surrogate.  ``beta_schedule`` is a list of smoothing
    parameters; larger β approximates the true max more closely.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1000.0,
            2000.0,
            5000.0,
            10000.0,
            20000.0,
            50000.0,
            100000.0,
            200000.0,
        ]

    # initialise logits – they are the optimisation variables
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on the simplex
            conv = _conv_fft(a, a)
            m = np.max(conv)
            shift = m                      # stabilisation trick
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max = shift + log(sum_exp)/beta
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp                  # soft weights, sum to 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]   # factor 2 already accounted

            # chain rule through soft‑max (logits → a)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) polishing on the exact surrogate.
    Operates directly on the true max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (pair contributions)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _subgradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """
    Long projected sub‑gradient descent on the exact surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Random walk in log‑softmax space with a linear cooling schedule.
    """
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        progress = it / max_iters
        T = 0.07 * (1.0 - progress) + 1e-6
        sigma = 0.15 * (1.0 - progress) + 1e-4

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val or math.exp(-(proposal_val - cur_val) / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
    return best_seq, best_val


def _hill_climbing_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap pair‑wise logit swap hill‑climbing.
    """
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val


def _smooth_seq(
    seq: np.ndarray,
    kernel_width: int,
    repeats: int,
    deadline: float,
) -> np.ndarray:
    """
    Apply a simple moving‑average smoothing kernel repeatedly.
    ``kernel_width`` must be odd.  Each smoothing preserves non‑negativity;
    after each pass the vector is re‑normalised to sum to 1.
    """
    if kernel_width < 3:
        return seq
    k = kernel_width
    kernel = np.ones(k, dtype=np.float64) / k
    a = seq.copy()
    for _ in range(repeats):
        if time.time() > deadline:
            break
        a = np.convolve(a, kernel, mode="same")
        # re‑normalise to stay on the simplex
        s = a.sum()
        if s > 0:
            a /= s
    return a

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Initialisation & timing
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Problem size – a power of two gives fast FFTs
    # --------------------------------------------------------------
    n = 4096  # 2^12, easy for FFTs and enough flexibility

    # --------------------------------------------------------------
    # 2. Seed pool construction
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            a = np.asarray(previous, dtype=np.float64)
            if a.size == n:
                prev_arr = a
            else:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # quick surrogate evaluation
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    # ---- phase 1: L‑BFGS with continuation ----
    beta_schedule = [
        1.0,
        5.0,
        20.0,
        80.0,
        200.0,
        500.0,
        1000.0,
        2000.0,
        5000.0,
        10000.0,
        20000.0,
        50000.0,
        100000.0,
        200000.0,
    ]

    # run L‑BFGS on the *best* few seeds (not all 20 – reserve time)
    for idx, seed_vec in enumerate(top_seeds[:5]):
        if time.time() > deadline:
            print("[Time] Deadline reached before finishing L‑BFGS runs.")
            break
        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        # aim for ~0.05 s per 1000 L‑BFGS iterations on average
        max_iter_beta = int(min(8000, remaining * 0.07 * 1e5))
        max_iter_beta = max(2500, max_iter_beta)

        print(f"\n--- L‑BFGS on seed #{idx + 1} (max_iter/β = {max_iter_beta}) ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(
                seed_vec,
                n,
                deadline,
                beta_schedule=beta_schedule,
                max_iter_per_beta=max_iter_beta,
            )
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] failed on seed #{idx + 1}: {e}")
            continue

        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

        # ---- short exp‑gradient polish after each L‑BFGS run ----
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_exp = int(min(500_000, remaining * 0.04 * 1e5))
            max_exp = max(200_000, max_exp)
            seq_exp, val_exp = _exp_gradient_opt(
                global_best_seq, n, deadline, max_iters=max_exp
            )
            if val_exp < global_best_val:
                global_best_seq, global_best_val = seq_exp.copy(), val_exp
                print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Long‑run global polishing (if time permits)
    # --------------------------------------------------------------
    # 4a) Exponentiated‑gradient (long run)
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(800_000, remaining * 0.05 * 1e5))
        max_exp = max(300_000, max_exp)
        print("\n--- Long exponentiated‑gradient polishing ---")
        seq_exp, val_exp = _exp_gradient_opt(
            global_best_seq, n, deadline, max_iters=max_exp
        )
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp.copy(), val_exp
            print(f"[ExpGrad‑Long] improved to {global_best_val:.6f}")

    # 4b) Projected sub‑gradient descent (long run)
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub = int(min(2_000_000, remaining * 0.05 * 1e5))
        max_sub = max(600_000, max_sub)
        print("\n--- Long projected sub‑gradient descent ---")
        seq_sub, val_sub = _subgradient_opt(
            global_best_seq, n, deadline, max_iters=max_sub
        )
        if val_sub < global_best_val:
            global_best_seq, global_best_val = seq_sub.copy(), val_sub
            print(f"[SubGrad‑Long] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sa = int(min(1_200_000, remaining * 0.04 * 1e5))
        max_sa = max(300_000, max_sa)
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_opt(
            global_best_seq, n, deadline, max_iters=max_sa
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Simple smoothing passes (moving‑average) – sometimes helps
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # try a few kernel widths; take the best improvement
        best_local_seq = global_best_seq.copy()
        best_local_val = global_best_val
        for width in (5, 9, 15):
            seq_smooth = _smooth_seq(
                best_local_seq, kernel_width=width, repeats=2, deadline=deadline
            )
            val_smooth = _objective_surrogate(seq_smooth)
            if val_smooth < best_local_val:
                best_local_seq, best_local_val = seq_smooth, val_smooth
                print(f"[Smooth] width={width} improved to {best_local_val:.6f}")
        if best_local_val < global_best_val:
            global_best_seq, global_best_val = best_local_seq, best_local_val

    # --------------------------------------------------------------
    # 7. Local hill‑climbing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = int(min(600_000, remaining * 0.04 * 1e5))
        max_hc = max(200_000, max_hc)
        print("\n--- Hill‑climbing fine‑tuning ---")
        seq_hc, val_hc = _hill_climbing_opt(
            global_best_seq, n, deadline, max_iters=max_hc
        )
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc.copy(), val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
