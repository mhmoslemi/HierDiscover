"""
Improved multi‑stage optimizer for the step‑function objective

The algorithm builds a diverse pool of analytic windows (uniform,
Hann, Hamming, Blackman, Kaiser, exponential, beta, power‑law,
Gaussian and raised‑cosine‑power families).  The best few seeds are
refined with a log‑sum‑exp smoothed L‑BFGS, followed by a long
exponentiated‑gradient (mirror‑descent) phase that works on the exact
max‑convolution.  Afterwards a *mass‑inflation* linear‑programming step
is applied; the new direction is mixed with the current iterate using a
line‑search over the mixing factor α, which yields a stronger reduction
of the peak of the self‑convolution.  A second EG polish and a short
final L‑BFGS run finish the pipeline.  All stages respect the supplied
time budget (default 1000 s) and the best sequence is stored in the
global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (ℓ₁‑norm = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate used during optimisation (ℓ₁‑normalised vectors)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred at the middle; σ = sigma_frac·n."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(- ((idx - centre) ** 2) / (2.0 * sigma * sigma))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine_power_window(n: int, p: float) -> np.ndarray:
    """(0.5·(1‑cos))ᵖ – a raised‑cosine window."""
    i = np.arange(n, dtype=np.float64)
    base = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    a = base ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of max‑convolution."""
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy optimiser required for L‑BFGS.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights from the smoothed max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑softmax domain)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # simple AdaGrad‑style decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step with α‑line‑search
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return Toeplitz pattern (rows, cols, offset) for convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """Solve  max Σ g_j  s.t. g≥0, (f⋆g)_k ≤ M .   Returns g and its ℓ₁‑mass."""
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximise Σg  ⇔ minimise -Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 5) -> np.ndarray:
    """
    Perform several LP mass‑inflation steps.
    For each LP we mix the current iterate with the LP solution using a
    line‑search over α∈(0,0.5] to minimise the new max‑convolution.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_norm = g / sum_g                     # ℓ₁‑normalise the LP solution

        # ----- line‑search over mixing factor α -----
        alphas = np.linspace(0.0, 0.5, 11)[1:]   # exclude 0
        best_alpha = 0.0
        best_M = M

        for alpha in alphas:
            cand = (1.0 - alpha) * seq + alpha * g_norm
            cand = cand / cand.sum()
            cand_M = _conv_fft(cand, cand).max()
            if cand_M < best_M:
                best_M = cand_M
                best_alpha = alpha

        if best_alpha == 0.0:
            # no improvement found – stop
            break

        seq = (1.0 - best_alpha) * seq + best_alpha * g_norm
        seq = seq / seq.sum()
        print(f"[LP] improved max conv from {M:.6e} to {best_M:.6e} (α={best_alpha:.3f})")

    return seq

def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 5.0, 30)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.1, 6.0, 13)
    betas = np.linspace(0.1, 6.0, 13)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.5, 6.0, 35):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals


def _grid_search_gaussian(n: int, deadline: float):
    vals = []
    sigmas = np.linspace(0.05, 0.30, 12)   # fraction of n
    for sf in sigmas:
        if time.time() > deadline:
            break
        vec = _gaussian_window(n, sf)
        vals.append((vec, _objective_surrogate(vec), ('gauss', sf)))
    return vals


def _grid_search_raised_cosine(n: int, deadline: float):

    powers = np.linspace(0.5, 6.0, 12)
    for p in powers:
        if time.time() > deadline:

        vals.append((vec, _objective_surrogate(vec), ('raisedcos', p)))

def _grid_search_kaiser(n: int, deadline: float):
    vals = []
    betas = np.linspace(2.0, 20.0, 12)
    for b in betas:
        if time.time() > deadline:

        vals.append((vec, _objective_surrogate(vec), ('kaiser', b)))
    return vals

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the supplied wall‑clock
    budget and stores the best sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size (4000 has proven to be a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation – a rich mixture of families
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best, if any
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser14', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # parametric sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))
    candidate_pool.extend(_grid_search_gaussian(n, deadline))
    candidate_pool.extend(_grid_search_raised_cosine(n, deadline))
    candidate_pool.extend(_grid_search_kaiser(n, deadline))

    # Random Dirichlet draws with varied concentration
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])           # sort by surrogate (lower is better)
    top_k = 6
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement of the elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=6000,
            beta_schedule=None   # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing (first pass)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(5_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (pass 1, max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad 1] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming mass‑inflation with α‑line‑search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming mass‑inflation (with line‑search) ---")
        seq_lp = _lp_inflation_step(global_best_seq, deadline, max_outer_iters=6)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Second exponentiated‑gradient polishing (after LP)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp2 = int(min(5_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (pass 2, max_iters={max_exp2}) ---")
        seq_exp2, val_exp2 = _exp_gradient_opt(global_best_seq, n, deadline,
                                               max_iters=max_exp2)
        if val_exp2 < global_best_val:
            global_best_seq = seq_exp2.copy()
            global_best_val = val_exp2
            print(f"[ExpGrad 2] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final short L‑BFGS refinement (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS refinement ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            max_iter_per_beta=4000,
            beta_schedule=[1e5, 5e5, 1e6]
        )
        val_final = _objective_surrogate(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Final L‑BFGS] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Stochastic restarts (lightweight)
    # ------------------------------------------------------------------
    restarts = 3
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{4 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS
        seq_restart = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0],
                                             max_iter_per_beta=2000)
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            # polish with EG
            seq_restart, val_restart = _exp_gradient_opt(seq_restart, n, deadline,
                                                         max_iters=250_000)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 9. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
