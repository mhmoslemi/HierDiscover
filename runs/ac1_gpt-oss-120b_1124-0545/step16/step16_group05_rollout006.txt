"""
Improved step‑function optimizer.

The algorithm follows the proven multi‑stage scheme but with
more aggressive hyper‑parameters, extra polishing passes and an
extended linear‑program “mass‑inflation’’ loop.  The search works on
probability vectors (ℓ₁‑norm = 1) and uses three optimisation blocks:

* Diverse parametric families + random Dirichlet draws → a rich seed pool.
* Smooth L‑BFGS on a log‑sum‑exp surrogate (β schedule up to 1e7,
  12 000 L‑BFGS iterations per β) → local minima of the smooth problem.
* Exact sub‑gradient (exponentiated‑gradient) polishing, several
  LP‑inflation steps (with a tiny line‑search on the mixing factor),
  and stochastic restarts.

All stages respect the supplied time budget (default 1000 s) and
store the best sequence found in the global variable `height_sequence_1`.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as _sp_opt
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = 2·n·max(conv) for a ℓ₁‑normalised vector (∑seq = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1 / (i+1)^p   (i=0 … n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on the discrete index set."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 -
         0.5 * np.cos(2.0 * np.pi * i / (n - 1.0)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _kaiser_window(n: int, beta_param: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta_param * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _triangular_decreasing(n: int) -> np.ndarray:
    """Linear decay from 1 to 0."""
    a = (n - np.arange(n, dtype=np.float64)) / n
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian (μ = (n‑1)/2)."""
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            800.0, 3200.0, 12800.0, 51200.0,
            2e5, 8e5, 3e6, 1e7
        ]

    v = np.log(start_vec + 1e-30)          # parameters in log‑softmax space

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)                       # max value for stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smoothed max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 2n‑1
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _sp_opt.minimize(fun=fun,
                                   x0=v,
                                   method='L-BFGS-B',
                                   jac=True,
                                   callback=callback,
                                   options={'maxiter': max_iter_per_beta,
                                            'disp': False,
                                            'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact sub‑gradient (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses log‑softmax parametrisation (i.e. multiplicative updates).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (only the maximal shift)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decreasing step‑size (AdaGrad‑like)
        eta = 0.8 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress reports (kept cheap)
        if it % 500_000 == 0:
            print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step
# ----------------------------------------------------------------------
_conv_pattern_cache = {}   # cache for Toeplitz pattern


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        s.t.      g ≥ 0,
                  (f * g)_k ≤ M   ∀k .
    Returns optimal g (ℓ₁‑normalised) and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # coefficient f_i for every entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximise Σg  ⇔ minimise -Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is fast for this moderate size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
        return g_opt, g_opt.sum()
    else:
        # fallback – return the current vector (no improvement possible)
        return f.copy(), f.sum()


def _line_search_mix(seq: np.ndarray, g_scaled: np.ndarray, M: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a few mixing factors t∈{0.5,0.25,0.125,…} and keep the best reduction.
    Returns the new sequence and its max‑convolution.
    """
    best_seq = seq
    best_M = M
    for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
        if time.time() > deadline:
            break
        new_seq = (1.0 - t) * seq + t * g_scaled   # already ℓ₁‑normalised
        new_M = np.max(_conv_fft(new_seq, new_seq))
        if new_M < best_M:
            best_M = new_M
            best_seq = new_seq
    return best_seq, best_M


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 10) -> np.ndarray:
    """
    Repeated LP mass‑inflation with line‑search on the mixing factor.
    Stops when the LP cannot increase ℓ₁‑mass or when no improvement is seen.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 0.5 and max_outer_iters > 0:
        max_outer_iters -= 1
        conv = _conv_fft(seq, seq)
        M = conv.max()

        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            # LP cannot inject extra mass → we are at a stationary point
            break

        # renormalise g to unit ℓ₁‑mass
        g_scaled = g / sum_g

        # try a few mixing factors and keep the best
        new_seq, new_M = _line_search_mix(seq, g_scaled, M, deadline)

        if new_M < M - 1e-12:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            # no improvement – terminate early
            break
    return seq


# ----------------------------------------------------------------------
# Seed generation utilities
# ----------------------------------------------------------------------
def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 12.0, 36)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.3, 6.0, 40):
        if time.time() > deadline:
            break
        vec = _power_family(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.05, 8.0, 17)
    betas = np.linspace(0.05, 8.0, 17)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _random_dirichlet_draws(n: int, rng: np.random.Generator,
                           deadline: float, max_draws: int = 200):
    concs = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0]
    vals = []
    for conc in concs:
        for _ in range(max_draws // len(concs)):
            if time.time() > deadline:
                return vals
            vec = _random_dirichlet(n, conc, rng)
            vals.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))
    return vals


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  The routine respects the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # problem size – 4000 offers a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (diverse families + random draws)
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n),
                           _objective_surrogate(_uniform(n)),
                           ('uniform',)))

    # previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr,
                               _objective_surrogate(prev_arr),
                               ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta_param=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec,
                               _objective_surrogate(vec),
                               (win_name,)))

    # extra analytic families
    candidate_pool.append((_triangular_decreasing(n),
                           _objective_surrogate(_triangular_decreasing(n)),
                           ('triangular',)))
    candidate_pool.append((_gaussian(n, sigma=n / 6.0),
                           _objective_surrogate(_gaussian(n, sigma=n / 6.0)),
                           ('gaussian',)))

    # parametric sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))

    # random Dirichlet draws
    candidate_pool.extend(_random_dirichlet_draws(n, rng, deadline, max_draws=150))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (by surrogate)
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])        # lower surrogate = better
    top_k = 15
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} candidates, keeping top {top_k} seeds.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement of elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(start_vec=seed_vec,
                                         n=n,
                                         deadline=deadline,
                                         max_iter_per_beta=12000,
                                         beta_schedule=None)   # default schedule
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exact sub‑gradient polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to EG (≈70 % of the rest)
        max_exp_iters = int(min(8_000_000, remaining * 0.65 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming mass‑inflation refinement (multiple outer steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming mass‑inflation refinement ---")
        seq_lp = _lp_inflation_step(global_best_seq, deadline, max_outer_iters=8)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Stochastic restarts – perturbation + short L‑BFGS + EG
    # ------------------------------------------------------------------
    restarts = 4
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{4 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.18, size=n)   # moderate perturbation
        start_vec = _softmax(log_best + pert)

        # short L‑BFGS with a reduced β schedule
        seq_restart = _lbfgs_smooth_optimize(start_vec,
                                             n,
                                             deadline,
                                             max_iter_per_beta=8000,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                            200.0, 800.0, 3200.0])
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            # brief EG polishing
            seq_restart, val_restart = _exp_gradient_opt(seq_restart, n,
                                                         deadline,
                                                         max_iters=300_000)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final optional local search (small random tweaks)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local fine‑tuning (random coordinate tweaks) ---")
        log_a = np.log(global_best_seq + 1e-30)
        step_scale = 0.02
        for it in range(2000):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = step_scale * (rng.random() - 0.5)
            new_log = log_a.copy()
            new_log[i] += delta
            new_log[j] -= delta
            new_seq = _softmax(new_log)
            new_val = _objective_surrogate(new_seq)
            if new_val < global_best_val:
                global_best_seq = new_seq
                global_best_val = new_val
                log_a = new_log
                # shrink step size a bit to keep refinements stable
                step_scale *= 0.98

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
