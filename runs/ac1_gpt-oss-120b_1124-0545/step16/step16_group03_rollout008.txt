"""
step‑function optimiser
-----------------------

This implementation searches for a non‑negative sequence *a* that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where *conv* is the full linear convolution.
The search proceeds in several stages:

1. **Seed generation** – a rich pool of ℓ₁‑normalised candidates:
   uniform, triangular, Gaussian, power‑law, exponential, double‑exponential,
   Beta‑distribution families and many Dirichlet draws.  The previous best
   solution (if any) is also inserted.

2. **Coarse selection** – evaluate the cheap surrogate
   `2·n·max(conv)` on all seeds and keep the best few.

3. **Continuation L‑BFGS** – smooth the non‑smooth max with a
   log‑sum‑exp (temperature β) and minimise the smooth surrogate with
   SciPy’s L‑BFGS‑B.  The schedule of β values becomes increasingly sharp,
   driving the solution towards the true max.

4. **Adam‑style mirror‑descent** – starting from the best L‑BFGS result,
   perform many Adam updates in log‑softmax space while gradually raising β.
   This refines the solution far beyond the short exponentiated‑gradient
   polish used previously.

5. **Local hill‑climbing** – cheap pair‑wise logit swaps that exploit any
   remaining tiny improvements.

All phases respect the global 1000 s budget (a 2 s safety margin is kept
for the final official evaluation).  The best sequence found is stored
in the global ``height_sequence_1`` variable and returned as a plain list
of floats.
"""

import time

import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled at the centre of each cell, then ℓ₁‑normalised."""
    xs = (np.arange(1, n + 1) - 0.5) / n  # centre points in (0,1)
    # avoid NaNs at the boundaries for α<1 or β<1
    eps = 1e-12
    xs = np.clip(xs, eps, 1 - eps)
    a = (xs ** (alpha - 1)) * ((1 - xs) ** (beta - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Beta family (adds extra shape flexibility) -------------------------
    for a in (0.3, 0.5, 1.0, 2.0, 5.0):
        for b in (0.3, 0.5, 1.0, 2.0, 5.0):
            seeds.append(_beta_family(n, a, b))

    # Random Dirichlet draws (different concentrations) --------------------
    conc_list = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])     # descending
            # a simple alternating placement heuristic
            sorted_desc = np.sort(raw)[::-1]
            out = np.empty_like(sorted_desc)
            left, right = 0, n - 1
            for idx, val in enumerate(sorted_desc):
                if idx % 2 == 0:
                    out[left] = val
                    left += 1
                else:
                    out[right] = val
                    right -= 1
            seeds.append(out)

            # thresholded sparsification
            th = np.quantile(raw, 0.7)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks -----------------------------------------------
    for blk in (80, 120, 150, 200):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly ---------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Refine `start_vec` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    # initialise logits – stay on the simplex via softmax
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # probability vector
            conv = _conv_fft(a, a)                     # full convolution
            shift = np.max(conv)                       # for numeric stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of the max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp                     # soft weights (sum to 1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style mirror‑descent optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   total_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimisation in log‑softmax space using a smooth max surrogate."""
    a = start_seq.copy()
    logit = np.log(a + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # β schedule – slowly tightens the smooth max
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  10000.0, 20000.0, 50000.0, 100000.0,
                  200000.0]

    # allocate iterations roughly equally over the schedule
    steps_per_beta = max(1, total_steps // len(beta_sched))
    step_counter = 0

    for beta in beta_sched:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            # current probability vector
            a = _softmax(logit)

            # convolution & smooth max
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_logit = a * (grad_a - dot)

            # Adam update
            step_counter += 1
            m = beta1 * m + (1.0 - beta1) * grad_logit
            v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
            m_hat = m / (1.0 - beta1 ** step_counter)
            v_hat = v / (1.0 - beta2 ** step_counter)

            # base learning rate; tuned empirically
            lr = 0.12 / (np.sqrt(v_hat) + eps_adam)
            logit = logit - lr * m_hat

            # occasional bookkeeping
            if step_counter % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val, best_seq = cur_val, a.copy()

    # Final check
    cur_val = _objective_surrogate(_softmax(logit))
    if cur_val < best_val:
        best_val, best_seq = cur_val, _softmax(logit)
    return best_seq, best_val

# Local hill‑climbing (pairwise logit swaps)
# ----------------------------------------------------------------------
def _hill_climbing_opt(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Very cheap pair‑wise logit swap hill‑climbing."""
    rng = np.random.default_rng()
    logit = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq, best_val = cur_seq.copy(), cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        sigma = 0.02 / math.sqrt(it)
        delta = rng.normal(scale=sigma)
        new_log = logit.copy()
        new_log[i] += delta
        new_log[j] -= delta
        new_seq = _softmax(new_log)
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            logit, cur_seq, cur_val = new_log, new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
    return best_seq, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`
    within the allotted time budget.  Returns the best sequence found.
    """
    # --------------------------------------------------------------
    # 0. Initialisation & time budget
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final call

    # --------------------------------------------------------------
    # 1. Problem size – a compromise between resolution and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Seed pool
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Collected {len(seeds)} raw seeds.")

    # cheap surrogate evaluation (true for ℓ₁‑normalised vectors)
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nBest surrogate values among raw seeds:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1:2d}: {val:.6f}")

    # --------------------------------------------------------------
    # 3. Coarse L‑BFGS refinement (heavy on the best seed)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    # Run L‑BFGS on the absolute best seed (more iterations)
    if _opt is not None:
        remaining = deadline - time.time()
        if remaining > 0:
            max_iter_per_beta = int(min(8000, remaining * 0.04 * 1e5))
            max_iter_per_beta = max(3000, max_iter_per_beta)
            print("\n--- L‑BFGS on the top seed (heavy) ---")
            seq_opt = _lbfgs_smooth_optimize(global_best_seq,
                                             n,
                                             deadline,
                                             max_iter_per_beta=max_iter_per_beta)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_seq, global_best_val = seq_opt, val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Adam‑style polishing (long run in log‑softmax space)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a generous chunk of the remaining budget to Adam
        steps = int(min(500_000, remaining * 0.25 * 1e5))
        steps = max(150_000, steps)
        print("\n--- Adam‑style polishing ---")
        seq_adam, val_adam = _adam_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            total_steps=steps)
        if val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[Adam] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Random perturbations + short L‑BFGS re‑starts
    # --------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # number of re‑starts limited by time
        restarts = 4
        for r in range(restarts):
            if time.time() > deadline:
                break
            perturb_sigma = 0.03 * (0.8 ** r)   # gradually shrinking noise
            perturbed_logits = np.log(global_best_seq + 1e-30) + perturb_sigma * rng.normal(size=n)
            perturbed_seq = _softmax(perturbed_logits)
            max_iter_per_beta = int(min(4000, remaining * 0.02 * 1e5))
            max_iter_per_beta = max(2000, max_iter_per_beta)
            print(f"\n--- Re‑start L‑BFGS #{r + 1} (σ={perturb_sigma:.4f}) ---")
            seq_opt = _lbfgs_smooth_optimize(perturbed_seq,
                                             n,
                                             deadline,
                                             max_iter_per_beta=max_iter_per_beta)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[Re‑L‑BFGS] surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_seq, global_best_val = seq_opt, val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Local hill‑climbing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc = int(min(400_000, remaining * 0.03 * 1e5))
        max_hc = max(100_000, max_hc)
        print("\n--- Hill‑climbing fine‑tuning ---")
        seq_hc, val_hc = _hill_climbing_opt(global_best_seq,
                                            n,
                                            deadline,
                                            max_iters=max_hc)
        if val_hc < global_best_val:
            global_best_seq, global_best_val = seq_hc, val_hc
            print(f"[HillClimb] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
