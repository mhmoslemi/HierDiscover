def _adam_smooth_optimize(start_vec, n, deadline, beta_schedule, iters_per_beta):
    # start_vec is probability distribution (sum=1)
    v = np.log(start_vec + 1e-30) # logits
    best_seq = start_vec.copy()
    best_val = _objective_exact(start_vec)
    # Adam parameters
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v) # second moment called v_2 (like vt)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = 0.05 # initial learning rate
    t = 0
    for beta in beta_schedule:
        if time.time() > deadline: break
        for it in range(iters_per_beta):
            if time.time() > deadline: break
            t += 1
            obj_smooth, grad_v = _smooth_objective(v, n, beta)
            # compute Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_2_hat = v_hat / (1 - beta2 ** t)
            # learning rate schedule: lr = lr0 / np.sqrt(t) perhaps
            lr = lr0 / np.sqrt(t)  # simple decay
            v = v - lr * m_hat / (np.sqrt(v_2_hat) + eps)

            if t % 5000 == 0:

                val = _objective_exact(a)
                if val < best_val:

        # after finishing beta stage we could re-evaluate best

    return best_seq, best_val

"""
Fast Adam‑based optimisation of the max‑convolution bound.

The algorithm works as follows

1. Build a modest pool of ℓ₁‑normalised “seed” vectors (uniform,
   geometric, power‑law, beta, sine‑window, random Dirichlet, …).  The
   best seed from the pool is used as a warm‑start.

2. For each seed run a continuation optimiser that minimises a
   log‑sum‑exp smooth approximation of the max‑convolution.
   The optimisation variable is the log‑probability vector `v`
   (soft‑max(v) yields the candidate heights).  Adam is used because
   it tolerates the noisy, non‑convex landscape and can perform many
   cheap iterations.

3. The smoothing parameter β is increased progressively; for each
   β a fixed number of Adam steps is carried out (the step‑size is
   slowly decayed so that the optimiser remains stable for large β).

4. After each optimisation run the exact (non‑smoothed) objective
   ``_objective_exact`` is evaluated and the best vector is kept.
   A brief stochastic “mass‑transfer” local search is performed to
   polish the final candidate.

All work respects the 1000 s budget.  The best sequence is stored in the
global name ``height_sequence_1`` and also returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (faster than np.convolve)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families that return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact objective / smoothed surrogate
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) / (sum(a)²),  (sum(a)=1 ⇒ 2·n·max)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv) / (np.sum(a) ** 2)


def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """
    Log‑sum‑exp smooth approximation of max(conv(a,a)).
    Returns (obj, grad) where grad is ∂obj/∂v.
    """
    a = _softmax(v)                     # probability vector (ℓ₁‑norm = 1)
    conv = _conv_fft(a, a)               # full convolution length = 2n‑1
    shift = np.max(conv)                 # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_term / sum_exp               # soft‑max over lags
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)           # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice length n

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float],
    iters_per_beta: int,
    lr0: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probability vector `v` to minimise the smoothed
    max‑convolution.  Returns the best exact candidate found.
    """
    # Initialise logits (soft‑max(v) = start_vec)
    v = np.log(start_vec + 1e-30)

    # Adam state
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)   # second moment accumulator
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # Keep the best exact candidate discovered
    best_a = start_vec.copy()
    best_val = _objective_exact(best_a)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj_smooth, grad_v = _smooth_objective(v, n, beta)

            # Adam updates (bias‑corrected)
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)

            # Learning‑rate decay: 1/√t, additionally damped for large β
            lr = lr0 / np.sqrt(t) / np.sqrt(beta)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

            # Occasionally evaluate the exact objective
            if t % 5000 == 0:
                a = _softmax(v)
                val = _objective_exact(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()

        # End of β stage – one more exact check
        a = _softmax(v)
        val = _objective_exact(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers) – polishing step
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 5000,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i → j if it helps."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(123456789)

    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_frac * a[i]
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _objective_exact(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand      # continue from improved point
    return best_a, best_val


# ----------------------------------------------------------------------
# Seed‑generation (small, diverse pool)
# ----------------------------------------------------------------------
def _collect_seeds(n: int, rng: np.random.Generator, prev: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exponential_decay(n, lam))
    for p in (0.0, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        seeds.append(_power_law(n, p))
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        for b in (0.5, 1.0, 1.5, 2.0, 3.0):
            seeds.append(_beta_shape(n, a, b))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=g))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any)
    if prev is not None and prev.size == n and prev.sum() > 0.0:
        seeds.append(prev / prev.sum())

    # Clean, enforce non‑negativity and ℓ₁‑normalisation
    cleaned = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` under a
    1000 s budget.  The optimiser is an Adam‑based continuation method
    combined with a final stochastic local search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0         # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimisation start: n={n}, budget≈{budget_s}s ===")

    # ------------------------------------------------------------------
    # Seed pool (include warm‑start if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick evaluation of the raw seeds – keep the best few for Adam
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best raw seed objective = {best_val:.6f}")

    # Sort seeds by exact objective and keep top‑k for intensive optimisation
    seeds.sort(key=_objective_exact)
    top_k = min(8, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # Adam continuation optimiser
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val

    # β schedule – start very smooth, end very sharp
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0,
    ]

    print("\n--- Adam continuation optimisation ---")
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing Adam runs.")
            break

        # allocate iteration budget for this seed (time‑aware)
        remaining = deadline - time.time()
        # aim for roughly 0.03 s per iteration → 30 k iterations ≈ 900 s,
        # we split proportionally among the remaining seeds
        iterations = int(min(70_000, max(5_000, remaining * 0.08)))
        iters_per_beta = max(2, iterations // len(beta_sched))

        print(f"\nRun #{idx + 1}/{top_k}: iters per β = {iters_per_beta}")
        seq_opt, val_opt = _adam_optimize(
            start_vec=seed_vec,
            n=n,
            deadline=deadline,
            beta_schedule=beta_sched,
            iters_per_beta=iters_per_beta,
            lr0=0.08,
        )
        print(f"[Adam] exact objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Short stochastic local‑search polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(30_000, remaining * 30))
        if iters > 0:
            print("\n--- Local‑search polishing ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.015,
            )
            print(f"[LocalSearch] objective = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_val = val_loc
                global_best_seq = seq_loc.copy()
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation through the provided evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {final_score:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in a possible next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
