def _perm_swap_opt(seq, n, deadline, max_iters=200000, rng=None):
    if rng is None: rng = np.random.default_rng()
    a = seq.copy()
    conv = _conv_fft(a, a)
    cur_max = conv.max()
    best_seq = a.copy()
    best_val = 2.0 * n * cur_max
    cur_val = best_val

    temperature = cur_max * 0.01 # small fraction

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        if i == j: continue
        xi = a[i]
        xj = a[j]
        if xi == xj:  # swapping identical values yields no improvement
            continue
        dx = xj - xi
        dy = xi - xj  # = -dx
        # store old conv slices
        old_i = conv[i:i+n].copy()
        old_j = conv[j:j+n].copy()
        # apply updates using original a (before swap)
        conv[i:i+n] += dx * a
        conv[j:j+n] += dy * a
        # swap a entries
        a[i], a[j] = xj, xi
        new_max = conv.max()
        new_val = 2.0 * n * new_max
        # accept if better or with probability
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability based on temperature
            prob = math.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:

        if accept:

            cur_max = new_max
            if new_val < best_val:

                best_seq = a.copy()
            # possibly cooling

            conv[j:j+n] = old_j
            a[i], a[j] = xi, xj  # swap back
            # maybe increase temperature? We just keep it
            temperature *= 1.00001
    return best_seq, best_val

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for an ℓ₁‑normalised sequence (i.e. 2·n·max(conv)).
    Returns the surrogate value directly, avoiding the final scaling by
    the sum (which is 1 for our normalised vectors).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ──────────────────────────────────────────────────────────────────────
# Analytic families (all return ℓ₁‑normalised vectors)
# ──────────────────────────────────────────────────────────────────────
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ──────────────────────────────────────────────────────────────────────
# Helper: a deterministic “large‑small” alternating ordering.
# ──────────────────────────────────────────────────────────────────────
def _alternating_large_small(vec: np.ndarray) -> np.ndarray:
    """
    Returns a permutation of *vec* where the largest weight is opposite the
    smallest, the second‑largest opposite the second‑smallest, etc.
    This arrangement minimises the sum of products a[i]*a[n‑1‑i] by the
    rearrangement inequality.
    """
    n = vec.shape[0]
    asc = np.sort(vec)          # ascending
    desc = asc[::-1]            # descending
    out = np.empty_like(vec)
    half = n // 2
    # pair large with small
    out[:half] = desc[:half]
    out[-half:] = asc[:half]
    if n % 2 == 1:
        # middle element gets the median
        out[half] = asc[half]
    return out


# ──────────────────────────────────────────────────────────────────────
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ──────────────────────────────────────────────────────────────────────
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective with L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 2000.0, 5000.0, 20000.0,
            50000.0, 200000.0, 500000.0, 1_000_000.0,
        ]

    # initialise logits ensuring numerical safety
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # max element (for stability)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (exact derivative of the smoothed max)
            w = exp_beta / sum_exp                    # soft‑max weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max parametrisation
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ──────────────────────────────────────────────────────────────────────
# Exact polishing via exponentiated‑gradient (mirror‑descent)
# ──────────────────────────────────────────────────────────────────────
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (the shift attaining the max)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ──────────────────────────────────────────────────────────────────────
# Projected sub‑gradient optimiser (exact max‑convolution)
# ──────────────────────────────────────────────────────────────────────
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ──────────────────────────────────────────────────────────────────────
# Permutation annealing (swap‑based hill‑climbing)
# ──────────────────────────────────────────────────────────────────────
def _perm_swap_opt(seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the *ordering* of a fixed multiset of heights.
    Swaps two positions, updates the convolution in O(n) thanks to the
    identity conv[i:i+n] += Δ·a, and accepts moves that lower the true
    objective (or with a temperature‑controlled probability).
    Returns the best sequence encountered and its surrogate objective.
    """
    a = seq.copy()
    conv = _conv_fft(a, a)                 # full convolution (length 2n‑1)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_seq = a.copy()
    best_val = cur_val

    # simple exponential cooling schedule
    temperature = cur_max * 0.05 if cur_max > 0 else 1e-4
    temp_decay = 0.99999

    # progress reporting
    report_every = max(1, max_iters // 20)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # propose a random swap (ensure i != j)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        xi = a[i]
        xj = a[j]
        if xi == xj:
            continue

        dx = xj - xi          # change for position i
        dy = -dx              # change for position j  (xi - xj)

        # store the slices we are about to modify (for possible revert)
        old_i = conv[i:i + n].copy()
        old_j = conv[j:j + n].copy()

        # apply the updates using the *old* a (before swapping)
        conv[i:i + n] += dx * a
        conv[j:j + n] += dy * a

        # perform the actual swap in the sequence
        a[i], a[j] = xj, xi

        new_max = conv.max()
        new_val = 2.0 * n * new_max

        # acceptance rule
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move with probability exp(-Δ/T)
            prob = math.exp(-(new_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
            # cool down a bit on successful moves
            temperature *= 0.99995
        else:
            # revert both the convolution and the permutation
            conv[i:i + n] = old_i
            conv[j:j + n] = old_j
            a[i], a[j] = xi, xj
            # warm up ever so slightly to keep exploration alive
            temperature *= 1.00001

        # occasional reporting
        if it % report_every == 0:
            print(f"[PermSwap] it={it:,} best={best_val:.6f} cur={cur_val:.6f} temp={temperature:.2e}")

    return best_seq, best_val


# ──────────────────────────────────────────────────────────────────────
# Grid‑search helpers (still useful for enriching the seed pool)
# ──────────────────────────────────────────────────────────────────────
def _grid_search_exponential(n: int, deadline: float) -> list:
    """Sweep exponential family over c ∈ [0.5, 3.0]."""
    vals = []
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float) -> list:
    """Coarse beta‑shape sweep (α,β ∈ [0.1,5] with 9 points each)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float) -> list:
    """Power‑law sweep p ∈ [-0.5,5] (31 points)."""
    vals = []
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals

# Main entry point
# ──────────────────────────────────────────────────────────────────────
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the previous best for
    # comparability; larger values tend to increase runtime.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Pull the previously‑found sequence (if any) for warm‑starting.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation (rich pool)
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    seed_pool = []

    # uniform baseline
    seed_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best, if we have one
    if prev_arr is not None:
        seed_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # parametric families
    seed_pool.extend(_grid_search_exponential(n, deadline))
    seed_pool.extend(_grid_search_beta(n, deadline))
    seed_pool.extend(_grid_search_power(n, deadline))

    # random Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            seed_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # additional deterministic large‑small ordering applied to the best
    # Dirichlet sample seen so far (helps reduce the principal product)
    if seed_pool:
        best_vec = min(seed_pool, key=lambda x: x[1])[0]
        alt_vec = _alternating_large_small(best_vec)
        seed_pool.append((alt_vec, _objective_surrogate(alt_vec), ('alternating',)))

    # ------------------------------------------------------------------
    # Keep the most promising seeds (top‑k)
    # ------------------------------------------------------------------
    seed_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in seed_pool[:top_k]]
    print(f"Generated {len(seed_pool)} seeds, retained top {top_k}.")

    # ------------------------------------------------------------------
    # 2️⃣  L‑BFGS continuation from each seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=12000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exact polishing (exponentiated‑gradient + sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp_iters = int(min(3_000_000, remaining * 0.08))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(1_200_000, remaining * 0.05))
        if max_sub_iters > 0:
            print("\n--- Final sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Permutation annealing (swap‑based hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to a few hundred thousand swaps depending on time left
        max_swap_iters = int(min(800_000, remaining * 0.20))
        if max_swap_iters > 0:
            print(f"\n--- Permutation annealing (max_iters={max_swap_iters}) ---")
            seq_perm, val_perm = _perm_swap_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_swap_iters,
                                                rng=rng)
            if val_perm < global_best_val:
                global_best_seq = seq_perm.copy()
                global_best_val = val_perm
                print(f"[PermSwap] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
