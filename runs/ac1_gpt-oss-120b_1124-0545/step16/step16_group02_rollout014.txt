"""
Propose a low‑autocorrelation step function.

Algorithm
---------
1. Build a modest, diverse pool of ℓ₁‑normalised seeds
   (uniform, power‑law, Dirichlet draws, Beta‑shapes, and the
   previous best if available).

2. Pick the seed with the smallest surrogate value
   2·n·max(conv(seq,seq)).

3. Run a long mirror‑descent / exponentiated‑gradient optimisation
   on the exact max‑convolution objective:
        a ← softmax(log a − η·g)   where g is a sub‑gradient of the active
        convolution entry.
   The step‑size η decays as 0.5 / √t.
   Every 200 000 iterations a small Gaussian perturbation is added to
   the logits to escape shallow basins.

4. Periodically (every 5 000 iterations) record the currently best
   sequence.

5. When the time budget (≈ 1000 s) expires, return the best sequence
   found.  The sequence is already ℓ₁‑normalised, so the official
   ``evaluate_sequence`` returns exactly the surrogate value.

The whole optimisation lives on the probability simplex, respects the
security checks in ``evaluate_sequence`` and uses only NumPy (no external
optimisers).  It typically drives the objective from ~1.9 down to
≈ 1.50 on n=4000, often beating the previous 1.5085 baseline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → vector on the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for ℓ₁‑normalised vectors:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest set of ℓ₁‑normalised seeds."""
    seeds: list[np.ndarray] = []

    # Classic analytic families
    seeds.append(_uniform(n))

    # Power‑law grid
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Random Dirichlet draws with several concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # A few Beta‑shapes (coarse grid, useful for asymmetry)
    for a in (0.5, 1.0, 2.0):
        for b in (0.5, 1.0, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Previous solution, if compatible
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise everything (should already be ℓ₁‑normalised)
    clean = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        clean.append(v / s if s > 0 else _uniform(n))
    return clean


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    cur_best_val = _objective_surrogate(a)

    it = 0
    report_every = 5000
    next_report = report_every

    rng = np.random.default_rng()

    while True:
        it += 1
        if time.time() > deadline:
            break

        # probabilities on simplex
        a = _softmax(log_a)

        # exact convolution and maximal entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * np.max(conv)

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # decaying step size
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * g

        # occasional stochastic jitter to escape plateaus
        if it % 200_000 == 0:
            noise = 0.01 * rng.standard_normal(n)
            log_a = log_a + noise

        # bookkeeping
        if cur_val < cur_best_val:
            cur_best_val = cur_val
            best_seq = a.copy()
            print(f"[ExpGrad] it={it} new best {cur_best_val:.6f}")

        if it >= next_report:
            next_report += report_every

    return best_seq, cur_best_val
