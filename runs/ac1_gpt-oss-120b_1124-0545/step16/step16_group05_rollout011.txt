def _adam_exp_gradient_opt(start_seq, n, deadline, max_iters=10_000_000):

    a = start_seq.copy()

    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)

    v = np.zeros_like(a)

    beta1 = 0.9

    beta2 = 0.999

    eps = 1e-8

    best_seq = a.copy()

    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):

        if time.time() > deadline:

            break

        conv = _conv_fft(a, a)

        max_idx = int(np.argmax(conv))

        # compute subgradient g

        g = np.zeros_like(a)

        i_start = max(0, max_idx - (n - 1))

        i_end = min(n - 1, max_idx)

        idx_i = np.arange(i_start, i_end + 1)

        rev_idx = max_idx - idx_i

        g[idx_i] = 2.0 * a[rev_idx]

        # Adam update

        m = beta1 * m + (1 - beta1) * g

        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)

        v_hat = v / (1 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)  # lr 0.2

        log_a = log_a - step * m_hat

        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)

        if cur_val < best_val:

            best_val = cur_val

            best_seq = a.copy()

        # optional progress print

        if it % 500_000 == 0:

            print(f"[AdamExpGrad] it={it}, best={best_val:.6f}")

    return best_seq, best_val

def _lp_improve(seq, deadline, max_outer_iters=10):

    n = len(seq)

    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:

        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)

        M = conv.max()

        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:

            break

        g_scaled = g / sum_g

        new_M = _conv_fft(g_scaled, g_scaled).max()

        if new_M < M:

            seq = g_scaled

            print(f"[LP] direct replace improved max conv {M:.6e} -> {new_M:.6e}")

            continue

        # no direct improvement; try mixing

        improved = False

        for t in (0.5, 0.25, 0.1, 0.05):

            new_seq = (1 - t) * seq + t * g_scaled

            new_seq = new_seq / new_seq.sum()

            new_M = _conv_fft(new_seq, new_seq).max()

            if new_M < M:

                seq = new_seq

                print(f"[LP] mixed t={t} improved max conv to {new_M:.6e}")

                improved = True

                break

        if not improved:

            break

    return seq

def _local_search(seq, n, deadline, max_trials=2000):

    best_seq = seq.copy()

    best_val = _objective_surrogate(seq)

    log_best = np.log(best_seq + 1e-30)

    for trial in range(max_trials):

        if time.time() > deadline:

            break

        # random perturbation of log coordinates (Gaussian)

        pert = np.random.default_rng().normal(scale=0.1, size=n)

        cand_log = log_best + pert

        cand = _softmax(cand_log)

        val = _objective_surrogate(cand)

        if val < best_val:

            best_val = val

            best_seq = cand.copy()

            log_best = cand_log

            # optionally print

            if trial % 200 == 0:

                print(f"[LocalSearch] trial {trial}, new best {best_val:.6f}")

    return best_seq, best_val

"""
Enhanced optimizer for the step‑function objective.

The algorithm builds on the previous multi‑stage pipeline
but adds several refinements:

* richer seeding (more parameter sweeps, Dirichlet draws)
* a more aggressive LP “mass‑inflation’’ step that
  directly adopts the scaled LP solution when it reduces the
  maximal autocorrelation, and otherwise mixes with decreasing
  step‑sizes.
* Adam‑style exponentiated‑gradient (mirror‑descent) instead of
  plain square‑root decay – faster convergence of the exact
  max‑convolution sub‑gradient.
* additional local stochastic search (Gaussian perturbations)
  after the main optimisation stages.
* tighter time‑checks and a smaller elite‑seed set so that more
  iterations can be performed inside the 1000 s budget.

The routine respects the supplied budget and returns a list
`height_sequence_1` containing the best sequence found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as _sp_opt
from scipy.optimize import linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max
            w = exp_beta / sum_exp                     # weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _sp_opt.minimize(fun=fun,
                                   x0=v,
                                   method='L-BFGS-B',
                                   jac=True,
                                   callback=callback,
                                   options={'maxiter': max_iter_per_beta,
                                            'disp': False,
                                            'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser (mirror descent)
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution using Adam step‑size."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)   # learning‑rate 0.2
        log_a = log_a - step * m_hat
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 500_000 == 0:
            print(f"[AdamEG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

# Linear‑programming “mass‑inflation’’ step (enhanced)
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        subject to g ≥ 0,
                   (f * g)_k ≤ M   ∀k .
    Returns the optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_improve(seq: np.ndarray,
                deadline: float,
                max_outer_iters: int = 10) -> np.ndarray:
    """
    Repeated LP “mass‑inflation’’ steps.
    If the scaled LP solution directly lowers the maximal convolution,
    it replaces the current sequence; otherwise a convex combination with
    decreasing mixing factors is attempted.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        # scaled LP solution (unit ℓ₁‑mass)
        g_scaled = g / sum_g
        new_M = _conv_fft(g_scaled, g_scaled).max()

        if new_M < M:
            seq = g_scaled
            print(f"[LP] direct replace reduced max‑conv {M:.6e} → {new_M:.6e}")
            continue

        # otherwise try mixing with a few step sizes
        improved = False
        for t in (0.5, 0.25, 0.1, 0.05):
            mixed = (1 - t) * seq + t * g_scaled
            mixed = mixed / mixed.sum()
            mixed_M = _conv_fft(mixed, mixed).max()
            if mixed_M < M:
                seq = mixed
                print(f"[LP] mixed (t={t}) improved max‑conv to {mixed_M:.6e}")
                improved = True
                break
        if not improved:
            break
    return seq

def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 6.0, 30)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.1, 6.0, 13)
    betas = np.linspace(0.1, 6.0, 13)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.5, 6.0, 35):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals

def _local_search(base_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_trials: int = 500) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq)
    log_best = np.log(best_seq + 1e-30)
    rng = np.random.default_rng(int(time.time()))
    for trial in range(max_trials):
        if time.time() > deadline:
            break
        pert = rng.normal(scale=0.08, size=n)
        cand_log = log_best + pert
        cand = _softmax(cand_log)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            log_best = cand_log
            if trial % 100 == 0:
                print(f"[LocalSearch] trial {trial} new best {best_val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  The routine respects the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load a previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (diverse families)
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # parameter sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # Random Dirichlet draws (broad concentration spectrum)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    print(f"Collected {len(candidate_pool)} seeds")

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])

"""
Enhanced optimizer for the step‑function objective.

The algorithm builds on the previous multi‑stage pipeline but adds
the following refinements:

* richer seeding (more parameter sweeps and Dirichlet draws);
* a more aggressive LP “mass‑inflation’’ step that directly adopts the
  scaled LP solution when it reduces the maximal autocorrelation,
  otherwise mixes with a set of decreasing step‑sizes;
* Adam‑style exponentiated‑gradient (mirror descent) instead of the
  plain square‑root decay, giving faster convergence of the exact
  max‑convolution sub‑gradient;
* an extra local stochastic search (Gaussian perturbations) after the
  main optimisation stages;
* tighter time‑checks and a smaller elite‑seed set, allowing more
  iterations inside the 1000 s budget.

The routine respects the supplied budget and stores the best sequence
in the global variable ``height_sequence_1`` which is finally returned
as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as _sp_opt
from scipy.optimize import linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _sp_opt.minimize(fun=fun,
                                   x0=v,
                                   method='L-BFGS-B',
                                   jac=True,
                                   callback=callback,
                                   options={'maxiter': max_iter_per_beta,
                                            'disp': False,
                                            'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser (mirror descent)
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution using an Adam‑style step."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / (np.sqrt(v_hat) + eps)
        log_a = log_a - step * m_hat
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if it % 500_000 == 0:
            print(f"[AdamEG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step (enhanced)
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        subject to g ≥ 0,
                   (f * g)_k ≤ M   ∀k .
    Returns the optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_improve(seq: np.ndarray,
                deadline: float,
                max_outer_iters: int = 10) -> np.ndarray:
    """
    Repeated LP “mass‑inflation’’ steps.
    If the scaled LP solution directly lowers the maximal convolution,
    it replaces the current sequence; otherwise a convex combination with
    decreasing mixing factors is attempted.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_scaled = g / sum_g
        new_M = _conv_fft(g_scaled, g_scaled).max()

        if new_M < M:
            seq = g_scaled
            print(f"[LP] direct replace reduced max‑conv {M:.6e} → {new_M:.6e}")
            continue

        improved = False
        for t in (0.5, 0.25, 0.1, 0.05):
            mixed = (1 - t) * seq + t * g_scaled
            mixed = mixed / mixed.sum()
            mixed_M = _conv_fft(mixed, mixed).max()
            if mixed_M < M:
                seq = mixed
                print(f"[LP] mixed (t={t}) improved max‑conv to {mixed_M:.6e}")
                improved = True
                break
        if not improved:
            break
    return seq

def _grid_search_exponential(n: int, deadline: float):

        rev = _rev_exp_family(n, c)

def _grid_search_beta(n: int, deadline: float):

    alphas = np.linspace(0.1, 6.0, 13)

                return vals
            vec = _beta_shape(n, a, b)

def _local_search(base_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_trials: int = 500) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq)
    log_best = np.log(best_seq + 1e-30)
    rng = np.random.default_rng(int(time.time()))
    for trial in range(max_trials):
        if time.time() > deadline:
            break
        pert = rng.normal(scale=0.08, size=n)
        cand_log = log_best + pert
        cand = _softmax(cand_log)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            log_best = cand_log
            if trial % 100 == 0:
                print(f"[LocalSearch] trial {trial} new best {best_val:.6f}")
    return best_seq, best_val

    # previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # parametric sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # Random Dirichlet draws (broad concentration spectrum)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    print(f"Collected {len(candidate_pool)} seeds")

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])  # lower surrogate is better
    top_k = min(6, len(candidate_pool))
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Keeping top {top_k} seeds for intensive optimisation")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop over elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before processing remaining seeds.")
            break
        print(f"\n--- Optimising seed #{idx + 1}/{top_k} ---")

        # 3a. Smooth L‑BFGS refinement
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None   # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        # 3b. LP “mass‑inflation’’ refinement (several passes)
        seq_opt = _lp_improve(seq_opt, deadline, max_outer_iters=8)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[LP] after refinement surrogate = {val_opt:.6f}")

        # 3c. Adam‑style exponentiated‑gradient polishing
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iters = int(min(7_000_000, remaining * 0.08 * 1e5))
            seq_opt, val_opt = _adam_exp_gradient_opt(seq_opt, n, deadline, max_iters=max_iters)
            print(f"[AdamEG] best surrogate after EG = {val_opt:.6f}")

        # 3d. Second LP pass (optional)
        seq_opt = _lp_improve(seq_opt, deadline, max_outer_iters=4)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[LP] second pass surrogate = {val_opt:.6f}")

        # 3e. Update global best if improved
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic restarts with short refinements
    # ------------------------------------------------------------------
    restarts = 4
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{5 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        perturb = rng.normal(scale=0.15, size=n)
        start_log = log_best + perturb
        start_vec = _softmax(start_log)

        # short L‑BFGS
        seq_restart = _lbfgs_smooth_optimize(
            start_vec, n, deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0],
            max_iter_per_beta=4000
        )
        val_restart = _objective_surrogate(seq_restart)

        # brief Adam‑EG
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iters = int(min(2_000_000, remaining * 0.05 * 1e5))
            seq_restart, val_restart = _adam_exp_gradient_opt(seq_restart, n, deadline, max_iters=max_iters)

        # LP polishing
        seq_restart = _lp_improve(seq_restart, deadline, max_outer_iters=3)
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            global_best_seq = seq_restart.copy()
            global_best_val = val_restart
            print(f"[Restart] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final local stochastic search
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Final local stochastic search ---")
        seq_local, val_local = _local_search(global_best_seq, n, deadline, max_trials=600)
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[LocalSearch] improved global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
