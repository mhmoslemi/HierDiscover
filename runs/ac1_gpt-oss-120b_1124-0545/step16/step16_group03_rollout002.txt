"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

within a 1000 s budget.

Algorithm
---------

* **Seed generation** – a diverse pool of L¹‑normalised candidates:
  uniform, triangular (center and edge versions), several Gaussians,
  power‑law, exponential / double‑exponential, cosine‑edge families,
  and many Dirichlet draws (both dense and sparse).

* **Smooth L‑BFGS refinement** – the non‑smooth max‑convolution is
  approximated by a log‑sum‑exp with a continuation parameter β.
  SciPy’s L‑BFGS‑B is run on the smoothed objective with a generous
  iteration budget and an aggressive β‑schedule (up to 1e5).  The
  optimiser works in log‑softmax space, i.e. the variables are logits
  `v` of a soft‑max that guarantees the simplex constraints.

* **Long projected sub‑gradient descent** – after L‑BFGS we polish the
  best solution with many sub‑gradient steps on the exact objective.
  Each step follows the sub‑gradient of the active convolution entry,
  a diminishing learning‑rate 𝜂ₜ=0.25/√t and a Euclidean projection
  onto the probability simplex.

* **Final evaluation** – the best vector found (stored in the global
  ``height_sequence_1``) is passed to the official ``evaluate_sequence``
  function and returned as a plain list of floats.

All phases respect the global deadline; if time runs out the best
solution obtained so far is returned.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate works on ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular_center(n: int) -> np.ndarray:
    """Standard triangle – peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _triangular_edge(n: int, power: float = 1.0) -> np.ndarray:
    """Inverted triangle – mass concentrated at the two edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.abs(i - c) / c
    a = a ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_edge(n: int, power: float = 1.0) -> np.ndarray:
    """Cosine that is 1 at the edges and 0 in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (n - 1))
    a = np.abs(a) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _half_cosine(n: int) -> np.ndarray:
    """Half‑cosine bump – high at the edges, low in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = (np.cos(np.pi * i / (n - 1)) + 1.0) / 2.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse collection of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular_center(n))
    seeds.append(_triangular_edge(n, power=1.0))
    seeds.append(_triangular_edge(n, power=2.0))
    for sigma in (0.08, 0.15, 0.22, 0.30):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0):
        seeds.append(_power(n, p))
    for rate in (0.002, 0.006, 0.010, 0.020):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))
    for power in (1.0, 2.0, 3.0):
        seeds.append(_cosine_edge(n, power))
    seeds.append(_half_cosine(n))

    # Random Dirichlet draws (different concentrations) --------------------
    for conc in (0.02, 0.05, 0.10, 0.30):
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                     # raw
            seeds.append(np.sort(raw)[::-1])                     # descending
            # sparsify
            th = np.quantile(raw, 0.75)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks ------------------------------------------------
    for blk in (80, 120, 200):
        for _ in range(4):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise robustly ---------------------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smoothed max‑conv."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]

    # initialise logits – optimisation variables live in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                    # stays on simplex
            conv = _conv_fft(a, a)                    # full linear convolution
            shift = np.max(conv)                      # numerical stability
            # smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = float(exp_beta.sum())
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t a
            w = exp_beta / sum_exp                    # soft‑weights, sum to 1
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # length‑n gradient

            # chain rule through soft‑max (logits → a)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def cb(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=cb,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]

    if time.time() < deadline and _opt is not None:

        # allocate the bulk of the remaining budget to this phase

        lbfgs_time = remaining * 0.60
        # rough guess: each L‑BFGS iteration (gradient+line‑search)
        # costs ~1 ms → allow up to 500 k iterations in total
        # we translate this into max iterations per β value

    # 4. Long sub‑gradient polishing (exact surrogate)
    # --------------------------------------------------------------
    if time.time() < deadline:
