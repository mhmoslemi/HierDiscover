"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm:

1. Build a few diverse seed vectors (uniform, power‑law, Gaussian,
   Dirichlet draws and an interpolated copy of the previous solution,
   if any) and keep the best according to the cheap surrogate
   ``2 * n * max(conv)`` (the sum is normalised to 1).

2. Refine the best seed with a *continuation Adam* optimiser in the
   log‑softmax (mirror‑descent) space.  The loss is the smooth
   log‑sum‑exp approximation of the max‑convolution, with a schedule of
   increasing temperature parameter β.  Adam’s adaptive steps help to
   escape shallow local minima and to converge faster than the older
   plain exponentiated‑gradient + L‑BFGS pipeline.

3. A short exact exponentiated‑gradient polish (sub‑gradient of the
   active convolution entry) is performed to squeeze out any remaining
   slack.

The whole routine respects the given time budget (default 1000 s) and
stores the best found sequence in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple analytic seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """Monotonically decreasing power‑law shape."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle."""
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw – a random point on the probability simplex."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential draw – heavy‑tailed random simplex point."""
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) + gradient
# ----------------------------------------------------------------------
def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    For logits v (log‑softmax variables) compute
        obj = 2·n·(max + log‑sum‑exp / β)
    and the gradient w.r.t. v.
    """
    a = _softmax(v)                       # probability vector
    conv = _conv_fft(a, a)                 # full convolution, length 2n‑1
    shift = np.max(conv)                   # stabilising shift
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smoothed objective
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # probability weights over the convolution entries
    w = exp_beta / sum_exp                 # length 2n‑1, sums to 1

    # gradient w.r.t. a : 2 * conv_rev(w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice that corresponds to ∂/∂a_i

    # chain rule through the soft‑max mapping
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return obj, grad_v


# ----------------------------------------------------------------------
# Continuation Adam optimiser (log‑softmax space)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iter_per_beta: int = 20_000):
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best probability vector found and its surrogate value.
    """
    if beta_schedule is None:
        # increasing temperature (more focus on the true max)
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]

    # initialise logits, Adam state and the best‑so‑far record
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iter_cnt = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        # a small base learning rate – Adam will adapt it
        base_lr = 0.03

        for _ in range(max_iter_per_beta):
            if time.time() > deadline:
                break

            obj, grad_v = _smooth_obj_grad(v, n, beta_val)

            # Adam updates
            iter_cnt += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** iter_cnt)
            v_hat = v2 / (1.0 - beta2 ** iter_cnt)

            v = v - base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep track of the best surrogate value seen so far
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the active entry)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the **exact** max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / np.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly 70 % of the remaining budget to Adam
        adam_deadline = time.time() + remaining * 0.70
        seq_adam, val_adam = _adam_optimize(best_seed,
                                            n,
                                            adam_deadline,
                                            max_iter_per_beta=30_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seed = seq_adam
            print(f"[Adam] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final exact EG polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        final_deadline = time.time() + remaining - 0.5   # tiny safety gap
        seq_final, val_final = _exp_gradient_opt(best_seed,
                                                 n,
                                                 final_deadline,
                                                 max_iters=400_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_val = val_final
            best_seed = seq_final
            print(f"[Final EG] Further improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. True evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seed.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
