"""
Propose a candidate step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The previous implementation used exponentiated‑gradient on the full FFT‑based
convolution.  Here we replace the long gradient phase by a fast
coordinate‑wise simulated‑annealing / hill‑climbing that updates the
convolution incrementally (O(n) per move instead of O(n log n)).  This
allows many more elementary steps within the 1000 s wall‑clock budget,
giving the optimiser a much larger search space and typically a lower
objective value.

The algorithm proceeds as follows:

1. **Initialisation** – start from the best known sequence (global variable
   ``height_sequence_1``) if it exists, otherwise from the uniform distribution.
2. **Full convolution** – compute the exact linear convolution once.
3. **Simulated annealing / greedy hill‑climbing** – repeatedly transfer a
   small amount of mass δ from a source index *i* to a target index *j*.
   The effect on the convolution is updated analytically (no FFT).  Moves that
   lower the surrogate `2·n·max(conv)` are always accepted; uphill moves are
   accepted with a temperature‑dependent probability.
   The temperature and the step‑size decay linearly with elapsed time.
   Every ``resync`` iterations we recompute the convolution from scratch and
   renormalise the vector to avoid accumulation of floating‑point errors.
4. **Polishing** – after the annealing phase we run a short exponentiated‑gradient
   (mirror‑descent) optimisation using the original FFT‑based routine to
   fine‑tune the final vector.
5. **Final evaluation** – the best sequence found is stored in the global
   variable ``height_sequence_1`` and returned as a plain Python list.

All operations respect the 1000 s wall‑clock limit; the final call to
``evaluate_sequence`` is performed a few seconds before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------
def _sa_optimize(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 resync: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Fast coordinate‑wise simulated annealing.
    Returns the best (seq, surrogate) encountered.
    """
    rng = np.random.default_rng(123456)

    # current state
    a = start_seq.copy()
    conv = np.convolve(a, a)          # exact linear convolution (length 2n‑1)
    cur_val = float(2.0 * n * np.max(conv))

    best_a = a.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    # annealing schedule constants
    T_start = max(cur_val * 0.1, 1e-3)   # a modest initial temperature
    T_end = 1e-8
    step_start = 0.5                    # up‑to 50 % of source mass may be moved initially
    step_end = 0.001

    # pre‑allocate a view of the whole vector for speed
    full_range = np.arange(n, dtype=np.int64)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time

        # linear cooling / step‑size decay
        T = T_start * (1.0 - progress) + T_end * progress
        step_factor = step_start * (1.0 - progress) + step_end * progress

        # --------------------------------------------------------------
        # propose a move: transfer δ from i (source) to j (target)
        # --------------------------------------------------------------
        i = rng.integers(0, n)                     # source index
        if a[i] <= 1e-15:        # source empty → retry
            continue

        j = rng.integers(0, n - 1)                 # target index (different)
        if j >= i:
            j += 1

        # choose a random amount – never exceed the source mass
        delta = step_factor * a[i] * rng.random()
        if delta <= 0.0:
            continue

        old_ai = a[i]
        old_aj = a[j]

        # ------------------------------------------------------------------
        # Incremental convolution update (uses the *old* a vector)
        # ------------------------------------------------------------------
        # slice contributions from the source (i) and target (j)
        conv[i:i + n] -= 2.0 * delta * a               # a still holds old values
        conv[j:j + n] += 2.0 * delta * a

        # quadratic self‑terms from the rank‑1 perturbation
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # ------------------------------------------------------------------
        # Apply the mass transfer to the sequence
        # ------------------------------------------------------------------
        a[i] = old_ai - delta
        a[j] = old_aj + delta

        # ------------------------------------------------------------------
        # Evaluate new surrogate
        # ------------------------------------------------------------------
        new_val = float(2.0 * n * np.max(conv))

        # ------------------------------------------------------------------
        # Acceptance test (Metropolis)
        # ------------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                if it % 5_000 == 0:
                    print(f"[SA] iter={it:7d} best={best_val:.6f} cur={cur_val:.6f}")
        else:
            # --------------------------------------------------------------
            # Revert the move (restore a and conv)
            # --------------------------------------------------------------
            # restore the probability vector
            a[i] = old_ai
            a[j] = old_aj

            # undo the convolution update – now a again contains the original values
            conv[i:i + n] += 2.0 * delta * a
            conv[j:j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta

        it += 1

        # ------------------------------------------------------------------
        # Periodically re‑synchronise to avoid drift (re‑compute full convolution)
        # ------------------------------------------------------------------
        if it % resync == 0:
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
                a_sum = 1.0
            else:
                a /= a_sum                     # enforce exact ℓ₁‑norm 1
            conv = np.convolve(a, a)
            cur_val = float(2.0 * n * np.max(conv))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA‑resync] iter={it:7d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to the reference implementation)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[EG]   it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and stores the best
    sequence found in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – the reference solution used 4000, we keep the same.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous construction if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            a0 = arr / arr.sum()
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1. Fast simulated‑annealing / hill‑climbing optimisation.
    # ------------------------------------------------------------------
    print("\n--- Simulated‑annealing phase (fast incremental updates) ---")
    seq_sa, val_sa = _sa_optimize(a0, n, deadline)
    print(f"[SA]   best surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 2. Short exponentiated‑gradient polishing (using exact FFT).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:                     # give at least a few seconds to the EG phase
        max_iters = int(min(500_000, remaining * 3000))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters≈{max_iters}) ---")
        seq_eg, val_eg = _exp_gradient_opt(seq_sa, n, deadline, max_iters=max_iters)
        if val_eg < val_sa:
            seq_sa, val_sa = seq_eg, val_eg
            print(f"[EG]   improved surrogate = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 3. Final official evaluation.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_sa.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_sa:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose the best sequence for the next round
    globals()["height_sequence_1"] = seq_sa.tolist()
    return seq_sa.tolist()
