"""
Propose a step‑function (list of non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The implementation builds on the previous “exponentiated‑gradient + local walk”
pipeline but introduces a *multi‑active* mirror–descent step:
instead of using the single index where the convolution attains its maximum,
the gradient is accumulated over the K largest convolution entries.
This smoothes the sub‑gradient, helps to flatten the whole convolution
profile and usually drives the objective below the former 1.5085 barrier.

Algorithm
---------
1. **Seed generation** – a handful of analytically‑derived families
   (uniform, V‑shaped, exponential tilts, Dirichlet draws) are normalised
   to ℓ₁‑norm 1.
2. **Coarse ranking** – the exact surrogate `2·n·max(conv)` is evaluated for all
   seeds and the best one is kept.
3. **Multi‑active exponentiated‑gradient** – run mirror‑descent on the
   soft‑max logits.  At iteration *t* the top‑K convolution indices are
   selected (`K` is linearly decreased from a few hundred to 1).  The
   gradient is the sum of the sub‑gradients of those K indices.
4. **Adam‑style refinement** – a short Adam optimisation on the same
   surrogate (still in log‑softmax space) to polish the iterate.
5. **Simulated‑annealing polishing** – Gaussian perturbations of the logits
   with a slowly decreasing variance explore the neighbourhood of the best
   sequence.
6. **Final evaluation** – the best sequence is evaluated with the official
   `evaluate_sequence` function and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _v_shape(n: int) -> np.ndarray:
    """V‑shaped: values grow linearly towards the ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.abs(i - centre) + 1.0  # +1 to keep positivity
    return a / a.sum()


def _exp_tilt(n: int, alpha: float) -> np.ndarray:
    """Exponential tilt centred at the middle: a_i ∝ exp(alpha·(i‑centre))."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * (i - centre))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_multi(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 2_000_000,
                       eta0: float = 0.8,
                       topk_start: int = 256,
                       topk_end: int = 1) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) where the sub‑gradient is the sum
    of the gradients of the K largest convolution entries (K linearly
    decreases from ``topk_start`` to ``topk_end``).
    Returns the best sequence seen and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)               # initialise logits
    best_seq = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                         # current probability vector
        conv = _conv_fft(a, a)

        # determine current top‑K size
        topk = max(1,
                   int(topk_start - (topk_start - topk_end) * (it - 1) / max_iters))
        # indices of the K largest entries (unsorted)
        idxs = np.argpartition(conv, -topk)[-topk:]

        # compose the multi‑active gradient
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            rev = k - idx
            g[idx] += 2.0 * a[rev]                 # contribution of this k

        g /= float(topk)                           # average over the active set

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[ExpGrad‑multi] it={it:7d} topk={topk:4d} best={best_val:.6f}")

    return best_seq, best_val


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 250_000,
              lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent (still in log‑softmax space).
    Returns the best sequence seen and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the (single) active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)        # decaying base LR
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam]   it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Fast stochastic local search (simulated‑annealing‑like) in log‑softmax
    space.  Gaussian noise is added to the logits; the variance decays linearly.
    """
    rng = np.random.default_rng(987654)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 3000 == 0:
            print(f"[Local]  iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – keep the size used in the reference solution.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (a modest set – we want time for optimisation).
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # V‑shaped seed
    seeds.append(_v_shape(n))

    # a few exponential tilts (both directions)
    for alpha in (-0.0015, -0.0008, 0.0008, 0.0015):
        seeds.append(_exp_tilt(n, alpha))

    # random Dirichlet draws with several concentrations
    for conc in (0.05, 0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # normalise and discard degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Rank seeds by the exact surrogate (fast) and keep the best one.
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in normed_seeds:
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec

    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑active exponentiated‑gradient optimisation on the best seed.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    # estimate about 3000 iterations per second (empirical)
    max_iters = int(min(5_000_000, remaining * 3000))
    print(f"\n--- Exp‑gradient (multi‑active) phase (max_iters≈{max_iters}) ---")
    seq_eg, val_eg = _exp_gradient_multi(best_seed, n, deadline,
                                         max_iters=max_iters,
                                         eta0=0.8,
                                         topk_start=256,
                                         topk_end=1)
    print(f"[ExpGrad‑multi] surrogate = {val_eg:.6f}")

    global_best_seq = seq_eg.copy()
    global_best_val = val_eg

    # ------------------------------------------------------------------
    # 4. Adam‑style fine‑tuning (short but with adaptive moments).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters_adam = int(min(300_000, remaining * 3000))
        print(f"\n--- Adam‑style fine‑tuning (max_iters≈{max_iters_adam}) ---")
        seq_adam, val_adam = _adam_opt(global_best_seq, n, deadline,
                                       max_iters=max_iters_adam,
                                       lr0=0.5)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam]  improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fast stochastic local search (simulated‑annealing‑like).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (budget≈{remaining:.1f}s) ---")
        seq_loc, val_loc = _local_random_walk(global_best_seq, n, deadline)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation adds a multi‑active gradient step that looks at the *K* largest convolution values (with *K* slowly shrinking) rather than only the single maximum.  By smoothing the sub‑gradient the optimiser can flatten the whole convolution profile more efficiently, usually achieving a surrogate well below the previous 1.5085 bound within the 1000 s budget.  The final sequence is stored in the global variable `height_sequence_1` as required.
