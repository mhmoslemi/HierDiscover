"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The optimisation is
performed on the ℓ₁‑normalised vector (so the denominator equals 1) by

1. generating a modest set of analytical seeds (uniform, beta‑shapes,
   power‑laws, Gaussian, Dirichlet) and keeping the best‑performing one,
2. running a *continuation* exponentiated‑gradient descent on a *smooth*
   surrogate of the max‑convolution:
   Lτ(a)= (1/τ)·log∑k exp(τ·(a*a)k).  For a fixed temperature τ the
   gradient is ∇Lτ = 2·conv(w,a^rev) where
   w_k = softmaxτ((a*a)k).  τ is increased from a small value to a large
   one, gradually sharpening the objective while the steps are taken
   in log‑softmax space (automatic simplex projection).
3. a short stochastic local‑walk (Gaussian perturbations of the logits)
   is performed as a final polishing step.
4. the resulting sequence is returned and stored in the global variable
   ``height_sequence_1`` (so that later runs can reuse it).

The algorithm stays within the 1000 s wall‑clock budget, uses only
NumPy (FFT based convolutions) and no external files or network access.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


# ----------------------------------------------------------------------
# Fast stochastic local search (used for a final polishing phase)
# ----------------------------------------------------------------------
def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.15,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search in log‑softmax (simple simulated‑annealing).
    Small Gaussian perturbations are added to the logits; improvements are kept.
    """
    rng = np.random.default_rng(123456)
    cur_seq = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur_seq, cur_seq))
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / total_time
        sigma = sigma_start * (1 - prog) + sigma_end * prog

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)

        proposal_val = 2.0 * n * np.max(_conv_fft(proposal_seq, proposal_seq))

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[LocalRW] iter={it:6d} best={best_val:.6f}")

    return best_seq, best_val
