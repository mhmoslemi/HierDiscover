def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – stays on the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate = 2·n·max(conv) for a sequence already ℓ₁‑normalised."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_window(n: int, sigma_factor: float = 0.30) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return a / a.sum()


def _triangular_shape(n: int) -> np.ndarray:
    """Linear ramp up then down (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = np.where(i <= half, i + 1, n - i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_permutation(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    p = rng.permutation(len(vec))
    return vec[p]

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                   # on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # Smooth max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # derivative of the smooth max term
            w = exp_beta / sum_exp                    # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]     # gradient wrt a

            # Back‑prop through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} – aborting further β steps.")
            break

    return _softmax(v)

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> np.ndarray:
    """Mirror‑descent on the exact max‑convolution (soft‑max parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)

    log_a = np.log(a + 1e-30)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # simple AdaGrad‑style decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best≈{best_val:.6f}")

    return best_seq

_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return sparse Toeplitz pattern for the convolution matrix (f * x)."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows, cols = [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows.append(np.full(length, k, dtype=np.int32))
        cols.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    offset = rows - cols          # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        s.t.         g ≥ 0,
                     (f * g)_k ≤ M   ∀k .
    Returns optimal g (already non‑negative) and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                     # coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                      # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        # fallback – keep the current vector
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _mix_with_optimal_t(f: np.ndarray, g: np.ndarray,
                       deadline: float, max_grid: int = 21) -> np.ndarray:
    """
    Find t∈[0,1] (grid search) that gives the smallest max‑convolution of
    (1‑t)·f + t·g.  The grid is coarse – we keep it cheap.
    """
    best_seq = f
    best_max = np.max(_conv_fft(f, f))
    ts = np.linspace(0.0, 1.0, max_grid)
    for t in ts:
        cand = (1.0 - t) * f + t * g
        cand = cand / cand.sum()          # safety – keep ℓ₁ = 1
        m = np.max(_conv_fft(cand, cand))
        if m < best_max:
            best_max = m
            best_seq = cand
    return best_seq


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 8) -> np.ndarray:
    """
    Perform several LP “mass‑inflation’’ iterations.
    After each LP we mix the current sequence with the LP‑solution using a
    tiny line‑search for the optimal convex‑combination coefficient.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 0.5 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            # LP could not add any ℓ₁‑mass – stationary point.
            break

        g_scaled = g / sum_g                # bring to ℓ₁ = 1
        new_seq = _mix_with_optimal_t(seq, g_scaled, deadline, max_grid=21)

        new_M = np.max(_conv_fft(new_seq, new_seq))
        if new_M < M - 1e-12:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            # no improvement – stop early
            break
    return seq

            perm = _random_permutation(base, rng)
            pool.append((perm, _objective_surrogate(perm, n), ('perm',)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising candidates (by exact surrogate)
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[1])
    top_k = 12
    elites = [vec for vec, _, _ in pool[:top_k]]
    print(f"Generated {len(pool)} seeds – keeping top {top_k} elites.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement of elite seeds
    # ------------------------------------------------------------------
    best_seq = elites[0].copy()
    best_val = _objective_surrogate(best_seq, n)
    print(f"[Init] best surrogate ≈ {best_val:.6f}")

    for idx, seed_vec in enumerate(elites):
        if time.time() > deadline:
            print("[Time] out before finishing all L‑BFGS refinements.")
            break
        print(f"\n--- L‑BFGS run on elite #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         max_iter_per_beta=10_000)
        val_opt = _objective_surrogate(seq_opt, n)
        print(f"[L‑BFGS] surrogate ≈ {val_opt:.6f}")
        if val_opt < best_val:
            best_seq, best_val = seq_opt.copy(), val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (exact max‑conv)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to EG (≈60 % of remaining seconds)
        max_iter = int(min(7_000_000, remaining * 0.6 * 1e5))
        print(f"\n--- EG polishing (max_iters={max_iter}) ---")
        seq_eg = _exp_gradient_opt(best_seq, n, deadline, max_iters=max_iter)
        val_eg = _objective_surrogate(seq_eg, n)
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print(f"[EG] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming “mass‑inflation’’ with line‑search mixing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- LP mass‑inflation refinement (with optimal t) ---")
        seq_lp = _lp_inflation_step(best_seq, deadline, max_outer_iters=6)
        val_lp = _objective_surrogate(seq_lp, n)
        if val_lp < best_val:
            best_seq, best_val = seq_lp.copy(), val_lp
            print(f"[LP] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Stochastic restarts – perturb + short L‑BFGS + EG
    # ------------------------------------------------------------------
    restarts = 3
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{3 - restarts} ---")
        # Gaussian perturbation in log‑space
        log_best = np.log(best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS (few βs, few iterations)
        seq_restart = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0],
                                             max_iter_per_beta=3000)
        val_restart = _objective_surrogate(seq_restart, n)

        if val_restart < best_val:
            # polish with a modest EG run
            seq_restart = _exp_gradient_opt(seq_restart, n, deadline,
                                            max_iters=250_000)
            val_restart = _objective_surrogate(seq_restart, n)
            if val_restart < best_val:
                best_seq, best_val = seq_restart.copy(), val_restart
                print(f"[Restart] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
