def _adam_optimize(start_vec, n, deadline,
                   beta_schedule, iters_per_beta,
                   lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    step = 0
    for beta in beta_schedule:
        for i in range(iters_per_beta):
            if time.time() > deadline:
                return _softmax(v)
            step += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_vals = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_vals)
            # gradient w.r.t a = 2 * conv_fft(rev_a, w)[n-1:n-1+n]
            w = exp_vals / sum_exp
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n-1:n-1+n]
            grad_v = a * (grad_a - np.dot(grad_a, a))
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1**step)
            v2_hat = v2 / (1 - beta2**step)
            adapt_lr = lr / (np.sqrt(v2_hat) + eps)
            v -= adapt_lr * m_hat
            # symmetrize
            v = (v + v[::-1]) / 2.0
    return _softmax(v)

def _subgradient_descent(start_seq, n, deadline, max_iters):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # step size schedule
        lr = 0.5 / np.sqrt(it)   # can be tuned
        a -= lr * g
        # project onto simplex
        a = _proj_simplex(a)
    return best_seq

"""
Improved stochastic‐gradient + Adam optimiser for the step‑function problem.

The algorithm proceeds in four phases (all respecting the 1000 s budget)

1. **Seed generation** – a large and diverse pool of ℓ₁‑normalised vectors built from
   analytic families (uniform, triangular, Gaussian, power‑law, sin², Beta)
   together with many Dirichlet draws.  The current best sequence (if any) is also
   added.

2. **Adam optimisation of a smooth max‑convolution surrogate** – each of the
   best *k* seeds is refined in the log‑softmax (simplex) parametrisation using
   an Adam optimiser on the log‑sum‑exp smooth approximation of ``max(conv(a,a))``.
   A long β‑schedule (up to 10⁶) is used, the learning‑rate is adapted per
   Adam‑step and the logits are symmetrised after every update.

3. **Projected sub‑gradient descent on the exact max‑convolution** – from the
   current global best a long sub‑gradient descent is run (projection onto the
   probability simplex) in order to push the true peak of the convolution lower.

4. **Final evaluation** – the best sequence found is evaluated with the official
   ``evaluate_sequence`` and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (surrogate assumes ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) pdf on {0,…,n‑1} – zero at the ends for α,β>1."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a[np.isnan(a)] = 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   iters_per_beta: int,
                   lr: float = 0.05) -> np.ndarray:
    """
    Refine ``start_vec`` (already ℓ₁‑normalised) using Adam on a
    log‑sum‑exp surrogate of ``max(conv(a,a))``.  The optimisation is
    performed in the log‑softmax (log‑probability) parametrisation so that
    the ℓ₁‑constraint is automatically satisfied.
    """
    # initialise logits
    v = np.log(start_vec + 1e-30)

    m = np.zeros_like(v)            # first moment
    v2 = np.zeros_like(v)           # second moment
    step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return _softmax(v)

            step += 1
            a = _softmax(v)                      # stays on simplex
            conv = _conv_fft(a, a)                # full convolution
            max_c = np.max(conv)
            shift = max_c                         # numeric stabilisation
            exp_vals = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_vals)
            w = exp_vals / sum_exp                # soft weights for the max

            # gradient wrt a (see derivation in the comment block above)
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # chain‑rule to logits
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # Adam update (per‑coordinate learning‑rate)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** step)
            v2_hat = v2 / (1.0 - beta2 ** step)

            adapt_lr = lr / (np.sqrt(v2_hat) + eps)
            v -= adapt_lr * m_hat

            # enforce symmetry – the optimum is known to be symmetric
            v = (v + v[::-1]) * 0.5

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)) for ℓ₁‑normalised vectors.
    Returns best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size (tuned a little larger than the original)
        lr = 0.5 / math.sqrt(it)
        a -= lr * g

        # projection onto probability simplex
        a = _proj_simplex(a)

    return best_seq, best_val

# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,

    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated time budget.

    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds for the final safe evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off of granularity vs speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")

    if previous is not None:

                prev_arr = tmp / tmp.sum()
        except Exception:
            prev_arr = None

    seeds: list[np.ndarray] = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.12, 0.18, 0.25):

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_power(n, p))
    seeds.append(_sin2(n))

    # Beta‑distribution shapes (α,β) both >0; those with α,β>1 are zero at the ends
    for (a, b) in ((0.5, 0.5), (0.5, 1.5), (1.5, 0.5), (2.0, 2.0), (2.0, 5.0), (5.0, 2.0)):
        seeds.append(_beta_dist(n, a, b))

    # Random Dirichlet draws with a spread of concentration parameters
    conc_list = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in conc_list:
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                         # raw draw
            seeds.append(np.sort(raw)[::-1])          # sorted descending
            seeds.append(rng.permuted(raw))           # random permutation

    # include previous best if available
    if prev_arr is not None:

    # ------------------------------------------------------------------

        if s > 0.0:

        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # a long β‑schedule; higher β sharpens the log‑sum‑exp approximation

        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,

        # aim for roughly 15 000 Adam steps per seed if time permits

        val_opt = _objective_surrogate(seq_opt)

        if val_opt < global_best_val:

            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # 5. Long projected sub‑gradient descent (exact max‑conv)
    # ------------------------------------------------------------------

                                                n,

                global_best_val = val_sub

    # ------------------------------------------------------------------

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
