"""
search for step heights using multi-start Entropic gradient + Adam + L‑BFGS.
"""

import time, math, numpy as np
try:
    import scipy.optimize as _opt
except Exception:
    _opt = None

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(v, 1.0 / len(v))

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s

def _random_dirichlet(n, conc, rng):
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s

def _random_exponential(n, rng):
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s

def _lowfreq_curve(n, num_freq, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _greedy_sidon_set(n, rng):
    # generate a random permutation of the integers 0..n-1 and greedily add.
    order = rng.permutation(n)
    sums = np.zeros(2 * n - 1, dtype=bool)
    support = []
    for x in order:
        # check if any sum x + y already used
        conflict = False
        for y in support:
            if sums[x + y]:
                conflict = True
                break
        if not conflict:
            # also check self sum
            if sums[2 * x]:
                continue
            # add
            support.append(x)
            for y in support:
                sums[x + y] = True
                sums[y + x] = True
            sums[2 * x] = True
    return np.array(support, dtype=int)

def _sidon_seed(n, rng, tries=5):
    # build a Sidon support and fill with random Dirichlet weights.
    best_seq = None
    best_val = np.inf
    for _ in range(tries):
        supp = _greedy_sidon_set(n, rng)
        m = supp.size
        if m == 0:
            continue
        # assign random Dirichlet weights on the support
        a = np.zeros(n, dtype=np.float64)
        weights = rng.dirichlet(np.full(m, 0.5)).astype(np.float64)
        a[supp] = weights
        # renormalise (should already sum to 1)
        a /= a.sum()
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a
    return best_seq

def _exp_gradient_opt(start_seq, n, deadline, max_iters=500_000, eta0=5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
    return best_seq, best_val

def _lbfgs_smooth_optimize(start_vec, n, deadline,
                           beta_schedule=None,
                           max_iter_per_beta=20000):
    if _opt is None:
        raise RuntimeError("SciPy optimiser not available")
    if beta_schedule is None:
        beta_schedule = [0.1, 0.5, 1, 2, 5, 10, 20, 50, 100,
                         200, 500, 1000, 2000, 5000, 10000,
                         20000, 50000, 100000, 200000, 500000,
                         1000000]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # log‑sum‑exp smoothing
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            w = exp_beta / sum_exp
            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print("[L‑BFGS] exception", e)
            break
    best_a = _softmax(v)
    return best_a

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The algorithm builds a rich pool of seeds,
    keeps the top few, runs a cheap exponentiated‑gradient polishing
    from each of them and finishes with a continuation L‑BFGS optimisation
    on a smooth max‑convolution surrogate.  The best sequence found
    (stored in the global ``height_sequence_1``) is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000  # length of the step function

    # ------------------------------------------------------------------
    # 1. Generate a wide variety of seeds
    # ------------------------------------------------------------------
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Analytic windows
    for gamma in np.linspace(0.2, 3.0, 12):
        # reuse sine_window defined earlier (omitted here for brevity)
        i = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
        a /= a.sum()
        seeds.append(a)

    # Random Dirichlet with a range of concentrations (including very sparse)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sidon‑set based seeds (few attempts, each with random weights)
    for _ in range(3):
        seq = _sidon_seed(n, rng, tries=3)
        if seq is not None:
            seeds.append(seq)

    # Re‑use previous solution if present
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (exact surrogate) and keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(7, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Seed evaluation finished – best surrogate among seeds = {evaluated[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑start optimisation
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = np.inf

    # each start gets a slice of the remaining time
    remaining_time = deadline - time.time()
    per_start_budget = remaining_time / (top_k + 1)  # +1 for final pass

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising start #{idx+1}/{top_k} ---")
        start_deadline = time.time() + per_start_budget * 0.7  # 70% of its slice for EG

        # 3a. Exponentiated‑gradient polishing (fast, cheap)
        seq_eg, val_eg = _exp_gradient_opt(seed_vec, n, start_deadline,

"""
Multi‑start optimisation of a non‑negative step‑function.

*   A large pool of analytic and random seeds is built.
*   The best few seeds are taken and each of them is polished with
    a cheap exponentiated‑gradient (mirror‑descent) run.
*   A continuation L‑BFGS optimiser works on a smooth
    max‑convolution surrogate (log‑sum‑exp) – a short run per seed,
    followed by a longer final run on the current global best.
*   The best sequence is stored in the global variable
    ``height_sequence_1`` and returned.

The optimiser respects a 1000 s wall‑time budget, printing progress and
the true value obtained from ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0 else np.full_like(v, 1.0 / len(v))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
       2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Sidon‑set helper (sparse support with almost unique pairwise sums)
# ----------------------------------------------------------------------
def _greedy_sidon_set(n: int, rng: np.random.Generator) -> np.ndarray:
    """Greedy construction of a Sidon set in {0,…,n‑1}."""
    order = rng.permutation(n)
    used = np.zeros(2 * n - 1, dtype=bool)
    support = []
    for x in order:
        # check collisions with current support
        conflict = False
        for y in support:
            if used[x + y]:
                conflict = True
                break
        if conflict or used[2 * x]:
            continue
        # accept x
        for y in support:
            used[x + y] = True
            used[y + x] = True
        used[2 * x] = True
        support.append(x)
    return np.asarray(support, dtype=int)


def _sidon_seed(n: int, rng: np.random.Generator, tries: int = 4) -> np.ndarray | None:
    """Create a few seeds with a Sidon support and random Dirichlet weights."""
    best_seq = None
    best_val = np.inf
    for _ in range(tries):
        supp = _greedy_sidon_set(n, rng)
        if supp.size == 0:
            continue
        # random positive weights on the support
        w = rng.dirichlet(np.full(supp.size, 0.5)).astype(np.float64)
        a = np.zeros(n, dtype=np.float64)
        a[supp] = w
        a /= a.sum()
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a
    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution surrogate with L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm builds a diverse pool of seeds,
    runs a cheap exponentiated‑gradient polishing from each of the best
    seeds, and finishes with a continuation L‑BFGS optimiser on a
    smoothed max‑convolution surrogate.  The best sequence found is stored
    in the global variable ``height_sequence_1`` and returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    n = 4000                                        # length of the step function

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # random Dirichlet (very sparse to moderately dense)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sidon‑set based seeds (few attempts)
    for _ in range(3):
        seq = _sidon_seed(n, rng, tries=3)
        if seq is not None:
            seeds.append(seq)

    # previous solution, if compatible
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (exact surrogate) and keep the top‑k
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:

        val = _objective_surrogate(vec)

    top_k = min(7, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

        eg_deadline = time.time() + per_start * 0.65

        else:

        print("\n--- Final L‑BFGS fine‑tuning ---")
        long_beta = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,

    except Exception as exc:  # pragma: no cover
