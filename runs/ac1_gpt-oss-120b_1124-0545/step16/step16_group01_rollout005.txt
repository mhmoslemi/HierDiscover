def _simulated_annealing_improve(seq, n, deadline):
    logs = np.log(seq + 1e-30)
    cur_seq = seq
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline:
        it += 1
        # schedule: progress = it / max_iters approx; but we don't know max_iters. Instead use temperature decaying with elapsed time.
        elapsed = time.time() - start_time; total = deadline - start_time
        progress = elapsed / total
        T = max(1e-6, 0.1 * (1 - progress))
        sigma = max(1e-5, 0.2 * (1 - progress))
        proposal_logs = logs + sigma * np.random.randn(n)
        proposal_seq = _softmax(proposal_logs)
        prop_val = _objective_surrogate(proposal_seq)
        delta = prop_val - cur_val
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            logs = proposal_logs
            cur_seq = proposal_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        # optionally perform occasional LBFGS polishing when improvement > threshold.
        if it % 50000 == 0 and time.time() < deadline:
            # attempt L-BFGS smoothing from current best
            seq_opt = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                             beta_schedule=[1,5,20,80,200,500,1_000,2_000,5_000,20_000,50_000,100_000,200_000,500_000,1_000_000],
                                             max_iter_per_beta=2000)
            val_opt = _objective_surrogate(seq_opt)

def _adam_optimize(start_seq, n, deadline, beta_schedule, iters_per_beta):

    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline: break
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / np.sum(exp_beta)   # soft weights
            # grad wrt a:
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update:
            m = beta1 * m + (1-beta1) * grad_v
            v_hat = beta2 * v_hat + (1-beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat_hat = v_hat / (1 - beta2**t)

            lr = 0.05 / (1 + 0.001 * t)   # decreasing LR
            v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)

            # evaluate surrogate occasionally
            if t % 5000 == 0:
                cur_a = _softmax(v)
                cur_val = _objective_surrogate(cur_a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_a.copy()
        if time.time() > deadline:
            break
    return best_seq, best_val

"""
Improved search for a non‑negative step‑function that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm consists of four phases (all within the 1000 s budget):

1. **Rich seed pool** – deterministic families (uniform, triangular,
   Gaussian, power‑law, U‑shaped, exponential) + many Dirichlet draws.
   Seeds are ℓ₁‑normalised and ranked by the cheap surrogate
   ``2·n·max(conv)`` (exact for ℓ₁‑normalised vectors).

2. **Adam optimisation on a smooth max‑convolution** – the non‑smooth
   ``max`` is approximated by a log‑sum‑exp with an increasing temperature
   (β‑schedule).  Adam (adaptive moments) is run directly in log‑softmax
   space, projecting automatically onto the probability simplex.

3. **Projected sub‑gradient descent** – fine‑grained descent on the true
   ``max(conv)`` (exact, non‑smooth) from the best Adam result.

4. **Final simulated‑annealing polish** – a short random‑walk in logit
   space to escape any remaining local minima.

The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by the official evaluator."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay from the edges toward the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution: large at the ends, small in the middle."""
    i = np.arange(n, dtype=np.float64) + 1.0
    rev = n - i + 1.0
    a = (i ** -p) + (rev ** -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   iters_per_beta: int = 2500) -> np.ndarray:
    """
    Adam on the log‑sum‑exp (smooth max) surrogate.
    Parameters
    ----------
    start_seq : np.ndarray
        ℓ₁‑normalised starting vector.
    n : int
        Length of the sequence.
    deadline : float
        Absolute time after which optimisation stops.
    beta_schedule : list[float] | None
        List of β values (temperature) – larger β → tighter approximation.
    iters_per_beta : int
        Upper bound on iterations per β value (subject to time budget).
    Returns
    -------
    np.ndarray
        The best sequence (still ℓ₁‑normalised) found during the run.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)          # first‑moment
    v_hat = np.zeros_like(v)      # second‑moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1

            a = _softmax(v)                     # stays on the simplex
            conv = _conv_fft(a, a)               # full conv
            max_c = np.max(conv)
            # numerically stable softmax over conv values
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp               # soft‑weights

            # gradient of the smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain‑rule through softmax

            # Adam update (in logit space)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat_hat = v_hat / (1.0 - beta2 ** t)

            # simple learning‑rate schedule, decaying with t
            lr = 0.05 / (1.0 + 0.001 * t)
            v = v - lr * m_hat / (np.sqrt(v_hat_hat) + eps)

            # occasional evaluation of the (exact) surrogate
            if t % 5000 == 0:
                cur_a = _softmax(v)
                cur_val = _objective_surrogate(cur_a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_a.copy()

    return best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000) -> np.ndarray:
    """
    Long sub‑gradient descent on the exact max‑convolution.
    Returns the best sequence obtained before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry (see original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        # Euclidean projection onto the probability simplex
        # (the classical O(n log n) algorithm)
        # – identical to _proj_simplex from the reference code.
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a = np.maximum(a - theta, 0.0)

    return best_seq


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (logit‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 250_000) -> np.ndarray:
    """
    Random walk in log‑softmax space with a simple cooling schedule.
    """
    rng = np.random.default_rng()
    logits = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling – starts hot, ends cold
        progress = it / max_iters
        T = max(1e-6, 0.07 * (1.0 - progress))
        sigma = max(1e-5, 0.15 * (1.0 - progress))

        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        prop_val = _objective_surrogate(proposal_seq)

        delta = prop_val - cur_val
        if delta < 0 or math.exp(-delta / T) > rng.random():
            logits = proposal_logits
            cur_seq = proposal_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq
