def _pairwise_refine(seq, n, deadline, rng, max_iters=500000):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    v = np.log(a + 1e-30)  # for stability but we will modify a directly
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to transfer = epsilon * a[i]; epsilon small random
        eps = rng.uniform(0, 0.01)  # up to 1% of mass at i
        delta = eps * a[i]
        if delta == 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegativity: a_new[i] >= 0; it is because delta <= a[i]
        # compute surrogate
        val_new = _objective_surrogate(a_new)
        if val_new < best_val:

"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

Algorithm Overview
------------------
1️⃣ **Rich Seed Generation** – many analytical families (uniform,
    exponential, reverse‑exponential, beta, power‑law, sinusoid,
    quadratic, arcsine) and a broad set of random Dirichlet / Gaussian‑mixture
    draws.  All seeds are ℓ₁‑normalised (sum = 1) so the surrogate
    objective equals ``2·n·max(conv)``.

2️⃣ **L‑BFGS continuation** – a smooth log‑sum‑exp surrogate of the max‑convolution
    is minimised from the most promising seeds.  The β‑schedule ends at 1e6,
    giving an extremely close approximation of the true max.

3️⃣ **Mirror‑descent polishing** – an exponentiated‑gradient (EG) optimiser on the
    exact non‑smooth objective (sub‑gradient of the active convolution entry).

4️⃣ **Simulated‑annealing hill‑climbing** – perturb the soft‑max logits with
    Gaussian noise, accepting downhill moves always and uphill moves with a
    temperature‑controlled probability.

5️⃣ **Pairwise mass‑transfer refinement** – repeatedly move a tiny amount of mass
    from a random index to another and keep the move only if it improves the
    exact surrogate.

All phases respect the global 1000 s wall‑clock budget.  The best sequence is
stored in ``height_sequence_1`` for possible reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a ℓ₁‑normalised sequence (i.e. 2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """decreasing exponential a_i ∝ r**i,  r = exp(-c/n)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """increasing exponential a_i ∝ r**(n-1-i)."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_family(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _quadratic_family(n: int) -> np.ndarray:
    """Parabolic shape a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_family(n: int) -> np.ndarray:
    """Arcsine‑like shape a_i ∝ 1/√((i+½)*(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_gaussian_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a few Gaussian bumps."""
    k = rng.integers(2, 5)           # number of components
    centers = rng.integers(0, n, size=k)
    stds = rng.uniform(5.0, n / 8.0, size=k) + 1e-8
    weights = rng.random(k)
    weights /= weights.sum()
    xs = np.arange(n, dtype=np.float64)
    a = np.zeros_like(xs)
    for w, mu, sigma in zip(weights, centers, stds):
        a += w * np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """Optimise the smoothed max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
            2000.0, 5000.0, 10000.0, 20000.0, 50000.0,
            100000.0, 200000.0, 500000.0, 1e6
        ]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (using the soft‑max log‑sum‑exp)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.8,
                    max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb soft‑max logits with Gaussian noise; accept downhill moves always,
    uphill moves with a temperature‑controlled probability.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 2.0
    sigma_decay = 0.99997
    temp_decay = 0.99997
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 200_000:
            sigma = sigma0
            temperature = 2.0
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Pairwise mass‑transfer local refinement
# ----------------------------------------------------------------------
def _pairwise_refine(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny amount of probability mass from one index to another.
    Keep the move only if it strictly improves the exact surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # transfer up to 1 % of the source mass
        eps = rng.uniform(0.0, 0.01)
        delta = eps * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = _objective_surrogate(a_new)
        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new  # accept move

    return best_seq, best_val
