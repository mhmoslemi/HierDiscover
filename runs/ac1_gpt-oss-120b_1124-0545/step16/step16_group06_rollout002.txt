    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # compute gradient of max (subgradient)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        # gradient w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx-(n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule to logits
        grad_v = a * (g - np.dot(g, a))

        # Adam update
        m = beta1 * m + (1-beta1) * grad_v
        v_hat = beta2 * v_hat + (1-beta2) * (grad_v**2)

        # recompute a
        a = _softmax(v)

        cur_val = _objective_surrogate(a)
        if cur_val<best_val:
            best_val=cur_val
            best_seq=a.copy()
    return best_seq,best_val

"""
Propose a step‑function (list of non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The runner works in three phases:

1️⃣  **Diverse seed pool** – analytic families (uniform, exponential,
    reversed‑exponential, power‑law, beta), many random Dirichlet draws,
    and a few sparse “spike” vectors.  All seeds are ℓ₁‑normalised.

2️⃣  **Smooth L‑BFGS refinement** – minimises a log‑sum‑exp
    surrogate of the max‑convolution.  The β‑schedule is extended up to
    1e7 to tighten the approximation.

3️⃣  **Non‑smooth polishing** – an Adam‑style optimiser on the exact
    max‑convolution (sub‑gradient w.r.t. soft‑max logits) followed by a
    short projected sub‑gradient run.

All iterates stay on the probability simplex, so the exact objective is
`2·n·max(conv)`.  The routine respects the 1000 s wall‑clock budget, prints
progress, stores the best sequence in the global variable
`height_sequence_1`, and returns it as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:          # pragma: no cover

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised sequence (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective with L‑BFGS‑B.
    Returns a probability simplex vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS phase.")

    if beta_schedule is None:
        # increasingly tight approximations of the max
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 5e5, 1e6,
                         5e6, 1e7]

    # work in log‑softmax space
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # true max
            # smooth approximation
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in comment above)
            w = exp_beta / sum_exp                   # softmax‑weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:            # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑style optimiser on the exact max‑convolution (non‑smooth)
# ----------------------------------------------------------------------
def _adam_subgrad_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      init_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam on the exact max‑convolution (sub‑gradient via active shift)."""
    a = start_seq.copy()
    logits = np.log(a + 1e-30)          # parameters in log‑space
    m1 = np.zeros_like(logits)
    m2 = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # recompute a from logits (ensures sum=1)
        a = _softmax(logits)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule through softmax
        dot = np.dot(g, a)
        grad_logits = a * (g - dot)

        # Adam update (no bias‑correction for speed)
        m1 = beta1 * m1 + (1.0 - beta1) * grad_logits
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_logits ** 2)

        step = init_lr / math.sqrt(it)   # simple decay
        logits = logits - step * m1 / (np.sqrt(m2) + eps)

        cur_val = _objective_surrogate(_softmax(logits))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(logits).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on the exact objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient (ordered pairs) for active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated wall‑clock budget.
    Returns the best sequence (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a little larger than the previous run
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0.  Retrieve a previously saved solution (if any) and adapt it
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1.  Build a rich pool of seeds
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    pool = []

    # uniform baseline
    pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous solution (if shape matches)
    if prev_vec is not None:
        pool.append((prev_vec, _objective_surrogate(prev_vec), ('previous',)))

    # exponential families (coarse grid)
    for c in np.linspace(0.5, 5.0, 20):
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        pool.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        pool.append((rev, _objective_surrogate(rev), ('revexp', c)))

    # power‑law family
    for p in np.linspace(-0.5, 5.0, 30):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        pool.append((vec, _objective_surrogate(vec), ('power', p)))

    # beta‑shape family (moderate grid)
    alphas = np.linspace(0.1, 4.0, 9)
    betas = np.linspace(0.1, 4.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            pool.append((vec, _objective_surrogate(vec), ('beta', a, b)))

    # random Dirichlet draws with various concentrations
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(25):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # sparse random “spike” vectors
    for k in (5, 10, 20, 50):
        for _ in range(12):
            if time.time() > deadline:
                break
            pos = rng.choice(n, size=k, replace=False)
            w = rng.exponential(scale=1.0, size=k)
            vec = np.zeros(n, dtype=np.float64)
            vec[pos] = w
            vec /= vec.sum()
            pool.append((vec, _objective_surrogate(vec), ('sparse', k)))

    # ------------------------------------------------------------------
    # 2.  Keep the most promising seeds
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [vec for vec, _, _ in pool[:top_k]]
    print(f"Generated {len(pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3.  L‑BFGS refinement from each seed
    # ------------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for i, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run #{i+1}/{len(top_seeds)} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline,
                                         max_iter_per_beta=5000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  High‑β final L‑BFGS refinement (tightening the max‑approx)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- High‑β L‑BFGS refinement ---")
        high_beta_sched = [1e5, 5e5, 1e6, 5e6, 1e7]
        seq_refined = _lbfgs_smooth_optimize(global_best, n, deadline,
                                             beta_schedule=high_beta_sched,
                                             max_iter_per_beta=3000)
        val_refined = _objective_surrogate(seq_refined)
        print(f"[High‑β] surrogate = {val_refined:.6f}")
        if val_refined < global_best_val:
            global_best = seq_refined.copy()
            global_best_val = val_refined
            print(f"[Update] after high‑β run = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Adam‑style sub‑gradient polishing on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(300_000, remaining * 0.1 * 1e5))
        print(f"\n--- Adam sub‑gradient polishing (max_iters={max_adam_iters}) ---")
        seq_adam, val_adam = _adam_subgrad_opt(global_best, n, deadline,
                                               max_iters=max_adam_iters,
                                               init_lr=0.5)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Update] after Adam = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final short projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(200_000, remaining * 0.05 * 1e5))
        if max_sub_iters > 0:
            print("\n--- Final projected sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best, n, deadline,
                                                max_iters=max_sub_iters)
            print(f"[SubGrad] surrogate = {val_sub:.6f}")
            if val_sub < global_best_val:
                global_best = seq_sub.copy()
                global_best_val = val_sub
                print(f"[Update] after final sub‑gradient = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7.  Evaluate with the official evaluator and store the result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
