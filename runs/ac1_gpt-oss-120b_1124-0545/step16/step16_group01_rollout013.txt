def _adam_optimize(start_vec, n, deadline, beta_schedule, lr=0.1, max_iters_per_beta=5000):  

Initialize logit v = np.log(start_vec+1e-30).  

Initialize m = np.zeros_like(v), v_hat = np.zeros_like(v).  

For each beta in schedule:

   for iter in range(max_iters_per_beta):
      if time.time()>deadline: break

      a = softmax(v)
      conv = conv_fft(a, a)
      shift = np.max(conv)  # for numerical stability
      exp_beta = np.exp(beta * (conv - shift))
      sum_exp = np.sum(exp_beta)
      smooth = shift + np.log(sum_exp) / beta
      obj = 2 * n * smooth

      w = exp_beta / sum_exp  # weights
      rev_a = a[::-1]
      grad_a = 2 * conv_fft(rev_a, w)[n-1:n-1+n]   # as earlier

      # chain rule
      grad_v = a * (grad_a - np.dot(grad_a, a))

      # Adam update
      m = beta1 * m + (1-beta1) * grad_v
      v_hat = beta2 * v_hat + (1-beta2) * (grad_v ** 2)
      m_hat = m / (1 - beta1**t)
      v_hat_corr = v_hat / (1 - beta2**t)
      v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

      # optionally track best objective; store best a

 Within loop, we compute best surrogate value. Keep best a and best obj.

"""
Fast Adam‑based minimiser for the step‑function bound.

The program builds a few diverse ℓ₁‑normalised seeds (previous best,
uniform, Dirichlet draws).  Each seed is refined in the log‑soft‑max
parameter space by minimising a smooth approximation of
``max(conv(a,a))``:

    smooth_β(a) = max(conv)  ≈  shift + log(∑exp(β·(conv‑shift))) / β

with a continuation on β = 10 → 1 000 000.  Adam (adaptive
first‑order method) is used because it works well with the noisy
gradient that arises from the soft‑max weighting.  The best sequence
found by any run is kept, optionally polished by a short sub‑gradient
descent, and finally handed to the official ``evaluate_sequence``.
The whole optimisation respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for ℓ₁‑normalised vectors:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with a common concentration parameter."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# L‑BFGS optimiser (fallback – used only if SciPy is available)
# ----------------------------------------------------------------------

    import scipy.optimize as _opt

    _opt = None

    if _opt is None:
        raise RuntimeError("SciPy is not available – L‑BFGS cannot run.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         20_000.0, 50_000.0, 100_000.0,
                         200_000.0, 500_000.0, 1_000_000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                   # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑conv surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float) -> np.ndarray:
    """
    Adam optimiser in log‑soft‑max space.
    Uses a continuation on the smoothing parameter β.
    Returns the best sequence (ℓ₁‑normalised) discovered.
    """
    v = np.log(start_vec + 1e-30)          # logits
    m = np.zeros_like(v)                   # first moment
    v_m = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    pow1, pow2 = 1.0, 1.0
    t = 0

    best_seq = None
    best_val = np.inf

    # β schedule – start gentle, finish very sharp
    beta_schedule = [10, 50, 200, 800, 3000, 10_000,
                     30_000, 100_000, 300_000, 1_000_000]

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # a modest, fixed budget per β (the inner loop also checks the deadline)
        max_iters_beta = 4000

        for _ in range(max_iters_beta):
            if time.time() > deadline:
                break

            t += 1
            pow1 *= beta1
            pow2 *= beta2

            a = _softmax(v)                     # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)

            # smooth approximation of the max (log‑sum‑exp)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # not needed explicitly, but kept for completeness
            # smooth_max = shift + np.log(sum_exp) / beta

            # gradient of smooth max w.r.t. conv entries = soft‑weights
            w = exp_beta / sum_exp

            # gradient of weighted sum of conv entries w.r.t. a
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # chain rule: gradient w.r.t. logits
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_m = beta2 * v_m + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - pow1)
            v_hat = v_m / (1.0 - pow2)

            lr = 0.05
            v -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # track the best surrogate seen so far
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        print(f"[Adam] β={beta:<9} best surrogate={best_val:.6f}")

    # fall‑back in case nothing was accepted (should never happen)
    if best_seq is None:
        best_seq = _softmax(v)
    return best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv)
# ----------------------------------------------------------------------


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on the exact max‑convolution surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # build sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        # Euclidean projection onto the simplex
        a = _proj_simplex(a)

    return best_seq, best_val


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the
    ``evaluate_sequence`` objective within the time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem dimension – 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a small set of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1. previous best (if it matches the current length)
    previous = globals().get("height_sequence_1")
    if previous is not None and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            seeds.append(prev_arr / prev_arr.sum())

    # 2. plain uniform vector
    seeds.append(_uniform(n))

    # 3. a handful of Dirichlet draws with various concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # Main optimisation loop – keep the best sequence seen
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = np.inf

    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            print("[Time] budget exhausted before processing all seeds.")
            break

        print(f"\n--- Optimising seed #{idx + 1} ---")

        # ------------------------------------------------------------------
        # First pass – quick L‑BFGS if SciPy is present (helps a lot)
        # ------------------------------------------------------------------
        seq_opt = None
        try:
            if _opt is not None:
                seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                                 n,
                                                 deadline,
                                                 beta_schedule=[1, 5, 20, 80, 200,
                                                                500, 1_000, 2_000,
                                                                5_000, 20_000,
                                                                50_000, 100_000,
                                                                200_000, 500_000,
                                                                1_000_000],
                                                 max_iter_per_beta=3000)
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] skipped due to exception: {e}")

        # ------------------------------------------------------------------
        # Adam refinement (always run)
        # ------------------------------------------------------------------
        start_vec = seq_opt if seq_opt is not None else seed_vec
        seq_refined = _adam_optimize(start_vec, n, deadline)

        cur_val = _objective_surrogate(seq_refined)
        print(f"[Result] surrogate = {cur_val:.6f}")

        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = seq_refined.copy()
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # If we still have time, perform a longer sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        remaining = deadline - time.time()
        # estimate a reasonable iteration budget (roughly 2 000 it/s)
        max_sub_iters = int(min(2_000_000, remaining * 2000))
        if max_sub_iters > 0:
            print("\n--- Long sub‑gradient descent (final polish) ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – should never happen
        global_best_seq = _uniform(n)

    true_val = evaluate_sequence(global_best_seq.tolist())
    print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
          f"true evaluate_sequence = {true_val:.6f}")

    # store for a possible future run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
