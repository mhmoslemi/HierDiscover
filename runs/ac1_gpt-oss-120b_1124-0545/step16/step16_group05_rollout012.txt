"""
Propose a (near‑optimal) non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines a rich set of analytic seeds, aggressive L‑BFGS
refinement on a smooth log‑sum‑exp surrogate, long exponentiated‑gradient
( mirror‑descent ) polishing, a few linear‑programming “mass‑inflation’’
steps with a line‑search on the mixing factor, and a final stochastic
hill‑climbing phase.  All stages respect a user‑supplied time budget
( default 1000 s ).  The best sequence found is stored in the global
variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective when `seq` is ℓ₁‑normalised (sum(seq)==1):
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()

        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smoothed max (see original implementation)
            w = exp_beta / sum_exp                     # weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> np.ndarray:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    # Simple AdaGrad‑style step‑size
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)          # decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq


# ----------------------------------------------------------------------
# LP “mass‑inflation’’ step with line‑search on mixing factor
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) for the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Solve:
        maximise Σ g_j
        s.t.  g ≥ 0,
              (f * g)_k ≤ M, ∀k.
    Return optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximize Σ g → minimize -Σ g
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, float(g_opt.sum())


def _lp_mix_step(seq: np.ndarray,
                 rows: np.ndarray, cols: np.ndarray, offset: np.ndarray,
                 deadline: float,
                 max_outer_iters: int = 8) -> np.ndarray:
    """
    Repeatedly solve the LP and perform a line‑search over the convex
    mixing factor `t`.  Stops when no improvement is made or time runs out.
    """
    n = len(seq)
    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = float(conv.max())
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_scaled = g / sum_g                     # normalise to ℓ₁‑mass 1

        # try a few mixing factors and keep the best one
        best_seq = seq
        best_val = _objective_surrogate(seq)

        for t in (0.5, 0.3, 0.7, 0.2, 0.8, 0.1, 0.9):
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_val = _objective_surrogate(new_seq)
            if new_val < best_val:
                best_seq = new_seq
                best_val = new_val

        if best_val < _objective_surrogate(seq) - 1e-12:
            seq = best_seq
            print(f"[LP] improved surrogate to {best_val:.6f}")
        else:
            break
    return seq


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                deadline: float,
                steps: int = 200_000,
                rng: np.random.Generator = None) -> np.ndarray:
    """Randomly move tiny probability mass between coordinates."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of the smaller entry
        eps = min(best_seq[i], best_seq[j]) * rng.uniform(0.0, 0.05)
        if eps <= 0.0:
            continue
        new_seq = best_seq.copy()
        # move from i to j
        if rng.random() < 0.5:
            delta = eps
            if new_seq[i] < delta:
                continue
            new_seq[i] -= delta
            new_seq[j] += delta
        else:
            delta = eps
            if new_seq[j] < delta:
                continue
            new_seq[j] -= delta
            new_seq[i] += delta
        # renormalise (should stay sum‑1, but tiny drift possible)
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    print(f"[HillClimb] final surrogate {best_val:.6f}")
    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimizes `evaluate_sequence`.  The routine respects the supplied
    time budget and stores the best sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – we stay with n=4000 (good trade‑off between precision
    # and LP memory consumption)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load a previous solution, if any (may be useful as a seed)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list) and len(previous) == n:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.sum() > 0.0:
            prev_arr = arr / arr.sum()

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of seed vectors
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # baseline uniform
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # exponential families (rich sweep)
    for c in np.linspace(0.2, 12.0, 30):          # extended range
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        candidate_pool.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        candidate_pool.append((rev, _objective_surrogate(rev), ('revexp', c)))

    # beta shape sweep (broader)
    alphas = np.linspace(0.1, 8.0, 16)
    betas = np.linspace(0.1, 8.0, 16)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            vec = _beta_shape(n, a, b)
            candidate_pool.append((vec, _objective_surrogate(vec), ('beta', a, b)))

    # power‑law sweep
    for p in np.linspace(0.0, 3.0, 31):
        if time.time() > deadline:
            break
        vec = _power(n, p)
        candidate_pool.append((vec, _objective_surrogate(vec), ('power', p)))

    # random Dirichlet draws spanning many concentrations
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:
        for _ in range(30):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (by surrogate)
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement of elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] out before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None               # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Rough estimate: ~250‑300 k iterations per second on typical hardware
        iters_per_sec = 250_000
        max_exp = int(min(5_000_000, remaining * iters_per_sec))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp = _exp_gradient_opt(global_best_seq, n, deadline, max_iters=max_exp)
        val_exp = _objective_surrogate(seq_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming mass‑inflation (with line‑search on mixing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming mass‑inflation refinement ---")
        rows, cols, offset = _get_conv_pattern(n)
        seq_lp = _lp_mix_step(global_best_seq, rows, cols, offset, deadline,
                              max_outer_iters=6)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Stochastic restarts + short L‑BFGS + EG
    # ------------------------------------------------------------------
    restarts = 6
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{6 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_vec = _softmax(log_best + pert)

        # short L‑BFGS
        seq_restart = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0],
                                             max_iter_per_beta=3000)
        val_restart = _objective_surrogate(seq_restart)
        if val_restart < global_best_val:
            # brief EG polishing
            seq_restart = _exp_gradient_opt(seq_restart, n, deadline,
                                            max_iters=200_000)
            val_restart = _objective_surrogate(seq_restart)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final hill‑climbing phase (cheap random local moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climbing refinement ---")
        seq_hc = _hill_climb(global_best_seq, deadline, steps=300_000, rng=rng)
        val_hc = _objective_surrogate(seq_hc)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
