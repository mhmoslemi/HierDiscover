"""
Propose a step‑function (list of non‑negative heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three phases:

1. **Diverse seed generation** – a rich collection of analytically‑derived
   probability vectors (uniform, beta, power‑law, Gaussian, sine‑shape,
   raised‑cosine, Dirichlet draws, plus the previous best if available) is
   built and ℓ₁‑normalised.

2. **Accelerated exponentiated‑gradient descent** – the best seed is refined
   for the bulk of the time budget.  The sub‑gradient of the
   “max‑convolution’’ is computed from *all* active indices (those attaining
   the maximal convolution value) and a mirror‑descent step is performed in
   log‑softmax space with a 1/√t step‑size schedule.  The best iterate seen
   is retained.

3. **Annealing‑style stochastic polishing** – a fast local random walk in
   log‑space (Gaussian perturbations with a slowly decreasing variance) is
   applied to the best vector from phase 2, keeping any improvement.

All operations respect the global time budget (default 1000 s).  The final
vector is stored in the global variable ``height_sequence_1`` and returned as
a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _sin_shape(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power   (non‑negative on the whole range)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _raised_cosine(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ (1‑cos(2π·i/(n‑1)))^power."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** power
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (active‑set version)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns the sub‑gradient of  max(conv(a,a))  w.r.t. a
    (the gradient is taken w.r.t. the active indices only) and the
    maximal convolution value.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # tolerance to collect all (numerically) tied active positions
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of indices i for which a_i participates in (a*a)[k]
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # contribution 2·a_rev_idx for each i
        g[idx_i] += 2.0 * a[rev_idx]
    return g, max_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Works in log‑softmax space, thus automatically staying on the probability
    simplex.  Returns the best vector seen along with its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # stay in log‑space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g, _ = _subgradient_max_conv(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500_000 == 0:
                print(f"[EG] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fast stochastic local search (annealing‑like)
# ----------------------------------------------------------------------
def _local_random_walk(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    sigma_start: float = 0.15,
    sigma_end: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations of the log‑probabilities with a linearly
    decreasing standard deviation.  Only strictly improving moves are kept.
    """
    rng = np.random.default_rng(987654321)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 5_000 == 0:
            print(f"[Local] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (seconds) and stores the
    resulting sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – we keep n = 4000 as a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve the previous solution (if any) and normalise it.
    # ------------------------------------------------------------------
    previous_vec = None
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation (a modest but diverse set).
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(30):
        a = 10 ** rng.uniform(-1, 1)   # 0.1 … 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.08, 0.5)
        seeds.append(_gaussian(n, sigma))

    # Sine‑shape and raised‑cosine families
    for p in (0.5, 1.0, 2.0, 4.0):
        seeds.append(_sin_shape(n, p))
        seeds.append(_raised_cosine(n, p))

    # Random Dirichlet draws – use a spread of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Normalise and filter degenerate vectors
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        tot = s.sum()
        if tot > 0.0:
            normed_seeds.append(s / tot)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Coarse ranking – keep the top few seeds w.r.t. the surrogate.
    # ------------------------------------------------------------------
    seed_vals = [_objective_surrogate(v) for v in normed_seeds]
    # sort indices by surrogate value (ascending)
    sorted_idx = np.argsort(seed_vals)
    top_k = 5
    top_indices = sorted_idx[:top_k]
    top_seeds = [normed_seeds[i] for i in top_indices]
    print(f"Top {top_k} seed surrogate values:")
    for i, idx in enumerate(top_indices):
        print(f"  #{i+1}: {seed_vals[idx]:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation – long exponentiated‑gradient on the best seed.
    # ------------------------------------------------------------------
    # Use the *best* seed as the primary start point; we still keep the other
    # top seeds as cheap warm‑starts later if time permits.
    primary_start = top_seeds[0]
    remaining = max(0.0, deadline - time.time())
    # estimate ~2500 EG iterations per second (empirical)
    max_iters = int(min(8_000_000, remaining * 2500))
    print(f"\n--- EG phase (max_iters≈{max_iters:,}) ---")
    seq_eg, val_eg = _exp_gradient_opt(primary_start, n, deadline, max_iters=max_iters, eta0=1.0)
    print(f"[EG] final surrogate = {val_eg:.6f}")

    best_seq = seq_eg.copy()
    best_val = val_eg

    # ------------------------------------------------------------------
    # 4. Optional secondary warm‑starts (shorter EG runs) from the other
    #    top seeds, only if we still have a comfortable time margin.
    # ------------------------------------------------------------------
    for i, start_vec in enumerate(top_seeds[1:], start=2):
        if time.time() > deadline:
            break
        # give each secondary run a modest budget (≈10 % of the remaining time)
        sec_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.1)
        iters = int(min(500_000, (sec_deadline - time.time()) * 2500))
        print(f"\n--- Secondary EG #{i} (iters≈{iters:,}) ---")
        seq_sec, val_sec = _exp_gradient_opt(start_vec, n, sec_deadline, max_iters=iters, eta0=0.8)
        if val_sec < best_val:
            best_seq = seq_sec.copy()
            best_val = val_sec
            print(f"[Secondary EG #{i}] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fast stochastic local search (annealing‑like) polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"\n--- Local random‑walk polishing (budget≈{deadline - time.time():.1f}s) ---")
        seq_loc, val_loc = _local_random_walk(best_seq, n, deadline)
        if val_loc < best_val:
            best_seq = seq_loc.copy()
            best_val = val_loc
            print(f"[Local] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
