"""
Iterative active‑set linear programming for low‑autocorrelation step functions.

The algorithm works with an ℓ¹‑normalised non‑negative vector `f`.
Given the current maximum autocorrelation `M = max(f*f)`, it solves a small
linear program that maximises the total mass `∑ g_j` under the pointwise
constraints ` (f * g)[k] ≤ M ` for a selected *active* set of lag indices.
The LP solution `g` is then ℓ¹‑normalised and mixed with `f`.  The mixing
strictly reduces the maximal autocorrelation, and the process is repeated
until the improvement stalls or the time budget expires.

A brief final polishing with a smooth‑max L‑BFGS optimiser (log‑softmax
parameterisation) is performed if time permits.

All arithmetic respects the limits of ``evaluate_sequence`` (non‑negative,
≤ 1000, and ℓ¹‑normalised ⇒ exact objective = 2·n·max(conv)).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP + L‑BFGS)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog, minimize
except Exception:  # pragma: no cover
    linprog = None
    minimize = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for ℓ¹‑normalised vectors:
    2·n·max(conv(seq,seq)) == official evaluate_sequence value.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# L‑BFGS smooth‑max optimiser (used only as a final polish)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns a probability vector (ℓ¹‑normalised).
    """
    if minimize is None:
        raise RuntimeError("SciPy minimise required for L‑BFGS polishing.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 1e5,
            2e5, 5e5, 1e6
        ]

    v = np.log(start_vec + 1e-30)          # logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)          # stays on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp (stabilised)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a
            w = exp_beta / sum_exp                         # weights on conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                     # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Active‑set LP + mixing optimiser
# ----------------------------------------------------------------------
def _solve_lp_active_set(f: np.ndarray,
                         M: float,
                         active_set: list[int],
                         var_upper: float = 1000.0) -> np.ndarray:
    """Solve the LP  max ∑ g_j  s.t. (f*g)[k] ≤ M for k∈active_set , g≥0, g≤var_upper."""
    n = f.size
    # Build A_ub (len(active_set) × n)
    rows = np.zeros((len(active_set), n), dtype=np.float64)
    for r, k in enumerate(active_set):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        cols = np.arange(i_start, i_end + 1, dtype=int)
        rows[r, cols] = f[k - cols]
    A_ub = rows
    b_ub = np.full(len(active_set), M, dtype=np.float64)

    bounds = [(0.0, var_upper) for _ in range(n)]
    c = -np.ones(n, dtype=np.float64)          # minimise -sum ⇒ maximise sum

    if linprog is None:
        raise RuntimeError("SciPy linprog required for LP step.")

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _lp_expand_step(f: np.ndarray,
                    M: float,
                    deadline: float,
                    max_active: int = 2000) -> np.ndarray:
    """
    One outer iteration:
      * solve LP with an active set (expanded until feasible)
      * return the feasible g vector.
    """
    n = f.size
    # start with the (single) most critical lag
    conv_f_f = _conv_fft(f, f)
    max_idx = int(np.argmax(conv_f_f))
    active_set = [max_idx]

    # iterative augment of active set until full feasibility
    while True:
        if time.time() > deadline:
            raise TimeoutError("Time budget exhausted during LP expansion.")

        g = _solve_lp_active_set(f, M, active_set)

        # check full feasibility
        conv_f_g = _conv_fft(f, g)
        viol = conv_f_g - M
        max_viol = np.max(viol)
        if max_viol <= 1e-12:
            return g

        # add the most violating index
        viol_idx = int(np.argmax(viol))
        if viol_idx not in active_set:
            active_set.append(viol_idx)
            if len(active_set) > max_active:
                # safety: fall back to full constraint set (unlikely)
                print("[LP] Active set hit limit, using full constraints.")
                active_set = list(range(2 * n - 1))
        else:
            # should not happen, but break to avoid infinite loop
            return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length 4000) that minimizes
    ``evaluate_sequence``.  The method combines an active‑set linear program
    that expands the ℓ¹‑mass under the current autocorrelation bound, followed
    by a convex mixing step and a short L‑BFGS polishing phase.
    """
    # -------------------- initialise --------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin

    n = 4000

    # start from the best known global solution if present, else uniform
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        f = np.asarray(previous, dtype=np.float64)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # safety: clip to the limits of evaluate_sequence and renormalise
    f = np.clip(f, 0.0, 1000.0)
    total = f.sum()
    if total <= 0.0:
        f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f = f / total

    # -------------------- initial objective --------------------
    conv_f_f = _conv_fft(f, f)
    M = np.max(conv_f_f)
    best_obj = 2.0 * n * M
    best_f = f.copy()
    print(f"[Init] objective ≈ {best_obj:.6f}")

    # -------------------- outer iterative refinement --------------------
    max_outer_iters = 30
    outer_it = 0
    while time.time() < deadline and outer_it < max_outer_iters:
        outer_it += 1
        # recompute bound
        conv_f_f = _conv_fft(f, f)
        M = np.max(conv_f_f)

        # --- LP expansion -------------------------------------------------
        try:
            g = _lp_expand_step(f, M, deadline)
        except TimeoutError:
            print("[Time] budget exhausted during LP.")
            break
        except Exception as exc:
            print(f"[LP] exception: {exc}")
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            print("[LP] obtained zero‑mass g, aborting.")
            break

        g_norm = g / sum_g                     # ℓ¹‑normalised candidate

        # --- convex mixing ------------------------------------------------
        # try a few mixing coefficients and keep the best reduction
        improved = False
        best_local_f = f
        best_local_M = M
        for t in (0.5, 0.3, 0.7, 0.9, 0.1):
            f_new = (1.0 - t) * f + t * g_norm
            conv_new = _conv_fft(f_new, f_new)
            M_new = np.max(conv_new)
            if M_new < best_local_M - 1e-12:
                best_local_f = f_new
                best_local_M = M_new
                improved = True

        if not improved:
            # No improvement found with any mixing; stop iteration
            print(f"[Iter {outer_it}] No improvement, terminating outer loop.")
            break

        f = best_local_f
        M = best_local_M
        cur_obj = 2.0 * n * M
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
            print(f"[Iter {outer_it}] obj={best_obj:.6f} (M={M:.9f}) sum_g={sum_g:.3f}")

        # early exit if we have already passed the target
        if best_obj < 1.503:
            print("[Target] reached desired bound.")
            break

    # -------------------- optional L‑BFGS polishing --------------------
    if time.time() < deadline and minimize is not None:
        try:
            print("[Polish] final L‑BFGS smooth‑max refinement")
            # short beta schedule to keep it cheap
            beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
            f_polished = _lbfgs_smooth_optimize(best_f, n, deadline,
                                                beta_schedule=beta_sched,
                                                max_iter_per_beta=4000)
            obj_polished = _objective_surrogate(f_polished)
            if obj_polished < best_obj:
                best_f = f_polished.copy()
                best_obj = obj_polished
                print(f"[Polish] improved to {best_obj:.6f}")
        except Exception as e:  # pragma: no cover
            print("[Polish] exception:", e)

    # -------------------- final evaluation --------------------
    try:
        true_val = evaluate_sequence(best_f.tolist())
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate≈{best_obj:.6f}, true evaluate_sequence={true_val}")

    # expose for possible warm‑start in the next call
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
