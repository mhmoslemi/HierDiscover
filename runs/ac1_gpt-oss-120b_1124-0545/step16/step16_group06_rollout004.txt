   true objective using a projected sub‑gradient method (mirror descent with
   a diminishing step‑size  ηₜ = 0.6/√t ).  The loop runs until the global deadline
   is reached, keeping the best iterate.

4️⃣ **Stochastic hill‑climbing** – a simulated‑annealing loop perturbs the
   logits of the current iterate with Gaussian noise.  Both downhill moves and
   occasional uphill moves (Metropolis criterion) are accepted.  The temperature
   and proposal‑scale decay slowly, allowing a fine‑grained exploration of the
   final basin.

All iterates stay on the probability simplex (ℓ₁‑normalised), guaranteeing that
the surrogate equals the exact evaluator `evaluate_sequence`.  The function
returns the best sequence found as a plain list of floats and stores it in the
global `height_sequence_1` for possible reuse by future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised sequence (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """
    Optimises the smoothed max‑convolution objective with L‑BFGS‑B.
    Returns an ℓ₁‑normalised vector (softmax of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy optimisation package is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0,
            100000.0, 200000.0, 500000.0, 1000000.0,
        ]

    # initialise logits (log‑softmax parametrisation)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # current probabilities
            conv = _conv_fft(a, a)                    # full convolution
            shift = np.max(conv)                      # subtract max for stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # ---------------------------------------------------------
            # Gradient w.r.t. a
            # ---------------------------------------------------------
            w = exp_beta / sum_exp                     # soft‑max over conv values
            rev_a = a[::-1]                            # reversed a for back‑conv
            full = _conv_fft(rev_a, w)                 # length = 3n-2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice the needed part

            # ---------------------------------------------------------
            # Chain rule through soft‑max
            # ---------------------------------------------------------
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (projected mirror‑descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Projected mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep the simplex automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see derivation in the description)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.6 / math.sqrt(it)                # diminishing step‑size
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.5,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb logits with Gaussian noise; accept improvements or Metropolis‑type
    uphill moves.  Very slow decay lets the search linger near the optimum.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996         # very gentle‑decay
    temp_decay = 0.99997

    no_improve = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # propose a new point in log‑space
        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        # accept if better, otherwise with probability exp(-(Δ)/T)
        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        # schedule decay
        sigma *= sigma_decay
        temperature *= temp_decay

        # if we are stuck for a while, re‑inject some noise
        if no_improve > 150_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate‑generation helpers (cheap families)
# ----------------------------------------------------------------------
def _grid_search_exponential(n: int, deadline: float) -> list:
    """Sweep exponential family over c ∈ [0.3, 4.0]."""
    vals = []
    cs = np.linspace(0.3, 4.0, 22)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev)))
    return vals


def _grid_search_beta(n: int, deadline: float) -> list:
    """Coarse beta‑shape sweep (α,β ∈ [0.1,5] with 11 points each)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 11)
    betas = np.linspace(0.1, 5.0, 11)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec)))
    return vals


def _grid_search_power(n: int, deadline: float) -> list:
    """Power‑law sweep p ∈ [-0.5,5] (35 points)."""
    vals = []
    for p in np.linspace(-0.5, 5.0, 35):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec)))
    return vals


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well and matches the previous best.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0️⃣  Retrieve previous solution, if present (allows warm‑starts)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Rich seed generation (fast surrogate evaluation)
    # ------------------------------------------------------------------
    print("\n--- Seed generation (surrogate) ---")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n))))

    # include previous best if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr)))

    # analytic families
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # random Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 40
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec)))

    # random exponential draws (heavy‑tailed)
    for _ in range(60):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        candidate_pool.append((vec, _objective_surrogate(vec)))

    print(f"Generated {len(candidate_pool)} candidates (pre‑filter).")

    # ------------------------------------------------------------------
    # 2️⃣  Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12                       # a modest number – L‑BFGS is expensive
    top_seeds = [vec for vec, _ in candidate_pool[:top_k]]
    print(f"Keeping top {top_k} seeds for L‑BFGS refinement.")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed L‑BFGS continuation from each seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during L‑BFGS.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=12000)   # a bit more work
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        # occasional random perturbation + second L‑BFGS pass (helps escape)
        if val_opt <= global_best_val and time.time() < deadline:
            v = np.log(seq_opt + 1e-30)
            v_pert = v + 0.02 * rng.normal(size=n)
            seq_pert = _softmax(v_pert)
            seq_opt2 = _lbfgs_smooth_optimize(seq_pert,
                                              n,
                                              deadline,
                                              max_iter_per_beta=8000)
            val_opt2 = _objective_surrogate(seq_opt2)
            if val_opt2 < val_opt:
                seq_opt, val_opt = seq_opt2, val_opt2
                print(f"[L‑BFGS‑pert] improved to {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient polishing (mirror descent) – long run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate most of the remaining budget to this precise phase
        max_exp_iters = int(min(3_500_000, remaining * 0.6 * 1e5))
        print(f"\n--- Exact sub‑gradient polishing (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Stochastic hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(5_000_000, remaining * 0.3 * 1e5))
        print(f"\n--- Stochastic hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq,
                                         n,
                                         deadline,
                                         rng,
                                         sigma0=0.4,
                                         max_iters=max_hc_iters)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final quick sub‑gradient polish (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(1_200_000, remaining * 0.1 * 1e5))
        if max_sub_iters > 0:
            print("\n--- Final sub‑gradient polish ---")
            seq_sub, val_sub = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=max_sub_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[FinalExp] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
