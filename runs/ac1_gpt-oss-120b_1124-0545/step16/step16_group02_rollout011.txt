def _exp_gradient_opt(start_seq, n, deadline, max_iters=500_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # compute grad wrt logit
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # step schedule
        eta = 0.5 / math.sqrt(it)

        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # optional print progress every some steps

    return best_seq, best_val

def _hill_climb(seq, n, deadline, max_iters=200_000):
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()
    for it in range(1, max_iters+1):

            break
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # propose delta
        # choose a random fraction of min(a_i, a_j)
        max_move = min(a[i], 0.5)  # limit to ensure positivity; can also limit to a[i]
        if max_move <= 0:
            continue
        # choose delta up to max_move * random factor
        delta = (rng.random() * 2 - 1) * max_move * 0.5  # move between -0.5*max_move and +0.5*max_move
        # Ensure non-negativity:
        if a[i] - delta < 0 or a[j] + delta < 0:

        a_new[i] -= delta
        a_new[j] += delta

        val_new = _objective_surrogate(a_new)
        if val_new < best_val:

            best_seq = a_new.copy()
            a = a_new

"""
search algorithm for a low‑max‑autocorrelation step function

The optimizer works on the ℓ₁‑normalised probability simplex.
It builds a large, diverse pool of analytic & random “seed’’ vectors,
keeps the best few and refines them with a mirror‑descent (exponentiated‑gradient)
on the exact max‑convolution objective.  Several random restarts of the
gradient phase and a final simple hill‑climber (pairwise mass transfer)
explore the neighbourhood of the best solution.  All computation respects a
global time‑budget (≈ 1000 s) and the best sequence found is stored in the
global variable ``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate: 2·n·max(conv(seq, seq)).  For ℓ₁‑normalised seq this
    equals the value returned by evaluate_sequence().
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                       # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random with exponential weights."""
    if k <= 0 or k > n:
        return _uniform(n)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _block_uniform(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a contiguous block of length block_frac·n."""
    block_len = max(1, int(round(block_frac * n)))
    start = rng.integers(0, n - block_len + 1)
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Largest Dirichlet draws placed alternately left/right."""
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]          # descending
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _collect_seeds_extended(n: int,
                            rng: np.random.Generator,
                            previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a large, diverse pool of ℓ₁‑normalised vectors.
    """
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.3, 0.5, 0.8, 1.0, 2.0, 3.0):
        seeds.append(_sine_window(n, gamma=g))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.10, 0.45, 7):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(10):
        seeds.append(_random_exponential(n, rng))

    # Sparse vectors
    for k in (max(1, int(0.01 * n)), max(1, int(0.03 * n)), max(1, int(0.07 * n))):
        for _ in range(4):
            seeds.append(_sparse_random(n, k, rng))

    # Block‑uniform vectors
    for bf in (0.05, 0.15, 0.30):
        for _ in range(4):
            seeds.append(_block_uniform(n, bf, rng))

    # Alternating Dirichlet seed
    seeds.append(_alternating_dirichlet(n, rng))

    # previous solution, if any
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # normalise sanity‑check
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser on the *exact* max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns (best_sequence, best_surrogate_value).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute convolution and locate active shift
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry (double‑counted convolution)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule through softmax (mirror‑descent on simplex)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # diminishing step size η_t = 0.5 / sqrt(t)
        eta = 0.5 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional status
            if it % 100_000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random‑restart wrapper for the exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _restart_exp_gradient(best_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         restarts: int = 8,
                         max_iters_per: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Perform several noisy restarts of the gradient optimiser.
    Returns the best sequence discovered.
    """
    v0 = np.log(best_seq + 1e-30)
    best_local_seq = best_seq.copy()
    best_local_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    for r in range(restarts):
        if time.time() > deadline:
            break
        # inject Gaussian noise into logits (strength decreases with r)
        noise_scale = 0.12 * (1.0 - r / restarts) + 0.02
        v_start = v0 + noise_scale * rng.standard_normal(n)
        start_seq = _softmax(v_start)

        seq_opt, val_opt = _exp_gradient_opt(start_seq, n, deadline,
                                             max_iters=max_iters_per)
        if val_opt < best_local_val:
            best_local_seq = seq_opt.copy()
            best_local_val = val_opt
            v0 = np.log(best_local_seq + 1e-30)   # centre further restarts here
            print(f"[Restart {r+1}] improved to {best_local_val:.6f}")

    return best_local_seq, best_local_val


# ----------------------------------------------------------------------
# Simple hill‑climber: move a tiny amount of mass between two coordinates
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex: propose a transfer of mass between two
    indices, accept only improvements.  Very cheap, useful for fine‑tuning.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a fraction of the smaller of a[i], a[j]
        max_move = min(a[i], a[j], 0.3)      # clamp to stay safe
        if max_move <= 0:
            continue
        # draw delta in [-max_move, +max_move] (sign decides direction)
        delta = (rng.random() * 2.0 - 1.0) * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0 or a_new[j] < 0.0:
            continue

        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new
        if it % 30_000 == 0:
            print(f"[HillClimb] it={it} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain Python list.
    """
    # ------------------------------------------------------------------
    # 0. Initialise RNG and timing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 2. Load previous solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 3. Seed generation (large, diverse pool)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Collected {len(seeds)} seed vectors.")

    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Gradient‑based refinement of the best seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] current best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached during seed refinement.")
            break

        # allocate a sensible number of iterations based on remaining time
        remaining = deadline - time.time()
        # estimate about 800 gradient steps per second (empirical)
        est_iters = int(min(500_000, remaining * 800))
        max_iters = max(120_000, est_iters)      # never go below a reasonable floor
        print(f"\n--- Exponentiated‑gradient on seed #{idx+1} (max_iters={max_iters}) ---")
        seq_opt, val_opt = _exp_gradient_opt(seed_vec, n, deadline,
                                             max_iters=max_iters)
        print(f"[ExpGrad] best surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random‑restart gradient phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random‑restart gradient refinement ---")
        seq_restart, val_restart = _restart_exp_gradient(global_best_seq, n,
                                                         deadline,
                                                         restarts=12,
                                                         max_iters_per=180_000)
        if val_restart < global_best_val:
            global_best_seq = seq_restart.copy()
            global_best_val = val_restart
            print(f"[Restart] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climbing ---")
        seq_hc, val_hc = _hill_climb(global_best_seq, n, deadline,
                                     max_iters=200_000)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation and return
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
