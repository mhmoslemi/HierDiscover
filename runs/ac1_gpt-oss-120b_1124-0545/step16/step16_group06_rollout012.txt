"""
minimize_convolution.py

Search for a non‑negative step‑function (list of heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The code follows a multi‑stage strategy:

1️⃣  Generate a diverse pool of analytic and random seeds (uniform,
    exponential, beta, power‑law, linear, sine, quadratic, Dirichlet …).

2️⃣  Run a long L‑BFGS‑B continuation on a smooth log‑sum‑exp surrogate of
    the max‑convolution, starting from the best seeds.

3️⃣  Refine the best candidate with an Adam optimiser on the same smooth
    surrogate (many more iterations than the previous L‑BFGS run).

4️⃣  A stochastic hill‑climbing (simulated‑annealing) loop directly on the
    exact surrogate ``_objective_surrogate`` (i.e. 2·n·max(conv)).

5️⃣  Short projected sub‑gradient polishing.

All vectors are kept on the probability simplex (ℓ₁‑normalised) by means
of a soft‑max parametrisation, so the evaluation objective equals
``2·n·max(conv)`` and is invariant under scaling.

The routine respects the 1000 s wall‑clock budget; after finishing it
stores the best sequence in the global name ``height_sequence_1`` and
returns it as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a ℓ₁‑normalised sequence:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _linear_increase(n: int) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _half_sine(n: int) -> np.ndarray:
    # sin(pi * (i+0.5) / n) is zero at both ends, maximal at centre
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _parabolic(n: int) -> np.ndarray:
    # inverted parabola, zero at ends, max at centre
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - ((2.0 * i / (n - 1)) - 1.0) ** 2
    a = np.clip(a, 0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 12000) -> np.ndarray:
    """
    Optimises the smoothed objective using L‑BFGS‑B (if SciPy is available).
    Returns an ℓ₁‑normalised vector (soft‑max of the final logits).
    """
    if _opt is None:
        raise RuntimeError("SciPy is required for L‑BFGS optimisation.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 2000.0, 5000.0, 20000.0,
            50000.0, 100000.0, 200000.0,
            500000.0, 1_000_000.0,
        ]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # --- gradient w.r.t. a ---
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # --- back‑prop through softmax ---
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on the same smooth surrogate (many cheap iterations)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta: float = 200_000.0,
              lr: float = 0.05,
              max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the logit space of the log‑sum‑exp surrogate.
    Returns best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    v = np.log(a + 1e-30)          # logits
    m = np.zeros_like(v)          # first moment
    v_hat = np.zeros_like(v)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        # gradient w.r.t. a
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat_corr = v_hat / (1.0 - beta2_pow)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing / simulated‑annealing optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.3,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb logits with Gaussian noise, accept better (or uphill with a tiny
    probability). Optimises the exact surrogate 2·n·max(conv).
    """
    v = np.log(start_seq + 1e-30)          # logits
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 0.5
    sigma_decay = 0.99994
    temp_decay = 0.99996
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 250_000:
            sigma = sigma0            # re‑inject diversity
            no_improve = 0

    return best_seq, best_val

# Projected sub‑gradient optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on the exact surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.3 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

        vec = _power(n, p)

    return vals

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the evaluation
    objective within the allocated time budget.  Returns the best sequence
    found as a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # Uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Previous best, if any
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Analytic families
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # Extra families
    extra_fams = [
        (_linear_decrease(n), 'lin_dec'),
        (_linear_increase(n), 'lin_inc'),
        (_half_sine(n), 'half_sine'),
        (_parabolic(n), 'parabola')
    ]
    for vec, tag in extra_fams:
        candidate_pool.append((vec, _objective_surrogate(vec), (tag,)))

    # Random Dirichlet draws (various concentrations)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # Random exponential draws (no normalisation needed – we normalise anyway)
    for _ in range(40):
        if time.time() > deadline:
            break
        vec = _random_exponential(n, rng)
        candidate_pool.append((vec, _objective_surrogate(vec), ('exp_draw',)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. L‑BFGS continuation from each seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         deadline,
                                         max_iter_per_beta=12000)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam smoothing optimisation (long run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_adam_iters = int(min(2_000_000, remaining * 1500))  # rough scaling
        print(f"\n--- Adam optimisation (max_iters={max_adam_iters}) ---")
        seq_adam, val_adam = _adam_opt(global_best_seq,
                                       n,
                                       deadline,
                                       beta=200_000.0,
                                       lr=0.05,
                                       max_iters=max_adam_iters)
        if val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(5_000_000, remaining * 1200))
        print(f"\n--- Stochastic hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq,
                                         n,
                                         deadline,
                                         rng,
                                         sigma0=0.3,
                                         max_iters=max_hc_iters)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_sub_iters = int(min(1_200_000, remaining * 800))
        if max_sub_iters > 0:
            print("\n--- Final sub‑gradient polishing ---")
            seq_sub, val_sub = _subgradient_opt(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_sub_iters)
            if val_sub < global_best_val:
                global_best_seq = seq_sub.copy()
                global_best_val = val_sub
                print(f"[SubGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
