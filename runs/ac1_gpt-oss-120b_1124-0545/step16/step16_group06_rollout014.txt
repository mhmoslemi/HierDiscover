def _hillclimb_incremental_opt(start_seq, n, deadline, rng):
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_seq = a.copy()
    best_val = cur_val

    sigma = 0.5
    sigma_decay = 0.99995
    temperature = 0.1
    temp_decay = 0.99995

    it = 0
    no_improve_counter = 0
    recompute_interval = 50000

    while time.time() < deadline:
        it += 1
        # select distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        donor = i
        recipient = j

        max_move = a[donor]
        if max_move <= 0:
            continue
        # step factor decreasing
        step_factor = sigma / math.sqrt(it)
        delta = max_move * step_factor * rng.random()
        if delta <= 0:
            continue

        # copy conv
        conv_candidate = conv.copy()
        # update conv_candidate
        conv_candidate[recipient:recipient+n] += 2.0 * delta * a
        conv_candidate[donor:donor+n] -= 2.0 * delta * a
        conv_candidate[2*recipient] += delta * delta
        conv_candidate[2*donor] += delta * delta
        conv_candidate[donor+recipient] -= 2.0 * delta * delta

        candidate_max = conv_candidate.max()
        candidate_val = 2.0 * n * candidate_max

        accept = False
        if candidate_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(candidate_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply move
            a[donor] -= delta
            a[recipient] += delta
            conv = conv_candidate
            cur_max = candidate_max
            cur_val = candidate_val
            no_improve_counter = 0

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        else:
            no_improve_counter += 1

        # decay sigma and temperature
        sigma *= sigma_decay
        temperature *= temp_decay

        # periodic recompute to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_fft(a, a)
            cur_max = conv.max()
            cur_val = 2.0 * n * cur_max

        # escape if stuck
        if no_improve_counter > 200_000:
            sigma = 0.5
            temperature = 0.1
            no_improve_counter = 0

    return best_seq, best_val

    * the previous best (global variable “height_sequence_1”) if it exists.

2️⃣  Smooth optimisation:
    * L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution.
    * a long β‑schedule (up to 1e6) and many L‑BFGS iterations per β.
    * works in the soft‑max (log‑probability) parametrisation, thus the
      ℓ₁‑norm is automatically 1.

3️⃣  Intensive incremental hill‑climbing:
    * The convolution is kept explicitly and updated in O(n) using the
      exact rank‑1 update formula for a move that transfers a tiny amount
      δ from index i to j.
    * Moves are accepted if they improve the true objective, otherwise they
      are accepted with a Metropolis probability depending on a slowly
      cooling temperature.
    * σ (the typical step size) and the temperature decay geometrically.
    * Every 50 000 iterations the full convolution is recomputed from scratch
      to erase accumulated floating‑point drift.
    * The best sequence seen is stored and finally reported.

All work is done on the probability simplex (sum = 1), therefore the
objective equals ``2·n·max(conv)`` and scaling invariance of the original
``evaluate_sequence`` is respected.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns a feasible non‑negative sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a ℓ₁‑normalised sequence (2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate) using L‑BFGS‑B
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """Optimise the smoothed max‑convolution objective with L‑BFGS‑B."""
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS stage.")

    # an aggressive β‑schedule aiming at a tighter approximation of max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        2000.0, 5000.0, 20000.0, 50000.0,
        100000.0, 200000.0, 500000.0, 1_000_000.0,
    ]

    v = np.log(start_vec + 1e-30)               # log‑softmax parametrisation
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp smoothing of the max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (see derivation in the original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a


# ----------------------------------------------------------------------
# Fast incremental hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_incremental_opt(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a tiny amount δ from index i to j,
    update the convolution with an O(n) rank‑1 formula, accept improvements
    (or uphill moves with a cooling temperature).  Returns the best
    sequence seen and its surrogate objective value.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_max = float(conv.max())
    cur_val = 2.0 * n * cur_max

    best_seq = a.copy()
    best_val = cur_val

    sigma = 0.5               # typical step‑size factor
    sigma_decay = 0.99995
    temperature = 0.1
    temp_decay = 0.99995

    it = 0
    no_improve = 0
    recompute_interval = 50000

    while time.time() < deadline:
        it += 1

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        donor, recip = i, j

        max_move = a[donor]
        if max_move <= 0.0:
            continue

        # step size decreasing with sqrt(it)
        step_factor = sigma / math.sqrt(it)
        delta = max_move * step_factor * rng.random()
        if delta <= 0.0:
            continue

        # --- propose via an inexpensive copy of conv -----------------
        conv_cand = conv.copy()
        # shift updates
        conv_cand[recip:recip + n] += 2.0 * delta * a
        conv_cand[donor:donor + n] -= 2.0 * delta * a
        # quadratic correction at the three special indices
        conv_cand[2 * recip] += delta * delta
        conv_cand[2 * donor] += delta * delta
        conv_cand[donor + recip] -= 2.0 * delta * delta

        cand_max = float(conv_cand.max())
        cand_val = 2.0 * n * cand_max

        # acceptance test
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(cand_val - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[recip] += delta
            conv = conv_cand
            cur_max = cand_max
            cur_val = cand_val
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        else:
            no_improve += 1

        # decay parameters
        sigma *= sigma_decay
        temperature *= temp_decay

        # periodic full recompute to fight floating‑point drift
        if it % recompute_interval == 0:
            conv = _conv_fft(a, a)
            cur_max = float(conv.max())
            cur_val = 2.0 * n * cur_max

        # occasional restart of the step‑size if stuck
        if no_improve > 200_000:
            sigma = 0.5
            temperature = 0.1
            no_improve = 0

        # early exit if we have already reached the target area
        if best_val < 1.503:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate generation helpers (coarse grids)
# ----------------------------------------------------------------------
def _grid_search_exponential(n: int, deadline: float) -> list:
    """Sweep exponential families over c ∈ [0.5, 3.0]."""
    vals = []
    cs = np.linspace(0.5, 3.0, 25)
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float) -> list:
    """Coarse β‑shape sweep (α,β ∈ [0.1,5] with 9 points each)."""
    vals = []
    alphas = np.linspace(0.1, 5.0, 9)
    betas = np.linspace(0.1, 5.0, 9)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float) -> list:
    """Power‑law sweep p ∈ [-0.5,5]."""
    vals = []
    for p in np.linspace(-0.5, 5.0, 31):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allocated budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0         # safety margin

    # Problem size – the previous best used n=4000, we keep it.
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution, if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    print("\n--- Seed generation ---")
    candidate_pool = []

    # uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # analytic families
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # random Dirichlet draws
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # keep the most promising seeds
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 2. Smooth optimisation (L‑BFGS) from each seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx+1} ---")
        try:
            seq_opt = _lbfgs_smooth_optimize(seed_vec, n, deadline)
            val_opt = _objective_surrogate(seq_opt)
            print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

            if val_opt < global_best_val:
                global_best_seq = seq_opt.copy()
                global_best_val = val_opt
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] failed on seed #{idx+1}: {e}")

    # ------------------------------------------------------------------
    # 3. Intensive incremental hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of the remaining budget to hill‑climbing
        max_hc_iters = int(1e7)   # effectively bounded by the time limit
        print(f"\n--- Incremental hill‑climbing (max_iters≈{max_hc_iters}) ---")
        seq_hc, val_hc = _hillclimb_incremental_opt(global_best_seq,
                                                    n,
                                                    deadline,
                                                    rng)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")
        else:
            print(f"[HillClimb] best value unchanged ({global_best_val:.6f})")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
