def _adam_optimize(start_seq, n, deadline):

    v = np.log(start_seq + eps)

    m = np.zeros_like(v)

    v2 = np.zeros_like(v)

    beta = 1.0

    max_beta = 1e6

    beta_steps = 2000

    iter = 0

    best_seq = start_seq.copy()

    best_val = _objective_surrogate(start_seq)

    beta_as_int = 1

    while time.time() < deadline:

        a = _softmax(v)

        # compute conv and gradient

        conv = _conv_fft(a, a)

        shift = np.max(conv)

        diff = conv - shift

        exp_beta = np.exp(beta * diff)

        sum_exp = exp_beta.sum()

        w = exp_beta / sum_exp

        # gradient wrt a

        rev_a = a[::-1]

        full = _conv_fft(rev_a, w)

        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient wrt v

        dot = np.dot(grad_a, a)

        grad_v = a * (grad_a - dot)

        # update Adam

        iter += 1

        m = beta1 * m + (1 - beta1) * grad_v

        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)

        m_hat = m / (1 - beta1 ** iter)

        v2_hat = v2 / (1 - beta2 ** iter)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # update best based on surrogate

        if iter % 500 == 0:

            cur_val = _objective_surrogate(a)

            if cur_val < best_val:

                best_val = cur_val

                best_seq = a.copy()

        # update beta

        if iter % beta_steps == 0 and beta < max_beta:

            beta = min(beta * 2, max_beta)

        # possibly break if time limit

    return best_seq, best_val

"""
Propose a (near) optimal non‑negative step‑function for the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Seed pool** – generate a diverse set of ℓ₁‑normalised candidates:
   uniform, sine‑, Kaiser‑ and beta‑shaped windows, random Dirichlet
   draws and low‑frequency cosine curves.

2. **Select the best seeds** according to the cheap exact surrogate
   ``2·n·max(conv(a,a))`` (i.e. the objective with the ℓ₁‑normalisation).

3. **Long Adam optimisation of a smoothed max‑convolution** – the
   variables live in log‑softmax space (guaranteeing non‑negativity
   and the simplex).  A smoothed max is used via log‑sum‑exp with a
   temperature parameter β that is doubled every few thousand steps.
   Adam (β₁=0.9, β₂=0.999) provides adaptive step‑sizes and converges
   much faster than plain exponentiated‑gradient.

4. **Fine L‑BFGS polishing** – the best vector from Adam is refined with
   SciPy’s L‑BFGS‑B on the same smoothed objective, using a long
   β‑schedule (up to 1e6) and many internal iterations.

5. **Return** the best vector found (global variable ``height_sequence_1``
   is updated for possible reuse).

The whole search respects a user supplied time budget (default 1000 s)
and always returns a valid list of non‑negative floats.
"""

import time
import math
import numpy as np

# optional SciPy import for the final L‑BFGS stage
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum = 1:
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Smoothed max‑convolution utilities (used by Adam & L‑BFGS)
# ----------------------------------------------------------------------
def _smoothed_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad_v) for the smoothed max‑convolution
    ``2·n·(shift + logsumexp/β)`` where ``shift = max(conv)``.
    The gradient is taken w.r.t. the log‑softmax parameters ``v``.
    """
    a = _softmax(v)                     # ensures a ≥ 0, sum = 1
    conv = _conv_fft(a, a)              # full length 2n‑1
    shift = np.max(conv)
    diff = conv - shift                  # ≤ 0
    exp_beta = np.exp(beta * diff)      # stable because max diff = 0
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp               # soft‑max weights over the convolution

    # gradient w.r.t. a: 2 * conv(reverse(a), w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # objective value (smoothed max)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   lr: float = 0.08,
                   max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam (in log‑softmax space) on the smoothed max‑convolution.
    Returns the best sequence seen (exact surrogate value).
    """
    eps = 1e-30
    v = np.log(start_seq + eps)          # initialise logits
    m = np.zeros_like(v)                 # first‑moment
    v2 = np.zeros_like(v)                # second‑moment
    beta1, beta2 = 0.9, 0.999
    t = 0

    # smoothing parameter β – start small and double regularly
    beta_smooth = 1.0
    beta_max = 1e6
    beta_inc_every = 4_000               # increase β every 4k iterations

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while time.time() < deadline and t < max_iter:
        t += 1

        # compute objective, gradient
        obj, grad_v = _smoothed_obj_and_grad(v, n, beta_smooth)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # periodically evaluate the exact surrogate (every few thousand iters)
        if t % 2000 == 0:
            a = _softmax(v)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] iter={t:,}  surrogate={best_val:.6f}")

        # increase smoothing temperature
        if (t % beta_inc_every) == 0 and beta_smooth < beta_max:
            beta_smooth = min(beta_smooth * 2.0, beta_max)

    # final check (in case we never hit the %2000 condition)
    a = _softmax(v)
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (identical to the earlier version but
# with a larger iteration budget)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution with L‑BFGS‑B.
    Variable lives in log‑softmax space (v). Returns a probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1e6]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            diff = conv - shift
            exp_beta = np.exp(beta_val * diff)
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
