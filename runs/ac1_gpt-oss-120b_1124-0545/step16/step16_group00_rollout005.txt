"""
Advanced multi‑start local search for the non‑negative step‑function
minimisation problem.

The algorithm works in three stages:

1. **Seed generation** – a diverse pool of analytic windows, random
   Dirichlet / exponential draws, low‑frequency cosine curves, a
   newly added alternating‑high‑low pattern and a sparse random support.
   All seeds are ℓ₁‑normalised to sum = 1.

2. **Smoothed‑Adam optimisation** – the exact max‑convolution is replaced
   by a temperature‑controlled soft‑max (log‑sum‑exp) and we perform
   Adam‑style mirror descent in the log‑softmax (probability‑simplex)
   domain.  The temperature β grows from 1 to 5·10⁴, letting the optimiser
   first smooth the landscape and then concentrate on the true max.
   The best sequence found by this stage is kept.

3. **Fine L‑BFGS polishing** – starting from the best Adam result we run
   a continuation L‑BFGS‑B optimiser on the same smoothed objective with
   an extended β‑schedule (up to 10⁵).  The final sequence is evaluated
   with the official ``evaluate_sequence`` function and stored in the
   global variable ``height_sequence_2`` for reuse.

The routine respects the supplied time budget (default 1000 s) and
always returns a valid list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)

try:

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple transformations
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
       2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_high_low(n: int, alpha: float = 0.8) -> np.ndarray:
    """Pattern: (1+α, 1-α, 1+α, …) normalised."""
    a = np.ones(n, dtype=np.float64)
    a[::2] *= (1.0 + alpha)
    a[1::2] *= (1.0 - alpha)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_sparse(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Choose m distinct indices, assign random positive weights, normalise."""
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.exponential(scale=1.0, size=m)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed‑Adam optimiser (mirrored‑descent in log‑softmax space)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     lr0: float = 0.5,
                     beta_start: float = 1.0,
                     beta_end: float = 5e4) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns (best_seq, best_surrogate_value).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # initialise logits
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- compute convolution and smoothed weights -----
        conv = _conv_fft(a, a)                 # length 2n‑1
        max_conv = np.max(conv)

        # temperature schedule (exponential growth)
        frac = it / max_iters
        beta = beta_start * ((beta_end / beta_start) ** frac)

        shift = max_conv
        exp_beta = np.exp(beta * (conv - shift))   # max entry → 1
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                      # soft‑max over conv entries

        # ----- gradient of smooth max w.r.t. a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # d/da of smooth max

        # ----- chain rule through softmax (logits) -----
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (math.sqrt(it) + 1.0)          # slowly decaying step size
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = _softmax(log_a)

        # ----- keep best -----
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report (only if enough time elapsed)
        if it % 100_000 == 0:
            print(f"[Adam] it={it:,} best={best_val:.6f} beta={beta:.1f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variable lives in log‑softmax space (v), guaranteeing non‑negativity
    and the simplex constraint.  Returns a probability vector.
    """
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]

    # initialise logits
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)            # stays on simplex
            conv = _conv_fft(a, a)            # full convolution
            max_c = np.max(conv)
            shift = max_c                     # stabilise exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # soft‑max weights over convolution entries
            w = exp_beta / sum_exp

            # gradient wrt a : 2 * conv(reverse(a), w)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    best_a = _softmax(v)
    return best_a

# Seed generation (large, diverse pool)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_alternating_high_low(n, alpha=0.6))

    # Sine windows – moderate grid
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian windows – finer σ grid
    for sigma in np.linspace(0.09, 0.25, 14):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))

    # Beta‑shape family (coarse grid)
    for a in np.linspace(0.2, 4.0, 6):
        for b in np.linspace(0.2, 4.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Sparse random supports (different sparsities)
    for m in (80, 120, 160):
        for _ in range(3):
            seeds.append(_random_sparse(n, m, rng))

    # Include previous solution if present and same length
    prev = globals().get("height_sequence_2")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    return seeds

# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_2`` and returns it as a plain list of floats.
    """
    # Initialise RNG and timing
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 1.5

    # Choose problem size – 4000 provides a good trade‑off
    n = 4000

    print(f"\n=== Starting optimisation (n={n}) ===")
    # ------------------------------------------------------------------
    # 1. Generate a broad set of seeds
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds using the cheap surrogate and keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long smoothed‑Adam polishing (majority of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + remaining * 0.65   # ~65 % to Adam
        print("\n--- Smoothed‑Adam optimisation ---")
        seq_adam, val_adam = _adam_smooth_opt(global_best_seq,
                                              n,
                                              adam_deadline,
                                              max_iters=2_000_000,
                                              lr0=0.5,
                                              beta_start=1.0,
                                              beta_end=5e4)
        if val_adam < global_best_val:
            global_best_seq = seq_adam
            global_best_val = val_adam
            print(f"[Adam] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine L‑BFGS polishing on the remaining time budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- L‑BFGS fine‑tuning (continuation) ---")
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=beta_schedule,
                                           max_iter_per_beta=15_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs
            global_best_val = val_lbfgs
            print(f"[L‑BFGS] refined surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final short Adam polish (use any leftover time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Final short Adam polish ---")
        final_deadline = time.time() + remaining
        seq_final, val_final = _adam_smooth_opt(global_best_seq,
                                                n,
                                                final_deadline,
                                                max_iters=200_000,
                                                lr0=0.3,
                                                beta_start=1.0,
                                                beta_end=2e4)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final Adam] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
