def _exp_gradient_opt(start_seq, n, deadline, max_iters=None, eta0=0.5):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        # compute gradient
        conv = _conv_fft(a,a)
        max_idx = int(np.argmax(conv))
        # compute gradient for max entry
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        idx = np.arange(start, end+1)
        rev = max_idx - idx
        # gradient exact: a[rev] + a[idx]
        g = np.zeros_like(a)
        g[idx] = a[rev] + a[idx]
        # update log-space
        eta = eta0 / math.sqrt(it+1)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        it += 1
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Better gradient‑based search for the step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The optimiser works directly on the simplex via a log‑softmax (mirror‑descent)
parameterisation.  The sub‑gradient of the *active* convolution entry is
computed exactly (the previous code missed the “self‑term”).  Multiple
initial seeds are generated, the best few are refined with an exponentiated‑gradient
run, a longer Adam‑style fine‑tuning follows and finally a short stochastic
local‑walk polishes the result.

All computation uses real FFTs for fast full convolution (≈ O(n log n)).
The final sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate = 2·n·max(conv) for a vector normalised to ℓ₁‑norm 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the **active** entry of the convolution (the one
    attaining the current maximum).  Result has the same length as ``a``.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    i = int(np.argmax(conv))

    start = max(0, i - (n - 1))
    end = min(n - 1, i)
    idx = np.arange(start, end + 1, dtype=np.int64)
    rev = i - idx

    g = np.zeros_like(a)
    # derivative = a[rev] + a[idx] for each idx
    g[idx] = a[rev] + a[idx]
    return g


def _smooth_grad(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f(a) = 2·n·logsumexp(alpha·conv)/alpha
    where ``conv`` = a * a (full convolution).
    """
    n = a.shape[0]

    # 1) full convolution
    conv = _conv_fft(a, a)

    # 2) soft‑max weights w_i ∝ exp(alpha·conv_i)
    max_c = np.max(conv)
    exps = np.exp(alpha * (conv - max_c))
    w = exps / np.sum(exps)                     # shape (2n‑1,)

    # 3) term1 = Σ_i w_i·a_{i‑k}
    w_rev = w[::-1]
    full1 = _conv_fft(a, w_rev)                # length 3n‑2
    # indices to pick: (len(w_rev)-1) – k
    idx = (len(w_rev) - 1) - np.arange(n, dtype=np.int64)
    term1 = full1[idx]

    # 4) term2 = a_k· Σ_{i=k}^{k+n‑1} w_i   (sliding window of length n)
    ones = np.ones(n, dtype=np.float64)
    full2 = _conv_fft(w, ones)                 # length 3n‑2
    # sliding‑window sum for each k is at position k + n‑1
    win = full2[np.arange(n, dtype=np.int64) + n - 1]
    term2 = a * win

    return term1 + term2


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) using the *exact* sub‑gradient
    of the active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space => simplex is enforced automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _grad_max_conv(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[EG] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              lr0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent (still performed in log‑softmax space)
    using the exact gradient of the active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _grad_max_conv(a)

        # Adam moments (still on the log‑space gradient)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 150_000,
                             eta0: float = 0.7,
                             alpha_start: float = 1.0,
                             alpha_end: float = 300.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient with a *smooth* max (log‑sum‑exp) that is
    annealed from a low to a high ``alpha`` value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for alpha (more aggressive later)
        progress = it / max_iters
        alpha = alpha_start * (1 - progress) + alpha_end * progress

        # gradient of the smoothed objective
        g = _smooth_grad(a, alpha)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30_000 == 0:
                print(f"[SmoothEG] it={it:7d} α={alpha:.1f} best={best_val:.6f}")

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.12,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Small stochastic walk in the log‑softmax space.  The variance slowly
    decreases, turning the procedure into a deterministic hill‑climb.
    """
    rng = np.random.default_rng(987654)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / total
        sigma = sigma_start * (1 - prog) + sigma_end * prog

        logits = np.log(cur_seq + 1e-30)
        prop_logits = logits + sigma * rng.normal(size=n)
        prop_seq = _softmax(prop_logits)
        prop_val = _objective_surrogate(prop_seq)

        if prop_val < cur_val:
            cur_seq = prop_seq
            cur_val = prop_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 5_000 == 0:
            print(f"[Local] it={it:6d} best={best_val:.6f}")

    return best_seq, best_val

# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats).  The function respects a wall‑clock
    budget of ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 2.0

    # problem size – the reference solution used n = 4000
    n = 4000

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution if it exists.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                previous_vec = arr / arr.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # 1. Rich seed generation
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if previous_vec is not None:
        seeds.append(previous_vec)

    # beta‑shape draws
    for _ in range(25):
        a = 10 ** rng.uniform(-1, 1)       # 0.1 … 10
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(20):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # Gaussian draws
    for _ in range(15):
        sigma = rng.uniform(0.07, 0.45)
        seeds.append(_gaussian(n, sigma))

    # random Dirichlet draws (various concentrations)
    for conc in (0.03, 0.1, 0.3, 0.7, 1.5):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # normalise (they are already normalised but be safe)
    normed_seeds = []
    for s in seeds:
        s = np.asarray(s, dtype=np.float64)
        total = s.sum()
        if total > 0.0:
            normed_seeds.append(s / total)

    print(f"\n--- Seed generation complete: {len(normed_seeds)} candidates ---")

    # ------------------------------------------------------------------
    # 2. Quick ranking – keep the best few seeds
    # ------------------------------------------------------------------
    surrogate_vals = [_objective_surrogate(v) for v in normed_seeds]
    # pick the top K seeds (lower surrogate is better)
    K = 5
    top_idx = np.argsort(surrogate_vals)[:K]
    top_seeds = [normed_seeds[i] for i in top_idx]

    print(f"Best surrogate among seeds = {surrogate_vals[top_idx[0]]:.6f}")

    # ------------------------------------------------------------------
    # 3. Short EG runs on each of the top seeds (exploration)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    for s_id, seed_vec in enumerate(top_seeds):
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate a modest budget per seed (≈ 30 k iters or time‑based)
        max_iters = int(min(80_000, remaining * 3000))
        print(f"\n--- EG on seed {s_id+1}/{len(top_seeds)} (max_iters≈{max_iters}) ---")
        seq, val = _exp_gradient_opt(seed_vec, n, deadline,
                                     max_iters=max_iters, eta0=0.6)
        if val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[Seed {s_id+1}] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long smooth‑EG run on the current best (fine‑grained annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 3000))
        print(f"\n--- Smooth EG phase (max_iters≈{max_iters}) ---")
        seq, val = _smooth_exp_gradient_opt(global_best_seq, n, deadline,
                                            max_iters=max_iters,
                                            eta0=0.7,
                                            alpha_start=1.0,
                                            alpha_end=300.0)
        if val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[SmoothEG] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Adam‑style fine‑tuning (adaptive moments)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 3000))
        print(f"\n--- Adam‑style fine‑tuning (max_iters≈{max_iters}) ---")
        seq, val = _adam_opt(global_best_seq, n, deadline,
                             max_iters=max_iters, lr0=0.5)
        if val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[Adam] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic local walk
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (≈{remaining:.1f}s left) ---")
        seq, val = _local_random_walk(global_best_seq, n, deadline)
        if val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[Local] improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
