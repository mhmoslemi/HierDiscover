"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm is deliberately simple and spends almost the whole time budget on a
continuous‑mirrored Adam optimisation in the log‑softmax (simplex) space.
It starts from the best known sequence (if any) or from a uniform distribution,
then repeatedly

* evaluates the exact max‑convolution via an FFT,
* computes a sub‑gradient of the max‑convolution,
* maps this gradient to the log‑softmax parameters,
* updates the parameters with Adam (per‑coordinate step‑size adaptation).

Only the best few iterates (according to the cheap surrogate
`2·n·max(conv)`) are kept and finally evaluated with the official
`evaluate_sequence`.  The whole routine respects the 1000 s budget and
requires only NumPy and standard library modules.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # Guard against pathological zero‑sum (should not happen)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax)
# ----------------------------------------------------------------------
def _adam_optimize_on_simplex(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 5_000_000,
                               lr0: float = 0.15,
                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Run Adam on a smooth surrogate of max(conv) while staying on the probability
    simplex via the softmax parametrisation.

    Returns the best sequence seen (as a probability vector) and its surrogate
    value 2·n·max(conv).
    """
    # initialise logits from the start sequence
    v = np.log(start_seq + 1e-30)

    m = np.zeros_like(v)          # first moment
    v_exp = np.zeros_like(v)      # second moment
    best_seq = None
    best_val = math.inf

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- simplex point -----
        a = _softmax(v)            # sum(a) == 1, a_i >= 0

        # ----- exact max‑convolution -----
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # keep the best seen
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[Adam] it={it:,d} best surrogate={best_val:.6f}")

        # ----- sub‑gradient of the active entry -----
        g_a = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g_a[idx_i] = 2.0 * a[rev_idx]

        # ----- map to log‑softmax gradient -----
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # ----- Adam update in logit space -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_exp = beta2 * v_exp + (1.0 - beta2) * (grad_v * grad_v)

        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it

        m_hat = m / bias_correction1
        v_hat = v_exp / bias_correction2

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        v -= step

    # fallback in case the loop never produced a candidate
    if best_seq is None:
        best_seq = _softmax(v)

    return best_seq, best_val
