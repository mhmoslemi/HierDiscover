"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy is a fast “seed explosion” followed by a long
mirror‑descent (exponentiated‑gradient) phase and a final
simulated‑annealing polish.  Compared with the previous implementation
it

* draws many more random Dirichlet seeds (including very sparse
  concentrations) and keeps only the best few,
* runs the smooth L‑BFGS phase on only the very best seed,
* performs a far longer exponentiated‑gradient optimisation,
* runs a slower‑cooling hill‑climb on the final candidate.

All phases respect the 1000 s wall‑clock budget.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution with a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised sequence (i.e. 2·n·max(conv))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution optimiser using mirror‑descent on the
    simplex (softmax logits).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent step in log‑space
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        eta = 0.6 / math.sqrt(it)          # learning‑rate schedule
        log_a = log_a - eta * grad_v
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (perturb logits)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.5,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation of the soft‑max logits with a very slow
    cooling schedule.  Accepts improvements and occasional uphill
    moves.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996          # slower decay than the previous run
    temp_decay = 0.99998

    no_improve = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        # occasional restart of sigma if stuck
        if no_improve > 250_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val

    """
    if _opt is None:
        # SciPy not available – just return the start vector.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0, 5_000.0,
                         10_000.0, 20_000.0, 50_000.0, 100_000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # smoothed max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    the supplied ``evaluate_sequence`` objective.
    The routine respects the 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    # --------------------------------------------------------------
    # Problem size – keep a moderate dimension that still allows
    # a fine‑grained search.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0.  Restore previous best if it exists (helps continuity)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0:
                prev_arr = arr / np.sum(arr)
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1.  Massive seed generation (only cheap evaluations)
    # --------------------------------------------------------------
    print("\n--- Seed generation (fast) ---")
    top_k = 30                # keep only the best 30 seeds
    heap: list[tuple[float, np.ndarray]] = []   # max‑heap storing (-value, vector)

    def push_candidate(vec: np.ndarray):
        """Insert vector into heap, keeping only the best ``top_k``."""
        val = _objective_surrogate(vec)
        # heap stores (negative value, vector) to make a max‑heap
        entry = (-val, vec)
        if len(heap) < top_k:
            heapq.heappush(heap, entry)
        else:
            if entry[0] > heap[0][0]:     # larger negative -> smaller value
                heapq.heapreplace(heap, entry)

    # Uniform baseline
    push_candidate(_uniform(n))

    # Previous solution if present
    if prev_arr is not None:
        push_candidate(prev_arr)

    # Analytic families – a denser grid than in the previous run
    for c in np.linspace(0.3, 4.0, 30):
        push_candidate(_exp_family(n, c))
        push_candidate(_rev_exp_family(n, c))

    for p in np.linspace(-0.4, 5.0, 40):
        push_candidate(_power(n, p))

    alphas = np.linspace(0.1, 5.0, 15)
    betas = np.linspace(0.1, 5.0, 15)
    for a_ in alphas:
        for b_ in betas:
            push_candidate(_beta_shape(n, a_, b_))

    # Random Dirichlet draws – many concentrations & many draws
    concs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05,
             0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per_conc = 150
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            # occasionally also try a shuffled version (different ordering)
            push_candidate(vec)
            if rng.random() < 0.25:       # 25 % chance of a shuffled copy
                shuffled = rng.permuted(vec)
                push_candidate(shuffled)

    # Retrieve the best seeds (as plain numpy arrays)
    seed_list = [entry[1] for entry in sorted(heap, key=lambda e: -e[0])]
    print(f"Collected {len(seed_list)} top seeds (kept {top_k}).")
    global_best_seq = seed_list[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Light L‑BFGS refinement – only on the absolute best seed
    # --------------------------------------------------------------
    if time.time() < deadline and _opt is not None:
        print("\n--- L‑BFGS refinement on the best seed ---")
        refined = _lbfgs_smooth_optimize(global_best_seq,
                                         n,
                                         deadline,
                                         max_iter_per_beta=12_000)
        val_refined = _objective_surrogate(refined)
        print(f"[L‑BFGS] refined surrogate = {val_refined:.6f}")
        if val_refined < global_best_val:
            global_best_seq = refined.copy()
            global_best_val = val_refined
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Long exponentiated‑gradient mirror‑descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the bulk of the remaining time to this phase
        max_exp_iters = int(min(8_000_000,
                                remaining * 0.13 * 1e5))   # heuristic scaling
        print(f"\n--- Exponentiated‑gradient (max_iters={max_exp_iters}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq,
                                             n,
                                             deadline,
                                             max_iters=max_exp_iters)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Stochastic hill‑climbing (very slow cooling)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(6_000_000,
                               remaining * 0.07 * 1e5))
        print(f"\n--- Hill‑climbing (max_iters={max_hc_iters}) ---")
        seq_hc, val_hc = _hill_climb_opt(global_best_seq,
                                         n,
                                         deadline,
                                         rng,
                                         sigma0=0.4,
                                         max_iters=max_hc_iters)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5.  Final evaluation with the official evaluator
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Save for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
