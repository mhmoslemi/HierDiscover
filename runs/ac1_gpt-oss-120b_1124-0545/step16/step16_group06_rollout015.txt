"""
Improved search for a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy expands the original three‑phase pipeline:

1️⃣  **Massive diversified seeding** – analytic families (uniform,
    exponential, reverse‑exponential, beta, power, sinc², cosine‑window)
    plus many random Dirichlet draws with a log‑uniform concentration.
2️⃣  **Fast mirror‑descent polishing** – operates directly on the exact
    max‑convolution using the sub‑gradient of the active entry.
    The iterate lives in the probability simplex via the log‑softmax
    (exponentiated‑gradient) parametrisation.
3️⃣  **Stochastic hill‑climbing** – Gaussian perturbations of the logits
    with a simple simulated‑annealing acceptance rule, exploiting the
    remaining time budget.

All stages respect the 1000 s wall‑clock limit and keep the sequence
non‑negative.  The best found sequence is stored in the global variable
``height_sequence_1`` and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sum to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a *probability* vector (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ r^i  with  r = exp(-c/n) ."""
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    i = np.arange(n, dtype=np.float64)
    a = r ** (n - 1 - i)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sinc_sq(n: int) -> np.ndarray:
    """a_i ∝ sinc(i/(n‑1))² .  np.sinc uses sin(πx)/(πx)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sinc(i / (n - 1)) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent / exponentiated‑gradient optimiser (exact max‑conv)
# ----------------------------------------------------------------------
def _mirror_descent_opt(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact max‑convolution.
    ``start_seq`` must be a probability vector (sum=1, non‑negative).
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (softmax parametrisation)
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent (exponentiated‑gradient) step
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * grad_log

        # occasional progress check
        if it % 10_000 == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()
                print(f"[MD] iter {it:,} – new best surrogate = {best_val:.6f}")

    # final projection onto simplex (safety)
    best_seq = best_seq / best_seq.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    sigma0: float = 0.3,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Perturb the log‑softmax logits with Gaussian noise.
    Accept moves that improve the exact surrogate or with a small
    probabilistic uphill move (simulated annealing).
    """
    v = np.log(start_seq + 1e-30)          # logits
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996
    temp_decay = 0.99999
    no_improve = 0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Hill] iter {it:,} – improved surrogate = {best_val:.6f}")
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        if no_improve > 200_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the ``evaluate_sequence`` objective within the given
    time budget.  Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # problem size – the previous best used 4000, we keep it
    n = 4000

    # ------------------------------------------------------------------
    # 0.  Retrieve previous solution, if any (useful for warm‑starts)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1.  Rich seed pool
    # ------------------------------------------------------------------
    print("\n--- Seed generation (wide) ---")
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_cosine_window(n))
    seeds.append(_sinc_sq(n))

    if prev_arr is not None:
        seeds.append(prev_arr)

    # exponential sweep (dense)
    for c in np.linspace(0.2, 5.0, 48):
        seeds.append(_exp_family(n, c))
        seeds.append(_rev_exp_family(n, c))

    # beta shape – random log‑uniform parameters
    for _ in range(250):
        a = 10 ** rng.uniform(-2, 2)   # α ∈ [1e‑2, 1e2]
        b = 10 ** rng.uniform(-2, 2)   # β ∈ [1e‑2, 1e2]
        seeds.append(_beta_shape(n, a, b))

    # power‑law draws
    for _ in range(250):
        p = rng.uniform(-0.5, 5.0)
        seeds.append(_power(n, p))

    # Dirichlet draws with many concentration levels
    concs = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(150):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # 2.  Evaluate all seeds, keep the best few
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    top_k = 8
    scored = []

    for vec in seeds:
        if time.time() > deadline:
            break
        # safety: enforce positivity and normalisation
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 0.01:
            continue
        a = vec / s
        val = _objective_surrogate(a)
        scored.append((val, a))

    scored.sort(key=lambda x: x[0])
    for val, a in scored[:top_k]:
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"Seed stage: best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Mirror‑descent polishing (exact max‑conv)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a generous slice of the remaining budget
        max_md_iters = int(min(800_000, remaining * 200))   # heuristic
        print(f"\n--- Mirror‑descent polishing (≤{max_md_iters:,} iters) ---")
        seq_md, val_md = _mirror_descent_opt(best_seq, n, deadline, max_iters=max_md_iters)
        if val_md < best_val:
            best_val = val_md
            best_seq = seq_md.copy()
            print(f"[MD] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Stochastic hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        max_hc_iters = int(min(4_000_000, remaining * 250))   # heuristic
        print(f"\n--- Stochastic hill‑climbing (≤{max_hc_iters:,} iters) ---")
        seq_hc, val_hc = _hill_climb_opt(best_seq, n, deadline, rng,
                                         sigma0=0.3, max_iters=max_hc_iters)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[HillClimb] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final exact evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
