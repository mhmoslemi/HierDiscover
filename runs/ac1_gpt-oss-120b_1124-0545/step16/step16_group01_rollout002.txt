"""
Propose a non‑negative step‑function (probability vector) that minimises

    2·n·max(conv(a, a)) / (sum(a)²)

The algorithm:
1. Build a rich pool of candidate vectors from analytic families
   (uniform, triangular, Gaussian, power‑law, sin², beta) and
   random Dirichlet draws.
2. Keep the best few candidates according to the cheap surrogate
   `2·n·max(conv)` (the exact objective for a normalised vector).
3. Refine each top candidate with a continuation L‑BFGS‑B optimiser that
   minimises a smooth log‑sum‑exp approximation of the max‑convolution.
4. Run a long projected sub‑gradient descent on the *exact* max‑convolution
   starting from the best L‑BFGS result.
5. Perform a few random restarts (logit‑space perturbations) followed by
   short sub‑gradient runs to escape possible local minima.
6. Return the best sequence found (stored globally as ``height_sequence_1``).

The whole search respects a 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (only needed for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the discrete grid, ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0⁰ or division by zero – clip extremes
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # Clip huge values caused by α<1 or β<1 (the official evaluator caps at 1000)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s > 0.0:
        return a / s
    else:
        return _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine a ℓ₁‑normalised vector with L‑BFGS‑B on the smooth max‑conv."""
    if _opt is None:
        # SciPy not present – fall back to the start vector.
        print("[L‑BFGS] SciPy not available – skipping optimisation.")
        return start_vec.copy()

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                # stays on the simplex
            conv = _conv_fft(a, a)                # full convolution
            max_c = np.max(conv)
            # smooth approximation of the max using log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max wrt a
            w = exp_beta / sum_exp                # soft‑weights (sum to 1)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2, but we need centre part
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv) for ℓ₁‑normalised vectors."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:7d} best={best_val:.6f}")

        # sub‑gradient of the active entry (all i that contribute to max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)          # classic diminishing step‑size
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random logit‑space perturbation followed by short sub‑gradient runs
# ----------------------------------------------------------------------
def _perturb_and_refine(base_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """Create a logit‑perturbed copy of *base_seq* and run a short sub‑gradient."""
    logits = np.log(base_seq + 1e-30)
    sigma = 0.8  # magnitude of random logit noise
    pert_logits = logits + sigma * rng.normal(size=n)
    pert_seq = _softmax(pert_logits)
    return _subgradient_opt(pert_seq, n, deadline, max_iters=max_iters)


# ----------------------------------------------------------------------
# Main entry point required by the challenge framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    builds a diverse seed pool, refines the best seeds with L‑BFGS,
    follows with a long sub‑gradient descent, and finally tries a few
    random restarts.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 1. Seed pool construction
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.10, 0.15, 0.20, 0.25, 0.30):
        seeds.append(_gaussian(n, sigma))
    for p in (0.2, 0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_power(n, p))
    seeds.append(_sin2(n))

    # Beta families (including the classic arcsine density)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        for beta in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
            seeds.append(_beta_distribution(n, alpha, beta))

    # Random Dirichlet draws with a spread of concentration parameters
    conc_list = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in conc_list:
        for _ in range(3):                     # fewer draws – keeps runtime low
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previously discovered best sequence (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size != n:
            # interpolate / resample to the new length
            idx_src = np.linspace(0, arr.size - 1, n)
            arr = np.interp(idx_src, np.arange(arr.size), arr, left=0.0, right=0.0)
        if arr.sum() > 0.0:
            seeds.append(arr / arr.sum())

    # Normalise robustly and make sure everything is non‑negative
    clean_seeds = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (cheap surrogate ranking)
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement on the top seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # Allocate roughly half of the remaining budget to L‑BFGS
    time_for_lbfgs = (deadline - time.time()) * 0.5
    # Simple per‑seed deadline slicing
    sec_per_seed = time_for_lbfgs / max(1, len(top_seeds))

    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS.")
            break

        slice_dead = min(deadline, time.time() + sec_per_seed)
        print(f"\n--- L‑BFGS run on seed #{idx+1} (time slice ≈ {sec_per_seed:.1f}s) ---")
        seq_opt = _lbfgs_smooth_optimize(seed_vec,
                                         n,
                                         slice_dead,
                                         beta_schedule=beta_sched,
                                         max_iter_per_beta=2500)
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_seq = seq_opt.copy()
            best_val = val_opt
            print(f"[Update] new global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long projected sub‑gradient descent (the heavy‑lifting phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Rough estimate: one sub‑gradient step ≈ 0.001 s (for n=8000)
        est_iters = int(min(800_000, remaining / 0.0012))
        est_iters = max(100_000, est_iters)  # guarantee a respectable run
        print(f"\n--- Sub‑gradient descent (≈ {est_iters} iters) ---")
        seq_sub, val_sub = _subgradient_opt(best_seq, n, deadline, max_iters=est_iters)
        if val_sub < best_val:
            best_seq = seq_sub.copy()
            best_val = val_sub
            print(f"[SubGrad] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random logit‑perturbation restarts (optional, time‑aware)
    # ------------------------------------------------------------------
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest chunk per restart
        max_iters = int(min(150_000, remaining / 0.0015))
        if max_iters < 10_000:
            break
        print(f"\n--- Random restart #{r+1} (sub‑grad iters={max_iters}) ---")
        seq_try, val_try = _perturb_and_refine(best_seq, n, deadline, rng, max_iters=max_iters)
        if val_try < best_val:
            best_seq = seq_try.copy()
            best_val = val_try
            print(f"[Restart] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
