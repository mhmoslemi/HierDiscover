def _adam_opt(start_vec, n, deadline, beta_schedule, steps_per_beta):
  v = log(start)
  m = np.zeros_like(v)
  v2 = np.zeros_like(v)
  t=0
  for beta in schedule:
    for step in range(steps_per_beta):
      if time.time() > deadline: break
      # compute gradient as earlier
      a = softmax(v)
      conv = conv_fft(a,a)
      shift = np.max(conv)
      exp_beta = np.exp(beta*(conv-shift))
      sum_exp = exp_beta.sum()
      # obj = 2 n (shift + log(sum_exp)/beta)
      # gradient w.r.t a as per earlier
      w = exp_beta / sum_exp
      rev_a = a[::-1]
      full = conv_fft(rev_a, w) # length 3n-2
      grad_a = 2.0 * full[n-1:n-1+n]
      # chain rule as earlier
      dot = np.dot(grad_a, a)
      grad_v = a * (grad_a - dot)
      # Adam updates
      t += 1
      m = beta1*m + (1-beta1)*grad_v
      v2 = beta2*v2 + (1-beta2)*(grad_v**2)
      m_hat = m/(1-beta1**t)
      v2_hat = v2/(1-beta2**t)
      v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    # after each beta maybe reduce lr
  return softmax(v)

"""
Improved three‑phase optimiser for the step‑function problem.

Phase 1 – rich, randomised seed generation (Dirichlet draws with many
concentrations, a few analytic families and the previous best if any).

Phase 2 – long L‑BFGS‑B runs on a smooth log‑sum‑exp surrogate of the
max‑convolution.  The β‑schedule is finer and the maximum number of
iterations per β is increased (up to 15000) compared with the reference
implementation.  This gives the optimiser much more time to approach the
true non‑smooth optimum.

Phase 3 – high‑precision mirror‑descent on the exact objective (sub‑gradient
w.r.t. the ℓ₁‑simplex).  The step‑size follows ηₜ = 0.6/√t with a
back‑tracking line‑search that accepts any strictly improving move.
After the mirror descent a short stochastic hill‑climb finishes the
polish.

All iterates stay on the probability simplex (ℓ₁‑normalised), so the
evaluation objective coincides with ``2·n·max(conv)``.  The routine
respects the 1000 s wall‑clock limit, prints progress information and
stores the best vector in the global name ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy (only for L‑BFGS‑B)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (efficient for large n)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an ℓ₁‑normalised vector:
       2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ r^i with r = exp(-c/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ r^{n‑1‑i}."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (n - 1 - np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS (smooth max‑convolution) optimiser
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """Return (obj, grad_v) for the log‑sum‑exp smooth surrogate."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)

    # log‑sum‑exp of (beta * (conv - shift))
    diff = conv - shift            # ≤ 0
    # avoid overflow: if beta*diff is too small, exp becomes 0 → ok
    exp_beta = np.exp(beta * diff)
    sum_exp = exp_beta.sum()
    # surrogate value: 2·n·(shift + log(sum_exp)/beta)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t a (see derivation in the reference code)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain‑rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000) -> np.ndarray:
    """Run L‑BFGS‑B on the smooth surrogate.
       Returns an ℓ₁‑normalised vector (softmax of the final logits)."""
    if _opt is None:
        raise RuntimeError("SciPy is required for the L‑BFGS phase.")

    if beta_schedule is None:
        # finer schedule – more steps at large β
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 2000.0, 8000.0, 30000.0,
                         100000.0, 300000.0, 1e6]

    # initialise log‑softmax parameters
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            return _smooth_obj_and_grad(v_local, n, beta_val)

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-13})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Precise mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _mirror_descent_max(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (log‑parameter) on the exact non‑smoothed objective.
    Uses a back‑tracking line‑search that only accepts strictly
    improving moves.
    Returns the best sequence seen and its surrogate objective.
    """
    # initialise logits (log‑softmax)
    log_a = np.log(start_seq + 1e-30)
    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # sub‑gradient for the active shift
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # base step size that decays like 1/√t
        eta = 0.6 / math.sqrt(it)

        # back‑tracking line‑search (max 8 tries)
        for _ in range(8):
            v_new = log_a - eta * grad_v
            a_new = _softmax(v_new)
            conv_new = _conv_fft(a_new, a_new)
            new_val = 2.0 * n * np.max(conv_new)

            if new_val < cur_val:               # accept a strict improvement
                log_a = v_new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a_new.copy()
                break
            eta *= 0.5   # shrink step and retry
        # if none of the tries improve we keep the old point

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing on logits (final polish)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                sigma0: float = 0.4,
                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Small Gaussian perturbations of the logits with a tiny simulated‑annealing
    acceptance rule.  Works directly on the exact surrogate.
    """
    v = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    sigma = sigma0
    temperature = 1.0
    sigma_decay = 0.99996
    temp_decay = 0.99997
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        v_prop = v + sigma * rng.normal(size=n)
        a_prop = _softmax(v_prop)
        val_prop = _objective_surrogate(a_prop)

        accept = False
        if val_prop < cur_val:
            accept = True
        else:
            prob = math.exp(-(val_prop - cur_val) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            v = v_prop
            cur_seq = a_prop
            cur_val = val_prop
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        else:
            no_improve += 1

        sigma *= sigma_decay
        temperature *= temp_decay

        # occasional re‑boost to escape flat regions
        if no_improve > 150_000:
            sigma = sigma0
            no_improve = 0

    return best_seq, best_val

def _generate_seeds(n: int, rng: np.random.Generator, deadline: float):
    """Create a diverse list of seeds (vector, surrogate value)."""
    seeds = []

    # deterministic families
    seeds.append((_uniform(n), _objective_surrogate(_uniform(n))))
    for c in np.linspace(0.3, 4.0, 20):
        if time.time() > deadline:
            break
        seeds.append((_exp_family(n, c), _objective_surrogate(_exp_family(n, c))))
        seeds.append((_rev_exp_family(n, c), _objective_surrogate(_rev_exp_family(n, c))))
    for a in np.linspace(0.2, 5.0, 12):
        for b in np.linspace(0.2, 5.0, 12):
            if time.time() > deadline:
                break
            seeds.append((_beta_shape(n, a, b), _objective_surrogate(_beta_shape(n, a, b))))
    for p in np.linspace(-0.4, 5.0, 30):
        if time.time() > deadline:
            break
        seeds.append((_power_law(n, p), _objective_surrogate(_power_law(n, p))))

    # random Dirichlet draws
    concs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    draws_per = 50
    for conc in concs:
        for _ in range(draws_per):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            seeds.append((vec, _objective_surrogate(vec)))

    # previous solution if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append((arr, _objective_surrogate(arr)))
        except Exception:
            pass

    return seeds

    return global_best_seq.tolist()
