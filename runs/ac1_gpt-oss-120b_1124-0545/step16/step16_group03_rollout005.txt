"""
Propose a step‑function that beats the previous 1.5085 bound.
The method combines:
  * a rich pool of analytic/draw seeds,
  * a fast surrogate objective (2·n·max(conv) for ℓ₁‑normalised vectors),
  * Adam‑style mirror‑descent in log‑softmax space on the exact max‑convolution,
  * a final stochastic local search (logit‑perturbations with cooling).
All phases respect the global 1000 s deadline and keep the best sequence found.
"""

# ----------------------------------------------------------------------
# Imports & low‑level utilities
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray, n: int) -> float:
    """
    Exact surrogate value 2·n·max(conv) for a sequence that sums to 1.
    The true objective `evaluate_sequence` coincides with this when the
    sequence is ℓ₁‑normalised.
    """
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective_true(seq: np.ndarray) -> float:
    """Calls the official evaluator (expects a Python list)."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Analytic families (always ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a

# Seed pool

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse list of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Analytic families ----------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma in (0.04, 0.07, 0.10, 0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))

    for p in (0.0, 0.2, 0.5, 0.8, 1.2, 2.0):
        seeds.append(_power(n, p))

    for rate in (0.0005, 0.001, 0.0025, 0.005, 0.01, 0.02):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Dirichlet draws (various concentrations) ---------------------------
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                                 # raw
            seeds.append(np.sort(raw)[::-1])                 # descending
            # random permutation of the raw draw
            seeds.append(raw[rng.permutation(n)])

    # Sparse blocks ---------------------------------------------------------
    for blk in (80, 120, 160, 200):
        for _ in range(3):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous solution ----------------------------------------------------
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Normalise everything robustly ----------------------------------------
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned

# Adam‑style mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam updates in log‑softmax space.
    Returns the best sequence found and its surrogate value.
    """
    w = np.log(start_seq + 1e-30)            # logits
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                               # base learning rate
    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq, n)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        # probabilities
        p = _softmax(w)

        # convolution & active shift
        conv = _conv_fft(p, p)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(p)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        rev = k - idx
        g[idx] = 2.0 * p[rev]               # diagonal gets 2* which is fine

        # gradient w.r.t. logits (mirror‑descent)
        dot = np.dot(g, p)
        grad = p * (g - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule – simple decay
        lr = lr0 / (1 + 1e-5 * t)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional true (surrogate) check
        if t % 5000 == 0:
            cur_val = _surrogate(p, n)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = p.copy()
    return best_seq, best_val

# Simple stochastic local search (logit perturbations)
# ----------------------------------------------------------------------
def _stochastic_refine(best_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Random walk in log‑softmax space with a linear cooling of the
    perturbation size. Accepts only strict improvements of the surrogate.
    """
    rng = np.random.default_rng()
    logit = np.log(best_seq + 1e-30)
    cur_seq = best_seq.copy()
    cur_val = _surrogate(cur_seq, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linearly decay sigma from 0.15 → 1e‑4
        sigma = 0.15 * (1 - it / max_iters) + 1e-4
        proposal_logits = logit + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _surrogate(proposal_seq, n)
        if proposal_val < cur_val:
            logit = proposal_logits
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < _surrogate(best_seq, n):
                best_seq = cur_seq.copy()
    return best_seq, cur_val
