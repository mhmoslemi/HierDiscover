"""
Hybrid optimisation for the step‑function minimisation problem.

The pipeline (all steps respect the global deadline) is:

1.  Initialise a reasonably good λ‑logit vector (previous best,
    Gaussian, power‑law …) and normalise it to an ℓ₁‑vector.
2.  Continuation L‑BFGS on the smooth log‑sum‑exp surrogate of the
    max‑convolution.  The β‑schedule is long and the per‑β iteration
    budget is generous, so we converge close to a stationary point.
3.  Exact mirror‑descent (exponentiated‑gradient) on the real
    max‑convolution surrogate for many iterations – cheap polishing.
4.  Linear‑programming “dual” step (the classic f‑g alternation):
    given the current f, solve   maximise Σ gᵢ   subject to
    (f * g)ₖ ≤ M  (M = max (f * f)) and g ≥ 0.
    Then move f ← (1‑t) f + t g (t≈0.5, decaying if no improvement).
    A few alternations dramatically lower the peak of the convolution.
5.  Optional long sub‑gradient descent as a final clean‑up.
6.  One call to the official ``evaluate_sequence`` to obtain the true
    score; the best ℓ₁‑normalised vector is stored as ``height_sequence_1``.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
import scipy.linalg as _la
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (working with ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact 2·n·max(conv) for a sequence with sum=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _objective(seq: np.ndarray) -> float:
    """True objective used by evaluate_sequence (no normalisation)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = float(np.max(b))
    sum_a = float(np.sum(seq))
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-rate * d)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_uniform_block(n: int, block_size: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of `block_size` indices."""
    idx = rng.choice(n, size=block_size, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / block_size
    return a


# ----------------------------------------------------------------------
# Seed pool construction
# ----------------------------------------------------------------------
def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.05, 0.12, 0.18, 0.25, 0.35):
        seeds.append(_gaussian(n, sigma))
    for p in (0.0, 0.25, 0.5, 0.75, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))
    for rate in (0.001, 0.003, 0.006, 0.01, 0.015):
        seeds.append(_exponential(n, rate))
        seeds.append(_double_exponential(n, rate))

    # Dirichlet draws
    conc_list = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for conc in conc_list:
        for _ in range(4):
            raw = _random_dirichlet(n, conc, rng)
            seeds.append(raw)                         # raw
            seeds.append(np.sort(raw)[::-1])         # descending
            # a sparse version (keep only the top 30 %)
            th = np.quantile(raw, 0.7)
            sparse = np.where(raw > th, raw, 0.0)
            if sparse.sum() > 0:
                seeds.append(sparse / sparse.sum())

    # Sparse uniform blocks
    for blk in (80, 100, 120, 150):
        for _ in range(4):
            seeds.append(_sparse_uniform_block(n, blk, rng))

    # Previous best
    if previous is not None and previous.size == n and previous.sum() > 0:
        seeds.append(previous / previous.sum())

    # Normalise robustly
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0.0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """Refine ``start_vec`` (ℓ₁‑normalised) with L‑BFGS‑B on a smooth max‑conv."""
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         10000.0, 20000.0, 50000.0, 100000.0,
                         200000.0, 500000.0]

    # optimisation variables are logits (soft‑max enforces simplex)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)                     # full convolution
            m = np.max(conv)
            shift = m                                   # stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth max w.r.t. a
            w = exp_beta / sum_exp                      # soft weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser (exact surrogate)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent polish on the exact max‑conv surrogate."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.5 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming “dual” refinement
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_outer: int = 6,
               t_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Alternating LP improvement:

    *Given* f (ℓ₁‑normalised) compute M = max(f*f).  Solve
    maximise Σ gᵢ  s.t. (f*g)ₖ ≤ M, g ≥ 0.
    Then update f ← (1‑t)f + t g, renormalise, and repeat.
    """
    best_seq = f.copy()
    best_val = _objective_surrogate(best_seq)
    t = t_initial

    for outer in range(max_outer):
        if time.time() > deadline:
            break

        # compute current bound M
        conv_fg = _conv_fft(best_seq, best_seq)
        M = float(np.max(conv_fg))

        # Build the convolution matrix A where A @ g = f * g
        # A has shape (2n‑1, n) and constant descending diagonals.
        data = []
        offsets = []
        # Only non‑zero diagonals need to be added.
        for d, val in enumerate(best_seq):
            if val == 0.0:
                continue
            data.append(np.full(n, val, dtype=np.float64))
            offsets.append(d)          # row‑col = d
        A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear programme: maximise Σ gᵢ  ⇔ minimise −Σ gᵢ
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        lp_res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                              method='highs', options={'presolve': True})
        if not lp_res.success:
            print("[LP] linprog failed:", lp_res.message)
            break
        g = lp_res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()

        # Mix and renormalise
        new_f = (1.0 - t) * best_seq + t * g
        new_f = new_f / new_f.sum()
        new_val = _objective_surrogate(new_f)

        if new_val < best_val - 1e-9:
            best_seq, best_val = new_f, new_val
            print(f"[LP] iter {outer+1}: improved surrogate = {best_val:.6f}")
            # keep mixing factor moderate
            t = min(0.9, t * 1.1)
        else:
            # no improvement → shrink the step
            t *= 0.5
            if t < 1e-3:
                break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact surrogate)
# ----------------------------------------------------------------------
def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """Long sub‑gradient descent on 2·n·max(conv)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.25 / math.sqrt(it)
        a -= lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    # --------------------------------------------------------------
    # 0. Setup
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 1. Problem size – 4000 offers a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Seed pool (use previous best if it exists)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    seeds = _collect_seeds(n, rng, prev_arr)
    # evaluate surrogate for all seeds and keep the best few
    evaluated = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 30
    best_candidates = [vec for vec, _ in evaluated[:top_k]]
    print(f"[Init] Best surrogate among raw seeds = {evaluated[0][1]:.6f}")

    # --------------------------------------------------------------
    # 3. L‑BFGS refinement on the most promising seed
    # --------------------------------------------------------------
    # start from the absolute best raw seed
    cur_seq = best_candidates[0].copy()
    cur_val = _objective_surrogate(cur_seq)
    print(f"[Start] surrogate = {cur_val:.6f}")

    # long continuation schedule
    beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0,
                  500.0, 1000.0, 2000.0, 5000.0,
                  10000.0, 20000.0, 50000.0,
                  100000.0, 200000.0, 500000.0]

    # generous per‑β budget (few thousand iterations)
    cur_seq = _lbfgs_smooth_optimize(cur_seq, n, deadline,
                                     beta_schedule=beta_sched,
                                     max_iter_per_beta=8000)
    cur_val = _objective_surrogate(cur_seq)
    print(f"[L‑BFGS] surrogate after continuation = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 4. Exact mirror‑descent polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(2_000_000, remaining * 0.08 * 1e5))
        iters = max(300_000, iters)
        cur_seq, cur_val = _exp_gradient_opt(cur_seq, n, deadline,
                                             max_iters=iters)
        print(f"[ExpGrad] surrogate after polishing = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 5. Linear‑programming dual refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_seq, cur_val = _lp_refine(cur_seq, n, deadline,
                                      max_outer=8, t_initial=0.5)
        print(f"[LP] surrogate after dual steps = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 6. Long sub‑gradient descent (final clean‑up)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(2_500_000, remaining * 0.07 * 1e5))
        cur_seq, cur_val = _subgradient_opt(cur_seq, n, deadline,
                                            max_iters=iters)
        print(f"[SubGrad] final surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 7. Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = cur_seq.tolist()
    return cur_seq.tolist()
