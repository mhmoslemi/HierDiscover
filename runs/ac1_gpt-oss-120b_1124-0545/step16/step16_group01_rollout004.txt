"""
Evolutionary + smooth‑max L‑BFGS optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Rich seed pool** – analytic families (uniform, triangular, Gaussian,
   power‑law, sinusoidal, exponential, beta, window functions) together with
   many Dirichlet draws.  All vectors are ℓ₁‑normalised, i.e. they represent
   probability distributions.

2. **Population evolution** – a compact population (≈30) is evolved on the
   exact surrogate 2·n·max(conv) (for normalised vectors this equals the
   true objective).  Off‑springs are created in log‑softmax space by recombining
   two parents and adding isotropic Gaussian noise whose standard deviation
   decays with the remaining time budget.  The best individuals survive
   (elitism) and the process repeats until only a few seconds remain.

3. **Polishing** – the current global best is refined with a continuation
   L‑BFGS‑B optimiser that minimises a smooth log‑sum‑exp approximation of the
   max‑convolution.  The optimiser runs for the remaining time budget.

The result is returned as a plain list of non‑negative floats.  All numbers are
clamped to [0, 1000] automatically by ``evaluate_sequence``; the optimiser works
on the simplex, so the sum is exactly 1.
"""

import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None  # the framework will raise if optimisation is unavailable


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _make_conv_helper(n: int):
    """Pre‑compute FFT length for a given n and return a convex‑max evaluator."""
    L = 2 * n - 1
    N = _next_pow2(L)

    def max_conv(seq: np.ndarray) -> float:
        """Maximum entry of conv(seq, seq).  ``seq`` must be non‑negative and sum to 1."""
        fft_seq = np.fft.rfft(seq, n=N)
        conv = np.fft.irfft(fft_seq * fft_seq, n=N)[:L]
        # numerical noise may produce tiny negative values – ignore them
        return float(conv.max())

    return max_conv


# ----------------------------------------------------------------------
# Objective helpers (surrogate = exact objective for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray, n: int, max_conv_func) -> float:
    """Exact objective for a normalised vector: 2·n·max(conv(seq,seq))."""
    return float(2.0 * n * max_conv_func(seq))


# ----------------------------------------------------------------------
# Analytic families (return normalised probability vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _sin2(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = np.sin(np.pi * x) ** 2
    a /= a.sum()
    return a


def _exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / rate)
    a /= a.sum()
    return a


def _exponential_symmetric(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-np.abs(i - c) / rate)
    a /= a.sum()
    return a


def _beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) pdf on the discrete grid {0,…,n‑1}."""
    x = np.linspace(0, 1, n, dtype=np.float64)
    # add a small epsilon to avoid 0⁰ when α or β < 1
    eps = 1e-12
    a = (x + eps) ** (alpha - 1) * (1 - x + eps) ** (beta - 1)
    a /= a.sum()
    return a


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimisation (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Refine a normalised vector with L‑BFGS‑B on a smooth max‑conv surrogate."""
    if _opt is None:
        raise RuntimeError("SciPy optimiser is required for L‑BFGS.")

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    # initialise logits (log‑softmax parameters)
    v = np.log(start_vec + 1e-30)

    N_fft = _next_pow2(2 * n - 1)                      # FFT length (shared)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            # ---------------------------------------------------------
            # smooth max of the convolution using log‑sum‑exp
            # ---------------------------------------------------------
            fft_a = np.fft.rfft(a, n=N_fft)
            conv = np.fft.irfft(fft_a * fft_a, n=N_fft)[: 2 * n - 1]
            shift = np.max(conv)                       # numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # ---------------------------------------------------------
            # gradient of the smooth max w.r.t. a
            # ---------------------------------------------------------
            w = exp_beta / sum_exp                      # soft weights (summing to 1)
            # grad_a = 2 * conv_{reversed a} (w)  (as derived in the paper)
            rev_a = a[::-1]
            fft_rev = np.fft.rfft(rev_a, n=N_fft)
            fft_w = np.fft.rfft(w, n=N_fft)
            full = np.fft.irfft(fft_rev * fft_w, n=N_fft)[: 2 * n - 1]
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # callback to respect the global deadline
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Evolutionary search + L‑BFGS polishing for the step‑function objective.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Leave a safety margin for the final evaluate_sequence call and polishing.
    safety = 10.0
    deadline = start_time + budget_s - safety

    # ------------------------------------------------------------------
    # Problem size – choose a moderate n for good trade‑off speed / quality.
    # ------------------------------------------------------------------
    n = 4000
    max_conv = _make_conv_helper(n)                 # fast max‑convolution

    # ------------------------------------------------------------------
    # 1. Build a diverse seed pool (analytic families + random draws)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for sigma in (0.10, 0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma))
    for p in (0.2, 0.5, 1.0, 2.0, 4.0):
        seeds.append(_power(n, p))
    seeds.append(_sin2(n))
    for rate in (300.0, 600.0, 1200.0):
        seeds.append(_exponential(n, rate))
        seeds.append(_exponential_symmetric(n, rate))
    for a, b in ((0.5, 0.5), (0.2, 0.8), (1.5, 1.5), (2.0, 5.0), (5.0, 2.0)):
        seeds.append(_beta_family(n, a, b))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # Random Dirichlet draws with a range of concentration parameters
    conc_list = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0)
    for conc in conc_list:
        for _ in range(8):
            raw = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(raw)                     # raw
            seeds.append(np.sort(raw)[::-1])      # sorted descending

    # Include the previous best if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Normalise all seeds (ensure they are proper probability vectors)
    # ------------------------------------------------------------------
    clean_seeds = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            clean_seeds.append(v / s)
        else:
            clean_seeds.append(_uniform(n))
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 3. Evaluate seeds cheaply (exact surrogate = true objective for sum = 1)
    # ------------------------------------------------------------------
    seed_vals = [_objective_surrogate(v, n, max_conv) for v in seeds]
    order = np.argsort(seed_vals)
    # keep a modest elite set for the EA
    elite_size = 30
    population = [seeds[i].copy() for i in order[:elite_size]]
    best_idx = order[0]
    best_seq = seeds[best_idx].copy()
    best_val = seed_vals[best_idx]

    print(f"[Init] elite size = {elite_size}, best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Evolutionary optimisation (log‑softmax recombination + Gaussian mutation)
    # ------------------------------------------------------------------
    pop_size = elite_size
    elite_keep = 10                     # number of individuals kept unchanged each generation
    sigma_start = 0.5
    sigma_end = 0.01

    generation = 0
    while time.time() < deadline - 5.0:          # keep a few seconds for final polish
        generation += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = min(1.0, elapsed / total)
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        # ------------------------------------------------------------------
        # create children
        # ------------------------------------------------------------------
        children = []
        n_children = pop_size - elite_keep
        for _ in range(n_children):
            # parent selection (uniform among current population)
            i1, i2 = rng.choice(pop_size, size=2, replace=False)
            p1 = population[i1]
            p2 = population[i2]

            # work in log‑softmax space
            log1 = np.log(p1 + 1e-30)
            log2 = np.log(p2 + 1e-30)
            child_logit = 0.5 * (log1 + log2) + sigma * rng.normal(size=n)
            child = _softmax(child_logit)

            val = _objective_surrogate(child, n, max_conv)
            children.append((val, child))

        # ------------------------------------------------------------------
        # form next generation: keep elite_keep best of current pop + children
        # ------------------------------------------------------------------
        combined = [( _objective_surrogate(v, n, max_conv), v) for v in population[:elite_keep]]
        combined.extend(children)
        combined.sort(key=lambda x: x[0])
        population = [seq for _, seq in combined[:pop_size]]

        # ------------------------------------------------------------------
        # track global best
        # ------------------------------------------------------------------
        cur_best_val, cur_best_seq = combined[0]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_seq = cur_best_seq.copy()
            print(f"[Gen {generation:4d}] new global best surrogate = {best_val:.6f}")

        # occasionally print progress even if no improvement
        if generation % 100 == 0:
            print(f"[Gen {generation:4d}] best={best_val:.6f}, sigma={sigma:.4f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with continuation L‑BFGS‑B
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 1.0 and _opt is not None:
        print("\n[Polish] L‑BFGS refinement started")
        # we give each β level a share of the remaining budget
        max_iter_per_beta = max(1000, int(remaining_time * 2000 / len(
            [1,5,20,80,200,500,1_000,2_000,5_000,20_000,50_000,100_000,200_000,500_000,1_000_000])))
        polished = _lbfgs_smooth_optimize(best_seq,
                                          n,
                                          deadline,
                                          beta_schedule=[
                                              1.0, 5.0, 20.0, 80.0, 200.0,
                                              500.0, 1_000.0, 2_000.0, 5_000.0,
                                              20_000.0, 50_000.0, 100_000.0,
                                              200_000.0, 500_000.0, 1_000_000.0,
                                          ],
                                          max_iter_per_beta=max_iter_per_beta)
        polished_val = _objective_surrogate(polished, n, max_conv)
        print(f"[Polish] polished surrogate = {polished_val:.6f}")
        if polished_val < best_val:
            best_seq = polished
            best_val = polished_val

    # ------------------------------------------------------------------
    # 6. Final evaluation with the official evaluator (ensures clamping, etc.)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑start in the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
