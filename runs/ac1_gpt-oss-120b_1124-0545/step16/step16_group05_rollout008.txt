"""
Improved optimizer for the step‑function bound

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)²)

All candidates are ℓ₁‑normalised (∑a = 1) – then the objective
reduces to `2·n·max(conv)`.  The code follows a multi‑stage pipeline
similar to the reference implementation, but with two key upgrades:

1. **Richer seeding.**  In addition to the classic windows,
   exponential / power‑law families we also generate
   Tukey‑window and Gaussian‑window families and a finer sweep of
   parameters.  Random Dirichlet draws with a wide range of
   concentration parameters are also added.

2. **“Min‑M’’ linear‑programming inflation.**  Instead of maximising the
   ℓ₁‑mass of a feasible vector (the original mass‑inflation step) we
   directly solve the LP

        minimise   M
        s.t.       (f * g)_k ≤ M   ∀k,
                   g ≥ 0,
                   ∑ g = 1,

   which yields a vector `g` that provably reduces the maximal
   convolution value.  The LP is solved with SciPy’s HiGHS driver
   (sparse matrix, fast for n≈4000).  The resulting `g` is mixed with
   the current iterate and the process is repeated a few times.

The remainder of the pipeline (smooth L‑BFGS on a log‑sum‑exp
surrogate, long exponentiated‑gradient polishing, stochastic restarts)
is kept but hyper‑parameters are tuned for a tighter budget.

The function respects the supplied wall‑clock budget (`budget_s`) and
stores the best sequence globally as ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective (no ℓ₁ normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = 2·n·max(conv) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window.  0 ≤ alpha ≤ 1."""
    if alpha <= 0:
        return _uniform(n)
    if alpha >= 1:
        # alpha=1 gives a Hann window
        return _hann_window(n)
    x = np.arange(n) / (n - 1)
    w = np.ones_like(x)
    first = x < alpha / 2
    w[first] = 0.5 * (1 + np.cos(np.pi * (2 * x[first] / alpha - 1)))
    last = x > 1 - alpha / 2
    w[last] = 0.5 * (1 + np.cos(np.pi * (2 * (x[last] - 1) / alpha + 1)))
    return w / w.sum()


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the window."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution.
    `start_vec` must be ℓ₁‑normalised.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5,
            5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                       # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp of the convolution entries
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = minimize(fun=fun,
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter_per_beta,
                                    'disp': False,
                                    'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (entropic parametrisation).
    Returns best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Simple AdaGrad‑style step‑size schedule
        eta = 0.6 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Convolution pattern (Toeplitz matrix) cache – used by LP stages
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


# ----------------------------------------------------------------------
# Linear‑programming “min‑M’’ inflation step
# ----------------------------------------------------------------------
def _solve_minmax_lp(f: np.ndarray,
                     rows: np.ndarray,
                     cols: np.ndarray,
                     offset: np.ndarray,
                     deadline: float):
    """
    Solve
        minimise   M
        s.t.       (f * g)_k ≤ M      ∀k,
                   g ≥ 0,
                   ∑ g = 1 .
    Returns the optimal g and the attained M.
    """
    n = len(f)

    # Build the sparse matrix that maps g → f * g
    data = f[offset]                             # picks the coefficient f_i for each entry
    A_f = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Append the column for -M (so that A_f * g - M ≤ 0)
    neg_one_col = sp.csr_matrix((-np.ones(2 * n - 1),
                                 (np.arange(2 * n - 1), np.full(2 * n - 1, n))),
                                 shape=(2 * n - 1, n + 1))
    A_ub = sp.hstack([A_f, neg_one_col])  # shape (2n‑1, n+1)

    c = np.concatenate([np.zeros(n), np.array([1.0])])  # minimise M
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]  # g ≥ 0, M ≥ 0

    # Solve with HiGHS (fast for our moderate size)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception as e:  # pragma: no cover
        print("[LP‑min] linprog raised:", e)
        return None, None

    if not res.success or res.x is None:
        return None, None

    g_opt = np.maximum(res.x[:n], 0.0)
    M_opt = float(res.x[-1])
    return g_opt, M_opt


def _lp_minmax_inflation(seq: np.ndarray,
                         deadline: float,
                         max_outer_iters: int = 8) -> np.ndarray:
    """
    Repeatedly solve the min‑M LP, blend the result with the current
    iterate and keep the improvement.  Stops when no further reduction
    is observed or the time budget is exhausted.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M_cur = float(conv.max())

        g_opt, M_opt = _solve_minmax_lp(seq, rows, cols, offset, deadline)
        if g_opt is None:
            break

        # If the LP cannot improve the maximal convolution, stop.
        if M_opt >= M_cur - 1e-14:
            break

        # Blend a fraction of g_opt into the current sequence.
        # A moderate mixing factor (0.5) works well; we also try a smaller
        # factor if the first attempt does not lower the true max.
        for t in (0.5, 0.25):
            new_seq = (1 - t) * seq + t * g_opt
            new_seq /= new_seq.sum()
            new_M = float(_conv_fft(new_seq, new_seq).max())
            if new_M < M_cur:
                seq = new_seq
                print(f"[LP‑min] improved max conv from {M_cur:.6e} to {new_M:.6e} (t={t})")
                break
        else:
            # No improvement with either blending factor → stop.
            break

    return seq


# ----------------------------------------------------------------------
# Seed‑generation helpers (grid sweeps etc.)
# ----------------------------------------------------------------------
def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.1, 8.0, 40)      # richer sweep extended to 8
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.1, 10.0, 20)
    betas = np.linspace(0.1, 10.0, 20)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vec = _beta_shape(n, a, b)
            vals.append((vec, _objective_surrogate(vec), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.5, 8.0, 45):
        if time.time() > deadline:
            return vals
        vec = _power(n, p)
        vals.append((vec, _objective_surrogate(vec), ('power', p)))
    return vals


def _grid_search_tukey(n: int, deadline: float):
    vals = []
    for alpha in np.linspace(0.0, 1.0, 11):
        if time.time() > deadline:
            break
        vec = _tukey_window(n, alpha)
        vals.append((vec, _objective_surrogate(vec), ('tukey', alpha)))
    return vals


def _grid_search_gaussian(n: int, deadline: float):
    vals = []
    centre = (n - 1) / 2.0
    for factor in np.linspace(0.05, 0.5, 10):
        if time.time() > deadline:
            break
        sigma = factor * centre
        vec = _gaussian_window(n, sigma)
        vals.append((vec, _objective_surrogate(vec), ('gauss', factor)))
    return vals


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 strikes a good balance for LP and FFT work.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Load a previous solution if it exists (same n)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Seed generation – diverse analytic families + random draws
    # --------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # Uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Include previous best if available
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # Parameter sweeps (richer than the previous run)
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))
    candidate_pool.extend(_grid_search_tukey(n, deadline))
    candidate_pool.extend(_grid_search_gaussian(n, deadline))

    # Random Dirichlet draws (wide concentration spectrum)
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # --------------------------------------------------------------
    # 2. Keep the most promising seeds (sorted by surrogate value)
    # --------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # --------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement on elite seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=6000,                 # a solid budget per β
            beta_schedule=None                      # default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (time‑aware)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Allocate most of the remaining time (≈70 %) to EG polishing.
        max_exp = int(min(3_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Linear‑programming “min‑M’’ inflation
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming min‑M inflation ---")
        seq_lp = _lp_minmax_inflation(global_best_seq, deadline, max_outer_iters=8)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP‑min] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Small post‑LP EG polish (fast, catches any leftover slack)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp2 = int(min(800_000, remaining * 0.04 * 1e5))
        print(f"\n--- Post‑LP EG polish (max_iters={max_exp2}) ---")
        seq_post, val_post = _exp_gradient_opt(global_best_seq, n, deadline,
                                               max_iters=max_exp2)
        if val_post < global_best_val:
            global_best_seq = seq_post.copy()
            global_best_val = val_post
            print(f"[Post‑LP EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Stochastic restarts – perturb + short L‑BFGS + EG
    # --------------------------------------------------------------
    restarts = 4
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{4 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS with a compact β schedule
        seq_restart = _lbfgs_smooth_optimize(
            start_vec,
            n,
            deadline,
            beta_schedule=[1.0, 5.0, 20.0, 80.0, 200.0, 500.0],
            max_iter_per_beta=2500
        )
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            # quick EG polishing for the restarted candidate
            seq_restart, val_restart = _exp_gradient_opt(seq_restart, n, deadline,
                                                         max_iters=200_000)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
