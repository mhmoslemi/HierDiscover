"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Seed generation** – a modest collection of analytic windows,
   Gaussian / beta‑shape families and a few random Dirichlet draws.
   All seeds are ℓ₁‑normalised (sum = 1) and the best few are kept.

2. **Adam optimisation on a smoothed‑max surrogate** – the variable lives in
   log‑softmax space guaranteeing non‑negativity and the simplex constraint.
   A temperature schedule β grows from ≈0.5 to ≈5·10⁴, turning the smooth
   log‑sum‑exp approximation of the max into the true max.  Adam (with
   bias‑correction and a 1/√t learning‑rate decay) follows the gradient of
   the surrogate.  The true objective (using the exact max) is monitored and
   the best vector seen is retained.

3. **Final evaluation** – the best vector found is passed to the official
   ``evaluate_sequence`` function and stored in the global variable
   ``height_sequence_1`` for possible reuse.

The optimiser runs until the time budget (default 1000 s) is exhausted,
always returning a valid list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution, softmax)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact surrogate (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(conv(a, a))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on smoothed‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_start: float = 0.5,
                   beta_end: float = 5e4,
                   lr0: float = 0.5,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam in log‑softmax space.
    The smooth max is approximated by a log‑sum‑exp with a temperature β
    that grows from ``beta_start`` to ``beta_end`` during the run.
    Returns the best vector seen and its exact surrogate value.
    """
    # initialise log‑probabilities
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    v_m = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    best_seq = start_vec.copy()
    best_val = _objective_surrogate(best_seq)

    start_opt = time.time()
    while time.time() < deadline and t < max_iters:
        t += 1
        # schedule β exponentially with the fraction of time elapsed
        elapsed = time.time() - start_opt
        total = deadline - start_opt
        frac = elapsed / total if total > 0 else 0.0
        beta = beta_start * (beta_end / beta_start) ** frac

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)              # full convolution, length 2n‑1
        maxc = np.max(conv)

        # track the best exact value (no smoothing!)
        cur_val = 2.0 * n * maxc
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if t % 10000 == 0:
                print(f"[Adam] iter={t:,}, β={beta:.1f}, best={best_val:.6f}")

        # smooth‑max using log‑sum‑exp
        shifted = conv - maxc                # ≤ 0, max entry becomes 0
        exp_beta = np.exp(beta * shifted)    # stabilised
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp               # weights (sum to 1)

        # gradient w.r.t. a : 2 * conv(reverse(a), w)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)      # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # learning‑rate schedule (1/√t) – helps later convergence
        lr = lr0 / math.sqrt(t)

        # Adam bias‑corrected update
        m = beta1 * m + (1 - beta1) * grad_v
        v_m = beta2 * v_m + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v_m / (1 - beta2 ** t)
        v -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_seq, best_val

        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Adam optimisation – divide remaining time among the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        per_seed = remaining / len(top_seeds)
        for idx, seed_vec in enumerate(top_seeds):
            seed_deadline = time.time() + per_seed
            print(f"\n--- Adam optimisation (seed #{idx+1}) ---")
            seq_opt, val_opt = _adam_optimize(seed_vec,
                                               n,
                                               seed_deadline,
                                               beta_start=0.5,
                                               beta_end=5e4,
                                               lr0=0.5,
                                               max_iters=5_000_000)
            print(f"[Seed {idx+1}] best surrogate = {val_opt:.6f}")
            if val_opt < global_best_val:
                global_best_val = val_opt
                global_best_seq = seq_opt.copy()
                print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for potential reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
