"""
Improved step‑function optimizer

The goal is to minimise 2·n·max(conv(a,a))/(sum(a)²) with a non‑negative
sequence `a`.  The algorithm works as follows (all within the given
wall‑clock budget, default 1000 s):

1. **Problem size** – we use a fixed length `n = 4000`.  This gives a good
   trade‑off between expressive power and per‑iteration cost.

2. **Rich seeding** – a small pool of candidates is built from
   * classic windows (Hann, Hamming, Blackman, Kaiser);
   * exponential families (forward / reverse);
   * a *Gaussian* family (new in this version);
   * a fine Dirichlet sweep; and the previous best solution if any.
   The surrogate objective `2·n·max(conv)` (i.e. the true objective with the
   ℓ₁‑norm already fixed to 1) is used to keep only the most promising
   seeds.

3. **Smooth L‑BFGS** – each seed is refined with L‑BFGS‑B on a
   log‑sum‑exp smooth approximation of the max‑convolution.  A modest
   continuation schedule (`β = 1 … 1e5`) is employed.

4. **Long exponentiated‑gradient (mirror‑descent) polishing** – a
   sub‑gradient descent on the exact max‑convolution is performed for as
   many iterations as time permits.  The step‑size follows an
   AdaGrad‑style decay `η = 0.8/√t`.

5. **Linear‑programming mass‑inflation** – repeated LP steps (up to 12
   outer iterations) maximise the ℓ₁‑mass of a vector `g` under the
   constraint `(f * g) ≤ M` where `M` is the current maximal convolution
   value.  The inflated vector is mixed back into `f` (mixing factor ½)
   and the process repeats, yielding a provable reduction of `M`.

6. **Second EG polishing** – after the LP phase we run a shorter EG
   pass to exploit any slack created by the mixing.

7. **Stochastic restarts** – a few random perturbations of the best
   solution are fed through a short L‑BFGS + EG cycle to escape shallow
   local minima.

8. **Final evaluation** – the best sequence is stored in the global
   variable ``height_sequence_1`` and returned as a plain list.

All heavy work is done with NumPy / SciPy (FFT for convolutions,
sparse matrices for the LP).  The implementation respects the 1000 s
budget and prints progress information.
"""

import scipy.sparse as sp

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (real) convolution via FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no ℓ₁‑normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for an ℓ₁‑normalised vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = np.i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _gaussian_window(n: int, sigma_factor: float) -> np.ndarray:
    """Symmetric Gaussian (normalised) with σ = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    if sigma <= 0:
        return _uniform(n)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    import scipy.optimize as _opt

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            1e4, 2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                     # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                    # weights in the smoothed max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)

# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (softmax parametrisation)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.8 / math.sqrt(it)       # a slightly larger step size
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols                     # i = k - j
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        subject to g ≥ 0,
                   (f * g)_k ≤ M   ∀k .
    Returns the optimal g and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray, deadline: float,
                       max_outer_iters: int = 12) -> np.ndarray:
    """
    Perform several “mass‑inflation’’ LP steps.
    Each iteration solves the LP described above, rescales the solution
    to unit ℓ₁‑mass and mixes it with the current iterate.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            # No ℓ₁‑mass increase possible.
            break

        g_scaled = g / sum_g                     # renormalise to sum = 1
        t = 0.5                                   # blending factor
        new_seq = (1 - t) * seq + t * g_scaled
        new_seq = new_seq / new_seq.sum()         # safety normalisation

        new_M = _conv_fft(new_seq, new_seq).max()
        if new_M < M:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            # try a smaller blend
            t = 0.25
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M:
                seq = new_seq
                print(f"[LP] improved (small t) max conv to {new_M:.6e}")
            else:
                break
    return seq

def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.2, 6.0, 35)      # richer sweep for exponentials
    for c in cs:
        if time.time() > deadline:
            break
        vec = _exp_family(n, c)
        vals.append((vec, _objective_surrogate(vec), ('exp', c)))
        rev = _rev_exp_family(n, c)
        vals.append((rev, _objective_surrogate(rev), ('revexp', c)))
    return vals


def _grid_search_gaussian(n: int, deadline: float):
    vals = []
    sigmas = np.linspace(0.04, 0.45, 20)   # sigma as fraction of n
    for s in sigmas:
        if time.time() > deadline:
            break
        vec = _gaussian_window(n, s)
        vals.append((vec, _objective_surrogate(vec), ('gauss', s)))
    return vals


def _grid_search_dirichlet(n: int, deadline: float, rng: np.random.Generator):
    vals = []
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            vals.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))
    return vals

        max_exp2 = int(min(1_500_000, remaining * 0.05 * 1e5))
        print(f"\n--- Second EG polishing (max_iters={max_exp2}) ---")
        seq_exp2, val_exp2 = _exp_gradient_opt(global_best_seq, n, deadline,
                                               max_iters=max_exp2)
        if val_exp2 < global_best_val:
            global_best_seq = seq_exp2.copy()
            global_best_val = val_exp2
            print(f"[ExpGrad2] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Stochastic restarts
    # ------------------------------------------------------------------
    restarts = 5
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{5 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.12, size=n)
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS
        seq_restart = _lbfgs_smooth_optimize(start_vec, n, deadline,
                                             beta_schedule=[1.0, 5.0, 20.0, 80.0],
                                             max_iter_per_beta=2000)
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            # short EG polishing
            seq_restart, val_restart = _exp_gradient_opt(seq_restart, n, deadline,
                                                         max_iters=250_000)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
