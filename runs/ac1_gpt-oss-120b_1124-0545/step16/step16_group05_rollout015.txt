"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The search combines:

* a rich pool of analytical families (uniform, beta, power, exponential,
  classic DSP windows) and many random Dirichlet draws;
* a coarse selection of the most promising seeds;
* a long L‑BFGS pass on the smooth log‑sum‑exp surrogate of the max‑convolution;
* an exact exponentiated‑gradient (mirror‑descent) phase that runs until the
  time budget is exhausted;
* a few “mass‑inflation’’ LP refinements with an adaptive mixing factor;
* a final stochastic hill‑climbing stage that transfers tiny amounts of mass
  between random positions.

All stages keep the sequence ℓ₁‑normalised (sum = 1) – the objective is
scale‑invariant – and respect the supplied wall‑clock budget.
The best sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as _sp_opt
from scipy.optimize import linprog
from scipy.special import i0

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the official evaluator (no normalisation assumed).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate = exact objective when the vector is ℓ₁‑normalised (∑a = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(-c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0))
    )
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a / a.sum()

# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    if beta_schedule is None:
        # aggressive schedule, ends with a very sharp approximation of the max
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights in the smoothed max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _sp_opt.minimize(fun=fun,
                                   x0=v,
                                   method='L-BFGS-B',
                                   jac=True,
                                   callback=callback,
                                   options={'maxiter': max_iter_per_beta,
                                            'disp': False,
                                            'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _exp_gradient_until_deadline(start_seq: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 alpha: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution.
    Runs until the time limit is hit.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        max_idx = int(np.argmax(conv))

        # sub‑gradient: g = 2·conv_rev(a, e_max) where e_max is the one‑hot at max_idx
        rev_a = a[::-1]
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[max_idx] = 1.0
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]  # length n

        # basic step‑size schedule (AdaGrad‑style)
        eta = alpha / math.sqrt(it)
        log_a = log_a - eta * grad
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional status report
        if it % 250_000 == 0:
            print(f"[ExpGD] it={it:,d} best={best_val:.6f}")

    return best_seq, best_val

_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """
    Linear programme:
        maximise   Σ g_j
        s.t.       g ≥ 0,
                   (f * g)_k ≤ M   ∀k .
    Returns the optimal g (as a dense vector) and its ℓ₁‑mass.
    """
    n = len(f)
    data = f[offset]                     # picks the appropriate coefficient f_i
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                      # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()                 # fallback – no improvement
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray,
                       deadline: float,
                       max_outer_iters: int = 8) -> np.ndarray:
    """
    Perform a handful of LP “mass‑inflation’’ refinements.
    After each LP solve we search a small 1‑D grid of mixing factors
    t ∈ {0.5,0.25,0.125,0.0625} and keep the best.
    """
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 1.0 and max_outer_iters > 0:
        max_outer_iters -= 1

        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:                # LP cannot increase ℓ₁‑mass
            break

        g_scaled = g / sum_g                     # renormalise to sum = 1

        # try a handful of mixing factors
        best_seq = None
        best_M = M
        for t in (0.5, 0.25, 0.125, 0.0625):
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < best_M:
                best_M = new_M
                best_seq = new_seq
        if best_seq is not None:
            seq = best_seq
            print(f"[LP] improved max conv → {best_M:.6e}")
        else:
            break
    return seq


# ----------------------------------------------------------------------
# Local random hill‑climbing (mass‑transfer) stage
# ----------------------------------------------------------------------
def _local_hill_climb(seq: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Small random mass transfers between two positions.
    Accept only strict improvements (greedy descent) – cheap but
    effective for fine‑tuning the final sequence.
    """
    cur = seq.copy()
    cur_val = _objective_surrogate(cur)

    it = 0
    while time.time() < deadline and it < max_iters:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # perturb magnitude scaled to current typical entry size
        eps = rng.uniform(-0.002, 0.002) * cur[i]
        if cur[i] - eps < 0.0:
            continue
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # ℓ₁‑mass unchanged, but re‑normalise for numerical safety
        new = new / new.sum()
        new_val = _objective_surrogate(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
        it += 1
        if it % 50_000 == 0:
            print(f"[Hill] it={it:,d} best={cur_val:.6f}")

    return cur, cur_val


# ----------------------------------------------------------------------
# Seeding helpers
# ----------------------------------------------------------------------
def _grid_search_exponential(n: int, deadline: float):
    vals = []
    cs = np.linspace(0.5, 4.0, 30)
    for c in cs:
        if time.time() > deadline:
            break
        vals.append((_exp_family(n, c), _objective_surrogate(_exp_family(n, c)), ('exp', c)))
        vals.append((_rev_exp_family(n, c), _objective_surrogate(_rev_exp_family(n, c)), ('revexp', c)))
    return vals


def _grid_search_beta(n: int, deadline: float):
    vals = []
    alphas = np.linspace(0.1, 6.0, 13)
    betas = np.linspace(0.1, 6.0, 13)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                return vals
            vals.append((_beta_shape(n, a, b), _objective_surrogate(_beta_shape(n, a, b)), ('beta', a, b)))
    return vals


def _grid_search_power(n: int, deadline: float):
    vals = []
    for p in np.linspace(-0.5, 6.0, 35):
        if time.time() > deadline:
            return vals
        vals.append((_power(n, p), _objective_surrogate(_power(n, p)), ('power', p)))
    return vals

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine observes the supplied time budget and
    stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good balance
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution, if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (a diverse pool)
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # baseline uniform
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # previous best (if available)
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # classic DSP windows
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # parametric sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # random Dirichlet draws across a wide concentration spectrum
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 60
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep the most promising seeds (coarse selection)
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 14
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Generated {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement of elite seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None          # use default aggressive schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exponentiated‑gradient polishing (runs until time runs out)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (full‑budget) ---")
        seq_exp, val_exp = _exp_gradient_until_deadline(global_best_seq,
                                                        n,
                                                        deadline,
                                                        alpha=0.8)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGD] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming mass‑inflation refinements (more outer loops)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming mass‑inflation refinement ---")
        seq_lp = _lp_inflation_step(global_best_seq, deadline, max_outer_iters=8)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Minor stochastic restarts (perturb + short L‑BFGS + EG)
    # ------------------------------------------------------------------
    restarts = 3
    while restarts > 0 and time.time() < deadline:
        restarts -= 1
        print(f"\n--- Stochastic restart #{3 - restarts} ---")
        log_best = np.log(global_best_seq + 1e-30)
        pert = rng.normal(scale=0.10, size=n)          # smaller perturbation
        start_log = log_best + pert
        start_vec = _softmax(start_log)

        # short L‑BFGS (very few β values)
        seq_restart = _lbfgs_smooth_optimize(start_vec,
                                              n,
                                              deadline,
                                              beta_schedule=[1.0, 5.0, 20.0, 80.0],
                                              max_iter_per_beta=2000)
        val_restart = _objective_surrogate(seq_restart)

        if val_restart < global_best_val:
            # polish with a moderate EG run
            seq_restart, val_restart = _exp_gradient_until_deadline(seq_restart,
                                                                     n,
                                                                     deadline,
                                                                     alpha=0.6)
            if val_restart < global_best_val:
                global_best_seq = seq_restart.copy()
                global_best_val = val_restart
                print(f"[Restart] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climbing fine‑tuning ---")
        seq_hc, val_hc = _local_hill_climb(global_best_seq,
                                           n,
                                           deadline,
                                           rng,
                                           max_iters=300_000)
        if val_hc < global_best_val:
            global_best_seq = seq_hc.copy()
            global_best_val = val_hc
            print(f"[HillClimb] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
