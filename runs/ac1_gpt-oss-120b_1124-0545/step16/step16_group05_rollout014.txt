def _targeted_move(a):

    max_idx = int(conv.argmax())
    # compute product per i

    idxs = np.arange(i_start, i_end+1)
    rev_idx = max_idx - idxs
    prod = a[idxs] * a[rev_idx]
    i0 = idxs[np.argmax(prod)]
    j0 = rev_idx[np.argmax(prod)]

    # pick random target positions not i0 or j0
    cand = rng.choice(n, size=2, replace=False)
    while cand[0] in (i0, j0) or cand[1] in (i0, j0):
        cand = rng.choice(n, size=2, replace=False)
    # amount to move
    delta = min(a[i0], a[j0]) * 0.5 # move half of smallest
    # ensure delta > 0
    if delta < 1e-12:
        return a, False
    a_new = a.copy()
    # move delta from both i0 and j0 to cand positions equally
    a_new[i0] -= delta
    a_new[j0] -= delta
    a_new[cand[0]] += delta
    a_new[cand[1]] += delta
    return a_new, True

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Return smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))

def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective for a probability vector (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _rev_exp_family(n: int, c: float) -> np.ndarray:
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()

def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()

def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()

def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = np.i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()

def _dpss_window(n: int, NW: float = 2.5) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window, ℓ₁‑normalised."""
    # scipy.signal.windows.dpss returns array of length n
    a = windows.dpss(n, NW, sym=True)
    a = np.asarray(a, dtype=np.float64)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()

# Smooth L‑BFGS optimisation (log‑sum‑exp surrogate)

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy optimiser required for L‑BFGS.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5, 2e5,
            5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on the simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp of (beta * (conv - shift))
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max
            w = exp_beta / sum_exp                     # weights
            rev_a = a[::-1]
            # full convolution of rev_a with w gives the gradient wrt a (size 3n‑2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through the soft‑max
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            # print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Exact sub‑gradient exponentiated‑gradient optimiser
# ---------------------------------------------------------------
def _exp_gradient_opt(seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution using correct sub‑gradient."""
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute sub‑gradient of the active entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        grad = np.zeros_like(a)
        # gradient component = a[rev_idx] + a[idx_i]
        grad[idx_i] = a[rev_idx] + a[idx_i]

        # multiplicative (mirror) update in log‑space
        eta = 0.6 / math.sqrt(it)          # step‑size schedule
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)
        log_a = log_a - eta * grad_v
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # optional progress print
            if it % 250_000 == 0:
                print(f"[ExpGrad] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

_conv_pattern_cache = {}

def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset

def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    n = len(f)
    data = f[offset]                         # coefficient for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # maximise Σg ↔ minimise −Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()

def _lp_inflation_step(seq: np.ndarray, deadline: float,
                       max_outer_iters: int = 5) -> np.ndarray:
    """Perform a few LP “mass‑inflation’’ refinements."""
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1
        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_scaled = g / sum_g
        t = 0.5
        new_seq = (1 - t) * seq + t * g_scaled
        new_seq = new_seq / new_seq.sum()
        new_M = _conv_fft(new_seq, new_seq).max()
        if new_M < M:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            t = 0.25
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M:
                seq = new_seq
                print(f"[LP] improved (small t) max conv to {new_M:.6e}")
            else:
                break
    return seq

# Targeted local search (few thousand hill‑climbing steps)
# ---------------------------------------------------------------
def _targeted_local_search(seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from the pair that contributes most
    to the current max‑conv to two random locations. Accept only strict
    improvements. Returns the best sequence seen and its surrogate value."""
    a = seq.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_max = conv.max()
    best_val = 2.0 * n * best_max
    rng = np.random.default_rng(20231101)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # recompute conv only when a changes
        conv = _conv_fft(a, a)
        max_idx = int(conv.argmax())

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        product = a[idx_i] * a[rev_idx]
        # pick the pair with largest product (largest contribution)
        k = np.argmax(product)
        i0 = idx_i[k]

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no normalisation assumed)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return math.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate = exact objective when seq sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_family(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(‑c·i/n) for i = 0 … n‑1."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _rev_exp_family(n: int, c: float) -> np.ndarray:
    """Reverse exponential – a_i ∝ exp(‑c·(n‑1‑i)/n)."""
    eps = c / n
    r = math.exp(-eps)
    a = r ** (np.arange(n, dtype=np.float64)[::-1])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1.0))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1.0))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1.0)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float = 14.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = np.i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _dpss_window(n: int, NW: float = 2.5) -> np.ndarray:
    """Slepian (DPSS) window, ℓ₁‑normalised."""
    a = windows.dpss(n, NW, sym=True)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000) -> np.ndarray:
    """L‑BFGS‑B on the log‑sum‑exp surrogate of the max‑convolution."""
    try:
        import scipy.optimize as _opt
    except Exception as e:  # pragma: no cover
        raise RuntimeError("SciPy optimiser required for L‑BFGS.") from e

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1e3, 2e3, 5e3,
            1e4, 2e4, 5e4, 1e5, 2e5,
            5e5, 1e6, 2e6, 5e6, 1e7,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parametrisation

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)                      # stays on simplex
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights in smoothed max
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)

_conv_pattern_cache = {}


def _get_conv_pattern(n: int):
    """Return (rows, cols, offset) describing the Toeplitz convolution matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]

    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_list.append(np.full(length, k, dtype=np.int32))
        cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offset = rows - cols
    _conv_pattern_cache[n] = (rows, cols, offset)
    return rows, cols, offset


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      rows: np.ndarray, cols: np.ndarray, offset: np.ndarray):
    """Linear programme:
       maximise Σ g_j  s.t. g ≥ 0, (f * g)_k ≤ M ∀k.
    """
    n = len(f)
    data = f[offset]                         # picks the coefficient f_i for each entry
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                          # minimise -Σg → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success and res.x is not None:
        g_opt = np.maximum(res.x, 0.0)
    else:
        g_opt = f.copy()
    return g_opt, g_opt.sum()


def _lp_inflation_step(seq: np.ndarray, deadline: float,
                       max_outer_iters: int = 5) -> np.ndarray:
    """Perform a few “mass‑inflation’’ LP steps."""
    n = len(seq)
    rows, cols, offset = _get_conv_pattern(n)

    while time.time() < deadline - 2.0 and max_outer_iters > 0:
        max_outer_iters -= 1
        conv = _conv_fft(seq, seq)
        M = conv.max()
        g, sum_g = _solve_lp_max_sum(seq, M, rows, cols, offset)

        if sum_g <= 1.0 + 1e-12:
            break

        g_scaled = g / sum_g                     # renormalise to sum = 1
        t = 0.5
        new_seq = (1 - t) * seq + t * g_scaled
        new_seq = new_seq / new_seq.sum()
        new_M = _conv_fft(new_seq, new_seq).max()
        if new_M < M:
            seq = new_seq
            print(f"[LP] improved max conv from {M:.6e} to {new_M:.6e}")
        else:
            t = 0.25
            new_seq = (1 - t) * seq + t * g_scaled
            new_seq = new_seq / new_seq.sum()
            new_M = _conv_fft(new_seq, new_seq).max()
            if new_M < M:
                seq = new_seq
                print(f"[LP] improved (small t) max conv to {new_M:.6e}")
            else:
                break
    return seq

def _targeted_local_search(seq: np.ndarray,

                           max_iters: int = 30000) -> tuple[np.ndarray, float]:

    max‑convolution column. Accept only strict improvements."""
    a = seq.copy()
    conv = _conv_fft(a, a)
    best_max = conv.max()
    best_val = 2.0 * n * best_max
    best_seq = a.copy()

        it += 1
        conv = _conv_fft(a, a)

        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i

        prod = a[idx_i] * a[rev_idx]
        k = np.argmax(prod)
        i0 = idx_i[k]
        j0 = rev_idx[k]

            continue

        cand = rng.choice(n, size=2, replace=False)

            cand = rng.choice(n, size=2, replace=False)

        a_new = a.copy()
        a_new[i0] -= delta
        a_new[j0] -= delta
        a_new[cand[0]] += delta
        a_new[cand[1]] += delta

        if (a_new < 0).any():
            continue

        new_max = _conv_fft(a_new, a_new).max()
        if new_max < best_max - 1e-12:
            a = a_new
            best_seq = a_new.copy()
            best_max = new_max
            best_val = 2.0 * n * best_max
            if it % 2000 == 0:
                print(f"[LocalSearch] it={it} best surrogate={best_val:.6f}")

    return best_seq, best_val

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Multi‑stage optimizer for the step‑function objective.

    1.  Generate a diverse pool of ℓ₁‑normalised seeds (windows, DPSS,
        exponential / beta / power families, random Dirichlet draws,
        previous best).
    2.  Keep the best handful and refine each with a smooth L‑BFGS optimiser.
    3.  Polish the current best with a long exponentiated‑gradient pass that
        uses the exact sub‑gradient of the max‑convolution.
    4.  Apply a few LP “mass‑inflation’’ steps.
    5.  Perform a targeted local‑search hill‑climbing phase.
    6.  Finish with a short ultra‑high‑β L‑BFGS polish.
    7.  Return the best sequence (stored globally as ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if any
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    print("\n=== Seed generation ===")
    candidate_pool = []

    # Uniform baseline
    candidate_pool.append((_uniform(n), _objective_surrogate(_uniform(n)), ('uniform',)))

    # Previous best
    if prev_arr is not None:
        candidate_pool.append((prev_arr, _objective_surrogate(prev_arr), ('previous',)))

    # Classic DSP windows + DPSS
    for win_name, win_func in [
        ('hann', _hann_window),
        ('hamming', _hamming_window),
        ('blackman', _blackman_window),
        ('kaiser', lambda nn: _kaiser_window(nn, beta=14.0)),
        ('dpss', lambda nn: _dpss_window(nn, NW=2.5)),
    ]:
        vec = win_func(n)
        candidate_pool.append((vec, _objective_surrogate(vec), (win_name,)))

    # Parametric sweeps
    candidate_pool.extend(_grid_search_exponential(n, deadline))
    candidate_pool.extend(_grid_search_beta(n, deadline))
    candidate_pool.extend(_grid_search_power(n, deadline))

    # Random Dirichlet draws
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_conc = 30
    for conc in concs:
        for _ in range(draws_per_conc):
            if time.time() > deadline:
                break
            vec = _random_dirichlet(n, conc, rng)
            candidate_pool.append((vec, _objective_surrogate(vec), ('dirichlet', conc)))

    # ------------------------------------------------------------------
    # 2. Keep top seeds
    # ------------------------------------------------------------------
    candidate_pool.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [vec for vec, _, _ in candidate_pool[:top_k]]
    print(f"Collected {len(candidate_pool)} seeds, keeping top {top_k}.")

    # ------------------------------------------------------------------
    # 3. Smooth L‑BFGS refinement
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing all L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run on seed #{idx + 1} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            max_iter_per_beta=8000,
            beta_schedule=None  # use aggressive default schedule
        )
        val_opt = _objective_surrogate(seq_opt)
        print(f"[L‑BFGS] surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_exp = int(min(3_000_000, remaining * 0.07 * 1e5))
        print(f"\n--- Exponentiated‑gradient polishing (max_iters={max_exp}) ---")
        seq_exp, val_exp = _exp_gradient_opt(global_best_seq, n, deadline,
                                             max_iters=max_exp)
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[ExpGrad] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Linear‑programming mass‑inflation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Linear‑programming mass‑inflation refinement ---")
        seq_lp = _lp_inflation_step(global_best_seq, deadline, max_outer_iters=4)
        val_lp = _objective_surrogate(seq_lp)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[LP] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Targeted local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Targeted local search (hill‑climbing) ---")
        seq_loc, val_loc = _targeted_local_search(global_best_seq, n,
                                                  deadline, max_iters=30000)
        if val_loc < global_best_val:
            global_best_seq = seq_loc.copy()
            global_best_val = val_loc
            print(f"[Local] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final ultra‑high‑β L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS polish with ultra‑high β ---")
        final_seq = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            max_iter_per_beta=4000,
            beta_schedule=[1e7, 2e7, 5e7, 1e8, 2e8, 5e8, 1e9]
        )
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq.copy()
            global_best_val = final_val
            print(f"[Final L‑BFGS] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
