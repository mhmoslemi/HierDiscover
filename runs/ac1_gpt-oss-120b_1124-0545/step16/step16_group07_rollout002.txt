Search for a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the full linear convolution.  The evaluation function

Algorithm
---------
1. **Seed generation** – build a modest collection of ℓ₁‑normalised vectors:

   few random Dirichlet draws.

2. **Seed ranking** – evaluate the exact surrogate

3. **Smooth‑max gradient descent** – minimise a differentiable
   approximation of the max‑convolution.

        F_τ(a) = τ·log(∑_k exp(conv_k / τ))

   The gradient of F_τ w.r.t. the soft‑max logits is obtained analytically
   and requires two FFT‑based convolutions per iteration.  The temperature
   τ is annealed through a short schedule (0.05 → 0.0005).  The iterates
   live in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑normalisation.

4. **Local random walk** – a final simulated‑annealing‑like polishing
   adds decreasing Gaussian noise to the logits and keeps improving moves.

5. **Evaluation & return** – the best sequence encountered is stored in the
   global variable ``height_sequence_1`` and returned as a plain list of
   floats.

All phases respect the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate = 2·n·max(conv) for an ℓ₁‑normalised vector.
    ``seq`` must already sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle; sigma = sigma_frac·n."""
    sigma = max(sigma_frac, 1e-12) * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc``."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _smooth_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Annealed smooth‑max gradient descent in log‑softmax space.
    The temperature τ is reduced through a short schedule.
    Returns the best sequence seen and its surrogate value.
    """
    # initialise logits (log‑softmax parameters)
    log_a = np.log(start_seq + 1e-30)

    best_seq = None
    best_val = np.inf
    it_total = 0

    # temperature schedule (larger τ → smoother landscape, smaller τ → true max)
    taus = [0.05, 0.02, 0.008, 0.002, 0.0005]

    for tau in taus:
        if time.time() > deadline:
            break

        # run as many iterations as time permits (capped for safety)
        while time.time() < deadline:
            it_total += 1
            # current probability vector
            a = _softmax(log_a)

            # 1) convolution a*a
            conv = _conv_fft(a, a)

            # 2) soft‑max weights for the smooth max (w = softmax(conv/τ))
            v = conv / tau
            vmax = np.max(v)
            exp_v = np.exp(v - vmax)
            w = exp_v / np.sum(exp_v)

            # 3) gradient w.r.t. a:
            #    grad_a = 2 * (w ⋆ a_rev)[n‑1 : n‑1+n]
            conv_rev = _conv_fft(w, a[::-1])
            grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

            # 4) gradient w.r.t. logits (chain rule through softmax)
            dot = np.dot(a, grad_a)
            grad_log = a * (grad_a - dot)

            # 5) step size (η_t = η₀ / √t) – η₀ is set to 0.5
            eta = 0.5 / math.sqrt(it_total)
            log_a = log_a - eta * grad_log

            # periodic reporting & best‑track
            if it_total % 50000 == 0:
                cur_seq = _softmax(log_a)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()
                    print(f"[SmoothGrad] τ={tau:.5g} it={it_total:,} best={best_val:.6f}")

    # make sure we have a sequence to return
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val


def _local_random_walk(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       sigma_start: float = 0.12,
                       sigma_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search in log‑softmax space.
    Gaussian perturbations of decreasing variance are added to the logits;
    improvements are kept.
    """
    rng = np.random.default_rng(987654)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        sigma = sigma_start * (1 - progress) + sigma_end * progress

        logits = np.log(cur_seq + 1e-30)
        proposal_logits = logits + sigma * rng.normal(size=n)
        proposal_seq = _softmax(proposal_logits)
        proposal_val = _objective_surrogate(proposal_seq)

        if proposal_val < cur_val:
            cur_seq = proposal_seq
            cur_val = proposal_val
            if cur_val < best_val:
                best_seq = cur_seq.copy()
                best_val = cur_val

        it += 1
        if it % 2000 == 0:
            print(f"[LocalWalk] iter={it:,} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose problem size.
    # If a previous solution exists we double its length to give the optimiser
    # more degrees of freedom (larger n often yields a lower constant).
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        n0 = len(prev)
        n = n0 * 2                     # e.g. 4000 → 8000
    else:
        n = 4000                       # fallback size

    print(f"\n=== Problem size: n = {n} ===")

    # ------------------------------------------------------------------
    # 0. Retrieve previous solution and build an up‑sampled seed.
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if isinstance(prev, list) and len(prev) > 0:
        prev_arr = np.asarray(prev, dtype=np.float64)
        # linear interpolation to the new length
        x_old = np.arange(prev_arr.size, dtype=np.float64)
        x_new = np.linspace(0, prev_arr.size - 1, n)
        upsampled = np.interp(x_new, x_old, prev_arr)
        upsampled = upsampled / upsampled.sum()
        seeds.append(upsampled)

    # few random Dirichlet seeds (different concentrations give diverse shapes)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # a handful of beta‑shape seeds
    for _ in range(15):
        a = 10 ** rng.uniform(-1, 1)
        b = 10 ** rng.uniform(-1, 1)
        seeds.append(_beta_shape(n, a, b))

    # a few power‑law seeds
    for _ in range(10):
        p = rng.uniform(0.0, 4.0)
        seeds.append(_power(n, p))

    # ------------------------------------------------------------------
    # 1. Rank seeds by the exact surrogate (fast) and keep the best one.
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = np.inf
    for vec in seeds:
        v = _objective_surrogate(vec)
        if v < best_seed_val:
            best_seed_val = v
            best_seed = vec
    print(f"Initial best surrogate (raw seed) = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth‑max gradient descent (annealed temperature).
    # ------------------------------------------------------------------
    seq_smooth, val_smooth = _smooth_gradient_opt(best_seed, n, deadline)
    print(f"[SmoothGrad] Surrogate after annealing = {val_smooth:.6f}")

    global_best_seq = seq_smooth.copy()
    global_best_val = val_smooth

    # ------------------------------------------------------------------
    # 3. Fast stochastic local‑walk polishing (uses remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"\n--- Local random‑walk polishing (≈{remaining:.1f}s left) ---")
        seq_local, val_local = _local_random_walk(global_best_seq, n, deadline)
        if val_local < global_best_val:
            global_best_seq = seq_local.copy()
            global_best_val = val_local
            print(f"[LocalWalk] Improved surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official evaluator.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse by the next round
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
