"""
Propose a step‑function sequence that (hopefully) beats the previous best
≈1.508534.  The strategy is a diversified seed generation (including very
skewed exponential / power‑law / beta families and ultra‑sparse patterns),
followed by:

* continuation L‑BFGS on a log‑sum‑exp surrogate of the max‑convolution,
* several restarts of exponentiated‑gradient (mirror descent) on the exact
  non‑smooth objective,
* a short simulated‑annealing phase that adds stochastic perturbations in
  log‑space,
* a cheap mass‑transfer local search,
* a final high‑β L‑BFGS polishing step.

All stages respect a 1000 s budget and update a global ``height_sequence_1``
variable for warm‑starts.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a vector on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors, sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decrease(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    s = a.sum()
    return a / s


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) + a2 * np.cos(4 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Place equal mass on `frac*n` randomly chosen indices (minimum 2)."""
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _piecewise_constant(n: int, splits: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑constant shape with `splits` blocks."""
    # generate random block lengths that sum to n
    cuts = np.sort(rng.choice(np.arange(1, n), size=splits - 1, replace=False))
    boundaries = np.concatenate(([0], cuts, [n]))
    a = np.empty(n, dtype=np.float64)
    for i in range(splits):
        start, end = boundaries[i], boundaries[i + 1]
        mass = rng.random()  # raw block weight
        a[start:end] = mass
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _rotate(v: np.ndarray, shift: int) -> np.ndarray:
    return np.roll(v, shift)


# ----------------------------------------------------------------------
# Objective helpers (exact objective and sub‑gradient)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    n = len(a)
    conv = _conv_fft(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (returns the gradient of the
    max‑convolution term only; the prefactor 2·n/(sum^2) is omitted as it
    does not affect the direction)."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and continuation L‑BFGS
# ----------------------------------------------------------------------
def _smooth_objective(v: np.ndarray, n: int, beta: float):
    """Log‑sum‑exp smooth approximation of max(conv(a,a)).  Returns (obj, grad)
    where the gradient is w.r.t. logits `v`."""
    a = _softmax(v)                     # probability vector (ℓ₁‑norm = 1)
    conv = _conv_fft(a, a)              # full linear convolution
    shift = np.max(conv)                # stability shift
    exp_term = np.exp(beta * (conv - shift))
    sum_exp = exp_term.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original code)
    w = exp_term / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float] | None = None,
    max_iter_per_beta: int = 2500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smooth surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            10000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # initialise logits

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta_val = float(beta)

        def fun(v_local):
            return _smooth_objective(v_local, n, beta_val)

        def callback(v_local):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "disp": False, "ftol": 1e-12},
            )
            v = res.x
        except KeyboardInterrupt:
            print(f"[L‑BFGS] deadline reached during β={beta_val}")
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} (β={beta_val})")
            break

    return _softmax(v)   # return probability vector (ℓ₁‑norm = 1)


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated gradient) optimizer
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """Mirror descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        g = _subgradient_maxconv(a)
        eta = eta0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        a = _softmax(log_a)

        if it % 200_000 == 0:
            cur = _objective_exact(a)
            print(f"[ExpGrad] it={it:,} obj={cur:.6f}")
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    cur = _objective_exact(a)
    if cur < best_val:
        best_val = cur
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (adds stochastic jumps in log‑space)
# ----------------------------------------------------------------------
def _simulated_annealing_optimize(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    total_iters: int = 200_000,
    init_temp: float = 0.1,
    final_temp: float = 1e-6,
    noise_scale: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Simple SA: perturb log‑probabilities, accept with Metropolis rule."""
    a_cur = start_a.copy()
    log_cur = np.log(a_cur + 1e-30)
    best_a = a_cur.copy()
    best_val = _objective_exact(a_cur)

    rng = np.random.default_rng(123456789)

    for it in range(1, total_iters + 1):
        if time.time() > deadline:
            break

        # linear temperature schedule
        T = init_temp * (1 - it / total_iters) + final_temp * (it / total_iters)

        # propose a new point
        noise = rng.normal(scale=noise_scale, size=n)
        log_prop = log_cur + noise * T
        a_prop = _softmax(log_prop)

        val_cur = _objective_exact(a_cur)
        val_prop = _objective_exact(a_prop)

        # Metropolis acceptance
        if val_prop < val_cur or rng.random() < np.exp(-(val_prop - val_cur) / max(T, 1e-12)):
            a_cur = a_prop
            log_cur = log_prop
            if val_prop < best_val:
                best_val = val_prop
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(f"[SA] it={it:,} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local‑search (mass transfers)
# ----------------------------------------------------------------------
def _local_search(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    iters: int = 30_000,
    step_frac: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climbing: move a small fraction of mass from i → j."""
    a = start_a.copy()
    best_val = _objective_exact(a)
    best_a = a.copy()
    rng = np.random.default_rng(987654321)

    for it in range(iters):
        if time.time() > deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] <= 0.0:
            continue

        delta = step_frac * a[i]
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        val = _objective_exact(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()
            a = a_candidate   # keep improving point

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed pool construction (now with expanded parameter ranges)
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of ℓ₁‑normalised vectors of length n."""
    seeds: list[np.ndarray] = []

    # deterministic families (including more extreme parameters)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decrease(n))

    # exponential decay: very steep and very mild
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        seeds.append(_exponential_decay(n, lam))

    # power‑law: wide range
    for p in np.linspace(0.0, 10.0, 31):
        seeds.append(_power_law(n, p))

    # beta shape – explore a broader rectangle
    for a in np.linspace(0.2, 8.0, 9):
        for b in np.linspace(0.2, 8.0, 9):
            seeds.append(_beta_shape(n, a, b))

    # sine window with stronger tapering
    for gamma in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0):
        seeds.append(_sine_window(n, gamma=gamma))

    # Gaussian smoothing (several sigmas)
    for sigma in np.linspace(0.08, 0.45, 8):
        seeds.append(_gaussian(n, sigma))

    # windows
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (2, 4, 6, 8, 10, 12, 15, 20):
        seeds.append(_kaiser_window(n, beta))

    # Random Dirichlet draws – wide concentration range
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Very sparse patterns
    for frac in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3):
        for _ in range(5):
            seeds.append(_random_sparse(n, frac, rng))

    # Piecewise‑constant blocks
    for splits in (2, 3, 4, 5):
        for _ in range(6):
            seeds.append(_piecewise_constant(n, splits, rng))

    # Low‑frequency cosine mixtures
    for nf in (2, 3, 4, 5, 6):
        for _ in range(8):
            t = np.arange(n, dtype=np.float64)
            a = np.zeros(n, dtype=np.float64)
            for k in range(1, nf + 1):
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.exponential(scale=1.0)
                a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
                seeds.append(a)

    # Pairwise mixtures (expand the pool)
    base = np.arange(len(seeds))
    for _ in range(800):
        i, j = rng.choice(base, size=2, replace=False)
        alpha = rng.random()
        seeds.append(_mix_shapes(seeds[i], seeds[j], alpha))

    # Rotations of a few promising seeds
    for _ in range(40):
        idx = rng.integers(len(seeds))
        shift = rng.integers(1, n)
        seeds.append(_rotate(seeds[idx], shift))

    # Include the previous solution if any
    if previous is not None and previous.size == n and previous.sum() > 0.0:
        seeds.append(previous / previous.sum())

    # Clean & normalise
    cleaned: list[np.ndarray] = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = _uniform(n)
            s = v.sum()
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising ``evaluate_sequence`` under the
    1000 s budget.

    The function follows a multi‑stage scheme:
    1. Generate a huge, diverse seed pool (including very skewed and sparse
       candidates).
    2. Run continuation L‑BFGS on the smooth log‑sum‑exp surrogate for the
       best seeds.
    3. Perform many exponentiated‑gradient (mirror‑descent) restarts on the
       exact non‑smooth objective.
    4. A simulated‑annealing refinement that adds stochastic jumps in
       log‑space.
    5. Cheap stochastic local‑search (mass transfers).
    6. Final high‑β L‑BFGS polishing.
    The best sequence found is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two speeds up the FFTs a bit
    # ------------------------------------------------------------------
    n = 4096
    print(f"\n=== Optimisation start (n={n}, budget≈{budget_s}s) ===")

    # ------------------------------------------------------------------
    # Gather seeds (warm‑start from previous if available)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seed vectors.")

    # ------------------------------------------------------------------
    # Quick evaluation of seeds, keep the best few
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    for vec in seeds:
        val = _objective_exact(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best objective among raw seeds = {best_val:.6f}")

    # sort and keep top‑k for L‑BFGS stage
    seeds.sort(key=_objective_exact)
    top_k = min(20, len(seeds))
    top_seeds = seeds[:top_k]

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_val = best_val
    beta_sched = [
        1.0, 5.0, 20.0, 80.0, 200.0,
        500.0, 1000.0, 2000.0, 5000.0,
        10000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0, 1_000_000.0,
    ]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            print("[Time] deadline reached before finishing L‑BFGS runs.")
            break
        print(f"\n--- L‑BFGS run #{idx + 1}/{top_k} ---")
        seq_opt = _lbfgs_smooth_optimize(
            seed_vec,
            n,
            deadline,
            beta_schedule=beta_sched,
            max_iter_per_beta=2500,
        )
        val_opt = _objective_exact(seq_opt)
        print(f"[L‑BFGS] objective = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Update] New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multiple EG restarts (exact objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        # decide how many restarts we can afford (roughly 0.12 s per 1 k iters)
        runs = 12
        iters_per_run = int(min(500_000, remaining * 0.12 / runs))
        print("\n--- EG restarts (exact objective) ---")
        for r in range(runs):
            if time.time() > deadline:
                break
            # random Dirichlet start with a random concentration
            conc = rng.uniform(0.01, 5.0)
            start_vec = _random_dirichlet(n, conc, rng)
            seq_exp, val_exp = _exp_gradient_optimize(
                start_vec,
                n,
                deadline,
                max_iters=iters_per_run,
                eta0=2.5,
            )
            print(f"[EG restart {r + 1}] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after EG restart = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long EG run from current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(3_000_000, remaining * 0.18))
        if max_iters > 0:
            print("\n--- Long EG optimisation (starting from current best) ---")
            seq_exp, val_exp = _exp_gradient_optimize(
                global_best_seq,
                n,
                deadline,
                max_iters=max_iters,
                eta0=3.0,
            )
            print(f"[Long EG] best = {val_exp:.6f}")
            if val_exp < global_best_val:
                global_best_seq = seq_exp.copy()
                global_best_val = val_exp
                print(f"[Update] New best after long EG = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (stochastic jumps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_iters = int(min(250_000, remaining * 40))
        if sa_iters > 0:
            print("\n--- Simulated‑annealing refinement ---")
            seq_sa, val_sa = _simulated_annealing_optimize(
                global_best_seq,
                n,
                deadline,
                total_iters=sa_iters,
                init_temp=0.12,
                final_temp=1e-7,
                noise_scale=0.35,
            )
            print(f"[SA] best = {val_sa:.6f}")
            if val_sa < global_best_val:
                global_best_seq = seq_sa.copy()
                global_best_val = val_sa
                print(f"[Update] New best after SA = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – tiny stochastic local search (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        iters = int(min(40_000, remaining * 30))
        if iters > 0:
            print("\n--- Local‑search (mass transfers) ---")
            seq_loc, val_loc = _local_search(
                global_best_seq,
                n,
                deadline,
                iters=iters,
                step_frac=0.018,
            )
            print(f"[LocalSearch] best = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_seq = seq_loc.copy()
                global_best_val = val_loc
                print(f"[Update] New best after local search = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – final polishing L‑BFGS with very high β values
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_beta = max(800, int(min(2500, remaining * 0.06 * 1e5)))
        print("\n--- Final polishing L‑BFGS (high β) ---")
        seq_final = _lbfgs_smooth_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0],
            max_iter_per_beta=per_beta,
        )
        val_final = _objective_exact(seq_final)
        print(f"[Final‑L‑BFGS] objective = {val_final:.6f}")
        if val_final < global_best_val:
            global_best_seq = seq_final.copy()
            global_best_val = val_final
            print(f"[Update] New best after final polishing = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for potential warm‑starts in subsequent rounds
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
