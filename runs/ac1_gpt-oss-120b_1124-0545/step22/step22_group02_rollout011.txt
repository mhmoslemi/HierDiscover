def flatten_loss(v):
    a = softmax(v)
    conv = conv_fft(a,a)
    avg = (a.sum()**2) / (2*n -1)  # note S = 1, avg = 1/(2n-1)
    diff = conv - avg
    loss = np.sum(diff**2)
    # grad w.r.t a:
    rev_a = a[::-1]
    full = conv_fft(rev_a, diff)  # same as gradient step
    grad_a = 2 * full[n-1:n-1+n]
    # chain rule
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return loss, grad_v

"""
Minimalist step‑function optimiser (≈ 1 000 s budget)

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)²)
for a non‑negative vector a.  Since the expression is scale‑invariant we
optimise over the probability simplex (∑a=1).  The algorithm proceeds in
the following stages

1.   Build a modest but diverse seed library (uniform, beta–U‑shapes,
     power‑law, exponential, DPSS …).  Evaluate the cheap surrogate
     2·n·max(conv) and keep the K best seeds.

2.   For each seed run a *flat‑convolution* phase:
     minimise the L² deviation of the convolution from its average
     (a constant) using a L‑BFGS optimiser on the log‑softmax
     parametrisation (if SciPy is present) or a vanilla mirror‑descent
     otherwise.

3.   Refine the flattened vectors with exponentiated‑gradient (mirror‑
     descent) on the exact max‑convolution (multi‑active set).  This
     directly reduces the objective.

4.   A short simulated‑annealing polishing finishes the search.

5.   A final heavy EG polishing is performed on the overall best.

All updates stay on the simplex via a soft‑max re‑parameterisation,
time‑checks guard the 1000 s wall‑clock budget and the best sequence is
stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for a fast L‑BFGS flat‑convolution step)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the flat‑convolution loss
# ----------------------------------------------------------------------
def _lbfgs_flatten_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iter: int = 2500) -> np.ndarray:
    """
    Minimises ∑_k (conv_k - avg)²  (avg = 1/(2n‑1) for ∑a=1)
    using L‑BFGS on the log‑softmax parametrisation.
    Returns a normalised vector (∑=1).  Falls back to a simple
    mirror‑descent if SciPy is unavailable.
    """
    if _opt is None:
        # ---- simple mirror‑descent fallback --------------------------------
        a = start_vec.copy()
        log_a = np.log(a + 1e-30)
        best_a = a.copy()
        best_val = np.inf
        denom = float(2 * n - 1)
        for it in range(1, max_iter + 1):
            if time.time() > deadline:
                break
            conv = _conv_fft(a, a)
            avg = 1.0 / denom
            diff = conv - avg
            rev_a = a[::-1]
            full = _conv_fft(rev_a, diff)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # mirror‑descent step
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)
            eta = 5.0 / math.sqrt(it)
            log_a -= eta * grad_log
            a = _softmax(log_a)
            val = np.sum(diff * diff)
            if val < best_val:
                best_val = val
                best_a = a.copy()
        return best_a

    # ---- L‑BFGS version --------------------------------------------------
    v0 = np.log(start_vec + 1e-30)          # log‑softmax parameters

    denom = float(2 * n - 1)                # number of convolution entries
    avg_const = 1.0 / denom                # desired constant value for ∑a=1

    def fun(v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        diff = conv - avg_const
        loss = np.sum(diff * diff)

        # gradient w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return loss, grad_v

    def callback(_):
        if time.time() > deadline:
            raise KeyboardInterrupt()

    try:
        res = _opt.minimize(fun=fun,
                            x0=v0,
                            method='L-BFGS-B',
                            jac=True,
                            callback=callback,
                            options={'maxiter': max_iter,
                                     'disp': False,
                                     'ftol': 1e-13})
        return _softmax(res.x)
    except KeyboardInterrupt:
        # Return the current iterate (still feasible)
        a = _softmax(v0)
        return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses all indices whose value is within `tol`·max of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        if ns.sum() == 0.0:
            attempts += 1
            continue
        ns /= ns.sum()

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the lower bound (≈ 1.0001)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Restore previous best if it exists (helps warm‑starts)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.shape == (n,):
                prev_arr = pa / pa.sum() if pa.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a compact, rich seed set
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))

    # beta shapes – we focus on very small alphas / betas (U‑shaped)
    for a in (0.05, 0.1, 0.2, 0.5):
        for b in (0.05, 0.1, 0.2, 0.5):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (log‑spaced rates, emphasis on heavy tails)
    for rate in np.logspace(-5, -0.5, 12):
        seeds.append(_exp_decay(n, rate))

    # power‑law tails – wide range of exponents
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # logistic (sigmoid) shapes – shifted to place mass at the ends
    for scale in (0.5, 1.0, 2.0):
        for mid in (n * 0.2, n * 0.8):
            seeds.append(_logistic_shape(n, scale, mid))

    # random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.01, 0.1, 0.5):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Previous best (if any) and tiny perturbations thereof
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(3):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the top few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Process each seed (flatten → EG → SA)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a share of the remaining budget proportional to remaining seeds
        remaining = deadline - time.time()
        share = remaining * 0.14 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        # ----- a) L‑BFGS flattening (L² deviation from constant) -------------
        seq_flat = _lbfgs_flatten_opt(seed_vec, n, sub_deadline, max_iter=2500)
        val_flat = _objective_surrogate(seq_flat)

        # ----- b) exponentiated‑gradient on the exact max‑convolution ----------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_flat,
                                                    n,
                                                    sub_deadline,
                                                    max_iters=600_000,
                                                    eta0=10.0,
                                                    tol=1e-9)
        else:
            seq_eg, val_eg = seq_flat, val_flat

        # ----- c) short simulated‑annealing polish ---------------------------
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_eg,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=90_000,
                                                  start_sigma=0.0012,
                                                  T_start=0.012,
                                                  T_end=3e-9)
        else:
            seq_sa, val_sa = seq_eg, val_eg

        # pick the best among the three candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_flat, val_flat),
            (seq_eg, val_eg),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (heavy EG)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=5_000_000,
                                                       eta0=6.0,
                                                       tol=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
