def _kaiser_window(n, beta):
    from scipy.special import i0
    i = np.arange(n)
    alpha = (n-1)/2.0
    t = (i - alpha)/alpha
    w = i0(beta * np.sqrt(1 - t*t)) / i0(beta)
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s

def _adam_exact_opt(start_seq, deadline, iters=200000, lr=0.1):

   # log_a initial
   log_a = np.log(start_seq + 1e-30)

   # Adam state m, v as before
   ...
   for it in range(iters):
        if time.time() > deadline: break
        # compute a
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        conv = _conv_fft(a,a)
        max_val = conv.max()
        max_idx = int(np.argmax(conv))
        # subgrad g_i = 2*a[rev_idx] as earlier
        g_raw = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g_raw[idx_i] = 2 * a[rev_idx]
        # gradient w.r.t log
        grad_log = a * (g_raw - np.dot(g_raw, a))
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        lr_adj = lr/(np.sqrt(v_hat) + eps)
        log_a -= lr_adj * m_hat
        # optionally clip, etc.
        # track best val
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val,best_seq = cur_val, a.copy()
   return best_seq,best_val

"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm
---------
1. Generate a rich pool of candidate probability vectors (uniform, exponential,
   power‑law, beta, Dirichlet, classic window functions, Kaiser windows, random
   piecewise‑linear shapes and the previous best if available).  All vectors are
   normalised to sum to 1 and forced to be strictly positive (tiny epsilon).

2. Evaluate the cheap surrogate objective  ``2·n·max(conv(a,a))`` for each candidate,
   keep the top‐k (default 10) seeds.

3. For every top seed run a two‑phase mirror‑descent optimisation in the
   log‑softmax (entropy‑mirror) space:

   * Phase A – smoothed Adam on a log‑sum‑exp approximation of the max
     convolution.  The smoothing parameter β is increased stepwise up to
     several hundred‑thousands, allowing a smooth landscape early on and a
     sharp focus later.

   * Phase B – exact Adam on the true max‑convolution (sub‑gradient).  The
     gradient w.r.t. the log‑probabilities is
       ∇θ = a·(g − ⟨g,a⟩)   with   g_i = 2·a_{k‑i}
     where k is the index of the current maximal convolution entry.

   Both phases keep track of the best exact surrogate value seen.

4. After the seed‑wise optimisation, a long exact‑Adam polish is performed on
   the overall best vector using the remaining time budget.

5. Finally a short swap‑based simulated‑annealing tweak (permutation anneal)
   runs for a few seconds to shave off any remaining excess.

All heavy work uses an FFT‑based convolution (O(n log n)).  The routine obeys
the 1000 s time budget, prints progress information and stores the resulting
list in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from scipy.special import i0

# ----------------------------------------------------------------------
# Helper – FFT based full linear convolution (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Normalisation utilities
# ----------------------------------------------------------------------
_EPS_POS = 1e-12


def _positive_normalise(v: np.ndarray) -> np.ndarray:
    """Force strict positivity, then renormalise to sum = 1."""
    v = np.maximum(v, _EPS_POS)
    total = v.sum()
    if total == 0.0:
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / total
    return v


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised probability vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return _positive_normalise(a)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return _positive_normalise(a)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return _positive_normalise(a)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return _positive_normalise(a)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return _positive_normalise(a)


def _hann_window(n: int) -> np.ndarray:
    return _positive_normalise(np.hanning(n).astype(np.float64))


def _hamming_window(n: int) -> np.ndarray:
    return _positive_normalise(np.hamming(n).astype(np.float64))


def _blackman_window(n: int) -> np.ndarray:
    return _positive_normalise(np.blackman(n).astype(np.float64))


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Classic Kaiser window (beta≥0)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    t = (i - centre) / centre  # in [-1,1]
    w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
    return _positive_normalise(w)


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 30_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth approximation of max(conv) using log‑sum‑exp."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800,
                         51_200, 200_000, 800_000]   # extended final β

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # log‑softmax parameters

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            # soft‑max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                     # weights ≈ softmax

            # gradient w.r.t. a (exact for smoothed objective)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to log‑space
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -50.0, 50.0)        # keep numerics stable

            # track best exact surrogate
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_exact_opt(start_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 2_500_000,
                    eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (Adam) on the true max‑convolution."""
    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = eta0
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # probability vector
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        # full convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry: g_i = 2 * a_{k‑i}
        g_raw = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g_raw[idx] = 2.0 * a[rev]

        # gradient w.r.t. log‑softmax parameters
        grad_log = a * (g_raw - np.dot(g_raw, a))

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / (np.sqrt(v_hat) + eps)

        log_a -= lr * m_hat
        log_a = np.clip(log_a, -50.0, 50.0)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """Swap‑only annealing for a few seconds."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found (as a plain list of floats) and stores it
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the time budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous global solution to use as a seed (if present).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = _positive_normalise(arr)
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build an extensive pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    for rate in (0.001, 0.002, 0.0035, 0.005):
        seeds.append(_exp_decay(n, rate))
    for rate in (0.0015, 0.0025, 0.004):
        seeds.append(_sym_exp_decay(n, rate))
    for p in (0.5, 0.7, 1.0, 1.3):
        seeds.append(_power(n, p))

    # Beta‑shaped families (random parameters)
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (wide range of concentration)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.8, 1.5, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Classic window functions
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for beta in (1.0, 2.5, 5.0, 8.0, 12.0, 20.0):
        seeds.append(_kaiser_window(n, beta))

    # Random piecewise‑linear shapes
    for nb in (5, 10, 15):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds; keep the best few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])

    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation per top seed (smooth Adam → exact Adam)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate roughly 60 % of remaining budget to the per‑seed optimisation
    remaining_total = deadline - time.time()
    per_seed_budget = (0.6 * remaining_total) / max(1, top_k)

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        sub_deadline = time.time() + per_seed_budget
        print(f"\n[Seed {idx+1}] time budget ≈ {per_seed_budget:.1f}s")

        # Phase A – smoothed Adam
        seq_smooth, val_smooth = _adam_smooth_opt(seed_vec,
                                                  sub_deadline,
                                                  iters_per_beta=20_000)

        # Phase B – exact Adam (if time left)
        if time.time() < sub_deadline:
            seq_exact, val_exact = _adam_exact_opt(seq_smooth,
                                                   sub_deadline,
                                                   max_iters=800_000,
                                                   eta0=0.8)
        else:
            seq_exact, val_exact = seq_smooth, val_smooth

        # Choose the better of the two phases for this seed
        local_seq, local_val = (seq_smooth, val_smooth) if val_smooth < val_exact else (seq_exact, val_exact)

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long exact‑Adam polishing (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long exact‑Adam polishing ---")
        final_seq, final_val = _adam_exact_opt(global_best_seq,
                                                deadline,
                                                max_iters=3_000_000,
                                                eta0=0.6)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation annealing tweak (a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 15.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
