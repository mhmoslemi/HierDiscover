"""
Propose a step–function (non‑negative list of heights) that minimizes the
``evaluate_sequence`` objective.

Algorithm Overview
------------------
1. **Warm‑start** – load the previous best sequence if it exists.
2. **Seed generation** – evaluate a rich collection of deterministic and
   random probability vectors (uniform, arcsine, triangular, beta‑shapes,
   Dirichlet draws, geometric/linear decays).  Keep the best one.
3. **Random Dirichlet search** – keep drawing Dirichlet samples of various
   concentrations while time permits.
4. **LP‑refinement** – given the current vector *f*, let
   `M = max(conv(f,f))`.  Solve the linear program  

        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (element‑wise)
                   g_j ≥ 0

   The LP is linear because conv(f,g) is linear in *g*.  The solution
   typically has Σ g_j > 1; after scaling `g ← g / Σ g` we obtain a vector
   that respects the same bound with a *smaller* peak.  Mix the new vector
   with the old one (`f ← (1‑t) f + t g`) and repeat a few times (the mix
   weight `t` is 0.5).  This is the “alternating LP” method described in
   the statement and often yields a noticeable drop in the surrogate.
5. **Exponentiated‑gradient fine‑tuning** – run a mirror‑descent optimizer
   on the exact ℓ∞‑convolution objective using the active lag’s exact
   sub‑gradient.
6. **Light‑weight random walk** – move a tiny amount of mass between two
   coordinates and keep the move if it improves the surrogate.
7. **Final evaluation** – call the provided ``evaluate_sequence`` and store
   the best sequence in the global variable ``height_sequence_1``.

All steps respect the overall time budget (default 1000 s).  The code
uses only NumPy, SciPy (for the LP) and the standard library.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Assumes sum(seq)=1.  Returns 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    eps = 1e-12
    a_sum = a.sum()
    if a_sum <= eps:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (mirror‑descent on the exact max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 8.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                # work in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient for the active lag (see analysis for factor 2)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating LP method)
# ----------------------------------------------------------------------
def _precompute_conv_matrix_indices(n: int):
    """Return (rows, cols) for the convolution matrix A where A·g = conv(f,g)."""
    rows = np.add.outer(np.arange(n, dtype=np.int32), np.arange(n, dtype=np.int32)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    return rows, cols


def _lp_optimize(f: np.ndarray, M: float, rows: np.ndarray,
                 cols: np.ndarray, deadline: float):
    """
    Solve:
        maximise   Σ g_j
        subject to   conv(f, g) ≤ M   (element‑wise, 2n‑1 constraints)
                     g_j ≥ 0
    Returns the raw optimal g (not yet normalised) or None on failure.
    """
    n = len(f)
    # data = repeated f for each column
    data = np.tile(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # minimise -Σ g  → maximise Σ g
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Linprog may occasionally be slow; we abort early if the global deadline is near.
    # The call itself has no built‑in timeout, so we simply check the remaining budget
    # after the solver returns.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print("[LP] Solver failure:", res.message)
        return None
    return res.x

            conv_f = _conv_fft(best_seq, best_seq)
            M = float(conv_f.max())

            g_raw = _lp_optimize(best_seq, M, rows, cols, deadline)
            if g_raw is None:
                break

            sum_g = g_raw.sum()
            if sum_g <= 1.0 + 1e-12:
                print(f"[LP {lp_it}] No improvement possible (Σg ≈ {sum_g:.6f})")
                break

            # scale to a proper probability vector
            g = g_raw / sum_g

            # mixing weight – 0.5 works well in practice
            t = 0.5
            cand = (1.0 - t) * best_seq + t * g
            cand /= cand.sum()   # numerical safety

            cand_val = _objective_surrogate(cand)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[LP {lp_it}] improved surrogate = {best_val:.6f}")
            else:
                print(f"[LP {lp_it}] no improvement (candidate {cand_val:.6f})")
                # If no improvement, still try a smaller mixing weight next round
                # (this simple strategy keeps the loop cheap)
                # but we break early to avoid wasted work
                break

    # ------------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient fine‑tuning (optional but fast)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient fine‑tuning ---")
        remaining = deadline - time.time()
        # allocate a fraction of the remaining budget to this stage
        max_iters = int(2_000_000 * (remaining / (budget_s - (time.time() - start_time))))
        max_iters = max(300_000, min(max_iters, 5_000_000))

        seq_opt, val_opt = _exp_gradient_opt(best_seq, n,
                                             deadline,
                                             max_iters=max_iters,
                                             eta0=8.0)
        print(f"[EG] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Final cheap random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n,
                                                    deadline,
                                                    attempts=250_000,
                                                    sigma=0.02)
        print(f"[Local] surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑start in a later call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
