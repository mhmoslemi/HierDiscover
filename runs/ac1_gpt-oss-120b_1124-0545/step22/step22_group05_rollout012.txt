def _simplex_proj(v):
    """Project onto the probability simplex."""
    # from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? then set to uniform?
        return np.full_like(v, 1.0/n)
    rho = ind[cond].max() - 1
    theta = cssv[cond][ -1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v):
    # ensure non-negative
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if len(rho) == 0:
        # all v_i <=0
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Advanced smooth‑gradient descent for the max‑self‑convolution objective.

The algorithm:
  1️⃣  Build a diverse pool of probability vectors (uniform, arcsine,
      triangular, beta‑shapes, geometric, Dirichlet draws, small
      perturbations).  All vectors are normalised to sum = 1.
  2️⃣  Keep the best seed (lowest exact surrogate 2·n·max(conv)).
  3️⃣  Perform a *mirror‑descent* on the **smooth approximation**
      of the max function:
           fβ(a) = (1/β)·log Σ exp(β·(a*a)_k) .
      The gradient of fβ is
           ∇fβ = 2·correlate(softmax(β·conv), a) .
      The update is carried out on log‑probabilities (exponential
      mirror map), i.e.  log_a ← log_a – η·∇fβ  followed by a soft‑max
      to stay on the simplex.  β starts small (≈0.5) and is multiplied
      slowly (×1.001) up to a large value (≈200) so that the
      approximation sharpens gradually.
      A decreasing stepsize η_t = η₀/√t (η₀≈3.0) together with an
      Adam‑style momentum term makes the search robust and fast.
  4️⃣  When the time budget is almost exhausted a cheap random local
      walk (tiny mass transfers) is performed to polish the solution.
  5️⃣  The best sequence found is finally evaluated with the supplied
      ``evaluate_sequence`` and saved in the global ``height_sequence_1``
      variable for warm‑starts.

All heavy work is done with real FFTs (O(n log n)) and the code
respectfully respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1, all entries >0)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s != 0.0 else np.full_like(v, 1.0 / v.size)


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    """
    n = len(seq)
    # linear convolution with FFT
    N = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(a_fft * a_fft, n=N)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (return probability vectors, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with a scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric‑decay (rate∈(0,1))."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    return noisy / s if s > 0 else seq.copy()


# ----------------------------------------------------------------------
# Mirror‑descent with a smooth max‑approximation
# ----------------------------------------------------------------------
def _smooth_gradient_descent(start_a: np.ndarray,
                             n: int,
                             deadline: float,
                             eta0: float = 3.0,
                             beta0: float = 0.5,
                             beta_max: float = 200.0,
                             beta_inc: float = 1.001,
                             max_iters: int = 10_000_000):
    """
    Mirror‑descent on fβ(a) = (1/β)·log Σ exp(β·conv_k) .
    Returns the best sequence (as a probability vector) and its exact surrogate.
    """
    # mantain log‑probabilities (mirror map = negative entropy)
    log_a = np.log(start_a + 1e-30)
    best_a = start_a.copy()
    best_val = _objective_surrogate(best_a)

    beta = beta0
    N = _next_pow2(3 * n - 2)          # FFT size sufficient for all convolutions

    # Adam‑style moments (all on simplex, so we work with the gradient on logits)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        a = _softmax(log_a)                 # current probability vector
        # --- self‑convolution (a*a) ---
        a_fft = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(a_fft * a_fft, n=N)
        conv = conv_full[: 2 * n - 1]        # length 2n‑1

        # --- smooth max via softmax over conv entries ---
        max_c = np.max(conv)
        # stable softmax on β·conv
        exp_vals = np.exp(beta * (conv - max_c))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp               # weights, still sum=1, length 2n‑1

        # --- gradient wrt a : g = 2·correlate(w, a) (valid part) ---
        # correlation via FFT: g = 2 * (a * rev_w)[n-1:2n-1] reversed
        rev_w = w[::-1]
        rev_w_fft = np.fft.rfft(rev_w, n=N)
        conv_rev_full = np.fft.irfft(a_fft * rev_w_fft, n=N)
        conv_rev = conv_rev_full[: 3 * n - 2]          # length 3n‑2
        g = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]   # length n

        # --- Adam step on logits (mirror‑descent) ---
        eta = eta0 / math.sqrt(it)                     # classic 1/√t schedule
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        log_a = log_a - step

        # evaluation of exact surrogate (no extra work – we already have conv)
        surrogate = 2.0 * n * max_c
        if surrogate < best_val:
            best_val = surrogate
            best_a = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[Descent] iter {it:6d}, β={beta:.2f}, surrogate={surrogate:.6f}")

        # increase β (smoothness → sharper)
        beta = min(beta_max, beta * beta_inc)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random fraction of mass
    from one index to another and keep the move if it improves the exact
    surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # safeguard against negative due to rounding
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list of
    floats.  The search respects ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size (default 4000, can be overridden)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Try to warm‑start from a previously stored solution
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣ Build an assorted pool of seeds (all probability vectors)
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n)]
    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, a, b))

    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    for rate in (0.9999, 0.9995, 0.999, 0.998):
        seeds.append(_geometric(n, rate))

    seeds.append(_linear_decreasing(n))

    for sigma in (0.01, 0.005):
        seeds.append(_perturb(_uniform(n), rng, sigma=sigma))

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate all seeds and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Optional large‑scale Dirichlet sampling (still cheap)
    # ------------------------------------------------------------------
    draw_limit = 8_000
    draws = 0
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print(f"[Dir] new best surrogate = {best_val:.6f} (draw {draws})")
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Smooth mirror‑descent optimisation (core refinement)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Smooth mirror‑descent optimisation ---")
        remaining = deadline - time.time()
        # very generous iteration budget (will be cut by the deadline)
        max_iters = int(5_000_000 * (remaining / (budget_s - (time.time() - start_time))))
        max_iters = max(500_000, min(max_iters, 12_000_000))

        seq_opt, val_opt = _smooth_gradient_descent(best_seq, n,
                                                    deadline,
                                                    eta0=3.0,
                                                    beta0=0.5,
                                                    beta_max=200.0,
                                                    beta_inc=1.001,
                                                    max_iters=max_iters)
        print(f"[Descent] exact surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Updated global best (descent)")

    # ------------------------------------------------------------------
    # 4️⃣ Light random local refinement (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n,
                                                    deadline,
                                                    attempts=300_000,
                                                    sigma=0.02)
        print(f"[Local] exact surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑starts
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
