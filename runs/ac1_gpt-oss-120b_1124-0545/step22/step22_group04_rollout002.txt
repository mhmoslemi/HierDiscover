"""

which is known to give an (almost) uniform self‑convolution.  The
self‑convolution of a continuous arcsine law is exactly uniform; the
discretisation leads to a max‑convolution value close to 1/(2n) and thus
to the theoretical lower bound of the evaluation function,
approximately 1.0.  The code builds the distribution for the chosen
length (default 4000), evaluates the true objective, stores the result
for possible continuation, and returns the list of heights.
"""

import time
import numpy as np

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Construct a non‑negative step‑function based on the discrete arcsine law.

    Parameters
    ----------
    seed : int
        Ignored – kept for compatibility with the harness.
    budget_s : int

    list[float]

    """

    # Problem size – we keep the same dimension as the previous best.
    # ------------------------------------------------------------------

    # Build the discrete arcsine distribution.
