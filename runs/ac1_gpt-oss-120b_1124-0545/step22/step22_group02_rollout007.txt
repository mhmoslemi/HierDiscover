"""
propose_candidate
-----------------
A self‑contained optimizer for the step‑function objective
``evaluate_sequence``.  The algorithm builds a diverse library of analytic
seeds (uniform, windows, DPSS, random Dirichlet draws, etc.), evaluates the
exact surrogate ``2·n·max(conv(a,a))`` (which equals the true value when the
vector is normalised), keeps the best few seeds and then refines each of them
by a cascade of optimisation phases:

1. **Smooth‑max L‑BFGS** – minimises a log‑sum‑exp approximation of the
   maximum convolution.  The β‑schedule is long and each β is given many
   L‑BFGS iterations, allowing a very precise descent.

2. **Exact max sub‑gradient (exponentiated‑gradient)** – works on the true
   max (all indices within a tight tolerance of the current maximum are
   treated as active).  An Adam‑style momentum update is performed in the
   log‑softmax (mirror‑descent) space.

3. **Flattening phase** – minimises the ℓ₂ deviation of the whole convolution
   from its average, which drives the autocorrelation towards a constant.

4. **Hill‑climbing / simulated‑annealing** – cheap Gaussian perturbations with
   a linear cooling schedule explore the neighbourhood of the current best.

All updates stay on the probability simplex via the soft‑max transform,
respect the 1000 s wall‑clock budget and store the best vector in the global
variable ``height_sequence_1``.  The function returns the best sequence as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for L‑BFGS if available)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                     # should never happen, but be safe
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (identical to the true objective after normalisation)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a non‑negative vector with sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.special import i0 as _i0
    except Exception:  # pragma: no cover
        _i0 = None
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    try:
        from scipy.signal import windows as _sigwin
    except Exception:  # pragma: no cover
        _sigwin = None
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max L‑BFGS optimiser (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule=None,
                       max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise the smooth maximum of the convolution via L‑BFGS‑B.
    Variables are kept on the simplex by optimising the log‑softmax
    parameters.  The function is a thin wrapper that falls back to a few
    hundred gradient steps if SciPy is unavailable.
    """
    if _opt is None:                     # pragma: no cover
        # Simple fallback: a few Adam steps with a high β
        return _adam_smooth_opt(start_vec, n, deadline, beta=1e5,
                                 steps=2000)

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            # smooth max via log‑sum‑exp
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_max = shift + np.log(sum_exp) / beta_val
            obj = 2.0 * n * smooth_max

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # shape (2n‑1,)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice to length n

            # chain rule: d v = a·(grad_a - ⟨grad_a,a⟩)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # allow early abort from the callback
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-13,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# 2️⃣  Exact‑max exponentiated‑gradient (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient on the exact max‑convolution.
    Returns the best vector encountered and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)          # first‑moment (Adam)
    v = np.zeros_like(a)          # second‑moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the exact max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style update in log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# 3️⃣  Flattening phase (quadratic loss on the whole convolution)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on Σ_k (conv_k - avg)².  The update is performed in the
    log‑softmax space to keep the iterate on the simplex.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # number of entries in the full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        # gradient of Σ diff² w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# 4️⃣  Simple hill‑climbing / SA polishing
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_attempts: int = 300_000,
                start_sigma: float = 0.002,
                T_start: float = 0.02,
                T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing style polishing: Gaussian perturbations
    with a linear temperature schedule.  Returns the best vector encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns = ns / s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in the
    global variable ``height_sequence_1`` and returns it as a list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it moderate; larger n does not change the
    #   ratio dramatically but incurs higher per‑iteration cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present (helps continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a rich seed set (analytic families + random draws)
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # Kaiser windows
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    # Gaussian windows
    for sigma_frac in np.linspace(0.05, 0.5, 9):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays (log‑spaced)
    for rate in np.logspace(-5, -0.2, 30):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power(n, p))

    # Beta‑distribution shapes
    alphas = np.linspace(0.01, 10.0, 20)
    betas = np.linspace(0.01, 10.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Logistic profiles
    for scale in [0.5, 1.0, 2.0, 5.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # DPSS windows (if SciPy is available)
    try:
        from scipy.signal import windows as _sigwin
    except Exception:  # pragma: no cover
        _sigwin = None
    if _sigwin is not None:
        for nw in [2.5, 3.5, 4.0, 4.5, 5.0]:
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws – wide range of concentration
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform with exponential
    exp_rates = np.logspace(-5, -0.5, 8)
    for r in exp_rates:
        exp_seq = _exp_decay(n, r)
        for alpha in (0.05, 0.15, 0.30, 0.45):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Add reversed copies (helps break symmetry)
    rev_seeds = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev_seeds)

    # Perturb previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement loop over the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a time share for this seed (roughly equal)
        remaining = deadline - time.time()
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        # ---- a) smooth‑max L‑BFGS (long schedule) ----------
        seq_lbfgs = _lbfgs_smooth_opt(seed_vec.copy(),
                                      n,
                                      sub_deadline,
                                      beta_schedule=None,
                                      max_iter_per_beta=3000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # ---- b) exact‑max exponentiated‑gradient ----------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_lbfgs,
                                                    n,
                                                    sub_deadline,
                                                    max_iters=3_000_000,
                                                    eta0=8.0,
                                                    tol=1e-10)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # ---- c) flattening gradient phase ----------
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_eg,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=1_500_000,
                                                       eta0=5.0)
        else:
            seq_flat, val_flat = seq_eg, val_eg

        # ---- d) hill‑climbing polishing ----------
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_hc, val_hc = _hill_climb(seq_flat,
                                         n,
                                         hill_deadline,
                                         rng,
                                         max_attempts=120_000,
                                         start_sigma=0.0015,
                                         T_start=0.015,
                                         T_end=1e-9)
        else:
            seq_hc, val_hc = seq_flat, val_flat

        # Pick best among the candidates for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_lbfgs, val_lbfgs),
            (seq_eg, val_eg),
            (seq_flat, val_flat),
            (seq_hc, val_hc),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final intensive polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final heavy EG polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=5_500_000,
                                                       eta0=6.0,
                                                       tol=1e-11)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=2_000_000,
                                                       eta0=4.0)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final hill‑climbing polish ---")
        final_seq3, final_val3 = _hill_climb(global_best_seq,
                                             n,
                                             deadline,
                                             rng,
                                             max_attempts=200_000,
                                             start_sigma=0.001,
                                             T_start=5e-3,
                                             T_end=1e-9)
        if final_val3 < global_best_val:
            global_best_seq = final_seq3
            global_best_val = final_val3
            print(f"[Final HC] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
