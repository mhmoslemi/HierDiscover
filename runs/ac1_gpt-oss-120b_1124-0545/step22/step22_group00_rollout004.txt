Algorithm
---------
1. **Initialisation** – reuse the previous global solution if present,
   otherwise start from a uniform vector of length ``n`` (`n`=4000 by
   default).  The vector is stored as a probability distribution
   (ℓ₁‑normalised).

2. **Broad parametric seeding** – generate a few hundred candidate
   vectors from a variety of analytic families (Gaussian, Beta,
   power‑law, sine‑window, exponential decay, random Dirichlet draws,
   mixtures).  All candidates are normalised and immediately evaluated
   with the cheap surrogate ``2·n·max(conv(a,a))``.

3. **Select the most promising seeds** – keep the best ``k`` seeds
   (default ``k=8``).  For each seed run a two‑stage local optimiser:

   * **L‑BFGS on a smooth‑max surrogate** – minimise a log‑sum‑exp
     approximation of the max‑convolution.  The smoothing parameter
     β is increased on a schedule and the optimiser is allowed many
     iterations (≈4000 per β).  This stage uses the exact gradient of
     the smooth surrogate.

   * **Adam‑style exponentiated‑gradient (mirror‑descent)** – starting
     from the L‑BFGS result, perform a long run of Adam on the exact
     sub‑gradient of the max‑convolution (full active set).  The update
     is performed in log‑softmax space, guaranteeing non‑negativity and
     ℓ₁‑normalisation at every step.

   The best sequence among the seeds is kept as the *global best*.

4. **Final polishing** – run a high‑budget Adam pass on the current
   global best followed by a high‑budget simulated‑annealing sweep.
   Both phases respect the remaining time budget.

5. **Return** – the best sequence found is stored in the global variable
   ``height_sequence_1`` and also returned as a plain ``list[float]``.
   The final true value is printed using ``evaluate_sequence``.

All heavy work respects the user‑supplied time budget (default 1000 s).
The code contains no lambda expressions and all helpers are defined at
module level.
"""

import time
import math
import numpy as np

# Optional SciPy import (used for L‑BFGS)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (surrogate) – assumes sum(a) == 1
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a, a)) for a probability vector."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (full active set)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)
    count = active.sum()
    if count == 0.0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)               # length = 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser on the exact max
# ----------------------------------------------------------------------
def _adam_expgd(start_a: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 400_000,
                lr: float = 0.5,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent with Adam moment estimates on the exact max‑conv."""
    log_a = np.log(start_a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq, n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # sub‑gradient (full active set)
        grad_a = _subgradient_max_conv(a)

        # mirror‑gradient
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑softmax variables)
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      beta_schedule=None,
                      max_iter_per_beta: int = 3500) -> np.ndarray:
    """Minimise a log‑sum‑exp surrogate of max(conv(a,a))."""
    if _opt is None:
        # SciPy not available – fall back to the start vector
        return _softmax(start_vec)

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0,
            1_000_0000.0
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() >= deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """Gaussian perturbations with a linear temperature schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns, n)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Parametric seeding utilities
# ----------------------------------------------------------------------
def _generate_parametric_seeds(n: int,
                               rng: np.random.Generator,
                               num: int = 600) -> list[np.ndarray]:
    """Randomly sample a variety of analytic families."""
    seeds = []

    for _ in range(num):
        choice = rng.integers(0, 7)  # 0..6
        if choice == 0:           # Gaussian – log‑uniform sigma
            sigma_frac = 10 ** rng.uniform(-2.0, -0.2)   # 0.01 … 0.63
            seeds.append(_gaussian(n, sigma_frac))
        elif choice == 1:         # Beta – log‑uniform α,β
            alpha = 10 ** rng.uniform(-1.0, 1.0)        # 0.1 … 10
            beta = 10 ** rng.uniform(-1.0, 1.0)
            seeds.append(_beta_shape(n, alpha, beta))
        elif choice == 2:         # Power‑law – uniform p
            p = rng.uniform(0.0, 5.0)
            seeds.append(_power(n, p))
        elif choice == 3:         # Sine window – uniform γ
            gamma = rng.uniform(0.2, 5.0)
            seeds.append(_sine_window(n, gamma))
        elif choice == 4:         # Exponential left – log‑uniform rate
            rate = 10 ** rng.uniform(-4.0, -0.2)
            seeds.append(_exp_decay_left(n, rate))
        elif choice == 5:         # Exponential right – log‑uniform rate
            rate = 10 ** rng.uniform(-4.0, -0.2)
            seeds.append(_exp_decay_right(n, rate))
        else:                     # Random Dirichlet – wide concentration range
            conc = 10 ** rng.uniform(-4.0, -0.2)       # 0.0001 … 0.63
            seeds.append(_random_dirichlet(n, conc, rng))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 0.2  # tiny safety margin

    # --------------------------------------------------------------
    # Determine problem size – reuse previous solution if present.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim != 1 or prev_arr.size == 0:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 4000
        print(f"\n[Info] No previous solution – using n={n}")
        base_seq = _uniform(n)
    else:
        n = int(prev_arr.shape[0])
        print(f"\n[Info] Re‑using previous solution of length n={n}")
        # normalise (the previous vector may already be normalised)
        s = prev_arr.sum()
        base_seq = prev_arr / s if s > 0 else _uniform(n)

    # --------------------------------------------------------------
    # 1. Broad seeding – deterministic families + random parametric draws
    # --------------------------------------------------------------
    seeds = []

    # deterministic families (a modest selection)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.10))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_beta_shape(n, alpha=0.5, beta=0.5))
    seeds.append(_beta_shape(n, alpha=2.0, beta=2.0))
    seeds.append(_power(n, p=0.5))
    seeds.append(_power(n, p=2.0))
    seeds.append(_exp_decay_left(n, rate=0.001))
    seeds.append(_exp_decay_right(n, rate=0.001))

    # mixtures with uniform
    for r in (0.001, 0.01, 0.1):
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        seeds.append(_mix_shapes(_uniform(n), exp_l, 0.3))
        seeds.append(_mix_shapes(_uniform(n), exp_r, 0.3))
        seeds.append(_mix_shapes(_uniform(n), exp_l, 0.7))
        seeds.append(_mix_shapes(_uniform(n), exp_r, 0.7))

    # random Dirichlet draws (sparse and dense)
    for conc in (0.0005, 0.001, 0.005, 0.02, 0.1):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # add the previous best (if any) and a few perturbations
    seeds.append(base_seq)
    for _ in range(5):
        pert = base_seq + 0.004 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s > 0.0:
            seeds.append(pert / s)

    # additional parametric random seeds
    seeds.extend(_generate_parametric_seeds(n, rng, num=500))

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec, n)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] best surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.7f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq, n)
    print(f"\n[Init] global best surrogate = {global_best_val:.7f}")

    # --------------------------------------------------------------
    # 4. Multi‑start local refinement (L‑BFGS → Adam EG)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate a modest slice per seed (at least a few seconds)
        sub_deadline = time.time() + max(4.0, remaining * 0.15 / max(1, (top_k - si)))

        # ---- L‑BFGS on smooth‑max surrogate ----
        if _opt is not None:
            seq_lbfgs = _lbfgs_smooth_opt(seed_vec, n, sub_deadline,
                                          beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0,
                                                         200.0, 500.0, 1_000.0,
                                                         2_000.0, 5_000.0,
                                                         20_000.0, 50_000.0,
                                                         100_000.0, 200_000.0,
                                                         500_000.0, 1_000_000.0,
                                                         2_000_000.0],
                                          max_iter_per_beta=3500)
            val_lbfgs = _objective_surrogate(seq_lbfgs, n)
        else:
            seq_lbfgs, val_lbfgs = seed_vec, _objective_surrogate(seed_vec, n)

        # ---- Adam‑style EG on exact max ----
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_expgd(seq_lbfgs, n, sub_deadline,
                                            max_iters=250_000,
                                            lr=0.6)
        else:
            seq_adam, val_adam = seq_lbfgs, val_lbfgs

        # choose best among raw seed, L‑BFGS, Adam EG
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec, n)),
            (seq_lbfgs, val_lbfgs),
            (seq_adam, val_adam)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.7f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.7f}")

    # --------------------------------------------------------------
    # 5. Final polishing – intensive Adam EG + SA
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive Adam‑EG polishing (full‑active) ---")
        final_seq, final_val = _adam_expgd(global_best_seq,
                                           n,
                                           deadline,
                                           max_iters=800_000,
                                           lr=0.5)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam‑EG] improved to {global_best_val:.7f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high‑budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=400_000,
                                                          start_sigma=8e-4,
                                                          T_start=3e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.7f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.7f}, "
              f"true evaluate_sequence = {true_val:.7f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
