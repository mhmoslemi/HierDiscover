"""
Improved step‑function optimiser.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0

Because the objective is scale‑invariant we work with ℓ₁‑normalised
vectors, i.e. minimise 2·n·max(conv(a,a)) under ∑a=1, a≥0.
The optimiser consists of three phases:

1️⃣  Generate a large, diverse pool of analytic seeds (uniform,
    triangular, sine‑windows, Gaussians, exponentials,
    power‑laws, beta‑shapes and many Dirichlet draws).
    Keep the best seed according to the exact surrogate
    `2·n·max(conv)`.

2️⃣  Polish the seed with a mirror‑descent (exponentiated‑gradient)
    optimiser that uses the *exact* sub‑gradient of the active
    convolution entry.  The only correction to the earlier code
    is the treatment of the diagonal term – it must be `2·a_j`
    (the previous implementation used `a_j` and was therefore
    sub‑optimal).  The step‑size follows η₀/√t.

3️⃣  While time permits, escape from local minima by small random
    perturbations followed by a short polishing run.

All heavy lifting uses FFT‑based convolution, the algorithm
stores the final sequence in the global variable ``height_sequence_1``
so that later runs can continue from the current best.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (used for random escapes)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent) with *exact* sub‑gradient.
# ----------------------------------------------------------------------
def _exp_gradient_opt_correct(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 10_000_000,
                              eta0: float = 8.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (guaranteed feasibility) and
    uses the correct sub‑gradient (diagonal term = 2·a_j).
    """
    a = start_seq.copy()
    # initialise log‑parameters; add tiny epsilon to avoid log(0)
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution of a with itself
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active convolution index

        # exact sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # every pair contributes 2·a[rev_idx]; the diagonal (i==rev_idx)
        # is naturally handled because rev_idx == idx_i ⇒ 2·a_j (correct).
        g[idx_i] = 2.0 * a[rev_idx]

        # mirror‑descent update in log‑space
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)               # <g, a>
        grad_log = a * (g - dot)         # ∇_log objective
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"  [Iter {it}] new best surrogate = {best_val:.6f}")

    return best_seq, best_val

        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the single best seed
    # --------------------------------------------------------------
    best_seed = None
    best_seed_val = float("inf")
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec

    if best_seed is None:
        # fall back to uniform if something went wrong
        best_seed = _uniform(n)
        best_seed_val = _objective_surrogate(best_seed)

    print(f"\n[Init] best seed surrogate = {best_seed_val:.6f}")

    # --------------------------------------------------------------
    # 3. Polish the best seed with the correct mirror‑descent optimiser
    # --------------------------------------------------------------
    print("\n--- Polishing phase (exponentiated gradient) ---")
    seq, val = _exp_gradient_opt_correct(best_seed, n, deadline,
                                         max_iters=10_000_000,
                                         eta0=8.0)

    print(f"\n[Polish] surrogate after polishing = {val:.6f}")

    # --------------------------------------------------------------
    # 4. Random‑perturbation escapes while time permits
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # generate a modestly perturbed copy of the current best
        pert = seq + 0.001 * rng.standard_normal(n)
        pert = np.maximum(pert, 0.0)
        pert = _project_simplex(pert)

        sub_deadline = time.time() + 8.0          # short burst
        seq_tmp, val_tmp = _exp_gradient_opt_correct(pert, n,
                                                     sub_deadline,
                                                     max_iters=500_000,
                                                     eta0=8.0)
        if val_tmp < val:
            seq, val = seq_tmp, val_tmp
            print(f"[Escape] improved surrogate = {val:.6f}")

    # --------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
