"""
Optimise a non‑negative step‑function (a probability vector) that
minimises  ``evaluate_sequence``.
The algorithm works in three stages:

1. Build a modest set of analytic / random seeds (uniform, exponential,
   power‑law, beta, Dirichlet, plus a down‑sampled version of the previous
   best if available) and keep the few best according to the exact objective
   2·n·max(conv(a,a)) (the sum of a is normalised to 1).

2. For each of those top seeds run a long exponentiated‑gradient (mirror‑
   descent) on the *exact* max‑convolution.  The update is performed in the
   log‑softmax space:  log_a ← log_a – η·∇, a = softmax(log_a).  η decays as
   η₀/√t and the loop runs until the allocated time slice expires.

3. Use a short Adam optimiser on a log‑sum‑exp smoothed version of the
   max‑convolution to polish the best candidate found by the EG phase.

All work respects a global time budget (default 1000 s).  The final
distribution is stored in the global variable ``height_sequence_1`` and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective (sum of the sequence is 1)
# ----------------------------------------------------------------------
def _objective_exact(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    a = np.asarray(seq, dtype=np.float64)
    s = a.sum()
    if s <= 0.0:
        return np.inf
    a = a / s
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (always ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Helper: resample an arbitrary vector to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _resample_to(vec: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(vec)
    if old_n == new_n:
        a = np.asarray(vec, dtype=np.float64)
        s = a.sum()
        return a / s if s > 0 else _uniform(new_n)
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    a = np.interp(new_x, old_x, vec)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(new_n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Maintains log‑softmax parameters (log_a) and updates with a decreasing
    step‑size η = η₀/√t.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # keep in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_exact(a)

    it = 0
    n = len(a)

    while True:
        if time.time() > deadline:
            break
        it += 1

        # recover probability vector
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        # convolution and active index (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry:
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing learning‑rate
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # keep log_a numerically bounded
        log_a = np.clip(log_a, -30.0, 30.0)

        cur_val = _objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     iters_per_beta: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth approximation of the max‑convolution.
    The smooth max follows a log‑sum‑exp with a temperature β that is
    increased along a fixed schedule.
    """
    beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200, 200_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_exact(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probability vector
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # convolution
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                    # weights (soft max)

            # gradient w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. log‑softmax parameters (mirror‑descent)
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_exact(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine builds a pool of analytic / random
    seeds, refines the most promising ones with a long exponentiated‑gradient
    descent on the exact max‑convolution, uses a short Adam‑based smoothing
    as final polish and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 1024 gives a good speed / quality trade‑off.
    # --------------------------------------------------------------
    n = 1024

    # --------------------------------------------------------------
    # 1. Build a pool of seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0018))
    seeds.append(_exp_decay(n, rate=0.0032))
    seeds.append(_sym_exp_decay(n, rate=0.0024))
    seeds.append(_power(n, p=0.7))
    seeds.append(_power(n, p=1.2))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.03, 0.08, 0.2, 0.6, 1.0, 2.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # previous global solution – down‑sample to current n
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1:
                down = _resample_to(arr, n)
                seeds.append(down)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Quick evaluation of the surrogate (exact objective) and keep top‑k
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_exact(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (exact objective) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient optimisation on the best seeds
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best_seq)
    print(f"\n[Init] global best exact = {global_best_val:.6f}")

    remaining_total = deadline - time.time()
    time_per_seed = remaining_total / max(1, top_k)

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        sub_deadline = time.time() + time_per_seed
        print(f"\n[Seed {idx+1}] EG optimisation (until {sub_deadline - start_time:.1f}s)")

        seq_opt, val_opt = _exp_gradient_opt_full(seed_vec,
                                                  sub_deadline,
                                                  eta0=6.0)

        print(f"[Seed {idx+1}] best exact = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print("    *** New global best ***")

    # --------------------------------------------------------------
    # 4. Short Adam‑smooth polishing with any remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam smoothing (polish) ---")
        seq_adam, val_adam = _adam_smooth_opt(global_best_seq, deadline,
                                              iters_per_beta=30_000)
        if val_adam < global_best_val:
            global_best_seq = seq_adam
            global_best_val = val_adam
            print(f"[Polish] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final evaluation with the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nExact surrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                 # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation stays within the time budget, works with any previously stored `height_sequence_1`, uses a fast FFT‑based convolution, runs a long exponentiated‑gradient descent on the exact max‑convolution (the most influential part), finishes with a short Adam‑based smoothing pass, and finally returns the best non‑negative step‑function found.
