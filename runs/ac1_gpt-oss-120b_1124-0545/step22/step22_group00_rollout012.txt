    n=len(f)
    # Build A matrix using scipy.linalg.toeplitz
    c = np.concatenate([f, np.zeros(n-1)])
    r = np.concatenate([f[:1], np.zeros(n-1)])   # first element f[0] then zeros
    A = scipy.linalg.toeplitz(c, r)   # shape (2n-1, n)

    # Use linprog minimization with c = -np.ones(n)

        if sumg > 1e-12:
            # rescale to sum 1
            g = g / sumg
        else:
            # fallback to f
            return f
    else:
        # fallback
        return f

    new_f = (1 - t) * f + t * g

    return new_f

"""
Improved optimiser for the step‑function upper bound.

The algorithm:
1. Generate a diverse set of seed sequences (analytic families,
   random Dirichlet draws and a down‑sampled version of the previous
   best solution).  All seeds are normalised to sum = 1 and evaluated
   with the cheap surrogate 2·n·max(conv(a,a)).

2. Pick the best seed and perform a *linear‑programming* improvement
   step.  For the current vector f we solve

        maximise   ∑ g_j
        subject to  (f * g)[k] ≤ M   for all k
                    g_j ≥ 0,

   where M = max(f * f).  The LP pushes mass into positions that have
   a small overlap with f.  The solution g is mixed with f:
        f ← (1−t)·f + t·g  (t modest, decreasing with the iteration).

   Repeating this step quickly drives down the maximum convolution.
   The LP is solved with ``scipy.optimize.linprog`` using the “highs”
   backend; the constraint matrix is a Toeplitz matrix built from f.

3. After the LP phase we run a long exponentiated‑gradient (mirror‑
   descent) optimisation on the exact max‑convolution (the same
   routine that the reference implementation used).  This fine‑tunes
   the solution on the simplex.

4. A short simulated‑annealing sweep is used as a final polish.

All stages respect the global 1000 s time budget.  The best sequence
found is stored in the global variable ``height_sequence_1`` and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
except Exception:  # pragma: no cover
    _opt = None
    _linalg = None

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full output)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# ----------------------------------------------------------------------
# Surrogate objective (exact when sum==1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)                # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / active.sum()
    return grad

# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent) on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

# ----------------------------------------------------------------------
# Simulated annealing (kept from previous version)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray,
             M: float,
             t: float,
             rng: np.random.Generator) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t. (f * g)[k] ≤ M ∀k,   g ≥ 0.
    Return the mixed vector (1‑t)·f + t·g (renormalised).
    """
    if _opt is None or _linalg is None:
        return f  # fall back to identity if SciPy missing

    n = len(f)
    # Build Toeplitz matrix that encodes the convolution constraints.
    # First column: f followed by (n‑1) zeros.
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # First row: f[0] followed by zeros.
    row = np.empty(n, dtype=np.float64)
    row[0] = f[0]
    row[1:] = 0.0

    A = _linalg.toeplitz(col, row)                 # shape (2n‑1, n)

    # Right‑hand side – a tiny slack avoids numerical infeasibility.
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

    # Linear programme: maximise Σ g_j  <=> minimise −Σ g_j .
    c = -np.ones(n, dtype=np.float64)

    # Bounds: g_j ≥ 0, no upper bound.
    bounds = [(0.0, None)] * n

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return f

    if not res.success:
        # If LP fails we simply return the original vector.
        return f

    g = res.x
    if g.sum() <= 1e-12:
        return f

    # Normalise g to sum = 1 before mixing.
    g = g / g.sum()

    mixed = (1.0 - t) * f + t * g
    mixed = mixed / mixed.sum()
    return mixed

def _lp_improve(start_seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeated LP improvement – each iteration mixes the current vector
    with the LP solution using a decreasing step size.
    Returns the best sequence seen and its surrogate value.
    """
    f = start_seq.copy()
    best_val = _objective_surrogate(f)

    t_initial = 0.5
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(f, f)
        M = conv.max()

        t = t_initial * (0.9 ** (it - 1))
        new_f = _lp_step(f, M, t, rng)

        new_val = _objective_surrogate(new_f)
        if new_val + 1e-12 < best_val:
            f = new_f
            best_val = new_val
            print(f"[LP] iteration {it:2d} surrogate = {best_val:.6f}")
        else:
            # No progress – stop early.
            break

    return f, best_val

# ----------------------------------------------------------------------
# Down‑sampling helper (used when a previous solution is longer)
# ----------------------------------------------------------------------
def _downsample(vec: np.ndarray, target_n: int) -> np.ndarray:
    n = len(vec)
    if n == target_n:
        return vec.copy()
    if n % target_n == 0:
        factor = n // target_n
        reshaped = vec.reshape(target_n, factor)
        new = reshaped.sum(axis=1)
    else:
        # Simple linear binning via cumulative sums.
        cum = np.cumsum(vec)
        new = np.empty(target_n, dtype=np.float64)
        total = cum[-1]
        for i in range(target_n):
            lo = total * i / target_n
            hi = total * (i + 1) / target_n
            new[i] = cum[np.searchsorted(cum, hi, side='right') - 1]
            if i > 0:
                new[i] -= cum[np.searchsorted(cum, lo, side='right') - 1]
        # The above may be a rough approximation; ensure non‑negative.
        new = np.maximum(new, 0.0)
    new = new / new.sum()
    return new

    seeds.append(_gaussian(target_n, sigma_frac=0.12))
    seeds.append(_gaussian(target_n, sigma_frac=0.25))

    # beta‑skewed grid

                                                target_n,
                                                eg_deadline,

        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[EG] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final polish ---")
        sa_seq, sa_val = _simulated_annealing(global_best_seq,
                                              target_n,
                                              deadline,
                                              rng,
                                              max_attempts=300_000,
                                              start_sigma=8e-4,
                                              T_start=3e-3,
                                              T_end=1e-9)
        if sa_val < global_best_val:
            global_best_seq = sa_seq
            global_best_val = sa_val
            print(f"[SA] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Evaluate with the official function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store result for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Improved optimiser for the step‑function upper bound.

The algorithm consists of four stages that respect a 1000 s budget:

1. **Broad seeding** – many analytic families (uniform, triangular,
   sine‑window, Gaussian, beta, exponential, power‑law) together with a
   handful of Dirichlet draws are generated.  All vectors are
   ℓ₁‑normalised (sum = 1) and evaluated with the cheap surrogate
   `2·n·max(conv(a,a))`, which equals the true score when the sum is 1.

2. **Linear‑programming refinement** – for the current best vector `f`
   we compute `M = max(conv(f,f))` and solve

        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M   for every shift k
                    g_j ≥ 0 .

   The constraints are encoded as a Toeplitz matrix; the LP is solved
   with ``scipy.optimize.linprog`` (high‑s).  The solution `g` is mixed
   back into `f` as `f ← (1‑t)·f + t·g` with a decreasing step size `t`.
   A few iterations quickly reduce the maximum convolution.

3. **Exponentiated‑gradient polishing** – mirror‑descent on the exact
   max‑convolution (the same update used in the reference code) is run
   for a long time to fine‑tune the solution on the simplex.

4. **Simulated‑annealing polish** – a final high‑budget SA pass explores
   the neighbourhood of the current best vector.

The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP step)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
except Exception:  # pragma: no cover
    _opt = None
    _linalg = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (exact max for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Return the (averaged) sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / active.sum()
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent) on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution,
    using the full active set.  Returns the best sequence and its surrogate.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad_a = _subgradient_max_conv(a)          # shape (n,)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)              # chain‑rule

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming improvement utilities
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray,
             M: float,
             t: float,
             rng: np.random.Generator) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t. (f * g)[k] ≤ M   (for all k),   g ≥ 0.
    Blend the solution with the current vector:
        f ← (1‑t)·f + t·g   (renormalised).
    """
    if _opt is None or _linalg is None:
        return f  # fallback if SciPy is unavailable

    n = len(f)

    # Construct Toeplitz matrix that implements the convolution constraints.
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.empty(n, dtype=np.float64)
    row[0] = f[0]
    row[1:] = 0.0
    A = _linalg.toeplitz(col, row)                 # shape (2n‑1, n)

    # Right‑hand side – we give a tiny slack to avoid infeasibility due to round‑off.
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

    # Linear programme: minimise –Σ g_j  (equivalent to maximise Σ g_j).
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return f

    if not res.success:
        return f

    g = res.x
    if g.sum() <= 1e-12:
        return f

    g = g / g.sum()                       # normalise to sum = 1
    mixed = (1.0 - t) * f + t * g
    mixed = mixed / mixed.sum()
    return mixed


def _lp_improve(start_seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 20) -> tuple[np.ndarray, float]:
    """
    Repeated LP improvement – each iteration mixes the current vector
    with the LP solution using a decreasing step size.
    Returns the best sequence seen and its surrogate value.
    """
    f = start_seq.copy()
    best_val = _objective_surrogate(f)
    t0 = 0.5

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(f, f)
        M = conv.max()
        t = t0 * (0.9 ** (it - 1))
        new_f = _lp_step(f, M, t, rng)
        new_val = _objective_surrogate(new_f)

        if new_val + 1e-12 < best_val:
            f = new_f
            best_val = new_val
            print(f"[LP] iteration {it:2d} surrogate = {best_val:.6f}")
        else:
            # No progress – stop early.
            break
    return f, best_val


# ----------------------------------------------------------------------
# Down‑sampling helper (used when a previous solution is longer)
# ----------------------------------------------------------------------
def _downsample(vec: np.ndarray, target_n: int) -> np.ndarray:
    """
    Reduce the length of ``vec`` to ``target_n`` by aggregating consecutive
    entries (exactly when the original length is a multiple of the target).
    The result is renormalised to sum = 1.
    """
    n = len(vec)
    if n == target_n:
        return vec.copy()
    if n % target_n == 0:
        factor = n // target_n
        new = vec.reshape(target_n, factor).sum(axis=1)
    else:
        # Simple truncation – cheap and works for our use‑case.
        new = vec[:target_n].copy()
    new = np.maximum(new, 0.0)
    if new.sum() == 0.0:
        return _uniform(target_n)
    return new / new.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a moderate problem size – 2000 gives a good trade‑off
    # --------------------------------------------------------------
    target_n = 2000

    # Try to reuse a previous solution (down‑sample if it is longer).
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and arr.sum() > 0.0:
                init_from_prev = _downsample(arr, target_n)
                print(f"[Info] Loaded previous solution (len {arr.size}) "
                      f"and down‑sampled to n={target_n}.")
            else:
                init_from_prev = None
        except Exception:
            init_from_prev = None
    else:
        init_from_prev = None

    # --------------------------------------------------------------
    # 1. Broad seeding
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(target_n))
    seeds.append(_triangular(target_n))
    seeds.append(_sine_window(target_n, gamma=1.0))
    seeds.append(_sine_window(target_n, gamma=2.0))
    seeds.append(_gaussian(target_n, sigma_frac=0.12))
    seeds.append(_gaussian(target_n, sigma_frac=0.25))

    # beta‑skewed grid
    alphas = np.linspace(0.2, 6.0, 9)
    betas = np.linspace(0.2, 6.0, 9)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(target_n, a_, b_))

    # exponential decays
    rates = np.logspace(-4, -0.2, 16)
    for r in rates:
        seeds.append(_exp_decay_left(target_n, r))
        seeds.append(_exp_decay_right(target_n, r))

    # power‑law family
    ps = np.linspace(0.0, 4.5, 9)
    for p in ps:
        seeds.append(_power(target_n, p))

    # random Dirichlet draws – explore very sparse and dense regimes
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2):
        for _ in range(4):
            seeds.append(_random_dirichlet(target_n, conc, rng))

    # mixtures with uniform
    for r in rates[::5]:
        exp_l = _exp_decay_left(target_n, r)
        exp_r = _exp_decay_right(target_n, r)
        for a_mix in (0.1, 0.35, 0.6):
            seeds.append(_mix_shapes(_uniform(target_n), exp_l, a_mix))
            seeds.append(_mix_shapes(_uniform(target_n), exp_r, a_mix))

    # previous best (if available)
    if init_from_prev is not None:
        seeds.append(init_from_prev)

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:                                   # pragma: no cover
        raise RuntimeError("No valid seed was generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Linear‑programming refinement (few cheap iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline - 30.0   # keep a buffer for later phases
        print("\n--- LP refinement stage ---")
        seq_lp, val_lp = _lp_improve(global_best_seq, sub_deadline, rng, max_iters=20)
        if val_lp < global_best_val:
            global_best_seq = seq_lp
            global_best_val = val_lp
            print(f"[LP] new global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (long run)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        eg_deadline = deadline - 5.0
        seq_eg, val_eg = _exp_gradient_opt_full(global_best_seq,
                                                target_n,
                                                eg_deadline,
                                                max_iters=2_500_000,
                                                eta0=6.0)
        if val_eg < global_best_val:
            global_best_seq = seq_eg
            global_best_val = val_eg
            print(f"[EG] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing final polish ---")
        sa_seq, sa_val = _simulated_annealing(global_best_seq,
                                              target_n,
                                              deadline,
                                              rng,
                                              max_attempts=300_000,
                                              start_sigma=8e-4,
                                              T_start=3e-3,
                                              T_end=1e-9)
        if sa_val < global_best_val:
            global_best_seq = sa_seq
            global_best_val = sa_val
            print(f"[SA] improved surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
