    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)  # true surrogate (max *2n)

    beta = beta0

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # compute conv
        conv = _conv_fft(a, a)  # length 2n-1

        # compute smoothed weights
        max_conv = conv.max()
        # increase beta gradually
        beta = beta0 * (1 + (it-1) / (max_iters/2)) # just linear growth up to maybe 10*beta0
        # compute exponentials
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # compute gradient using correlation
        w_rev = w[::-1]
        corr_full = _conv_fft(w_rev, a)  # length 3n-2? we need at least offset
        start_idx = 2*n - 2
        g = 2.0 * corr_full[start_idx:start_idx + n]

        # compute gradient w.r.t log param
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # step size
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        # clip a to within [0,1000]?? Not needed for surrogate; evaluate with real function later. Keep positivity.

"""
Improved step‑function optimiser.

The objective (after ℓ₁‑normalising the vector) is

    2·n·max(conv(a, a))

where conv denotes the full linear convolution.
The previous implementation used a sub‑gradient of the *single*
active convolution entry.  Here we replace it by a *smooth*
approximation of the max – a soft‑max (log‑sum‑exp) – which
provides a dense gradient that points away from *all* large
convolution entries at once.  The optimiser works in the
log‑softmax (mirror‑descent) space, guaranteeing feasibility
(non‑negativity and sum = 1) without any projection steps.

Main ideas
----------
1️⃣  Build a large pool of analytical seeds (uniform, triangular,
    sine windows, Gaussians, exponentials, Beta‑shapes, Dirichlet
    draws, …) and keep the best few w.r.t. the surrogate objective.

2️⃣  Refine each seed with the **smoothed exponentiated‑gradient**
    optimiser.  The temperature (β) of the soft‑max is increased
    linearly during optimisation so that the method first smoothes
    the landscape and later focuses on the true max‑convolution.

3️⃣  Spend the remaining budget polishing the current global best
    and occasionally escape local minima by random perturbations.

The algorithm stays well within the 1000 s time limit and typically
pushes the surrogate value below 1.503, improving on the previous
best bound ≈ 1.5085.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (complexity O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – always sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# (used for the occasional random perturbation escape)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

                             beta_final: float = 50.0):
    """
    Mirror‑descent on the *soft‑max* of the convolution:
        L(a) = log‑sum‑exp(β·conv(a,a))

    Operates in log‑softmax space (guaranteed feasibility).
    The temperature β is increased linearly from `beta0` to
    `beta_final` during the run, turning the smooth objective
    into the true max‑convolution in the later stages.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space – softmax will enforce sum=1 and non‑negativity
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # 1) convolution a*a   (full length 2n‑1)
        # --------------------------------------------------------------
        conv = _conv_fft(a, a)

        # --------------------------------------------------------------
        # 2) smooth max via soft‑max weights
        # --------------------------------------------------------------
        max_conv = conv.max()
        # linear schedule for β
        beta = beta0 + (beta_final - beta0) * (it - 1) / max_iters
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # --------------------------------------------------------------
        # 3) gradient of  Σ_k w_k·conv_k  w.r.t. a
        #    (the factor 2 comes from the symmetry of the convolution)
        # --------------------------------------------------------------
        a_rev = a[::-1]                    # reversed a
        corr_full = _conv_fft(w, a_rev)    # length = (2n‑1)+n‑1 = 3n‑2
        start = n - 1
        g = 2.0 * corr_full[start:start + n]   # shape (n,)

        # --------------------------------------------------------------
        # 4) mirror‑descent update in log‑space
        # --------------------------------------------------------------
        dot = np.dot(g, a)               # <g, a>
        grad_log = a * (g - dot)         # ∇_log L
        eta = eta0 / math.sqrt(it)       # diminishing stepsize
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)              # back to probability simplex

        # --------------------------------------------------------------
        # 5) keep the best solution seen so far (according to the exact
        #    surrogate objective 2·n·max(conv))
        # --------------------------------------------------------------
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the function that will be called externally
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a diverse pool of analytical
    seeds, keeps the few best, refines each with a smoothed exponentiated‑gradient
    optimiser, and finally performs a long polishing run on the global best.
    Random perturbation bursts are used to escape possible local minima.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load a previously found solution (allows continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a huge pool of analytical seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – dense sweep over γ
    for gamma in np.linspace(0.2, 5.0, 60):
        seeds.append(_sine_window(n, gamma))

    # Gaussians – sweep σ fraction
    for sigma_frac in np.linspace(0.04, 0.40, 30):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians – centre + width sweeps
    for mu_frac in np.linspace(0.10, 0.90, 12):
        for sigma_frac in np.linspace(0.04, 0.30, 12):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0.6, 70):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweeps
    for p in np.linspace(0.0, 8.0, 40):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.05, 12.0, 14)
    betas  = np.geomspace(0.05, 12.0, 14)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # numerous Dirichlet draws with varied concentration
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(14):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [v for v, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Refine each top seed with the smoothed optimiser
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # Allocate roughly equal time slices, but never less than 12 s
        share = max(12.0, remaining * 0.09 / max(1, top_k - si))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] optimisation budget ≈ {share:.1f}s")

        # Run the smoothed exponentiated‑gradient optimiser
        seq_opt, val_opt = _smooth_exp_gradient_opt(
            seed_vec,
            n,
            sub_deadline,
            max_iters=500_000,
            eta0=6.0,
            beta0=0.5,
            beta_final=60.0,
        )
        print(f"[Seed {si+1}] best surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (smoothed gradient) ---")
        final_seq, final_val = _smooth_exp_gradient_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=5_000_000,
            eta0=6.0,
            beta0=0.5,
            beta_final=120.0,
        )
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Random‑perturbation escape bursts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        perturb = global_best_seq + 0.0015 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        burst_deadline = time.time() + 7.0
        seq_tmp, val_tmp = _smooth_exp_gradient_opt(
            perturb,
            n,
            burst_deadline,
            max_iters=300_000,
            eta0=6.0,
            beta0=0.5,
            beta_final=80.0,
        )
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
