"""
Improved search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes three ideas:

1. **Quick multi‑start exploration** – a few short exponentiated‑gradient
   runs (mirror‑descent on the exact max‑convolution) from diverse
   random starts (previous best, uniform, Dirichlet samples).

2. **Flattening optimisation** – a longer exponentiated‑gradient phase
   that minimises the *L2* deviation of the autocorrelation from the
   perfectly flat target 1/(2 n−1).  For a probability vector the
   objective is
        L(a)=∑_k ( (a∗a)[k] - 1/(2n−1) )² .
   Its gradient can be written as a convolution, which we compute with
   FFTs and update the vector in the log‑domain (exponentiated gradient)
   while keeping it on the simplex.

3. **Fine‑tuning with simulated annealing** – tiny mass transfers between
   random coordinates with a geometric cooling schedule.

The whole routine respects the global 1000 s budget, stores the best
sequence in the global variable ``height_sequence_1`` and returns it as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient on the exact *max*–convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Single‑active shift for speed.
    Returns the best (ℓ₁‑normalised) sequence encountered and its surrogate.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # safe log‑domain

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())                     # the worst shift

        # gradient of the active max‑convolution (ignoring constant 2·n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = a[k - idx]                         # vectorised

        eta = eta0 / math.sqrt(it)                  # √‑decay

        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase‑2: exponentiated‑gradient minimising L2 deviation of autocorrelation
# ----------------------------------------------------------------------
def _exp_gradient_opt_l2(start_seq: np.ndarray,
                         deadline: float,
                         eta0: float = 5.0,
                         max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent minimising
        L(a) = Σ_k ( (a*a)[k] - 1/(2n‑1) )² .
    The gradient is computed via a second convolution.
    Returns the best (ℓ₁‑normalised) sequence and its surrogate.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # constant target autocorrelation (flat spectrum)
    target = 1.0 / (2 * n - 1)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) current autocorrelation
        conv = _conv_fft(a, a)               # length 2n‑1
        err = conv - target                  # deviation from flat target

        # 2) gradient of L2 loss: 2 * (err ⋆ a)   (cross‑correlation)
        # cross‑correlation via convolution with reversed a
        grad_raw = _conv_fft(err, a[::-1])[n - 1:n - 1 + n]
        grad = 2.0 * grad_raw                # factor 2 from d/dx Σ e²

        # normalise the gradient to avoid exploding steps
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            grad = grad / max_abs

        eta = eta0 / math.sqrt(it)

        log_a -= eta * grad
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.05,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Transfer a tiny amount of probability mass between two random coordinates.
    Temperature and perturbation size decay geometrically.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # geometric cooling
        frac = attempts / max_attempts
        T = T_start * (1.0 - frac) + T_end * frac
        sigma = sigma_start * (1.0 - frac) + 1e-6 * frac

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        amt = sigma * rng.random() * a[i]   # move a tiny fraction
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
