assistantfinal**Solution Overview**

The task is to find a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimizes  

\[
F(a)=\frac{2\,n\;\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the discrete linear convolution.  
Because the denominator cancels the scaling of `a`, we may restrict ourselves to probability
vectors (`a_i≥0`, `∑a_i=1`).  In this formulation  

\[
F(a)=2\,n\;\max_{k}(a*a)_k .
\]

The function is convex (maximum of a family of convex quadratics) but it lives in a
4000‑dimensional simplex, so we use a combination of **smooth‑max Adam optimisation**, an
**exact exponentiated‑gradient (mirror‑descent) stage**, and a **large‑scale random
refinement**.  In addition to the seeds used previously we generate a **family of symmetric
Beta‑distributions** (the arcsine distribution corresponds to `α=β=0.5`).  By sweeping
`α∈[0.05,2]` we obtain high‑quality initial points that are already better than the
previous best seed.

The algorithm proceeds in four timed stages:

1. **Seed generation & selection** – uniform, triangular, Dirichlet draws,
   symmetric Beta‑distributions (α∈{0.05,…,2}).  
   The surrogate `S(a)=2 n max(conv(a,a))` is evaluated with an FFT‑based convolution.
2. **Adam on a smooth approximation of the max** – a schedule of increasing
   smoothing parameters `β` (from 1 to 10⁷) and a generous inner‑iteration budget.
3. **Exact exponentiated‑gradient polishing** – mirror‑descent on the true max‑convolution
   with a step‑size `η_t = η₀/√t`.  The loop runs until the global deadline is reached.
4. **Intensive local random walk** – repeatedly move a small random amount of mass
   from one coordinate to another; only improvements are kept.

All stages stop when the allocated time expires, guaranteeing that a candidate is
returned before the 1000 s limit.

```python
"""
Non‑negative step‑function optimiser.

The optimiser builds a diverse pool of seeds (uniform, triangular, Dirichlet,
symmetric Beta‑distributions), keeps the best one according to the surrogate
objective S(a)=2·n·max(conv(a,a)), then refines it in three timed phases:

1. Adam on a smooth‑max surrogate (increasing β, many inner iterations).
2. Exact exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
3. Greedy random local search (mass transfer between two coordinates).

All phases respect the global deadline (≤ 1000 s).  The final sequence is stored
in the global variable ``height_sequence_1`` and also returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    For α = 0.5 we obtain the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n  # centre of the i‑th bin, lies in (0,1)
    # avoid 0/1 exactly – the half‑offset already guarantees it's safe
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Numerical safety: enforce non‑negativity
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        # fallback – uniform
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    # initialise log‑probabilities (logits)
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a schedule of increasingly sharp smooth‑max parameters
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # increase the inner budget when we have still a lot of time left
        max_iter_beta = 8000 if (deadline - time.time()) > 30 else 2000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk on the simplex:
    repeatedly move a small random amount from one coordinate to another.
    Only strictly improving moves are kept.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # size of the transfer – proportional to the source coordinate
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm respects the supplied time budget (≈1000 s) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so that we always finish before the external timeout
    deadline = start_time + budget_s - 2.0

    # problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a potential seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – the key new ingredient
    beta_alphas = [
        0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of seeds – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam smoothing on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth max) ---")
        # give roughly half of the remaining budget to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact exponentiated‑gradient polishing (sub‑gradient on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        # allocate the majority of the remaining budget
        polishing_deadline = deadline - 5.0  # keep a few seconds for final steps
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        # the number of attempts is scaled to the time that is still left
        time_left = deadline - time.time()
        # a rough estimate: one attempt ≈ 0.0005 s
        approx_attempts = int(time_left / 0.0005)
        approx_attempts = max(50_000, min(300_000, approx_attempts))
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=approx_attempts,
                                                sigma=0.02)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
