```
def _adam_opt(start_seq, n, deadline, max_iters=100000, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)  # shape n
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        grad = _subgradient_max_conv(a)
        # chain rule to log domain
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)
        # update moments
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # update log_a
        log_a -= lr * m_hat / (np.sqrt(v_hat)+eps)
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val; best_seq = a.copy()
    return best_seq, best_val
```

"""
Optimiser for the step‑function bound.

The optimizer performs a broad seed sweep (including fine Beta
distributions, double‑exponential envelopes, raised‑cosine windows and
random Dirichlet draws), evaluates a cheap surrogate
`2·n·max(conv(a,a))`, then refines the top seeds with a two‑phase local
optimiser:
  1. Adam mirror‑descent on the exact sub‑gradient of the max‑convex
     objective.
  2. Smooth‑max L‑BFGS (log‑sum‑exp) to exploit curvature.
A short simulated‑annealing pass is applied afterwards, and finally the
best solution is up‑sampled to a larger dimension and polished again.
All work respects the 1000 s budget; the best sequence found is stored
in the global variable ``height_sequence_1`` so that later runs can
continue the optimisation.
"""

import time
import math
import numpy as np

# Optional SciPy import for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of max(conv) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    count = active.sum()
    if count == 0:
        return np.zeros_like(a)
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)            # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Adam mirror‑descent optimiser on the exact max‑convex objective
# ----------------------------------------------------------------------
def _adam_opt_full(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam mirror‑descent (log‑space) minimising  max(conv(a,a)).
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # sub‑gradient of max(conv)
        grad_a = _subgradient_max_conv(a)
        # mirror‑gradient (log‑softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (optional decay)
        lr = lr0 / math.sqrt(it)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise the log‑sum‑exp smooth approximation of max(conv) using
    L‑BFGS‑B in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA with Gaussian perturbations. Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_sym(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (alpha - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_double(n: int, rate: float) -> np.ndarray:
    """Symmetric double exponential envelope."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-rate * i)
    right = np.exp(-rate * (n - 1 - i))
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int) -> np.ndarray:
    """Raised‑cosine window (full support)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq

    """Return the (averaged) sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    count = active.sum()
    if count == 0:
        return np.zeros_like(a)
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)               # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Adam mirror‑descent optimiser (full‑active sub‑gradient)
# ----------------------------------------------------------------------
def _adam_opt_full(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam mirror‑descent (log‑space) on the exact max‑convolution objective.
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_sym(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (alpha - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_double(n: int, rate: float) -> np.ndarray:
    """Symmetric sum of two exponentials (left+right)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-rate * i)
    right = np.exp(-rate * (n - 1 - i))
    a = left + right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int) -> np.ndarray:
    """Raised‑cosine window (full support)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, followed by ℓ₁‑normalisation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3.0          # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – reuse previous solution if present.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
                n = arr.shape[0]
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    print(f"\n[Info] Using problem size n = {n}")

    # ------------------------------------------------------------------
    # 1. Broad seeding – diverse analytic families + random draws
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_raised_cosine(n))

    # fine symmetric Beta sweep (including U‑shaped regimes)
    betas = np.concatenate((
        np.arange(0.05, 0.55, 0.05),   # 0.05 … 0.5
        np.arange(0.6, 4.1, 0.4)      # 0.6 … 4.0
    ))
    for a in betas:
        seeds.append(_beta_sym(n, a))

    # double‑exponential envelopes (symmetric)
    rates = np.logspace(-4, -0.2, 25)
    for r in rates:
        seeds.append(_exp_double(n, r))

    # random Dirichlet draws – explore various concentration levels
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best (if any)
    if prev_arr is not None:
        base = prev_arr.copy()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        now = time.time()
        if now >= deadline - 12.0:          # reserve time for final polishing
            break
        remaining = deadline - now
        sub_deadline = now + max(5.0, remaining * 0.7 / max(1, (top_k - si)))

        print(f"\n[Seed {si+1}/{top_k}] Refinement (deadline in {sub_deadline - now:.1f}s)")

        # (a) Adam mirror‑descent
        seq_adam, val_adam = _adam_opt_full(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters=180_000,
                                            lr0=0.6)

        # (b) Smooth‑max L‑BFGS (if time permits)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_adam,
                                               n,
                                               sub_deadline,
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_adam, val_adam

        # (c) Short simulated‑annealing
        hill_deadline = sub_deadline - 0.4
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=120_000,
                                                  start_sigma=0.002,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among the refinements (including original seed)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – optional up‑sampling to a larger dimension
    # ------------------------------------------------------------------
    best_true_val = None
    try:
        best_true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception on the current best:", exc)

    if time.time() < deadline - 8.0:
        large_n = max(n * 2, n + 2000)   # roughly double the size
        print(f"\n[Upsample] Polishing at larger size n = {large_n}")
        up_seq = _upsample(global_best_seq, large_n)
        large_deadline = deadline - 2.0

        seq_large, _ = _adam_opt_full(up_seq,
                                      large_n,
                                      large_deadline,
                                      max_iters=300_000,
                                      lr0=0.5)

        if time.time() < large_deadline - 2.0:
            seq_large, _ = _simulated_annealing(seq_large,
                                                large_n,
                                                large_deadline,
                                                rng,
                                                max_attempts=150_000,
                                                start_sigma=8e-4,
                                                T_start=3e-3,
                                                T_end=1e-9)

        try:
            true_val_large = evaluate_sequence(seq_large.tolist())
            print(f"[Upsample] true evaluate_sequence = {true_val_large:.6f}")
            if best_true_val is None or true_val_large < best_true_val:
                global_best_seq = seq_large
                best_true_val = true_val_large
                print("    >>> Improved after up‑sampling!")
        except Exception as exc:
            print("[Warning] evaluate_sequence failed on up‑sampled candidate:", exc)

    # ------------------------------------------------------------------
    # 6. Final official evaluation (if not already done)
    # ------------------------------------------------------------------
    if best_true_val is None:
        try:
            best_true_val = evaluate_sequence(global_best_seq.tolist())
        except Exception as exc:  # pragma: no cover
            print("[Warning] final evaluate_sequence raised:", exc)
            best_true_val = float('inf')

    print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, true evaluate_sequence = {best_true_val:.6f}")

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
