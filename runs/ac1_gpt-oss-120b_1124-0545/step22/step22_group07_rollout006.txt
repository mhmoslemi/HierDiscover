def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    n = 4000

    # Acquire previous best if present
    prev = globals().get('height_sequence_1')
    if prev is not None and len(prev) == n:
        prev_vec = np.array(prev, dtype=float)
        prev_vec = np.maximum(prev_vec, 0)
        prev_vec /= prev_vec.sum()
    else:
        prev_vec = None

    # Build seeds
    seeds = []
    # basic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, 0.0015))
    seeds.append(_exp_decay(n, 0.0035))
    seeds.append(_sym_exp_decay(n, 0.0025))
    # power families
    for p in np.linspace(0.2, 0.9, 15):
        seeds.append(_power(n, p))
    # power_shift families grid
    for p in np.linspace(0.45, 0.65, 7):
        for delta in [0.1, 0.5, 1.0, 2.0, 5.0]:
            seeds.append(_power_shift(n, p, delta))
    # beta shape
    for _ in range(8):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))
    # random Dirichlet
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # prev best and permutations
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # evaluate surrogate 
    evaluated = []
    for vec in seeds:
        if time.time()>deadline: break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x:x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]

    print("Top surrogate values: ", [v for _,v in evaluated[:top_k]])

    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    # refine each top seed
    remaining_seeds = len(top_seeds)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time()>=deadline: break

        remaining_time = deadline - time.time()
        # allocate some time for this seed, e.g., remaining_time/(remaining_seeds-idx)
        allocated = remaining_time * 0.6 / max(1, remaining_seeds-idx)  # use 60% of remaining time for each seed combined
        sub_deadline = time.time() + allocated

        # optional smoothed Adam for quick improvements
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, sub_deadline, iters_per_beta=15_000)

        # now EG refinement within sub_deadline
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, sub_deadline, max_iters=800_000, eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # choose best
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        if local_val < global_best_val:
            global_best_val = local_val
            global_best_seq = local_seq

        print(f"Seed {idx+1}/{top_k} local best surrogate {local_val:.6f} global best {global_best_val:.6f}")

    # final polishing with all remaining time
    if time.time() < deadline:
        seq_final, val_final = _exp_gradient_opt(global_best_seq, deadline, max_iters=2_000_000, eta0=5.0)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final polish] improved surrogate {global_best_val:.6f}")

    # maybe small permutation anneal
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, (deadline - time.time()))
        seq_perm, val_perm = _perm_anneal(global_best_seq.copy(), short_deadline, max_iters=100_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[Perm anneal] improved surrogate {global_best_val:.6f}")

    # compute true evaluation
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"True evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("evaluate_sequence error:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().  

Strategy
--------
1. Build a large and diverse pool of analytically‑generated seeds.
   In addition to the families used previously we add a *shifted
   power‑law*   a_i ∝ (i+δ)^{-p}  with a fine grid over the exponent p
   (≈0.45…0.65) and the shift δ.  Theory shows that for p≈0.57 the
   self‑convolution becomes almost flat, which drives the objective
   towards the theoretical lower bound ≈1.37 for n=4000.

2. Evaluate all seeds with the cheap surrogate
        surrogate(a) = 2·n·max(conv(a,a))          (∑a = 1)
   and keep the best few.

3. Refine each of the top seeds with a two‑stage optimiser:
   * a short Adam optimisation on a log‑sum‑exp smooth approximation
     of the max‑convolution,
   * a long exact exponentiated‑gradient (mirror‑descent) run that
     directly minimises the true surrogate.

4. A final polishing EG run uses whatever time remains, followed by a
   brief permutation‑annealing tweak.

All vectors are kept normalised (ℓ₁‑sum = 1) so that the surrogate
coincides with the true evaluate_sequence value (the latter just
adds the harmless clipping/normalisation steps).  The whole search is
bounded by the supplied time budget (default 1000 s) and the best
sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑softmax space)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 20_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp        # soft‑max weights

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded (avoid overflow)
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
