def _simulated_annealing(start_seq, n, deadline, init_temp=0.03, final_temp=1e-6, max_iter=200000):
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = surrogate(seq)
    temp = init_temp
    rng = np.random.default_rng()
    # compute cooling factor per iteration: alpha = (final_temp/init_temp)**(1/max_iter)
    if max_iter > 0:
        cooling_rate = (final_temp / init_temp) ** (1.0 / max_iter)
    else:
        cooling_rate = 1.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose delta proportional to seq[i] times a random factor
        max_delta = seq[i] * 0.5  # limit delta to not exceed half of mass at i
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        # apply move
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # compute surrogate
        val = surrogate(new_seq)
        diff = val - best_val
        # if improved accept new as best
        if val < best_val:
            best_seq = new_seq.copy()
            best_val = val
            seq = new_seq
            continue
        # else accept with prob exp(-diff/temp)
        if temp > 0:
            prob = math.exp(-diff / temp)
            if rng.random() < prob:
                seq = new_seq
        # update temperature
        temp *= cooling_rate

    return best_seq, best_val

"""
Propose a candidate step‑function.

The search consists of the following stages (each respects the overall
time budget):

1️⃣  Seed generation – a variety of deterministic and random simplex vectors
   (uniform, arcsine‑like, triangular, Dirichlet with many
   concentrations and a Gaussian‑square seed).  The best seed is kept.

2️⃣  Adam optimisation on a smooth approximation of the max‑convolution.
   The schedule of the smoothing parameter β is dense and the inner loop
   is enlarged compared with the reference implementation.

3️⃣  Exact exponentiated‑gradient polishing (sub‑gradient descent on the
   true max‑convolution).  The number of iterations is scaled to the
   remaining time.

4️⃣  Simulated‑annealing local search – random mass transfers between two
   coordinates are accepted with Metropolis probability, allowing the
   algorithm to escape shallow local minima.

5️⃣  Tiny stochastic walk on the simplex (the “local search” from the
   reference code) to squeeze any final improvement.

The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.  The final
value is verified with the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    For a probability vector (sum = 1) this is exactly the objective
    returned by ``evaluate_sequence``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a == 0.0:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed factories (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian_square(n: int, rng: np.random.Generator) -> np.ndarray:
    """Square of a standard normal vector, then normalised."""
    x = rng.standard_normal(n)
    a = np.square(x) + 1e-12
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    surrogate of 2·n·max(conv(a,a)).  Returns the best probability vector
    seen and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # A dense schedule of smoothing parameters (β)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # A slightly larger inner loop than in the reference code
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # Smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # Gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) ‑ 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # Convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update (learning rate schedule decays with β)
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # A modest, β‑aware learning‑rate
            lr = 0.12 / (1.0 + 0.2 * math.log1p(beta))
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Occasionally evaluate the *exact* surrogate to keep the best
            if it % 1000 == 0:
                cur_val = _surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement (deterministic walk)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex:
    repeatedly move a small random amount from one coordinate to another
    and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search (Metropolis acceptance)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float = 0.02,
                         final_temp: float = 1e-6,
                         max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style annealing on the simplex.
    Mass is transferred between two positions; the move is accepted
    if it improves the surrogate or with probability exp(‑Δ/T).
    """
    rng = np.random.default_rng()
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _surrogate(seq)

    # geometric cooling
    if max_iter > 0:
        cooling = (final_temp / init_temp) ** (1.0 / max_iter)
    else:
        cooling = 1.0
    temp = init_temp

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # propose a transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = seq[i] * 0.5   # never move more than half the source mass
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        val = _surrogate(new_seq)
        diff = val - best_val

        if val < best_val:
            best_seq, best_val = new_seq.copy(), val
            seq = new_seq
        else:
            # Metropolis accept with probability exp(-Δ/T)
            if temp > 0.0 and rng.random() < math.exp(-diff / temp):
                seq = new_seq

        temp *= cooling  # cool down

    return best_seq, best_val

        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
