"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1️⃣  **Seed generation** – a large, diverse pool of analytically‑parameterised
    vectors (uniform, triangular, sine windows, Gaussians, exponentials,
    power‑laws, beta‑shapes, U‑shapes, two‑level piecewise, many Dirichlet draws
    and the previous best solution).  All vectors are ℓ₁‑normalised.

2️⃣  **Local refinement** – each of the K best seeds (by the surrogate
    objective ``2·n·max(conv)``) is locally optimised with a fast
    *Adam* optimiser working in the log‑softmax (mirror‑descent) space.
    The exact sub‑gradient of the active convolution entry (the max) is used.
    Optionally a short permutation hill‑climb is performed before Adam.

3️⃣  **Polishing & escapes** – the current global best is polished for the
    remaining time with a long Adam run.  Random perturbations followed by
    short Adam bursts are repeatedly tried to escape possible local minima.

All steps respect the time budget (default 1000 s).  The final sequence is
stored in the global variable ``height_sequence_1`` for possible continuation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return the surrogate objective 2·n·max(conv(seq,seq)) for a vector
    with sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _u_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped (low in centre, high at the edges) vector."""
    c = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = (np.abs(i - c) + 1.0) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_level_piecewise(n: int, edge_frac: float, high_factor: float) -> np.ndarray:
    """
    Two‑level piecewise shape:
    - edges of total length edge_frac·n have weight high_factor,
    - the middle has weight 1.
    """
    e = max(1, int(edge_frac * n / 2))
    a = np.ones(n, dtype=np.float64)
    a[:e] = high_factor
    a[-e:] = high_factor
    return a / a.sum()


# ----------------------------------------------------------------------
# Helper: simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on log‑softmax space (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_log_opt(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 300_000,
                  eta0: float = 5.0,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the surrogate objective.
    Operates in log‑softmax space to keep feasibility (a_i ≥ 0, sum=1).

    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # gradient w.r.t. log‑softmax parameters
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # project back to simplex via softmax
        a = _softmax(log_a)

        cur_val = 2.0 * n * np.max(conv)   # surrogate
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill‑climb (very short, cheap exploration)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Quick local search that swaps two entries or adds a tiny Gaussian perturbation.
    Returns the best sequence observed.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule (tiny temperature)
    T_start = 1e-4 * cur_val
    T_end = 1e-7 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:  # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                     # small Gaussian perturbation
            sigma = 0.0008 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

    for conc in conc_vals:
        for _ in range(14):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) + some noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > global_deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine each top seed
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= global_deadline:
            break

        # allocate roughly equal share of the remaining time to each seed
        remain = global_deadline - time.time()
        share = max(12.0, remain * 0.12 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) cheap deterministic ordering (descending) – useful for Adam
        seq_desc = np.sort(seed_vec)[::-1]

        # b) brief permutation hill‑climb
        seq_hc, val_hc = _perm_hill_climb(seed_vec, n, sub_deadline, rng,
                                          max_iters=100_000)

        # c) Adam polish of three variants (original, descending, hill‑climb)
        variants = [seed_vec, seq_desc, seq_hc]
        local_best_seq = None
        local_best_val = np.inf

        for vi, var in enumerate(variants):
            if time.time() >= sub_deadline:
                break
            # split remaining time equally among remaining variants
            time_left = sub_deadline - time.time()
            var_share = time_left / (len(variants) - vi)
            var_deadline = time.time() + var_share

            seq_opt, val_opt = _adam_log_opt(var, n, var_deadline,
                                             max_iters=350_000,
                                             eta0=7.0)
            if val_opt < local_best_val:
                local_best_val = val_opt
                local_best_seq = seq_opt

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")
        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # --------------------------------------------------------------
    remaining_time = global_deadline - time.time()
    if remaining_time > 30.0:
        print("\n--- Final long polishing (Adam) ---")
        final_seq, final_val = _adam_log_opt(global_best_seq, n,
                                             global_deadline,
                                             max_iters=4_500_000,
                                             eta0=6.5)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Random‑perturbation escape attempts while time permits
    # --------------------------------------------------------------
    while time.time() < global_deadline - 5.0:
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 9.0
        seq_tmp, val_tmp = _adam_log_opt(perturb, n,
                                         sub_deadline,
                                         max_iters=400_000,
                                         eta0=6.0)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the seed pool (adding U‑shapes, two‑level piecewise families, more Dirichlet draws), replaces the plain exponentiated‑gradient step with an Adam‑style mirror descent (often converges faster), keeps a light permutation hill‑climb, allocates the time budget more flexibly, and performs a final long Adam polishing together with occasional random‑perturbation escapes.  All operations respect the 1000 s limit and return a valid non‑negative sequence.
