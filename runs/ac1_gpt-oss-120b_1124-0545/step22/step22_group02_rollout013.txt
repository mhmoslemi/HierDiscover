"""
Improved step‑function optimizer.

The routine builds a very large library of analytic windows (including
uniform, triangular, sine, Kaiser, Tukey, DPSS, Gaussian, power‑law,
Beta‑distribution, exponential, logistic, etc.), evaluates them with the
exact surrogate 2·n·max(conv) (where the vector is ℓ₁‑normalised) and
keeps the best few.  The most promising seeds are then refined by a
pipeline:

1. Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution,
   using a large active set (all indices within a small relative tolerance
   of the current maximum) and Adam‑style momentum.
2. Smooth‑max optimisation via L‑BFGS‑B on a log‑sum‑exp surrogate, with a
   long β‑schedule that drives the smooth approximation to the true maximum.
3. “Flattening” gradient descent that minimises the quadratic deviation of
   the convolution from its average, i.e. pushes the whole autocorrelation
   towards a constant.
4. Short simulated‑annealing polishing.

All stages respect the global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a plain Python list.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (assumes sum(seq)==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (tapered cosine) window – SciPy fallback to manual implementation."""
    if _sigwin is not None:
        a = _sigwin.tukey(n, alpha=alpha)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)

    # Manual implementation (works for any n≥1)
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.ones_like(i)
    if alpha <= 0:
        a[:] = 1.0
    elif alpha >= 1:
        a = np.sin(np.pi * i) ** 2
    else:
        # rise
        rise = i < alpha / 2
        a[rise] = 0.5 * (1 + np.cos(np.pi * (2 * i[rise] / alpha - 1)))
        # fall
        fall = i > (1 - alpha / 2)
        a[fall] = 0.5 * (1 + np.cos(np.pi * (2 * i[fall] / alpha - 2 / alpha + 1)))
        # middle part stays 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, i=0..n‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_monotone(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random decreasing vector by sorting i.i.d. uniforms."""
    a = rng.random(n)
    a.sort()
    a = a[::-1]                     # decreasing
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution (multi‑active)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose value is within tol·max of the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)

        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]

        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """L‑BFGS‑B on a log‑sum‑exp smooth max of the convolution."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0, 10_000_000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                 # smooth max weights
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on Σ (conv_k – avg)² – pushes the whole convolution flat."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # number of entries in the full conv

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # average of the convolution entries
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """Simple Gaussian‑perturbation SA with a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine follows the strategy described in the module docstring and
    respects the provided time budget (default 1000 s).  The best sequence
    found is stored globally as ``height_sequence_1`` and returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives a richer search space
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous solution if it matches the current size
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            tmp = np.asarray(previous, dtype=np.float64)
            if tmp.shape == (n,):
                prev_arr = tmp.copy()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a very rich seed library (analytic + random)
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # Basic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # Kaiser with a finer grid of betas
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0, 16.0]:
        seeds.append(_kaiser_window(n, beta))

    # Tukey windows (different tapers)
    for alpha in [0.0, 0.2, 0.5, 0.8, 1.0]:
        seeds.append(_tukey_window(n, alpha))

    # Gaussian shapes
    for sigma_frac in np.linspace(0.03, 0.5, 12):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays (log‑spaced rates)
    for rate in np.logspace(-5, -0.1, 35):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails (larger exponent range)
    for p in np.linspace(0.0, 12.0, 25):
        seeds.append(_power(n, p))

    # Beta‑distribution shapes – a coarser but broader sweep
    alphas = [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
    betas  = [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Logistic profiles
    for scale in [0.3, 0.6, 1.0, 2.0, 4.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # DPSS windows (if scipy.signal is available)
    if _sigwin is not None:
        for nw in [2.5, 3.0, 3.5, 4.0, 4.5, 5.0]:
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws with a wide concentration range
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random monotone decreasing seeds
    for _ in range(6):
        seeds.append(_random_monotone(n, rng))

    # Mixtures of uniform with exponential tails
    for rate in np.logspace(-5, -0.5, 6):
        exp_seq = _exp_decay(n, rate)
        for alpha in (0.1, 0.3, 0.5, 0.7):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Include the reversed version of every seed (helps break symmetry)
    rev_seeds = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev_seeds)

    # Perturbation of the previous solution (if available)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement – more budget for the better seeds
    # ------------------------------------------------------------------
    # Allocate decreasing shares of the remaining budget: top seed gets
    # the most time, the worst of the top‑k gets the least.
    weight_sum = top_k * (top_k + 1) / 2
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        share = remaining * (top_k - si) / weight_sum
        sub_deadline = time.time() + share

        # ---- a) exponentiated‑gradient (multi‑active) ----------
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=600_000,
                                                eta0=9.0,
                                                tol=1e-9)

        # ---- b) smooth L‑BFGS ----------
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=None,
                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # ---- c) flattening gradient ----------
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=400_000,
                                                       eta0=6.0)
        else:
            seq_flat, val_flat = seq_lbfgs, val_lbfgs

        # ---- d) short simulated annealing ----------
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_flat,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=100_000,
                                                  start_sigma=0.001,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_flat, val_flat

        # Pick the best among all candidates for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_flat, val_flat),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (heavy EG + flattening + SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG (multi‑active) polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=6_000_000,
                                                       eta0=7.0,
                                                       tol=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=1_500_000,
                                                       eta0=5.0)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.0008,
                                                         T_start=4e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
