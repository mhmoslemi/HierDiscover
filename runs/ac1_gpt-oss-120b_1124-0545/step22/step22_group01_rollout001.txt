"""
Hybrid optimizer for the step‑function attack.

The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a))²

which, after ℓ₁‑normalising the vector (so sum(a)=1), is simply
`2·n·max(conv(a,a))`.  The algorithm works in three phases:

1️⃣  A large pool of analytic families (uniform, triangular, sine‑window,
    Gaussian, super‑Gaussian, logistic, power‑law, beta‑shapes and many
    Dirichlet draws) is generated, normalised and evaluated by the cheap
    surrogate objective `2·n·max(conv)`.  The best `top_k` seeds are kept.

2️⃣  Each retained seed is locally refined by a short permutation hill‑climb
    and two mirror‑descent variants:
       • classical exponentiated‑gradient (EG) with a √t‑decay step size,
       • an Adam‑style EG that adapts the learning rate per‑coordinate.
    The best local result becomes a new global incumbent.

3️⃣  The current global best is polished for the remaining budget with the
    Adam‑EG optimiser, while occasional random perturbations + short EG runs
    are used to escape shallow local minima.

All iterates are kept in the probability simplex via the softmax (log‑space)
or Euclidean projection.  The final vector is returned as a plain ``list``
and stored in the global variable ``height_sequence_1`` for possible
continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, guarantees non‑negative entries that sum to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(-(i - centre) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _super_gaussian(n: int, sigma_frac: float, p: float) -> np.ndarray:
    """exp( -|x-μ|^p / σ^p ), p≥2 gives flatter tops."""
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - mu) ** p / (sigma ** p))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient for the active convolution index
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idx: int) -> np.ndarray:
    """Exact sub‑gradient of max(conv) at the given active index."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, active_idx - (n - 1))
    i_end = min(n - 1, active_idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = active_idx - idx_i
    # off‑diagonal entries counted twice
    g[idx_i] = 2.0 * a[rev_idx]
    # diagonal entries counted only once
    diag_mask = idx_i == rev_idx
    if np.any(diag_mask):
        g[idx_i[diag_mask]] = a[idx_i[diag_mask]]
    return g


# ----------------------------------------------------------------------
# Mirror‑descent: classic EG
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Exponentiated‑gradient (log‑space) mirror descent with √t decay."""
    a = start_seq.copy()
    n = a.shape[0]
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = _subgradient(a, k)

        dot = np.dot(g, a)               # <g,a>
        grad_log = a * (g - dot)         # ∇_log objective

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val

# Mirror‑descent: Adam‑style EG
# ----------------------------------------------------------------------
def _exp_gradient_opt_adam(start_seq: np.ndarray,
                           deadline: float,
                           max_iters: int = 300_000,
                           lr: float = 1.0,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8):
    """Adam‑adapted exponentiated‑gradient in log‑space."""
    a = start_seq.copy()
    n = a.shape[0]
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = _subgradient(a, k)

        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update on log‑space
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val

# Simple permutation‑based hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (fraction of current best)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, len(cur_seq), size=2)
            if i != j:
                nxt = cur_seq.copy()
                nxt[i], nxt[j] = nxt[j], nxt[i]
            else:
                continue
        else:                                         # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            nxt = cur_seq + sigma * rng.normal(size=len(cur_seq))
            nxt = np.maximum(nxt, 0.0)
            s = nxt.sum()
            if s == 0.0:
                continue
            nxt = nxt / s

        val = _objective_surrogate(nxt)

        if val < cur_val:
            cur_seq = nxt
            cur_val = val
            if val < best_val:
                best_seq = nxt.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = nxt
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows
    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma))

    # Gaussian family
    for sigma_frac in np.linspace(0.05, 0.30, 20):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians
    for mu_frac in np.linspace(0.15, 0.85, 8):
        for sigma_frac in np.linspace(0.05, 0.25, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays
    for rate in np.logspace(-5, 0.5, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law
    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power_law(n, p))

    # beta shapes
    alphas = np.geomspace(0.1, 10.0, 10)
    betas  = np.geomspace(0.1, 10.0, 10)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # logistic
    for scale in np.linspace(0.05, 0.5, 12):
        seeds.append(_logistic(n, scale))

    # super‑Gaussian (p = 2,3,4)
    for p in (2.0, 3.0, 4.0):
        for sigma_frac in np.linspace(0.08, 0.30, 10):
            seeds.append(_super_gaussian(n, sigma_frac, p))

    # random Dirichlet draws
    conc_vals = [0.01, 0.03, 0.07, 0.15, 0.3, 0.7, 1.0, 2.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) – added later in main routine
    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds many analytic seeds, keeps the
    best few, refines each by hill‑climbing and two mirror‑descent variants,
    and finally spends the remaining time polishing the global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present (useful for continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Generate a large pool of analytic seeds
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng)

    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.8f}")

    # ------------------------------------------------------------------
    # 3️⃣  Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 4️⃣  Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate roughly 10 % of the remaining time per remaining seed,
        # but never less than 12 s.
        share = max(12.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}/{top_k}] refinement budget ≈ {share:.1f}s")

        # a) deterministic descending order (cheap heuristic)
        seq_desc = np.sort(seed_vec)[::-1]

        # b) short permutation hill‑climb
        seq_hc, val_hc = _perm_hill_climb(seed_vec, sub_deadline, rng,
                                          max_iters=120_000)

        # c) three variants through mirror‑descent
        variants = [seed_vec, seq_desc, seq_hc]
        local_best_seq = None
        local_best_val = np.inf

        for vi, var in enumerate(variants):
            if time.time() >= sub_deadline:
                break
            # split remaining time equally among remaining variants
            time_left = sub_deadline - time.time()
            var_share = time_left / (len(variants) - vi)
            var_deadline = time.time() + var_share

            # classic exponentiated‑gradient
            seq_opt1, val_opt1 = _exp_gradient_opt(var, var_deadline,
                                                   max_iters=400_000,
                                                   eta0=6.0)

            # Adam‑style EG
            seq_opt2, val_opt2 = _exp_gradient_opt_adam(var, var_deadline,
                                                        max_iters=300_000,
                                                        lr=1.0)

            # pick the better of the two
            if val_opt1 < val_opt2:
                seq_opt, val_opt = seq_opt1, val_opt1
            else:
                seq_opt, val_opt = seq_opt2, val_opt2

            if val_opt < local_best_val:
                local_best_val = val_opt
                local_best_seq = seq_opt

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.8f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (Adam EG) ---")
        final_seq, final_val = _exp_gradient_opt_adam(
            global_best_seq,
            deadline,
            max_iters=4_000_000,
            lr=1.2)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 6️⃣  Random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:   # keep a few seconds for final eval
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 7.0
        seq_tmp, val_tmp = _exp_gradient_opt_adam(
            perturb,
            sub_deadline,
            max_iters=350_000,
            lr=1.0)

        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.8f}")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, "
              f"true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
