"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Seed generation** – a diverse collection of analytically‑derived and
   random probability vectors (uniform, exponential, power, beta, Dirichlet,
   piecewise‑linear, sinusoidal, previous best, etc.).

2. **Pick the most promising seed** – evaluate the cheap surrogate
   ``2·n·max(conv(p,p))`` for every seed and keep the one with smallest
   value.

3. **Long smoothed Adam optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution.  The temperature ``β`` is raised
   gradually (exponential schedule) and Adam’s adaptive steps are used.
   The variables live in the log‑softmax (mirror‑descent) space, guaranteeing
   the probability simplex constraint.

4. **Exact exponentiated‑gradient polish** – switch to the true
   non‑smoothed objective.  The sub‑gradient is taken from *all*
   near‑maximal convolution entries (active set) to avoid getting stuck on
   a single binding constraint.  An exponential‑mirror update with a
   decaying learning‑rate is performed.

5. **Short permutation‑annealing tweak** – random swaps are accepted
   according to a simple simulated‑annealing schedule, often shaving the
   final surrogate a little more.

6. **Return the best sequence** – the final vector is stored in the global
   variable ``height_sequence_1`` and also returned as a plain list of
   ``float``.  The whole procedure respects the global wall‑time budget
   (default 1000 s).

All heavy work (convolutions) is performed with an FFT‑based implementation
(`_conv_fft`) which runs in O(n log n).  The code does not read/write any
files and uses only the standard library plus NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, length = n+m‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective (probability vector -> 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_surrogate(p: np.ndarray) -> float:
    """Return 2·n·max(p * p) for a probability vector (sum = 1)."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised probability vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_power(n: int, power: float) -> np.ndarray:
    """Bell‑shaped sine window (zero at both ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    """(1 - cos) Hann‑like window."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using Adam.
    Uses the log‑sum‑exp smooth max with temperature β.
    """
    if beta_schedule is None:
        # geometric increase, ending at a very sharp temperature
        beta_schedule = [1, 5, 25, 125, 625, 3125, 15625,
                         78125, 390625, 1953125]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # log‑softmax parameters

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)                     # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                    # soft‑max weights

            # gradient of smooth max w.r.t a  (see derivation in the statement)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t log‑softmax parameters (chain rule)
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded
            log_a = np.clip(log_a, -30.0, 30.0)

            # keep track of the best *exact* surrogate seen
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0,
                     tol_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution.
    Uses the active set of indices whose value is within
    ``tol_factor`` of the current maximum.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        tol = max_val * tol_factor
        active = np.where(conv >= max_val - tol)[0]

        # build sub‑gradient from the active set
        n = len(a)
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.intp)
            rev_idx = k - idx
            g[idx] += 2.0 * a[rev_idx]          # contribution of each active diagonal

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng(0)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method generates a rich set of seeds, selects the best one, then
    performs a long smoothed‑Adam optimisation followed by an exact
    exponentiated‑gradient polish and a brief permutation‑annealing tweak.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good quality / speed trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous global solution, if any – helps continuation runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.clip(arr, 0.0, 1000.0)
                s = prev_vec.sum()
                if s > 0:
                    prev_vec = prev_vec / s
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a fairly large pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0010))
    seeds.append(_exp_decay(n, rate=0.0030))
    seeds.append(_sym_exp_decay(n, rate=0.0020))
    seeds.append(_power(n, p=0.6))
    seeds.append(_power(n, p=1.4))
    seeds.append(_cosine_window(n))
    seeds.append(_sin_power(n, power=0.5))
    seeds.append(_sin_power(n, power=2.0))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.3, 0.8, 1.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random piecewise‑linear shapes
    for nb in (4, 9):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best seed
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    best_seed, best_seed_val = evaluated[0]
    print("\n--- Seed summary (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:5]):
        print(f"  #{i+1}: {v:.6f}")

    print(f"\n[Init] chosen seed surrogate = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Main smoothed Adam optimisation (majority of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.70 * remaining   # 70 % of what is left
    print(f"\n[Adam] allocated time ≈ {adam_deadline - time.time():.1f}s")
    seq_adam, val_adam = _adam_smooth_opt(best_seed,
                                          adam_deadline,
                                          iters_per_beta=40_000)

    print(f"[Adam] best surrogate after Adam = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact EG polish (use whatever time is still available)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[EG] starting exact exponentiated‑gradient polish")
        seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                           deadline,
                                           max_iters=2_000_000,
                                           eta0=6.0,
                                           tol_factor=5e-5)
        if val_eg < val_adam:
            best_seq, best_val = seq_eg, val_eg
            print(f"[EG] improved surrogate = {best_val:.6f}")
        else:
            best_seq, best_val = seq_adam, val_adam
            print("[EG] no improvement over Adam")
    else:
        best_seq, best_val = seq_adam, val_adam

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 15.0
        print("\n[Perm] final short permutation annealing")
        seq_last, val_last = _perm_anneal(best_seq.copy(),
                                          short_deadline,
                                          max_iters=90_000)
        if val_last < best_val:
            best_seq, best_val = seq_last, val_last
            print(f"[Perm] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
