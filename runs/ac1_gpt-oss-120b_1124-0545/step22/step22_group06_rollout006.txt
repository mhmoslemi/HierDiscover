"""
A fast, multi‑stage optimizer for the step‑function evaluation problem.

The algorithm proceeds as follows

1️⃣  Build a diverse pool of *seeds*:
    – classic uniform / triangular / arcsine shapes,
    – a parametric Beta‑U‑shaped family a_i ∝ ((i+½)(n‑½‑i))^{α‑1},
    – low‑concentration Dirichlet draws,
    – the previously saved solution (if any).

2️⃣  Evaluate the cheap smooth surrogate  `2·n·max(conv(a,a))` for all seeds
    and keep the top‑K (default K = 8).

3️⃣  For every top seed run an **Adam optimiser** on a log‑softmax
    parametrisation of the *log‑sum‑exp* smooth approximation of the max‑convolution.
    The schedule uses many β‑values (up to 10⁷) and a larger number of inner
    iterations (8000 per β) than the reference implementation.

4️⃣  Take the best result and polish it with an **exponentiated‑gradient (mirror‑descent)**
    on the *exact* max‑convolution (sub‑gradient at the active index).  The
    number of iterations scales with the remaining time budget.

5️⃣  Finish with a **tiny stochastic walk** on the simplex (local random
    refinement) and, if time permits, a few short Adam restarts from
    random perturbations of the current best.

All stages respect the global deadline (budget_s seconds).  The final
sequence (non‑negative, any scale) is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level convolution utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    Result length = len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact smooth surrogate used throughout the search:
        2·n·max(conv(seq, seq))
    The sequence is assumed to be already normalised (∑seq = 1),
    so this value equals the denominator of ``evaluate_sequence``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑U‑shaped density:
        a_i ∝ ((i+½)(n‑½‑i))^{α‑1}
    For α = 0.5 we recover the arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    # (i+0.5)*(n-0.5-i) ≥ 0 → never zero because i∈[0,n‑1]
    base = (i + 0.5) * (n - 0.5 - i) + eps
    a = np.power(base, alpha - 1.0)
    # Guard against potential NaNs (should not happen)
    a = np.where(np.isnan(a), 0.0, a)
    tot = a.sum()
    if tot == 0.0:
        return _uniform(n)
    return a / tot


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iter_beta: int = 8000,
                    beta_schedule: List[float] = None,
                    lr: float = 0.15) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    approximation of   2·n·max(conv(a,a)).
    Returns the best probability vector seen and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1000.0,
            2000.0, 5000.0,
            20000.0, 50000.0,
            100000.0, 200000.0, 500000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0,
        ]

    # initialise logits (mirror space) from the starting probability vector
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # Smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # Gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # Convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodically evaluate the *exact* surrogate
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the currently active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # relative move proportional to the current mass at i
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored in the global variable ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate call
    deadline = start_time + budget_s - 2.0

    # Problem size (default 4000).  Larger n gives a larger search space
    # and sometimes better values, but also costs more time.
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Add a parametric beta‑U‑shaped family
    for alpha in np.linspace(0.2, 0.9, 15):
        seeds.append(_beta_shape(n, float(alpha)))

    # A handful of low‑concentration Dirichlet draws (very U‑shaped)
    for conc in (0.01, 0.05, 0.1, 0.2):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Initial surrogate evaluation – keep the K best seeds
    # ------------------------------------------------------------------
    K = 8                     # how many seeds survive to the heavy optimisation
    evaluated = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec.astype(np.float64)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        evaluated.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
    evaluated.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in evaluated[:K]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam smoothing on each of the top seeds
    # ------------------------------------------------------------------
    if time.time() < deadline and top_seeds:
        print("\n--- Adam smoothing (smooth max) on top seeds ---")
        # allocate roughly half of the remaining time to this phase
        remaining = deadline - time.time()
        adam_budget = remaining * 0.5
        per_seed_deadline = time.time() + adam_budget / max(1, len(top_seeds))
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_smoothing(
                seed_vec, n,
                per_seed_deadline,
                max_iter_beta=8000,
                lr=0.15
            )
            print(f" [Adam] seed #{idx + 1:02d}: surrogate = {val_adam:.6f}")
            if val_adam < best_val:
                best_val, best_seq = val_adam, seq_adam.copy()
                print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣  Exact exponentiated‑gradient polishing (sub‑gradient of true max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0   # keep a few seconds for the final step
        remaining_time = polishing_deadline - time.time()
        # Scale the allowed number of iterations with the amount of time left.
        total_budget = deadline - start_time
        base_iters = 2_500_000
        max_iters = int(base_iters * remaining_time / total_budget)
        max_iters = max(250_000, min(max_iters, 5_000_000))
        seq_eg, val_eg = _exp_gradient_opt(
            best_seq, n,
            polishing_deadline,
            max_iters=max_iters,
            eta0=5.0
        )
        print(f" [EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 4️⃣  Optional short Adam restarts from random perturbations
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Short Adam restarts from perturbed states ---")
        # Give about 10 % of the remaining budget to this exploratory phase
        restarts_deadline = deadline - 5.0
        while time.time() < restarts_deadline:
            # Create a small random perturbation of the current best
            noise = rng.normal(scale=0.03, size=n)
            pert = best_seq + noise
            pert = np.clip(pert, 0.0, None)
            if pert.sum() == 0.0:
                continue
            pert = pert / pert.sum()
            # Very short Adam run (just a few hundred iterations)
            short_deadline = time.time() + 2.0   # 2 seconds per restart
            seq_tmp, val_tmp = _adam_smoothing(
                pert, n,
                short_deadline,
                max_iter_beta=2000,      # far fewer iterations
                lr=0.12
            )
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, seq_tmp.copy()
                print(f"   >>> Restart improved surrogate to {best_val:.6f}")
            if time.time() > restarts_deadline:
                break

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(
            best_seq, n, deadline,
            attempts=200_000,
            sigma=0.02
        )
        print(f" [Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
