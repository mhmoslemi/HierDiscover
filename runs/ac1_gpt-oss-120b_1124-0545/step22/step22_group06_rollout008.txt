from scipy.special import betainc

def beta_distribution(n, alpha):
    xs = np.arange(n+1) / n
    cdf = betainc(alpha, alpha, xs)  # shape (n+1,)
    a = cdf[1:] - cdf[:-1]
    # ensure sum to 1
    a = a / a.sum()
    return a
```

```python
"""
Search for a step‑function minimizing evaluate_sequence.

The algorithm:
1. Choose a relatively large n (default 12000) and generate a diverse
   collection of probability vectors:
   - uniform, triangular, exact arcsine (Beta(0.5,0.5)),
   - Beta(α,α) for several α values,
   - a few Dirichlet draws.
   All vectors sum to 1.
2. Evaluate the exact surrogate 2·n·max(conv) for each seed and keep the
   best one.
3. Refine the best seed with Adam optimisation on the smooth log‑sum‑exp
   approximation of the max‑convolution.  The schedule of smoothing
   parameters is longer and the inner loop runs twice as many iterations
   as before.
4. Polishing with exponentiated‑gradient (mirror descent) on the true
   max‑convolution.  The surrogate is evaluated only every 500 steps so
   many more sub‑gradient updates fit into the budget.
5. A cheap stochastic walk on the simplex (tiny local search) for the
   remaining time.
6. Return the best sequence (converted to a plain list) and store it in
   the global variable ``height_sequence_1`` for the next invocation.
"""

import time
import math
import numpy as np
from scipy.special import betainc

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) distribution discretised on n equal intervals."""
    xs = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    cdf = betainc(alpha, alpha, xs)   # regularised incomplete beta
    a = cdf[1:] - cdf[:-1]
    # Numerical safety – clamp negatives that may arise from rounding
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        return _uniform(n)
    return a / total


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    surrogate of   2·n·max(conv(a,a)).
    Returns the best sequence (as a probability vector) and its exact
    surrogate value seen during the run.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # extended beta schedule – final β is huge to essentially recover the true max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 200_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # increase number of inner iterations per β (twice the original amount)
    max_iter_beta = 8000

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n-1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation (every 1000 steps)
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0,
                     eval_gap: int = 500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Sub‑gradient is taken at the current active index k = argmax(conv).
    The surrogate is evaluated only every ``eval_gap`` iterations to save
    time.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * np.max(conv))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % eval_gap == 0:
            cur_val = float(2.0 * n * np.max(conv))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final evaluation (just in case we never hit an eval point)
    conv = _conv_fft(a, a)
    final_val = float(2.0 * n * np.max(conv))
    if final_val < best_val:
        best_seq, best_val = a.copy(), final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 150_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a move that respects the simplex
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

"""
Search for a step‑function minimizing ``evaluate_sequence``.

The algorithm

1. Choose a relatively large n (default 12000) – a larger n reduces the
   discretisation error of the optimal continuous solution.
2. Build a diverse pool of probability vectors:
   – uniform, triangular,
   – the exact arcsine law (Beta(½,½)),
   – several Beta(α,α) densities for α∈[0.3,1.0],
   – a few Dirichlet draws.
3. Evaluate the exact surrogate 2·n·max(conv) for each seed and keep the
   best one.
4. Refine the best seed with Adam optimisation on a smooth
   log‑sum‑exp approximation of the max‑convolution.  The inner loop is
   twice as long as the reference implementation and the smoothing
   schedule reaches very large β to recover the true max.
5. Polish the result with exponentiated‑gradient (mirror descent) on the
   *exact* max‑convolution.  The (expensive) surrogate is evaluated only
   every few hundred steps, allowing many more sub‑gradient updates.
6. A cheap stochastic walk on the simplex (tiny random local search)
   uses the remaining time to squeeze out any final improvement.
7. The final sequence (still a probability vector, hence sum = 1) is
   returned and stored in the global ``height_sequence_1`` for the next
   call.
"""

import time
import math
import numpy as np
from scipy.special import betainc

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) on n equal bins – discretised CDF differences."""
    xs = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    cdf = betainc(alpha, alpha, xs)            # regularised incomplete beta
    a = cdf[1:] - cdf[:-1]
    a = np.maximum(a, 0.0)                     # clip tiny negatives
    total = a.sum()
    if total <= 0.0:
        return _uniform(n)
    return a / total


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of
    2·n·max(conv).  Returns the best vector seen (probability vector) and
    its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # A fairly long schedule – it ends with a huge β so that the smooth
    # max becomes essentially the hard max.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 50_000_000.0, 200_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # twice the original number of inner iterations per β
    max_iter_beta = 8000

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0,
                     eval_gap: int = 500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient at the
    active index).  The surrogate is evaluated only every ``eval_gap``
    iterations to keep the run fast.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = float(2.0 * n * np.max(conv))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % eval_gap == 0:
            cur_val = float(2.0 * n * np.max(conv))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final evaluation
    conv = _conv_fft(a, a)
    final_val = float(2.0 * n * np.max(conv))
    if final_val < best_val:
        best_seq, best_val = a.copy(), final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 150_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: move a small random amount from
    one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
