"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm
---------
1. **Problem size** – we keep n = 4000 (the same size used by the
   reference solution).  The objective for a probability vector
   (sum = 1) is 2·n·max(conv(a,a)), so we work directly on the simplex.

2. **Initial pool** – a small diversified set of seeds:
   * the previous best sequence (if any);
   * exponential decays with several rates;
   * a symmetric triangular shape;
   * the uniform distribution.
   All seeds are normalised to sum‑to‑1.

3. **Accelerated mirror descent** – each seed is refined by an
   Adam‑style optimizer on the log‑softmax parametrisation.
   *Variables*: v = log(a) (a = softmax(v)).
   *Gradient*: for the active convolution index k* we have
        g_a = 2·C_{k*}·a,   C_{k*}[i,j]=1 iff i+j=k*.
   The chain‑rule gives ∇_v = a·(g_a – (g_a·a)).
   Adam’s moments (m, v) are maintained and the step size is
   decayed automatically.

   The optimiser runs until the global time limit is approached
   (or a maximum iteration count).  The best vector seen for the
   seed is kept.

4. **Final polishing** – a classic exponentiated‑gradient (mirror
   descent) pass (the same sub‑gradient used by the reference code)
   is applied to the best vector found, using a long decreasing
   stepsize schedule.  This often squeezes a few extra digits.

5. **Evaluation & storage** – the final vector is evaluated with the
   provided ``evaluate_sequence`` function, printed and stored in the
   global variable ``height_sequence_1`` for possible continuation
   runs.

The whole routine is self‑contained, uses only NumPy (and SciPy if
available for an optional smoothed L‑BFGS warm‑start) and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the middle, zero at ends."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax variables
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the exact max‑convolution objective.
    Returns the best sequence seen (as a probability vector) and its
    surrogate value.
    """
    # initialise in log‑space
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # convolution and location of the current maximum
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of max entry w.r.t. a:
        # g_a[i] = 2 * a[k - i]  if 0 <= k-i < n else 0
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        # chain rule: ∇_v = a * (g_a - (g_a·a))
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate surrogate
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to the reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin (≈5 s)
    deadline = start_time + budget_s - 5.0

    # Problem dimension
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    # 1) previous best, if available
    prev = globals().get("height_sequence_1")

        except Exception:

    # 2) exponential decays with several rates
    for rate in (0.0005, 0.001, 0.002, 0.005):

    if not candidates:
        candidates.append(_exp_decay(n, rate=0.001))

    # ------------------------------------------------------------------
    # Optimise each candidate with Adam and keep the best
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    # Allocate a rough budget per candidate (dynamic)
    remaining_time = deadline - time.time()
    max_iters_per_candidate = 300_000  # generous default

    for idx, cand in enumerate(candidates, start=1):
        # Normalise to simplex
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:

        cand = cand / cand.sum()

        # Dynamically shrink iteration budget if we are running out of time

        iters = max(iters, 10_000)   # always allow a minimal run

        seq, val = _adam_opt(cand, n, deadline, max_iters=iters,

        print(f"  → surrogate after Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"  ** new global best (surrogate) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient (high‑iteration pass)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – generate a simple decay if everything failed
        best_overall_seq = _exp_decay(n, rate=0.001)
        best_overall_seq = best_overall_seq / best_overall_seq.sum()

        polished_seq, polished_val = _exp_gradient_opt(

            best_overall_val = polished_val

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs

    return best_overall_seq.tolist()
