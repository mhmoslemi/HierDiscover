"""
Exact (conv‑max) optimisation with a multi‑active subgradient,
followed by an aggressive simulated‑annealing fine‑tune.
The optimiser works on the ℓ₁‑normalised vector (the objective
2·n·max(conv)/∑a² is scale‑invariant) and respects the 1000 s budget.

Algorithm
---------
1. Load the previously found vector (global ``height_sequence_1``) if any,
   otherwise start from a uniform distribution of length 4000.
2. Phase‑1 – cheap exponentiated‑gradient (single worst shift).  This
   gets us quickly into a good basin.
3. Phase‑2 – sub‑gradient descent where all shifts attaining the current
   maximum are active.  This removes the bias of a single‑active update
   and typically pushes the value a little lower.
4. Phase‑3 – simulated‑annealing with a large initial temperature and
   relatively large mass‑transfer sigma.  The SA explores the neighbourhood
   of the best point found by the previous phases.
5. Return the best sequence (as a plain list) and store it in the global
   ``height_sequence_1`` for the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when ∑a = 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase‑1 – exponentiated‑gradient with a *single* active shift
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())                     # single active shift

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = a[k - idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase‑2 – sub‑gradient using *all* active (max‑tight) shifts
# ----------------------------------------------------------------------
def _subgradient_multi(start_seq: np.ndarray,
                       deadline: float,
                       eta0: float = 8.0,
                       max_iters: int = 500_000,
                       tol_rel: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient where the gradient is the sum of contributions
    from every shift attaining the current maximal convolution value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        # indices with value within a tiny relative tolerance of the max
        tol = max(tol_rel, M * tol_rel)
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient: sum_i a_{k-i} over all active k
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] += a[k - idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase‑3 – simulated‑annealing refinement (larger moves)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 800_000,
                                T_start: float = 0.05,
                                T_end: float = 1e-9,
                                sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """SA that moves a (potentially) larger amount of probability mass."""
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # move a *fraction* of the donor mass (up to sigma of its current value)
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        # evaluate surrogate (fast, no clipping)
        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the 1000 s overall budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (height vector) that minimises
    ``evaluate_sequence``.  Returns the best list found within the
    allocated time budget.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Initialise – reuse previous solution if available, otherwise uniform
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            s = arr.sum()
            if s <= 0.0:
                raise ValueError
            seq0 = arr / s                     # ℓ₁‑normalise
            n = seq0.shape[0]
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    print(f"\n=== Optimisation started – n = {n} ===")
    print(f"Initial surrogate value: {_surrogate_obj(seq0):.6f}")

    # --------------------------------------------------------------
    # Phase‑1: exponentiated‑gradient (single‑active shift)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    eg_deadline = time.time() + max(5.0, 0.30 * remaining)   # ~30 % of time
    print("\n--- Phase 1: Exponentiated‑gradient (single‑active) ---")
    seq_eg, val_eg = _exp_gradient_opt_long(seq0, eg_deadline,
                                            eta0=5.0,
                                            max_iters=5_000_000)
    print(f"[Phase 1] surrogate = {val_eg:.6f}")

    # --------------------------------------------------------------
    # Phase‑2: multi‑active sub‑gradient (exact max‑convolution)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + max(5.0, 0.30 * remaining)   # another ~30 %
        print("\n--- Phase 2: Multi‑active sub‑gradient descent ---")
        seq_sub, val_sub = _subgradient_multi(seq_eg, sub_deadline,
                                              eta0=8.0,
                                              max_iters=500_000)
        print(f"[Phase 2] surrogate = {val_sub:.6f}")
    else:
        seq_sub, val_sub = seq_eg, val_eg

    # --------------------------------------------------------------
    # Phase‑3: simulated‑annealing fine‑tuning (larger moves)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_deadline = time.time() + remaining
        print("\n--- Phase 3: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(seq_sub, sa_deadline,
                                                     rng,
                                                     max_attempts=800_000,
                                                     T_start=0.05,
                                                     T_end=1e-9,
                                                     sigma_start=0.01)
        print(f"[Phase 3] surrogate = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_sub, val_sub

    # --------------------------------------------------------------
    # Choose the best sequence discovered
    # --------------------------------------------------------------
    best_seq, best_val = (seq_sa, val_sa) if val_sa < val_sub else (seq_sub, val_sub)

    print(f"\n=== Final surrogate after polishing = {best_val:.6f} ===")

    # --------------------------------------------------------------
    # True evaluation (may differ slightly due to clipping/round‑off)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
