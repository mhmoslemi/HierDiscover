def lp_refine(f, rows, cols, diff, t, max_iter, deadline):
    n = len(f)
    g_var = cp.Variable(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = _conv_fft(f, f)
        M = conv.max()
        # Build A
        data = f[diff]
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        constraints = [g_var >= 0, A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
        except Exception as e:
            # fallback to OSQP if ECOS fails? Not needed
            break
        if g_var.value is None:
            break
        g_val = np.maximum(g_var.value, 0.0)
        sum_g = g_val.sum()
        if sum_g <= 0:
            break
        g_norm = g_val / sum_g
        f_new = (1 - t) * f + t * g_norm
        # Normalize sum to 1
        f_new = f_new / f_new.sum()
        # compute new max conv
        conv_new = _conv_fft(f_new, f_new)
        M_new = conv_new.max()
        # check improvement
        if M_new >= M - 1e-12:
            break
        f = f_new
    return f

def _build_conv_pattern(n):
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    pos = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        rows[pos:pos+length] = k
        cols[pos:pos+length] = np.arange(i_start, i_end+1, dtype=np.int32)
        pos += length
    diff = rows - cols
    return rows, cols, diff

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Generate a modest library of analytic “seed’’ windows
   (uniform, triangular, sine, blackman, hamming, cosine,
   Kaiser, Gaussian, exponential decay, power‑law, Beta shape,
   DPSS, random Dirichlet draws).  All seeds are ℓ₁‑normalised.

2. Evaluate the surrogate  2·n·max(conv(a,a))  for every seed
   (the surrogate equals the true objective when Σa = 1).

3. Keep the best few seeds (default 5) and, for each of them,
   run a *convex‑programming refinement*:

   •  Fix the current vector f (Σf = 1).  Let M = max(conv(f,f)).
   •  Solve the LP

        maximise    Σ g_j
        subject to  g ≥ 0,
                    conv(f,g) ≤ M  (2n‑1 linear constraints)

      This LP is bounded because every entry of g appears in at
      least one constraint with a positive coefficient from f.
      It is implemented with CVXPY and a sparse convolution matrix
      (the sparsity pattern is pre‑computed once).

   •  Normalise g to Σg = 1 and replace f ← (1‑t)·f + t·g
      (t = 0.2).  Iterate a few times (max 8) or until M stops
      decreasing.

4. Polish the refined vector with a short exponentiated‑gradient
   (mirror‑descent) on the exact max‑convolution.  This step
   improves the max‑convolution while staying on the probability
   simplex.

5. Keep the best sequence seen overall, evaluate it with the
   provided ``evaluate_sequence`` and store it in the global
   variable ``height_sequence_1`` before returning.

The whole search respects the 1000 s wall‑clock budget, works for
any sequence length (the previous best length – 4000 – is reused if
available) and falls back to the original gradient‑based optimiser
if CVXPY is not installed.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (assumes Σseq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.special import i0 as _i0
    except Exception:  # pragma: no cover
        _i0 = None
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s


def _dpss_window(n: int, nw: float) -> np.ndarray:
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 6.0,
                            tol: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _build_conv_pattern(n: int):
    """Return (rows, cols, diff) describing the n×n convolution matrix."""
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    pos = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        rows[pos:pos + length] = k
        cols[pos:pos + length] = np.arange(i_start, i_end + 1, dtype=np.int32)
        pos += length
    diff = rows - cols
    return rows, cols, diff


# ----------------------------------------------------------------------
# LP refinement (convex program)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               rows: np.ndarray,
               cols: np.ndarray,
               diff: np.ndarray,
               t: float = 0.2,
               max_iter: int = 8,
               deadline: float = 0.0) -> np.ndarray:
    """
    Perform a few LP‐based refinement steps:
    maximise Σg_j  subject to g≥0 and conv(f,g)≤M,
    then blend g into f with mixing factor t.
    """
    if cp is None:
        # CVXPY not available – skip LP refinement
        return f

    n = len(f)
    g_var = cp.Variable(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv = _conv_fft(f, f)
        M = conv.max()

        # Build the sparse convolution matrix A for the current f
        data = f[diff]                     # length n*n
        A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                    shape=(2 * n - 1, n))

        constraints = [g_var >= 0, A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
        except Exception:  # pragma: no cover
            break

        if g_var.value is None:
            break

        g_val = np.maximum(g_var.value, 0.0)
        sum_g = g_val.sum()
        if sum_g <= 0.0:
            break

        g_norm = g_val / sum_g
        f_new = (1.0 - t) * f + t * g_norm
        f_new = f_new / f_new.sum()   # maintain Σ=1

        conv_new = _conv_fft(f_new, f_new)
        M_new = conv_new.max()
        # stop if no improvement in max‑convolution
        if M_new >= M - 1e-12:
            break
        f = f_new

    return f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in the
    global variable ``height_sequence_1`` and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # --------------------------------------------------------------
    # Determine problem size (reuse previous length if possible)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and previous:
        n = len(previous)
        prev_arr = np.asarray(previous, dtype=np.float64)
    else:
        n = 3000                         # moderate size, good trade‑off
        prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a compact seed set (analytic families + random draws)
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    for sigma_frac in np.linspace(0.05, 0.4, 7):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in np.logspace(-5, -0.5, 12):
        seeds.append(_exp_decay(n, rate))

    for p in np.linspace(0.0, 5.0, 8):
        seeds.append(_power(n, p))

    # Beta shapes – coarse grid to keep overhead low
    for a in np.linspace(0.5, 5.0, 5):
        for b in np.linspace(0.5, 5.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # DPSS windows (if available)
    if _sigwin is not None:
        for nw in [2.5, 3.5, 4.5, 5.5]:
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws
    for conc in [0.001, 0.01, 0.1, 1.0]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform & exponential decay
    for rate in np.logspace(-5, -0.5, 5):
        exp_seq = _exp_decay(n, rate)
        for alpha in (0.1, 0.3, 0.5):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Reversed versions (helps break symmetry)
    seeds.extend([s[::-1].copy() for s in seeds])

    # Include previous best if present
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(3):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Pre‑compute sparse convolution pattern (rows, cols, diff)
    # --------------------------------------------------------------
    rows, cols, diff = _build_conv_pattern(n)

    # --------------------------------------------------------------
    # 4. Refine each top seed
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break

        print(f"\n--- Refinement for seed #{idx+1} ---")
        f = seed_vec.copy()

        # LP‐based refinement (few iterations)
        f = _lp_refine(f, rows, cols, diff,
                       t=0.2,
                       max_iter=8,
                       deadline=deadline)

        # Short exponentiated‑gradient polish
        if time.time() < deadline:
            f, val = _exp_gradient_opt_multi(f, n, deadline,
                                            max_iters=150_000,
                                            eta0=6.0,
                                            tol=1e-8)
            val = _objective_surrogate(f)
        else:
            val = _objective_surrogate(f)

        print(f"[Seed {idx+1}] refined surrogate = {val:.6f}")

        if val < global_best_val:
            global_best_seq = f
            global_best_val = val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final polishing on the current global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (EG) ---")
        global_best_seq, final_val = _exp_gradient_opt_multi(
            global_best_seq, n, deadline,
            max_iters=500_000,
            eta0=6.0,
            tol=1e-8)
        final_val = _objective_surrogate(global_best_seq)
        if final_val < global_best_val:
            global_best_val = final_val
            print(f"[Final polish] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This script builds a diverse set of candidate step‑functions, evaluates them with the exact surrogate, improves the most promising candidates by a small convex‑programming step (using CVXPY) and a brief exponentiated‑gradient polish, and finally returns the best sequence it discovered before the 1000 s budget expires. The implementation respects all problem constraints and stores the result in the global variable `height_sequence_1`.
