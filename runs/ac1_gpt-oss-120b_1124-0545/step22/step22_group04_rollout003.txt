def optimize_seq(start_seq, deadline, max_iters, eta0=1.0, use_adam=False):
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    if use_adam:
        m = np.zeros_like(log_a)
        v = np.zeros_like(log_a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t = 0

    best_seq = a.copy()
    best_val = surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = conv_fft(a, a)
        max_c = conv.max()
        # active indices
        # tolerance relative to max; use absolute tolerance maybe 1e-12
        tol = 1e-12
        active = np.where(conv >= max_c - tol)[0]

        g_a = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx-(n-1))
            i_end = min(n-1, idx)
            # i = i_start..i_end
            i_range = np.arange(i_start, i_end+1)
            rev_i = idx - i_range
            g_a[i_range] += 2.0 * a[rev_i]

        # scale g: maybe average over active count
        if len(active) > 1:
            g_a /= len(active)

        # compute gradient w.r.t logits
        dot = np.dot(g_a, a)
        grad_log = a * (g_a - dot)

        if use_adam:
            t += 1
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            # learning rate schedule: eta = eta0 / sqrt(it) maybe
            eta = eta0 / math.sqrt(it)
            log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        else:
            eta = eta0 / math.sqrt(it)
            log_a -= eta * grad_log

        a = softmax(log_a)

        # occasionally evaluate
        if it % 500 == 0:
            val = surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
    return best_seq, best_val
```

def _mirror_descent_opt(start_seq, deadline, max_iters=1_000_000, eta0=5.0, use_adam=False):
    a = start_seq.copy()
    n = len(a)
    # compute initial logit
    log_a = np.log(a + 1e-30)

    # Adam state
    if use_adam:
        m = np.zeros_like(log_a)
        v = np.zeros_like(log_a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t = 0

    best_a = a.copy()
    conv = _conv_fft(a,a)
    best_val = 2.0*n*np.max(conv)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_c = conv.max()
        # active indices tolerance
        tol = max_c * 1e-12
        active = np.where(conv >= max_c - tol)[0]

        g_a = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i_range = np.arange(i_start, i_end + 1)
            rev_i = idx - i_range
            g_a[i_range] += 2.0 * a[rev_i]
        if len(active) > 1:
            g_a /= len(active)

        # gradient w.r.t logit
        dot = np.dot(g_a, a)
        grad_log = a * (g_a - dot)

        if use_adam:
            t += 1
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            eta = eta0 / math.sqrt(it)
            log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        else:
            eta = eta0 / math.sqrt(it)
            log_a -= eta * grad_log

        a = _softmax(log_a)

        # evaluate periodically
        if it % 500 == 0:
            cur_val = 2.0*n*np.max(_conv_fft(a,a))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    # final evaluation
    final_val = 2.0*n*np.max(_conv_fft(best_a, best_a))
    return best_a, final_val

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution x*y via real FFT.
    Length of the result is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple soft‑max (maps ℝⁿ → simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; result sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact, no smoothing): 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _surrogate(a: np.ndarray, n: int) -> float:
    """Return 2·n·max( a convolved with itself )."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (sub‑gradient of the max‑convolution)
# ----------------------------------------------------------------------
def _mirror_descent_opt(start_seq: np.ndarray,
                        deadline: float,
                        max_iters: int = 10_000_000,
                        eta0: float = 10.0,
                        use_adam: bool = False) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector on the simplex using mirror descent.
    The objective is the exact max‑convolution (no smoothing).
    If ``use_adam`` is True, an Adam‑style adaptive step size is used.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    n = a.size
    # work in log‑space so the simplex constraint is automatic
    log_a = np.log(a + 1e-30)

    # Adam state
    if use_adam:
        m = np.zeros_like(log_a)
        v = np.zeros_like(log_a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0

    # best‑so‑far bookkeeping
    best_a = a.copy()
    best_val = _surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- convolution and active set ----
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # active indices – use a tiny relative tolerance to capture ties
        tol = max_c * 1e-12
        active = np.where(conv >= max_c - tol)[0]

        # ---- sub‑gradient of the active max‑entries ----
        g_a = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            rev_i = idx - i_range
            g_a[i_range] += 2.0 * a[rev_i]
        if len(active) > 1:
            g_a /= len(active)

        # gradient w.r.t. logits (mirror‑descent update)
        dot = np.dot(g_a, a)
        grad_log = a * (g_a - dot)

        # ---- step‑size / Adam ----
        if use_adam:
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            eta = eta0 / math.sqrt(it)
            log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        else:
            eta = eta0 / math.sqrt(it)
            log_a -= eta * grad_log

        # back to the simplex
        a = _softmax(log_a)

        # ---- occasional evaluation (every 500 iters) ----
        if it % 500 == 0:
            cur_val = _surrogate(a, n)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

    # final evaluation on the best candidate
    final_val = _surrogate(best_a, n)
    return best_a, final_val


# ----------------------------------------------------------------------
# Very cheap hill‑climbing on the exact max‑convolution
# ----------------------------------------------------------------------
def _hill_climb_opt(start_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local greedy search: repeatedly move a small amount of mass from a
    larger coordinate to a smaller one.  Accept only improvements (with a
    tiny probability of uphill moves to escape tiny plateaus).
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_val = _surrogate(a, n)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # pick donor i (probability proportional to its current weight)
        i = rng.integers(0, n)
        if a[i] == 0.0:
            continue
        # pick a distinct recipient j
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the donor mass
        eps = a[i] * rng.uniform(0.01, 0.07)  # 1‑7 % of donor
        if eps <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        cur_val = _surrogate(cand, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = cand.copy()
            a = cand                     # accept move
        else:
            # very occasional uphill acceptance (helps escape tiny plateaus)
            if rng.random() < 0.001:
                a = cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a 4000‑element non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine follows the multi‑phase strategy
    described in the module doc‑string and respects the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    n = 4000

    # ------------------------------------------------------------------
    # Initialise from a stored solution if present; otherwise exponential.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        init = np.asarray(prev, dtype=np.float64)
        init = np.maximum(init, 0.0)
        if init.sum() == 0.0:
            init = np.full(n, 1.0 / n)
        else:
            init = init / init.sum()
    else:
        # exponential decay – a simple decent starting point
        init = np.exp(-0.001 * np.arange(n, dtype=np.float64))
        init = init / init.sum()

    # ------------------------------------------------------------------
    # Phase 1 – short random‑restart mirror descent explorations
    # ------------------------------------------------------------------
    best_seq = init.copy()
    best_val = _surrogate(best_seq, n)
    print(f"[Init] surrogate = {best_val:.6f}")

    # allocate roughly 20 % of the remaining budget for restarts
    restart_budget = (deadline - time.time()) * 0.20
    restart_start = time.time()
    restart_idx = 0
    while time.time() < deadline - 30.0 and (time.time() - restart_start) < restart_budget:
        # each restart gets at most ~25 s (or the remaining slice)
        slice_time = min(25.0, deadline - time.time() - 30.0)
        sub_deadline = time.time() + slice_time

        # random Dirichlet init – concentration controls sparsity
        alpha = rng.uniform(0.05, 3.0)
        a0 = rng.dirichlet(np.full(n, alpha))

        seq, val = _mirror_descent_opt(a0,
                                       sub_deadline,
                                       max_iters=500_000,
                                       eta0=5.0,
                                       use_adam=False)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Restart {restart_idx}] Improved surrogate = {best_val:.6f}")

        restart_idx += 1

    # ------------------------------------------------------------------
    # Phase 2 – long final polishing (Adam‑accelerated mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (Adam) ---")
        final_seq, final_val = _mirror_descent_opt(best_seq,
                                                   deadline,
                                                   max_iters=10_000_000,
                                                   eta0=10.0,
                                                   use_adam=True)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq.copy()
            print(f"[Polish] Improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap hill‑climbing tweak (if a few seconds remain)
    # ------------------------------------------------------------------
    if time.time() + 15.0 < deadline:
        print("\n--- Hill‑climbing refinement ---")
        seq_hc, val_hc = _hill_climb_opt(best_seq,
                                         deadline,
                                         max_iters=200_000)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()
            print(f"[Hill‑climb] Improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
