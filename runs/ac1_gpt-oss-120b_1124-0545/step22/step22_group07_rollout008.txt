"""
Propose a step‑function (non‑negative heights) minimizing ``evaluate_sequence``.

Algorithm
---------
1. Build a modest pool of analytic / random seeds (uniform, beta‑shape,
   exponential, power‑law, Gaussian, Dirichlet, plus the previous best if any).
   The beta shape with parameters (0.5,0.5) is the continuous arcsine density,
   a known “U‑shaped’’ candidate that tends to flatten the self‑convolution.

2. Pick the seed with the lowest *surrogate* objective
   (the exact objective for a probability vector: 2·n·max(conv)).

3. Run a short smoothed Adam optimisation in log‑softmax space to obtain a
   smoother starting point.

4. Run a long exponentiated‑gradient (mirror‑descent) procedure on the exact
   max‑convolution.  The update is performed on the log‑probabilities,
   includes a decaying step size ηₜ = η₀/√t, a small isotropic Gaussian noise
   (σₜ = σ₀/√t) to escape shallow local minima and a simple momentum term.

5. Finish with a brief permutation‑annealing (swap‑moves) to polish the final
   vector.

All heavy work uses an FFT‑based convolution (O(n log n)).  The routine obeys
the global time budget (default 1000 s) and stores the final sequence in the
global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real, length = 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (for a probability vector: sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when ``seq`` sums to 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised np.ndarray
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {1,…,n} (discrete, then normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smoothed version of max(conv).
    The smoothing uses a log‑sum‑exp with temperature β.
    """
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200, 200_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)  # log‑softmax parameters

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probability vector
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp  # soft‑max weights

            # gradient w.r.t a (see main script for derivation)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to log‑softmax space
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution,
# with a tiny amount of Gaussian noise and simple momentum.
# ----------------------------------------------------------------------
def _exp_gradient_opt_noise(start_seq: np.ndarray,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 8.0,
                            noise0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Adds decaying isotropic Gaussian noise to the log‑probabilities and
    a simple momentum term to accelerate convergence.
    """
    rng = np.random.default_rng()
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    prev_update = np.zeros_like(log_a)
    gamma = 0.9  # momentum coefficient

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # build probability vector from log‑softmax
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        # convolution and active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active diagonal (see comment in analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx
        g[idx] = 2.0 * a[rev]

        # step‑size and noise decay
        eta = eta0 / math.sqrt(it)
        sigma = noise0 / math.sqrt(it)

        # momentum + noise update in log‑space
        update = -eta * g + gamma * prev_update
        update += sigma * rng.normal(size=n)

        log_a += update
        prev_update = update

        # keep log‑values bounded
        log_a = np.clip(log_a, -30.0, 30.0)

        # evaluate current surrogate (cheap, no extra convolution)
        cur_val = _objective_surrogate(np.exp(log_a - np.max(log_a)).astype(np.float64) / np.exp(log_a - np.max(log_a)).sum())
        if cur_val < best_val:
            best_val = cur_val
            best_seq = np.exp(log_a - np.max(log_a))
            best_seq /= best_seq.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

    evaluated.sort(key=lambda x: x[1])
    best_seed, best_val = evaluated[0]
    print(f"[Init] best surrogate from seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Short smoothed Adam optimisation (helps get out of very bad regions)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        adam_deadline = time.time() + min(30.0, time_left / 3.0)
        print("[Adam] running smoothed Adam optimiser")
        seq_adam, val_adam = _adam_smooth_opt(best_seed, adam_deadline,
                                             beta_schedule=[10, 50, 200, 800, 3200, 12800],
                                             iters_per_beta=20_000)
        if val_adam < best_val:
            best_seed = seq_adam
            best_val = val_adam
            print(f"    Adam improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main exponentiated‑gradient optimisation with noise and momentum
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 20.0:
        eg_deadline = time.time() + time_left - 20.0   # reserve ~20 s for final tweak
        print("[EG] running exponentiated‑gradient with noise")
        seq_eg, val_eg = _exp_gradient_opt_noise(best_seed, eg_deadline,
                                                 max_iters=2_500_000,
                                                 eta0=8.0,
                                                 noise0=0.2)
        if val_eg < best_val:
            best_seed = seq_eg
            best_val = val_eg
            print(f"    EG improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final short permutation annealing (swap moves)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        anneal_deadline = time.time() + time_left
        print("[Anneal] final permutation‑annealing")
        seq_final, val_final = _perm_anneal(best_seed.copy(),
                                           anneal_deadline,
                                           max_iters=120_000)
        if val_final < best_val:
            best_seed = seq_final
            best_val = val_final
            print(f"    Anneal improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided ``evaluate_sequence`` and store answer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seed.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seed.tolist()
    return best_seed.tolist()
