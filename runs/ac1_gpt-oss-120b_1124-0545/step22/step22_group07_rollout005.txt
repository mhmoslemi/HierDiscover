import numpy as np, math, time, sys
from scipy.optimize import minimize

_global_deadline = None

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n):
    return np.full(n, 1.0 / n)

def _exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()

def _sym_exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()

def _power(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n+1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()

def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _piecewise_linear(n, num_breaks, rng):
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    return a / a.sum()

class _LbfgsObjective:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def __call__(self, log_a):
        # softmax to probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()
        # convolution
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        beta = self.beta
        # compute softmax weights
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp
        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[self.n - 1 : self.n - 1 + self.n]
        # gradient wrt log_a
        grad_log = a * (grad_a - np.dot(grad_a, a))
        # smoothed objective value: M_beta = shift + log(sum_exp)/beta
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log

def _lbfgs_opt(start_seq, beta_schedule, max_iter_per_beta, deadline):
    # start_seq: probability vector (sum=1)
    log_a = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(start_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj = _LbfgsObjective(n, beta)

        # compute remaining time and estimate a safe max_iter based on it
        remaining = deadline - time.time()
        max_iter = max_iter_per_beta
        # L-BFGS may call the function multiple times; limit
        opts = {'maxiter': max_iter, 'disp': False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method='L-BFGS-B',
                           jac=True,
                           options=opts)
        except Exception as e:
            # in case optimizer aborts
            break
        log_a = res.x
        # produce probability vector
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # continue with the same log_a for next beta
    return best_seq, best_val

def _exp_gradient_opt(start_seq, deadline, max_iters=2000000, eta0=5.0):
    # identical to earlier (copy from baseline) but with possible tweak
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # renormalize
        log_a = np.clip(log_a, -30.0, 30.0)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _perm_anneal(start_seq, deadline, max_iters=50000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]
        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that minimises
    evaluate_sequence().  The method builds a modest pool of analytic / random
    seeds, selects the most promising ones, refines them with a continuation
    of L‑BFGS‑B on a soft‑max surrogate (gradually increasing the smoothing
    parameter β), finishes with a long exponentiated‑gradient polish and a
    brief permutation‑annealing tweak.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 2.0  # small safety margin
    deadline = _global_deadline

    # problem size – 4000 points strikes a balance between quality and speed
    n = 4000

    # retrieve previous solution if present (continuation)
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build seed pool
    # ------------------------------------------------------------------
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, 0.0015))
    seeds.append(_exp_decay(n, 0.0035))
    seeds.append(_sym_exp_decay(n, 0.0025))
    seeds.append(_power(n, 0.6))
    seeds.append(_power(n, 1.1))
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))
    for nb in (5, 10):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))
    for _ in range(6):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds.append(_beta_shape(n, a, b))
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate ranking – keep a few best seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement with L‑BFGS (continuation on β)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    beta_schedule = [10, 80, 400, 2000, 10000, 50000]  # progressively sharper
    max_iter_per_beta = 250  # modest; L‑BFGS does a few line‑search steps per iter

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate about a fifth of remaining time for each seed (adjusted)
        share = remaining * 0.2 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # Follow up with a short exponentiated‑gradient polish if time permits
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                              sub_deadline,
                                              max_iters=500_000,
                                              eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        local_seq, local_val = (seq_lbfgs, val_lbfgs) if val_lbfgs < val_eg else (seq_eg, val_eg)

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing on the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation annealing tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 15.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=60_000)
        if val_last < global_best_val:

"""
Search for a non‑negative step‑function (probability vector) that minimises
evaluate_sequence().  The method builds a modest pool of analytic / random
seeds, selects the most promising ones, refines each with a continuation of
L‑BFGS‑B on a smooth log‑sum‑exp surrogate (gradually increasing the
smoothing parameter β), follows with a long exponentiated‑gradient polish
and ends with a short permutation‑annealing tweak.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global deadline – used by the L‑BFGS objective to avoid overrunning.
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq sums to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return a normalised probability vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


def _second(pair):
    return pair[1]


# ----------------------------------------------------------------------
# L‑BFGS surrogate objective (log‑sum‑exp smoothing).
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable needed by scipy.optimize.minimize."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        beta = self.beta

        # smooth max via log‑sum‑exp (weights)
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp

        # gradient w.r.t. a (2 * convolution of reversed a with w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value: M_β = shift + log(sum_exp)/β
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: run L‑BFGS on a sequence of increasingly sharp
    log‑sum‑exp surrogates.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:                      # any interruption – stop early
            break

        log_a = res.x
        # back‑project to probability vector
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (exponentiated gradient).
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max entry (see original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # keep logs bounded to avoid overflow / underflow
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation annealing (swap moves only).
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The pipeline is:

    1. generate a pool of analytic / random seeds;
    2. keep the top few according to the exact surrogate;
    3. refine each with a continuation L‑BFGS on a log‑sum‑exp surrogate;
    4. polish the current global best with a long exponentiated‑gradient run;
    5. finish with a very short permutation‑annealing tweak;
    6. store and return the best sequence.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 2.0     # small safety margin
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Problem size – 4000 points offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) to keep continuity across runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a modest pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
    ]

    # Dirichlet draws with a broad range of concentration parameters.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random piecewise‑linear shapes.
    for nb in (5, 10):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Beta‑shaped families.
    for _ in range(6):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds.append(_beta_shape(n, a, b))

    # Include the previous solution and a few of its random permutations.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=_second)
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement with continuation L‑BFGS.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    beta_schedule = [10, 80, 400, 2000, 10000, 50000]   # increasingly sharp
    max_iter_per_beta = 250                           # modest per‑β budget

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        share = remaining * 0.2 / max(1, (top_k - idx))   # roughly 20 % per seed
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ----- continuation L‑BFGS -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # ----- short EG polish (if time left) -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                              sub_deadline,
                                              max_iters=500_000,
                                              eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # choose the better of the two refinements
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing on the remaining budget.

    if time.time() < deadline:

                                                 eta0=5.0)
        if final_val < global_best_val:

            global_best_val = final_val

    # ------------------------------------------------------------------
    if time.time() < deadline:

                                          short_deadline,
                                          max_iters=60_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # 6. Evaluate with the provided function and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())

            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"

        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
