"""
Improved search for a step‑function minimising ``evaluate_sequence``.

Algorithm
---------
1. Build a diverse pool of initial probability vectors (uniform,
   arcsine‑like, triangular, a few Dirichlet draws and the previous
   solution if available).  Keep the three best seeds according to the
   smooth surrogate 2·n·max(conv).

2. For each of the selected seeds run a *long* Adam optimiser on the
   log‑softmax (mirror) parametrisation.  The smoothing parameter β is
   increased gradually; for each β we perform many inner updates
   (≈8000) and store the best exact surrogate seen.

3. Starting from the globally best vector, run a *sub‑gradient* descent
   on the true max‑convolution (additive updates on the simplex).  The
   step‑size decays as η₀/√t and we project onto the simplex after each
   update.

4. Polishing: an exponentiated‑gradient (mirror‑descent) pass similar
   to the reference code but with a larger iteration budget.

5. Fine mixing: line‑search convex combinations of the current best
   vector with a few simple shapes (uniform, arcsine, triangular).  The
   combination that yields the smallest exact surrogate is kept.

6. Tiny stochastic walk (random local search) on the simplex with a
   decreasing step‑size to squeeze the last few bits.

7. The final sequence (still a probability vector, i.e. sums to 1) is
   handed to ``evaluate_sequence`` and stored as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:          # extreme safety – should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  ``seq`` must sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in log‑softmax space of the log‑sum‑exp smooth
    approximation of 2·n·max(conv).
    Returns the best probability vector and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)          # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a fairly dense β schedule, ending at 1e8
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0,
        2_000_000.0, 5_000_000.0, 10_000_000.0,
        20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        max_iter_beta = 8000                     # more inner work per β
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                  # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            shifted = beta * (conv - max_c)       # ≤ 0, safe exponentiation
            exp_shift = np.exp(shifted)
            w = exp_shift / exp_shift.sum()       # softmax on β·conv

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = n + (2n‑1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted               # already the true gradient

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # simple schedule: slowly shrink learning‑rate for large β
            lr = 0.15 if beta < 1e6 else 0.08

            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep exact best every few steps
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final clean‑up
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *true* max‑convolution (additive updates)
# ----------------------------------------------------------------------


def _subgradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 400_000,
                     eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact max‑convolution.
    Variables live on the simplex; after each step we project back by
    clipping negatives and re‑normalising.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        a = a - eta * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:               # numerical safeguard – re‑initialise
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (mirrored sub‑gradient)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the genuine max‑convolution (same update as in the
    reference code).  Returns the best exact surrogate seen.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement (stochastic walk)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small random
    amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Mixing of two probability vectors (convex combination)
# ----------------------------------------------------------------------


def _mix_search(base_seq: np.ndarray,
                mixes: list[np.ndarray],
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple line‑search over convex combinations of ``base_seq`` with each
    vector in ``mixes``.  Returns the best mixture found.
    """
    best_seq = base_seq.copy()
    best_val = _objective_surrogate(best_seq)
    alphas = np.linspace(0.0, 1.0, 101)  # 0 % … 100 % in steps of 1 %

    for mix in mixes:
        for a in alphas:
            if time.time() > deadline:
                break
            cand = (1.0 - a) * base_seq + a * mix
            # already normalised because both sum to 1
            val = _objective_surrogate(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
    return best_seq, best_val
