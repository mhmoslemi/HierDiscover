"""
Propose a better step‑function (height sequence).

The algorithm improves on the previous version by
* using the correct sub‑gradient of the exact max‑convolution
  (the old code missed a factor 2 and the diagonal case);
* adding a *smooth* approximation of the max (log‑sum‑exp) so that all
  “almost‑worst’’ shifts influence the update;
* performing a few independent restarts with random perturbations of the
  best sequence found so far;
* finishing each restart with a simulated‑annealing fine‑tuning.

All operations stay on the probability simplex (non‑negative, ℓ₁‑norm = 1)
by means of an exponentiated‑gradient (multiplicative‑weights) update.
The code respects the 1000 s overall budget and stores the best
construction in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (surrogate) – assumes sum(a)=1
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for an ℓ₁‑normalised vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the *smooth* max‑convolution
# ----------------------------------------------------------------------
def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a,a)  with respect to a
    (w is a probability vector of length 2·n‑1).
    The result is  2· Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    L = len(w)
    w_rev = w[::-1]
    conv = _conv_fft(a, w_rev)                 # length n+L‑1
    idx = L - 1 - np.arange(n)                 # L‑1‑i  for i=0…n‑1
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_smoothed(start_seq: np.ndarray,
                               deadline: float,
                               max_iters: int = 8000,
                               eta0: float = 3.0,
                               tau0: float = 0.12,
                               tau_min: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a *smooth* approximation of
    the max‑convolution.  The surrogate (2·n·max(conv)) is monitored and
    the best vector seen is returned.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                  # keep in log‑space

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    tau = tau0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact convolution and its maximum
        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # smooth max: weights = soft‑max over the convolution values
        # (subtract max to avoid overflow)
        shifted = (conv - max_c) / max(tau, 1e-12)
        w = np.exp(shifted)
        w = w / w.sum()

        # gradient of the smooth objective
        grad = _gradient_weighted(a, w)

        # learning‑rate schedule (mirror‑descent)
        eta = eta0 / math.sqrt(it)

        # EG update in log‑space
        log_a -= eta * grad
        a = _softmax(log_a)

        # decay smoothing parameter
        tau = max(tau_min, tau * 0.9995)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 300_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.004) -> tuple[np.ndarray, float]:
    """
    Tiny mass transfers between random coordinates, accepting worse moves
    according to a linear cooling schedule.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear temperature decay
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # move a tiny fraction of the mass of i to a random j
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
