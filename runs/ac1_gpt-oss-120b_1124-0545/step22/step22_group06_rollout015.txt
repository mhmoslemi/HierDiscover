"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search consists of three stages:

1. **Diverse deterministic seeds** – besides the uniform, triangular and
   Dirichlet seeds used previously, we generate monotone power‑law and
   exponential decays: a_i ∝ (n‑i)^p, a_i ∝ (i+1)^{-p} and a_i ∝ r^i.
   These families are known to concentrate mass near one side of the
   interval, which drastically lowers the maximal linear convolution
   (the bottleneck of the objective).  All seeds are normalised to a
   probability vector (sum = 1) and evaluated with a fast FFT‑based surrogate
   `2·n·max(conv)`.

2. **Polishing the best seeds** – the top‑k seeds (by surrogate value) are
   refined with the same Adam‑smooth‑max optimiser and an exact
   exponentiated‑gradient (mirror descent) phase that directly minimises the
   true ``max(conv)``.  The schedules are slightly stretched compared to the
   reference implementation, allowing more optimisation steps within the
   1000 s budget.

3. **Final tiny stochastic walk** – a cheap random‑local‑search that moves a
   tiny amount of mass between two coordinates and keeps improvements.

The whole pipeline is timed; the best sequence found before the deadline
is stored in the global variable ``height_sequence_1`` and returned as a list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(seq, seq))   (seq is assumed to sum to 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_decreasing(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{-p}    (decreasing power‑law, a_0 largest).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a = a / a.sum()
    return a


def _reverse_power_decreasing(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (n-i)^{p}    (also decreasing, but the values are the reverse
    of a simple power‑law, useful for symmetry with the convolution).
    """
    i = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    a = i ** p
    a = a / a.sum()
    return a


def _exponential_decay(n: int, r: float) -> np.ndarray:
    """
    a_i ∝ r^{i}   with 0<r<1.  The decay factor r is chosen close to 1
    so that many entries stay non‑zero while still providing a strong
    asymmetry.
    """
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    surrogate of   2·n·max(conv(a,a)).
    Returns the best probability vector and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # expanded schedule – we go further into the sharp regime
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        1e5, 2e5, 5e5,
        1e6, 2e6, 5e6,
        1e7, 2e7, 5e7, 1e8,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        max_iter_beta = 6000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n-1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasionally keep the best exact surrogate seen
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence found within the time budget
    is stored in the global variable ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size (default 4000)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # power‑law families (both directions)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        seeds.append(_power_decreasing(n, p))
        seeds.append(_reverse_power_decreasing(n, p))

    # exponential decays
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995):
        seeds.append(_exponential_decay(n, r))

    # a few Dirichlet draws with different concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Initial evaluation of seeds (fast surrogate)
    # ------------------------------------------------------------------
    print("\n--- Initial seed evaluation (surrogate) ---")
    evaluated = []
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        evaluated.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    # keep the top‑k seeds for polishing
    evaluated.sort(key=lambda x: x[0])
    top_k = 3
    candidates = [vec for _, vec in evaluated[:top_k]]
    best_val = evaluated[0][0]
    best_seq = evaluated[0][1].copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam smoothing on each top candidate
    # ------------------------------------------------------------------
    polished = []
    if time.time() < deadline:
        print("\n--- Adam smoothing (smooth max) on top seeds ---")
        # allocate roughly half of remaining time to Adam passes
        remaining = deadline - time.time()
        per_candidate_deadline = time.time() + remaining / max(1, len(candidates))
        for idx, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            dline = min(per_candidate_deadline,
                        deadline - 5.0)   # keep a few seconds for later phases
            seq_adam, val_adam = _adam_smoothing(cand, n, dline)
            polished.append((val_adam, seq_adam))
            print(f"   [Adam #{idx+1}] best exact surrogate = {val_adam:.6f}")
            if val_adam < best_val:
                best_val, best_seq = val_adam, seq_adam.copy()
                print("      >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣  Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0
        # scale the number of iterations to the remaining time
        remaining_time = polishing_deadline - time.time()
        max_iters = int(2_000_000 *
                        remaining_time /
                        (deadline - start_time))
        max_iters = max(250_000, min(max_iters, 3_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=5.0)
        print(f"   [EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("      >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=120_000,
                                                sigma=0.02)
        print(f"   [Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("      >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
