"""
A fast, robust optimiser for the “max‑convolution” objective.

The goal is to minimise
    2·n·max(conv(a, a))   (the value returned by ``evaluate_sequence``)
subject to a_i ≥ 0 and Σ a_i = 1 (the scale does not matter because the
objective is homogeneous).

The optimiser consists of three stages:

1. **Initialisation** – try a few simple analytic families (exponential,
   power‑law) and keep the best one as a probability vector.

2. **Smoothing → L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of the
   max‑convolution using SciPy’s L‑BFGS‑B.  A continuation over a schedule of
   temperatures β gradually tightens the surrogate.  The number of L‑BFGS
   iterations is chosen adaptively so that a few minutes of the total budget
   remain for the fine‑tuning stage.

3. **Fine‑tuning with Adam** – work directly on the exact max‑convolution.
   The gradient of the active entry is computed analytically; Adam (with
   momentum) is applied to the log‑softmax parameters, guaranteeing that the
   iterate stays on the simplex.  The loop runs until the time budget is
   exhausted, constantly keeping the best vector seen.

After the main phase a handful of short random‑restart Adam runs are performed
to escape possible shallow local minima, and finally a cheap permutation‑based
hill‑climber is applied.

The best vector is stored in the global variable ``height_sequence_1`` so that
future runs can continue from it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a, a)).  ``a`` must already sum to 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


def _max_conv_subgrad(a: np.ndarray):
    """
    Returns (max_value, subgradient w.r.t. a, index_of_max).
    Sub‑gradient of the active convolution entry:
        g_i = 2·a_{k-i}   for all i with 0≤i<n and 0≤k-i<n,
    where k is the index where the convolution attains its maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))
    max_val = conv[k]

    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = k - idx_i
    g[idx_i] = 2.0 * a[rev_idx]

    return max_val, g, k


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) – used by L‑BFGS and optionally by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed surrogate
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    The gradient is w.r.t. the log‑softmax variables ``v_local``.
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c                       # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the *exact* max‑convolution (sub‑gradient version)
# ----------------------------------------------------------------------
def _adam_max_opt(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  lr: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the exact max‑convolution objective.
    Operates in the log‑softmax space to keep the iterate on the simplex.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    max_val, _, _ = _max_conv_subgrad(a)
    best_val = 2.0 * n * max_val

    t = 0
    last_print = time.time()
    while time.time() < deadline:
        t += 1
        # exact sub‑gradient of the active convolution entry
        max_val, g, _ = _max_conv_subgrad(a)

        # current surrogate value
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # gradient w.r.t. log‑softmax variables
        grad_v = a * (g - np.dot(g, a))

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        # occasional progress report (every ~2 s)
        if time.time() - last_print > 2.0:
            print(f"[Adam] it={t:,}  surrogate={cur_val:.6f}")
            last_print = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based hill climbing (optional final tweak)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two entries or adds a tiny Gaussian perturbation.
    Accepts worsening moves with a simple annealing schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:          # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        else:                            # tiny Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)
        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq, cur_val = ns, val

    return best_seq, best_val
