Enhanced step‑function optimizer

Goal: find a non‑negative vector a (sum = 1) of length n≈4000 that
minimises   2·n·max(conv(a,a))   – the same quantity returned by
``evaluate_sequence`` (the value is invariant under scaling).

Improvements over the previous version
---------------------------------------
1. **Coarse‑to‑fine multi‑resolution search**:
   – First optimises a much smaller prototype (n≈512) with cheap
     exponentiated‑gradient iterations.
   – Upsamples the best prototype to the target length and uses it as
     a high‑quality seed for the full‑scale optimisation.
2. **Wider seed library** (including sine windows with γ<1,
   additional Kaiser β, and mixtures of uniform+exponential).
3. **More aggressive long‑run polishing**:
   – Larger iteration budgets for the multi‑active exponentiated‑gradient
     phase.
   – A longer “flattening” phase that drives the whole convolution
     towards a constant.
   – A final simulated‑annealing polish with a higher starting temperature.
4. **Simple Gaussian smoothing post‑process** – a few passes of a
   small Gaussian kernel often lower the peak of the autocorrelation
   without harming the ℓ₁‑norm.
5. **Dynamic time‑budget handling** – each stage receives a share of the
   remaining wall‑clock time, guaranteeing that the function always
   returns before the 1000 s limit.

All updates stay on the probability simplex either via a soft‑max
parameterisation (exponentiated‑gradient) or via an Euclidean projection
(used only internally by helper utilities).  The best sequence found is
stored in the global name ``height_sequence_1`` and returned as a plain
list.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects 1‑D array")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.nonzero(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))  (sum(seq)==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    # simple momentum (Adam‑like without bias correction)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices that are “almost” maximal
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient–descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)².
    The aim is to make the whole convolution as flat as possible.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of the full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent update (log‑space)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        ssum = ns.sum()
        if ssum == 0.0:
            attempts += 1
            continue
        ns /= ssum

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple Gaussian smoothing (post‑processing)
# ----------------------------------------------------------------------
def _gaussian_smooth(seq: np.ndarray, sigma: float = 1.0, passes: int = 3) -> np.ndarray:
    """
    Convolves the sequence with a small 1‑D Gaussian kernel a few times.
    The result is re‑normalised to sum = 1.
    """
    if sigma <= 0:
        return seq
    # build kernel of odd length covering ~3σ on each side
    radius = int(3 * sigma)
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()

    smoothed = seq.copy()
    for _ in range(passes):
        smoothed = np.convolve(smoothed, kernel, mode='same')
        smoothed = np.maximum(smoothed, 0.0)
        total = smoothed.sum()
        if total > 0:
            smoothed /= total
    return smoothed

        seq_opt, val_opt = _exp_gradient_opt_multi(seq0.copy(),
                                                   n_small,
                                                   sub_dead,
                                                   max_iters=250_000,
                                                   eta0=9.0,
                                                   tol=1e-9)
        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small_seq = seq_opt.copy()
        print(f"[Prototype {idx+1}/{len(top_small)}] surrogate = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 2. ---- Upsample prototype to full size ------------------------------------
    # ------------------------------------------------------------------
    if best_small_seq is None:
        prot_big = _uniform(n_big)
    else:
        # linear interpolation from n_small -> n_big
        xs = np.linspace(0, n_small - 1, n_big)
        prot_big = np.interp(xs, np.arange(n_small), best_small_seq)
        prot_big = np.maximum(prot_big, 0.0)
        s = prot_big.sum()
        if s > 0:
            prot_big /= s
        else:
            prot_big = _uniform(n_big)

    # ------------------------------------------------------------------
    # 3. ---- Large‑scale optimisation -------------------------------------------
    # ------------------------------------------------------------------
    print("\n=== Phase 2: large‑scale optimisation (n={}) ===".format(n_big))
    # Gather a rich set of seeds for the large problem
    seeds_big = [
        prot_big,
        _uniform(n_big),
        _triangular(n_big),
        _sine_window(n_big, gamma=0.5),
        _sine_window(n_big, gamma=0.8),
        _sine_window(n_big, gamma=1.2),
        _blackman_window(n_big),
        _hamming_window(n_big),
        _cosine_window(n_big),
        _kaiser_window(n_big, beta=2.0),
        _kaiser_window(n_big, beta=6.0),
        _kaiser_window(n_big, beta=12.0),
        _gaussian(n_big, sigma_frac=0.08),
        _gaussian(n_big, sigma_frac=0.20),
        _power(n_big, p=1.0),
        _power(n_big, p=3.0),
    ]

    # DPSS windows (if available)
    if _sigwin is not None:
        for nw in (3.0, 4.0, 5.0):
            seeds_big.append(_dpss_window(n_big, nw))

    # Random Dirichlet draws with a variety of concentrations
    for conc in (0.001, 0.01, 0.05, 0.2):
        for _ in range(2):
            seeds_big.append(_random_dirichlet(n_big, conc, rng))

    # Mixtures of uniform & exponential tails
    exp_seq = _exp_decay(n_big, rate=1e-4)
    for alpha in (0.1, 0.3, 0.5):
        seeds_big.append(_mix_shapes(_uniform(n_big), exp_seq, alpha))

    # Reversed copies to break symmetry
    seeds_big.extend([s[::-1].copy() for s in list(seeds_big)])

    # Quick surrogate evaluation of all large seeds
    big_evals = []
    for vec in seeds_big:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        big_evals.append((vec, val))

    big_evals.sort(key=lambda x: x[1])
    top_k = min(12, len(big_evals))
    top_seeds = [vec for vec, _ in big_evals[:top_k]]

    print("\nTop large seeds (surrogate values):")
    for i, (_, v) in enumerate(big_evals[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # Initialise global best with the best seed
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. ---- Multi‑start refinement on the large problem -------------------------
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair share of the remaining budget
        remaining_total = deadline - time.time()
        share = remaining_total * 0.11 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        # a) exponentiated‑gradient (multi‑active)
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                 n_big,
                                                 sub_deadline,
                                                 max_iters=350_000,
                                                 eta0=10.0,
                                                 tol=1e-9)

        # b) flattening gradient phase
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_eg,
                                                       n_big,
                                                       sub_deadline,
                                                       max_iters=250_000,
                                                       eta0=6.0)
        else:
            seq_flat, val_flat = seq_eg, val_eg

        # c) simulated annealing polish
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_flat,
                                                  n_big,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=100_000,
                                                  start_sigma=0.0012,
                                                  T_start=0.02,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_flat, val_flat

        # d) optional Gaussian smoothing (cheap)
        seq_smooth = _gaussian_smooth(seq_sa, sigma=0.8, passes=2)
        val_smooth = _objective_surrogate(seq_smooth)

        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_flat, val_flat),
            (seq_sa, val_sa),
            (seq_smooth, val_smooth),
        ]

        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. ---- Final polishing (longer runs) ---------------------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG (multi‑active) polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n_big,
                                                       deadline,
                                                       max_iters=5_000_000,
                                                       eta0=7.0,
                                                       tol=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n_big,
                                                       deadline,
                                                       max_iters=1_500_000,
                                                       eta0=5.0)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n_big,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.001,
                                                         T_start=7e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. ---- Optional final Gaussian smoothing (tiny extra boost) -------------
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_final = _gaussian_smooth(global_best_seq, sigma=0.5, passes=2)
        val_final = _objective_surrogate(seq_final)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final smooth] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. ---- Official evaluation via provided function ---------------------------
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
