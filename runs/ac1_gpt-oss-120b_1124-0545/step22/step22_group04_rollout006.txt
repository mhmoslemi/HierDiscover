def _adam_optimize(seq, beta, deadline, max_iters, lr=0.1):
    v = np.log(seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = seq.copy()
    best_val = _objective_surrogate(seq)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        obj, grad = _smooth_obj_and_grad(v, len(seq), beta)
        m = beta1 * m + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v2 / (1 - beta2**t)
        v = v - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Propose a high‑quality step‑function.

The optimiser works in three stages:
1.  Initialise from the previous best solution (or a simple exponential decay)
    and normalise to a probability vector (sum = 1).
2.  A long L‑BFGS phase minimises a log‑sum‑exp surrogate of the max‑convolution.
    The temperature (beta) is increased gradually; each stage performs as many
    iterations as time permits.
3.  An Adam‑based mirror‑descent stage works directly on the same surrogate
    with a very large beta (≈ 10⁶).  This is essentially a smoothed sub‑gradient
    descent on the true max‑convolution and is very aggressive at flattening the
    convolution peak.  The best sequence seen throughout the run is kept.

Optional cheap random‑restart refinements are performed if any time is left.
The final candidate is evaluated with the provided ``evaluate_sequence`` and
stored in the global variable ``height_sequence_1`` for possible continuation
runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple normalising utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)), where sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic initial guess (exponential decay)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed objective

        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)

    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c                     # stability shift

    # exp(beta*(c - max_c)) is safe: the largest term is 1, the rest are ≤1
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original code)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   beta: float,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam on the log‑variables (v = log a) for the smoothed surrogate.
    Returns the best sequence seen (as a probability vector) and its exact
    surrogate value.
    """
    n = len(start_seq)
    v = np.log(start_seq + 1e-30)   # log‑softmax variables
    m = np.zeros_like(v)            # first moment
    v2 = np.zeros_like(v)           # second moment
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v2 / (1.0 - beta2 ** t)

        # parameter update in log‑space
        v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to probability simplex
        a = _softmax(v)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a L‑BFGS smoothing phase with an extensive Adam
    fine‑tuning stage.  Random restarts are performed if time permits.
    The best vector found is stored in ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem dimension – a modest size gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise the sequence
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_seq = arr
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        # cheap exponential decay, sorted descending (better for the optimiser)
        init_seq = _exp_decay(n, rate=0.001)
        init_seq = np.sort(init_seq)[::-1]

    # Normalise to a probability vector (sum = 1)
    init_seq = np.maximum(init_seq, 0.0)
    init_seq = init_seq / init_seq.sum()

    cur_seq = init_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    print(f"[Init] surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smoothed surrogate (if SciPy is present)
    # --------------------------------------------------------------
    try:
        import scipy.optimize as _opt
    except Exception:                     # pragma: no cover
        _opt = None

    if _opt is not None:
        v = np.log(cur_seq + 1e-30)
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
            50.0, 100.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

        # roughly 20 % of the total budget for smoothing
        smoothing_deadline = deadline - 300.0   # keep ~5 min for Adam + restarts

        for beta in beta_schedule:
            if time.time() > smoothing_deadline:
                break
            max_iter_per_beta = 3500
            try:
                res = _opt.minimize(fun=_smooth_obj_and_grad,
                                    x0=v,
                                    args=(n, beta),
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': max_iter_per_beta,
                                             'disp': False,
                                             'ftol': 1e-12})
                v = res.x
            except Exception:   # pragma: no cover
                # If something goes wrong we simply break out of L‑BFGS.
                break

        cur_seq = _softmax(v)
        cur_val = _objective_surrogate(cur_seq)
        print(f"[After L‑BFGS] surrogate = {cur_val:.6f}")
    else:
        print("[Info] SciPy not available – skipping L‑BFGS phase")

    # --------------------------------------------------------------
    # Phase 2 – Adam fine‑tuning with a very large beta
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam fine‑tuning (high beta) ---")
        # use a cascade of increasing betas; the last one is extremely high
        betas_fine = [50.0, 200.0, 1_000.0, 5_000.0,
                      20_000.0, 100_000.0, 500_000.0, 1_000_000.0]

        # allocate roughly the remaining time (minus a safety margin)
        remaining = deadline - time.time() - 30.0   # keep ~30 s for restarts & final eval
        per_beta_time = max(5.0, remaining / len(betas_fine))

        for beta in betas_fine:
            if time.time() >= deadline - 30.0:
                break
            sub_deadline = time.time() + per_beta_time
            # the Adam loop respects the sub‑deadline internally
            cur_seq, cur_val = _adam_optimize(cur_seq,
                                              beta=beta,
                                              deadline=sub_deadline,
                                              max_iters=2_000_000,
                                              lr=0.12)
            print(f"  [beta={beta:.0e}] surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap random‑restart polishes (if time permits)
    # --------------------------------------------------------------
    while time.time() + 25.0 < deadline:
        # Create a noisy copy of the current best
        noisy = cur_seq + 0.003 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        # Run a short Adam sweep with a medium beta
        sub_deadline = time.time() + 22.0
        seq_rr, val_rr = _adam_optimize(noisy,
                                        beta=100_000.0,
                                        deadline=sub_deadline,
                                        max_iters=500_000,
                                        lr=0.10)
        if val_rr < cur_val:
            cur_seq, cur_val = seq_rr, val_rr
            print(f"[Random restart] improved surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # Final tiny permutation‑based hill‑climb (very fast)
    # --------------------------------------------------------------
    if time.time() + 8.0 < deadline:
        # one more round of simple swaps – only 30 k attempts
        max_iters = 30_000
        cur_seq_opt = cur_seq.copy()
        cur_val_opt = cur_val
        it = 0
        T_start = 1e-3 * cur_val_opt
        T_end = 1e-6 * cur_val_opt
        while time.time() < deadline and it < max_iters:
            it += 1
            T = T_start + (T_end - T_start) * (it / max_iters)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq_opt.copy()
            ns[i], ns[j] = ns[j], ns[i]
            val = _objective_surrogate(ns)
            if val < cur_val_opt or (T > 0 and np.exp(-(val - cur_val_opt) / T) > rng.random()):
                cur_seq_opt = ns
                cur_val_opt = val
        if cur_val_opt < cur_val:
            cur_seq, cur_val = cur_seq_opt, cur_val_opt
            print(f"[Permutation hill‑climb] improved surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for continuation runs
    globals()["height_sequence_1"] = cur_seq.tolist()
    return cur_seq.tolist()
