def _smooth_eg_opt(start_a, deadline, beta_schedule, eta0=0.5, max_iter_per_beta=2000):
    a = start_a.copy()
    best = a.copy()
    best_val = _surrogate_obj(a)
    it = 0
    for beta in beta_schedule:
        for k in range(max_iter_per_beta):
            if time.time()>deadline:
                return best, best_val
            conv = _conv_fft(a, a)
            # compute softmax w
            maxc = np.max(conv)  # for stability
            exps = np.exp(beta * (conv - maxc))
            w = exps / exps.sum()
            # gradient
            a_rev = a[::-1]
            grad_full = _conv_fft(w, a_rev)  # size 3n-1
            # slice out n elements
            grad = 2 * grad_full[len(a)-1 : len(a)-1 + len(a)]
            # eg update
            it += 1
            eta = eta0 / math.sqrt(it)

"""
Propose a step‑function height vector that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The optimiser works in three
phases under the 1000 s budget:

1. **Smooth exponentiated‑gradient descent** – a deterministic annealing
   on a soft‑max (log‑sum‑exp) surrogate of the max‑convolution.  The
   inverse temperature β is raised geometrically while a multiplicative
   (mirror‑descent) update drives the mass away from entries that
   contribute most to the surrogate.

2. **Exact max‑convolution EG** – a short mirror‑descent run that uses
   the true max‑convolution (single‑active shift) as a sub‑gradient.
   This refines the solution obtained in (1).

3. **Simulated‑annealing (SA) fine‑tuning** – tiny probability‑mass
   transfers between random coordinates, accepting worsening moves
   according to an exponential cooling schedule.

The best vector found is stored in the global name ``height_sequence_1``
and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase‑1: smooth exponentiated‑gradient descent (annealed β)
# ----------------------------------------------------------------------
def _smooth_exp_gradient(start_seq: np.ndarray,
                         deadline: float,
                         beta_start: float = 0.5,
                         beta_end: float = 500.0,
                         beta_steps: int = 12,
                         iters_per_beta: int = 1500,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Deterministic annealing on the log‑sum‑exp surrogate of max‑convolution.
    Uses a multiplicative (exponentiated‑gradient) update while keeping the
    vector on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    it = 0
    betas = np.geomspace(beta_start, beta_end, num=beta_steps)

    for beta in betas:
        # early exit if we ran out of time
        if time.time() > deadline:
            break

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            # 1) exact self‑convolution
            conv = _conv_fft(a, a)

            # 2) softmax weights w_i ∝ exp(β·conv_i)
            maxc = np.max(conv)                     # stability shift
            exps = np.exp(beta * (conv - maxc))
            w = exps / exps.sum()                    # shape: (2n‑1,)

            # 3) gradient of the smoothed max (∇t = Σ w_i ∇conv_i)
            #    ∇conv_i = 2·a shifted ⇒ use convolution with the reversed a
            rev = a[::-1]                            # length n
            grad_full = _conv_fft(w, rev)            # length 3n‑2
            mid = len(a) - 1
            grad = 2.0 * grad_full[mid:mid + len(a)]  # length n, non‑negative

            # 4) mirror‑descent (exponentiated‑gradient) update
            it += 1
            eta = eta0 / math.sqrt(it)               # decaying step size
            a = a * np.exp(-eta * grad)               # element‑wise
            s = a.sum()
            if s == 0.0 or np.isnan(s):
                a = np.full_like(a, 1.0 / len(a))
            else:
                a /= s                                 # renormalise to ℓ₁=1

            # keep the best solution seen so far
            val = _surrogate_obj(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase‑2: exponentiated‑gradient on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Uses only the single active (worst) shift for the sub‑gradient.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                     # safe log

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())                     # worst shift

        # sub‑gradient of max‑convolution (ignoring constant 2·n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = a[k - idx]                        # vectorised

        eta = eta0 / math.sqrt(it)

        log_a -= eta * g
        a = _softmax(log_a)                        # stays on the simplex

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase‑3: simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Simple SA: transfer a tiny amount of probability mass between two
    random coordinates.  Temperature and perturbation size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        amt = sigma * rng.random() * a[i]          # a tiny fraction of a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point – respects the 1000 s overall budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises ``evaluate_sequence``.
    Uses three optimisation phases (smooth EG, exact EG, SA) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Initialise – use a previously stored solution if available,
    # otherwise start from the uniform distribution.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            s = arr.sum()
            if s <= 0.0:
                raise ValueError
            seq0 = arr / s                     # ℓ₁‑normalise
            n = seq0.shape[0]
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    print(f"\n=== Optimisation started – n = {n} ===")
    print(f"Initial surrogate value: {_surrogate_obj(seq0):.6f}")

    # --------------------------------------------------------------
    # Phase‑1: smooth exponentiated‑gradient (annealed β)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.55 * remaining)
    print("\n--- Phase 1: Smooth EG on log‑sum‑exp surrogate ---")
    seq_smooth, val_smooth = _smooth_exp_gradient(
        start_seq=seq0,
        deadline=phase1_deadline,
        beta_start=0.5,
        beta_end=800.0,
        beta_steps=14,
        iters_per_beta=1200,
        eta0=0.6,
    )
    print(f"[Smooth EG] surrogate = {val_smooth:.6f}")

    # --------------------------------------------------------------
    # Phase‑2: exact max‑convolution EG (short polishing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase2_deadline = time.time() + max(5.0, 0.30 * remaining)
        print("\n--- Phase 2: Exact‑max EG (short) ---")
        seq_exact, val_exact = _exp_gradient_opt_long(
            start_seq=seq_smooth,
            deadline=phase2_deadline,
            eta0=4.0,
            max_iters=2_000_000,
        )
        print(f"[Exact EG] surrogate = {val_exact:.6f}")
    else:
        seq_exact, val_exact = seq_smooth, val_smooth

    # --------------------------------------------------------------
    # Phase‑3: simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase3_deadline = time.time() + remaining
        print("\n--- Phase 3: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(
            start_seq=seq_exact,
            deadline=phase3_deadline,
            rng=rng,
            max_attempts=500_000,
            T_start=0.02,
            T_end=1e-8,
            sigma_start=0.003,
        )
        print(f"[SA] surrogate = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_exact, val_exact

    # --------------------------------------------------------------
    # Choose the best sequence among all phases
    # --------------------------------------------------------------
    best_seq, best_val = seq_smooth, val_smooth
    if val_exact < best_val:
        best_seq, best_val = seq_exact, val_exact
    if val_sa < best_val:
        best_seq, best_val = seq_sa, val_sa

    print(f"\n=== Final surrogate after all phases = {best_val:.6f} ===")

    # --------------------------------------------------------------
    # True evaluation (may differ slightly due to rounding)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
