"""
Propose a candidate step‑function for the leaderboard.

The search consists of three phases

1. **Diverse initialisation** – a few deterministic vectors (exponential decay)
   and a few random Dirichlet draws are built and normalised to a probability
   vector (sum = 1).

2. **Exponentiated‑gradient (mirror‑descent) polishing** – each start is
   refined with a fast stochastic mirror‑descent on the exact objective
   (the active entry of the linear convolution).  Very short runs are used
   for the many starts; the best intermediate solution is then polished for
   the remaining time with a longer run.

3. **Optional local hill‑climb** – a cheap permutation / perturbation hill
   climber finishes the search.

All steps respect the 1000 s time budget, keep the sequence non‑negative,
and finally store the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Simple analytic families (used for cheap initial guesses)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate that the optimiser uses:
       2·n·max(conv(seq,seq))   (sum(seq) is 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the true max‑convolution
    objective.  Works in log‑space to stay on the simplex.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space variables for the softmax parametrisation
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry (see the analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation / perturbation hill climbing
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two positions or adds a tiny Gaussian perturbation.
    A simple annealing schedule (temperature decays linearly) permits occasional
    uphill moves.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # linear annealing schedule
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:               # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                 # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Normalisation utility (keeps the vector on the simplex)
# ----------------------------------------------------------------------
def _normalise(arr: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and rescale to sum = 1."""
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0.0:
        return np.full_like(arr, 1.0 / len(arr))
    return arr / s


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine creates several diverse initial points, polish each with a
    fast exponentiated‑gradient optimiser, keeps the best intermediate result,
    and finally performs a longer polish plus an optional hill‑climb.
    The best sequence is stored in the global ``height_sequence_1`` variable.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a short safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    n = 4000  # problem dimension
    # ------------------------------------------------------------------
    # collect a few starting vectors
    starts = []

    # 1) reuse previous best if it exists and has the right size
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                starts.append(_normalise(arr))
        except Exception:
            pass

    # 2) deterministic exponential decays
    starts.append(_exp_decay(n, rate=0.001))
    starts.append(_exp_decay(n, rate=0.003))
    starts.append(_exp_decay(n, rate=0.0005))

    # 3) a few random Dirichlet‑like draws (different concentration)
    for _ in range(3):
        alpha = rng.uniform(0.02, 2.0)          # concentration parameter
        # Gamma sampling produces Dirichlet components
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        starts.append(_normalise(g))

    # ------------------------------------------------------------------
    # Phase 1 – short EG polishing of each start (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    # allocate about half the budget to the exploratory runs
    explore_deadline = deadline - 250.0  # keep ~4 min for the final polish

    for init in starts:
        if time.time() > explore_deadline:
            break
        # each exploratory run gets at most ~30 s or the remaining time
        sub_deadline = min(time.time() + 30.0, explore_deadline)
        seq, val = _exp_gradient_opt(init, n, sub_deadline,
                                     max_iters=800_000, eta0=8.0)
        if val < best_val:
            best_val, best_seq = val, seq

    # Fallback – should never happen
    if best_seq is None:
        best_seq = _exp_decay(n, rate=0.001)
        best_seq = _normalise(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – long EG polishing of the current best (exploitation)
    # ------------------------------------------------------------------
    final_seq, final_val = _exp_gradient_opt(best_seq, n, deadline,
                                            max_iters=4_500_000,
                                            eta0=10.0)
    best_seq, best_val = final_seq, final_val

    # ------------------------------------------------------------------
    # Phase 3 – optional quick permutation hill‑climb
    # ------------------------------------------------------------------
    if time.time() + 15.0 < deadline:
        seq_hc, val_hc = _perm_hill_climb(best_seq.copy(),
                                          n,
                                          deadline,
                                          rng,
                                          max_iters=80_000)
        if val_hc < best_val:
            best_seq, best_val = seq_hc, val_hc

    # ------------------------------------------------------------------
    # Final true evaluation (the value that the leaderboard will see)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
