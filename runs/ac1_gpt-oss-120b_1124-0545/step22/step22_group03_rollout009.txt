    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a,a)
        k = int(conv.argmax())

        i_end = min(n-1, k)

        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:

"""
Optimise a non‑negative step‑function (height vector) that minimises

    2·n·max(conv(a, a)) / (sum(a) ** 2)

The search is split into three time‑budgeted phases:

1. **Adam‑style exponentiated‑gradient descent** – works on the exact
   worst shift of the auto‑convolution.  The optimisation is performed
   on the log‑probabilities (mirror‑descent on the simplex) with a
   classic Adam learning‑rate schedule (β₁=0.9, β₂=0.999).  This stage
   is very cheap (one FFT per iteration) and can run for many millions
   of iterations.

2. **Multi‑active mirror‑descent** – after the first phase the maximal
   convolution value is still slightly above the optimum.  All shifts
   whose convolution value lies within a small relative tolerance of the
   maximum are collected (typically a handful).  The gradient is the
   sum of the “single‑active’’ gradients for those shifts, i.e. each
   active shift contributes a slice of the current vector.  Updating
   with this richer sub‑gradient flattens the convolution peak further.

3. **Simulated‑annealing fine‑tuning** – tiny amounts of probability mass
   are transferred between random coordinates.  A linear cooling schedule
   (temperature and perturbation size) is used and improvements are
   always accepted; worsening moves are accepted with the usual Boltzmann
   probability.  This phase cleans up any remaining local irregularities.

Two different starts are tried (the previous best solution if it exists,
a uniform distribution, and a sparse Dirichlet draw).  The best of the
runs is stored in the global variable ``height_sequence_1`` and returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser (single active shift)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              eta: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities with the sub‑gradient of the single
    worst convolution shift.
    Returns the best (ℓ₁‑normalised) sequence encountered and its
    surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # initialise log‑weights safely
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution and worst shift
        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of max‑convolution (ignoring the factor 2)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = a[k - idx]

        # Adam updates on the log‑space vector
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent with *multiple* active shifts (uniform weight)
# ----------------------------------------------------------------------
def _exp_gradient_multi(start_seq: np.ndarray,
                       deadline: float,
                       eta0: float = 5.0,
                       epsilon_frac: float = 0.001,
                       max_active: int = 30,
                       max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using all convolution
    shifts whose value lies within ``epsilon_frac`` of the current
    maximum.  The gradient for each active shift is added together.
    Returns the best normalised sequence encountered and its surrogate.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        thresh = max_val - epsilon_frac * max_val

        active = np.where(conv >= thresh)[0]

        # If there are many active shifts, keep only the top ``max_active``
        if active.size > max_active:
            # select indices with largest conv values
            top_inds = np.argpartition(conv, -max_active)[-max_active:]
            active = top_inds

        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            # contribution a[k-idx] for all i in idx
            g[idx] += a[k - idx]

        # decaying learning‑rate schedule
        eta = eta0 / math.sqrt(it)

        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.05,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Transfer a tiny amount of probability mass between two random
    coordinates.  Temperature and perturbation size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)  # shrink perturbation together with T

        # choose a donor with non‑negligible mass
        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # move a tiny fraction of the donor's mass
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the overall 1000 s budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function tries a few different initialisations, runs an
    Adam‑style EG phase, a multi‑active EG phase and finally a short
    simulated‑annealing polish.  The best sequence found is stored in the
    global variable ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # leave a small safety margin for the final evaluation
    deadline = t0 + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise – try to reuse a previously discovered solution.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    init_sequences = []

    # initialise length (the previous runs used 4000)
    n = 4000

    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = arr / arr.sum()
                if arr.shape[0] == n:
                    init_sequences.append(arr)
        except Exception:
            pass

    # uniform start (always present)
    init_sequences.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a sparse Dirichlet initialisation – often helps escaping local minima
    init_sequences.append(rng.dirichlet(np.full(n, 0.3)))   # low concentration → spikier start

    best_seq = None
    best_surrogate = float("inf")

    # --------------------------------------------------------------
    # Run a short optimisation for each initialisation, keeping the
    # overall best vector.
    # --------------------------------------------------------------
    for ix, seq0 in enumerate(init_sequences):
        if time.time() >= deadline:
            break

        print(f"\n=== Restart {ix} – starting optimisation ===")
        # ----------------------------------------------------------
        # Phase 1 – Adam optimiser (single worst shift)
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        phase_deadline = time.time() + max(5.0, remaining * 0.55)   # ~55 % of the remaining budget
        print(f"Phase 1 (Adam) – time budget ≈ {phase_deadline - time.time():.1f}s")
        seq_adam, val_adam = _adam_opt(seq0, phase_deadline, eta=0.05)
        print(f"[Phase 1] surrogate = {val_adam:.6f}")

        # ----------------------------------------------------------
        # Phase 2 – multi‑active exponentiated‑gradient
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            phase_deadline = time.time() + max(5.0, remaining * 0.30)   # ~30 %
            print(f"Phase 2 (multi‑active EG) – time budget ≈ {phase_deadline - time.time():.1f}s")
            seq_multi, val_multi = _exp_gradient_multi(seq_adam, phase_deadline,
                                                       eta0=5.0,
                                                       epsilon_frac=0.001,
                                                       max_active=30,
                                                       max_iters=500_000)
            print(f"[Phase 2] surrogate = {val_multi:.6f}")
        else:
            seq_multi, val_multi = seq_adam, val_adam

        # ----------------------------------------------------------
        # Phase 3 – simulated‑annealing fine‑tuning
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            print(f"Phase 3 (SA) – time budget ≈ {remaining:.1f}s")
            seq_sa, val_sa = _simulated_annealing_refine(seq_multi,
                                                         time.time() + remaining,
                                                         rng,
                                                         max_attempts=500_000,
                                                         T_start=0.05,
                                                         T_end=1e-8,
                                                         sigma_start=0.005)
            print(f"[Phase 3] surrogate = {val_sa:.6f}")
        else:
            seq_sa, val_sa = seq_multi, val_multi

        # ----------------------------------------------------------
        # Keep the best of this restart
        # ----------------------------------------------------------
        if val_sa < best_surrogate:
            best_surrogate = val_sa
            best_seq = seq_sa
            print(f"*** New overall best surrogate = {best_surrogate:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the original ``evaluate_sequence`` (may
    # differ slightly from the surrogate because of rounding).
    # --------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No sequence was produced within the time budget.")

    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n>>> Best true evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
