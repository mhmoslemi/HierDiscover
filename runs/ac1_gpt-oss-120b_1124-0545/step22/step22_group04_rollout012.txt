"""
Improved search for a step‑function with a low evaluation score.

The optimisation problem is:
    minimise 2·n·max(conv(a,a)) / (sum(a)²)   with   a_i ≥ 0.
Because the objective is scale‑invariant we can keep the coefficients on the
probability simplex (sum = 1).  The central convolution entry (index n‑1) is
the inner product of the vector with its reverse; reducing this product while
keeping every single coefficient small lowers the overall max‑convolution.

Algorithm
---------
1.  Initialisation – start from the previously stored best sequence if it
    exists, otherwise from a simple exponential decay and normalise to sum=1.

2.  Adam optimisation on a *log‑softmax* parametrisation.
    The smooth surrogate
        2·n·(max(c) + log Σ exp(β·(c-max(c)))/β)
    (with c = a*a) is minimised by Adam for a schedule of β growing from
    0.5 → 1e6.  The parametrisation guarantees non‑negativity and the simplex
    constraint automatically.

3.  Projected sub‑gradient descent (the “exponentiated‑gradient” step of the
    previous solution) is run as a polishing phase.  It works directly on the
    exact max‑convolution and keeps the vector on the simplex via a softmax
    update.

4.  An intensive simulated‑annealing hill‑climb (swap moves and tiny Gaussian
    perturbations) explores the neighbourhood of the best vector found so far.
    The schedule accepts worsening moves with a decreasing temperature.

5.  If time permits a few short random‑restart Adam runs are performed; any
    improvement replaces the current best.

6.  Finally the true objective is obtained with ``evaluate_sequence``,
    the result is printed and the vector is stored in the global variable
    ``height_sequence_1`` for possible continuation runs.

The whole routine respects the 1000 s time budget; the final sequence is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv(seq,seq)).  (Sum is fixed to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – needed for Adam optimisation
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective
    using log‑sum‑exp smoothing.
    """
    a = _softmax(v_local)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights over the convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule:   ∂obj/∂v = a * (grad_a - (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best sequence seen (as a probability vector) and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for the smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update in the log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Works entirely on the simplex via a softmax‑style log‑space update.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space representation for the simplex‐projection
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (swap / perturb moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or adds a tiny Gaussian
    perturbation.  Worsening moves are accepted with a simple temperature schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # tiny Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # accept with probability exp(-(val‑cur_val)/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # problem size (the leaderboard uses n = 4000)
    n = 4000

    # ------------------------------------------------------------------
    # 1. Initialise – reuse previously found sequence if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_seq = np.maximum(arr, 0.0)
                init_seq = init_seq / init_seq.sum()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        # fall back to a simple exponential decay, sorted descending
        init_seq = _exp_decay(n, rate=0.001)
        init_seq = np.sort(init_seq)[::-1]   # descending order

    # ------------------------------------------------------------------
    # 2. Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    print("[Stage 1] Adam on smoothed objective")
    seq_adam, val_adam = _adam_opt(
        start_seq=init_seq,
        n=n,
        deadline=deadline,
        max_iters=200_000,
        beta_start=0.5,
        beta_end=1_000_000.0,
        alpha=0.5,
    )
    print(f"   after Adam   surrogate = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient polishing (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 2] Exact EG polishing")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            max_iters=5_000_000,
            eta0=10.0,
        )
        print(f"   after EG    surrogate = {val_eg:.6f}")
        if val_eg < val_adam:
            seq_adam, val_adam = seq_eg, val_eg

    # ------------------------------------------------------------------
    # 4. Hill‑climb with simulated‑annealing moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 3] Hill‑climb (swap / perturb)")
        seq_hc, val_hc = _hill_climb(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=600_000,
        )
        print(f"   after HC    surrogate = {val_hc:.6f}")
        if val_hc < val_adam:
            seq_adam, val_adam = seq_hc, val_hc

    # ------------------------------------------------------------------
    # 5. Short random‑restart Adam refinements (if time remains)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 4] Random‑restart Adam")
        noisy = seq_adam + 0.003 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=80_000,
            beta_start=0.5,
            beta_end=200_000.0,
            alpha=0.5,
        )
        print(f"   restart     surrogate = {val_rr:.6f}")
        if val_rr < val_adam:
            seq_adam, val_adam = seq_rr, val_rr
            print("   *** improvement found! ***")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_adam.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_adam:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq_adam.tolist()
    return seq_adam.tolist()
