Optimiser for the step‑function upper bound (target < 1.505).

Key ideas
---------

*   Keep the successful three‑stage pipeline of the reference code but
    *extend the seed families* (very sharp beta‑shapes, very fast
    exponentials, double Gaussian bumps) to explore more “U‑shaped”
    distributions that put mass near the ends of the interval.
*   Add a *single‑max* exponentiated‑gradient optimiser – instead of
    averaging the sub‑gradient over all active (near‑max) indices we
    descend using the exact gradient of the currently maximal convolution
    entry.  This yields a stronger descent direction for the true
    objective.
*   Run the classic full‑active EG optimiser as well and keep the best of
    the two.
*   Increase the time budget for the final polishing phase (more
    iterations, finer learning‑rate schedule) while still respecting the
    1000 s global limit.
*   Finally evaluate the best sequence with ``evaluate_sequence`` and store
    it in the global variable ``height_sequence_1`` for continuation across
    runs.

The routine is fully self‑contained, uses only NumPy (SciPy is optional)
and respects the time budget.

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (quadratic O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s != 0.0 else np.full_like(v, 1.0 / len(v))


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors:
       2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    a /= a.sum()
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a /= a.sum()
    return a


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    mixed /= mixed.sum()
    return mixed


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _double_bump(n: int, pos1: float, pos2: float, sigma_frac: float) -> np.ndarray:
    """Two narrow Gaussian bumps placed at fractions pos1, pos2 of the interval."""
    i = np.arange(n, dtype=np.float64)
    sigma = sigma_frac * n
    mu1 = pos1 * (n - 1)
    mu2 = pos2 * (n - 1)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    a /= a.sum()
    return a

# Sub‑gradient utilities
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient over all active (near‑max) indices."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    count = active.sum()
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


def _subgradient_max_conv_single(a: np.ndarray) -> np.ndarray:
    """Exact gradient of the *single* maximal convolution entry."""
    n = len(a)
    conv = _conv_fft(a, a)
    idx = int(np.argmax(conv))
    grad = np.zeros_like(a)
    # valid indices j satisfy 0 ≤ idx‑j < n
    js = np.where((idx - np.arange(n) >= 0) & (idx - np.arange(n) < n))[0]
    grad[js] = 2.0 * a[idx - js]
    return grad

def _exp_gradient_opt_full(start_seq: np.ndarray,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """EG using the *averaged* sub‑gradient (as in the reference code)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)

def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=len(a))
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects a hard time budget and
    stores the best found sequence in the global variable
    ``height_sequence_1`` for future runs.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size – reuse previous solution if available.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) > 0:
        n = len(prev)
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.sum() > 0.0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        # default size – a little smaller than the reference (speed vs quality)
        n = 3000
        prev_arr = None
        print(f"\n[Info] No previous solution – using n={n}")

    # ------------------------------------------------------------------
    # 1️⃣  Broad seeding – many analytic families + random draws
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for g in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_sine_window(n, gamma=g))
    for sf in (0.08, 0.12, 0.18, 0.25):
        seeds.append(_gaussian(n, sigma_frac=sf))

    # very sharp beta shapes (U‑shaped)
    for a in np.logspace(-2, 1, 10):
        for b in np.logspace(-2, 1, 10):
            seeds.append(_beta_shape(n, float(a), float(b)))

    # exponential decays – explore extreme rates
    for r in np.logspace(-5, 0.5, 12):
        seeds.append(_exp_decay_left(n, float(r)))
        seeds.append(_exp_decay_right(n, float(r)))

    # power‑law families
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power(n, float(p)))

    # double‑bump shapes – two narrow Gaussians near the ends
    for sigma_f in (0.02, 0.04, 0.06):
        seeds.append(_double_bump(n, 0.10, 0.90, sigma_f))
        seeds.append(_double_bump(n, 0.05, 0.95, sigma_f))

    # random Dirichlet draws (very sparse and very dense regimes)
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2, 1.0, 5.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best (if any)
    if prev_arr is not None:
        base = prev_arr.copy()
        seeds.append(base)
        for _ in range(10):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2️⃣ Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] Best surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Multi‑start local refinement on the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of remaining time for this seed
        remaining = deadline - time.time()
        per_seed = max(4.0,
                       remaining * 0.12 / max(1, (top_k - si)))
        sub_deadline = time.time() + per_seed

        # ----- (a) exponentiated‑gradient – full‑active ----- 
        seq_eg_full, val_eg_full = _exp_gradient_opt_full(
            seed_vec.copy(), sub_deadline, max_iters=350_000, eta0=7.0)

        # ----- (b) exponentiated‑gradient – single‑max ----- 
        if time.time() < sub_deadline:
            seq_eg_single, val_eg_single = _exp_gradient_opt_single(
                seed_vec.copy(), sub_deadline, max_iters=350_000, eta0=7.0)
        else:
            seq_eg_single, val_eg_single = seq_eg_full, val_eg_full

        # ----- (c) L‑BFGS smoothing ----- 
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(
                seq_eg_single, sub_deadline,
                beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0,
                               200.0, 500.0, 1_000.0, 2_000.0, 5_000.0],
                max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg_single, val_eg_single

        # ----- (d) simulated annealing ----- 
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(
                seq_lbfgs, hill_deadline, rng,
                max_attempts=200_000, start_sigma=0.0025,
                T_start=0.012, T_end=5e-9)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # Choose the best of the four refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg_full, val_eg_full),
            (seq_eg_single, val_eg_single),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final intensive polishing (more iterations, both EG variants)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing: EG (full‑active) ---")
        seq_full, val_full = _exp_gradient_opt_full(
            global_best_seq.copy(),
            deadline,
            max_iters=4_000_000,
            eta0=5.0)
        if val_full < global_best_val:
            global_best_seq, global_best_val = seq_full, val_full
            print(f"[Polish EG full] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing: EG (single‑max) ---")
        seq_single, val_single = _exp_gradient_opt_single(
            global_best_seq.copy(),
            deadline,
            max_iters=3_000_000,
            eta0=5.0)
        if val_single < global_best_val:
            global_best_seq, global_best_val = seq_single, val_single
            print(f"[Polish EG single] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final polishing: Simulated annealing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq,
            deadline,
            rng,
            max_attempts=400_000,
            start_sigma=8e-4,
            T_start=4e-3,
            T_end=1e-9)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[Polish SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
