"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm consists of four phases
(remaining time is split dynamically):

0️⃣  **Warm‑start & simple seeds** – uniform, arcsine, triangular,
    Dirichlet draws and a previously stored solution (if any).

1️⃣  **Parametric exploration** – generate many candidate shapes from
    two low‑dimensional families (Beta densities and a generalized
    arcsine power law).  Each candidate is evaluated with the exact
    ``2·n·max(conv)`` surrogate (computed by FFT).  The best candidate
    becomes the seed for the next phase.

2️⃣  **Smooth optimisation** – Adam on the log‑softmax (mirror) space
    of a log‑sum‑exp approximation of the max‑convolution.  The β
    schedule is richer and each β receives more inner iterations than
    in the reference implementation.

3️⃣  **Exact sub‑gradient polishing** – exponentiated‑gradient (mirror
    descent) on the true max‑convolution using a decreasing step‑size.

4️⃣  **Final stochastic walk** – cheap random “move ε from i to j”
    steps on the simplex to squeeze any remaining improvement.

The best sequence found (as a list of floats) is stored in the global
``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used throughout the search:
        2·n·max(conv(seq, seq))   (sum(seq) is assumed = 1)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density (β(½,½))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on the grid i+½.
    α,β > 0.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against overflow/underflow
    eps = 1e-14
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    # Remove any NaNs that can appear for extreme parameters
    a = np.where(np.isfinite(a), a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0)
    return a / a.sum()


def _power_arcsine(n: int, p: float) -> np.ndarray:
    """
    Generalised arcsine shape:
        a_i ∝ 1 / ((i+½)·(n‑½‑i))^p .
    p > 0; p = 0.5 reproduces the ordinary arcsine.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / ((i + 0.5) ** p * (n - 0.5 - i) ** p + eps)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp
    surrogate of 2·n·max(conv(a,a)).  Returns the best probability
    vector seen and its exact surrogate value.
    """
    # initialise logits from the start probabilities
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a richer β‑schedule (more gradual sharpening)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        2e4, 5e4,
        1e5, 2e5, 5e5,
        1e6, 2e6, 5e6,
        1e7, 2e7, 5e7,
        1e8,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # give each β a generous budget (≈8000 iterations)
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)        # extra factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12                     # a slightly more conservative step‑size
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation (every 1000 steps)
            if it % 1000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the active index.  Returns the best
    probability vector seen and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 120_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a tiny fraction of mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence found within the time
    budget is stored in the global variable ``height_sequence_1`` and
    returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin so we never overrun the limit
    deadline = start_time + budget_s - 1.0

    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # --------------------------------------------------------------
    # 0️⃣  Build a diverse initial pool (all vectors sum to 1)
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 0️⃣  Evaluate seeds, keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            continue
        vec = vec / vec_sum
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Parametric random search (Beta & power‑arcsine families)
    # --------------------------------------------------------------
    if time.time() < deadline:
        param_deadline = time.time() + 0.30 * (deadline - time.time())
        print("\n--- Parametric random search (Beta / power‑arcsine) ---")
        while time.time() < param_deadline:
            if rng.random() < 0.55:
                # Beta family – sample log‑uniform α,β
                log_alpha = rng.uniform(np.log(0.05), np.log(10.0))
                log_beta = rng.uniform(np.log(0.05), np.log(10.0))
                alpha = math.exp(log_alpha)
                beta = math.exp(log_beta)
                cand = _beta_distribution(n, alpha, beta)
                fam = "Beta"
            else:
                # Power‑arcsine family
                p = rng.uniform(0.15, 2.5)      # wide but reasonable range
                cand = _power_arcsine(n, p)
                fam = "Arc"
            # normalise and evaluate
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            val = _objective_surrogate(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"  [+] New best ({fam}) surrogate = {val:.6f}")
        print(f"[Param] best surrogate after random search = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam smoothing (smooth max) on the current best seed
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth max) ---")
        remain = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remain          # reserve a bit for later phases
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 3️⃣  Exact exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 4.0      # leave a tiny window for final walk
        remaining = polishing_deadline - time.time()
        # allocate iterations proportionally to the remaining fraction
        max_iters = int(2_500_000 * remaining / (deadline - start_time))
        max_iters = max(300_000, min(max_iters, 5_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=6.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        # allocate as many attempts as time permits (≈10 μs per try)
        time_left = deadline - time.time()
        # rough attempts estimate: 1e5 attempts ≈ 2 s for n=4000
        attempts = int(120_000 + 200_000 * time_left / (budget_s * 0.1))
        attempts = max(60_000, attempts)
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=attempts,
                                                sigma=0.018)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
