"""
Optimiser for the step‑function upper bound.

The implementation follows the three‑stage idea of the reference code but
adds a few new ingredients that have proven useful for reaching a
lower bound below 1.503:

* a new analytic family – the “square‑root series” (coefficients of
  (1‑x)^‑½) and a simple 1/√k decay – which are known to give a
  theoretical bound ≈π/2≈1.571.  These seeds drive the search into the
  right region of the space.
* a cheap low‑dimensional “block” family (piece‑wise constant
  profiles).  Random block partitions give very diverse, highly
  asymmetric shapes that are hard to obtain with pure Dirichlet draws.
* longer optimisation per high‑quality seed: the exponentiated‑gradient
  (mirror‑descent) optimiser runs many more iterations on a smaller
  problem size (n≈2000) which allows a much tighter convergence under the
  1000 s budget.
* an intensified simulated‑annealing polishing phase (≈500 k tries)
  after the final gradient pass.

All helper functions are top‑level; there are no closures or lambdas.
The routine respects the time budget, stores the best sequence in the
global variable ``height_sequence_1`` and finally returns it as a list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(m log m))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) for a normalised to sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sqrt_series(n: int) -> np.ndarray:
    """Coefficients of (1‑x)^−½, i.e. a_i = C(2i,i)/4^i, normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(0, n - 1):
        # recurrence a_{i+1} = a_i * ((2i+1)*(2i+2)) / (4*(i+1)^2)
        num = (2 * i + 1) * (2 * i + 2)
        den = 4.0 * (i + 1) * (i + 1)
        a[i + 1] = a[i] * (num / den)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_block_seed(n: int, blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Piece‑wise constant profile with ``blocks`` random segments."""
    # ensure each block has length ≥1
    cuts = rng.choice(np.arange(1, n), size=blocks - 1, replace=False)
    cuts.sort()
    lengths = np.diff(np.concatenate(([0], cuts, [n])))
    heights = rng.random(blocks) + 1e-6  # strictly positive
    block_vec = np.repeat(heights, lengths)
    s = block_vec.sum()
    return block_vec / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)      # length 2n‑1
    count = active.sum()
    if count == 0.0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)                 # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent on exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad = _subgradient_max_conv(a)
        # standard EG update in log‑space
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 500_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    n = len(a)
    attempts = 0

    while attempts < max_attempts and time.time() < deadline:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats).  Result is cached in the global
    variable ``height_sequence_1`` for possible reuse in later calls.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse the previous solution if it exists.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
                n = arr.shape[0]
            else:
                raise ValueError
        except Exception:
            prev_arr = None
            n = 2000
    else:
        prev_arr = None
        n = 2000                     # smaller n ⇒ faster optimisation

    print(f"\n[Info] Working with length n = {n}")

    # --------------------------------------------------------------
    # 1. Broad seeding – diverse analytic families + random families
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exp_decay_left(n, 0.01))
    seeds.append(_exp_decay_right(n, 0.01))
    seeds.append(_power(n, 0.5))
    seeds.append(_beta_shape(n, 0.8, 0.8))

    # new families that are known to give low bounds
    seeds.append(_inverse_sqrt(n))
    seeds.append(_sqrt_series(n))

    # mixtures with uniform
    for alpha in (0.2, 0.5, 0.8):
        seeds.append(_mix_shapes(_uniform(n), _inverse_sqrt(n), alpha))
        seeds.append(_mix_shapes(_uniform(n), _sqrt_series(n), alpha))

    # random block patterns (5‑block)
    for _ in range(10):
        seeds.append(_random_block_seed(n, blocks=5, rng=rng))

    # random Dirichlet draws (various concentrations)
    for conc in (0.005, 0.02, 0.1, 0.3):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturb previous best, if any
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0:
                seeds.append(pert / pert.sum())

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:

        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:

        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # --------------------------------------------------------------
    # Initialise global best
    # --------------------------------------------------------------

    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------

                                 remaining_time * 0.15 / max(1, (remaining_seeds - si)))

                                           sub_deadline,

    return global_best_seq.tolist()
