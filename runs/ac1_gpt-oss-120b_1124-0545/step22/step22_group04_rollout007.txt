"""
Optimisation of a non‑negative step‑function for the provided ``evaluate_sequence``.
---------------------------------------------------------------------------
The objective (up‑to an irrelevant scale) is

    2·n·max(conv(a,a)) / (sum(a)**2) .

Since the value is invariant under a global scaling of ``a`` we work with a
probability vector (sum = 1) and minimise the surrogate
    2·n·max(conv(a,a))

which is the L∞‑norm of the autocorrelation.  The optimiser consists of three
stages:

1. **Initialisation** – several sensible starting points (exponential decay,
   linear decay, uniform, the previous best if any) are generated and the three
   most promising (according to the cheap surrogate) are kept.

2. **Smooth L‑BFGS phase** – a log‑softmax parametrisation is optimised
   against a smooth Log‑Sum‑Exp approximation of the max.  The temperature
   ``beta`` is increased on a schedule, each step performing a modest number
   of L‑BFGS iterations.

3. **Exact ADAM polishing** – on the true non‑smooth objective we perform a
   mirror‑descent (exponentiated‑gradient) optimisation using the ADAM
   optimiser in the log‑space.  The method keeps the best sequence seen.
   After a long polishing we launch many short random‑restart refinements;
   each restart is re‑polished with ADAM and accepted if it improves the
   surrogate.

A final short permutation‑based hill‑climb is performed if time permits.
The best sequence is stored in the global variable ``height_sequence_1``
so that subsequent runs can continue from it.

The whole search respects a user‑provided time budget (default 1000 s) and
returns the best sequence discovered before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase
# ----------------------------------------------------------------------
try:                               # pragma: no cover
    import scipy.optimize as _opt
except Exception:                 # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, fully linear)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the quantity we actually optimise)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for cheap initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _linear_decay(n: int) -> np.ndarray:
    """a_i ∝ (n‑i), i = 0..n‑1, normalised to sum = 1."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) – objective & gradient for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad) of the smoothed objective
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # log‑sum‑exp part (numerically stable)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original notebook)
    w = exp_beta / sum_exp                     # weights that sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used by ADAM)
# ----------------------------------------------------------------------
def _active_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of 2·n·max(conv(a,a)) w.r.t. a.
    The active index is the one attaining the maximum.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    max_idx = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = max_idx - idx_i
    g[idx_i] = 2.0 * a[rev_idx]          # each contributing pair appears twice
    return g, max_idx


# ----------------------------------------------------------------------
# ADAM optimiser on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              max_iters: int = 5_000_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with ADAM updates in log‑space.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    n = len(a)

    # work in log‑softmax variables (stays on the simplex)
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        # sub‑gradient of the active convolution entry
        g, _ = _active_subgradient(a)

        # ADAM accumulation
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size that decays like 1/√t (helps stability later on)
        lr = lr0 / math.sqrt(t)

        # ADAM update in log‑space
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to the simplex
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight permutation hill‑climb (used at the very end)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two positions or applies a tiny Gaussian perturbation.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence found.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:      # swap move – the most common
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                         # tiny Gaussian jitter
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the optimiser used by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a short L‑BFGS smoothing phase with a long ADAM
    polishing phase and a large number of tiny random‑restart refinements.
    It works with a configurable dimension (default 6000) and stores the
    best sequence in the global variable ``height_sequence_1`` for later runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final evaluation / book‑keeping
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem dimension – a larger n gives a richer search space
    # --------------------------------------------------------------
    n = 6000

    # --------------------------------------------------------------
    # Initialise a pool of promising starting points
    # --------------------------------------------------------------
    init_pool = []

    # 1) use previous best if it exists (upsample / interpolate to current n)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape[0] == n:
            init_pool.append(prev_arr.copy() / prev_arr.sum())
        else:
            # simple linear interpolation to the new size
            old_x = np.linspace(0, 1, prev_arr.shape[0])
            new_x = np.linspace(0, 1, n)
            up = np.interp(new_x, old_x, prev_arr)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                init_pool.append(up / up.sum())

    # 2) a few analytic families
    init_pool.append(_exp_decay(n, rate=0.0005))
    init_pool.append(_exp_decay(n, rate=0.001))
    init_pool.append(_exp_decay(n, rate=0.002))
    init_pool.append(_linear_decay(n))
    init_pool.append(np.full(n, 1.0 / n))           # uniform
    init_pool.append(rng.dirichlet(np.ones(n)))    # random Dirichlet

    # --------------------------------------------------------------
    # Keep the three cheapest (according to the cheap surrogate)
    # --------------------------------------------------------------
    init_vals = [_objective(seq) for seq in init_pool]
    top_idx = np.argsort(init_vals)[:3]
    candidates = [init_pool[i] for i in top_idx]

    best_seq = None
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase 1 – smooth L‑BFGS on a Log‑Sum‑Exp surrogate
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]

    # allocate roughly 15 % of the budget to this phase
    smoothing_deadline = deadline - 200.0

    for start in candidates:
        if time.time() > smoothing_deadline:
            break

        v = np.log(start + 1e-30)   # log‑softmax variables
        for beta in beta_schedule:
            if time.time() > smoothing_deadline:
                break
            if _opt is None:
                break
            try:
                res = _opt.minimize(fun=_smooth_obj_and_grad,
                                    x0=v,
                                    args=(n, beta),
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': 2500,
                                             'disp': False,
                                             'ftol': 1e-12})
                v = res.x
            except Exception:       # pragma: no cover
                # In case the optimiser crashes – just continue.
                pass

        seq = _softmax(v)           # back to a probability vector
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Smoothing] new best surrogate = {best_val:.6f}")

    # If for any reason none of the candidates survived, fall back to a simple decay.
    if best_seq is None:
        best_seq = _exp_decay(n, rate=0.001)
        best_seq = best_seq / best_seq.sum()
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – intense ADAM polishing on the exact nonsmooth objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- ADAM polishing (main run) ---")
        best_seq, best_val = _adam_opt(best_seq,
                                       deadline,
                                       max_iters=5_000_000,
                                       lr0=0.2)
        print(f"[ADAM] surrogate after main polishing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – many short random‑restart refinements
    # --------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        # create a noisy copy of the current best
        noisy = best_seq + 0.005 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 20.0
        seq_r, val_r = _adam_opt(noisy,
                                 sub_deadline,
                                 max_iters=2_000_000,
                                 lr0=0.2)
        if val_r < best_val:
            best_seq, best_val = seq_r, val_r
            print(f"[Random restart] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final quick permutation hill‑climb (if time remains)
    # --------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        short_deadline = time.time() + 10.0
        seq_hc, val_hc = _perm_hill_climb(best_seq.copy(),
                                          n,
                                          short_deadline,
                                          rng,
                                          max_iters=80_000)
        if val_hc < best_val:
            best_seq, best_val = seq_hc, val_hc
            print(f"[Hill‑climb] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided black‑box routine
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
