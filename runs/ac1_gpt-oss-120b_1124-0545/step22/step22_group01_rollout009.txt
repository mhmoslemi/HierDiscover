def _adam_opt(start_seq, n, deadline, max_iters, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # gradient of max conv w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # Off-diagonal count twice
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]  # count only once
        # Subgradient of max conv j.s. of objective (just conv value)
        # But actual objective includes factor 2n; gradient w.r.t a same as for conv value.
        # Since scaling factor constant.
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Improved step‑function optimiser.

Goal: minimise 2·n·max(conv(a,a)) / (sum(a))².
The optimisation works as follows:

1️⃣  Build a large pool of analytical seeds (uniform, triangular, sine,
    Hann/Hamming/Blackman windows, Gaussians, exponentials, power‑laws,
    beta‑shapes and many Dirichlet draws).  All are ℓ₁‑normalised.

2️⃣  Evaluate the cheap surrogate 2·n·max(conv) and keep the best K.

3️⃣  From the best seeds generate a few random convex combinations
    (mixes of the top seeds) and re‑evaluate – this often yields a
    better starting point.

4️⃣  Refine every remaining seed with a **Adam‑style mirror descent**
    optimiser that works in the log‑softmax (probability‑simplex) space.
    The exact sub‑gradient of the active convolution entry is used;
    the update respects the simplex constraints automatically.

5️⃣  The current global best is polished for the remaining time with a
    long Adam run.  Occasionally short “escape” bursts are performed:
    a small Gaussian perturbation followed by a brief Adam optimisation.

All heavy work uses an FFT based convolution.  The final best vector is
stored in the global variable ``height_sequence_1`` for later continuation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simplex projection (used for tiny perturbations)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax (probability simplex) manifold
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr: float = 0.1,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the exact max‑convolution objective.
    Operates in log‑space, guaranteeing non‑negativity and sum‑to‑one.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    # initialise log‑probabilities (softmax will enforce feasibility)
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1

        # current feasible vector
        a = _softmax(log_a)

        # convolution and index of the active lag
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        # off‑diagonal entries appear twice
        g[idx_i] = 2.0 * a[rev_idx]

        # diagonal (i == rev_idx) counted only once
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # gradient of the surrogate (the factor 2·n is constant)
        dot = np.dot(g, a)                     # <g, a>
        grad_log = a * (g - dot)               # ∇_log (max conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # log‑space update
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate surrogate for the new a
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – a modest size that still leaves plenty of iterations.
    # --------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if it exists (helps continuations)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of analytical seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # sine windows – dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma))

    # Gaussians – sweep sigma fraction
    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians – centre & width sweep
    for mu_frac in np.linspace(0.15, 0.85, 10):
        for sigma_frac in np.linspace(0.05, 0.30, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponentials – log‑space sweep
    for rate in np.logspace(-5, 0.5, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweep
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.1, 10.0, 10)
    betas = np.geomspace(0.1, 10.0, 10)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # many Dirichlet draws with varied concentration
    conc_vals = np.geomspace(0.02, 5.0, 12)
    for conc in conc_vals:
        for _ in range(20):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the K best seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Generate a few random convex combinations of the elite seeds
    # ------------------------------------------------------------------
    combo_seeds = []
    for _ in range(150):
        weights = rng.dirichlet(np.ones(top_k))
        combo = sum(w * s for w, s in zip(weights, top_seeds))
        combo = combo / combo.sum()
        combo_seeds.append(combo)

    for vec in combo_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    # ------------------------------------------------------------------
    # 4. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Refine each elite seed with Adam optimisation
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly equal share (minimum 12 s per seed)
        share = max(12.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] Adam refinement budget ≈ {share:.1f}s")
        seq_opt, val_opt = _adam_opt(seed_vec, n, sub_deadline,
                                     max_iters=600_000,
                                     lr=0.12)
        print(f"   -> surrogate = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_seq, global_best_val = seq_opt, val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (Adam) ---")
        final_seq, final_val = _adam_opt(global_best_seq, n,
                                         deadline,
                                         max_iters=3_500_000,
                                         lr=0.08)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Random‑perturbation escape bursts while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)
        sub_deadline = time.time() + 7.0
        seq_tmp, val_tmp = _adam_opt(perturb, n, sub_deadline,
                                     max_iters=300_000,
                                     lr=0.09)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
