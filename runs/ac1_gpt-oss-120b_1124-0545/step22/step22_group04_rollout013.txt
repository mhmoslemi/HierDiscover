def project_onto_simplex(v):
    """projects vector onto simplex (sum=1, nonnegative)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Strategy
--------
1. **Rich initialisation** – generate a handful of analytically motivated
   distributions (uniform, exponential decay, inverse‑sqrt, and the
   *arcsine* family a_i ∝ 1/√((i+½)(n‑½‑i))) and a few random Dirichlet draws.
   All are normalised to a probability vector (sum = 1).

2. **Pick a promising seed** – evaluate the cheap surrogate 2·n·max(conv)
   for each seed and keep the best few.

3. **Smooth optimisation** – run L‑BFGS on the log‑sum‑exp surrogate of the
   max‑convolution for an increasing schedule of the smoothing parameter
   β (0.5 → 1e6).  The optimisation works in log‑softmax space, preserving
   non‑negativity and the simplex constraint.

4. **Exact polishing** – refine the result with a projected sub‑gradient
   method on the *true* max‑convolution objective.  The update is
   additive, followed by a fast O(n log n) projection onto the simplex.

5. **Random‑restart polishing** – repeatedly add a small Gaussian noise to
   the current best, polish it again with the sub‑gradient method and keep
   any improvement.  This occupies the remaining time budget.

6. **Optional short permutation hill‑climb** – a cheap simulated‑annealing
   style local search for the last few seconds.

All heavy work is timed; the routine stops gracefully before the 1000 s
budget expires and stores the final sequence in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# optional SciPy import – used only for the L‑BFGS smoothing phase
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_raw(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (sum is not forced to 1).
    Returns ``np.inf`` for degenerate inputs.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate used while the sequence is a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (used for cheap initial guesses)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _inv_sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def _arcsine_shape(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ ( 1 / √((i+½)(n‑½‑i)) )^p , normalised."""
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = raw ** p
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) of the smoothed objective:
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # log‑sum‑exp part, numerically stable
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # shape (2n‑1,)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # first n components

    # chain rule for softmax parametrisation
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (projected additive updates)
# ----------------------------------------------------------------------
def _proj_subgradient_opt(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Keeps the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_raw(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _project_onto_simplex(a)

        cur_val = _objective_raw(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent optimiser used in the reference implementation
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective using log‑space
    multiplicative updates (exponentiated‑gradient).  Returns the best
    sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space variables – start from the current distribution
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two entries or adds a tiny Gaussian perturbation.
    Accepts worsening moves according to a simple annealing schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # tiny perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimiser – the routine invoked by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method builds several analytically motivated seeds, refines the best
    one with a smoothing L‑BFGS phase and a projected sub‑gradient polish,
    and finishes with a few random‑restart and permutation refinements.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension – a power of two makes the FFT fast
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates (all normalised)
    # ------------------------------------------------------------------
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) simple exponential decay (the reference baseline)
    candidates.append(_exp_decay(n, rate=0.001))

    # 3) inverse‑sqrt shape
    candidates.append(_inv_sqrt_shape(n))

    # 4) arcsine family with a few exponents
    for p in (0.8, 1.0, 1.2):
        candidates.append(_arcsine_shape(n, p=p))

    # 5) a couple of random Dirichlet draws
    for _ in range(2):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Pick the most promising seed according to the cheap surrogate
    # ------------------------------------------------------------------
    scored = [(c, _objective_surrogate(c)) for c in candidates]
    scored.sort(key=lambda x: x[1])
    best_seq, best_score = scored[0]
    print(f"[Init] best surrogate (2·n·max conv) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation with L‑BFGS on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    v = np.log(best_seq + 1e-30)          # logits for the softmax
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
        50.0, 100.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]

    # reserve ~5 min for polishing and final tweaks
    smoothing_deadline = deadline - 300.0
    for beta in beta_schedule:
        if time.time() > smoothing_deadline:
            break

        max_iter_per_beta = 1500               # cheap but many β steps
        if _opt is not None:
            try:
                res = _opt.minimize(fun=_smooth_obj_and_grad,
                                    x0=v,
                                    args=(n, beta),
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': max_iter_per_beta,
                                             'ftol': 1e-12,
                                             'disp': False})
                v = res.x
            except Exception:   # pragma: no cover
                # If something goes wrong we simply continue with the current v.
                pass
        else:
            # SciPy unavailable – skip the smoothing phase.
            break

    cur_seq = _softmax(v)                     # back to a probability vector
    cur_val = _objective_surrogate(cur_seq)
    print(f"[After smoothing] surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – precise projected sub‑gradient polishing (the bulk of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Projected sub‑gradient polishing ---")
        cur_seq, cur_val = _proj_subgradient_opt(cur_seq, n, deadline,
                                                 max_iters=500_000,
                                                 eta0=5.0)
        print(f"[After PGD] surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – random‑restart refinements (if time permits)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        noisy = cur_seq + 0.005 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _exp_gradient_opt(noisy, n, sub_deadline,
                                           max_iters=2_000_000,
                                           eta0=10.0)
        if val_rr < cur_val:
            cur_seq, cur_val = seq_rr, val_rr
            print(f"[Random restart] improved surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – a quick permutation hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        short_deadline = time.time() + 10.0
        seq_last, val_last = _perm_hill_climb(cur_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=80_000)
        if val_last < cur_val:
            cur_seq, cur_val = seq_last, val_last
            print(f"[Final permutation] improved surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate the *true* objective via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = cur_seq.tolist()
    return cur_seq.tolist()
