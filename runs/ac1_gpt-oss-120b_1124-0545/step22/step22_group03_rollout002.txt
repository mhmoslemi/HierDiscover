"""
Improved search for a low‑peak auto‑convolution step‑function.

Algorithm
---------

1. **Warm‑start** – reuse a previously found sequence if available,
   otherwise start from a uniform distribution.

2. **Parametric sweep** – generate a few families of candidate
   distributions (Beta‑shaped with exponent α∈[‑0.9,0] and symmetric
   geometric shapes).  The exact objective 2·n·max(conv(a,a)) is
   evaluated; the best candidate seeds the next phase.

3. **Exponentiated‑gradient with a smooth max** –
   instead of using only the worst shift (as in the reference code)
   we minimise a differentiable surrogate
        Sβ(a)= (1/β)·log Σ_k exp(β·(a∗a)_k) .
   The gradient of Sβ can be written as a * (w+w_rev) where w are the
   soft‑max weights of the convolution entries.  Updating in log‑space
   (mirror‑descent) keeps the vector on the probability simplex.
   β is increased from a small value (≈2) to a large value (≈200) so the
   surrogate gradually approaches the true max.

4. **Fine‑grained simulated‑annealing** – tiny random mass transfers
   between two coordinates with a linear cooling schedule.  Only the
   exact objective is used here, and the best neighbour is kept.

The whole routine respects the 1000 s budget, prints progress, stores
the best sequence globally as ``height_sequence_1`` and returns it as a
plain ``list[float]``.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers (FFT based full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (exact when Σ a = 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under ℓ₁‑normalisation."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
# ----------------------------------------------------------------------
def _beta_shape_seq(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape:
        a_i ∝ (i+1)^{α}·(n-i)^{α}
    for α∈ℝ.  α<0 concentrates mass at the ends,
    α>0 concentrates near the centre.
    The vector is ℓ₁‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    # shift to avoid zero at the borders
    vals = ((i + 1) * (n - i)) ** alpha
    # Guard against overflow/underflow
    vals = np.where(vals == 0, 1e-300, vals)
    total = np.sum(vals)
    if total == 0.0:
        vals = np.ones_like(vals)
        total = n
    return vals / total


def _symmetric_geometric_seq(n: int, r: float) -> np.ndarray:
    """
    Symmetric geometric shape:
        a_i ∝ r^{min(i, n-1-i)}   with 0<r≤1.
    r close to 1 → uniform, r→0 → mass at the edges.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    vals = np.power(r, d)
    total = np.sum(vals)
    if total == 0.0:
        vals = np.ones_like(vals)
        total = n
    return vals / total


# ----------------------------------------------------------------------
# Exponentiated‑gradient using a smooth approximation of max
# ----------------------------------------------------------------------
def _exp_gradient_smooth(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 20_000,
                         eta0: float = 5.0,
                         beta_start: float = 2.0,
                         beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smooth max
    S_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).
    The schedule for β is linear from beta_start to beta_end.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # initialise log‑weights safely
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # -----------------------------------------------------------------
        # 1. Compute convolution and smooth weight vector w
        # -----------------------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        m = np.max(conv)
        beta = beta_start + (beta_end - beta_start) * it / max_iters
        # stable soft‑max on the convolution entries
        w = np.exp(beta * (conv - m))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum                           # Σ w = 1

        # -----------------------------------------------------------------
        # 2. Gradient of the smooth max wrt a
        #    grad = Σ_k w_k·∂conv_k/∂a  = a * (w + w_rev)  (central part)
        # -----------------------------------------------------------------
        w_sym = w + w[::-1]                         # length 2n‑1
        # full convolution a ∗ w_sym  → length 3n‑2
        full_grad = _conv_fft(a, w_sym)
        start = len(w_sym) // 2                     # = n‑1
        grad = full_grad[start:start + n]           # central n entries

        # -----------------------------------------------------------------
        # 3. Mirror‑descent update in log‑space
        # -----------------------------------------------------------------
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad
        a = _softmax(log_a)

        # -----------------------------------------------------------------
        # 4. Track the best exact objective seen so far
        # -----------------------------------------------------------------
        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 150_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Transfer a small amount of probability mass between two random coordinates.
    Temperature T and perturbation size sigma decay linearly.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] < 1e-15:
            attempts += 1
            continue

        # amount to move: a fraction of a[i]
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the overall 1000 s budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (height vector) that
    minimises evaluate_sequence().  The routine loads any previously
    found best vector stored in the global ``height_sequence_1``,
    performs a parametric sweep, a smooth‑max exponentiated‑gradient
    optimisation and a final simulated‑annealing polishing phase.
    The best sequence is stored globally and returned as a list.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # leave a small safety margin for the runner
    deadline = t0 + budget_s - 2.0

    # -----------------------------------------------------------------
    # 0. Determine problem size and start point
    # -----------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0 or np.sum(arr) <= 0.0:
                raise ValueError
            seq0 = arr / np.sum(arr)                     # ℓ₁‑normalize
            n = seq0.shape[0]
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # default length – large enough to exhibit fine structure but fast to handle
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Optimisation started – n = {n} ===")
    print(f"Initial exact surrogate value: {_surrogate_obj(seq0):.6f}")

    # -----------------------------------------------------------------
    # 1. Parametric sweep (Beta‑shapes + symmetric geometric)
    # -----------------------------------------------------------------
    candidates = [seq0]  # keep the current start as a candidate as well
    alphas = [-0.9, -0.8, -0.7, -0.6, -0.55, -0.5, -0.45, -0.4, -0.3, -0.2, -0.1, 0.0]
    for a in alphas:
        cand = _beta_shape_seq(n, a)
        candidates.append(cand)

    r_vals = [0.99, 0.95, 0.9, 0.85, 0.8, 0.7, 0.6]
    for r in r_vals:
        cand = _symmetric_geometric_seq(n, r)
        candidates.append(cand)

    best_seq = seq0.copy()
    best_val = _surrogate_obj(best_seq)

    for idx, cand in enumerate(candidates):
        val = _surrogate_obj(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if (idx + 1) % 5 == 0:
            print(f"  Sweep [{idx + 1}/{len(candidates)}] – best value so far: {best_val:.6f}")

    print(f"\nAfter parametric sweep best surrogate = {best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. Exponentiated‑gradient on smooth max (major share of the budget)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        eg_deadline = time.time() + max(5.0, 0.70 * remaining)
        print("\n--- Phase 1: Smooth‑max exponentiated‑gradient ---")
        seq_eg, val_eg = _exp_gradient_smooth(best_seq, eg_deadline,
                                              max_iters=30_000,
                                              eta0=5.0,
                                              beta_start=2.0,
                                              beta_end=200.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
    else:
        seq_eg, val_eg = best_seq, best_val

    # -----------------------------------------------------------------
    # 3. Simulated‑annealing polishing
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Phase 2: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(seq_eg, deadline, rng,
                                                     max_attempts=150_000,
                                                     T_start=0.02,
                                                     T_end=1e-8,
                                                     sigma_start=0.01)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_eg, val_eg

    # -----------------------------------------------------------------
    # 4. Pick the best of the two refinements
    # -----------------------------------------------------------------
    if val_sa < val_eg:
        best_seq = seq_sa
        best_val = val_sa
    else:
        best_seq = seq_eg
        best_val = val_eg

    print(f"\n=== Final exact surrogate after polishing = {best_val:.6f} ===")

    # -----------------------------------------------------------------
    # 5. Final evaluation with the provided evaluate_sequence()
    # -----------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
