```
def _project_simplex(v):
    # project onto probability simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

The goal is to find a non‑negative vector `a` (the “step‑function heights”) that minimises  

\[
\text{eval}(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\]

where `*` is the linear convolution.  
Because `evaluate_sequence` clips negatives and caps at 1000, we simply optimise on
the simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge0 .
\]

The previous code used an exponentiated‑gradient on the *hard* maximum.
Here we improve it by

1. **Richer initial pool** – many deterministic shapes and a huge number of
   Dirichlet draws with different concentrations.
2. **Smooth‑max gradient descent** – we replace the hard max by a
   differentiable soft‑max  
   \[
   \phi_\beta(a)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta (a*a)[k]}\Bigr)
   \]
   whose gradient is a weighted sum of all lags.  
   This yields a much finer descent direction than using a single active lag.
3. **Multi‑start optimisation** – the best 5 seeds are each refined
   independently; the total time budget is split automatically.
4. **Final cheap local walk** – a random walk on the simplex that accepts any
   improvement (tiny mass moves).
5. **Time‑aware design** – every stage checks the deadline; the whole search
   finishes well before the 1000 s limit.

All heavy work is done with FFT‑based convolutions (`O(n log n)`).  
Projection onto the simplex uses the classic O(n log n) algorithm
(Duchi et al., 2008).

The function `propose_candidate` follows the required signature and stores the
best sequence in the global variable `height_sequence_1` for warm‑starting later
runs.

```python
"""
Propose a non‑negative step function (list of heights) that minimises
evaluate_sequence.

Algorithm
---------
1. Build a large diverse pool of seeds (uniform, arcsine, triangular,
   several beta shapes, geometric/linear decays, perturbations, many
   Dirichlet draws).  All are normalised to sum = 1.
2. Keep the best few seeds (by the exact surrogate 2·n·max(conv)).
3. For each of the top‑K seeds run a *smooth‑max* gradient descent:
      – use φβ(a)= (1/β)·log Σ exp(β·conv) as a smooth approximation of max.
      – gradient:  ∇φβ = 2·conv(a, w_rev) on the segment of length n,
        where w = softmax(β·conv).
      – additive step on the simplex (projected gradient descent),
        step size ηₜ = η₀ / √t, β increased linearly to β_max.
   The best vector of all runs is retained.
4. Light‑weight random‑local‑search (tiny mass transfers) for a final polish.
5. Return the best sequence (and store it globally as height_sequence_1).

The code respects the 1000 s budget; each stage stops as soon as the
deadline is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (outputs a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq)),  assuming sum(seq) == 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with scalar concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric decay (rate ∈ (0,1))."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _cosine_shape(n: int) -> np.ndarray:
    """A smooth cosine hill – useful as a seed."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """
    Add small Gaussian noise, clip negatives and renormalise.
    """
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 250_000,
                        eta0: float = 0.6,
                        beta_start: float = 1.0,
                        beta_end: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation of the max‑convolution.
    Returns the best sequence found (sum = 1) and its exact surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # pre‑compute exponent schedule
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact convolution (needed for the surrogate & for the soft‑max)
        conv = _conv_fft(a, a)

        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

        # ---------- smooth‑max weights ----------
        # β grows linearly from beta_start to beta_end
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iters)

        maxc = np.max(conv)
        # stable softmax over the lag values
        exp_weights = np.exp(beta * (conv - maxc))
        w = exp_weights / exp_weights.sum()          # shape (2n‑1,)

        # ---------- gradient ----------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        # compute by a single convolution with the reversed weight vector
        conv_w_rev = _conv_fft(a, w[::-1])           # length 3n‑2
        grad = 2.0 * conv_w_rev[n - 1:2 * n - 1][::-1]  # shape (n,)

        # ---------- step ----------
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _simplex_projection(a)                  # stay on the simplex

    # final check (in case the last iteration improved)
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random amount from i→j,
    keep the move if it improves the exact surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a relative fraction of mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

    # Warm‑start from a previously found solution (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:

    seeds.append(_cosine_shape(n))

    # keep only the K best

    top_seeds = top_seeds[:top_k]
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    draw_limit = 150_000               # many more than the previous run
    draws = 0
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    print("\n--- Random Dirichlet sampling ---")
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print(f"  [Dir] new best surrogate = {best_val:.6f} (draw {draws})")
        # keep a few elite samples for later multi‑start
        if len(top_seeds) < top_k or val < top_seeds[-1][0]:
            top_seeds.append((val, vec.copy()))
            top_seeds.sort(key=lambda x: x[0])
            top_seeds = top_seeds[:top_k]
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Multi‑start smooth‑max gradient descent
    # ------------------------------------------------------------------
    print("\n--- Multi‑start smooth‑max gradient descent ---")
    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – skip optimisation
        seq_opt, val_opt = best_seq, best_val
    else:
        # give each start roughly equal time
        num_starts = min(len(top_seeds), 5)
        iters_per_start = 250_000
        per_start_deadline = time.time() + remaining / num_starts
        seq_opt, val_opt = best_seq, best_val

        for sidx, (_, start_vec) in enumerate(top_seeds[:num_starts]):
            if time.time() > deadline:
                break
            print(f"  [GD] start #{sidx+1}/{num_starts}")
            # run descent; cap iterations so we respect the overall deadline
            limit = min(iters_per_start,
                        int( (deadline - time.time()) * 0.9 ))
            cand_seq, cand_val = _smooth_max_descent(start_vec,
                                                     n,
                                                     deadline,
                                                     max_iters=limit,
                                                     eta0=0.6,
                                                     beta_start=1.0,
                                                     beta_end=2000.0)
            print(f"     -> surrogate = {cand_val:.6f}")
            if cand_val < val_opt:
                seq_opt, val_opt = cand_seq, cand_val
                print("        >>> Updated global best (gradient descent)")

    # ------------------------------------------------------------------
    # 4️⃣ Final cheap random local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_local, val_local = _local_random_search(seq_opt, n,
                                                    deadline,
                                                    attempts=500_000,
                                                    sigma=0.015)
        print(f"[Local] surrogate = {val_local:.6f}")
        if val_local < val_opt:
            seq_opt, val_opt = seq_local, val_local
            print("        >>> Updated global best (local walk)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_opt.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_opt:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for warm‑start in a possible subsequent call
    globals()["height_sequence_1"] = seq_opt.tolist()
    return seq_opt.tolist()
```
