"""
Propose a step‑function (non‑negative list of heights) that minimises the
``evaluate_sequence`` objective.

The search consists of three stages:

1️⃣ **Rich seed generation** – many deterministic shapes (uniform,
   arcsine, triangular, beta‑distributions, simple decays) and a large
   pool of random mixtures (beta + beta, perturbed uniform, Dirichlet
   draws, smoothed previous best).  All seeds are normalised to sum = 1
   and the best surrogate value is kept.

2️⃣ **Adaptive exponentiated‑gradient (AdaGrad‑mirror) optimisation** –
   we work in log‑space (mirror descent w.r.t. KL divergence) and use an
   AdaGrad‑style per‑coordinate learning‑rate schedule
   ``η_i = η₀ / sqrt(∑ g_i²)``.
   The sub‑gradient is the active convolution lag:
   ``g_i = 2·a_{k‑i}`` for indices that contribute to the maximal
   auto‑convolution.  The method runs until the time budget is almost
   exhausted, with a simple restart mechanism when no improvement has
   been seen for a long interval.

3️⃣ **Fine‑grained random local walk** – a cheap stochastic walk that
   moves a tiny amount of mass between two coordinates and accepts
   strict improvements.  This cleans up any remaining local irregularities.

The whole procedure respects the given ``budget_s`` (default 1000 s) and
stores the best sequence in the global ``height_sequence_1`` variable for
future warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(seq, seq)),  assuming sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _mixture_beta(n: int,
                  alpha1: float, beta1: float,
                  alpha2: float, beta2: float,
                  weight: float,
                  rng: np.random.Generator) -> np.ndarray:
    """Convex mixture of two beta densities."""
    a1 = _beta_distribution(n, alpha1, beta1)
    a2 = _beta_distribution(n, alpha2, beta2)
    mixed = weight * a1 + (1.0 - weight) * a2
    return mixed / mixed.sum()


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric decay (0 < rate < 1)."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * rate ** i
    eps = 1e-12
    a_sum = a.sum()
    if a_sum <= eps:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _smooth(seq: np.ndarray, kernel: np.ndarray) -> np.ndarray:
    """Convolve with a small kernel and renormalise."""
    conv = np.convolve(seq, kernel, mode='same')
    conv = np.maximum(conv, 0.0)
    s = conv.sum()
    if s <= 0.0:
        return seq.copy()
    return conv / s


def _random_dirichlet(n: int, conc: float,
                      rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adaptive exponentiated‑gradient optimisation (AdaGrad‑mirror)
# ----------------------------------------------------------------------
def _exp_gradient_adagrad(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          eta0: float = 8.0,
                          restart_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution with an AdaGrad learning‑rate.
    ``start_seq`` must already sum to 1.
    Returns the best sequence encountered and its surrogate value.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # working in log‑softmax space
    # AdaGrad accumulator (per‑coordinate sum of squared sub‑gradients)
    acc_sq = np.full_like(a, 1e-12)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # bookkeeping for restarts
    last_improve = 0

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- convolution & active lag -----
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # ----- sub‑gradient of the active lag -----
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # ----- AdaGrad update in log‑space -----
        acc_sq += g * g
        step = eta0 / (np.sqrt(acc_sq) + 1e-12)   # element‑wise step size
        log_a = log_a - step * g
        a = _softmax(log_a)                        # projects back onto simplex

        # ----- record best -----
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            last_improve = it

        # ----- simple restart if stuck for a long time -----
        if it - last_improve > restart_interval:
            # perturb the current best to try a new region
            log_a = np.log(best_seq + 1e-30) + 0.1 * rng.normal(size=n)
            a = _softmax(log_a)
            acc_sq.fill(1e-12)
            last_improve = it

        # safety net: break if we have exhausted the whole budget
        if it >= 5_000_000:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random amount of mass from
    one coordinate to another and keep the move only if it improves the
    surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a relative amount from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
