```python
"""
Adam‑based optimiser for the max‑self‑convolution bound.

The optimiser works in the log‑softmax (entropy) parametrisation of a
non‑negative ℓ₁‑normalised vector `a`.  A smooth approximation of the
maximum ‑operator is built from a log‑sum‑exp with an annealing schedule
over the smoothing parameter β.  An Adam optimiser updates the
log‑probabilities using the gradient of the smoothed objective (exact
gradient of the log‑sum‑exp).  After each β‑stage the current best
vector is recorded using the exact max‑convolution surrogate.
A final high‑budget simulated‑annealing sweep is applied to escape any
remaining local minima.  The routine respects a 1000 s wall‑clock
budget and stores the best found sequence in the global variable
`height_sequence_2` for future runs.
"""

import time
import math
import numpy as np

# optional SciPy import (not used in this version)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

def _smooth_max_conv(a: np.ndarray, beta: float):
    """
    Smoothed max of the convolution using log‑sum‑exp.

    Returns a tuple (smooth_max, grad_a) where grad_a = d/d a smoothed_max.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    shift = conv.max()
    # log‑sum‑exp in a stable way
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smoothed max = shift + log(sum_exp)/beta
    smooth_max = shift + np.log(sum_exp) / beta

    # gradient wrt a: ∂/∂a_i (∑_k w_k conv_k) where w_k = exp(beta*(conv_k-shift))/sum_exp
    w = exp_beta / sum_exp                     # shape (2n-1,)
    # gradient of conv = 2 * (rev_a * w) where * denotes convolution
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice of length n
    return smooth_max, grad_a

def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   max_iters_per_beta: int = 25000,
                   lr: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimiser in log‑softmax space using the smoothed max‑convolution.
    Returns the best sequence (as a probability vector) and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
            10_000_000.0
        ]

    # initialise log‑probabilities
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        iters = max_iters_per_beta
        beta = float(beta)
        for _ in range(iters):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(log_a)

            smooth_max, grad_a = _smooth_max_conv(a, beta)

            # gradient w.r.t. log‑softmax variables (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            log_a -= step

            # evaluate exact surrogate occasionally (every 200 iter) to reduce cost
            if (t % 200) == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check at end of schedule
    final_a = _softmax(log_a)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.01,
                         T_end: float = 1e-9):
    """
    Simple simulated‑annealing refinement.
    Returns the best sequence and its exact surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    def _exp_right(rate):
```

"""
Adam‑based optimiser for the max‑self‑convolution bound.

The routine works in the log‑softmax (entropy) parametrisation of a
non‑negative ℓ₁‑normalised vector `a`.  A smooth approximation of the
maximum is built from a log‑sum‑exp with an annealing schedule over the
smoothing parameter β.  An Adam optimiser updates the log‑probabilities
using the gradient of the smoothed objective.  After each β‑stage the
current best vector is recorded using the exact max‑convolution surrogate.
A final high‑budget simulated‑annealing sweep refines the result.
The best sequence is stored in the global variable ``height_sequence_2``
for reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (not required for this version)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (exact max for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max of the convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max_conv(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, grad_a) where

        smooth_max = shift + log(sum(exp(beta * (conv - shift)))) / beta

    and grad_a = d/d a smooth_max.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta

    w = exp_beta / sum_exp                     # weight vector, length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice of length n
    return smooth_max, grad_a


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax variables) using the smoothed max
# ----------------------------------------------------------------------
def _adam_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule=None,
    max_iters_per_beta: int = 25000,
    lr: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Mirror‑descent (Adam) on the smoothed max‑convolution.
    Returns the best probability vector and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    log_a = np.log(start_seq + 1e-30)          # initialise log‑probabilities
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(log_a)

            smooth_max, grad_a = _smooth_max_conv(a, beta)

            # gradient w.r.t. log‑softmax variables (mirror‑descent form)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            log_a -= step

            # occasional exact evaluation (every 200 steps)
            if (t % 200) == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(log_a)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_attempts: int = 300_000,
    start_sigma: float = 0.002,
    T_start: float = 0.01,
    T_end: float = 1e-9,
):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence and its exact surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
