"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

*   Work with a *probability vector* (sum = 1); the objective is
    ``2·n·max(conv(a,a))`` and is invariant under scaling.
*   Choose a relatively large problem size (default n=8000).  With more
    degrees of freedom the maximal convolution can be made closer to its
    theoretical lower bound 1/(2n‑1).
*   Build a diverse pool of analytic seeds:
    – uniform,
    – Beta‑shaped (including the arcsine case α=β=0.5),
    – power‑law, exponential decay,
    – Dirichlet draws with various concentration parameters.
*   Rank the seeds by the *exact* surrogate objective (the same as the
    true objective for a normalised vector) and keep the best three.
*   For each of those seeds spend a share of the remaining budget in a
    two‑stage optimisation:
    1.  **Smoothed Adam** – minimise a log‑sum‑exp (smooth‑max) surrogate.
        The temperature β is increased gradually (10 → 2·10⁶) and we run a
        large number of FFT‑based convolution steps.
    2.  **Exact exponentiated‑gradient (EG)** – sub‑gradient descent on the
        true max‑convolution using the same mirror‑descent update in the
        log‑softmax (probability) space.
*   After all seeds are refined, run a final long EG polish on the current
    best candidate, followed by a short permutation‑annealing tweak.
*   The best sequence found is stored in the global variable
    ``height_sequence_1`` and also returned as a plain list of ``float``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) where sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the *exact* max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent in log‑softmax space using the exact max entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)          # keep away from log(0)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (2·a_{k-i})
        n = a.shape[0]
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # recover probabilities (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur = _objective_surrogate(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp smooth approximation of max(conv).
    Operates in the log‑softmax (probability) space.
    """
    if beta_schedule is None:
        # increasingly sharp temperatures
        beta_schedule = [10, 40, 160, 640, 2560, 10_240, 40_960,
                         163_840, 655_360, 2_621_440, 10_485_760]

    n = start_seq.shape[0]
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                # (2n‑1)‑vector, sums to 1

            # gradient w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # pick the central n entries

            # chain rule for log‑softmax
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded (prevents overflow)
            log_a = np.clip(log_a, -30.0, 30.0)

            cur = _objective_surrogate(a)
            if cur < best_val:
                best_val = cur
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = start_seq.shape[0]
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

            beta_schedule=[10, 40, 160, 640, 2560, 10_240,

                           10_485_760],
            iters_per_beta=30_000,
        )

        # ---- exact EG polishing on the same slice (if time left) -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                seq_adam,
                sub_deadline,
                max_iters=800_000,
                eta0=6.0,
            )
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final long EG polishing (uses all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            deadline,
            max_iters=2_500_000,
            eta0=5.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very short permutation tweak (a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 15.0
        seq_last, val_last = _perm_anneal(
            global_best_seq.copy(),
            short_deadline,
            max_iters=80_000,
        )
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the provided function and store result
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation enlarges the search space, enriches the seed pool (including the arcsine‑Beta shape known to flatten the sum distribution), dedicates the bulk of the time budget to a high‑resolution smoothed‑Adam phase followed by an exact exponentiated‑gradient polish, and finishes with a short permutation anneal. All computations respect the 1000 s time limit, and the best sequence is stored in the global variable `height_sequence_1`.
