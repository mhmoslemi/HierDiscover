"""
propose_candidate
-----------------
Search for a non‑negative step function that minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear (full) convolution.  The optimiser is
purely ``numpy`` based and works entirely in the probability simplex
(i.e. the vector is kept normalised to ℓ₁‑norm 1, making the denominator
equal to 1).  It consists of three complementary phases:

1. **Exponentiated‑gradient (mirror descent)** – a fast sub‑gradient
   method that moves the log‑weights in the direction of the active
   (maximal) convolution shift.  The learning‑rate decays as 1/√t.

2. **Global simulated‑annealing** – occasionally huge perturbations of
   the log‑weights are tried (Gaussian noise) and accepted with a
   classic Metropolis rule.  This helps to jump out of local minima
   that the EG step may get stuck in.

3. **Fine‑grained local annealing** – tiny mass transfers between two
   random coordinates (the classic “swap‑mass” move).  It polishes the
   solution found by the previous phases.

Multiple random restarts (including the current global best, a uniform
vector and a simple arcsine‑shaped vector) are tried; the best sequence
ever seen is stored in the global variable ``height_sequence_1`` and
returned.  The routine respects the overall time budget (default 1000 s)
by checking the deadline after each iteration.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical primitives (FFT‑based convolution, softmax, objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _surrogate_obj(a: np.ndarray) -> float:
    """
    Exact objective under the assumption sum(a)=1:
          2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# 1. Exponentiated‑gradient optimiser (single‑active shift)
# ----------------------------------------------------------------------
def _exp_gradient(start_seq: np.ndarray,
                  deadline: float,
                  eta0: float = 5.0,
                  max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the max‑convolution (single worst shift).
    Returns best (ℓ₁‑normalised) vector seen and its surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # avoid log(0)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(conv.argmax())                     # worst shift

        # gradient of max‑convolution w.r.t. a (ordered pairs counted twice)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        # Each ordered pair contributes a factor a_{k-i}; the pair (i,j)
        # and (j,i) are handled together by adding twice the term, except
        # when i == j (i == k//2) where we should add only once.
        grad = np.zeros_like(a)
        grad[idx] = a[k - idx]                     # first orientation
        # second orientation (i != k-i) – add again
        mask = idx != (k - idx)
        grad[idx[mask]] += a[k - idx[mask]]

        # learning‑rate decay
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a -= eta * grad
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# 2. Global simulated‑annealing (log‑weight perturbations)
# ----------------------------------------------------------------------
def _global_sa(start_seq: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_attempts: int = 2_000_000,
               T_start: float = 0.5,
               T_end: float = 1e-6,
               sigma_start: float = 0.2,
               sigma_end: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Large‑scale annealing: add Gaussian noise to the log‑weights,
    accept with Metropolis rule.  The schedule is linear in the
    progress variable (attempts / max_attempts).
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        prog = attempts / max_attempts
        T = T_start * (1.0 - prog) + T_end * prog
        sigma = sigma_start * (1.0 - prog) + sigma_end * prog

        # perturbation in log‑space
        log_a = np.log(a + 1e-30) + sigma * rng.normal(size=n)
        ns = _softmax(log_a)

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# 3. Local simulated‑annealing (tiny mass swaps)
# ----------------------------------------------------------------------
def _local_sa(start_seq: np.ndarray,
              deadline: float,
              rng: np.random.Generator,
              max_attempts: int = 500_000,
              T_start: float = 0.02,
              T_end: float = 1e-8,
              sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Classical SA: move a small amount of probability mass from one index
    to another.  Both temperature and perturbation magnitude decay
    linearly.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        prog = attempts / max_attempts
        T = T_start * (1.0 - prog) + T_end * prog
        sigma = sigma_start * (T / T_start)

        # donor index (must have some mass)
        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # amount to move – a small fraction of a[i]
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: a cheap arcsine‑shaped initialisation (often a good start)
# ----------------------------------------------------------------------
def _arcsine_initial(n: int) -> np.ndarray:
    """U‑shaped distribution proportional to 1/√(x·(1‑x))."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    inv = 1.0 / np.sqrt(idx * (n - idx + 0.5))
    inv /= inv.sum()
    return inv


# ----------------------------------------------------------------------
# Main entry point – respects the global time budget.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise the step‑function heights and return the best list found.
    The routine respects the overall 1000 s budget, stores the result in
    the global ``height_sequence_1`` variable and prints progress.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 1.5          # safety margin

    # --------------------------------------------------------------
    # Initialise – try to reuse a previously found solution,
    # otherwise start from several diverse seeds
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    init_candidates = []

    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and arr.sum() > 0.0:
                init_candidates.append(arr / arr.sum())
        except Exception:
            pass

    # a uniform start (always useful)
    n_default = 4000
    init_candidates.append(np.full(n_default, 1.0 / n_default, dtype=np.float64))

    # arcsine start (U‑shaped)
    init_candidates.append(_arcsine_initial(n_default))

    # a few random Dirichlet starts (different random seeds)
    for _ in range(2):
        rand = rng.random(n_default)
        rand /= rand.sum()
        init_candidates.append(rand)

    # --------------------------------------------------------------
    # Keep the overall best seen across all initialisations
    # --------------------------------------------------------------
    best_global_seq = None
    best_global_val = float("inf")

    # Allocate roughly 80 % of the remaining budget to optimisation,
    # 20 % to the final true evaluation / housekeeping.
    remaining = deadline - time.time()
    opt_deadline = time.time() + 0.80 * remaining

    # Distribute the optimisation time evenly among the initial candidates
    per_candidate_deadline = opt_deadline / len(init_candidates)

    print("\n=== Optimisation started – using {} initial candidates ===".format(
          len(init_candidates)))

    for idx, start_seq in enumerate(init_candidates):
        # ensure we respect the global deadline
        if time.time() > opt_deadline:
            break

        n = start_seq.shape[0]
        print(f"\n--- Candidate {idx + 1}/{len(init_candidates)} – n = {n} ---")
        print(f"Initial surrogate value: {_surrogate_obj(start_seq):.6f}")

        # ------- Phase 1 – Exponentiated gradient (long) -------------
        eg_deadline = time.time() + 0.55 * (per_candidate_deadline - time.time())
        seq_eg, val_eg = _exp_gradient(start_seq, eg_deadline,
                                       eta0=5.0, max_iters=5_000_000)
        print(f"[EG] best surrogate = {val_eg:.6f}")

        # ------- Phase 2 – Global simulated annealing -----------------
        remaining = per_candidate_deadline - (time.time() - t0)
        if remaining > 5.0:
            ga_deadline = time.time() + 0.30 * remaining
            seq_ga, val_ga = _global_sa(seq_eg, ga_deadline, rng,
                                        max_attempts=2_000_000,
                                        T_start=0.5, T_end=1e-6,
                                        sigma_start=0.2, sigma_end=1e-4)
            print(f"[GA] best surrogate = {val_ga:.6f}")
        else:
            seq_ga, val_ga = seq_eg, val_eg

        # ------- Phase 3 – Local simulated annealing (fine) ----------
        remaining = per_candidate_deadline - (time.time() - t0)
        if remaining > 3.0:
            la_deadline = time.time() + remaining
            seq_la, val_la = _local_sa(seq_ga, la_deadline, rng,
                                       max_attempts=500_000,
                                       T_start=0.02, T_end=1e-8,
                                       sigma_start=0.003)
            print(f"[LA] best surrogate = {val_la:.6f}")
        else:
            seq_la, val_la = seq_ga, val_ga

        # choose the best for this candidate
        candidate_best_seq = seq_la
        candidate_best_val = val_la

        if candidate_best_val < best_global_val:
            best_global_val = candidate_best_val
            best_global_seq = candidate_best_seq.copy()
            print(f"=> New global best surrogate = {best_global_val:.6f}")

    # --------------------------------------------------------------
    # Final true evaluation (might differ slightly due to rounding)
    # --------------------------------------------------------------
    final_seq = best_global_seq
    # Convert to plain Python list for the caller/evaluation
    final_list = final_seq.tolist()
    try:
        true_val = evaluate_sequence(final_list)
        print(f"\nTrue evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for later runs
    globals()["height_sequence_1"] = final_list
    print("\n=== Optimisation finished – best surrogate = {:.6f} ===".format(
          best_global_val))
    return final_list
