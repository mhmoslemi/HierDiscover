def _smooth_descent(a_start, n, deadline, alphas=[2,5,10,20,50,100], iters_per_alpha=2000, eta0=1.0):

    a = a_start.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    total_iters = 0

    for alpha in alphas:
        for itr in range(1, iters_per_alpha+1):
            # time check
            if time.time()>deadline:
                return best_seq, best_val

            # compute conv and gradient
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # compute weights safely
            exp_term = np.exp(alpha * (conv - max_c))  # avoid overflow
            w = exp_term / exp_term.sum()  # shape L
            a_rev = a[::-1]
            conv_wr = _conv_fft(w, a_rev)
            grad = 2 * conv_wr[n-1: n-1 + n]

            # projector gradient descent step
            it_num = total_iters + itr
            eta = eta0 / np.sqrt(it_num)  # diminishing step

            a_new = a - eta * grad
            a_new = _project_simplex(a_new)  # ensures non-neg sum=1

            a = a_new

            # evaluate surrogate occasionally maybe each 50 iters
            if itr % 100 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
        total_iters += iters_per_alpha

    return best_seq, best_val

"""
A fast “smooth‐max” projected gradient descent optimiser.

The original problem is to minimise

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0

After normalising ∑a_i = 1 the objective becomes 
    2·n·max(conv(a,a))

which is exactly the surrogate used throughout the optimiser.
The new optimiser works as follows:

1️⃣  Generate a modest pool of analytical seeds (exponential decays,
    uniform, random Dirichlet draws, and a few reversed versions).
2️⃣  Keep the few best seeds according to the exact surrogate.
3️⃣  For each seed run a *smooth‑max* projected gradient descent:
   – replace the non‑smooth max by a log‑sum‑exp
     f_α(a)= (1/α)·log∑_k exp(α·(a∗a)_k) .
   – the gradient can be obtained analytically and computed
     with two FFT‑based convolutions.
   – after each gradient step project back onto the probability simplex
     (non‑negative, sum = 1).
   – increase α progressively (1 → 2 → 5 → 10 → 20 → 40 → 80) so the
     optimisation first sees a smooth landscape and finally the true
     max‑objective.
4️⃣  The best sequence from the above stage is optionally polished
    with the exact exponentiated‑gradient (mirror‑descent) routine used
    in earlier submissions.
5️⃣  The final sequence is returned (and stored in ``height_sequence_1``)
    for possible continuation runs.

The whole routine respects the 1000 s time budget, constantly checks the
deadline and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (projected)
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        alphas=(1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0),
                        iters_per_alpha: int = 2500,
                        eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp smooth approximation
    of max(conv(a,a)).  Operates in the probability simplex.
    Returns the best sequence seen (according to the exact surrogate)
    and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    total_iters = 0
    for alpha in alphas:
        # schedule: fixed number of inner iterations per α
        for it in range(1, iters_per_alpha + 1):
            if time.time() > deadline:
                return best_seq, best_val

            # 1) convolution a∗a
            conv = _conv_fft(a, a)

            # 2) safe soft‑max weights (log‑sum‑exp trick)
            m = np.max(conv)
            exp_term = np.exp(alpha * (conv - m))
            w = exp_term / exp_term.sum()          # shape L = 2n‑1

            # 3) gradient of the smooth objective:
            #    ∇_a f = 2 * (w * a_rev)[n‑1 : n‑1+n]
            a_rev = a[::-1]
            conv_wr = _conv_fft(w, a_rev)          # length 2n‑1 + n‑1
            grad = 2.0 * conv_wr[n - 1: n - 1 + n]

            # 4) diminishing step size (AdaGrad‑style)
            total_iters += 1
            eta = eta0 / math.sqrt(total_iters)

            # 5) projected gradient step
            a = _project_simplex(a - eta * grad)

            # occasional exact evaluation to keep track of the best
            if it % 250 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final exact check (in case the last α was still far from the true max)
    cur_val = _objective_surrogate(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser with exact sub‑gradient (from earlier)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (same as the previous code)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        log_a = log_a - eta * grad_log
        a = np.exp(log_a)
        a = a / a.sum()                 # keep on simplex (softmax)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a small pool of diverse seeds,
    keeps the best few, refines each with a smooth‑max projected gradient
    descent, finishes with a short exact mirror‑descent polish and returns
    the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a finer resolution while still cheap.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load a previous solution (if any) for possible continuation.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            prev_arr = arr / arr.sum()

    # ------------------------------------------------------------------
    # 1. Build a modest pool of analytic seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0005))          # very gentle decay

    # exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0, 30):
        seeds.append(_exp_decay(n, rate))
        # also the reflected version (mass on the right side)
        seeds.append(_exp_decay(n, rate)[::-1])

    # random Dirichlet draws with several concentrations
    for conc in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(4):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick exact surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (exact surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # about 12 % of the remaining budget to this seed (min 15 s)
        share = max(15.0, remaining * 0.12 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) Smooth‑max projected gradient descent
        seq_smooth, val_smooth = _smooth_max_descent(
            seed_vec, n, sub_deadline,
            alphas=(1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0),
            iters_per_alpha=2500,
            eta0=0.7)

        # b) Short exact mirror‑descent polish on the result
        if time.time() < sub_deadline - 5.0:
            seq_opt, val_opt = _exp_gradient_opt(
                seq_smooth, n,
                sub_deadline - 5.0,
                max_iters=250_000,
                eta0=5.0)
            if val_opt < val_smooth:
                seq_smooth, val_smooth = seq_opt, val_opt

        print(f"[Seed {si+1}] local best surrogate = {val_smooth:.6f}")

        if val_smooth < global_best_val:
            global_best_seq = seq_smooth
            global_best_val = val_smooth
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (smooth‑max) ---")
        final_seq, final_val = _smooth_max_descent(
            global_best_seq, n, deadline,
            alphas=(10.0, 20.0, 40.0, 80.0, 120.0),
            iters_per_alpha=4000,
            eta0=0.4)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Small random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 8.0
        seq_tmp, val_tmp = _smooth_max_descent(
            perturb, n, sub_deadline,
            alphas=(20.0, 40.0, 80.0),
            iters_per_alpha=2500,
            eta0=0.5)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
