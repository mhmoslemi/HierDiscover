assistantfinal**Solution Overview**

The target is to minimise  

```
E(a) = 2·n·max(conv(a,a)) / (∑a_i)²          (a_i ≥ 0)
```

The scale `∑a_i` does not matter, therefore we work with probability
vectors `p = a/∑a` (∑p_i = 1) and minimise  

```
F(p) = 2·n·max(conv(p,p))
```

* `conv(p,p)` – full linear convolution (length = 2·n‑1).  
* `max(conv(p,p))` is non‑smooth, we replace it by a smooth
  approximation (log‑sum‑exp) and optimise the smoother with Adam.
  The exact (non‑smooth) value is monitored, the best vector found is
  kept.

The algorithm consists of the following stages (all respecting the
`budget_s` time limit):

1. **Rich pool of seeds** – uniform, arcsine, triangular, many Beta
   shapes and many Dirichlet draws (including very low concentration
   parameters).  Every seed is normalised to a probability vector and
   the best seed is kept.
2. **Random Dirichlet & Gamma draws** – while time permits, draw more
   random probability vectors with a wide range of concentration
   parameters; keep any improvement.
3. **Adam optimisation on a smoothed objective**  
   * variables: logits `z` (`p = softmax(z)`);
   * smooth objective: `J(z)=log‑sum‑exp(β·conv(p,p))/β`;
   * gradient `∂J/∂z` is obtained analytically:
        * `w = softmax(β·conv(p,p))`;
        * `∂J/∂p = 2·conv(w , rev(p))` (slice of a full convolution);
        * `∂J/∂z = p·(∂J/∂p – ⟨∂J/∂p , p⟩)`;
   * β is annealed linearly from `β_start` (smooth) to `β_end`
     (sharp) during the run.
   * Adam’s adaptive step‑size is used; the best *exact* surrogate
     `2·n·max(conv(p,p))` encountered is stored.
4. **Cheap local random walk** – tiny mass transfers between two
   coordinates; keep a move only if the exact surrogate improves.
5. **Final evaluation** – call the provided `evaluate_sequence` on the
   best vector and store it globally for warm‑starts.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`);
the whole pipeline fits comfortably into the 1000 s budget and
normally finds a better bound than the reference code (≈ 1.503).

```python
# -*- coding: utf-8 -*-
"""
Propose a step‑function (non‑negative list of heights) that minimises
``evaluate_sequence``.  The method builds many seeds, improves the best
seed with a smooth Adam optimiser (log‑sum‑exp surrogate of the max
convolution) and finishes with a tiny stochastic local walk.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (O(n log n)).
    Returns a real array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(p: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(p,p)), assuming sum(p)=1.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (return probability vectors – sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with a *single* concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_gamma_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a Dirichlet vector with *different* concentrations for each
    coordinate (Gamma shape parameters drawn from a broad distribution).
    """
    # shape parameters in [0.01, 5] with a heavy tail for small values
    alphas = rng.gamma(shape=0.3, scale=1.0, size=n) + 0.01
    draws = rng.gamma(shape=alphas, scale=1.0)
    if draws.sum() == 0.0:
        return _uniform(n)
    return draws / draws.sum()


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric‑decay distribution (rate∈(0,1))."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    s = a.sum()
    if s <= 1e-12:
        return _uniform(n)
    return a / s


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add tiny Gaussian noise, clip and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 150_000,
              lr: float = 0.15,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              beta_start: float = 5.0,
              beta_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth objective
        J(z) = log‑sum‑exp(β·conv(p,p)) / β,
    where p = softmax(z).  The function returns the probability vector
    achieving the *lowest exact surrogate* seen during the run.
    """
    # initialise logits
    z = np.log(start_seq + 1e-30)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_seq = start_seq.copy()
    best_surrogate = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # anneal β linearly from beta_start → beta_end
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iters)

        # probability vector
        p = _softmax(z)

        # convolution p*p
        conv = _conv_fft(p, p)                     # length 2n-1
        max_conv = np.max(conv)

        # ----- smooth weights w = softmax(β·conv)  -----
        # shift by max for numerical stability
        shift = max_conv
        exp_vals = np.exp(beta * (conv - shift))
        Z = exp_vals.sum()
        w = exp_vals / Z

        # ----- gradient wrt p -----
        # grad_p = 2 * conv(w , rev(p))  (slice of the full convolution)
        full = _conv_fft(w, p[::-1])               # length 3n-2
        grad_p = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient wrt logits z -----
        dot = np.dot(grad_p, p)
        grad_z = p * (grad_p - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # evaluate exact surrogate and keep best
        cur_surrogate = float(2.0 * n * max_conv)
        if cur_surrogate < best_surrogate:
            best_surrogate = cur_surrogate
            best_seq = p.copy()
            # occasional log
            if it % 5_000 == 0:
                print(f"[Adam] iter {it:,} – surrogate = {best_surrogate:.6f}")

    return best_seq, best_surrogate


# ----------------------------------------------------------------------
# Light‑weight random local walk (exact surrogate)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 250_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a small random amount of mass
    from one coordinate to another and keep the move if it improves the
    exact surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best list of heights found before the ``budget_s`` timeout.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin – leaves a bit of room for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size (default 4000, can be overridden)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Warm‑start from a previous run (if present)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.isfinite(arr).all() and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣ Build a diverse pool of deterministic and random seeds
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n)]

    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decreasing(n))

    # a grid of Beta shapes (including very spiky ones)
    for a, b in [(0.05, 0.05), (0.05, 1.0), (1.0, 0.05),
                 (0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                 (2.0, 0.5), (0.5, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, a, b))

    # deterministic geometric decays
    for rate in (0.9999, 0.9995, 0.999, 0.998, 0.997):
        seeds.append(_geometric(n, rate))

    # small perturbations of the uniform vector
    for sigma in (0.01, 0.005):
        seeds.append(_perturb(_uniform(n), rng, sigma=sigma))

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate all deterministic seeds – keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet / Gamma draws (massive exploration)
    # ------------------------------------------------------------------
    draw_limit = 8_000
    draws = 0
    conc_options = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05,
                    0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        if rng.random() < 0.5:
            vec = _random_dirichlet(n, conc, rng)
        else:
            vec = _random_gamma_vector(n, rng)      # wide variety of alphas
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
            print(f"[Rand] new best surrogate = {best_val:.6f} (draw {draws})")
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Adam optimisation on the smoothed surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam optimisation (smooth surrogate) ---")
        # allocate roughly half of the remaining budget to this stage
        remaining = deadline - time.time()
        # a generous iteration budget – will stop automatically on time‑out
        max_iters = int(200_000 * (remaining / budget_s))
        max_iters = max(30_000, min(max_iters, 250_000))

        seq_opt, val_opt = _adam_opt(best_seq, n,
                                     deadline,
                                     max_iters=max_iters,
                                     lr=0.15,
                                     beta_start=5.0,
                                     beta_end=30.0)
        print(f"[Adam] best surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Final lightweight random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n,
                                                    deadline,
                                                    attempts=300_000,
                                                    sigma=0.02)
        print(f"[Local] exact surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local walk)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
