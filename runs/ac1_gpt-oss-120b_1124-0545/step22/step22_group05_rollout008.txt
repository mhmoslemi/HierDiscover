"""
Propose a step‑function (non‑negative list of heights) that minimizes the
provided ``evaluate_sequence`` objective.

Algorithm
---------

1️⃣  Build a rich initial pool:
    * uniform, arcsine, triangular, geometric, linear decay.
    * a dense grid of Beta(α,β) shapes (α,β∈[0.1,2] with step 0.2) – this
      explores the whole family of symmetric power‑law densities.
    * a handful of random Dirichlet draws (many concentration values) and
      a warm‑start from the previous run if available.

   All candidates are normalised to sum = 1 and the exact surrogate
   ``2·n·max(conv)`` (sum already fixed) is evaluated; the best seed is kept.

2️⃣  **Weighted exponentiated‑gradient (mirror descent).**  
   Instead of using the sub‑gradient from a single active lag, we use a
   *soft‑max* weighting of **all** convolution lags:
   ```
       w_k ∝ exp(λ·conv_k)     (λ grows from a modest value to a large one)
   ```
   The gradient of the smoothed objective is the weighted sum of the
   individual lag‑gradients and can be computed by a second FFT‑based
   convolution.  Updates are performed in log‑space (entropy mirror map) with
   an Adam‑style adaptive step size; the iterate is projected back onto the
   probability simplex via a stable soft‑max.

   The loop runs until the time budget is exhausted (or a safe iteration cap
   is reached).  The best sequence seen during the run (according to the
   exact surrogate) is retained.

3️⃣  **Cheap local random walk** – tiny mass transfers between random
   coordinates are accepted only when the true surrogate improves.  This
   usually squeezes a few more digits out of the best sequence.

4️⃣  The final sequence (still normalised to sum = 1) is returned; the caller
   will feed it to ``evaluate_sequence`` which rescales it if necessary.

The whole procedure respects the 1000 s budget; heavy optimisation (step 2)
automatically stops when the deadline is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq, seq)), assuming sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all vectors sum to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    # avoid the all‑zero vector for extreme parameters
    if a.sum() == 0.0:
        a = _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (soft‑max weighting)
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Returns (g, max_conv) where ``g`` is the gradient of the smoothed
    objective (soft‑max with temperature 1/lam) w.r.t. the probability vector
    ``a``, and ``max_conv`` = max(conv(a,a)).
    """
    n = len(a)
    # 1. convolution a∗a
    conv = _conv_fft(a, a)                # length 2n‑1
    max_conv = float(np.max(conv))

    # 2. soft‑max weights over the lags
    #   w_k ∝ exp(lam·conv_k)
    lc = lam * conv
    lc_max = np.max(lc)
    exp_lc = np.exp(lc - lc_max)
    w = exp_lc / exp_lc.sum()             # still length 2n‑1, sums to 1

    # 3. gradient = 2 * (a ⋆ w)  (cross‑correlation)
    #    which can be obtained via a convolution with the reversed weight.
    w_rev = w[::-1]
    grad_conv = _conv_fft(a, w_rev)       # length 3n‑2
    # slice to obtain the part that corresponds to indices 0 … n‑1
    grad_slice = grad_conv[n - 1:2 * n - 1]    # length n
    g = 2.0 * grad_slice[::-1]                # reverse to get correct order
    return g, max_conv


# ----------------------------------------------------------------------
# Weighted exponentiated‑gradient optimisation (Adam on log‑probabilities)
# ----------------------------------------------------------------------
def _exp_gradient_weighted(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 3_000_000,
                           eta0: float = 4.0,
                           lam_start: float = 200.0,
                           lam_end: float = 4000.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam‑style adaptation on the smoothed surrogate.
    ``start_seq`` must sum to 1.  Returns the best sequence seen and its
    surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    # initial surrogate (exact max‑convolution)
    conv0 = _conv_fft(a, a)
    best_val = float(2.0 * n * np.max(conv0))

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear schedule for λ (temperature)
        lam = lam_start + (lam_end - lam_start) * (it / max_iters)

        # gradient of smoothed max‑conv
        g, max_conv = _weighted_gradient(a, lam)

        # Adam update on log‑variables
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)       # raw step size (decreasing)

        log_a = log_a - eta * (m_hat / (np.sqrt(v_hat) + eps))
        a = _softmax(log_a)              # projects back onto the simplex

        # exact surrogate for this iterate (no extra FFT needed – we already have max_conv)
        cur_val = float(2.0 * n * max_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[Weighted EG] iter {it:7d}: surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search (exact surrogate)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: move a random fraction of mass from
    one coordinate to another and keep the move if the exact surrogate improves.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val
