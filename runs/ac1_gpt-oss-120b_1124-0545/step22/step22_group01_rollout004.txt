def _smooth_exp_gradient_opt(start_seq, n, deadline, tau, max_iters=1e6):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute convolution
        conv = _conv_fft(a, a)  # returns length 2n-1
        # compute softmax weights p
        # to avoid overflow, subtract max
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        p = exp_vals / exp_vals.sum()

        # compute gradient raw
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)  # length 3n-2
        # slice
        slice_start = n - 1
        g_raw = 2.0 * conv_rev[slice_start: slice_start + n]  # 2 * Σ_j a_j p_{i+j}
        diag_idx = 2 * np.arange(n)
        # ensure diag idx within bounds
        g_raw -= a * p[diag_idx]

        # compute mirror descent update
        dot = np.dot(g_raw, a)
        grad_log = a * (g_raw - dot)
        eta = 5.0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        # numeric stability
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

"""
Optimise a non‑negative step‑function (a vector of heights) for the
objective used by ``evaluate_sequence``:

    f(a) = 2·n·max(conv(a,a)) / (sum(a))².

The algorithm works in the probability‑simplex (∑a=1, a≥0) and
optimises a *smooth* approximation of the L∞‑norm:

    Sτ(a) = τ·log(∑_k exp(conv_k / τ))

so that   ∂Sτ/∂a = Σ_k p_k ∂conv_k/∂a,
with p_k the soft‑max weights.
A mirror‑descent (exponentiated‑gradient) update is performed
in log‑space, guaranteeing feasibility at every step.

A short temperature schedule (τ decreasing from 2·10⁻² down to 5·10⁻⁴)
is used, together with occasional random perturbations to escape
local minima.  The routine starts from the best sequence already stored
in the global variable ``height_sequence_1`` (if any); otherwise a
Dirichlet draw is used.

The function respects the supplied time budget (default 1000 s) and
stores the final sequence back into ``height_sequence_1`` for possible
continuations.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real, linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seeds (fallback when no previous solution is available)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a strictly positive probability vector from Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth‑max exponentiated‑gradient optimiser
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             tau: float = 0.01,
                             max_iters: int = 800_000,
                             eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *smooth* max‑convolution.
    Operates in log‑softmax space (guaranteed feasibility).

    Parameters
    ----------
    start_seq : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    n : int
        Length of the vector.
    deadline : float
        Absolute time (seconds since epoch) after which the routine
        must stop.
    tau : float
        Temperature for the soft‑max; smaller → closer to true max.
    max_iters : int
        Hard iteration limit.
    eta0 : float
        Initial learning‑rate multiplier (used as η₀/√t).

    Returns
    -------
    best_seq, best_val
        The best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑parameters; softmax will enforce the simplex
    log_a = np.log(a + 1e-30)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # 1️⃣  a * a  (full linear convolution)
        conv = _conv_fft(a, a)                     # length 2n‑1

        # 2️⃣  soft‑max weights for the smooth max
        if tau <= 0:
            # hard max – only the largest entry gets weight 1
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            vmax = conv.max()
            # scaling by τ after subtraction of the max for stability
            w = np.exp((conv - vmax) / tau)
            p = w / w.sum()

        # 3️⃣  gradient of smooth max w.r.t. a
        a_rev = a[::-1]                           # reversed a
        conv_rev = _conv_fft(a_rev, p)            # length 3n‑2
        # slice that corresponds to Σ_j a_j p_{i+j}
        slice_start = n - 1
        g_raw = 2.0 * conv_rev[slice_start:slice_start + n]
        # diagonal correction (i == j) – count only once
        diag_idx = 2 * np.arange(n)               # positions 0,2,4,…,2n‑2
        g_raw -= a * p[diag_idx]

        # 4️⃣  mirror‑descent (exponentiated‑gradient) update in log‑space
        dot = np.dot(g_raw, a)                    # ⟨g,a⟩
        grad_log = a * (g_raw - dot)              # ∇_log F
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)                       # back to simplex

        # keep the best observed surrogate value
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
