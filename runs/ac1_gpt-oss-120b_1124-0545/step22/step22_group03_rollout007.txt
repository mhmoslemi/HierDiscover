    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    best_val = _surrogate_obj(a)
    it = 0
    while it < max_iters and time.time() < sub_deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active shifts (within tolerance relative)
        active = np.where(conv >= max_val - 1e-12)[0]  # exact ties may be few
        # optional also include shifts within small relative tolerance
        if len(active) == 0:
            active = np.array([conv.argmax()], dtype=int)
        # compute gradient sum over active
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            idx = np.arange(i_start, i_end+1, dtype=np.int64)

            g[idx] += a[k - idx]

        eta = eta0 / math.sqrt(it)

        cur_val = _surrogate_obj(a)

            best_val = cur_val
            best_a = a.copy()

"""
Propose candidate step‑function heights using multi‑start exponentiated‑gradient
and simulated‑annealing.  Several initial shapes (uniform, triangular,
Hann, random Dirichlet, previous best) are tried; each is refined with a
gradient‐descent phase that uses the subgradient of the max‑convolution
(including all active shifts) followed by a temperature‑controlled SA that
moves tiny probability mass between indices.  The best vector found within
the overall time budget is returned.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))

def _exp_grad_multi(start_seq: np.ndarray,
                    deadline: float,
                    eta0: float = 5.0,
                    eps_active: float = 1e-12,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using a sub‑gradient of the max‑convolution.
    The sub‑gradient is the average of gradients of all shifts that achieve the current
    maximum (within a tiny tolerance).  Returns the best sequence encountered.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # initialise log‑weights
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active shifts: those that attain the maximum (allow tiny numerical slack)
        active = np.where(conv >= max_val - eps_active)[0]

        if active.size == 0:
            active = np.array([conv.argmax()], dtype=np.int64)

        # compute sub‑gradient = average of the gradients of active shifts
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            g[idx] += a[k - idx]
        g = g / active.size

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Classic SA: move a tiny amount of probability mass between two random indices.
    Temperature and perturbation size decay linearly.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def _build_initial_vectors(n: int, rng: np.random.Generator,
                           prev: np.ndarray | None) -> list[np.ndarray]:
    """Create a collection of diversified start vectors."""
    vectors = []

    if prev is not None:
        vectors.append(prev)

    # Uniform
    vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Triangular (symmetric, peak in middle)
    t = np.arange(1, n // 2 + 1, dtype=np.float64)

    else:

    hann /= hann.sum()
    vectors.append(hann)

    # Random Dirichlet samples – sparse and dense
    for alpha in (0.5, 2.0):
        vec = rng.dirichlet(np.full(n, alpha))
        vectors.append(vec)

    return vectors

Propose a height vector that minimises ``evaluate_sequence``.

The algorithm uses several diversified start vectors (previous best,
uniform, triangular, Hann window and Dirichlet samples).  Each start is
refined in two stages:

1. **Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution**  
   – the sub‑gradient is the *average* of the gradients of all shifts that
     are currently maximal (a small tolerance is used to collect an active
     set).  This gives a more informative descent direction than using a
     single worst shift.

2. **Simulated‑annealing fine‑tuning** – tiny mass transfers between random
   indices with a linear temperature/sigma schedule.

The total time budget (default 1000 s) is split among a handful of
restarts; the best sequence found overall is returned (and stored in the
global ``height_sequence_1`` for possible reuse).  If time remains after the
restarts a final fine‑tuning pass is performed on the current best.
"""

    """Full linear convolution via a real FFT (O(n log n))."""

    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)

    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

        # Choose the best of the two phases for this restart
        if val_sa < val_eg:
            cand_seq, cand_val = seq_sa, val_sa
        else:
            cand_seq, cand_val = seq_eg, val_eg

        print(f"Best surrogate this restart = {cand_val:.6f}")

        if cand_val < global_best_val:
            global_best_val = cand_val
            global_best_seq = cand_seq.copy()
            print(f"=> New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning on the overall best (if time remains)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and global_best_seq is not None:
        print("\n=== Final fine‑tuning of the global best ===")
        final_deadline = time.time() + 0.60 * remaining
        seq_ft, val_ft = _exp_grad_multi(global_best_seq, final_deadline,
                                         eta0=2.0, max_iters=5_000_000)
        # Optional short SA after the final EG
        if deadline - time.time() > 5.0:
            seq_ft_sa, val_ft_sa = _simulated_annealing_refine(
                seq_ft, deadline, rng,
                max_attempts=300_000,
                T_start=0.02,
                T_end=1e-9,
                sigma_start=0.003)
            if val_ft_sa < val_ft:
                seq_ft, val_ft = seq_ft_sa, val_ft_sa

        if val_ft < global_best_val:
            global_best_val = val_ft
            global_best_seq = seq_ft.copy()
            print(f"--> Improved global surrogate after fine‑tuning = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # True evaluation (may differ slightly from the surrogate)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\nTrue evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the best sequence for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
