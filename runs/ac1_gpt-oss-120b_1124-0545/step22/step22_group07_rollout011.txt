"""
A fast, aggressive search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The idea is:

1.  Build a fairly large pool of analytic / random “seed” vectors.
2.  Pick the few best (by the cheap surrogate 2·n·max(conv)).
3.  For each of them run a long smoothed‑Adam optimisation with a
    gradually increasing temperature β (log‑sum‑exp smooth max).
4.  Follow up with a very long exact‑max exponentiated‑gradient (mirror‑descent)
    polishing stage.
5.  Finish with a short permutation‑annealing tweak.
6.  Return the best sequence found (also stored in the global variable
    ``height_sequence_1``).

The optimisation works in log‑softmax space, so the probability simplex
constraint (non‑negative + sum = 1) is always satisfied.
All time is managed with a hard deadline (default 1000 s); the routine
automatically backs off to whatever time is left.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution via real FFT (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when the sum of the vector = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact score for a probability vector (sum == 1): 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using Adam.
    The surrogate uses log‑sum‑exp with temperature β.
    """
    if beta_schedule is None:
        # start very smooth, finish very sharp
        beta_schedule = [5, 25, 125, 625, 3_125,
                         15_625, 78_125, 390_625, 1_953_125]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # exact convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)                 # numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                # soft‑max weights

            # gradient of the surrogate w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to obtain gradient w.r.t. log‑softmax variables
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded (avoid overflow/underflow)
            log_a = np.clip(log_a, -30.0, 30.0)

            # track best *exact* objective seen so far
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Operates in the log‑softmax space (a_i = exp(log_a_i)/Σ).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (current max entry)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis notes)
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very short permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The strategy is:
        * generate a diverse pool of analytic / random seeds,
        * keep the few best according to the cheap surrogate,
        * run a long smoothed‑Adam optimisation on each of them,
        * follow up with a very long exact‑max exponentiated‑gradient
          polish,
        * finish with a short permutation‑annealing tweak,
        * store the result in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – a couple of seconds before the hard limit
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 points gives a good balance of quality / speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) – helps continuation runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a reasonably large pool of seeds (analytic families,
    #    Dirichlet draws, piecewise‑linear shapes, previous best)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.001))
    seeds.append(_exp_decay(n, rate=0.003))
    seeds.append(_exp_decay(n, rate=0.007))
    seeds.append(_power_law(n, p=0.4))
    seeds.append(_power_law(n, p=0.8))
    seeds.append(_power_law(n, p=1.2))
    seeds.append(_power_law(n, p=1.6))

    # sweep a finer grid for power‑law and exponential families
    for p in np.linspace(0.1, 2.0, 15):
        seeds.append(_power_law(n, p))
    for rate in np.logspace(np.log10(5e-4), np.log10(2e-2), 12):
        seeds.append(_exp_decay(n, rate))

    # beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – provide diverse “random” shapes
    for conc in (0.05, 0.2, 0.6, 1.5, 3.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random piecewise‑linear shapes (few breakpoints)
    for nb in (3, 6, 9):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop: Adam‑smooth + EG polish for each top seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time
        remaining = deadline - time.time()
        # spend a larger fraction on the first (best) seed, less on later
        share = remaining * (0.45 if idx == 0 else 0.25) / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ---- smoothed Adam optimisation --------------------------------
        seq_adam, val_adam = _adam_smooth_opt(seed_vec,
                                              sub_deadline,
                                              iters_per_beta=35_000)

        # ---- exact‑max EG run on the same time slice --------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                               sub_deadline,
                                               max_iters=800_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        local_seq, local_val = (seq_adam, val_adam) if val_adam < val_eg else (seq_eg, val_eg)

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (uses all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:          # keep a few seconds for final tweak
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline - 10.0,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (a few seconds) – may shave a bit
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        short_deadline = time.time() + 15.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=70_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
