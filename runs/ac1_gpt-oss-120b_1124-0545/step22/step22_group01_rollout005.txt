def _smooth_exp_gradient_opt(start_seq, n, deadline, beta_start=1.0, beta_max=2000.0, eta0=1.0):
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    it = 0
    beta = beta_start
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # compute softmax weights
        # avoid overflow: subtract max_c
        w_raw = np.exp(beta * (conv - max_c))
        w = w_raw / w_raw.sum()
        # compute gradient
        a_rev = a[::-1]
        grad = 2.0 * _conv_fft(w, a_rev)[n-1:n-1+n]
        # mirror descent update in log-space
        # compute dot(g,a)
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)
        # update best
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # schedule beta: increase gently
        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.2)
    return best_a, best_val

"""
Improved step‑function optimiser (alternative to the previous approach).

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0

Because the objective is homogeneous we may restrict ourselves to probability
vectors (sum = 1).  The problem then becomes minimising the peak of the
auto‑convolution of a probability distribution.

Key ideas
~~~~~~~~~
1. **Smoothed max** – replace the non‑differentiable max by a
   temperature‑controlled soft‑max (log‑sum‑exp).  Its gradient is a
   weighted sum of *all* convolution entries, which gives a much richer
   direction than the sub‑gradient that only looks at the single active
   index.

2. **Mirror‑descent in the simplex** – we work in log‑probability space,
   i.e. `log_a = log(a)`.  An entropic mirror update (`a = softmax(log_a)`)
   automatically enforces non‑negativity and the ℓ₁‑normalisation.

3. **Annealing the temperature** – start with a low β (very smooth) to
   explore, then increase it geometrically so the optimisation concentrates
   on the true maximum as time progresses.

4. **Warm‑starts and diversification** – we reuse the previous best
   solution (if any), a few analytic windows (uniform, triangular, Hann,
   Blackman, sine‑window) and several Dirichlet draws.  The best few seeds
   are refined with the smoothed optimiser; the current global best is
   polished for the remaining time.

The optimiser is deliberately simple, uses only FFT‑based convolutions
(O(n log n)) and fits comfortably inside the 1000 s budget while typically
exceeding the previous bound of ≈1.5085.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Project onto simplex (fallback, not used in the main loop)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser (mirror descent)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             beta_start: float = 1.0,
                             beta_max: float = 2000.0,
                             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smoothed version of `max(conv)`.
    Works in log‑space (entropy mirror) so constraints are always satisfied.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1

        # exact convolution a*a
        conv = _conv_fft(a, a)
        max_c = conv.max()

        # soft‑max weights over all convolution entries
        w_raw = np.exp(beta * (conv - max_c))   # stability: max_c removed
        w = w_raw / w_raw.sum()

        # gradient of the smoothed objective:
        # g_j = 2 * Σ_k w_k * a_{k-j}
        a_rev = a[::-1]                     # reversed a
        grad = 2.0 * _conv_fft(w, a_rev)[n - 1:n - 1 + n]

        # mirror‑descent step in log‑space
        dot = np.dot(grad, a)               # ⟨grad,a⟩
        grad_log = a * (grad - dot)         # ∇_log f
        eta = eta0 / math.sqrt(it)          # decreasing step size
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)                 # back to the simplex

        # keep the best observed point
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # anneal temperature (increase β)
        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.2)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Original exponentiated‑gradient optimiser (kept as a fast polish)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the exact sub‑gradient of the true max‑convolution.
    Operates in log‑space (entropy mirror).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))  # active index

        # sub‑gradient of the active entry
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent update
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
