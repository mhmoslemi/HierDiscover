"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.
The algorithm follows three ideas that extend the previous baseline:

1. **Richer initial pool** – besides the uniform/arc‑sine/triangular families we
   add power‑law and mixed (convex combination) seeds, and we always keep a
   copy of the previous solution (if any) resized to the current length.

2. **Two‑phase exponentiated‑gradient (mirror descent)** – the first phase uses a
   large step‑size (η₀≈8) to quickly approach a good region, the second phase
   uses a much smaller step‑size (η₀≈0.5) together with the same exact
   sub‑gradient (the active lag of the autocorrelation) to perform a fine
   optimisation.  Both phases stop automatically when the time budget is
   exhausted.

3. **Intensive random walk** – after the gradient phases we finish with a cheap
   local search that repeatedly moves a tiny amount of mass from one index to
   another (accept‑only‑improvements).  Because the objective is essentially
   ``2·n·max(conv(a,a))`` for a normalised vector, this walk can still
   discover tiny improvements that the gradient steps may miss.

All operations respect the simplex constraint (non‑negative entries that sum
to one).  The code runs completely self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` function, and adapts its workload to the
remaining time so that the 1000 s budget is never exceeded.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, softmax, surrogate objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (projects onto the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search.
    For a probability vector (sum = 1) it is
        2·n·max(conv(seq,seq))
    which equals the value returned by ``evaluate_sequence``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _geometric(n: int, rate: float) -> np.ndarray:
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    a_sum = a.sum()
    if a_sum <= 1e-12:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, exponent)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with a scalar concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add tiny Gaussian noise and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix(seq1: np.ndarray, seq2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w*seq1 + (1-w)*seq2, then renormalise."""
    mixed = w * seq1 + (1.0 - w) * seq2
    s = mixed.sum()
    if s <= 0.0:
        return seq1.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact sub‑gradient of max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 8.0):
    """
    Mirror–descent on the true ``max(conv)`` objective.
    Sub‑gradient is taken from the (possibly multiple) active lags.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)            # stay in log‑softmax space
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # active set – all lags within a tiny absolute tolerance of the maximum
        eps = max(1e-15, 1e-12 * max_c)
        act = np.where(conv >= max_c - eps)[0]

        # exact sub‑gradient for each active lag
        g = np.zeros_like(a)
        for k in act:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            idx = np.arange(i0, i1 + 1, dtype=int)
            rev = k - idx
            # contribution a[rev] + a[idx]
            g[idx] += a[rev] + a[idx]

        if len(act) > 1:
            g /= len(act)                  # any convex combination works

        # step size – classic √t schedule
        eta = eta0 / math.sqrt(it)

        # log‑space additive update, then project back onto the simplex
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Lightweight random local refinement (accept‑only‑improvements)
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """Move a tiny random fraction of mass from i to j and keep the move if it improves."""
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta relative to the source mass
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

        print("\n--- Random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                deadline,
                                                attempts=400_000,
                                                sigma=0.015)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for a possible warm‑start in a subsequent call.
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
