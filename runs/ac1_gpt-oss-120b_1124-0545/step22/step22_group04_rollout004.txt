    # Represent as matrix A shape (2n-1, n)

        # For each column i, f coeff appears in rows k=i..i+n-1 (i offset)
        start_row = i

def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical edge case – return uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum is 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    maxb = np.max(conv)
    sum_a = np.sum(seq)
    # sum_a should be 1 but keep formula generic
    return float(2.0 * n * maxb / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Surrogate for L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) of the smoothed objective:
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c                     # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # work in log‑space (log‑softmax variables)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (sum over all i,j with i+j=max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected gradient descent (Euclidean) on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
            n: int,
            deadline: float,
            max_iters: int = 2_000_000,
            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Euclidean projected gradient descent on the exact max‑convolution.
    Returns best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a_tmp = a - eta * g
        a_tmp = _proj_simplex(a_tmp)

        cur_val = _objective(a_tmp)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_tmp.copy()

        a = a_tmp

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fast permutation based hill climbing (simulated annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two positions or adds a tiny Gaussian noise.
    Accepts worsening moves with a simple annealing schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            ns = _proj_simplex(ns)

        val = _objective(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic families for cheap initialisation
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) for i = 0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines several restarts, a short L‑BFGS surrogate phase,
    mirror‑descent and projected‑gradient polishing, plus a final permutation
    tweak.  It respects the given time budget and stores the best result in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Preserve a tiny safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1. previous best, if available and of correct size
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # 2. exponential decay (the default used previously)
    candidates.append(_exp_decay(n, rate=0.001))

    # 3. a few random Dirichlet draws (different concentration parameters)
    for alpha in (0.3, 0.7, 1.5, 3.0):
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        if g.sum() == 0:
            continue
        candidates.append(g / g.sum())

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS smoothing on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    max_iter_per_beta = 2000                     # cheap enough for many restarts

    best_seq = None
    best_val = float('inf')

    for idx, init_seq in enumerate(candidates):
        # ensure proper simplex
        seq = _proj_simplex(init_seq)

        print(f"\n[Candidate {idx+1}/{len(candidates)}] initial surrogate = {_objective(seq):.6f}")

        # log‑softmax variables for L‑BFGS
        v = np.log(seq + 1e-30)

        for beta in beta_schedule:
            if time.time() > deadline:
                break
            if _opt is None:
                break
            try:
                res = _opt.minimize(fun=_smooth_obj_and_grad,
                                    x0=v,
                                    args=(n, beta),
                                    method='L-BFGS-B',
                                    jac=True,
                                    options={'maxiter': max_iter_per_beta,
                                             'disp': False,
                                             'ftol': 1e-12})
                v = res.x
            except Exception:   # pragma: no cover
                break

        seq = _softmax(v)                # back to probability vector
        val = _objective(seq)
        print(f"[Candidate {idx+1}] after smoothing surrogate = {val:.6f}")

        # keep the best after smoothing
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    if best_seq is None:
        # fallback – should never happen
        best_seq = _exp_decay(n, rate=0.001)

    # ------------------------------------------------------------------
    # Phase 2 – intensive mirror‑descent polishing (EG)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- EG polishing (mirror‑descent) ---")
        best_seq, best_val = _exp_gradient_opt(best_seq, n, deadline,
                                               max_iters=4_000_000,
                                               eta0=12.0)
        print(f"[After EG] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2b – Euclidean projected‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- PGD polishing (Euclidean) ---")
        best_seq, pgd_val = _pgd_opt(best_seq, n, deadline,
                                     max_iters=2_000_000,
                                     eta0=6.0)
        if pgd_val < best_val:
            best_val = pgd_val
            print(f"[After PGD] improved surrogate = {best_val:.6f}")
        else:
            print(f"[After PGD] no improvement (best still {best_val:.6f})")

    # ------------------------------------------------------------------
    # Phase 3 – short random‑restart refinements
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        # create a noisy copy of the current best
        noisy = best_seq + 0.003 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        noisy = _proj_simplex(noisy)

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _exp_gradient_opt(noisy, n, sub_deadline,
                                           max_iters=1_500_000,
                                           eta0=12.0)
        if val_rr < best_val:
            best_seq, best_val = seq_rr, val_rr
            print(f"[Random restart] improved surrogate = {best_val:.6f}")

        # also try a PGD polish on the same noisy start
        if time.time() + 5.0 < deadline:
            sub_deadline2 = time.time() + 5.0
            seq_rr2, val_rr2 = _pgd_opt(seq_rr, n, sub_deadline2,
                                        max_iters=500_000,
                                        eta0=6.0)
            if val_rr2 < best_val:
                best_seq, best_val = seq_rr2, val_rr2
                print(f"[RR PGD] further improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final permutation tweak (very fast)
    # ------------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        short_deadline = time.time() + 10.0
        seq_last, val_last = _perm_hill_climb(best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=80_000)
        if val_last < best_val:
            best_seq, best_val = seq_last, val_last
            print(f"[Final permutation] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate the *true* objective via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
