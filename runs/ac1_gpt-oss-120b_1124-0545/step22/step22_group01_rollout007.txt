assistantfinal**Solution Overview**

The problem is to minimise  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

The scale of `a` does not matter, therefore we can **ℓ₁‑normalise** the vector
(`∑a = 1`).  After normalisation the objective becomes  

\[
F(a)=2\,n\;\max_k (a*a)_k .
\]

The task is thus a *min‑max* problem: minimise the maximum entry of the
autocorrelation of a non‑negative probability vector.

The previous optimiser already performs a long exponentiated‑gradient (mirror
descent) run.  To push the bound a little further we keep the same
simple‑yet‑effective machinery but change two things:

1. **Smooth max** – instead of using only the single active index we take a few
   of the largest convolution entries and form a smooth approximation of the
   max with a temperature `β`.  
   The gradient of this “soft‑max” is a *weighted* sum of the gradients of the
   selected entries, which reduces the zig‑zag behaviour of pure sub‑gradient
   descent.

2. **Time‑focused optimisation** – we discard the heavyweight seed‑generation
   stage and devote almost the whole 1000 s budget to a long polish of the
   best known sequence (the one stored in `height_sequence_1`).  
   The remaining time is spent on short random escapes and re‑optimisation.

The optimiser works entirely in the log‑domain (exponentiated‑gradient /
mirror‑descent) which guarantees feasibility (non‑negativity and exact
ℓ₁‑normalisation).  All heavy work is performed with an FFT‑based convolution,
and the gradient for the smooth max is built from at most a few active
convolution indices, so each iteration is cheap and we can afford millions of
updates inside the time limit.

The final sequence is stored in the global variable `height_sequence_1` for
possible continuation runs.

"""
Improved optimiser for the step‑function bound.

It loads the previous best sequence (if any), then runs a long
mirror‑descent optimisation that minimises a *smoothed* maximum of the
autocorrelation.  The smoothing (soft‑max) uses the top few convolution
entries, which gives a more stable descent direction than a single active
constraint.  Random perturbations are tried while time permits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; output sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector with sum == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex ( Euclidean )
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Mirror‑descent optimiser with a smoothed max (soft‑max)
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 10_000_000,
                    eta0: float = 6.0,
                    beta: float = 8.0,
                    top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *soft‑max* of the convolution.

    The soft‑max approximates   max_k (a*a)_k   and uses the `beta`
    temperature.  Only the `top_k` largest convolution entries are taken,
    which makes a gradient cheap to compute (O(top_k·n) per iteration).
    """
    a = np.copy(start_seq)
    # safety normalisation
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space (exponentiated‑gradient) variables
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution – needed to know the current maximum values
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # pick the `top_k` indices (ties are handled automatically)
        k = min(top_k, conv.shape[0])
        idxs = np.argpartition(-conv, k - 1)[:k]
        # soft‑max weights over the selected entries
        vals = conv[idxs]
        w = np.exp(beta * (vals - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / k)
        else:
            w = w / w_sum

        # gradient of the smooth max – weighted sum of the individual gradients
        g = np.zeros_like(a)

        for kk, weight in zip(idxs, w):
            i_start = max(0, kk - (n - 1))
            i_end = min(n - 1, kk)
            if i_start > i_end:
                continue
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev = kk - idx_i

            # off‑diagonal contribution (appears twice in the convolution)
            g[idx_i] += 2.0 * weight * a[rev]

            # diagonal entries (i == rev) must be counted only once
            diag = idx_i == rev
            if np.any(diag):
                # we added 2·weight·a[i]; keep only weight·a[i]
                g[idx_i[diag]] -= weight * a[idx_i[diag]]

        # mirror‑descent update in log‑space
        dot = np.dot(g, a)                     # <g, a>
        grad_log = a * (g - dot)                # ∇_log f
        eta = eta0 / math.sqrt(it)              # classic 1/√t schedule
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)                     # stays on the simplex

        # keep the best observed sequence
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fallback pure exponentiated‑gradient optimiser (identical to the previous version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 5.0):
    """
    Classic exponentiated‑gradient (mirror‑descent) optimiser that uses the
    *exact* sub‑gradient of the active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    `evaluate_sequence`.  The routine loads the previous best solution,
    spends the majority of the time polishing it with the smoothed‑max
    mirror‑descent, and finally tries a few random escapes.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1.  Load previous solution or fall back to the uniform vector
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) > 0:
        arr = np.asarray(prev, dtype=np.float64)
        n = arr.shape[0]
        a0 = np.maximum(arr, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / s
        print(f"Loaded previous sequence of length {n}")
    else:
        # default to a 4000‑long uniform distribution
        n = 4000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print("No previous sequence found – starting from uniform.")

    print(f"Initial surrogate objective = {_objective_surrogate(a0):.6f}")

    # ----------------------------------------------------------------
    # 2.  Long polishing phase with the smoothed‑max optimiser
    # ----------------------------------------------------------------
    # We spend roughly 80 % of the time on this phase
    polish_deadline = start_time + (deadline - start_time) * 0.80
    best_seq, best_val = _smooth_max_opt(a0, n, polish_deadline,
                                         max_iters=10**9,
                                         eta0=6.0,
                                         beta=8.0,
                                         top_k=5)
    print(f"[Polish] best surrogate after smoothed‑max = {best_val:.6f}")

    # ----------------------------------------------------------------
    # 3.  Finish with the classic exponentiated‑gradient optimiser
    # ----------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_val = _exp_gradient_opt(best_seq, n, deadline,
                                                max_iters=10**9,
                                                eta0=5.0)
        if final_val < best_val:
            best_seq, best_val = final_seq, final_val
            print(f"[Final] improved surrogate = {best_val:.6f}")

    # ----------------------------------------------------------------
    # 4.  Random‑perturbation escapes while time permits
    # ----------------------------------------------------------------
    while time.time() < deadline - 5.0:       # keep a few seconds for final eval
        pert = best_seq + 0.001 * rng.standard_normal(n)
        pert = np.maximum(pert, 0.0)
        pert = _project_simplex(pert)

        sub_deadline = time.time() + 8.0      # short burst
        seq_tmp, val_tmp = _smooth_max_opt(pert, n, sub_deadline,
                                           max_iters=400_000,
                                           eta0=6.0,
                                           beta=8.0,
                                           top_k=5)
        if val_tmp < best_val:
            best_seq, best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {best_val:.6f}")

    # ----------------------------------------------------------------
    # 5.  Final official evaluation
    # ----------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
