def _project_simplex(w):
    # ensure nonnegative and sum=1 while keeping closeness to w in L2 sense
    u = np.sort(w)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(w)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    return np.maximum(w - theta, 0)

"""
Propose a low‑value step‑function.

Algorithm Overview
------------------
1. **Initial guess** – start from the previously found sequence (if any) or
   from a set of cheap analytic families (exponential, power‑law,
   linear‑cutoff).  A small grid search over the family parameters
   identifies the best seed.

2. **Adam on a smoothed surrogate** – minimise a log‑sum‑exp approximation
   of  max(conv(a,a))  using the Adam optimiser.
   The temperature `beta` is increased gradually; for each `beta`
   a few thousand Adam steps are performed.  The best vector seen
   (according to the exact max‑convolution) is kept.

3. **Exact sub‑gradient polishing** – a mirror‑descent (exponentiated‑gradient)
   routine using the true sub‑gradient of the active convolution entry
   (identical to the original `*_exp_gradient_opt`) refines the solution.

4. **Random‑restart polishing** – while time permits, a few noisy copies of the
   current best are polished with the same sub‑gradient method; improvements
   are kept.

5. **Final evaluation** – the true `evaluate_sequence` function is called on
   the resulting sequence; the sequence is stored globally for possible
   continuation runs.

The whole routine respects the 1000 s budget and only uses NumPy (and SciPy
if available for the optional L‑BFGS fallback).  All helper functions are
defined at module level as required.
"""

import time
import math
import numpy as np

# optional SciPy – only used for a L‑BFGS fallback (not required)
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (the quantity we finally minimise)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) for a normalised seq (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (cheap seeds)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0⋯n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _linear_cutoff(n: int, k: int) -> np.ndarray:
    """a_i = max(0, (k-i)/k) for i<k, zero afterwards; normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, (k - i) / k)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _mixed_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """Convex combination of two exponentials."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-r1 * i) + (1.0 - w) * np.exp(-r2 * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – used by Adam / L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (obj, grad) of the smoothed objective
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c                       # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (mirror‑descent on max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space update (exponentiated‑gradient)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small permutation‑based hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or perturbs the vector.
    Accepts worsening moves with a simple annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # annealing schedule parameters
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:            # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                               # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_optimize(v0: np.ndarray,
                   n: int,
                   deadline: float,
                   betas: list[float],
                   iters_per_beta: int = 5_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax variables v, using the smoothed log‑sum‑exp
    surrogate.  Returns the best probability vector and its exact surrogate
    value.
    """
    v = v0.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    best_seq = None
    best_val = np.inf

    for beta in betas:
        # how many Adam steps to allocate for this beta?  more for larger beta
        max_iter = iters_per_beta
        if beta > 5_000:
            max_iter = int(iters_per_beta * 2)
        for _ in range(max_iter):
            if time.time() > deadline:
                return best_seq, best_val

            t += 1
            obj, grad = _smooth_obj_and_grad(v, n, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # learning‑rate schedule – larger beta ⇒ smaller steps
            lr = 1.0 / (beta + 1.0)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

            # evaluate exact surrogate periodically (every 50 steps)
            if (t & 31) == 0:
                seq = _softmax(v)
                val = _objective_surrogate(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
    # final evaluation of the last iterate
    if best_seq is None:
        best_seq = _softmax(v)
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a cheap parametric grid search, an Adam optimiser
    on a smoothed surrogate, and a final exact sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")

            if arr.shape == (n,):
                init_seq = arr.copy()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    # ------------------------------------------------------------------
    # Cheap parametric families + tiny grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # start from the previous solution if we have one
    if init_seq is not None:
        val = _objective_surrogate(init_seq)
        best_seq, best_val = init_seq, val
        print(f"[Init] previous best surrogate = {best_val:.6f}")

    # exponential decay rates
    rates = np.geomspace(3e-4, 5e-3, num=30)   # 0.0003 … 0.005
    for r in rates:
        seq = _exp_decay(n, r)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_seq, best_val = seq, val

    # power‑law exponents
    powers = np.linspace(0.5, 4.5, num=30)
    for p in powers:
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_seq, best_val = seq, val

    # linear cut‑off lengths
    ks = np.linspace(int(0.05 * n), int(0.5 * n), num=20, dtype=int)
    for k in ks:
        seq = _linear_cutoff(n, k)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_seq, best_val = seq, val

    # mixed exponentials
    for _ in range(15):
        r1 = rng.uniform(3e-4, 5e-3)
        r2 = rng.uniform(3e-4, 5e-3)
        w = rng.random()
        seq = _mixed_exp(n, r1, r2, w)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_seq, best_val = seq, val

    print(f"[After grid] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam smoothing phase ---")
        # start from the best of the grid search
        v0 = np.log(best_seq + 1e-30)

        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
            50.0, 100.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
