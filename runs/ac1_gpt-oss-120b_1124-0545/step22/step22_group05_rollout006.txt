"""
Step‑function optimizer (2024‑11‑25)

The objective to minimise is

    J(a) = 2·n·max_k (a * a)[k]          (a lives on the probability simplex)

where “*” is the full linear convolution.  The problem is convex (a convex
quadratic objective inside a max) and can be tackled by mirror‑descent on the
simplex (exponentiated‑gradient).  The previous baseline used a plain
sub‑gradient (active‑lag only) which converges slowly.  Here we improve it by
optimising a *smoothed* version of the max (log‑sum‑exp) so that the gradient
takes into account many lags simultaneously; this gives a much richer
direction and typically reaches a lower value in the same wall‑clock time.

Algorithm outline
-----------------
1.  Build a modest pool of deterministic and random seeds (uniform,
    arcsine, triangular, several beta‑shapes, geometric decays, Dirichlet
    draws).  All vectors are normalised to sum = 1.

2.  Keep the best seed according to the exact surrogate
    `2·n·max(conv(a,a))`.

3.  Run a *smooth* exponentiated‑gradient descent:
        •  At iteration t compute the full auto‑convolution c = a*a.
        •  Form weights w_k = softmax(β_t·c_k) (β increases from a tiny
           value to a large one, turning the smooth objective into the true
           max).
        •  The gradient of the smooth objective is
              g = 2·(w ⋆ a_rev)[n‑1 : n‑1+n],
           i.e. a convolution of the weight vector with the reversed a.
        •  Update the log‑probabilities:
              log_a ← log_a – η_t·g   (η_t = η₀/√t)
           and project back to the simplex with a softmax.
        •  Periodically evaluate the exact surrogate and keep the best point.

4.  Starting from the best point obtained in (3) run the *exact*
    exponentiated‑gradient (active‑lag sub‑gradient) for a short while –
    this removes the smoothing bias.

5.  Finish with a cheap random local walk on the simplex.

All stages respect the overall 1000 s budget.  The final sequence is stored
in the global ``height_sequence_1`` variable and returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution) – the quantity we finally
# minimise (assuming sum(a)=1).
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    eps = 1e-12
    s = a.sum()
    if s <= eps:
        return _uniform(n)
    return a / s


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp) used in the accelerated phase
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of   (1/β)·log Σ_k exp(β·conv_k)   w.r.t. a.
    """
    # stable softmax over convolution values
    maxc = np.max(conv)
    exp_term = np.exp(beta * (conv - maxc))
    w = exp_term / exp_term.sum()                 # shape (2n‑1,)

    # convolution of the weight vector with the reversed a
    a_rev = a[::-1]
    g_full = _conv_fft(w, a_rev)                # length = 3n‑2
    offset = (len(conv) - 1) // 2                # = n‑1
    g = 2.0 * g_full[offset:offset + len(a)]
    return g

                             n: int,

                             eta0: float = 8.0,
                             beta0: float = 5.0,
                             beta_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *smoothed* max‑convolution.
    The temperature β is slowly increased from beta0 to beta_max,
    turning the smooth objective into the true max.
    """
    a = start_seq.copy()
    eps = 1e-30
    log_a = np.log(a + eps)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)                               # exact auto‑conv
        beta = beta0 + (beta_max - beta0) * (it / max_iters)  # linear schedule
        g = _smooth_gradient(a, conv, beta)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)                                 # project to simplex

        # occasional exact evaluation (cheap compared to a full iteration)
        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact active‑lag sub‑gradient optimiser (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Classical mirror‑descent using the sub‑gradient of the *active* lag.
    """
    a = start_seq.copy()
    eps = 1e-30
    log_a = np.log(a + eps)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active lag

        # sub‑gradient of conv_k (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light random local walk on the simplex
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 250_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny fraction from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    the supplied ``evaluate_sequence`` objective.  The search respects the
    ``budget_s`` time limit and stores the best sequence in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored solution (if compatible)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 0️⃣ Build a rich pool of deterministic and random seeds
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n)]
    if prev_arr is not None:
        seeds.append(prev_arr)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))

    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, a, b))

    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    for rate in (0.9999, 0.9995, 0.999, 0.998):
        seeds.append(_geometric(n, rate))

    seeds.append(_linear_decreasing(n))

    for sigma in (0.01, 0.005):
        seeds.append(_perturb(_uniform(n), rng, sigma=sigma))

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate all seeds and keep the best (exact surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet search (many draws while time permits)
    # ------------------------------------------------------------------
    draw_limit = 10_000
    draws = 0
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print(f"[Dir] new best surrogate = {best_val:.6f} (draw {draws})")
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Smoothed exponentiated‑gradient optimisation (main engine)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Smoothed exponentiated‑gradient optimisation ---")
        remaining = deadline - time.time()
        # allocate the bulk of remaining time to this stage
        max_iters = int(min(1_500_000, remaining * 1200))   # rough 0.8 ms / iter budget
        max_iters = max(200_000, max_iters)                # at least a few hundred k

        seq_smooth, val_smooth = _smooth_exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            eta0=10.0,
            beta0=5.0,
            beta_max=250.0,
        )
        print(f"[Smooth] best surrogate after smoothing = {val_smooth:.6f}")
        if val_smooth < best_val:
            best_val, best_seq = val_smooth, seq_smooth.copy()
            print("   >>> Updated global best (smooth)")

    # ------------------------------------------------------------------
    # 4️⃣ Exact active‑lag exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exact gradient refinement (active lag) ---")
        remaining = deadline - time.time()
        max_iters = int(min(2_000_000, remaining * 1500))
        max_iters = max(300_000, max_iters)

        seq_exact, val_exact = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            eta0=8.0,
        )
        print(f"[Exact] best surrogate = {val_exact:.6f}")
        if val_exact < best_val:
            best_val, best_seq = val_exact, seq_exact.copy()
            print("   >>> Updated global best (exact)")

    # ------------------------------------------------------------------
    # 5️⃣ Final lightweight random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            attempts=300_000,
            sigma=0.02,
        )
        print(f"[Local] best surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for warm‑start of a possible next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
