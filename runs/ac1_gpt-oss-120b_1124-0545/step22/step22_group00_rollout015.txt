"""
Propose a candidate step–function that beats the previous bound (≈1.5085).

The key observation is that for a probability vector a (ℓ₁‑normalised),
the quantity to minimise is

    S = 2·n·max_k (a * a)[k]

where (a * a) is the linear convolution.
If we choose a decreasing polynomial shape

    a_i ∝ (n - i)^{p}      (i = 0 … n‑1)

the convolution peaks at the centre with size approximately

    max ≈ ((p+1)²·B(p+1,p+1)) / n       (B = beta‑function)

and the extreme left‑most entry contributes (p+1)²/n².
The optimal p solves   B(p+1,p+1) ≈ 1/n.
For n≈4000 this happens for p≈5–6, giving S≈0.03 – far below the target.

The implementation therefore
  * builds a fine grid of Beta‑shaped vectors (α=1, β≈5‑8),
  * evaluates the cheap surrogate 2·n·max(conv) on each,
  * optionally refines the best one with a short exponentiated‑gradient run,
  * returns the best vector found (stored in the global ``height_sequence_1``).

All work finishes well within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (real FFT) – O(n log n)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple families used as seeds
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ i^{α‑1}·(n‑i+1)^{β‑1}  (i runs 1…n)"""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _poly_desc_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^{p}   – pure decreasing polynomial (α=1, β=p+1)"""
    i = np.arange(n, dtype=np.float64)          # i = 0 … n‑1
    a = (n - i) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / active.sum()
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 150_000,
                           eta0: float = 6.0):
    """Mirror‑descent (EG) on the exact max‑convolution, full active set."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad_a = _subgradient_max_conv(a)          # (n,)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)              # chain rule
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = np.exp(log_a - np.max(log_a))          # stable soft‑max
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine builds a fine grid of
    decreasing‑polynomial Beta shapes (α=1, β≈5‑8), evaluates the cheap
    surrogate 2·n·max(conv), optionally refines the best seed with a short
    exponentiated‑gradient pass, and returns the best sequence found.
    The result is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0               # a safety margin

    # --------------------------------------------------------------
    # Problem size – keep the previous length if a solution exists.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = arr.shape[0]
            if n > 0 and arr.sum() > 0.0:
                base = arr / arr.sum()
            else:
                base = None
                n = 4000
        except Exception:
            base = None
            n = 4000
    else:
        n = 4000
        base = None

    print(f"\n[Info] Using n = {n}")

    # --------------------------------------------------------------
    # 1. Build a diverse set of seeds – focus on pure decreasing powers.
    # --------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))

    # pure polynomial decreasing shapes (p = exponent)
    for p in np.linspace(0.5, 8.0, 31):          # 31 candidates, cheap
        seeds.append(_poly_desc_shape(n, p))

    # Beta shapes with α=1 (pure decreasing) and a fine β grid
    for beta in np.linspace(5.0, 8.0, 31):        # includes the optimal region
        seeds.append(_beta_shape(n, 1.0, beta))

    # a few random Dirichlet draws (exploration)
    for conc in (0.001, 0.005, 0.02, 0.1):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # add previous solution (if any) for local improvement
    if base is not None:
        seeds.append(base)

    # --------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # ensure non‑negative and normalised
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        # surrogate = 2·n·max(conv)
        conv = _conv_fft(vec, vec)
        val = float(2.0 * n * np.max(conv))
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        # tiny progress report
        if (idx + 1) % 20 == 0:
            print(f"[Seed {idx+1}/{len(seeds)}] current best surrogate = {best_val:.6f}")

    print(f"\n[Seed scan] best surrogate after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Optional short EG refinement of the current best seed
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined_seq, refined_val = _exp_gradient_opt_full(
            best_seq, n, deadline, max_iters=120_000, eta0=5.0
        )
        if refined_val < best_val:
            best_seq, best_val = refined_seq, refined_val
            print(f"[Refine] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final evaluation with the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Result ===\nSurrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("evaluate_sequence raised an exception:", exc)

    # store for later calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
