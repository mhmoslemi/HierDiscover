"""
Advanced step‑function optimizer (improved)

Goal: minimise 2·n·max(conv(a,a))  (with ∑a = 1, a_i ≥ 0).  
Approach:
  1. Build a rich but modest library of analytic windows (uniform,
     triangular, Kaiser, Gaussian, DPSS, etc.) and a few random Dirichlet
     draws – all normalised to the probability simplex.
  2. Pick the best seed (smallest exact surrogate 2·n·max(conv)).
  3. Refine it with three complementary optimisation phases, all
     respecting the simplex via a soft‑max (exponentiated‑gradient)
     parametrisation:
        • Smooth max optimisation with L‑BFGS‑B on a log‑sum‑exp surrogate,
          using a schedule of increasing β (≈ true max) and many L‑BFGS
          iterations per β.
        • Exponentiated‑gradient (mirror descent) on the exact max‑convolution
          using the sub‑gradient from the currently maximal lag.
        • Quadratic “flattening’’ of the whole convolution (minimise the
          variance of conv(a,a) around its mean) – drives the autocorrelation
          towards a constant.
  4. A short simulated‑annealing polish (Gaussian perturbations on the
     simplex) to escape possible tiny local minima.
  5. Return the best sequence found (stored globally as height_sequence_1)
     and also report the true value from evaluate_sequence.

The routine adapts to the remaining wall‑clock budget (default 1000 s)
and works for sequence lengths up to ~10 k in the given time budget.
"""

import time
import math
import numpy as np

    _opt = None

    from scipy.special import i0 as _i0

except Exception:  # pragma: no cover

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet vector with symmetric concentration."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Uses the (single) index attaining the current maximum as active set.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    # work in log‑space to stay on simplex
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # gradient of the active max (sub‑gradient)
        idx = np.arange(n, dtype=np.int64)
        rev_idx = max_idx - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        # contribution from the convolution term
        g[mask] = a[rev_idx[mask]]
        g = 2.0 * g                     # because conv = a * a is symmetric

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the quadratic loss Σ (conv_k – avg)².
    Drives the whole convolution towards a constant.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full linear convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when ∑a = 1
        diff = conv - avg

        rev_a = a[::-1]
        # gradient of Σ diff² w.r.t. a (using convolution)
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # convert to log‑space (mirror‑descent on simplex)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B.
    Parameterises the probability vector with a soft‑max of free variables.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            # shift = max(conv) for numerical stability
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximation of max
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max wrt a
            w = exp_beta / sum_exp                 # weights over each lag
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice to length n

            # chain rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian SA on the simplex.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

            break

        sub_deadline = time.time() + remaining * 0.95   # keep a tiny slack

        # ---- Phase A: smooth L‑BFGS (log‑sum‑exp) ----
        print("\n--- L‑BFGS smooth optimisation ---")
        seq_lbfgs = _lbfgs_smooth_optimize(global_best_seq, n, sub_deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < global_best_val:
            global_best_seq, global_best_val = seq_lbfgs, val_lbfgs
            print(f"[L‑BFGS] new global best surrogate = {global_best_val:.6f}")

        if time.time() >= deadline:
            break

        # ---- Phase B: exponentiated‑gradient on exact max‑convolution ----
        print("\n--- Exponentiated‑gradient (max‑conv) refinement ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=600_000,
                                                 eta0=8.0,
                                                 tol=1e-9)
        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg, val_eg
            print(f"[EG] new global best surrogate = {global_best_val:.6f}")

        if time.time() >= deadline:
            break

        # ---- Phase C: flattening (quadratic loss) ----
        print("\n--- Flattening gradient phase ---")
        seq_flat, val_flat = _flatten_gradient_opt(global_best_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=300_000,
                                                   eta0=5.0)
        if val_flat < global_best_val:
            global_best_seq, global_best_val = seq_flat, val_flat
            print(f"[Flatten] new global best surrogate = {global_best_val:.6f}")

        if time.time() >= deadline:
            break

        # ---- Phase D: short simulated annealing polish ----
        print("\n--- Simulated‑annealing polish ---")
        seq_sa, val_sa = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              rng,
                                              max_attempts=120_000,
                                              start_sigma=0.0015,
                                              T_start=0.01,
                                              T_end=5e-9)
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] new global best surrogate = {global_best_val:.6f}")

        # break after one full cycle – the loop guard will allow a second
        # if time permits but we avoid endless repeats.
        break

    # --------------------------------------------------------------
    # Final evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
