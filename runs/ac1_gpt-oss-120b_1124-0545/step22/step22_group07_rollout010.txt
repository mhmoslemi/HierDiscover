"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm
---------
1. Build a fairly large pool of analytic seeds (uniform, exponential,
   power‑law, beta‑distribution) and a dense family of *mixtures* of the
   uniform distribution with a symmetric beta shape.  The mixture parameters
   (beta‑exponent and mixing weight) are scanned on a small grid – this
   “low‑dimensional” search can discover shapes that make the self‑convolution
   flatter than the generic random seeds.

2. Keep the best few seeds (according to the exact surrogate
   ``2·n·max(conv)``) and polish each of them with a fast exponentiated‑gradient
   (EG) optimiser that works directly on the non‑smooth objective.
   The EG update is performed in log‑softmax space, guaranteeing non‑negativity
   and the unit‑sum constraint.

3. The currently best candidate is then refined for the rest of the time
   budget with a long EG run (the step size follows the classic
   ``η_t = η0 / sqrt(t)`` schedule).

4. Finally a very short permutation‑annealing (swap moves) is applied – it
   occasionally yields a tiny extra improvement.

All heavy work uses an FFT based convolution (``np.fft.rfft``) which is
``O(n log n)`` per iteration.  The code respects the global time budget
(≈1000 s) and stores the result in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``seq``."""
    return float(np.max(_conv_fft(seq, seq)))


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    return 2.0 * n * _max_conv(seq)


# ----------------------------------------------------------------------
# Analytic seed families (all return an ℓ₁‑normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete symmetric Beta(α,β) on {0,…,n‑1}.  All entries >0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (a_i = exp(log_a_i) / Σ).
    """
    a = start_seq.copy()
    n = len(a)

    # safeguard against zeros – start_seq should be strictly positive
    eps = 1e-30
    log_a = np.log(a + eps)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute full self‑convolution, locate active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule η_t = η0 / √t
        eta = eta0 / math.sqrt(it)

        # EG update in log‑space
        log_a -= eta * g
        # map back to probabilities (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 70_000) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the exact surrogate objective."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method builds a pool of analytic seeds (including a dense
    mixture‑of‑uniform + beta family), refines the most promising seeds with a
    short exponentiated‑gradient run, then performs a long EG polishing on the
    current best until the time budget runs out, followed by a quick permutation
    annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Try to reuse a previous solution if it exists (helps continuation runs)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                if prev_vec.sum() > 0:
                    prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Build a diverse pool of seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0018))
    seeds.append(_exp_decay(n, rate=0.0032))
    seeds.append(_sym_exp_decay(n, rate=0.0024))
    seeds.append(_power(n, p=0.7))
    seeds.append(_power(n, p=1.2))

    # Random Dirichlet with a wide range of concentrations
    for conc in (0.03, 0.08, 0.2, 0.6, 1.0, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random piecewise‑linear shapes
    for nb in (4, 9):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Some random Beta shapes (both symmetric and asymmetric)
    for _ in range(10):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # --------------------------------------------------------------
    # 2. Grid search over mixtures of Uniform + symmetric Beta(α,α)
    #    The mixture weight λ∈[0,1] interpolates between uniform (λ=1)
    #    and a pronounced U‑shape (λ≈0).
    # --------------------------------------------------------------
    uniform_vec = _uniform(n)
    alphas = np.geomspace(0.02, 3.0, num=30)  # dense on log‑scale
    lambdas = np.linspace(0.0, 1.0, num=21)   # 0 % … 100 % uniform

    for alpha in alphas:
        beta_vec = _beta_shape(n, alpha, alpha)   # symmetric
        for lam in lambdas:
            mix = lam * uniform_vec + (1.0 - lam) * beta_vec
            # mix is already normalised (lam+(1‑lam)=1)
            seeds.append(mix)

    # previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # --------------------------------------------------------------
    # 3. Evaluate all seeds (surrogate) and keep the top few
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 4. Short EG polishing for each top seed (budget‑aware)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate roughly an equal share of the remaining budget
        remaining = deadline - time.time()
        # ensure we always leave some time for the final polishing
        share = min(remaining * 0.12, remaining - 30.0)  # keep ~30 s for final steps
        sub_deadline = time.time() + max(5.0, share)    # at least a few seconds

        print(f"\n[Seed {idx+1}] time slice ≈ {share:.1f}s")

        seq_refined, val_refined = _exp_gradient_opt(seed_vec,
                                                     sub_deadline,
                                                     max_iters=800_000,
                                                     eta0=8.0)

        if val_refined < global_best_val:
            global_best_seq = seq_refined
            global_best_val = val_refined
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Long EG polishing for the remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=5_000_000,
                                                 eta0=8.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Very short permutation‑annealing tweak (a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 18.0
        seq_perm, val_perm = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Evaluate with the provided ``evaluate_sequence`` and store result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
