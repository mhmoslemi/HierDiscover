def _neighbor_move(seq, deadline, max_iters=200_000):
    n = len(seq)
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    T0 = cur_val
    T1 = cur_val * 1e-4

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = T0 + (T1 - T0) * (it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = cur_seq[i] * rng.random() * 0.5  # up to half of i's mass
        if eps <= 0:
            continue
        ns = cur_seq.copy()
        ns[i] -= eps
        ns[j] += eps
        val = _objective_surrogate(ns)
        if val < cur_val or (T>0 and rng.random() < math.exp(-(val - cur_val)/T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val

"""
Propose an improved step‑function sequence.

The strategy is:

1. Build a rich pool of analytic seeds, including a dense grid of
   exponential‑decay rates, power‑law families, beta‑shapes, random
   Dirichlet draws and a two‑exponential mixture.  All candidates are
   normalised to sum = 1 (the objective of evaluate_sequence is scale‑invariant).

2. Evaluate the exact surrogate objective (2·n·max(conv)) for every seed
   and keep the best few (top_k).

3. For each of the top seeds run a *smoothed Adam* optimiser on the
   log‑softmax parameters.  The smoothing is a log‑sum‑exp version of the
   max‑convolution with a β‑schedule that ends at a very large value
   (≈ 8·10⁵).  More iterations per β are performed than in the previous
   solution.

4. Polish the Adam result with a long *exponentiated‑gradient* run on the
   exact max‑convolution (sub‑gradient on the true max).  The step size
   schedule is slightly more aggressive.

5. After all seeds have been processed, spend the remaining budget on a final
   long EG polishing of the current global best.

6. Finish with a short permutation‑annealing tweak (swap moves) to shave
   off any remaining noise.

All heavy work is performed with FFT‑based convolution (O(n log n)).
The function respects the supplied time budget (default 1000 s) and stores
the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution (real, linear, full length = n+m‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (sum = 1 vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (always return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_two_mix(n: int,
                 rate1: float,
                 rate2: float,
                 alpha: float,
                 rng: np.random.Generator) -> np.ndarray:
    """Mixture of two exponentials: α·exp(−rate1·i)+(1−α)·exp(−rate2·i)."""
    i = np.arange(n, dtype=np.float64)
    a = alpha * np.exp(-rate1 * i) + (1.0 - alpha) * np.exp(-rate2 * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_800_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the active entry)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)      # log‑softmax parameters

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)                 # exact self‑convolution
        max_idx = int(np.argmax(conv))         # index of the active diagonal

        # sub‑gradient for the active entry
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g                     # mirror step in log‑space
        a = np.exp(log_a - np.max(log_a))    # improve stability
        a /= a.sum()                         # project onto the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using Adam.
    The objective is 2·n·M_β where M_β is the log‑sum‑exp smooth max.
    """
    if beta_schedule is None:
        # start smoothing, finish with a very sharp approximation of the true max
        beta_schedule = [2, 10, 50, 200, 800, 3_200, 12_800,
                         51_200, 200_000, 800_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                                 # base learning‑rate factor
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp            # soft‑max weights

            # gradient w.r.t a
            rev_a = a[::-1]
            # full convolution rev_a * w has length = 3n‑2, but we only need the middle n entries
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t log‑softmax parameters (chain rule)
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)   # prevent overflow

            # track best exact objective seen so far
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method builds a large, diverse pool of analytic / random seeds, selects the
    most promising ones, refines them with a smoothed Adam optimiser, polishes with
    a long exponentiated‑gradient run, and finally applies a short permutation anneal.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Gather previous solution if present – useful for continuation runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                if prev_vec.sum() > 0:
                    prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build an extensive pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))

    # fine grid of exponential‑decay rates (log‑space)
    exp_rates = np.logspace(np.log10(2e-4), np.log10(5e-3), num=30)
    for rate in exp_rates:
        seeds.append(_exp_decay(n, rate))

    # power‑law family
    for p in np.linspace(0.4, 2.0, num=30):
        seeds.append(_power(n, p))

    # beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear (moderate number of break points)
    for nb in (4, 8):
        for _ in range(3):
            seeds.append(_piecewise_linear(n, nb, rng))

    # two‑exponential mixtures (add extra expressive power)
    for _ in range(15):
        r1 = rng.uniform(2e-4, 5e-3)
        r2 = rng.uniform(2e-4, 5e-3)
        alpha = rng.random()
        seeds.append(_exp_two_mix(n, r1, r2, alpha, rng))

    # previous best and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop (Adam smooth + EG polishing)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (≈ 70 % of what is left divided equally)
        remaining = deadline - time.time()
        share = remaining * 0.7 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ---- smoothed Adam optimisation --------------------------------
        seq_adam, val_adam = _adam_smooth_opt(seed_vec,
                                              sub_deadline,
                                              iters_per_beta=60_000)

        # ---- exponentiated‑gradient polish (exact max) -----------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                               sub_deadline,
                                               max_iters=1_200_000,
                                               eta0=7.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        local_seq, local_val = (seq_adam, val_adam) if val_adam < val_eg else (seq_eg, val_eg)

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (may shave a little more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(20.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:    # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
